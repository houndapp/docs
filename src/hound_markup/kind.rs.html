1//! [`ContentKind`]: the coarse "what is in this message" vocabulary used to
2//! filter a search or a channel's media view.
3//!
4//! A kind is a *fact about the stored message*, derived once by
5//! [`crate::ParsedMessage`] from the source plus the facts the source alone
6//! cannot carry (see [`crate::MessageContext`]). The set is deliberately
7//! small and fixed in this crate rather than in a schema: a new kind is a
8//! variant plus a detection arm, never a database migration.
9
10/// One coarse content facet of a message.
11///
12/// The enum is `#[non_exhaustive]` because it is a derived index, not a wire
13/// contract: matching on it must keep working when a kind is added.
14#[non_exhaustive]
15#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
16pub enum ContentKind {
17    /// The body carries at least one `http(s)` link — a bare autolink, or
18    /// the destination of labelled link or image syntax (which renders as
19    /// literal text, but still names a URL).
20    Link,
21    /// A picture is actually attached or embedded. Never inferred from image
22    /// *syntax*: `![x](https://…)` is a URL the server has not verified, and
23    /// promising a picture that turns out to be a 404 is worse than promising
24    /// nothing.
25    Image,
26    Video,
27    Audio,
28    /// An attachment that is not an image, video or audio file.
29    File,
30    /// At least one link unfurled to a preview card.
31    Embed,
32    /// A fenced or indented code block.
33    Code,
34    /// A `||spoiler||` anywhere in the body.
35    Spoiler,
36}
37
38impl ContentKind {
39    /// Every kind, in [`Ord`] order — the order
40    /// [`crate::ParsedMessage::kinds`] is sorted into.
41    pub const ALL: &'static [ContentKind] = &[
42        ContentKind::Link,
43        ContentKind::Image,
44        ContentKind::Video,
45        ContentKind::Audio,
46        ContentKind::File,
47        ContentKind::Embed,
48        ContentKind::Code,
49        ContentKind::Spoiler,
50    ];
51
52    /// The stable string form, used verbatim as a database column value and
53    /// as a search filter token.
54    pub fn as_str(self) -> &'static str {
55        match self {
56            ContentKind::Link => "link",
57            ContentKind::Image => "image",
58            ContentKind::Video => "video",
59            ContentKind::Audio => "audio",
60            ContentKind::File => "file",
61            ContentKind::Embed => "embed",
62            ContentKind::Code => "code",
63            ContentKind::Spoiler => "spoiler",
64        }
65    }
66
67    /// The inverse of [`ContentKind::as_str`]; `None` for any other input.
68    ///
69    /// Not `std::str::FromStr`: a row written by a build that knew a kind this
70    /// one does not is skipped, not an error to propagate.
71    pub fn parse_str(s: &str) -> Option<Self> {
72        Some(match s {
73            "link" => ContentKind::Link,
74            "image" => ContentKind::Image,
75            "video" => ContentKind::Video,
76            "audio" => ContentKind::Audio,
77            "file" => ContentKind::File,
78            "embed" => ContentKind::Embed,
79            "code" => ContentKind::Code,
80            "spoiler" => ContentKind::Spoiler,
81            _ => return None,
82        })
83    }
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89
90    #[test]
91    fn as_str_round_trips_over_all() {
92        for kind in ContentKind::ALL {
93            assert_eq!(ContentKind::parse_str(kind.as_str()), Some(*kind));
94        }
95        assert_eq!(ContentKind::parse_str("nope"), None);
96    }
97
98    /// `ALL` is what a reindex iterates and what a filter UI lists; a kind
99    /// missing from it is invisible everywhere without being a compile error.
100    #[test]
101    fn all_is_sorted_and_complete() {
102        let mut sorted = ContentKind::ALL.to_vec();
103        sorted.sort();
104        sorted.dedup();
105        assert_eq!(sorted, ContentKind::ALL);
106    }
107}
