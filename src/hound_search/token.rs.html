1//! Highlight spans over the query source: which characters of the exact string
2//! the user is typing carry which meaning.
3//!
4//! ## Offsets
5//!
6//! Ranges are half-open `[start, end)` in **UTF-16 code units**, the unit a
7//! Dart `String` indexes by and therefore what a `TextEditingValue`'s selection
8//! and span boundaries are measured in — the same convention as
9//! [`hound_markup::StyleRun`].
10//!
11//! Tokens are non-overlapping and ascending. They do not tile the input: the
12//! whitespace between chunks produces no token, and a chunk the parser dropped
13//! entirely still reports its spans so the field can mark it.
14
15use std::ops::Range;
16
17use serde::{Deserialize, Serialize};
18
19/// One highlightable span of query source.
20#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
21pub struct Token {
22    /// Half-open `[start, end)` in UTF-16 code units.
23    pub range: Range<u32>,
24    pub kind: TokenKind,
25}
26
27/// What a [`Token`]'s span is.
28///
29/// `#[non_exhaustive]` because the vocabulary tracks the query language, not a
30/// wire contract: a new filter shape must not break a client's `match`.
31#[non_exhaustive]
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
33pub enum TokenKind {
34    /// A bare word term.
35    Word,
36    /// A quoted phrase term, quotes included. An unterminated quote runs to the
37    /// end of the input.
38    Phrase,
39    /// The leading `-` of a negated term. Never part of the term's own span.
40    Negation,
41    /// A recognised filter key **and its `:`** — `from:`, `has:`. An
42    /// unrecognised key is not a filter, so `foo:bar` is one [`TokenKind::Word`].
43    FilterKey,
44    /// A filter's value, quotes included when it was quoted. Absent when the
45    /// value is empty (`from:` at the end of the input).
46    FilterValue,
47    /// A filter value that is a `hound:<kind>/<uuid>` reference — the resolved
48    /// pill form the UI writes for `from:`/`in:`/`mentions:`. Replaces
49    /// [`TokenKind::FilterValue`] for that span.
50    Reference,
51}
