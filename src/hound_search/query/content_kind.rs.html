1//! Wire form for [`ContentKind`].
2//!
3//! `hound_markup` carries no serde dependency (it is parsed on both sides of
4//! the FFI boundary, where serde buys nothing), so the `has:` list travels as
5//! its stable string vocabulary — [`ContentKind::as_str`], the same values the
6//! database column holds — rather than as a derived representation that would
7//! pin the enum's variant order to the wire.
8
9use hound_markup::ContentKind;
10use serde::de::Error as _;
11use serde::{Deserialize, Deserializer, Serialize, Serializer};
12
13pub(super) fn serialize<S: Serializer>(
14    kinds: &[ContentKind],
15    serializer: S,
16) -> Result<S::Ok, S::Error> {
17    kinds
18        .iter()
19        .map(|kind| kind.as_str())
20        .collect::<Vec<_>>()
21        .serialize(serializer)
22}
23
24/// An unrecognised kind is an error, not a skip: dropping it would silently
25/// widen the search the caller asked for.
26pub(super) fn deserialize<'de, D: Deserializer<'de>>(
27    deserializer: D,
28) -> Result<Vec<ContentKind>, D::Error> {
29    Vec::<String>::deserialize(deserializer)?
30        .into_iter()
31        .map(|name| {
32            ContentKind::parse_str(&name)
33                .ok_or_else(|| D::Error::custom(format!("unknown content kind `{name}`")))
34        })
35        .collect()
36}
