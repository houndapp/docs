1//! The video **layer model** — the single source of truth mapping a layer combo to
2//! a moq track name + wire [`TrackKind`], and the pure helpers that turn a receiver's
3//! tile size / bandwidth into a [`QualityRequest`].
4//!
5//! Two scalability axes, deliberately mapped to two different transport mechanisms:
6//!
7//! * **Spatial** ([`SpatialLayer`]) = a SEPARATE moq track. The relay is forward-no-
8//!   transcode + group-atomic, so per-receiver selection is per-*track* subscription;
9//!   the source layer keeps its original track name (an un-upgraded peer never learns
10//!   about layers and keeps working), and the low layer is an additional track a
11//!   constrained / small-tile receiver subscribes to instead.
12//! * **Temporal** ([`TemporalLayer`]) = a per-FRAME marking WITHIN one spatial track
13//!   (a base frame is self-decodable; an enhancement frame depends on base and is
14//!   droppable). Not part of the track identity — see the producer-side gating.
15//!
16//! Keeping this here (not in `call.rs`) keeps the wire/string juggling in one tested
17//! place and out of the IO glue.
18
19use crate::TrackKind;
20use crate::congestion::{QualityRequest, camera_bitrate_bps, screen_bitrate_bps};
21
22/// Spatial scalability layer — each maps to its own moq track.
23#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
24pub enum SpatialLayer {
25    /// The full-quality stream (original track name; always published).
26    Source,
27    /// A reduced-resolution stream a constrained receiver takes instead.
28    Low,
29}
30
31/// Temporal scalability layer — a per-frame marking, not a track.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
33pub enum TemporalLayer {
34    /// Self-decodable base frame (depended on by enhancement frames).
35    Base = 0,
36    /// Droppable enhancement frame (depends only on base; never depended upon).
37    Enhanced = 1,
38}
39
40impl TemporalLayer {
41    /// Decode a wire temporal id: `0` = base, anything else = enhancement.
42    pub fn from_id(id: u8) -> Self {
43        if id == 0 { Self::Base } else { Self::Enhanced }
44    }
45
46    /// The wire temporal id.
47    pub fn id(self) -> u8 {
48        self as u8
49    }
50}

52/// Which media source a video track belongs to. Mirrors the control-plane `Source`
53/// (camera vs screen) but stays in `hound-media` so this crate needs no control dep.
54#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
55pub enum VideoSource {
56    Camera,
57    Screen,
58}
59
60/// The moq `track_name` for a (source, spatial) pair — the subscribe/publish key
61/// under the per-peer broadcast path. The source layers keep the original names so
62/// a peer that never learns about layers is unaffected.
63pub fn track_name(source: VideoSource, spatial: SpatialLayer) -> &'static str {
64    match (source, spatial) {
65        (VideoSource::Camera, SpatialLayer::Source) => "cam",
66        (VideoSource::Camera, SpatialLayer::Low) => "cam_l0",
67        (VideoSource::Screen, SpatialLayer::Source) => "screen",
68        (VideoSource::Screen, SpatialLayer::Low) => "screen_l0",
69    }
70}
71
72/// The wire [`TrackKind`] for a (source, spatial) pair.
73pub fn track_kind(source: VideoSource, spatial: SpatialLayer) -> TrackKind {
74    match (source, spatial) {
75        (VideoSource::Camera, SpatialLayer::Source) => TrackKind::CamVideo,
76        (VideoSource::Camera, SpatialLayer::Low) => TrackKind::CamVideoLow,
77        (VideoSource::Screen, SpatialLayer::Source) => TrackKind::ScrVideo,
78        (VideoSource::Screen, SpatialLayer::Low) => TrackKind::ScrVideoLow,
79    }
80}

82/// One below-source rung: a geometry (16:9 reference height) and a frame rate. Its
83/// *price* is not stored — it comes from the content-aware bitrate model
84/// ([`rung_bps`]), so a rung and its cost can never drift apart.
85///
86/// A rung is a nominal shape, not an encode geometry: the producer fits it to its
87/// own capture aspect, so a non-16:9 source is never stretched — by HEIGHT
88/// (`fit_encode_dims`) where the aspect is fixed and the height is the ladder's
89/// unit (the camera), and by AREA (`fit_encode_area`) where it is not and the
90/// rung's price is what must be held (a screen share). See [`Self::pixels`].
91#[derive(Debug, Clone, Copy, PartialEq, Eq)]
92pub struct VideoRung {
93    /// 16:9 reference height (the encoder fits this to the source's own aspect).
94    pub height: u32,
95    pub fps: u32,
96}
97
98impl VideoRung {
99    /// The rung's own pixel count at its 16:9 reference geometry.
100    pub const fn pixels(self) -> u64 {
101        (self.height as u64) * 16 / 9 * (self.height as u64)
102    }
103
104    /// The largest tile this rung may be asked to fill, in **physical** pixels:
105    /// its own area times [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`].
106    pub const fn covers_up_to_pixels(self) -> u64 {
107        self.pixels() * TILE_UPSCALE_TOLERANCE_NUM / TILE_UPSCALE_TOLERANCE_DEN
108    }
109}
110
111/// How far a rung may be stretched to fill a tile before the next rung up is worth
112/// its extra bits — an **area** ratio of 3/2, i.e. ~22% of linear upscale in each
113/// dimension. Expressed as an integer fraction so the boundary is exact.
114pub const TILE_UPSCALE_TOLERANCE_NUM: u64 = 3;
115pub const TILE_UPSCALE_TOLERANCE_DEN: u64 = 2;
116
117/// The **thumbnail** rung: a grid cell, a filmstrip stamp, an off-screen tile.
118/// 320×180 @ 15 fps ≈ 130 kbps — LiveKit ships `h180` at this geometry, Jitsi's
119/// bridge defaults its thumbnails to 180p, and Daily publishes 320×180 @ 10 fps.
120/// Against the 1 Mbps flat low rung this replaces it is **4× fewer pixels and half
121/// the frames**, which is where the large-call receive-CPU win comes from.
122pub const THUMB_RUNG: VideoRung = VideoRung {
123    height: 180,
124    fps: 15,
125};
126
127/// The **low** rung: a mid-sized tile that is still below source. 640×360 @ 20 fps
128/// ≈ 420 kbps — the old `cam_l0` geometry, re-priced (it was 1 Mbps at 30 fps).
129pub const LOW_RUNG: VideoRung = VideoRung {
130    height: 360,
131    fps: 20,
132};
133
134/// The **only** below-source screen rung. Screen share degrades by
135/// `DegradePriority::Legibility` — drop frames before pixels — so a small share
136/// tile keeps 360p (a 180p thumbnail of shared *text* is unreadable, where a 180p
137/// thumbnail of a face is fine) and gives up frame rate instead: 15 fps is the
138/// same floor the screen ladder's legibility ordering uses. ≈200 kbps against the
139/// 1 Mbps it used to cost, at identical resolution.
140pub const SCREEN_LOW_RUNG: VideoRung = VideoRung {
141    height: 360,
142    fps: 15,
143};
144
145/// A visible tile larger than this — in **physical** pixels — takes the source
146/// rather than any below-source rung: it is past what the richest low rung
147/// ([`LOW_RUNG`], and [`SCREEN_LOW_RUNG`] at the same geometry) can fill within
148/// [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`].
149///
150/// Worked examples, in the PHYSICAL pixels the UI actually reports (its own 16 px
151/// bucketing already applied, so these are the exact numbers that arrive), for the
152/// two shapes the shipped UI produces — `tile_grid.dart`'s
153/// `maxCrossAxisExtent: 360` logical grid cell and its 96 px-tall filmstrip:
154///
155/// | surface | @ DPR 1 | @ DPR 2 |
156/// |---|---|---|
157/// | grid cell | 368×208 = 76 544 px → **thumb** | 720×400 = 288 000 px → **low** |
158/// | filmstrip stamp | 176×96 = 16 896 px → **thumb** | 336×192 = 64 512 px → **thumb** |
159/// | expanded / stage | 1280×720 = 921 600 px → **source** | **source** |
160///
161/// The 2× grid cell is the case this boundary exists for: it is a genuine 720-wide
162/// surface, and filling it from a 320-wide thumbnail would be a 2.25× linear
163/// upscale of a face.
164pub const TILE_SOURCE_MIN_PIXELS: u64 = LOW_RUNG.covers_up_to_pixels();
165
166/// What one below-source rung costs for this content, from the bits-per-pixel
167/// model in [`crate::congestion`]. Camera and screen are priced by different
168/// models: screen content (text, flat regions, sharp edges) compresses several
169/// times better than motion video, so pricing a shared window like a talking head
170/// would oversize it.
171pub fn rung_bps(source: VideoSource, rung: VideoRung) -> u64 {
172    match source {
173        VideoSource::Camera => camera_bitrate_bps(rung.height, rung.fps),
174        VideoSource::Screen => screen_bitrate_bps(rung.height, rung.fps),
175    }
176}
177
178/// What the **source** rung costs — the top-rung bitrate the client publishes this
179/// content at (`CAM_BITRATE` / `SCREEN_BITRATE` in `crates/app/client/src/call/mod.rs`,
180/// and the `source_bps` the producer's [`crate::plan_layers`] coalesces against).
181///
182/// Stated as constants rather than run back through [`rung_bps`]: the camera figure
183/// is the anchor the whole bits-per-pixel model was *solved from* (a test below pins
184/// the identity), and the screen figure is a published ceiling the ladder climbs
185/// toward rather than a modelled point. [`crate::allocate`] needs both so a `Source`
186/// demand can be priced on the same scale as a below-source rung.
187pub const CAM_SOURCE_BPS: u64 = 1_500_000;
188pub const SCREEN_SOURCE_BPS: u64 = 2_500_000;
189
190/// The price of taking the SOURCE track for this content — see [`CAM_SOURCE_BPS`].
191pub fn source_bps(source: VideoSource) -> u64 {
192    match source {
193        VideoSource::Camera => CAM_SOURCE_BPS,
194        VideoSource::Screen => SCREEN_SOURCE_BPS,
195    }
196}

198/// The below-source rungs available for `source`, **cheapest first**. Every caller
199/// walks this in order; the ascending-price invariant is asserted in debug builds
200/// by [`debug_assert_ascending`].
201pub(crate) fn low_rungs(source: VideoSource) -> &'static [VideoRung] {
202    match source {
203        VideoSource::Camera => &[THUMB_RUNG, LOW_RUNG],
204        // Legibility: no thumbnail rung for shared content — see [`SCREEN_LOW_RUNG`].
205        VideoSource::Screen => &[SCREEN_LOW_RUNG],
206    }
207}

209/// The ladder must be strictly ascending in price AND in area, or "cheapest rung
210/// that covers the tile" and "richest rung this cap affords" stop being inverses of
211/// each other and a tile could round-trip to a rung it never asked for.
212fn debug_assert_ascending(source: VideoSource, rungs: &[VideoRung]) {
213    debug_assert!(
214        rungs
215            .windows(2)
216            .all(|w| rung_bps(source, w[0]) < rung_bps(source, w[1])
217                && w[0].pixels() < w[1].pixels()),
218        "{source:?} rung ladder is not strictly ascending: {rungs:?}"
219    );
220}

222/// Map a rendered tile's **physical** pixel size + visibility to a
223/// [`QualityRequest`]. This is the *demand* half of demand-driven simulcast — it
224/// never asks for more than the tile can use, which is exactly the receiver-CPU win
225/// (a thumbnail pulls the low track).
226///
227/// `width`/`height` are physical (device) pixels, not logical ones: decode and
228/// render cost scale with the pixels the GPU actually paints, so the same 360-wide
229/// grid cell is a thumbnail on a 1× display and a genuine 720-wide surface on a 2×
230/// one. The UI multiplies by `MediaQuery.devicePixelRatioOf` before reporting.
231///
232/// The cap is **derived from the tile's own pixel count**, not a constant: the tile
233/// takes the cheapest rung that covers it (within
234/// [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`]), priced through
235/// [`rung_bps`]; past the richest rung it takes the source. A 1 Mbps flat cap —
236/// what this used to return for *every* sub-360p tile, including a 171×96 filmstrip
237/// stamp — is 6–12× the per-tile budget Zoom, Daily and LiveKit all converged on,
238/// and it was the dominant cost of a large call.
239pub fn tile_quality_request(
240    source: VideoSource,
241    width: u32,
242    height: u32,
243    visible: bool,
244) -> QualityRequest {
245    let rungs = low_rungs(source);
246    debug_assert_ascending(source, rungs);
247    let pixels = (width as u64) * (height as u64);
248    // Not rendered: the receiver subscribes to NOTHING for this tile
249    // (`call::subscribe::pick_layer`), and reports the cheapest rung flagged so the
250    // producer's fold ignores it while anyone is actually watching — yet still
251    // sizes the layer when every requester is hidden (see [`crate::plan_layers`]).
252    if !visible {
253        return QualityRequest::Hidden(rung_bps(source, rungs[0]));
254    }
255    rungs
256        .iter()
257        .find(|r| pixels <= r.covers_up_to_pixels())
258        .map_or(QualityRequest::Source, |r| {
259            QualityRequest::Capped(rung_bps(source, *r))
260        })
261}

263/// Invert a coalesced cap back into the rung the producer should encode: the
264/// richest rung this cap can pay for, never below the cheapest one.
265///
266/// The producer cannot simply look the cap up: `RequestQuality { cap_bps }` carries
267/// a *bitrate*, and the cap that arrives has been folded ([`most_constrained`])
268/// with the receiver's bandwidth-driven demand (`DownlinkDemand`), which produces
269/// arbitrary values — and it is a **cross-receiver minimum** over everyone still
270/// rendering the track (see [`crate::plan_layers`]). So the mapping is "what fits",
271/// not "what was asked for" — and a cap below even the cheapest rung still yields
272/// that rung rather than nothing, because the encoder's own bitrate is then set to
273/// the cap and the geometry is the floor the ladder allows (the same
274/// never-degenerate-toward-zero contract as `DownlinkDemand`'s `MIN_CAP_BPS`).
275pub fn low_rung_for_cap(source: VideoSource, cap_bps: u64) -> VideoRung {
276    let rungs = low_rungs(source);
277    debug_assert_ascending(source, rungs);
278    rungs
279        .iter()
280        .rev()
281        .find(|r| rung_bps(source, **r) <= cap_bps)
282        .copied()
283        .unwrap_or(rungs[0])
284}

286/// How many remote video peers a call must have before a peer that has been
287/// *observed silent* is priced a rung below what its tile alone would buy.
288///
289/// **20**, which is where Jitsi's `conference-last-n-limits { 20 = 20, 30 = 15,
290/// 50 = 10, 90 = 2 }` starts tightening: selectivity is a function of room size,
291/// because a small call can simply afford everyone and the *visible* cost of being
292/// wrong about who matters (a 2× grid cell filled from a 180p thumbnail is a
293/// noticeable 2.25× upscale of a face) is not worth paying there. At 8 peers the
294/// size-derived rungs cost ~3.4 Mbps and are affordable; at 30 they are ~12 Mbps
295/// and are not. Below this line ranking only ever *promotes*.
296pub const SPEAKER_DEMOTE_MIN_PEERS: usize = 20;
297
298/// Modulate a tile's own demand by where its peer sits in the speaking ranking —
299/// the receiver-side Last-N lever, applied as a **modifier on the size-derived
300/// rung** rather than as a replacement for it.
301///
302/// * the **dominant** speaker moves up one rung (a grid cell's `THUMB` becomes
303///   `LOW`), which is Jitsi's `onstage-preferred-height-px = 360` against its
304///   `default-max-height-px = 180`, expressed in this ladder's terms;
305/// * a peer **observed silent** for the whole long window, in a crowded call, moves
306///   down one rung;
307/// * everyone else — including every peer we have no speaking evidence about, e.g.
308///   an un-upgraded client that never reports a level — is left exactly as the tile
309///   priced it.
310///
311/// Three properties make this composable with everything P1/P2 built, and each is
312/// pinned by a test below:
313///
314/// 1. **The layer axis is invariant.** `Hidden` and `Source` pass through untouched
315///    and a `Capped` demand stays `Capped`, so modulation can never turn a
316///    subscription on, off, or into a different track. It moves a *cap*, which the
317///    `LayerHysteresis` same-axis path commits immediately — so RECEIVER-side, P2's
318///    churn bounds are not merely respected, they are not engaged (no resubscribe,
319///    no pipeline restart, no dwell consumed).
320///
321///    That is only the receiver's half, and the producer's is not free: a sustained
322///    handover moves the cross-receiver MINIMUM cap (`plan_layers`), so the promoted
323///    peer's shared low layer changes rung — an encoder rebuild plus an IDR to every
324///    subscriber of it. Bounded, not absent: at most one such rebuild per
325///    [`crate::GEOMETRY_REBUILD_DWELL_US`] per producer, which is exactly why
326///    [`crate::speaking::SWITCH_DWELL_US`] is pinned to that dwell.
327/// 2. **Never above the ladder, never below it.** A promotion stops at the richest
328///    below-source rung (`Source` is the tile-size signal's decision alone — a user
329///    staring at an expanded tile gets sharp video whether or not they are talking),
330///    and a demotion stops at the cheapest.
331/// 3. **Silence must be evidence.** `SpeakerTier::Unknown` is never demoted; only
332///    `Silent`, which the ranker only reports after watching a peer for its whole
333///    long window. Same two-tier trust rule as `TileDemand`'s reported-vs-seeded hide.
334///
335/// Apply this to the TILE demand, before folding with the receiver's own bandwidth
336/// demand (`most_constrained`): a promotion must never lift a receiver above what
337/// its link estimate allows.
338pub fn speaker_adjusted(
339    source: VideoSource,
340    req: QualityRequest,
341    tier: crate::speaking::SpeakerTier,
342    peers: usize,
343) -> QualityRequest {
344    use crate::speaking::SpeakerTier;
345    let QualityRequest::Capped(bps) = req else {
346        return req;
347    };
348    let rungs = low_rungs(source);
349    debug_assert_ascending(source, rungs);
350    // Where this cap sits on the ladder — the same "richest rung it can afford"
351    // inversion the producer runs ([`low_rung_for_cap`]), as an index.
352    let at = rungs
353        .iter()
354        .rposition(|r| rung_bps(source, *r) <= bps)
355        .unwrap_or(0);
356    let want = match tier {
357        SpeakerTier::Dominant => (at + 1).min(rungs.len() - 1),
358        SpeakerTier::Silent if peers >= SPEAKER_DEMOTE_MIN_PEERS => at.saturating_sub(1),
359        _ => at,
360    };
361    if want == at {
362        req
363    } else {
364        QualityRequest::Capped(rung_bps(source, rungs[want]))
365    }
366}

368/// Fold two demands into the more-constrained one. `Hidden` beats everything (a
369/// tile nobody is looking at cannot become one that is by way of a bandwidth
370/// estimate), `Capped` beats `Source`, and two of a kind take the lower cap. Used
371/// to combine a tile-size demand with a bandwidth demand so the receiver can never
372/// over-request beyond what either signal allows.
373pub fn most_constrained(a: QualityRequest, b: QualityRequest) -> QualityRequest {
374    use QualityRequest::{Capped, Hidden, Source};
375    match (a, b) {
376        // Hidden wins, at the lower of the two caps it could be carrying.
377        (Hidden(x), Hidden(y)) => Hidden(x.min(y)),
378        (Hidden(x), Capped(y)) | (Capped(y), Hidden(x)) => Hidden(x.min(y)),
379        (Hidden(x), Source) | (Source, Hidden(x)) => Hidden(x),
380        (Source, other) | (other, Source) => other,
381        (Capped(x), Capped(y)) => Capped(x.min(y)),
382    }
383}

457    /// THE re-pricing contract: a tile's cap comes from its own PHYSICAL pixel
458    /// count. The two shapes the shipped UI renders are checked at both display
459    /// densities, because the same logical cell is a different amount of decode and
460    /// render work on a 1× and a 2× display — which is the whole reason the demand
461    /// is reported in physical pixels.
462    #[test]
463    fn tile_cap_is_priced_from_the_tile() {
464        use VideoSource::Camera;
465        let thumb = rung_bps(Camera, THUMB_RUNG);
466        let low = rung_bps(Camera, LOW_RUNG);
467        assert!(thumb < low, "the ladder must be strictly ascending");
468
469        // The exact reports the UI sends for the shipped grid cell, bucketing
470        // included (`ui/test/demand_test.dart` pins the other side of this pair).
471        // At 1× it is a thumbnail; at 2× it is a real 720-wide surface and takes
472        // LOW — filling it from 180p would be a 2.25× linear upscale of a face.
473        assert_eq!(cap(tile_quality_request(Camera, 368, 208, true)), thumb);
474        assert_eq!(cap(tile_quality_request(Camera, 720, 400, true)), low);
475        // The filmstrip stamp stays on the thumbnail rung at BOTH densities — it is
476        // 96 logical px tall by design.
477        assert_eq!(cap(tile_quality_request(Camera, 176, 96, true)), thumb);
478        assert_eq!(cap(tile_quality_request(Camera, 336, 192, true)), thumb);

480        assert_eq!(cap(tile_quality_request(Camera, 320, 180, true)), thumb);
481        assert_eq!(cap(tile_quality_request(Camera, 640, 360, true)), low);
482        // And it is a real saving, not a rename: an order of magnitude under the

484        assert!(thumb * 6 < 1_000_000, "thumbnail rung still costs {thumb}");

487    /// The tolerance is an AREA ratio, and the boundary sits exactly where the

489    #[test]
490    fn a_rung_covers_its_own_area_times_the_tolerance() {
491        use VideoSource::Camera;
492        assert_eq!(THUMB_RUNG.pixels(), 320 * 180);

510        let boundary = TILE_SOURCE_MIN_PIXELS;

589    /// Round trip: what a tile asks for is what the producer builds. This is the
590    /// property that makes the ladder work end to end — the demand crosses the
591    /// wire as a bare bitrate, so a rung that could not be recovered from its own
592    /// price would silently encode at the wrong geometry.
593    ///
594    /// Every case here is a tile shape the ladder genuinely produces (both display
595    /// densities of the two shipped surfaces), so a rung is never asserted for a
596    /// size that could not ask for it.
597    #[test]
598    fn every_tile_demand_round_trips_to_its_rung() {
599        use VideoSource::Camera;
600        for (tile, want) in [
601            ((176u32, 96u32), THUMB_RUNG), // filmstrip stamp @1x
602            ((336, 192), THUMB_RUNG),      // filmstrip stamp @2x
603            ((320, 180), THUMB_RUNG),      // the rung's own geometry
604            ((368, 208), THUMB_RUNG),      // grid cell @1x
605            ((480, 272), LOW_RUNG),        // a mid-sized tile
606            ((640, 360), LOW_RUNG),        // the rung's own geometry
607            ((720, 400), LOW_RUNG),        // grid cell @2x
608        ] {
609            let asked = cap(tile_quality_request(Camera, tile.0, tile.1, true));
610            assert_eq!(
611                low_rung_for_cap(Camera, asked),
612                want,
613                "a {}×{} tile asked {asked} bps and got the wrong rung back",
614                tile.0,
615                tile.1
616            );
617        }
618        // A hidden tile round-trips too — its carried cap is a real rung price.
619        let QualityRequest::Hidden(bps) = tile_quality_request(Camera, 368, 208, false) else {
620            panic!("a hidden tile must report Hidden");
621        };
622        assert_eq!(low_rung_for_cap(Camera, bps), THUMB_RUNG);
623    }

654    /// THE composability property: modulation moves a cap, never a track. If this
655    /// ever fails, ranking has become able to subscribe, unsubscribe or layer-switch
656    /// a tile behind the hysteresis's back — and P2's churn bounds stop holding.
657    #[test]
658    fn speaker_modulation_never_changes_the_layer_axis() {
659        use crate::speaking::SpeakerTier::{Dominant, Recent, Silent, Unknown};
660        use VideoSource::{Camera, Screen};
661        let axis = |r: QualityRequest| match r {
662            QualityRequest::Hidden(_) => 0,
663            QualityRequest::Capped(_) => 1,
664            QualityRequest::Source => 2,
665        };
666        for source in [Camera, Screen] {
667            for req in [
668                QualityRequest::Hidden(130_000),
669                QualityRequest::Capped(rung_bps(source, THUMB_RUNG)),
670                QualityRequest::Capped(rung_bps(source, LOW_RUNG)),
671                QualityRequest::Capped(1), // a bandwidth-crushed cap under every rung
672                QualityRequest::Capped(u64::MAX),
673                QualityRequest::Source,
674            ] {
675                for tier in [Dominant, Recent, Unknown, Silent] {
676                    for peers in [1, 8, 32] {
677                        let out = speaker_adjusted(source, req, tier, peers);
678                        assert_eq!(
679                            axis(out),
680                            axis(req),
681                            "{source:?} {req:?} at {tier:?} in a {peers}-peer call \
682                             changed layer axis to {out:?}"
683                        );
684                        // And whatever it produced is a rung the producer can build.
685                        if let QualityRequest::Capped(bps) = out {
686                            assert!(bps > 0, "a modulated cap must stay payable");
687                        }
688                    }
689                }
690            }
691        }
692    }
