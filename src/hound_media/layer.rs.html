1//! The video **layer model** — the single source of truth mapping a layer combo to
2//! a moq track name + wire [`TrackKind`], and the pure helpers that turn a receiver's
3//! tile size / bandwidth into a [`QualityRequest`].
4//!
5//! Two scalability axes, deliberately mapped to two different transport mechanisms:
6//!
7//! * **Spatial** ([`SpatialLayer`]) = a SEPARATE moq track. The relay is forward-no-
8//!   transcode + group-atomic, so per-receiver selection is per-*track* subscription;
9//!   the source layer keeps its original track name (an un-upgraded peer never learns
10//!   about layers and keeps working), and the low layer is an additional track a
11//!   constrained / small-tile receiver subscribes to instead.
12//! * **Temporal** ([`TemporalLayer`]) = a per-FRAME marking WITHIN one spatial track
13//!   (a base frame is self-decodable; an enhancement frame depends on base and is
14//!   droppable). The marking is not part of the track identity — one publication
15//!   carries both layers — but the SUBSCRIPTION is: the relay republishes a derived
16//!   base-layer-only track (`services/sfu::derive`) that a receiver takes instead of
17//!   the whole one. [`TemporalView`] is that choice, and it is deliberately a
18//!   different type from [`TemporalLayer`]: one says what a frame IS, the other says
19//!   which subset of them was asked for.
20//!
21//! The two axes are asymmetric in who decides. A spatial rung is a track the
22//! PRODUCER had to be asked to encode (`RequestQuality`, coalesced across every
23//! receiver by [`crate::plan_layers`]); a temporal view is carved out of an existing
24//! publication by the relay, so it is decided by one receiver alone and never
25//! travels upstream. That asymmetry is why [`TemporalView`] rides
26//! [`crate::VideoAllocation`] rather than [`crate::QualityRequest`] — see its doc.
27//!
28//! Keeping this here (not in `call.rs`) keeps the wire/string juggling in one tested
29//! place and out of the IO glue.

34/// Spatial scalability layer — each maps to its own moq track.
35#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
36pub enum SpatialLayer {
37    /// The full-quality stream (original track name; always published).
38    Source,
39    /// A reduced-resolution stream a constrained receiver takes instead.
40    Low,
41}

43/// Temporal scalability layer — a per-frame marking, not a track.
44#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
45pub enum TemporalLayer {
46    /// Self-decodable base frame (depended on by enhancement frames).
47    Base = 0,
48    /// Droppable enhancement frame (depends only on base; never depended upon).
49    Enhanced = 1,
50}
51
52impl TemporalLayer {
53    /// Decode a wire temporal id: `0` = base, anything else = enhancement.
54    pub fn from_id(id: u8) -> Self {
55        if id == 0 { Self::Base } else { Self::Enhanced }
56    }
57
58    /// The wire temporal id.
59    pub fn id(self) -> u8 {
60        self as u8
61    }
62
63    /// Whether a frame at this layer may be dropped without breaking the decode
64    /// of anything that follows. The whole L1T2 contract in one predicate: base
65    /// frames are referenced, enhancement frames never are.
66    pub fn discardable(self) -> bool {
67        matches!(self, Self::Enhanced)
68    }
69}

71/// The number of bits the wire temporal id occupies in the frame envelope's
72/// reserved byte (`hound_transport`'s 16-byte MoQ object header, byte 3).
73///
74/// Three bits — RFC 9626's frame-marking `TID` width — so the field can carry an
75/// L1T3 ladder later without a wire change, even though only L1T2 is produced
76/// today. The remaining 5 bits stay zero and are reserved; a decoder MUST mask
77/// rather than compare the whole byte, or a future flag in bit 3 reads as a
78/// temporal id of 8.
79pub const TEMPORAL_ID_MASK: u8 = 0b0000_0111;

81/// The **temporal view** a receiver subscribes to — a different thing from
82/// [`TemporalLayer`], which is the per-frame marking on the wire.
83///
84/// [`TemporalLayer`] answers "what is this frame?"; `TemporalView` answers "which
85/// subset of a track's frames did I ask for?". The two meet at the SFU: a
86/// [`Self::BaseOnly`] subscription is served from a **derived** track
87/// ([`base_track_name`]) carrying only [`TemporalLayer::Base`] objects, filtered
88/// on hound's own envelope byte — no transcode, no bitstream parse.
89///
90/// This axis deliberately does **not** travel upstream on `RequestQuality`. The
91/// producer publishes one full-rate track either way (the base layer is carved
92/// out of it by the relay, not encoded separately), so telling the producer would
93/// only invite it to change what it *encodes* — and a `Capped` demand is folded
94/// into the cross-receiver minimum, which is exactly the legibility trap
95/// [`SCREEN_MID_RUNG`] documents. It is a receiver-local subscription choice and
96/// stays one.
97///
98/// Ordered cheapest-first so `BaseOnly < Full` falls out of a comparison.
99#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Default)]
100pub enum TemporalView {
101    /// Base-layer frames only: same resolution, half the frame rate, ~70% of the
102    /// bitrate. Served from the relay's derived `…_t0` track.
103    BaseOnly,
104    /// Every frame of the track (TL0 + TL1) — the original track name.
105    #[default]
106    Full,
107}

109/// Track-name suffix of the relay's derived base-layer-only track. One string,
110/// shared by the client's subscribe path and the SFU's deriver, so the two can
111/// never disagree about what a `_t0` track is called.
112pub const TEMPORAL_BASE_SUFFIX: &str = "_t0";
113
114/// The derived base-layer track name for `track` (`"cam"` → `"cam_t0"`). Used by
115/// the SFU to name the broadcast it republishes and by the receiver to name the
116/// track it subscribes to.
117pub fn base_track_name(track: &str) -> String {
118    format!("{track}{TEMPORAL_BASE_SUFFIX}")
119}

121/// Whether `track` is already a derived base-layer track — the guard that stops
122/// the SFU deriving `cam_t0_t0` off its own output.
123pub fn is_base_track_name(track: &str) -> bool {
124    track.ends_with(TEMPORAL_BASE_SUFFIX)
125}

127/// What fraction of a track's bitrate the base layer carries, as an exact integer
128/// ratio: **0.7**, which is the `BaseLayerBitRateFraction` the VideoToolbox
129/// encoder is configured with (`vt_encoder.rs`). Half the frames at ~70% of the
130/// bits, because the base layer's inter-frame distance doubles — dropping half
131/// the frames does not halve the bytes.
132///
133/// Stated here rather than run through [`crate::congestion::camera_bitrate_bps`]
134/// at half fps for the same reason [`CAM_SOURCE_BPS`] is a constant: the price of
135/// the base layer is decided by the **encoder's split**, not by the ladder's
136/// bits-per-pixel model, and a receiver that budgeted it any other way would be
137/// budgeting for bytes that are not on the wire.
138pub const BASE_LAYER_BITRATE_NUM: u64 = 7;
139pub const BASE_LAYER_BITRATE_DEN: u64 = 10;
140
141/// The price of the base-layer-only view of a stream costing `full_bps`.
142pub const fn base_only_bps(full_bps: u64) -> u64 {
143    full_bps / BASE_LAYER_BITRATE_DEN * BASE_LAYER_BITRATE_NUM
144}

146/// Whether a `(source, spatial)` track has a base-layer-only sibling the receiver
147/// may subscribe to — i.e. whether the SFU derives a `…_t0` for it.
148///
149/// **Camera source track only**, and both halves of that are deliberate:
150///
151/// * **Camera, not screen.** Screen degrades by `DegradePriority::Legibility` —
152///   frames before pixels — and its ladder has already spent that budget:
153///   [`SCREEN_LOW_RUNG`] is 15 fps and [`SCREEN_MID_RUNG`] is the sharp rung a
154///   reader wants. Halving a 15 fps share to 7.5 fps is a slideshow of someone's
155///   scrolling editor, and halving the 30 fps mid rung lands on the same 15 fps
156///   the floor already offers at a fifth of the price. There is no step here
157///   worth a second derived track.
158/// * **The source track, not `_l0`.** The gap the camera ladder has no rung for
159///   is [`LOW_RUNG`] (420 kbps) → source (1.5 Mbps) — the same "one enormous
160///   step" shape [`SCREEN_MID_RUNG`] was built for. Below that, `_l0`'s rungs are
161///   already at the frame-rate floor for a recognisable face (the thumbnail is
162///   15 fps; halved it is 7.5), and every extra derived track puts the SFU on the
163///   per-object path for another stream — the cost this design is otherwise
164///   careful to keep proportional to what is actually watched.
165///
166/// So one derived track per camera publisher, buying the ladder a 720p rung at
167/// half frame rate for ~70% of the source's price.
168pub fn base_view_offered(source: VideoSource, spatial: SpatialLayer) -> bool {
169    matches!(
170        (source, spatial),
171        (VideoSource::Camera, SpatialLayer::Source)
172    )
173}

175/// The moq `track_name` for a (source, spatial, view) triple, or `None` when that
176/// track has no such view ([`base_view_offered`]). [`track_name`] is the
177/// `TemporalView::Full` case.
178pub fn track_name_for(
179    source: VideoSource,
180    spatial: SpatialLayer,
181    view: TemporalView,
182) -> Option<&'static str> {
183    match view {
184        TemporalView::Full => Some(track_name(source, spatial)),
185        // Spelled as a literal rather than built from `track_name` + the suffix so
186        // the name is a `&'static str` like every other track name; the test below
187        // pins the two constructions equal.
188        TemporalView::BaseOnly => match (source, spatial) {
189            (VideoSource::Camera, SpatialLayer::Source) => Some("cam_t0"),
190            _ => None,
191        },
192    }
193}

225/// One below-source rung: a geometry (16:9 reference height) and a frame rate. Its
226/// *price* is not stored — it comes from the content-aware bitrate model
227/// ([`rung_bps`]), so a rung and its cost can never drift apart.
228///
229/// A rung is a nominal shape, not an encode geometry: the producer fits it to its
230/// own capture aspect, so a non-16:9 source is never stretched — by HEIGHT
231/// (`fit_encode_dims`) where the aspect is fixed and the height is the ladder's
232/// unit (the camera), and by AREA (`fit_encode_area`) where it is not and the
233/// rung's price is what must be held (a screen share). See [`Self::pixels`].
234#[derive(Debug, Clone, Copy, PartialEq, Eq)]
235pub struct VideoRung {
236    /// 16:9 reference height (the encoder fits this to the source's own aspect).
237    pub height: u32,
238    pub fps: u32,
239}
240
241impl VideoRung {
242    /// The rung's own pixel count at its 16:9 reference geometry.
243    pub const fn pixels(self) -> u64 {
244        (self.height as u64) * 16 / 9 * (self.height as u64)
245    }
246
247    /// The largest tile this rung may be asked to fill, in **physical** pixels:
248    /// its own area times [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`].
249    pub const fn covers_up_to_pixels(self) -> u64 {
250        self.pixels() * TILE_UPSCALE_TOLERANCE_NUM / TILE_UPSCALE_TOLERANCE_DEN
251    }
252}
253
254/// How far a rung may be stretched to fill a tile before the next rung up is worth
255/// its extra bits — an **area** ratio of 3/2, i.e. ~22% of linear upscale in each
256/// dimension. Expressed as an integer fraction so the boundary is exact.
257pub const TILE_UPSCALE_TOLERANCE_NUM: u64 = 3;
258pub const TILE_UPSCALE_TOLERANCE_DEN: u64 = 2;
259
260/// The **thumbnail** rung: a grid cell, a filmstrip stamp, an off-screen tile.
261/// 320×180 @ 15 fps ≈ 130 kbps — LiveKit ships `h180` at this geometry, Jitsi's
262/// bridge defaults its thumbnails to 180p, and Daily publishes 320×180 @ 10 fps.
263/// Against the 1 Mbps flat low rung this replaces it is **4× fewer pixels and half
264/// the frames**, which is where the large-call receive-CPU win comes from.
265pub const THUMB_RUNG: VideoRung = VideoRung {
266    height: 180,
267    fps: 15,
268};

270/// The **low** rung: a mid-sized tile that is still below source. 640×360 @ 20 fps
271/// ≈ 420 kbps — the old `cam_l0` geometry, re-priced (it was 1 Mbps at 30 fps).
272pub const LOW_RUNG: VideoRung = VideoRung {
273    height: 360,
274    fps: 20,
275};

277/// The **floor** below-source screen rung. Screen share degrades by
278/// `DegradePriority::Legibility` — drop frames before pixels — so a small share
279/// tile keeps 360p (a 180p thumbnail of shared *text* is unreadable, where a 180p
280/// thumbnail of a face is fine) and gives up frame rate instead: 15 fps is the
281/// same floor the screen ladder's legibility ordering uses. ≈200 kbps against the
282/// 1 Mbps it used to cost, at identical resolution.
283pub const SCREEN_LOW_RUNG: VideoRung = VideoRung {
284    height: 360,
285    fps: 15,
286};

288/// The **intermediate** screen rung: 720p @ 30 fps ≈ 1.1 Mbps, sitting between
289/// [`SCREEN_LOW_RUNG`] (200 kbps) and the 2.5 Mbps source.
290///
291/// It exists to shrink the screen ladder's one enormous step. With 360p15 as the
292/// only below-source rung a share's whole ladder was `200 kbps → 2.5 Mbps`, and
293/// [`crate::allocate_video`]'s cheapest-first upgrade phase therefore left every
294/// budget in a 2.3 Mbps-wide band buying the share nothing at all: it went from
295/// "barely legible" straight to "more than the link has". Two steps of 900 kbps
296/// and 1.4 Mbps instead, which is what makes a 5 Mbps receiver watching a share
297/// plus two cameras land on a sharp 720p30 share rather than a 360p15 one.
298///
299/// 720p30 is also the legibility ladder's own next stop — `screen_rungs`' floor
300/// resolution is 720p ("below that text is unreadable") — so the intermediate rung
301/// is legible content at a real frame rate, not a compromise between the two.
302///
303/// **It is a BUDGET rung, not a SIZE rung.** A tile's own size never asks for it:
304/// [`tile_quality_request`] stops a screen tile's size-derived walk at
305/// [`SCREEN_LOW_RUNG`]'s coverage, so a share's size demand is still exactly
306/// floor-or-`Source`. This rung is reachable only through [`crate::allocate_video`]
307/// — as an upgrade step a `Source`-demanding share takes when the budget affords
308/// 1.1 Mbps but not 2.5, and as the step a deficit walks back down through. The
309/// reason is the coalesced minimum, and it is load-bearing: a `Source` demand is
310/// immune to it (`plan_layers` routes it to the source track), a `Capped` demand is
311/// not, so a stage share that asked for `Capped(1.1 Mbps)` alongside anyone
312/// rendering a grid-cell share would be *folded down* to the 360p15 floor and
313/// rendered 2× upscaled — which is the legibility failure `SCREEN_LOW_RUNG` exists
314/// to prevent, inflicted on the one receiver actually reading the text. See
315/// [`tile_quality_request`]'s walk.
316pub const SCREEN_MID_RUNG: VideoRung = VideoRung {
317    height: 720,
318    fps: 30,
319};
320
321/// A visible tile larger than this — in **physical** pixels — takes the source
322/// rather than any below-source rung: it is past what the richest rung a tile's
323/// SIZE may select ([`LOW_RUNG`], and [`SCREEN_LOW_RUNG`] at the same geometry) can
324/// fill within [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`].
325///
326/// One boundary for both ladders, and it stayed one when [`SCREEN_MID_RUNG`]
327/// landed: that rung is reachable from the allocator, never from a tile's size, so
328/// the screen ladder's size boundary is still its floor rung's coverage — the same
329/// 345 600 px, since the two floors share the 360p geometry. See
330/// [`tile_quality_request`] for why the size walk must stop there.
331///
332/// Worked examples, in the PHYSICAL pixels the UI actually reports (its own 16 px
333/// bucketing already applied, so these are the exact numbers that arrive), for the
334/// two shapes the shipped UI produces — `tile_grid.dart`'s
335/// `maxCrossAxisExtent: 360` logical grid cell and its 96 px-tall filmstrip:
336///
337/// | surface | @ DPR 1 | @ DPR 2 |
338/// |---|---|---|
339/// | grid cell | 368×208 = 76 544 px → **thumb** | 720×400 = 288 000 px → **low** |
340/// | filmstrip stamp | 176×96 = 16 896 px → **thumb** | 336×192 = 64 512 px → **thumb** |
341/// | expanded / stage | 1280×720 = 921 600 px → **source** | **source** |
342///
343/// The 2× grid cell is the case this boundary exists for: it is a genuine 720-wide
344/// surface, and filling it from a 320-wide thumbnail would be a 2.25× linear
345/// upscale of a face.
346pub const TILE_SOURCE_MIN_PIXELS: u64 = LOW_RUNG.covers_up_to_pixels();

348/// What one below-source rung costs for this content, from the bits-per-pixel
349/// model in [`crate::congestion`]. Camera and screen are priced by different
350/// models: screen content (text, flat regions, sharp edges) compresses several
351/// times better than motion video, so pricing a shared window like a talking head
352/// would oversize it.
353pub fn rung_bps(source: VideoSource, rung: VideoRung) -> u64 {
354    match source {
355        VideoSource::Camera => camera_bitrate_bps(rung.height, rung.fps),
356        VideoSource::Screen => screen_bitrate_bps(rung.height, rung.fps),
357    }
358}

360/// What the **source** rung costs — the top-rung bitrate the client publishes this
361/// content at (`CAM_BITRATE` / `SCREEN_BITRATE` in `crates/app/client/src/call/mod.rs`,
362/// and the `source_bps` the producer's [`crate::plan_layers`] coalesces against).
363///
364/// Stated as constants rather than run back through [`rung_bps`]: the camera figure
365/// is the anchor the whole bits-per-pixel model was *solved from* (a test below pins
366/// the identity), and the screen figure is a published ceiling the ladder climbs
367/// toward rather than a modelled point. [`crate::allocate`] needs both so a `Source`
368/// demand can be priced on the same scale as a below-source rung.
369pub const CAM_SOURCE_BPS: u64 = 1_500_000;
370pub const SCREEN_SOURCE_BPS: u64 = 2_500_000;
371
372/// The price of taking the SOURCE track for this content — see [`CAM_SOURCE_BPS`].
373pub fn source_bps(source: VideoSource) -> u64 {
374    match source {
375        VideoSource::Camera => CAM_SOURCE_BPS,
376        VideoSource::Screen => SCREEN_SOURCE_BPS,
377    }

380/// The below-source rungs available for `source`, **cheapest first**. Every caller
381/// walks this in order; the ascending-price invariant is asserted in debug builds
382/// by [`debug_assert_ascending`].
383pub(crate) fn low_rungs(source: VideoSource) -> &'static [VideoRung] {
384    match source {
385        VideoSource::Camera => &[THUMB_RUNG, LOW_RUNG],
386        // Legibility: no thumbnail rung for shared content — the floor is 360p and
387        // gives up frame rate (see [`SCREEN_LOW_RUNG`]) — and one intermediate rung
388        // above it so the climb to the source is not a single 2.3 Mbps leap
389        // ([`SCREEN_MID_RUNG`], an allocator step only: `tile_quality_request`
390        // never asks for it from a tile's size).
391        VideoSource::Screen => &[SCREEN_LOW_RUNG, SCREEN_MID_RUNG],
392    }
393}

395/// The ladder must be strictly ascending in price AND in area, or "cheapest rung
396/// that covers the tile" and "richest rung this cap affords" stop being inverses of
397/// each other and a tile could round-trip to a rung it never asked for.
398fn debug_assert_ascending(source: VideoSource, rungs: &[VideoRung]) {
399    debug_assert!(
400        rungs
401            .windows(2)
402            .all(|w| rung_bps(source, w[0]) < rung_bps(source, w[1])
403                && w[0].pixels() < w[1].pixels()),
404        "{source:?} rung ladder is not strictly ascending: {rungs:?}"
405    );
406}

408/// Map a rendered tile's **physical** pixel size + visibility to a
409/// [`QualityRequest`]. This is the *demand* half of demand-driven simulcast — it
410/// never asks for more than the tile can use, which is exactly the receiver-CPU win
411/// (a thumbnail pulls the low track).
412///
413/// `width`/`height` are physical (device) pixels, not logical ones: decode and
414/// render cost scale with the pixels the GPU actually paints, so the same 360-wide
415/// grid cell is a thumbnail on a 1× display and a genuine 720-wide surface on a 2×
416/// one. The UI multiplies by `MediaQuery.devicePixelRatioOf` before reporting.
417///
418/// The cap is **derived from the tile's own pixel count**, not a constant: the tile
419/// takes the cheapest rung that covers it (within
420/// [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`]), priced through
421/// [`rung_bps`]; past the richest rung it takes the source. A 1 Mbps flat cap —
422/// what this used to return for *every* sub-360p tile, including a 171×96 filmstrip
423/// stamp — is 6–12× the per-tile budget Zoom, Daily and LiveKit all converged on,
424/// and it was the dominant cost of a large call.
425pub fn tile_quality_request(
426    source: VideoSource,
427    width: u32,
428    height: u32,
429    visible: bool,
430) -> QualityRequest {
431    let rungs = low_rungs(source);
432    debug_assert_ascending(source, rungs);
433    let pixels = (width as u64) * (height as u64);
434    // Not rendered: the receiver subscribes to NOTHING for this tile
435    // (`call::subscribe::pick_layer`), and reports the cheapest rung flagged so the
436    // producer's fold ignores it while anyone is actually watching — yet still
437    // sizes the layer when every requester is hidden (see [`crate::plan_layers`]).
438    if !visible {
439        return QualityRequest::Hidden(rung_bps(source, rungs[0]));
440    }
441    // Past the richest rung a tile's SIZE may select — one boundary for both
442    // ladders, see [`TILE_SOURCE_MIN_PIXELS`] — the tile takes the source. For the
443    // camera this is just the walk below running off the end of the ladder. For a
444    // SCREEN it is a real cut: it stops the walk at the floor rung even though
445    // [`SCREEN_MID_RUNG`] would cover a tile up to 1 382 400 px, so a share's
446    // size-derived demand is only ever `Capped(floor)` or `Source`.
447    //
448    // Why the cut, rather than letting the size walk reach the intermediate rung:
449    // `Source` is immune to the producer's cross-receiver MINIMUM and `Capped` is
450    // not. A stage-sized share tile that asked for `Capped(1.1 Mbps)` would be
451    // folded (`crate::plan_layers`) against every other receiver's share demand —
452    // a grid-cell viewer's 200 kbps wins the min — and `call::subscribe::pick_layer`
453    // hands any capped demand the LOW track unconditionally while the `_l0` decl
454    // exists. The reader staring at 1280×720 of shared text would then be watching
455    // a 360p15 layer upscaled 2×: the exact legibility failure the screen ladder is
456    // built around, caused by someone else's grid cell. Demanding `Source` keeps
457    // that receiver on the source track, and the allocator is free to step it DOWN
458    // to the intermediate rung when the budget cannot afford 2.5 Mbps — a decision
459    // made from this receiver's own link, not from another receiver's layout.
460    if pixels > TILE_SOURCE_MIN_PIXELS {
461        return QualityRequest::Source;
462    }
463    rungs
464        .iter()
465        .find(|r| pixels <= r.covers_up_to_pixels())
466        .map_or(QualityRequest::Source, |r| {
467            QualityRequest::Capped(rung_bps(source, *r))
468        })
469}

471/// Invert a coalesced cap back into the rung the producer should encode: the
472/// richest rung this cap can pay for, never below the cheapest one.
473///
474/// The producer cannot simply look the cap up: `RequestQuality { cap_bps }` carries
475/// a *bitrate*, and the cap that arrives has been folded ([`most_constrained`])
476/// with the receiver's bandwidth-driven demand (`DownlinkDemand`), which produces
477/// arbitrary values — and it is a **cross-receiver minimum** over everyone still
478/// rendering the track (see [`crate::plan_layers`]). So the mapping is "what fits",
479/// not "what was asked for" — and a cap below even the cheapest rung still yields
480/// that rung rather than nothing, because the encoder's own bitrate is then set to
481/// the cap and the geometry is the floor the ladder allows (the same
482/// never-degenerate-toward-zero contract as `DownlinkDemand`'s `MIN_CAP_BPS`).
483pub fn low_rung_for_cap(source: VideoSource, cap_bps: u64) -> VideoRung {
484    let rungs = low_rungs(source);
485    debug_assert_ascending(source, rungs);
486    rungs
487        .iter()
488        .rev()
489        .find(|r| rung_bps(source, **r) <= cap_bps)
490        .copied()
491        .unwrap_or(rungs[0])
492}

494/// How many remote video peers a call must have before a peer that has been
495/// *observed silent* is priced a rung below what its tile alone would buy.
496///
497/// **20**, which is where Jitsi's `conference-last-n-limits { 20 = 20, 30 = 15,
498/// 50 = 10, 90 = 2 }` starts tightening: selectivity is a function of room size,
499/// because a small call can simply afford everyone and the *visible* cost of being
500/// wrong about who matters (a 2× grid cell filled from a 180p thumbnail is a
501/// noticeable 2.25× upscale of a face) is not worth paying there. At 8 peers the
502/// size-derived rungs cost ~3.4 Mbps and are affordable; at 30 they are ~12 Mbps
503/// and are not. Below this line ranking only ever *promotes*.
504pub const SPEAKER_DEMOTE_MIN_PEERS: usize = 20;
505
506/// Modulate a tile's own demand by where its peer sits in the speaking ranking —
507/// the receiver-side Last-N lever, applied as a **modifier on the size-derived
508/// rung** rather than as a replacement for it.
509///
510/// * the **dominant** speaker moves up one rung (a grid cell's `THUMB` becomes
511///   `LOW`), which is Jitsi's `onstage-preferred-height-px = 360` against its
512///   `default-max-height-px = 180`, expressed in this ladder's terms;
513/// * a peer **observed silent** for the whole long window, in a crowded call, moves
514///   down one rung;
515/// * everyone else — including every peer we have no speaking evidence about, e.g.
516///   an un-upgraded client that never reports a level — is left exactly as the tile
517///   priced it.
518///
519/// Three properties make this composable with everything P1/P2 built, and each is
520/// pinned by a test below:
521///
522/// 1. **The layer axis is invariant.** `Hidden` and `Source` pass through untouched
523///    and a `Capped` demand stays `Capped`, so modulation can never turn a
524///    subscription on, off, or into a different track. It moves a *cap*, which the
525///    `LayerHysteresis` same-axis path commits immediately — so RECEIVER-side, P2's
526///    churn bounds are not merely respected, they are not engaged (no resubscribe,
527///    no pipeline restart, no dwell consumed).
528///
529///    That is only the receiver's half, and the producer's is not free: a sustained
530///    handover moves the cross-receiver MINIMUM cap (`plan_layers`), so the promoted
531///    peer's shared low layer changes rung — an encoder rebuild plus an IDR to every
532///    subscriber of it. Bounded, not absent: at most one such rebuild per
533///    [`crate::GEOMETRY_REBUILD_DWELL_US`] per producer, which is exactly why
534///    [`crate::speaking::SWITCH_DWELL_US`] is pinned to that dwell.
535/// 2. **Never above the ladder, never below it.** A promotion stops at the richest
536///    below-source rung (`Source` is the tile-size signal's decision alone — a user
537///    staring at an expanded tile gets sharp video whether or not they are talking),
538///    and a demotion stops at the cheapest.
539///
540///    **A screen share is exempt entirely.** Speaking is evidence about a *face*,
541///    not about shared content: whether the sharer is the one talking says nothing
542///    about how legible their slides need to be, and a share's rungs are 900 kbps
543///    and 1.4 Mbps apart — moving one on a speaking transition would hand the
544///    producer an encoder rebuild plus an IDR to every subscriber for a signal that
545///    was never about that track. This used to be enforced by cardinality alone (a
546///    one-rung screen ladder has nothing to move); [`SCREEN_MID_RUNG`] ended that,
547///    so the carve-out is now explicit and pinned by a test below.
548/// 3. **Silence must be evidence.** `SpeakerTier::Unknown` is never demoted; only
549///    `Silent`, which the ranker only reports after watching a peer for its whole
550///    long window. Same two-tier trust rule as `TileDemand`'s reported-vs-seeded hide.
551///
552/// Apply this to the TILE demand, before folding with the receiver's own bandwidth
553/// demand (`most_constrained`): a promotion must never lift a receiver above what
554/// its link estimate allows.
555pub fn speaker_adjusted(
556    source: VideoSource,
557    req: QualityRequest,
558    tier: crate::speaking::SpeakerTier,
559    peers: usize,
560) -> QualityRequest {
561    use crate::speaking::SpeakerTier;
562    // Shared content is never re-ranked by who is speaking — see the carve-out in
563    // this function's contract. Must come FIRST: it is a property of the SOURCE, not
564    // of the demand shape or of how many rungs the screen ladder happens to have.
565    if matches!(source, VideoSource::Screen) {
566        return req;
567    }
568    let QualityRequest::Capped(bps) = req else {
569        return req;
570    };
571    let rungs = low_rungs(source);
572    debug_assert_ascending(source, rungs);
573    // Where this cap sits on the ladder — the same "richest rung it can afford"
574    // inversion the producer runs ([`low_rung_for_cap`]), as an index.
575    let at = rungs
576        .iter()
577        .rposition(|r| rung_bps(source, *r) <= bps)
578        .unwrap_or(0);
579    let want = match tier {
580        SpeakerTier::Dominant => (at + 1).min(rungs.len() - 1),
581        SpeakerTier::Silent if peers >= SPEAKER_DEMOTE_MIN_PEERS => at.saturating_sub(1),
582        _ => at,
583    };
584    if want == at {
585        req
586    } else {
587        QualityRequest::Capped(rung_bps(source, rungs[want]))
588    }
589}

591/// Fold two demands into the more-constrained one. `Hidden` beats everything (a
592/// tile nobody is looking at cannot become one that is by way of a bandwidth
593/// estimate), `Capped` beats `Source`, and two of a kind take the lower cap. Used
594/// to combine a tile-size demand with a bandwidth demand so the receiver can never
595/// over-request beyond what either signal allows.
596pub fn most_constrained(a: QualityRequest, b: QualityRequest) -> QualityRequest {
597    use QualityRequest::{Capped, Hidden, Source};
598    match (a, b) {
599        // Hidden wins, at the lower of the two caps it could be carrying.
600        (Hidden(x), Hidden(y)) => Hidden(x.min(y)),
601        (Hidden(x), Capped(y)) | (Capped(y), Hidden(x)) => Hidden(x.min(y)),
602        (Hidden(x), Source) | (Source, Hidden(x)) => Hidden(x),
603        (Source, other) | (other, Source) => other,
604        (Capped(x), Capped(y)) => Capped(x.min(y)),
605    }
606}

1043    /// Round trip: what a tile asks for is what the producer builds. This is the
1044    /// property that makes the ladder work end to end — the demand crosses the
1045    /// wire as a bare bitrate, so a rung that could not be recovered from its own
1046    /// price would silently encode at the wrong geometry.
1047    ///
1048    /// Every case here is a tile shape the ladder genuinely produces (both display
1049    /// densities of the two shipped surfaces), so a rung is never asserted for a
1050    /// size that could not ask for it.
1051    #[test]
1052    fn every_tile_demand_round_trips_to_its_rung() {
1053        use VideoSource::Camera;
1054        for (tile, want) in [
1055            ((176u32, 96u32), THUMB_RUNG), // filmstrip stamp @1x
1056            ((336, 192), THUMB_RUNG),      // filmstrip stamp @2x
1057            ((320, 180), THUMB_RUNG),      // the rung's own geometry
1058            ((368, 208), THUMB_RUNG),      // grid cell @1x
1059            ((480, 272), LOW_RUNG),        // a mid-sized tile
1060            ((640, 360), LOW_RUNG),        // the rung's own geometry
1061            ((720, 400), LOW_RUNG),        // grid cell @2x
1062        ] {
1063            let asked = cap(tile_quality_request(Camera, tile.0, tile.1, true));
1064            assert_eq!(
1065                low_rung_for_cap(Camera, asked),
1066                want,
1067                "a {}×{} tile asked {asked} bps and got the wrong rung back",
1068                tile.0,
1069                tile.1
1070            );
1071        }
1072        // A hidden tile round-trips too — its carried cap is a real rung price.
1073        let QualityRequest::Hidden(bps) = tile_quality_request(Camera, 368, 208, false) else {
1074            panic!("a hidden tile must report Hidden");
1075        };
1076        assert_eq!(low_rung_for_cap(Camera, bps), THUMB_RUNG);
1077
1078        // The screen half round-trips too, and its full range is TWO answers: a
1079        // share tile's size only ever produces the legibility floor or `Source`.
1080        // [`SCREEN_MID_RUNG`] is deliberately unreachable from here — it is an
1081        // allocator step (see `tile_quality_request`), and the round trip that
1082        // matters for it is the cap inversion, pinned in
1083        // `a_cap_buys_the_richest_rung_it_can_afford`.
1084        use VideoSource::Screen;
1085        for (tile, want) in [
1086            ((368u32, 208u32), Some(SCREEN_LOW_RUNG)), // grid cell @1x
1087            ((640, 360), Some(SCREEN_LOW_RUNG)),       // the rung's own geometry
1088            ((720, 400), Some(SCREEN_LOW_RUNG)),       // grid cell @2x, legible floor
1089            ((960, 540), None),                        // a half-window share
1090            ((1280, 720), None),                       // a stage share
1091        ] {
1092            let req = tile_quality_request(Screen, tile.0, tile.1, true);
1093            match want {
1094                Some(rung) => assert_eq!(
1095                    low_rung_for_cap(Screen, cap(req)),
1096                    rung,
1097                    "a {}×{} screen tile asked {req:?} and got the wrong rung back",
1098                    tile.0,
1099                    tile.1
1100                ),
1101                None => assert_eq!(
1102                    req,
1103                    QualityRequest::Source,
1104                    "a {}×{} screen tile is past the size walk's floor-or-source cut",
1105                    tile.0,
1106                    tile.1
1107                ),
1108            }
1109        }
1110        let QualityRequest::Hidden(bps) = tile_quality_request(Screen, 368, 208, false) else {
1111            panic!("a hidden screen tile must report Hidden");
1112        };
1113        assert_eq!(low_rung_for_cap(Screen, bps), SCREEN_LOW_RUNG);
1114    }
