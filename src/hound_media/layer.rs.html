1//! The video **layer model** — the single source of truth mapping a layer combo to
2//! a moq track name + wire [`TrackKind`], and the pure helpers that turn a receiver's
3//! tile size / bandwidth into a [`QualityRequest`].
4//!
5//! Two scalability axes, deliberately mapped to two different transport mechanisms:
6//!
7//! * **Spatial** ([`SpatialLayer`]) = a SEPARATE moq track. The relay is forward-no-
8//!   transcode + group-atomic, so per-receiver selection is per-*track* subscription;
9//!   the source layer keeps its original track name (an un-upgraded peer never learns
10//!   about layers and keeps working), and the low layer is an additional track a
11//!   constrained / small-tile receiver subscribes to instead.
12//! * **Temporal** ([`TemporalLayer`]) = a per-FRAME marking WITHIN one spatial track
13//!   (a base frame is self-decodable; an enhancement frame depends on base and is
14//!   droppable). Not part of the track identity — see the producer-side gating.
15//!
16//! Keeping this here (not in `call.rs`) keeps the wire/string juggling in one tested
17//! place and out of the IO glue.
18
19use crate::TrackKind;
20use crate::congestion::{QualityRequest, camera_bitrate_bps, screen_bitrate_bps};
21
22/// Spatial scalability layer — each maps to its own moq track.
23#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
24pub enum SpatialLayer {
25    /// The full-quality stream (original track name; always published).
26    Source,
27    /// A reduced-resolution stream a constrained receiver takes instead.
28    Low,
29}
30
31/// Temporal scalability layer — a per-frame marking, not a track.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
33pub enum TemporalLayer {
34    /// Self-decodable base frame (depended on by enhancement frames).
35    Base = 0,
36    /// Droppable enhancement frame (depends only on base; never depended upon).
37    Enhanced = 1,
38}
39
40impl TemporalLayer {
41    /// Decode a wire temporal id: `0` = base, anything else = enhancement.
42    pub fn from_id(id: u8) -> Self {
43        if id == 0 { Self::Base } else { Self::Enhanced }
44    }
45
46    /// The wire temporal id.
47    pub fn id(self) -> u8 {
48        self as u8
49    }
50}

52/// Which media source a video track belongs to. Mirrors the control-plane `Source`
53/// (camera vs screen) but stays in `hound-media` so this crate needs no control dep.
54#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
55pub enum VideoSource {
56    Camera,
57    Screen,
58}
59
60/// The moq `track_name` for a (source, spatial) pair — the subscribe/publish key
61/// under the per-peer broadcast path. The source layers keep the original names so
62/// a peer that never learns about layers is unaffected.
63pub fn track_name(source: VideoSource, spatial: SpatialLayer) -> &'static str {
64    match (source, spatial) {
65        (VideoSource::Camera, SpatialLayer::Source) => "cam",
66        (VideoSource::Camera, SpatialLayer::Low) => "cam_l0",
67        (VideoSource::Screen, SpatialLayer::Source) => "screen",
68        (VideoSource::Screen, SpatialLayer::Low) => "screen_l0",
69    }
70}

72/// The wire [`TrackKind`] for a (source, spatial) pair.
73pub fn track_kind(source: VideoSource, spatial: SpatialLayer) -> TrackKind {
74    match (source, spatial) {
75        (VideoSource::Camera, SpatialLayer::Source) => TrackKind::CamVideo,
76        (VideoSource::Camera, SpatialLayer::Low) => TrackKind::CamVideoLow,
77        (VideoSource::Screen, SpatialLayer::Source) => TrackKind::ScrVideo,
78        (VideoSource::Screen, SpatialLayer::Low) => TrackKind::ScrVideoLow,
79    }
80}

82/// One below-source rung: a geometry (16:9 reference height) and a frame rate. Its
83/// *price* is not stored — it comes from the content-aware bitrate model
84/// ([`rung_bps`]), so a rung and its cost can never drift apart.
85///
86/// A rung is a nominal shape, not an encode geometry: the producer fits it to its
87/// own capture aspect, so a non-16:9 source is never stretched — by HEIGHT
88/// (`fit_encode_dims`) where the aspect is fixed and the height is the ladder's
89/// unit (the camera), and by AREA (`fit_encode_area`) where it is not and the
90/// rung's price is what must be held (a screen share). See [`Self::pixels`].
91#[derive(Debug, Clone, Copy, PartialEq, Eq)]
92pub struct VideoRung {
93    /// 16:9 reference height (the encoder fits this to the source's own aspect).
94    pub height: u32,
95    pub fps: u32,
96}
97
98impl VideoRung {
99    /// The rung's own pixel count at its 16:9 reference geometry.
100    pub const fn pixels(self) -> u64 {
101        (self.height as u64) * 16 / 9 * (self.height as u64)
102    }
103
104    /// The largest tile this rung may be asked to fill, in **physical** pixels:
105    /// its own area times [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`].
106    pub const fn covers_up_to_pixels(self) -> u64 {
107        self.pixels() * TILE_UPSCALE_TOLERANCE_NUM / TILE_UPSCALE_TOLERANCE_DEN
108    }
109}
110
111/// How far a rung may be stretched to fill a tile before the next rung up is worth
112/// its extra bits — an **area** ratio of 3/2, i.e. ~22% of linear upscale in each
113/// dimension. Expressed as an integer fraction so the boundary is exact.
114pub const TILE_UPSCALE_TOLERANCE_NUM: u64 = 3;
115pub const TILE_UPSCALE_TOLERANCE_DEN: u64 = 2;
116
117/// The **thumbnail** rung: a grid cell, a filmstrip stamp, an off-screen tile.
118/// 320×180 @ 15 fps ≈ 130 kbps — LiveKit ships `h180` at this geometry, Jitsi's
119/// bridge defaults its thumbnails to 180p, and Daily publishes 320×180 @ 10 fps.
120/// Against the 1 Mbps flat low rung this replaces it is **4× fewer pixels and half
121/// the frames**, which is where the large-call receive-CPU win comes from.
122pub const THUMB_RUNG: VideoRung = VideoRung {
123    height: 180,
124    fps: 15,
125};
126
127/// The **low** rung: a mid-sized tile that is still below source. 640×360 @ 20 fps
128/// ≈ 420 kbps — the old `cam_l0` geometry, re-priced (it was 1 Mbps at 30 fps).
129pub const LOW_RUNG: VideoRung = VideoRung {
130    height: 360,
131    fps: 20,
132};
133
134/// The **floor** below-source screen rung. Screen share degrades by
135/// `DegradePriority::Legibility` — drop frames before pixels — so a small share
136/// tile keeps 360p (a 180p thumbnail of shared *text* is unreadable, where a 180p
137/// thumbnail of a face is fine) and gives up frame rate instead: 15 fps is the
138/// same floor the screen ladder's legibility ordering uses. ≈200 kbps against the
139/// 1 Mbps it used to cost, at identical resolution.
140pub const SCREEN_LOW_RUNG: VideoRung = VideoRung {
141    height: 360,
142    fps: 15,
143};
144
145/// The **intermediate** screen rung: 720p @ 30 fps ≈ 1.1 Mbps, sitting between
146/// [`SCREEN_LOW_RUNG`] (200 kbps) and the 2.5 Mbps source.
147///
148/// It exists to shrink the screen ladder's one enormous step. With 360p15 as the
149/// only below-source rung a share's whole ladder was `200 kbps → 2.5 Mbps`, and
150/// [`crate::allocate_video`]'s cheapest-first upgrade phase therefore left every
151/// budget in a 2.3 Mbps-wide band buying the share nothing at all: it went from
152/// "barely legible" straight to "more than the link has". Two steps of 900 kbps
153/// and 1.4 Mbps instead, which is what makes a 5 Mbps receiver watching a share
154/// plus two cameras land on a sharp 720p30 share rather than a 360p15 one.
155///
156/// 720p30 is also the legibility ladder's own next stop — `screen_rungs`' floor
157/// resolution is 720p ("below that text is unreadable") — so the intermediate rung
158/// is legible content at a real frame rate, not a compromise between the two.
159///
160/// **It is a BUDGET rung, not a SIZE rung.** A tile's own size never asks for it:
161/// [`tile_quality_request`] stops a screen tile's size-derived walk at
162/// [`SCREEN_LOW_RUNG`]'s coverage, so a share's size demand is still exactly
163/// floor-or-`Source`. This rung is reachable only through [`crate::allocate_video`]
164/// — as an upgrade step a `Source`-demanding share takes when the budget affords
165/// 1.1 Mbps but not 2.5, and as the step a deficit walks back down through. The
166/// reason is the coalesced minimum, and it is load-bearing: a `Source` demand is
167/// immune to it (`plan_layers` routes it to the source track), a `Capped` demand is
168/// not, so a stage share that asked for `Capped(1.1 Mbps)` alongside anyone
169/// rendering a grid-cell share would be *folded down* to the 360p15 floor and
170/// rendered 2× upscaled — which is the legibility failure `SCREEN_LOW_RUNG` exists
171/// to prevent, inflicted on the one receiver actually reading the text. See
172/// [`tile_quality_request`]'s walk.
173pub const SCREEN_MID_RUNG: VideoRung = VideoRung {
174    height: 720,
175    fps: 30,
176};
177
178/// A visible tile larger than this — in **physical** pixels — takes the source
179/// rather than any below-source rung: it is past what the richest rung a tile's
180/// SIZE may select ([`LOW_RUNG`], and [`SCREEN_LOW_RUNG`] at the same geometry) can
181/// fill within [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`].
182///
183/// One boundary for both ladders, and it stayed one when [`SCREEN_MID_RUNG`]
184/// landed: that rung is reachable from the allocator, never from a tile's size, so
185/// the screen ladder's size boundary is still its floor rung's coverage — the same
186/// 345 600 px, since the two floors share the 360p geometry. See
187/// [`tile_quality_request`] for why the size walk must stop there.
188///
189/// Worked examples, in the PHYSICAL pixels the UI actually reports (its own 16 px
190/// bucketing already applied, so these are the exact numbers that arrive), for the
191/// two shapes the shipped UI produces — `tile_grid.dart`'s
192/// `maxCrossAxisExtent: 360` logical grid cell and its 96 px-tall filmstrip:
193///
194/// | surface | @ DPR 1 | @ DPR 2 |
195/// |---|---|---|
196/// | grid cell | 368×208 = 76 544 px → **thumb** | 720×400 = 288 000 px → **low** |
197/// | filmstrip stamp | 176×96 = 16 896 px → **thumb** | 336×192 = 64 512 px → **thumb** |
198/// | expanded / stage | 1280×720 = 921 600 px → **source** | **source** |
199///
200/// The 2× grid cell is the case this boundary exists for: it is a genuine 720-wide
201/// surface, and filling it from a 320-wide thumbnail would be a 2.25× linear
202/// upscale of a face.
203pub const TILE_SOURCE_MIN_PIXELS: u64 = LOW_RUNG.covers_up_to_pixels();
204
205/// What one below-source rung costs for this content, from the bits-per-pixel
206/// model in [`crate::congestion`]. Camera and screen are priced by different
207/// models: screen content (text, flat regions, sharp edges) compresses several
208/// times better than motion video, so pricing a shared window like a talking head
209/// would oversize it.
210pub fn rung_bps(source: VideoSource, rung: VideoRung) -> u64 {
211    match source {
212        VideoSource::Camera => camera_bitrate_bps(rung.height, rung.fps),
213        VideoSource::Screen => screen_bitrate_bps(rung.height, rung.fps),
214    }
215}

217/// What the **source** rung costs — the top-rung bitrate the client publishes this
218/// content at (`CAM_BITRATE` / `SCREEN_BITRATE` in `crates/app/client/src/call/mod.rs`,
219/// and the `source_bps` the producer's [`crate::plan_layers`] coalesces against).
220///
221/// Stated as constants rather than run back through [`rung_bps`]: the camera figure
222/// is the anchor the whole bits-per-pixel model was *solved from* (a test below pins
223/// the identity), and the screen figure is a published ceiling the ladder climbs
224/// toward rather than a modelled point. [`crate::allocate`] needs both so a `Source`
225/// demand can be priced on the same scale as a below-source rung.
226pub const CAM_SOURCE_BPS: u64 = 1_500_000;
227pub const SCREEN_SOURCE_BPS: u64 = 2_500_000;
228
229/// The price of taking the SOURCE track for this content — see [`CAM_SOURCE_BPS`].
230pub fn source_bps(source: VideoSource) -> u64 {
231    match source {
232        VideoSource::Camera => CAM_SOURCE_BPS,
233        VideoSource::Screen => SCREEN_SOURCE_BPS,
234    }
235}

237/// The below-source rungs available for `source`, **cheapest first**. Every caller
238/// walks this in order; the ascending-price invariant is asserted in debug builds
239/// by [`debug_assert_ascending`].
240pub(crate) fn low_rungs(source: VideoSource) -> &'static [VideoRung] {
241    match source {
242        VideoSource::Camera => &[THUMB_RUNG, LOW_RUNG],
243        // Legibility: no thumbnail rung for shared content — the floor is 360p and
244        // gives up frame rate (see [`SCREEN_LOW_RUNG`]) — and one intermediate rung
245        // above it so the climb to the source is not a single 2.3 Mbps leap
246        // ([`SCREEN_MID_RUNG`], an allocator step only: `tile_quality_request`
247        // never asks for it from a tile's size).
248        VideoSource::Screen => &[SCREEN_LOW_RUNG, SCREEN_MID_RUNG],
249    }
250}
251
252/// The ladder must be strictly ascending in price AND in area, or "cheapest rung
253/// that covers the tile" and "richest rung this cap affords" stop being inverses of
254/// each other and a tile could round-trip to a rung it never asked for.
255fn debug_assert_ascending(source: VideoSource, rungs: &[VideoRung]) {
256    debug_assert!(
257        rungs
258            .windows(2)
259            .all(|w| rung_bps(source, w[0]) < rung_bps(source, w[1])
260                && w[0].pixels() < w[1].pixels()),
261        "{source:?} rung ladder is not strictly ascending: {rungs:?}"
262    );
263}

265/// Map a rendered tile's **physical** pixel size + visibility to a
266/// [`QualityRequest`]. This is the *demand* half of demand-driven simulcast — it
267/// never asks for more than the tile can use, which is exactly the receiver-CPU win
268/// (a thumbnail pulls the low track).
269///
270/// `width`/`height` are physical (device) pixels, not logical ones: decode and
271/// render cost scale with the pixels the GPU actually paints, so the same 360-wide
272/// grid cell is a thumbnail on a 1× display and a genuine 720-wide surface on a 2×
273/// one. The UI multiplies by `MediaQuery.devicePixelRatioOf` before reporting.
274///
275/// The cap is **derived from the tile's own pixel count**, not a constant: the tile
276/// takes the cheapest rung that covers it (within
277/// [`TILE_UPSCALE_TOLERANCE_NUM`]/[`TILE_UPSCALE_TOLERANCE_DEN`]), priced through
278/// [`rung_bps`]; past the richest rung it takes the source. A 1 Mbps flat cap —
279/// what this used to return for *every* sub-360p tile, including a 171×96 filmstrip
280/// stamp — is 6–12× the per-tile budget Zoom, Daily and LiveKit all converged on,
281/// and it was the dominant cost of a large call.
282pub fn tile_quality_request(
283    source: VideoSource,
284    width: u32,
285    height: u32,
286    visible: bool,
287) -> QualityRequest {
288    let rungs = low_rungs(source);
289    debug_assert_ascending(source, rungs);
290    let pixels = (width as u64) * (height as u64);
291    // Not rendered: the receiver subscribes to NOTHING for this tile
292    // (`call::subscribe::pick_layer`), and reports the cheapest rung flagged so the
293    // producer's fold ignores it while anyone is actually watching — yet still
294    // sizes the layer when every requester is hidden (see [`crate::plan_layers`]).
295    if !visible {
296        return QualityRequest::Hidden(rung_bps(source, rungs[0]));
297    }
298    // Past the richest rung a tile's SIZE may select — one boundary for both
299    // ladders, see [`TILE_SOURCE_MIN_PIXELS`] — the tile takes the source. For the
300    // camera this is just the walk below running off the end of the ladder. For a
301    // SCREEN it is a real cut: it stops the walk at the floor rung even though
302    // [`SCREEN_MID_RUNG`] would cover a tile up to 1 382 400 px, so a share's
303    // size-derived demand is only ever `Capped(floor)` or `Source`.
304    //
305    // Why the cut, rather than letting the size walk reach the intermediate rung:
306    // `Source` is immune to the producer's cross-receiver MINIMUM and `Capped` is
307    // not. A stage-sized share tile that asked for `Capped(1.1 Mbps)` would be
308    // folded (`crate::plan_layers`) against every other receiver's share demand —
309    // a grid-cell viewer's 200 kbps wins the min — and `call::subscribe::pick_layer`
310    // hands any capped demand the LOW track unconditionally while the `_l0` decl
311    // exists. The reader staring at 1280×720 of shared text would then be watching
312    // a 360p15 layer upscaled 2×: the exact legibility failure the screen ladder is
313    // built around, caused by someone else's grid cell. Demanding `Source` keeps
314    // that receiver on the source track, and the allocator is free to step it DOWN
315    // to the intermediate rung when the budget cannot afford 2.5 Mbps — a decision
316    // made from this receiver's own link, not from another receiver's layout.
317    if pixels > TILE_SOURCE_MIN_PIXELS {
318        return QualityRequest::Source;
319    }
320    rungs
321        .iter()
322        .find(|r| pixels <= r.covers_up_to_pixels())
323        .map_or(QualityRequest::Source, |r| {
324            QualityRequest::Capped(rung_bps(source, *r))
325        })
326}

328/// Invert a coalesced cap back into the rung the producer should encode: the
329/// richest rung this cap can pay for, never below the cheapest one.
330///
331/// The producer cannot simply look the cap up: `RequestQuality { cap_bps }` carries
332/// a *bitrate*, and the cap that arrives has been folded ([`most_constrained`])
333/// with the receiver's bandwidth-driven demand (`DownlinkDemand`), which produces
334/// arbitrary values — and it is a **cross-receiver minimum** over everyone still
335/// rendering the track (see [`crate::plan_layers`]). So the mapping is "what fits",
336/// not "what was asked for" — and a cap below even the cheapest rung still yields
337/// that rung rather than nothing, because the encoder's own bitrate is then set to
338/// the cap and the geometry is the floor the ladder allows (the same
339/// never-degenerate-toward-zero contract as `DownlinkDemand`'s `MIN_CAP_BPS`).
340pub fn low_rung_for_cap(source: VideoSource, cap_bps: u64) -> VideoRung {
341    let rungs = low_rungs(source);
342    debug_assert_ascending(source, rungs);
343    rungs
344        .iter()
345        .rev()
346        .find(|r| rung_bps(source, **r) <= cap_bps)
347        .copied()
348        .unwrap_or(rungs[0])
349}

351/// How many remote video peers a call must have before a peer that has been
352/// *observed silent* is priced a rung below what its tile alone would buy.
353///
354/// **20**, which is where Jitsi's `conference-last-n-limits { 20 = 20, 30 = 15,
355/// 50 = 10, 90 = 2 }` starts tightening: selectivity is a function of room size,
356/// because a small call can simply afford everyone and the *visible* cost of being
357/// wrong about who matters (a 2× grid cell filled from a 180p thumbnail is a
358/// noticeable 2.25× upscale of a face) is not worth paying there. At 8 peers the
359/// size-derived rungs cost ~3.4 Mbps and are affordable; at 30 they are ~12 Mbps
360/// and are not. Below this line ranking only ever *promotes*.
361pub const SPEAKER_DEMOTE_MIN_PEERS: usize = 20;
362
363/// Modulate a tile's own demand by where its peer sits in the speaking ranking —
364/// the receiver-side Last-N lever, applied as a **modifier on the size-derived
365/// rung** rather than as a replacement for it.
366///
367/// * the **dominant** speaker moves up one rung (a grid cell's `THUMB` becomes
368///   `LOW`), which is Jitsi's `onstage-preferred-height-px = 360` against its
369///   `default-max-height-px = 180`, expressed in this ladder's terms;
370/// * a peer **observed silent** for the whole long window, in a crowded call, moves
371///   down one rung;
372/// * everyone else — including every peer we have no speaking evidence about, e.g.
373///   an un-upgraded client that never reports a level — is left exactly as the tile
374///   priced it.
375///
376/// Three properties make this composable with everything P1/P2 built, and each is
377/// pinned by a test below:
378///
379/// 1. **The layer axis is invariant.** `Hidden` and `Source` pass through untouched
380///    and a `Capped` demand stays `Capped`, so modulation can never turn a
381///    subscription on, off, or into a different track. It moves a *cap*, which the
382///    `LayerHysteresis` same-axis path commits immediately — so RECEIVER-side, P2's
383///    churn bounds are not merely respected, they are not engaged (no resubscribe,
384///    no pipeline restart, no dwell consumed).
385///
386///    That is only the receiver's half, and the producer's is not free: a sustained
387///    handover moves the cross-receiver MINIMUM cap (`plan_layers`), so the promoted
388///    peer's shared low layer changes rung — an encoder rebuild plus an IDR to every
389///    subscriber of it. Bounded, not absent: at most one such rebuild per
390///    [`crate::GEOMETRY_REBUILD_DWELL_US`] per producer, which is exactly why
391///    [`crate::speaking::SWITCH_DWELL_US`] is pinned to that dwell.
392/// 2. **Never above the ladder, never below it.** A promotion stops at the richest
393///    below-source rung (`Source` is the tile-size signal's decision alone — a user
394///    staring at an expanded tile gets sharp video whether or not they are talking),
395///    and a demotion stops at the cheapest.
396///
397///    **A screen share is exempt entirely.** Speaking is evidence about a *face*,
398///    not about shared content: whether the sharer is the one talking says nothing
399///    about how legible their slides need to be, and a share's rungs are 900 kbps
400///    and 1.4 Mbps apart — moving one on a speaking transition would hand the
401///    producer an encoder rebuild plus an IDR to every subscriber for a signal that
402///    was never about that track. This used to be enforced by cardinality alone (a
403///    one-rung screen ladder has nothing to move); [`SCREEN_MID_RUNG`] ended that,
404///    so the carve-out is now explicit and pinned by a test below.
405/// 3. **Silence must be evidence.** `SpeakerTier::Unknown` is never demoted; only
406///    `Silent`, which the ranker only reports after watching a peer for its whole
407///    long window. Same two-tier trust rule as `TileDemand`'s reported-vs-seeded hide.
408///
409/// Apply this to the TILE demand, before folding with the receiver's own bandwidth
410/// demand (`most_constrained`): a promotion must never lift a receiver above what
411/// its link estimate allows.
412pub fn speaker_adjusted(
413    source: VideoSource,
414    req: QualityRequest,
415    tier: crate::speaking::SpeakerTier,
416    peers: usize,
417) -> QualityRequest {
418    use crate::speaking::SpeakerTier;
419    // Shared content is never re-ranked by who is speaking — see the carve-out in
420    // this function's contract. Must come FIRST: it is a property of the SOURCE, not
421    // of the demand shape or of how many rungs the screen ladder happens to have.
422    if matches!(source, VideoSource::Screen) {
423        return req;
424    }
425    let QualityRequest::Capped(bps) = req else {
426        return req;
427    };
428    let rungs = low_rungs(source);
429    debug_assert_ascending(source, rungs);
430    // Where this cap sits on the ladder — the same "richest rung it can afford"
431    // inversion the producer runs ([`low_rung_for_cap`]), as an index.
432    let at = rungs
433        .iter()
434        .rposition(|r| rung_bps(source, *r) <= bps)
435        .unwrap_or(0);
436    let want = match tier {
437        SpeakerTier::Dominant => (at + 1).min(rungs.len() - 1),
438        SpeakerTier::Silent if peers >= SPEAKER_DEMOTE_MIN_PEERS => at.saturating_sub(1),
439        _ => at,
440    };
441    if want == at {
442        req
443    } else {
444        QualityRequest::Capped(rung_bps(source, rungs[want]))
445    }
446}

448/// Fold two demands into the more-constrained one. `Hidden` beats everything (a
449/// tile nobody is looking at cannot become one that is by way of a bandwidth
450/// estimate), `Capped` beats `Source`, and two of a kind take the lower cap. Used
451/// to combine a tile-size demand with a bandwidth demand so the receiver can never
452/// over-request beyond what either signal allows.
453pub fn most_constrained(a: QualityRequest, b: QualityRequest) -> QualityRequest {
454    use QualityRequest::{Capped, Hidden, Source};
455    match (a, b) {
456        // Hidden wins, at the lower of the two caps it could be carrying.
457        (Hidden(x), Hidden(y)) => Hidden(x.min(y)),
458        (Hidden(x), Capped(y)) | (Capped(y), Hidden(x)) => Hidden(x.min(y)),
459        (Hidden(x), Source) | (Source, Hidden(x)) => Hidden(x),
460        (Source, other) | (other, Source) => other,
461        (Capped(x), Capped(y)) => Capped(x.min(y)),
462    }
463}

1027    #[test]
1028    fn most_constrained_takes_the_floor() {
1029        use QualityRequest::{Capped, Hidden, Source};
1030        assert_eq!(most_constrained(Source, Source), Source);
1031        // Source is the least constrained — the other wins.
1032        assert_eq!(most_constrained(Source, Capped(500)), Capped(500));
1033        assert_eq!(most_constrained(Capped(500), Source), Capped(500));
1034        // Two caps → the lower (more constrained).
1035        assert_eq!(most_constrained(Capped(800), Capped(500)), Capped(500));
1036        // Hidden outranks everything: a link estimate cannot make an unmounted
1037        // tile visible again.
1038        assert_eq!(most_constrained(Hidden(500), Source), Hidden(500));
1039        assert_eq!(most_constrained(Source, Hidden(500)), Hidden(500));
1040        assert_eq!(most_constrained(Hidden(500), Capped(800)), Hidden(500));
1041        assert_eq!(most_constrained(Capped(300), Hidden(500)), Hidden(300));
1042        assert_eq!(most_constrained(Hidden(800), Hidden(500)), Hidden(500));
