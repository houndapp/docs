1//! # Receiver-side downlink demand
2//!
3//! Complements [`crate::layer::tile_quality_request`] (rendered-tile-size demand)
4//! with the BANDWIDTH half of demand-driven simulcast: when *this* receiver's
5//! downlink can't keep up with a video track, ask the producer for the low layer.
6//! The two demands are folded with [`crate::layer::most_constrained`] by the
7//! caller — this module only decides the bandwidth half.
8//!
9//! ## Signal: frame-sequence gaps, not the sender's own bitrate
10//!
11//! Each wire frame ([`crate::fragment::Frame`]) carries a per-track `frame_seq`.
12//! A gap in the sequence a receiver observes means frames were lost **between the
13//! sender and this receiver** — unlike a raw delivered-bytes rate, it is not
14//! confounded by the sender's own encoder adapting its bitrate (which changes how
15//! BIG frames are, not how many of them arrive). [`FrameSeqWindow`] folds arrivals
16//! into a windowed `(received, expected)` count; [`DownlinkDemand`] turns that
17//! ratio into a [`QualityRequest`](crate::congestion::QualityRequest).
18//!
19//! ## Thresholds: retuned after a cratering regression
20//!
21//! An earlier receive-side decider (`DownlinkEstimator`/`QualityRequestDecider`,
22//! removed as dead code — see commit history) capped a receiver at 0.85× the
23//! source bitrate. That turned a MINOR dip into a resolution cliff: a healthy 1:1
24//! 720p call would crater to 360p on a transient wobble, because 0.85 is barely
25//! below "normal" jitter in the estimate. This decider caps only a **genuinely
26//! poor** link:
27//!
28//! * drop to [`QualityRequest::Capped`] once delivery falls below **1/4** of source
29//! * restore [`QualityRequest::Source`] only once delivery climbs back above **1/2**
30//!   of source
31//!
32//! The wide `[1/4, 1/2]` band is deliberate anti-flap hysteresis — a link
33//! wobbling anywhere in that band (including the specific case of ~76% delivery,
34//! which the OLD 0.85 threshold would have capped) holds its current demand. Note
35//! also: with a single receiver, the *sender's own* loss-based AIMD
36//! ([`crate::congestion::RateController`], driven by `MediaFeedback` reports)
37//! already right-sizes the one stream it's publishing — this decider only earns
38//! its keep once a call has enough heterogeneous receivers that no single
39//! sender-side rate can serve everyone (see the module docs on the call site for
40//! how demand is gated).
41//!
42//! Sans-IO: time enters only as an explicit `now_us` argument (mirrors
43//! [`crate::congestion::RateController`] / [`crate::send_budget::SendBudget`]);
44//! no clock reads, no I/O.
45
46use crate::congestion::QualityRequest;
47
48/// Is `a` newer than `b` in a wrapping 32-bit sequence space (RFC 1982 serial
49/// comparison). A local copy of `crate::recovery`'s helper — this module has no
50/// dependency on `recovery` and the comparison is one line.
51fn seq_newer(a: u32, b: u32) -> bool {
52    a != b && a.wrapping_sub(b) < u32::MAX / 2
53}
54
55/// Bound on a single observed `frame_seq` gap. A restart (a fresh producer's
56/// `frame_seq` starts back at 0) or a corrupt/wild value must not be misread as
57/// millions of lost frames — far above any gap a live call could produce (a
58/// stalled pipeline is reaped and re-subscribed long before `frame_seq` could
59/// legitimately jump this far).
60const MAX_FRAME_GAP: u32 = 10_000;
61
62/// Per-track receive-completeness tracker: folds each arriving frame's wire
63/// `frame_seq` into a windowed `(received, expected)` count. One instance per
64/// subscribed video track — a fresh instance per subscription, so a resubscribe
65/// (a new producer's independent `frame_seq` space) starts clean.
66#[derive(Debug, Default, Clone, Copy)]
67pub struct FrameSeqWindow {
68    last_seq: Option<u32>,
69    received: u32,
70    expected: u32,
71    /// Same counts, never drained by [`Self::take`] — the running totals a
72    /// cumulative receiver-loss gauge is derived from (see [`Self::lost_estimate`]).
73    received_total: u64,
74    expected_total: u64,
75}
76
77impl FrameSeqWindow {
78    pub fn new() -> Self {
79        Self::default()
80    }
81
82    /// Record one arrived frame's wire `frame_seq`. A frame OLDER than the last
83    /// seen (reorder or duplicate) still counts toward `received` but does not
84    /// move the continuity baseline — it was already folded into an earlier gap,
85    /// so re-counting it as "expected" again would double-count.
86    pub fn observe(&mut self, seq: u32) {
87        self.received = self.received.saturating_add(1);
88        self.received_total = self.received_total.saturating_add(1);
89        let advance = match self.last_seq {
90            None => {
91                self.last_seq = Some(seq);
92                1
93            }
94            Some(last) if seq_newer(seq, last) => {
95                self.last_seq = Some(seq);
96                seq.wrapping_sub(last).min(MAX_FRAME_GAP)
97            }
98            Some(_) => 0, // reorder/duplicate: already accounted for
99        };
100        self.expected = self.expected.saturating_add(advance);
101        self.expected_total = self.expected_total.saturating_add(advance as u64);
102    }
103
104    /// Drain the window's `(received, expected)` counts and reset them. The
105    /// continuity baseline (`last_seq`) is NOT reset — a gap spanning a window
106    /// boundary is still detected correctly on the next `observe`.
107    pub fn take(&mut self) -> (u32, u32) {
108        let out = (self.received, self.expected);
109        self.received = 0;
110        self.expected = 0;
111        out
112    }
113
114    /// The CURRENT (undrained) window's shortfall: how many of the frames the
115    /// sequence space said to expect have not arrived yet. Non-destructive, so it
116    /// never disturbs the decider's window boundary.
117    pub fn window_deficit(&self) -> u32 {
118        self.expected.saturating_sub(self.received)
119    }
120
121    /// Frames lost between the sender and here since construction/[`Self::reset`] —
122    /// the running totals' shortfall, which is what a cumulative receiver-loss gauge
123    /// wants (the per-window deficit sawtooths to zero every evaluation).
124    ///
125    /// A gap is only revealed by the arrival AFTER it, so a frame still in flight
126    /// counts as lost until it lands; a reordered arrival then takes it back off the
127    /// total. Hence an *estimate*: it tracks reality within one reorder window
128    /// rather than being monotonic.
129    pub fn lost_estimate(&self) -> u64 {
130        self.expected_total.saturating_sub(self.received_total)
131    }
132
133    /// Forget everything, including the continuity baseline. Call when the
134    /// underlying subscription restarts and a fresh instance isn't already being
135    /// used (the normal case is a fresh [`FrameSeqWindow`] per subscription; this
136    /// exists for callers that reuse one instance across restarts).
137    pub fn reset(&mut self) {
138        *self = Self::default();
139    }
140}

142/// Tunables for [`DownlinkDemand`].
143#[derive(Debug, Clone, Copy)]
144pub struct DownlinkConfig {
145    /// How often the windowed `(received, expected)` count is turned into a
146    /// decision. 1 s mirrors the cadence of the other adaptation loops in this
147    /// crate ([`crate::congestion::RateConfig::increase_interval_us`] default,
148    /// the sender's 1 s adaptation tick).
149    pub eval_interval_us: i64,
150}
151
152impl Default for DownlinkConfig {
153    fn default() -> Self {
154        Self {
155            eval_interval_us: 1_000_000,
156        }
157    }
158}
159
160/// A `Capped` request never asks for less than this — mirrors
161/// [`crate::congestion::RateConfig`]'s default `min_bps` floor, so a genuinely
162/// terrible link still asks for a minimally-useful low layer instead of
163/// degenerating toward zero.
164const MIN_CAP_BPS: u64 = 100_000;
165
166/// A new `Capped` value while already capped is only re-emitted (and the demand
167/// updated) once it has moved at least this much relative to the standing value —
168/// otherwise ordinary window-to-window wobble in a poor-but-stable link would spam
169/// the producer with a new cap every eval tick.
170const CAP_CHATTER_NUM: u64 = 1;
171const CAP_CHATTER_DEN: u64 = 10;
172
173/// The receiver's bandwidth-driven demand for one subscribed video track. Folds
174/// arriving frames' `frame_seq` into a [`FrameSeqWindow`] and, once per
175/// [`DownlinkConfig::eval_interval_us`], turns the window's delivery ratio into a
176/// [`QualityRequest`] with the `[source/4, source/2]` hysteresis band described in
177/// the module docs. Windows are evaluated by whichever of [`Self::observe_frame`]
178/// (a frame arrived) or [`Self::on_tick`] (time passed, possibly with no frames at
179/// all) crosses the due time first. Pure/sans-IO — both take `now_us` and
180/// `source_bps` explicitly.
181#[derive(Debug)]
182pub struct DownlinkDemand {
183    cfg: DownlinkConfig,
184    window: FrameSeqWindow,
185    next_eval_us: Option<i64>,
186    current: QualityRequest,
187}
188
189impl DownlinkDemand {
190    /// A fresh decider, optimistic (assumes it can take the source until its
191    /// window proves otherwise) — the same "prove impairment before asking for
192    /// less" default as the tile-demand half and the deleted decider it replaces.
193    pub fn new(cfg: DownlinkConfig) -> Self {
194        Self {
195            cfg,
196            window: FrameSeqWindow::new(),
197            next_eval_us: None,
198            current: QualityRequest::Source,
199        }
200    }
201
202    /// Record one arrived frame's wire `frame_seq` at local time `now_us`, against
203    /// the track's current `source_bps` (the top-rung bitrate the producer
204    /// encodes at — a fixed per-source constant in this codebase, mirroring what
205    /// the producer's own [`crate::congestion::SimulcastPlanner`] assumes). Folds
206    /// the frame into the current window; once [`DownlinkConfig::eval_interval_us`]
207    /// has elapsed since the last evaluation, turns the window into a (possibly
208    /// unchanged) decision. Always returns the standing demand — a faithful
209    /// projection of the last decision, matching [`Self::current`].
210    pub fn observe_frame(&mut self, seq: u32, now_us: i64, source_bps: u64) -> QualityRequest {
211        // Evaluate a DUE prior window BEFORE folding this frame in, so a window's
212        // last frame is still counted in the decision it triggers, and the frame
213        // that crosses the boundary correctly starts the next window instead of
214        // diluting the one it just closed.
215        match self.next_eval_us {
216            None => self.next_eval_us = Some(now_us + self.cfg.eval_interval_us),
217            Some(due) if now_us >= due => {
218                self.evaluate(source_bps);
219                self.next_eval_us = Some(now_us + self.cfg.eval_interval_us);
220            }
221            Some(_) => {}
222        }
223        self.window.observe(seq);
224        self.current
225    }
226
227    /// Advance the demand on a **time tick**, with no frame arrival.
228    ///
229    /// [`Self::observe_frame`] can only evaluate a window it was fed into, so a
230    /// downlink that degrades to *zero* delivery — the exact condition where this
231    /// receiver most needs to ask for the low layer — would otherwise never
232    /// evaluate at all and sit on the optimistic [`QualityRequest::Source`] default
233    /// forever. The caller drives this from its periodic reconcile/stall path;
234    /// calling it every tick alongside `observe_frame` is harmless (a window is
235    /// evaluated at most once, by whichever of the two crosses its due time).
236    pub fn on_tick(&mut self, now_us: i64, source_bps: u64) -> QualityRequest {
237        // `None` = no frame has ever arrived, so there is no window and no
238        // continuity baseline: nothing to decide (a subscription that has not
239        // started yet is not a starved one).
240        if let Some(due) = self.next_eval_us
241            && now_us >= due
242        {
243            self.evaluate(source_bps);
244            self.next_eval_us = Some(now_us + self.cfg.eval_interval_us);
245        }
246        self.current
247    }
248
249    /// Turn the current window into a decision and reset the window's counters
250    /// (not its continuity baseline).
251    fn evaluate(&mut self, source_bps: u64) {
252        let (received, expected) = self.window.take();
253        // An empty window on a stream that HAS delivered before is a starved link,
254        // not "no signal": ratio 0 so the state machine has a route from starved to
255        // `Capped`. Before the first-ever frame there is no baseline to compare
256        // against, so hold instead of manufacturing a decision from 0/0.
257        if expected == 0 && self.window.last_seq.is_none() {
258            return;
259        }
260        let ratio = if expected == 0 {
261            0.0
262        } else {
263            received as f64 / expected as f64
264        };
265        let estimate_bps =
266            ((ratio * source_bps as f64) as u64).clamp(MIN_CAP_BPS, source_bps.max(MIN_CAP_BPS));
267        let drop_below = source_bps / 4;
268        let restore_above = source_bps / 2;
269
270        self.current = match self.current {
271            QualityRequest::Source if estimate_bps < drop_below => {
272                QualityRequest::Capped(estimate_bps)
273            }
274            QualityRequest::Source => QualityRequest::Source,
275            QualityRequest::Capped(_) if estimate_bps >= restore_above => QualityRequest::Source,
276            QualityRequest::Capped(prev) => {
277                // Same layer, but only re-emit a moved cap on a MEANINGFUL move —
278                // otherwise ordinary wobble on a stable poor link re-sends the
279                // request every eval tick for no operational benefit.
280                let moved = prev.abs_diff(estimate_bps) * CAP_CHATTER_DEN > prev * CAP_CHATTER_NUM;
281                if moved {
282                    QualityRequest::Capped(estimate_bps)
283                } else {
284                    QualityRequest::Capped(prev)
285                }
286            }
287        };
288    }
289
290    /// The demand from the last [`Self::observe_frame`] / [`Self::on_tick`]
291    /// evaluation — a faithful projection, never recomputed.
292    pub fn current(&self) -> QualityRequest {
293        self.current
294    }
295
296    /// Frames lost on the downlink since this decider was created — see
297    /// [`FrameSeqWindow::lost_estimate`]. Independent of the window boundary, so a
298    /// caller can report it per arrival without perturbing the decision.
299    pub fn lost_estimate(&self) -> u64 {
300        self.window.lost_estimate()
301    }
302
303    /// The current (undrained) window's shortfall — see
304    /// [`FrameSeqWindow::window_deficit`].
305    pub fn window_deficit(&self) -> u32 {
306        self.window.window_deficit()
307    }
308
309    /// Reset to the optimistic default and forget all window/continuity state.
310    /// The normal way a restart is handled is a fresh [`DownlinkDemand`] per
311    /// subscription (see the call site); this is for a caller that reuses one
312    /// instance across restarts instead.
313    pub fn reset(&mut self) {
314        self.window.reset();
315        self.next_eval_us = None;
316        self.current = QualityRequest::Source;
317    }
318}

324    const SRC: u64 = 4_000_000; // a 4 Mbps "source" for these cases
325    const STEP_US: i64 = 1_000; // 1 ms between frames within a window
326    const TICK_US: i64 = 1_000_000; // the default eval interval
327
328    /// Feed `n` frames, in-sequence, into `dd`, dropping frames so that
329    /// EXACTLY `keep_num/keep_den` of them are delivered (a Bresenham-style
330    /// deterministic spread, not a run of drops at the start/end), all tightly
331    /// spaced (well under one eval interval). Then jumps `now_us` a full
332    /// [`TICK_US`] forward and feeds one more (always-kept) frame to CLOSE that
333    /// window — `observe_frame` evaluates a due window before folding the frame
334    /// that crossed the boundary, so this returns exactly the just-fed window's
335    /// decision, and the closing frame cleanly starts the next window. `seq` and
336    /// `now_us` both advance monotonically across calls, so repeated calls model
337    /// consecutive windows.
338    fn feed_ratio(
339        dd: &mut DownlinkDemand,
340        seq: &mut u32,
341        now_us: &mut i64,
342        n: u32,
343        keep_num: u32,
344        keep_den: u32,
345        source_bps: u64,
346    ) -> QualityRequest {
347        let mut acc = 0u32;
348        for _ in 0..n {
349            acc += keep_num;
350            let keep = acc >= keep_den;
351            if keep {
352                acc -= keep_den;
353                dd.observe_frame(*seq, *now_us, source_bps);
354            }
355            *seq = seq.wrapping_add(1);
356            *now_us += STEP_US;
357        }
358        *now_us += TICK_US;
359        let decision = dd.observe_frame(*seq, *now_us, source_bps);
360        *seq = seq.wrapping_add(1);
361        decision
362    }

369        let (mut seq, mut now) = (0u32, 0i64);

648    /// The cumulative loss estimate survives window drains (the per-window deficit
649    /// resets every evaluation; the gauge must not sawtooth back to zero).
650    #[test]
651    fn lost_estimate_accumulates_across_windows() {
652        let mut window = FrameSeqWindow::new();
653        // 0,1,3 → one frame (2) missing.
654        window.observe(0);
655        window.observe(1);
656        window.observe(3);
657        assert_eq!(window.window_deficit(), 1);
658        assert_eq!(window.lost_estimate(), 1);
659        let _ = window.take(); // window boundary
660        assert_eq!(window.window_deficit(), 0, "the window drained");
661        assert_eq!(window.lost_estimate(), 1, "the total did not");
662        // Another gap in the next window (4,5 missing) adds to the same total.
663        window.observe(6);
664        assert_eq!(window.lost_estimate(), 3);
665    }
