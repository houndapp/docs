1//! # Playout / jitter buffer
2//!
3//! Receives reassembled [`Frame`]s (which arrive reordered and with variable
4//! delay) and releases them to the decoder **in capture-timestamp order, on a
5//! steady playout schedule**. It absorbs network jitter and reordering and —
6//! crucially — is driven purely by a local playout clock, never by another
7//! stream's timestamps: gating video on the audio clock would freeze a peer's
8//! video whenever their audio is absent or stalled. This buffer never stalls
9//! on missing media.
10//!
11//! ## Playout model
12//!
13//! On the first frame, the buffer maps the sender's timeline onto the local
14//! clock with a fixed delay: `offset = local_now + target_delay − frame_ts`.
15//! Thereafter a frame with timestamp `t` plays at local time `t + offset`.
16//! Because that mapping is monotonic in `t`, releasing everything whose playout
17//! time has arrived naturally yields frames in timestamp order. A frame that
18//! shows up after its playout time has already passed (a late straggler) is
19//! dropped rather than delivered out of order; the decoder recovers at the next
20//! keyframe.
21//!
22//! That anchor is re-established from BOTH sides when the one-way delay moves
23//! under it — too deep (a join burst / sender-faster skew piles the buffer up) and
24//! too shallow (a sustained delay increase leaves every frame overdue the moment it
25//! lands). Both cost one deliberate playout step; neither fires on ordinary jitter,
26//! which [`crate::AdaptiveJitter`] absorbs by growing the depth instead.
27//!
28//! Sans-IO: the caller supplies `now_us` (local), so it is deterministic and
29//! testable against [`crate::sim`]. The buffer starts at a fixed `target_delay`;
30//! [`crate::AdaptiveJitter`] retargets it from observed jitter. Sender/receiver
31//! clock skew is not tracked (negligible over a call's duration).
32
33use std::collections::VecDeque;
34
35use crate::fragment::Frame;

37/// Absolute ceiling (µs) on how far ahead of real time the newest buffered item may be
38/// scheduled before the buffer force-resyncs (skips the backlog) — the floor for the
39/// `max(4×target_delay, …)` resync trigger, so even a tiny target can't let a burst
40/// pile up more than this. 400 ms: well above any legitimate jitter depth, so only a
41/// genuine backlog (SFU subscribe-flush / sender clock skew) trips it.
42const MAX_BUFFER_AHEAD_US: i64 = 400_000;
43
44/// Consecutive items that arrived already past their scheduled playout time before the
45/// anchor is rebuilt (the shallow-side mirror of the catch-up resync). Both this AND
46/// [`SHALLOW_RESYNC_HOLD_US`] must be satisfied.
47const SHALLOW_RESYNC_MIN_RUN: u32 = 5;
48
49/// How long (µs, on the arrival clock) the buffer must stay overdue-on-arrival before
50/// the anchor is rebuilt. A stall-then-flush burst restores the cushion within a frame
51/// or two of the flush (the queued frames land with their timestamps far ahead), so
52/// only a sustained one-way-delay STEP — where the cushion never comes back — survives
53/// 500 ms. Long enough to never fire on jitter, short enough that the cushion is back
54/// before the next tail event.
55const SHALLOW_RESYNC_HOLD_US: i64 = 500_000;
56
57/// A playout item carries a capture-clock timestamp the buffer schedules against.
58///
59/// Implemented for [`Frame`] (the audio/encoded-media path) and for any
60/// caller-defined wrapper (e.g. a decoded video frame on the receive path), so the
61/// playout *policy* lives here once and is never re-rolled in the I/O shell.
62pub trait Timestamped {
63    /// Capture-clock timestamp (µs) this item should play out at, relative to the
64    /// sender's timeline. The buffer maps it onto the local clock on first push.
65    fn timestamp_us(&self) -> i64;
66}
67
68impl Timestamped for Frame {
69    fn timestamp_us(&self) -> i64 {
70        self.timestamp_us
71    }
72}

74/// Buffers items and releases them in timestamp order on a playout schedule.
75///
76/// Generic over the item type `T: Timestamped` so the same policy serves audio
77/// (`PlayoutBuffer<Frame>`, the default) and decoded video without copying the
78/// payload — the buffer only ever reads `timestamp_us()` for ordering.
79#[derive(Debug)]
80pub struct PlayoutBuffer<T = Frame> {
81    target_delay_us: i64,
82    /// Items awaiting playout, sorted ascending by `timestamp_us`. A deque, not a
83    /// `Vec`: every release pops the FRONT, which on a `Vec` memmoves the whole
84    /// remaining buffer (up to `max_frames`) per frame. Insert is still a
85    /// binary-search + shift, but that is inherent to keeping it sorted.
86    frames: VecDeque<T>,
87    /// Local-clock offset: `playout_local = frame_ts + offset`. Set on first push.
88    offset: Option<i64>,
89    /// Timestamp of the last item released (or skipped); older items are stale.
90    last_released_ts: Option<i64>,
91    max_frames: usize,
92    /// Cumulative items dropped on [`Self::push`] because they arrived too LATE
93    /// (timestamp at/behind the last released) — a network late-arrival / loss
94    /// signal for the debug report. Diagnostic only; never affects scheduling.
95    late_drops: u64,
96    /// Cumulative items dropped on [`Self::push`] because the buffer hit `max_frames`
97    /// (drop-oldest) — backed-up / sender-faster clock-skew signal. Diagnostic only.
98    overflow_drops: u64,
99    /// Length of the current run of items that arrived already past their scheduled
100    /// playout time, and the arrival time the run started at. Any item that still had
101    /// cushion resets both. Drives the shallow re-anchor.
102    overdue_run: u32,
103    overdue_run_since_us: Option<i64>,
104    /// Cumulative items that arrived past their scheduled playout time — the "the
105    /// cushion is gone" signal (an eroded anchor releases every frame on the next
106    /// pump, so any jitter is an underflow). Diagnostic only.
107    overdue_arrivals: u64,
108    /// Cumulative shallow re-anchors (deliberate cushion rebuilds). Diagnostic only.
109    shallow_resyncs: u64,
110    /// Cumulative items dropped by [`Self::pop_latest_due`] because a NEWER item was
111    /// due in the same pump — the video "only the freshest frame matters" policy.
112    /// Not a fault (rendering them would only be overdrawn), but it is the bulk of
113    /// what a receive-side playout drop counter reports. Diagnostic only.
114    superseded_drops: u64,
115}

117impl<T: Timestamped> PlayoutBuffer<T> {
118    /// `target_delay_us` is the jitter-buffer depth (e.g. 100_000 = 100 ms):
119    /// larger absorbs more jitter/reorder at the cost of latency.
120    pub fn new(target_delay_us: i64) -> Self {
121        Self::with_capacity(target_delay_us, 256)
122    }
123
124    pub fn with_capacity(target_delay_us: i64, max_frames: usize) -> Self {
125        Self {
126            target_delay_us,
127            frames: VecDeque::new(),
128            offset: None,
129            last_released_ts: None,
130            max_frames: max_frames.max(1),
131            late_drops: 0,
132            overflow_drops: 0,
133            overdue_run: 0,
134            overdue_run_since_us: None,
135            overdue_arrivals: 0,
136            shallow_resyncs: 0,
137            superseded_drops: 0,
138        }
139    }
140
141    /// Insert an item received at local time `now_us`. Stale (already past) and
142    /// duplicate-timestamp items are dropped.
143    pub fn push(&mut self, now_us: i64, frame: T) {
144        let ts = frame.timestamp_us();
145        if let Some(last) = self.last_released_ts
146            && ts <= last
147        {
148            self.late_drops += 1; // arrived too late (already released past it)
149            return; // already released an item at/after this timestamp
150        }
151        if self.offset.is_none() {
152            self.offset = Some(now_us + self.target_delay_us - ts);
153        }
154
155        // Cushion this item still had when it landed: µs from arrival to its scheduled
156        // playout. Read BEFORE the insert (which moves the item); folded in below, once
157        // the too-deep branch has had its say.
158        let cushion_us = self.offset.map_or(0, |offset| (ts + offset) - now_us);
159
160        match self.frames.binary_search_by_key(&ts, |f| f.timestamp_us()) {
161            Ok(_) => return, // duplicate timestamp already buffered
162            Err(idx) => self.frames.insert(idx, frame),
163        }
164
165        // Bound memory: if we're badly backed up, drop the oldest (it would be
166        // released next, but we're choosing to skip ahead). Mark it released so
167        // a later copy can't reintroduce it out of order.
168        while self.frames.len() > self.max_frames {
169            let dropped = self.frames.pop_front().expect("len > max_frames >= 1");
170            self.overflow_drops += 1;
171            self.bump_last_released(dropped.timestamp_us());
172        }
173
174        // Catch-up resync: if the NEWEST buffered item is scheduled far in the future,
175        // the playout offset was anchored to a BURST (e.g. the SFU flushes a backlog of
176        // buffered frames the instant you subscribe) or to sustained sender-faster
177        // clock skew — so frames pile up seconds deep, play late, and overflow while the
178        // configured `target_delay` still reads small. Re-anchor to the newest item so
179        // we skip the stale backlog and resume near real time. One brief skip beats
180        // permanent multi-second latency + continuous overflow drops. Only fires on an
181        // egregious backlog (≫ the adaptive target), never on ordinary jitter.
182        let mut resynced = false;
183        if let Some(offset) = self.offset
184            && let Some(back_ts) = self.frames.back().map(|f| f.timestamp_us())
185        {
186            let ahead_us = (back_ts + offset) - now_us;
187            let ceiling = self
188                .target_delay_us
189                .saturating_mul(4)
190                .max(MAX_BUFFER_AHEAD_US);
191            if ahead_us > ceiling {
192                resynced = true;
193                let newest = self.frames.pop_back().expect("non-empty: just inserted");
194                let newest_ts = newest.timestamp_us();
195                let skipped = self.frames.len() as u64;
196                // The remaining items are all older than `newest` and sorted ascending,
197                // so the last one is the newest dropped — bumping `last_released` past it
198                // (one call, max-monotonic) blocks any straggler from reappearing.
199                if let Some(last_dropped) = self.frames.back().map(|f| f.timestamp_us()) {
200                    self.bump_last_released(last_dropped);
201                }
202                self.frames.clear();
203                self.overflow_drops = self.overflow_drops.saturating_add(skipped);
204                // Re-anchor: the newest item now plays `target_delay` from now.
205                self.offset = Some(now_us + self.target_delay_us - newest_ts);
206                self.frames.push_back(newest);
207            }
208        }
209
210        if resynced {
211            // The too-deep branch just rebuilt the anchor; any overdue run it was
212            // carrying is answered.
213            self.overdue_run = 0;
214            self.overdue_run_since_us = None;
215        } else {
216            self.note_arrival_cushion(now_us, cushion_us);
217        }
218    }
219
220    /// Shallow-side mirror of the catch-up resync. A sustained one-way-delay increase
221    /// (path change, bufferbloat that only partly drains) shifts every arrival past the
222    /// playout time the ORIGINAL anchor assigned it, so `pop_due` hands each frame
223    /// straight through with zero cushion and every subsequent jitter event is an
224    /// audible underflow — permanently, because nothing else ever moves `offset` down.
225    /// [`crate::AdaptiveJitter`] cannot rescue it: the inter-arrival estimator is differential,
226    /// so a *step* in delay leaves the measured jitter unchanged.
227    ///
228    /// So: count arrivals that were already overdue, and once the run is both long and
229    /// sustained ([`SHALLOW_RESYNC_MIN_RUN`] / [`SHALLOW_RESYNC_HOLD_US`]) rebuild the
230    /// anchor against the newest buffered item. Costs ONE deliberate ~`target_delay`
231    /// playout gap instead of a permanently dry buffer.
232    fn note_arrival_cushion(&mut self, now_us: i64, cushion_us: i64) {
233        // A negative target is a caller error that already means "release ASAP"; never
234        // fight it. Any item that still had cushion ends the run.
235        if cushion_us >= 0 || self.target_delay_us <= 0 {
236            self.overdue_run = 0;
237            self.overdue_run_since_us = None;
238            return;
239        }
240        self.overdue_arrivals = self.overdue_arrivals.saturating_add(1);
241        self.overdue_run = self.overdue_run.saturating_add(1);
242        let since_us = *self.overdue_run_since_us.get_or_insert(now_us);
243        if self.overdue_run < SHALLOW_RESYNC_MIN_RUN
244            || now_us.saturating_sub(since_us) < SHALLOW_RESYNC_HOLD_US
245        {
246            return;
247        }
248        let Some(anchor_ts) = self.frames.back().map(|f| f.timestamp_us()) else {
249            return;
250        };
251        self.offset = Some(now_us + self.target_delay_us - anchor_ts);
252        self.shallow_resyncs = self.shallow_resyncs.saturating_add(1);
253        self.overdue_run = 0;
254        self.overdue_run_since_us = None;
255    }
256
257    /// Release every item whose playout time has arrived, in timestamp order,
258    /// **appending** them to `out` (never cleared — the caller owns the buffer).
259    ///
260    /// This is the form the real-time audio mix pulls with: its bus thread ticks
261    /// every 10 ms and must not allocate, so it hands the same `Vec` back every
262    /// tick and this appends into it. [`Self::pop_due`] is the same release policy
263    /// with a fresh `Vec` per call, for callers not on an RT thread.
264    pub fn pop_due_into(&mut self, now_us: i64, out: &mut Vec<T>) {
265        let Some(offset) = self.offset else {
266            return;
267        };
268        while let Some(front) = self.frames.front() {
269            if front.timestamp_us() + offset > now_us {
270                break; // not due yet; the rest are later still
271            }
272            let frame = self.frames.pop_front().expect("front() just matched");
273            self.bump_last_released(frame.timestamp_us());
274            out.push(frame);
275        }
276    }
277
278    /// Release every item whose playout time has arrived, in timestamp order.
279    /// [`Self::pop_due_into`] into a fresh `Vec`; prefer that one on an RT thread.
280    pub fn pop_due(&mut self, now_us: i64) -> Vec<T> {
281        let mut out = Vec::new();
282        self.pop_due_into(now_us, &mut out);
283        out
284    }

286    /// Release the **latest** item whose playout time has arrived, dropping any
287    /// earlier still-due items as superseded — the right policy for *video*, where
288    /// only the freshest frame matters and rendering an older frame after a newer
289    /// one is pointless judder. Returns `None` when nothing is due yet.
290    ///
291    /// Equivalent in scheduling to [`Self::pop_due`] (same offset/stale rules), but
292    /// collapses a burst of due frames to the newest instead of emitting all of
293    /// them: a render surface shows one frame at a time, so if several came due in
294    /// one pump (the buffer was behind), the intermediate ones would only be
295    /// overdrawn. Each call still advances `last_released_ts` past every dropped
296    /// item so a later straggler can't reintroduce one out of order.
297    pub fn pop_latest_due(&mut self, now_us: i64) -> Option<T> {
298        let offset = self.offset?;
299        // Count the contiguous run of due items at the front.
300        let mut due = 0usize;
301        while let Some(front) = self.frames.get(due) {
302            if front.timestamp_us() + offset > now_us {
303                break;
304            }
305            due += 1;
306        }
307        if due == 0 {
308            return None;
309        }
310        // Drain all but the last due item; they are superseded by the newest.
311        for _ in 0..due - 1 {
312            let dropped = self.frames.pop_front().expect("counted due above");
313            self.superseded_drops = self.superseded_drops.saturating_add(1);
314            self.bump_last_released(dropped.timestamp_us());
315        }
316        let frame = self.frames.pop_front().expect("counted due above");
317        self.bump_last_released(frame.timestamp_us());
318        Some(frame)
319    }

321    /// Adjust the jitter-buffer depth at runtime toward `new_target_delay_us` (drive
322    /// it from a [`crate::JitterEstimator`] as measured jitter rises/falls). The playout
323    /// `offset` shifts by the delta so every buffered frame's due time moves
324    /// together: increasing depth delays playout (absorb more jitter), decreasing
325    /// advances it (lower latency). No-op before the first push (the offset is unset).
326    ///
327    /// Step it GRADUALLY — a large instant decrease can make queued frames due in the
328    /// past (a release burst); a large increase opens a gap. The caller ramps toward
329    /// the estimate in small increments; this primitive applies one step exactly.
330    pub fn retarget(&mut self, new_target_delay_us: i64) {
331        let delta = new_target_delay_us - self.target_delay_us;
332        self.target_delay_us = new_target_delay_us;
333        if let Some(offset) = self.offset.as_mut() {
334            *offset += delta;
335        }
336    }

338    /// The current jitter-buffer depth (µs).
339    pub fn target_delay_us(&self) -> i64 {
340        self.target_delay_us
341    }

355    /// Cumulative items that arrived already past their scheduled playout time — the
356    /// playout cushion is gone, so they release on the next pump with no protection.
357    /// The shallow-side counterpart of [`Self::overflow_drops`], and the signal a debug
358    /// report must key on: the depth *recommendation* cannot see an eroded anchor
359    /// (a delay STEP does not change inter-arrival jitter). Diagnostic.
360    pub fn overdue_arrivals(&self) -> u64 {
361        self.overdue_arrivals
362    }
363
364    /// Cumulative shallow re-anchors: the buffer rebuilt its cushion after a sustained
365    /// run of overdue arrivals. Each one is a deliberate ~`target_delay` playout gap,
366    /// so a nonzero-and-climbing count means the one-way delay is still moving.
367    /// Diagnostic.
368    pub fn shallow_resyncs(&self) -> u64 {
369        self.shallow_resyncs
370    }
371
372    /// Cumulative items [`Self::pop_latest_due`] dropped as superseded by a newer
373    /// due item. Expected on the video path (that IS the policy) — reported as part
374    /// of a receive-side playout-drop total, not as loss. Diagnostic.
375    pub fn superseded_drops(&self) -> u64 {
376        self.superseded_drops
377    }
378
379    /// Local-clock time (µs, the same domain as the `now_us` given to
380    /// [`Self::push`]/[`Self::pop_due`]) at which the EARLIEST buffered item comes
381    /// due — a *pull* consumer's wake deadline, so it can sleep until playout
382    /// instead of polling. `None` when nothing is buffered (nothing to wake for).
383    ///
384    /// The audio path never needs this (its pump is the sound card's 10 ms tick);
385    /// the video path has no such clock, so its pump must schedule itself.
386    pub fn next_due_us(&self) -> Option<i64> {
387        let offset = self.offset?;
388        self.frames.front().map(|f| f.timestamp_us() + offset)
389    }
390
391    /// Number of frames currently buffered.
392    pub fn len(&self) -> usize {
393        self.frames.len()
394    }
395
396    pub fn is_empty(&self) -> bool {
397        self.frames.is_empty()
398    }
399
400    fn bump_last_released(&mut self, ts: i64) {
401        self.last_released_ts = Some(match self.last_released_ts {
402            Some(prev) => prev.max(ts),
403            None => ts,
404        });
405    }

431        let mut b = PlayoutBuffer::new(DELAY);

443    fn late_drops_count_arrived_too_late_frames() {
444        let mut b = PlayoutBuffer::new(DELAY);

458        // 80 ms target. First frame anchors the offset at local now=0.
459        let mut b = PlayoutBuffer::new(80_000);
460        b.push(0, frame(0, 0));
461        // The SFU flushes a 5 s backlog: 250 frames (20 ms apart) all arriving at once.
462        for k in 1..=250 {

473            "the skipped backlog is counted as overflow drops ({})",
474            b.overflow_drops()
475        );
476        // Playout is near real time: the newest frame is due ~target_delay from now,
477        // not 5 s out. (Releasing at now=80ms+a margin yields the freshest frame.)
478        let due = b.pop_due(80_000 + 20_000);
479        assert!(!due.is_empty(), "the resynced newest frame plays promptly");
480    }

512    /// Drive `n` frames of 20 ms audio through the buffer at a steady cadence, each
513    /// arriving `transit_us` after its capture stamp, draining as the mix bus does.
514    /// Returns the local clock after the last push.
515    fn run_stream(b: &mut PlayoutBuffer<Frame>, start_ts: i64, n: i64, transit_us: i64) -> i64 {
516        let mut now = 0;
517        for k in 0..n {
518            let ts = start_ts + k * 20_000;
519            now = ts + transit_us;
520            b.push(now, frame(ts, (ts / 20_000) as u32));
521            let _ = b.pop_due(now);
522        }
523        now
524    }

526    /// THE sustained-delay-step case: mid-call the one-way delay jumps +300 ms (path
527    /// change / partly-drained bufferbloat). The anchor was taken before the step, so
528    /// every frame now lands PAST its scheduled playout and releases with zero cushion —
529    /// permanently, since the adaptive estimator is differential and sees no jitter
530    /// change. The buffer must notice the sustained run and rebuild its cushion.
531    #[test]
532    fn sustained_delay_step_re_anchors_and_restores_the_cushion() {
533        const TARGET: i64 = 80_000;
534        let mut b = PlayoutBuffer::new(TARGET);
535        // 2 s of steady arrivals: cushion sits at the full target.
536        run_stream(&mut b, 0, 100, 5_000);
537        assert_eq!(b.overdue_arrivals(), 0, "a steady link is never overdue");
538
539        // The step: every subsequent frame transits 305 ms instead of 5 ms.
540        run_stream(&mut b, 100 * 20_000, 40, 305_000);
541        assert_eq!(
542            b.shallow_resyncs(),
543            1,
544            "the sustained overdue run rebuilds the anchor exactly once (overdue {})",
545            b.overdue_arrivals()
546        );
547        // The debounce is bounded: ~500 ms of overdue arrivals, not the whole 800 ms.
548        assert!(
549            (SHALLOW_RESYNC_MIN_RUN as u64..=30).contains(&b.overdue_arrivals()),
550            "re-anchored within the debounce window, not later: {}",
551            b.overdue_arrivals()
552        );
553
554        // Cushion restored: the next frame at the SAME (stepped) delay is scheduled a
555        // full target ahead, so it is not due on arrival. (Other frames ARE due in the
556        // same pump — that queue IS the rebuilt cushion — so assert on this one.)
557        let ts = 140 * 20_000;
558        let now = ts + 305_000;
559        b.push(now, frame(ts, 140));
560        assert!(
561            !b.pop_due(now).iter().any(|f| f.frame_seq == 140),
562            "post-re-anchor arrivals must hold a cushion, not release on arrival"
563        );
564        assert!(
565            b.pop_due(now + TARGET).iter().any(|f| f.frame_seq == 140),
566            "and it comes due a full target after arrival"
567        );
568        // And it settles: no further re-anchors while the new delay holds.
569        run_stream(&mut b, 141 * 20_000, 100, 305_000);
570        assert_eq!(b.shallow_resyncs(), 1, "one rebuild, then steady");
571    }

574    /// (which [`AdaptiveJitter`] absorbs by growing the depth) nor a stall-then-flush
575    /// burst (the cushion returns with the flush) may spend a playout gap.

581        for k in 0..500i64 {
582            let ts = k * 20_000;
583            let now = ts + 50_000 + if k % 2 == 0 { 40_000 } else { -40_000 };

1006    /// End-to-end through the impairment simulator: jittered/reordered packets →
1007    /// reassembly → playout buffer must emit frames in non-decreasing timestamp
1008    /// order (the property the decoder needs).
1009    #[test]
1010    fn integration_jittered_packets_play_out_in_order() {
1011        use crate::packet::PacketHeader;
1012        use crate::sim::{NetworkConditions, SimLink};
1013        use crate::{HEADER_LEN, Packetizer, Reassembler};
1014
1015        let cond = NetworkConditions {
1016            jitter_us: 60_000,
1017            loss: 0.02,
1018            ..Default::default()
1019        };
1020        let mut link = SimLink::new(cond, 123);
1021        let mut pktzr = Packetizer::new(1);
1022        let mut reasm = Reassembler::new();
1023        let mut play = PlayoutBuffer::new(150_000);
1024
1025        let mut last_out_ts = i64::MIN;
1026        let mut released = 0;
1027        let mut now = 0i64;
1028        for f in 0..150u32 {
1029            let body: Vec<u8> = (0..4000u32).map(|i| (i ^ f) as u8).collect();
1030            for p in pktzr.packetize(
1031                TrackKind::CamVideo,
1032                Codec::Av1,
1033                f % 8 == 0,
1034                now,
1035                &body,
1036                1100,
1037            ) {
1038                link.send(now as u64, &p);
1039            }
1040            now += 33_000; // ~30 fps
1041            for p in link.poll(now as u64) {
1042                let h = PacketHeader::parse(&p).unwrap();
1043                if let Some(frame) = reasm.push(&h, &p[HEADER_LEN..]) {
1044                    play.push(now, frame);
1045                }
1046            }
1047            for frame in play.pop_due(now) {
1048                assert!(
1049                    frame.timestamp_us > last_out_ts,
1050                    "playout went backwards: {} after {}",
1051                    frame.timestamp_us,
1052                    last_out_ts
1053                );
1054                last_out_ts = frame.timestamp_us;
1055                released += 1;
1056            }
1057        }
1058        // Most frames make it through in order (a few dropped by 2% loss).
1059        assert!(released > 130, "only {released}/150 frames played out");
1060    }

1062    // ── PlayoutBuffer push/pop microbenchmark (per-frame receive hot path) ───
1063    //
1064    // Every received frame is pushed into the playout buffer and later popped on
1065    // the playout schedule; the cost is part of the receive-side per-frame
1066    // budget. This times a steady push-one/pop-due cycle at a 30 fps cadence.
1067    // Deterministic, no RNG, no IO. Not a correctness assertion (asserts only
1068    // that frames flowed through); the printout informs perf.md. Run:
1069    //   cargo test --release -p hound-media bench_playout -- --nocapture
1070    #[test]
1071    fn bench_playout_push_pop_cycle() {
1072        const N: u32 = 2_000;
1073        const STEP_US: i64 = 33_333; // ~30 fps
1074        let mut b = PlayoutBuffer::new(DELAY);
1075
1076        // Warm-up a few cycles.
1077        for i in 0..10i64 {
1078            b.push(i * STEP_US, frame(i * STEP_US, i as u32));
1079            let _ = b.pop_due(i * STEP_US);
1080        }
1081
1082        let mut popped = 0u64;
1083        let t0 = std::time::Instant::now();
1084        for i in 0..N as i64 {
1085            let now = i * STEP_US;
1086            b.push(now, frame(now, i as u32));
1087            popped += b.pop_due(now).len() as u64;
1088        }
1089        let per = t0.elapsed() / N;
1090        eprintln!(
1091            "PlayoutBuffer push+pop_due cycle: {per:?}/frame over {N} frames \
1092             ({popped} popped)"
1093        );
1094        assert!(popped > 0, "playout buffer released nothing");
1095    }

1097    // ─── Receiving from a SUPPRESSING audio publisher ───────────────────────────
1098    //
1099    // A publisher that stops sending during silence (`hound-client`'s
1100    // `call::policy::SilenceGate`) leaves TIMESTAMP GAPS in an otherwise ordinary
1101    // stream: one keepalive every 500 ms, then a jump back to 20 ms cadence when
1102    // speech resumes. The property below is what this buffer promises the resume, and
1103    // it hinges on the publisher keeping REAL capture timestamps — which is why it
1104    // lives here rather than next to the gate.
1105    //
1106    // What it does NOT cover is the receive pipeline BETWEEN the wire and this buffer:
1107    // hold this frame back, or hand the estimator the release instant instead of the
1108    // arrival instant, and every guarantee here is still met while the depth is
1109    // dragged to its cap anyway. That seam is only testable where it exists — see
1110    // `hound-client`'s `call::audio_rx` tests, which drive the real hold into a real
1111    // `AdaptiveJitter`/`PlayoutBuffer` pair and pin both the depth and the resume.
1112
1113    /// The audio playout depth (`PLAYOUT_TARGET_US` in `hound-client`).
1114    const AUDIO_TARGET: i64 = 80_000;
1115    /// The publisher's keepalive cadence while suppressed.
1116    const KEEPALIVE_US: i64 = 500_000;
1117
1118    /// **The first word after a long silence is not late.**
1119    ///
1120    /// The anchor is `offset = arrival + target − ts`, so a frame is overdue only if
1121    /// wall time advanced MORE than the sender's timestamps did. Across a suppressed
1122    /// window both advance by exactly the same amount (real capture timestamps, real
1123    /// arrival times), so the resumed frame lands with the full cushion — it must not
1124    /// be dropped as a late straggler, and it must not release early either.
1125    #[test]
1126    fn a_voiced_frame_after_a_ten_second_suppression_plays_on_the_normal_schedule() {
1127        let mut b = PlayoutBuffer::new(AUDIO_TARGET);
1128        // 1 s of speech, arriving 5 ms after capture.
1129        let mut seq = 0u32;
1130        for k in 0..50i64 {
1131            let ts = k * 20_000;
1132            b.push(ts + 5_000, frame(ts, seq));
1133            let _ = b.pop_due(ts + 5_000);
1134            seq += 1;
1135        }
1136        // 10 s suppressed: nothing but a keepalive every 500 ms.
1137        let quiet_start = 50 * 20_000;
1138        let mut t = quiet_start;
1139        while t < quiet_start + 10_000_000 - KEEPALIVE_US {
1140            t += KEEPALIVE_US;
1141            b.push(t + 5_000, frame(t, seq));
1142            let _ = b.pop_due(t + 5_000);
1143            seq += 1;
1144        }
1145        assert_eq!(b.late_drops(), 0, "keepalives are never late");
1146        assert_eq!(
1147            b.shallow_resyncs(),
1148            0,
1149            "a suppression gap must not look like a one-way-delay step"
1150        );
1151
1152        // Speech resumes. The first voiced frame must hold a FULL target of cushion.
1153        let resume_ts = quiet_start + 10_000_000;
1154        let arrival = resume_ts + 5_000;
1155        b.push(arrival, frame(resume_ts, seq));
1156        assert_eq!(
1157            b.late_drops(),
1158            0,
1159            "the first voiced frame is not a straggler"
1160        );
1161        assert!(
1162            !b.pop_due(arrival).iter().any(|f| f.frame_seq == seq),
1163            "it must not release on arrival — that would mean the cushion was gone"
1164        );
1165        assert!(
1166            b.pop_due(arrival + AUDIO_TARGET)
1167                .iter()
1168                .any(|f| f.frame_seq == seq),
1169            "and it comes due exactly one playout target after arrival, the same as \
1170             any frame in a live stream"
1171        );
1172    }
