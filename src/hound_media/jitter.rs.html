1//! # Inter-arrival jitter estimation → playout depth
2//!
3//! [`JitterEstimator`] tracks the link's actual jitter (the RFC 3550 smoothed
4//! mean plus a windowed tail peak) and recommends a [`PlayoutBuffer`] depth;
5//! [`AdaptiveJitter`] ramps a buffer toward that recommendation in bounded,
6//! asymmetric steps (grow fast, shrink slow). Sans-IO: arrivals enter as
7//! explicit `(now_us, frame_ts_us)` pairs, so the policy is deterministic.
8
9use crate::playout::{PlayoutBuffer, Timestamped};
10
11/// How long an observed transit-deviation PEAK keeps the recommended depth
12/// provisioned, on the arrival clock. The smoothed mean `J` forgets a rare spike in
13/// under a second, but the spikes that actually pop audio are RECURRING tail events —
14/// a ~100 ms connection stall every 30–60 s (a video IDR/delta burst bufferbloating
15/// the uplink delays every stream at once; stream priority cannot reorder a
16/// downstream router's queue). If the buffer relaxed between events, every recurrence
17/// would pop again; holding the peak for 90 s covers the observed recurrence interval,
18/// while a link that stays clean for 90 s relaxes back to `floor + 4·J`.
19const PEAK_WINDOW_US: i64 = 90_000_000;
20
21/// Sans-IO inter-arrival jitter estimator that recommends a [`PlayoutBuffer`] depth.
22///
23/// The fixed jitter-buffer depth is a latency-vs-smoothness compromise picked for an
24/// "average" link: too shallow and a jittery link stutters; too deep and a clean
25/// link pays needless latency — bad for a low-latency gaming call. This tracks the
26/// link's actual jitter and recommends a depth that follows it: shallow (low latency)
27/// when the link is steady, deeper only when it genuinely jitters.
28///
29/// Two signals feed the recommendation, and the depth covers BOTH:
30/// * **Mean jitter** — the RFC 3550 smoothed inter-arrival deviation: for consecutive
31///   frames `D = (arrival_i − arrival_j) − (ts_i − ts_j)` is the transit-time
32///   variation, and `J += (|D| − J) / 16`. Covers steady-state wobble.
33/// * **Tail peak** — the windowed maximum `|D|` over the last [`PEAK_WINDOW_US`]
34///   (a monotonic deque; O(1) amortized). Covers the rare-but-recurring stall the
35///   mean cannot see: one 100 ms event barely moves `J`, yet it is exactly what
36///   underflows a mean-sized buffer. (This is the NetEq-style peak mode; it is why
37///   production stacks run ~140–160 ms preferred depth on links that show tails.)
38///
39/// The recommended depth is `floor + max(4·J, peak)`, clamped to `[floor, max]`.
40/// The consumer ramps [`PlayoutBuffer::retarget`] toward this gradually. Pure.
41#[derive(Debug)]
42pub struct JitterEstimator {
43    prev_arrival_us: Option<i64>,
44    prev_ts_us: Option<i64>,
45    /// Smoothed |transit-time variation| (RFC 3550 `J`), in µs.
46    jitter_us: f64,
47    /// Windowed-max of `|D|`: (arrival_us, dev_us) kept strictly decreasing in dev
48    /// (a monotonic deque), expired past [`PEAK_WINDOW_US`]. Front = current peak.
49    peaks: std::collections::VecDeque<(i64, i64)>,
50    floor_us: i64,
51    max_us: i64,
52}
53
54impl JitterEstimator {
55    /// `floor_us` is the minimum depth (latency floor even on a perfect link);
56    /// `max_us` caps it (never add unbounded latency on a pathological link).
57    pub fn new(floor_us: i64, max_us: i64) -> Self {
58        Self {
59            prev_arrival_us: None,
60            prev_ts_us: None,
61            jitter_us: 0.0,
62            peaks: std::collections::VecDeque::new(),
63            floor_us: floor_us.max(0),
64            max_us: max_us.max(floor_us),
65        }
66    }
67
68    /// Fold one frame's arrival: `now_us` is the local receive time, `frame_ts_us` the
69    /// sender's capture timestamp. Out-of-order/duplicate stamps contribute a (valid)
70    /// deviation like any other — the estimate is `|D|`, sign-independent.
71    pub fn observe(&mut self, now_us: i64, frame_ts_us: i64) {
72        if let (Some(pa), Some(pt)) = (self.prev_arrival_us, self.prev_ts_us) {
73            let d = (now_us - pa) - (frame_ts_us - pt);
74            self.jitter_us += (d.unsigned_abs() as f64 - self.jitter_us) / 16.0;
75            // Windowed-max tail tracking (monotonic deque): drop entries this one
76            // dominates, append, expire the front past the window.
77            let dev = d.unsigned_abs().min(i64::MAX as u64) as i64;
78            while self.peaks.back().is_some_and(|&(_, v)| v <= dev) {
79                self.peaks.pop_back();
80            }
81            self.peaks.push_back((now_us, dev));
82            while self
83                .peaks
84                .front()
85                .is_some_and(|&(t, _)| t < now_us - PEAK_WINDOW_US)
86            {
87                self.peaks.pop_front();
88            }
89        }
90        self.prev_arrival_us = Some(now_us);
91        self.prev_ts_us = Some(frame_ts_us);
92    }
93
94    /// The recommended jitter-buffer depth (µs): `floor + max(4·J, peak)`, clamped to
95    /// `[floor, max]` — mean jitter sizes the steady state, the windowed tail peak
96    /// provisions for the recurring stall the mean cannot see.
97    pub fn recommended_delay_us(&self) -> i64 {
98        let need = (4.0 * self.jitter_us) as i64;
99        let est = self.floor_us + need.max(self.peak_us());
100        est.clamp(self.floor_us, self.max_us)
101    }
102
103    /// The current smoothed jitter estimate (µs) — for stats/debug.
104    pub fn jitter_us(&self) -> i64 {
105        self.jitter_us as i64
106    }
107
108    /// The windowed-max transit deviation (µs) — the tail the depth provisions for.
109    pub fn peak_us(&self) -> i64 {
110        self.peaks.front().map_or(0, |&(_, v)| v)
111    }
112}

114/// Absolute cap (µs) on the depth [`AdaptiveJitter::with_floor`] will grow to.
115/// Deliberately NOT a multiple of the floor: tying it to the floor made the headroom
116/// above the anchor shrink with the latency target, so the tighter the floor the less
117/// tail a link was allowed to have. 450 ms is the point past which a call has stopped
118/// being conversational — beyond it, added depth buys smoothness nobody wants — and it
119/// comfortably covers the recurring ~100 ms stalls the peak tracker provisions for.
120const MAX_ADAPTIVE_DEPTH_US: i64 = 450_000;
121
122/// Drives a [`PlayoutBuffer`]'s depth from measured jitter: folds each frame's
123/// arrival into a [`JitterEstimator`] and ramps the buffer toward the recommended
124/// depth in **bounded, asymmetric steps**. The IO shell just calls
125/// [`Self::on_frame`] per received frame; all the policy lives here, sans-IO.
126///
127/// ## Why asymmetric (grow fast, shrink slow)
128///
129/// A jitter spike that outruns the buffer causes an underrun — an audible gap or a
130/// video stall — so the buffer must **grow promptly** to get ahead of rising jitter.
131/// Shrinking only trades latency back once the link is durably calm, and shrinking
132/// too eagerly risks oscillation (and a release burst, since decreasing depth makes
133/// queued frames due earlier). So the up-step is large and the down-step small: the
134/// buffer chases jitter up quickly and relaxes down gently. Each call applies at most
135/// one step, so a single outlier can't lurch the buffer.
136#[derive(Debug)]
137pub struct AdaptiveJitter {
138    est: JitterEstimator,
139    grow_step_us: i64,
140    shrink_step_us: i64,
141}
142
143impl AdaptiveJitter {
144    /// `floor_us`/`max_us` bound the buffer depth (passed to the [`JitterEstimator`]);
145    /// `grow_step_us` is the max increase and `shrink_step_us` the max decrease applied
146    /// per observed frame. Make `grow_step_us` several times `shrink_step_us`.
147    pub fn new(floor_us: i64, max_us: i64, grow_step_us: i64, shrink_step_us: i64) -> Self {
148        Self {
149            est: JitterEstimator::new(floor_us, max_us),
150            grow_step_us: grow_step_us.max(0),
151            shrink_step_us: shrink_step_us.max(0),
152        }
153    }
154
155    /// A sensible default for real-time media: a `floor_us` latency floor, the
156    /// [`MAX_ADAPTIVE_DEPTH_US`] cap, grow at 10 ms/frame, shrink at 1 ms/frame.
157    /// (`JitterEstimator::new` raises the cap to the floor if the floor is larger.)
158    pub fn with_floor(floor_us: i64) -> Self {
159        Self::new(floor_us, MAX_ADAPTIVE_DEPTH_US, 10_000, 1_000)
160    }
161
162    /// Fold one frame's arrival (`now_us` local receive time, `frame_ts_us` the
163    /// sender's capture timestamp) and apply ONE bounded retarget step to `buf`
164    /// toward the current recommendation.
165    pub fn on_frame<T: Timestamped>(
166        &mut self,
167        buf: &mut PlayoutBuffer<T>,
168        now_us: i64,
169        frame_ts_us: i64,
170    ) {
171        self.est.observe(now_us, frame_ts_us);
172        let target = self.est.recommended_delay_us();
173        let current = buf.target_delay_us();
174        let delta = target - current;
175        let stepped = if delta >= 0 {
176            current + delta.min(self.grow_step_us)
177        } else {
178            current + delta.max(-self.shrink_step_us)
179        };
180        if stepped != current {
181            buf.retarget(stepped);
182        }
183    }
184
185    /// The current smoothed jitter estimate (µs) — for stats/debug.
186    pub fn jitter_us(&self) -> i64 {
187        self.est.jitter_us()
188    }
189
190    /// The windowed-max transit deviation (µs) the depth is provisioned for — the
191    /// half of [`Self::recommended_delay_us`] that the smoothed mean cannot explain.
192    /// Diagnostic: a recommendation stuck high with a low [`Self::jitter_us`] is the
193    /// peak holding it there, which is a very different problem from a jittery link.
194    pub fn peak_us(&self) -> i64 {
195        self.est.peak_us()
196    }
197
198    /// The depth the estimator currently recommends (`floor + 4·J`, clamped) — what
199    /// the buffer is ramping toward. A recommendation well above the current depth
200    /// (which the bounded grow step can't reach fast enough) is the "jitter buffer too
201    /// shallow for the link" signal the debug report keys on. It is only half the
202    /// shallow story: the estimator is differential, so it cannot see a one-way-delay
203    /// STEP eroding the anchor — [`PlayoutBuffer::overdue_arrivals`] is that half.
204    pub fn recommended_delay_us(&self) -> i64 {
205        self.est.recommended_delay_us()
206    }
207}

209#[cfg(test)]
210mod jitter_estimator_tests {
211    use super::{JitterEstimator, PEAK_WINDOW_US};
212
213    /// A perfectly steady 30 fps link (arrival cadence == capture cadence) has zero
214    /// transit variation → the estimator stays at the latency floor.
215    #[test]
216    fn steady_link_stays_at_floor() {
217        let mut e = JitterEstimator::new(20_000, 200_000);
218        for i in 0..100 {
219            let t = i * 33_333;
220            e.observe(t, t); // arrival tracks capture exactly
221        }
222        assert_eq!(e.recommended_delay_us(), 20_000, "no jitter → floor");
223    }
224
225    /// THE recurring-stall case (the tail event that pops audio): a link that
226    /// is calm except for a single ~100 ms stall. The smoothed mean barely moves
227    /// (4·J stays far below the stall), so a mean-only recommendation would leave the
228    /// buffer shallow and the NEXT stall would underflow it — the tail peak must
229    /// provision the depth immediately and hold it across the recurrence window.
230    #[test]
231    fn single_stall_provisions_tail_depth_immediately() {
232        let floor = 80_000;
233        let mut e = JitterEstimator::new(floor, 320_000);
234        // 3 s of perfectly steady 20 ms audio cadence…
235        let mut t = 0i64;
236        for _ in 0..150 {
237            e.observe(t, t);
238            t += 20_000;
239        }
240        assert_eq!(e.recommended_delay_us(), floor, "steady: at the floor");
241        // …then one frame arrives 100 ms late (the connection stall).
242        e.observe(t + 100_000, t);
243        assert!(e.peak_us() >= 100_000, "the stall is captured as the peak");
244        assert!(
245            e.recommended_delay_us() >= floor + 100_000,
246            "depth provisions for the tail at once (got {})",
247            e.recommended_delay_us()
248        );
249        assert!(
250            (4.0 * e.jitter_us() as f64) < 50_000.0,
251            "the mean alone would NOT have provisioned (4J = {})",
252            4 * e.jitter_us()
253        );
254    }
255
256    /// The tail hold is bounded: once the link stays clean past the peak window the
257    /// recommendation relaxes back to the floor (no permanent latency tax).
258    #[test]
259    fn tail_depth_expires_after_the_window() {
260        let floor = 80_000;
261        let mut e = JitterEstimator::new(floor, 320_000);
262        let mut t = 0i64;
263        for _ in 0..10 {
264            e.observe(t, t);
265            t += 20_000;
266        }
267        e.observe(t + 100_000, t); // the stall
268        assert!(e.recommended_delay_us() >= floor + 100_000);
269        // A clean link for > PEAK_WINDOW_US (90 s at 20 ms cadence).
270        for _ in 0..(PEAK_WINDOW_US / 20_000 + 100) {
271            t += 20_000;
272            e.observe(t, t);
273        }
274        assert_eq!(e.peak_us(), 0, "the peak expired with the window");
275        assert_eq!(
276            e.recommended_delay_us(),
277            floor,
278            "recommendation relaxed back to the floor"
279        );
280    }
281
282    /// A jittery link (arrivals deviate from the capture cadence) drives the
283    /// recommended depth above the floor, but never past the cap.
284    #[test]
285    fn jittery_link_deepens_within_cap() {
286        let mut e = JitterEstimator::new(20_000, 120_000);
287        // Capture every 33ms; arrivals alternate ±20ms around the cadence.
288        for i in 0..200 {
289            let ts = i * 33_333;
290            let arrival = ts + if i % 2 == 0 { 20_000 } else { -20_000 };
291            e.observe(arrival, ts);
292        }
293        let d = e.recommended_delay_us();
294        assert!(d > 20_000, "jitter must deepen the buffer: {d}");
295        assert!(d <= 120_000, "never past the cap: {d}");
296    }
297}

299#[cfg(test)]
300mod adaptive_jitter_tests {
301    use super::{AdaptiveJitter, MAX_ADAPTIVE_DEPTH_US, PlayoutBuffer, Timestamped};
302
303    struct Item(i64);
304    impl Timestamped for Item {
305        fn timestamp_us(&self) -> i64 {
306            self.0
307        }
308    }
309
310    /// A jittery link grows the buffer quickly; once it goes calm, the buffer relaxes
311    /// back down — but slower than it grew (the grow/shrink asymmetry).
312    #[test]
313    fn grows_fast_shrinks_slow() {
314        let floor = 20_000;
315        let mut adapt = AdaptiveJitter::new(floor, 200_000, 10_000, 1_000);
316        let mut buf: PlayoutBuffer<Item> = PlayoutBuffer::new(floor);
317        buf.push(0, Item(0)); // establish the offset so retarget takes effect
318
319        // Phase 1: heavy jitter — arrivals swing ±40 ms around a 33 ms cadence.
320        for i in 0..200i64 {
321            let ts = i * 33_333;
322            let arrival = ts + if i % 2 == 0 { 40_000 } else { -40_000 };
323            adapt.on_frame(&mut buf, arrival, ts);
324        }
325        let grown = buf.target_delay_us();
326        assert!(grown > floor, "jitter must deepen the buffer: {grown}");
327
328        // Phase 2: link goes perfectly steady — depth relaxes back toward the floor.
329        let mut t = 200 * 33_333;
330        for _ in 0..50 {
331            adapt.on_frame(&mut buf, t, t);
332            t += 33_333;
333        }
334        let relaxed = buf.target_delay_us();
335        assert!(
336            relaxed < grown,
337            "a calm link must relax the buffer: {relaxed}"
338        );
339        assert!(relaxed >= floor, "never below the floor: {relaxed}");
340
341        // The shrink over 50 calm frames is bounded by the (small) shrink step and so
342        // is far less than the growth it underwent — the asymmetry holds.
343        assert!(
344            relaxed > floor,
345            "slow shrink shouldn't reach the floor in 50 frames: {relaxed}"
346        );
347    }

349    /// Jitter that would recommend a depth ABOVE the configured cap holds the buffer
350    /// AT the cap and never exceeds it: the `JitterEstimator` clamps its
351    /// recommendation to `[floor, max]`, so even a pathologically jittery link can't
352    /// grow the buffer past `max_us` (bounded latency on a bad link).
353    #[test]
354    fn observed_jitter_over_cap_holds_at_cap() {
355        let floor = 20_000;
356        let max = 60_000; // a deliberately tight cap
357        let mut adapt = AdaptiveJitter::new(floor, max, 10_000, 1_000);
358        let mut buf: PlayoutBuffer<Item> = PlayoutBuffer::new(floor);
359        buf.push(0, Item(0)); // establish the offset
360
361        // Violent jitter: arrivals swing ±200 ms around a 33 ms cadence — the raw
362        // recommendation (floor + 4·J) would be far past the 60 ms cap.
363        for i in 0..400i64 {
364            let ts = i * 33_333;
365            let arrival = ts + if i % 2 == 0 { 200_000 } else { -200_000 };
366            adapt.on_frame(&mut buf, arrival, ts);
367            assert!(
368                buf.target_delay_us() <= max,
369                "buffer must never exceed the cap: {} > {max}",
370                buf.target_delay_us()
371            );
372        }
373        assert_eq!(
374            buf.target_delay_us(),
375            max,
376            "sustained over-cap jitter pins the buffer exactly at the cap"
377        );
378    }

380    /// `with_floor`'s cap is ABSOLUTE, not a multiple of the floor: a production 80 ms
381    /// floor must still be allowed to provision for a ~300 ms tail. (It used to be
382    /// `4 × floor` = 320 ms, which made the headroom above the anchor shrink with every
383    /// latency-floor reduction.)
384    #[test]
385    fn with_floor_cap_is_absolute_not_a_multiple_of_the_floor() {
386        let floor = 80_000; // PLAYOUT_TARGET_US
387        let mut adapt = AdaptiveJitter::with_floor(floor);
388        let mut buf: PlayoutBuffer<Item> = PlayoutBuffer::new(floor);
389        buf.push(0, Item(0));
390
391        // A 300 ms tail event: the peak tracker provisions floor + 300 ms = 380 ms,
392        // which the old 4×floor cap would have clipped to 320 ms.
393        let mut t = 0i64;
394        for _ in 0..10 {
395            adapt.on_frame(&mut buf, t, t);
396            t += 20_000;
397        }
398        adapt.on_frame(&mut buf, t + 300_000, t);
399        assert!(
400            adapt.recommended_delay_us() > floor * 4,
401            "the recommendation must clear the old 4×floor cap: {}",
402            adapt.recommended_delay_us()
403        );
404
405        // Still bounded: a pathological 2 s tail clamps at the absolute cap.
406        t += 400_000;
407        adapt.on_frame(&mut buf, t + 2_000_000, t);
408        assert_eq!(
409            adapt.recommended_delay_us(),
410            MAX_ADAPTIVE_DEPTH_US,
411            "and never past the absolute cap"
412        );
413        // The bounded grow step still walks the buffer up to it and stops there.
414        for _ in 0..200 {
415            t += 20_000;
416            adapt.on_frame(&mut buf, t, t);
417            assert!(buf.target_delay_us() <= MAX_ADAPTIVE_DEPTH_US);
418        }
419        assert_eq!(buf.target_delay_us(), MAX_ADAPTIVE_DEPTH_US);
420    }
421
422    /// One step never moves the buffer by more than the configured bound — a single
423    /// outlier can't lurch playout.
424    #[test]
425    fn one_step_is_bounded() {
426        let floor = 20_000;
427        let mut adapt = AdaptiveJitter::new(floor, 500_000, 10_000, 1_000);
428        let mut buf: PlayoutBuffer<Item> = PlayoutBuffer::new(floor);
429        buf.push(0, Item(0));
430        // Prime prev arrival/ts.
431        adapt.on_frame(&mut buf, 0, 0);
432        // A huge transit jump on the very next frame.
433        adapt.on_frame(&mut buf, 1_000_000, 33_333);
434        assert!(
435            buf.target_delay_us() - floor <= 10_000,
436            "a single frame can't grow the buffer past one grow-step"
437        );
438    }
