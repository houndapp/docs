1//! # Congestion control + adaptive quality
2//!
3//! Sans-IO pieces for graceful degradation under congestion:
4//!
5//! * [`RateController`] — a loss-based AIMD bandwidth estimator (the
6//!   well-known WebRTC loss controller): increase slowly when loss is low, hold
7//!   in the middle band, and back off fast when loss is high. Produces a target
8//!   send bitrate.
9//! * [`QualityLadder`] — maps an available bitrate to a discrete video [`Rung`]
10//!   with hysteresis (**drop fast, climb slow**) and a manual ceiling, falling
11//!   back to [`Quality::AudioOnly`] when even the lowest rung won't fit.
12//!
13//! Principle: **audio is sacred, video is the release valve.** This governs
14//! video only; audio runs at a fixed low bitrate underneath and is never
15//! starved. Both types are pure and clock-driven for deterministic tests.
16
17/// Tunables for [`RateController`]. Defaults mirror the classic WebRTC loss
18/// controller.
19#[derive(Debug, Clone)]
20pub struct RateConfig {
21    pub min_bps: u64,
22    pub max_bps: u64,
23    pub start_bps: u64,
24    /// Below this loss fraction, increase. Default 2%.
25    pub increase_loss: f64,
26    /// Above this loss fraction, decrease. Default 10%.
27    pub decrease_loss: f64,
28    /// Multiplicative increase per update step. Default 1.08.
29    pub increase_factor: f64,
30    /// Minimum spacing between *increase* steps ("up slow"). Decreases are not
31    /// rate-limited ("down fast").
32    pub increase_interval_us: i64,
33    /// Minimum spacing between increase steps for an interval that carried NO
34    /// traffic ([`RateController::on_feedback_idle`]). A zero loss reading over an
35    /// idle interval is not evidence of capacity — nothing was on the wire to lose
36    /// — so the estimate probes upward on a much slower ladder than the measured
37    /// one. Long enough to stop a paused-video controller from ramping open-loop
38    /// into an immediate re-collapse, short enough that a genuinely recovered link
39    /// still resumes in seconds.
40    pub idle_increase_interval_us: i64,
41}
42
43impl Default for RateConfig {
44    fn default() -> Self {
45        Self {
46            min_bps: 100_000,   // 100 kbps floor (≈ lowest video rung)
47            max_bps: 4_000_000, // 4 Mbps ceiling
48            start_bps: 600_000,
49            increase_loss: 0.02,
50            decrease_loss: 0.10,
51            increase_factor: 1.08,
52            increase_interval_us: 1_000_000,
53            idle_increase_interval_us: 4_000_000,
54        }
55    }
56}

58/// Loss-based AIMD send-rate estimator.
59#[derive(Debug)]
60pub struct RateController {
61    cfg: RateConfig,
62    target_bps: u64,
63    last_increase_us: i64,
64}
65
66impl RateController {
67    pub fn new(cfg: RateConfig) -> Self {
68        let target_bps = cfg.start_bps.clamp(cfg.min_bps, cfg.max_bps);
69        Self {
70            cfg,
71            target_bps,
72            last_increase_us: i64::MIN / 2,
73        }
74    }
75
76    /// Feed an observed loss fraction (0.0..=1.0) for the last interval; returns
77    /// the updated target bitrate.
78    ///
79    /// * loss > `decrease_loss`: back off immediately to `target * (1 - 0.5·loss)`.
80    /// * loss < `increase_loss`: grow by `increase_factor`, at most once per
81    ///   `increase_interval_us`.
82    /// * in between: hold.
83    pub fn on_feedback(&mut self, loss: f64, now_us: i64) -> u64 {
84        self.feed(loss, now_us, self.cfg.increase_interval_us)
85    }
86
87    /// Feedback for an interval in which the sender produced **nothing** (the
88    /// quality ladder paused video, or the capture loop was idle). Decreases and
89    /// holds behave exactly as [`Self::on_feedback`] — a congestion signal from
90    /// elsewhere on the shared connection still backs the estimate off at full
91    /// speed — but the *climb* runs on [`RateConfig::idle_increase_interval_us`]
92    /// instead of the measured cadence, because a zero loss fraction measured over
93    /// zero traffic proves nothing about capacity. Without this the controller
94    /// ramps open-loop while paused and re-crosses the resume threshold every few
95    /// seconds, which is the sender half of the video on/off blink.
96    pub fn on_feedback_idle(&mut self, loss: f64, now_us: i64) -> u64 {
97        self.feed(loss, now_us, self.cfg.idle_increase_interval_us)
98    }
99
100    fn feed(&mut self, loss: f64, now_us: i64, increase_interval_us: i64) -> u64 {
101        let loss = loss.clamp(0.0, 1.0);
102        if loss > self.cfg.decrease_loss {
103            let factor = 1.0 - 0.5 * loss;
104            self.target_bps = (self.target_bps as f64 * factor) as u64;
105            // Suppress an immediate re-increase after backing off.
106            self.last_increase_us = now_us;
107        } else if loss < self.cfg.increase_loss
108            && now_us - self.last_increase_us >= increase_interval_us
109        {
110            self.target_bps = (self.target_bps as f64 * self.cfg.increase_factor) as u64;
111            self.last_increase_us = now_us;
112        }
113        self.target_bps = self.target_bps.clamp(self.cfg.min_bps, self.cfg.max_bps);
114        self.target_bps
115    }
116
117    pub fn target_bps(&self) -> u64 {
118        self.target_bps
119    }
120
121    /// Move the bitrate ceiling at runtime — e.g. the quality ladder was
122    /// reconfigured to a richer (or capped) top rung, so the estimate is allowed to
123    /// climb higher (or must drop). The current estimate is re-clamped into the new
124    /// `[min, max]` so it never sits above the ceiling.
125    pub fn set_max_bps(&mut self, max_bps: u64) {
126        self.cfg.max_bps = max_bps.max(self.cfg.min_bps);
127        self.target_bps = self.target_bps.clamp(self.cfg.min_bps, self.cfg.max_bps);
128    }
129}

131/// One receiver's standing quality request for a video track, as the sender sees it.
132///
133/// Demand-driven simulcast: the sender publishes only the **source** until a receiver
134/// proves it cannot keep up and asks for less. The request is deliberately
135/// constrained — a receiver asks for AT MOST one tier, and only *below* the source
136/// (it can never ask for more than the publisher captures; a cap at/above source is
137/// just "I can take source").
138#[derive(Debug, Clone, Copy, PartialEq, Eq)]
139pub enum QualityRequest {
140    /// This receiver handles the source/top quality — no lower layer needed for it.
141    Source,
142    /// This receiver can sustain at most this many bits/sec; it wants a lower layer.
143    Capped(u64),
144    /// This receiver is **not rendering** the track at all (its tile unmounted or
145    /// scrolled out of the viewport). It still needs a layer to be subscribed to —
146    /// until off-screen tiles unsubscribe outright, falling back to the SOURCE
147    /// would be far worse — but it must NOT constrain the shared low layer that
148    /// receivers who *are* rendering depend on. The carried value is the cap it
149    /// would ask for if it were visible, used only when every requester is hidden.
150    ///
151    /// See [`plan_layers`] for how the fold treats it.
152    Hidden(u64),
153}
154
155/// What a sender should publish for a video track given ALL its receivers' requests
156/// ([`plan_layers`]). At most two streams: the source and one shared low layer.
157#[derive(Debug, Clone, Copy, PartialEq, Eq)]
158pub struct LayerPlan {
159    /// Encode + publish the source/top layer (at least one receiver can take it).
160    pub send_source: bool,
161    /// `Some(bps)` → also (or *only*, if `!send_source`) publish ONE shared low layer
162    /// sized for the most-constrained requester (so every requester can decode it).
163    /// `None` → no low layer demanded.
164    pub low_layer_bps: Option<u64>,
165}
166
167impl LayerPlan {
168    /// The zero-cost default: encode only the source, no low layer. Byte-identical
169    /// to a build without simulcast — the plan a sender acts on whenever no receiver
170    /// has asked for less, and the safe plan the planner publishes while actuation is
171    /// disabled (see [`SimulcastPlanner`]).
172    pub const SOURCE_ONLY: LayerPlan = LayerPlan {
173        send_source: true,
174        low_layer_bps: None,
175    };
176}

178/// Aggregate every receiver's [`QualityRequest`] into the sender's [`LayerPlan`].
179///
180/// The whole demand-driven, coalesced, source-droppable policy in one pure function:
181/// * a receiver wanting `Source` (or a cap ≥ source — it can only ask for *lower*)
182///   keeps the source alive;
183/// * receivers wanting less than source collapse to ONE shared low layer at the
184///   **minimum** of their caps, so the weakest can still decode it (bounded to two
185///   total streams — no per-receiver explosion);
186/// * a [`QualityRequest::Hidden`] receiver keeps the low layer alive but is
187///   **excluded from that minimum** while anyone is actually rendering. A tile
188///   nobody is looking at must not decide the geometry everyone else watches;
189/// * if NO receiver can take the source, `send_source` is `false` and only the low
190///   layer is published — the optimisation that stops encoding a stream nobody wants.
191///
192/// ## The cap is a cross-receiver MINIMUM
193///
194/// Among receivers that *are* rendering, this is still one shared layer sized to
195/// the most-constrained of them: a peer showing a 180p thumbnail and a peer showing
196/// a 360p tile both get 180p. That is inherent to publishing one low layer rather
197/// than one per receiver, and excluding hidden tiles (above) only removes the
198/// pathological case. The designed fix for the rest is a receiver-side **total
199/// budget allocator** that ranks tracks and hands out per-track rungs against a
200/// total bitrate and decode-instance budget, rather than each track independently
201/// picking a rung and the producer taking a global minimum.
202///
203/// With no receivers the source stays on (idle default; nothing to optimise yet).
204/// Pure/sans-IO → unit-tested below.
205pub fn plan_layers(requests: &[QualityRequest], source_bps: u64) -> LayerPlan {
206    let mut send_source = requests.is_empty();
207    let mut min_cap: Option<u64> = None;
208    let mut min_hidden: Option<u64> = None;
209    for req in requests {
210        match *req {
211            // "Only lower than what's being sent": a cap at/above source means this
212            // receiver can take the source itself.
213            QualityRequest::Source => send_source = true,
214            QualityRequest::Capped(bps) if bps >= source_bps => send_source = true,
215            QualityRequest::Capped(bps) => {
216                min_cap = Some(min_cap.map_or(bps, |m| m.min(bps)));
217            }
218            // Kept out of `min_cap` entirely: it may only decide the layer when it
219            // is the ONLY kind of requester left.
220            QualityRequest::Hidden(bps) => {
221                min_hidden = Some(min_hidden.map_or(bps, |m| m.min(bps)));
222            }
223        }
224    }
225    LayerPlan {
226        send_source,
227        low_layer_bps: min_cap.or(min_hidden),
228    }
229}

231/// Producer-side aggregator: tracks each receiver's standing [`QualityRequest`] for
232/// one published video track and yields the current [`LayerPlan`] via [`plan_layers`].
233///
234/// The sender feeds it every `QualityRequested` (peer → cap) and every peer-leave,
235/// and acts on the returned plan only when it *changes* — so a steady call costs
236/// nothing and a flapping receiver (already damped by its own send-side hysteresis)
237/// can at most toggle one layer. Peers are keyed by a plain `u32` (the session passes
238/// `PeerId.0`) so this core stays free of the control-plane types. Pure/sans-IO.
239#[derive(Debug)]
240pub struct SimulcastPlanner {
241    source_bps: u64,
242    /// Per-receiver standing request. A peer absent here has not asked for anything
243    /// → it is assumed to take the source (the default before any request arrives).
244    caps: std::collections::BTreeMap<u32, QualityRequest>,
245    /// The plan the current requests imply — what the sender publishes: a shared low
246    /// layer when a receiver asks for a below-source rung, source-only otherwise.
247    /// Recomputed on every request/roster change and acted on when it changes.
248    plan: LayerPlan,
249}
250
251impl SimulcastPlanner {
252    /// Start with the given source bitrate and no receiver requests (plan = source
253    /// only — the zero-cost common case).
254    pub fn new(source_bps: u64) -> Self {
255        Self {
256            source_bps,
257            caps: std::collections::BTreeMap::new(),
258            plan: plan_layers(&[], source_bps),
259        }
260    }
261
262    fn recompute(&mut self) -> bool {
263        let reqs: Vec<QualityRequest> = self.caps.values().copied().collect();
264        let next = plan_layers(&reqs, self.source_bps);
265        let changed = next != self.plan;
266        self.plan = next;
267        changed
268    }
269
270    /// Record a receiver's standing request; returns `true` if the resulting plan
271    /// changed (the sender re-publishes the low-layer target).
272    pub fn set_request(&mut self, peer: u32, req: QualityRequest) -> bool {
273        if self.caps.get(&peer) == Some(&req) {
274            return false;
275        }
276        self.caps.insert(peer, req);
277        self.recompute()
278    }
279
280    /// Drop a receiver (it left or unsubscribed); returns `true` if the plan changed.
281    pub fn remove_peer(&mut self, peer: u32) -> bool {
282        if self.caps.remove(&peer).is_none() {
283            return false;
284        }
285        self.recompute()
286    }
287
288    /// Reconcile against the live roster: drop caps for any peer not in `keep` (a
289    /// receiver that left without retracting its request). Returns `true` if the plan
290    /// changed. `keep` membership is tested via the closure so the caller needn't
291    /// allocate a set.
292    pub fn retain_peers(&mut self, keep: impl Fn(u32) -> bool) -> bool {
293        let before = self.caps.len();
294        self.caps.retain(|&peer, _| keep(peer));
295        if self.caps.len() == before {
296            return false;
297        }
298        self.recompute()
299    }
300
301    /// Update the source bitrate (the source rung re-negotiated); returns `true` if
302    /// the plan changed (a cap that was below the old source may now be ≥ the new one).
303    pub fn set_source_bps(&mut self, source_bps: u64) -> bool {
304        if self.source_bps == source_bps {
305            return false;
306        }
307        self.source_bps = source_bps;
308        self.recompute()
309    }
310
311    /// The peer ids currently requesting a CAPPED (below-source) layer — the receivers
312    /// the shared low layer serves. The sender's feedback partitioner excludes these
313    /// from the SOURCE adaptation so a low-layer receiver can't drag the source down
314    /// (see the simulcast notes).
315    pub fn capped_peers(&self) -> std::collections::BTreeSet<u32> {
316        self.caps
317            .iter()
318            .filter_map(|(&p, req)| match req {
319                QualityRequest::Capped(bps) | QualityRequest::Hidden(bps)
320                    if *bps < self.source_bps =>
321                {
322                    Some(p)
323                }
324                _ => None,
325            })
326            .collect()
327    }
328
329    /// The plan the sender must act on: a shared low layer when a receiver requests a
330    /// below-source rung, [`LayerPlan::SOURCE_ONLY`] otherwise. Safe because the source
331    /// encode is never dropped — the low layer is purely additive, so a constrained
332    /// receiver switches its own subscription to the low track without starving anyone.
333    pub fn plan(&self) -> LayerPlan {
334        self.plan
335    }
336}

338/// A discrete video quality preset.
339#[derive(Debug, Clone, Copy, PartialEq, Eq)]
340pub struct Rung {
341    pub bitrate_bps: u64,
342    pub width: u32,
343    pub height: u32,
344    pub fps: u32,
345}

347/// The selected quality for the current conditions.
348#[derive(Debug, Clone, Copy, PartialEq, Eq)]
349pub enum Quality {
350    Video(Rung),
351    /// Below the lowest video rung — drop video, keep audio.
352    AudioOnly,
353}

355/// Maps available bitrate to a [`Rung`] with hysteresis and a manual ceiling.
356#[derive(Debug)]
357pub struct QualityLadder {
358    /// Rungs sorted ascending by bitrate.
359    rungs: Vec<Rung>,
360    current: usize,
361    /// Whether the last [`Self::update`] decided [`Quality::AudioOnly`] (the
362    /// available bitrate fell below the lowest rung). Kept so [`Self::current`] is
363    /// a faithful projection of the last decision rather than silently reporting
364    /// `Video(rungs[0])` while video is actually dropped.
365    audio_only: bool,
366    /// Manual maximum rung index (inclusive); `None` = no cap.
367    ceiling: Option<usize>,
368    /// Headroom required to climb a rung (e.g. 1.15 = need 15% above its cost).
369    climb_margin: f64,
370    /// Consecutive [`Self::update`] calls that have cleared the resume threshold
371    /// while [`Quality::AudioOnly`]. Reset by any call that does not.
372    resume_streak: u32,
373    /// Doubling exponent on the resume dwell, raised each time a resume collapses
374    /// again before it settled. Bounded by [`RESUME_BACKOFF_MAX_SHIFT`].
375    resume_backoff_shift: u32,
376    /// Consecutive [`Self::update`] calls spent on video since the last resume.
377    video_streak: u32,
378}
379
380/// Headroom above `rungs[0]` required to RESUME video after
381/// [`Quality::AudioOnly`]. Dropping video is deliberately margin-free (drop fast),
382/// but resuming at the bare rung-0 bitrate puts the link *exactly* at the edge that
383/// just failed, so the first real frames re-collapse it — a stable on/off blink.
384/// Mirrors the rung-to-rung `climb_margin`: prove the link can carry the lowest
385/// rung *plus* headroom before putting traffic back on it.
386const RESUME_MARGIN: f64 = 1.15;
387
388/// How many consecutive [`QualityLadder::update`] calls must clear
389/// [`RESUME_MARGIN`] before video resumes. The estimate that drives a resume is
390/// built with no video traffic on the link (see
391/// [`RateController::on_feedback_idle`]), so it is the weakest evidence the
392/// controller ever acts on; requiring it to hold for a few ticks costs a
393/// second-scale delay on a genuine recovery and removes the single-tick flip.
394const RESUME_DWELL_TICKS: u32 = 3;
395
396/// Cap on the resume-dwell doubling. A link whose true capacity sits *below*
397/// `rungs[0]` cannot be fixed by a margin alone: every probe genuinely fits the
398/// estimate, puts real frames on the wire, and is knocked straight back down — a
399/// self-sustaining blink. Each resume that fails to settle doubles the dwell (3 →
400/// 6 → 12 → 24 calls), so the sender converges on "stay audio-only, probe rarely"
401/// instead of blinking, while a link that recovers for real still gets a probe
402/// within the capped window.
403const RESUME_BACKOFF_MAX_SHIFT: u32 = 3;
404
405/// Video ticks after which a resume counts as SETTLED — the link really could
406/// carry rung 0, so the next collapse starts a fresh episode at the base dwell
407/// rather than inheriting the old backoff.
408const RESUME_SETTLED_TICKS: u32 = 30;
409
410impl QualityLadder {
411    /// The default camera ladder: 180p15 → 360p20 → 360p30 → 540p30 → 720p30.
412    pub fn standard() -> Self {
413        Self::new(vec![
414            Rung {
415                bitrate_bps: 150_000,
416                width: 320,
417                height: 180,
418                fps: 15,
419            },
420            Rung {
421                bitrate_bps: 300_000,
422                width: 640,
423                height: 360,
424                fps: 20,
425            },
426            Rung {
427                bitrate_bps: 500_000,
428                width: 640,
429                height: 360,
430                fps: 30,
431            },
432            Rung {
433                bitrate_bps: 800_000,
434                width: 960,
435                height: 540,
436                fps: 30,
437            },
438            Rung {
439                bitrate_bps: 1_500_000,
440                width: 1280,
441                height: 720,
442                fps: 30,
443            },
444        ])
445    }
446
447    /// The default screen-share ladder — [`DegradePriority::Legibility`] capped at
448    /// 1080p60, i.e. `screen_for(ScreenQuality::default())`: 720p15 → 1080p15 →
449    /// 1080p30 → 1080p60 (climb resolution before fps; on the way down, drop fps
450    /// before resolution). Kept as a named constructor for callers that don't
451    /// expose the per-stream control; [`Self::screen_for`] is the configurable one.
452    pub fn screen() -> Self {
453        Self::screen_for(ScreenQuality::default())
454    }
455
456    /// Build a screen-share ladder for a per-stream [`ScreenQuality`] preference
457    /// (degrade priority + resolution/fps caps). See [`screen_rungs`] for the rung
458    /// construction; this just wraps the result. Pair with [`Self::reconfigure`]
459    /// to change the preference on a live stream without losing the bitrate
460    /// position.
461    pub fn screen_for(quality: ScreenQuality) -> Self {
462        Self::new(screen_rungs(quality))
463    }
464
465    /// Swap in a new rung set (e.g. from a mid-stream [`ScreenQuality`] change)
466    /// while preserving the *bitrate position*: the next [`Self::update`] re-picks
467    /// from the live available bitrate, so the only state to carry is "are we
468    /// paused (audio-only)". `current` is clamped into the new ladder; a stale
469    /// ceiling index is dropped (the new ladder's caps are baked into its rungs).
470    pub fn reconfigure(&mut self, rungs: Vec<Rung>) {
471        assert!(!rungs.is_empty(), "ladder needs at least one rung");
472        self.current = self.current.min(rungs.len() - 1);
473        self.ceiling = None;
474        self.rungs = rungs;
475        // The new ladder has its own rung-0 cost, so any resume progress — or
476        // backoff earned against the old floor — is meaningless. `video_streak`
477        // carries through: like `audio_only`, it describes the stream's current
478        // state, not the ladder's shape.
479        self.resume_streak = 0;
480        self.resume_backoff_shift = 0;
481    }
482
483    /// The BWE estimate at which the *top* rung becomes selectable: the top rung's
484    /// bitrate plus the [`Self::update`] climb-margin headroom. The rate controller
485    /// seeds its ceiling from this so it can actually climb to the richest rung on a
486    /// fast link. Without it a flat BWE cap below the top rung × margin would strand
487    /// the high rungs entirely — e.g. a 1440p120 top needs ~20 Mbps, so a 4 Mbps cap
488    /// would top out at 1080p15 and never offer the quality a gaming LAN can carry.
489    pub fn bwe_ceiling_bps(&self) -> u64 {
490        let top = self.rungs.last().map(|r| r.bitrate_bps).unwrap_or(0);
491        (top as f64 * self.climb_margin).ceil() as u64
492    }
493
494    /// `rungs` must be non-empty and sorted ascending by bitrate.
495    pub fn new(rungs: Vec<Rung>) -> Self {
496        assert!(!rungs.is_empty(), "ladder needs at least one rung");
497        Self {
498            rungs,
499            current: 0,
500            audio_only: false,
501            ceiling: None,
502            climb_margin: 1.15,
503            resume_streak: 0,
504            resume_backoff_shift: 0,
505            // A fresh ladder starts on video (see `current`), and that starting
506            // state is presumed settled — the first collapse of a call is a fresh
507            // episode, not a failed resume probe.
508            video_streak: RESUME_SETTLED_TICKS,
509        }
510    }
511
512    /// Set the manual ceiling by rung index (clamped). `None` removes the cap.
513    pub fn set_ceiling(&mut self, idx: Option<usize>) {
514        self.ceiling = idx.map(|i| i.min(self.rungs.len() - 1));
515    }
516
517    /// Pick the quality for the given available bitrate. Drops as many rungs as
518    /// needed immediately, but climbs at most one rung per call and only with
519    /// `climb_margin` headroom (and never above the ceiling).
520    ///
521    /// The [`Quality::AudioOnly`] boundary is **asymmetric**: video drops the
522    /// instant the estimate falls below `rungs[0]`, but resuming needs
523    /// [`RESUME_MARGIN`] of headroom sustained for [`RESUME_DWELL_TICKS`] calls —
524    /// doubled per failed resume (see [`RESUME_BACKOFF_MAX_SHIFT`]).
525    pub fn update(&mut self, available_bps: u64) -> Quality {
526        let max_idx = self
527            .ceiling
528            .unwrap_or(self.rungs.len() - 1)
529            .min(self.rungs.len() - 1);
530
531        if self.audio_only {
532            let resume_at = (self.rungs[0].bitrate_bps as f64 * RESUME_MARGIN).ceil() as u64;
533            if available_bps < resume_at {
534                self.resume_streak = 0;
535                self.current = 0;
536                return Quality::AudioOnly;
537            }
538            self.resume_streak = self.resume_streak.saturating_add(1);
539            if self.resume_streak < (RESUME_DWELL_TICKS << self.resume_backoff_shift) {
540                self.current = 0;
541                return Quality::AudioOnly;
542            }
543        } else if available_bps < self.rungs[0].bitrate_bps {
544            self.resume_backoff_shift = if self.video_streak >= RESUME_SETTLED_TICKS {
545                0 // video held for a long stretch — a fresh episode, not a blink
546            } else {
547                (self.resume_backoff_shift + 1).min(RESUME_BACKOFF_MAX_SHIFT)
548            };
549            self.video_streak = 0;
550            self.current = 0;
551            self.audio_only = true;
552            self.resume_streak = 0;
553            return Quality::AudioOnly;
554        }
555        self.audio_only = false;
556        self.resume_streak = 0;
557        self.video_streak = self.video_streak.saturating_add(1);
558        if self.current > max_idx {
559            self.current = max_idx;
560        }
561        // Drop fast: step down until the current rung fits.
562        while self.current > 0 && self.rungs[self.current].bitrate_bps > available_bps {
563            self.current -= 1;
564        }
565        // Climb slow: a single step, only with headroom and under the ceiling.
566        if self.current < max_idx {
567            let next = self.rungs[self.current + 1];
568            if available_bps as f64 >= next.bitrate_bps as f64 * self.climb_margin {
569                self.current += 1;
570            }
571        }
572        Quality::Video(self.rungs[self.current])
573    }
574
575    /// The quality from the last [`Self::update`] (a faithful projection: reports
576    /// [`Quality::AudioOnly`] if that was the last decision). Before any update,
577    /// reports the lowest video rung (the starting state).
578    pub fn current(&self) -> Quality {
579        if self.audio_only {
580            Quality::AudioOnly
581        } else {
582            Quality::Video(self.rungs[self.current])
583        }
584    }
585}

589/// Which dimension a screen-share ladder sacrifices **first** as bandwidth drops
590/// — a per-stream user choice, switchable on a live stream (not asked at stream
591/// start). Gameplay wants motion; an IDE/slides wants legibility.
592#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
593pub enum DegradePriority {
594    /// Hold frame rate, drop **resolution** first — smooth motion over sharpness
595    /// (gameplay, video). On the way up: climb fps at low res, then climb res.
596    Motion,
597    /// Hold resolution, drop **frame rate** first — readable detail over smooth
598    /// motion (text, IDEs, slides). On the way up: climb res, then climb fps.
599    #[default]
600    Legibility,
601}

603/// The user's maximum-resolution cap for a stream (the "Source / 1440p / 1080p /
604/// 720p / 480p" control). `Source` imposes no cap beyond the capture's own
605/// geometry — the session still clamps it to the encoder's native target.
606#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
607pub enum MaxResolution {
608    #[default]
609    Source,
610    P1440,
611    P1080,
612    P720,
613    P480,
614}

616impl MaxResolution {
617    /// Luma-height cap in pixels (`Source` ⇒ effectively unbounded; the caller
618    /// clamps to the real capture size).
619    pub fn height_cap(self) -> u32 {
620        match self {
621            MaxResolution::Source => u32::MAX,
622            MaxResolution::P1440 => 1440,
623            MaxResolution::P1080 => 1080,
624            MaxResolution::P720 => 720,
625            MaxResolution::P480 => 480,
626        }
627    }
628}

636    pub max_resolution: MaxResolution,
637    /// Frame-rate cap; the UI offers 30 / 60 / 120.

651/// Candidate ladder heights (16:9 luma height), ascending.
652const LADDER_HEIGHTS: [u32; 4] = [480, 720, 1080, 1440];
653/// Candidate ladder frame rates, ascending.
654const LADDER_FPS: [u32; 4] = [15, 30, 60, 120];
655
656/// Make `v` even (encoders want even luma dimensions).
657const fn even(v: u32) -> u32 {
658    v & !1
659}
660
661/// The 16:9 luma width for a height (even). Descriptive: the encoder fits the
662/// *source* aspect, so this is the reference geometry the rung reports.
663const fn width_16x9(height: u32) -> u32 {
664    even(height * 16 / 9)
665}
666
667/// Target bitrate (bits/s) for one screen rung at `height`×`fps`. A single
668/// model — `pixels × fps^0.72 × k`, rounded to 100 kbps — instead of a hand-tuned
669/// table per (res, fps), staying monotonic in both axes (the ladder ordering relies
670/// on it).
671///
672/// **Calibrated for SCREEN content, not camera video.** Screen content (text, UI,
673/// large static regions, sharp intra-friendly edges) compresses several times better
674/// than motion video; a camera-video rate table (1080p30 → ~8 Mbps) oversizes a
675/// screenshare and saturates the shared connection, starving everything else. `k` is
676/// set so 1080p30 ≈ 2.4 Mbps — in line with production-RTC screenshare rates over
677/// codecs less efficient than H.265, and ample for crisp text. Rungs: 720p15≈0.6M,
678/// 1080p15≈1.5M, 1080p30≈2.4M, 1080p60≈4.0M. The BWE ceiling tracks the top rung,
679/// so a fast link reaches it.
680pub(crate) fn screen_bitrate_bps(height: u32, fps: u32) -> u64 {
681    // 0.10 solves 1080p30 → ~2.4 Mbps (screen-content target).
682    const K: f64 = 0.10;
683    let pixels = (width_16x9(height) as f64) * (height as f64);
684    round_rung_bps(pixels * (fps as f64).powf(FPS_EXP) * K)
685}
686
687/// How a rung's rate scales with frame rate. Sub-linear: halving the frame rate
688/// does not halve the bits, because the residual between consecutive frames grows
689/// as they get further apart. Shared by both content models.
690const FPS_EXP: f64 = 0.72;
691
692/// How a CAMERA rung's rate scales with pixel count. Sub-linear for the same
693/// reason the fps term is: a small picture needs proportionally *more* bits per
694/// pixel (the per-macroblock and header overheads amortise worse, and downscaling
695/// concentrates detail), which is why the industry's 180p thumbnails are priced at
696/// ~3× the bits-per-pixel of their 720p sources rather than 1/16th of the bitrate.
697/// Solved from the two anchors below — see [`camera_bitrate_bps`].
698const CAM_PIXEL_EXP: f64 = 0.70;
699
700/// Calibration constant for [`camera_bitrate_bps`], solved so the model reproduces
701/// the shipped SOURCE rung exactly: 1280×720 @ 30 fps → 1500 kbps (`CAM_BITRATE`
702/// in `crates/app/client/src/call/mod.rs`). The whole camera ladder is therefore
703/// anchored on the one rate the client already publishes at.
704const CAM_BPP_K: f64 = 8.6567;
705
706/// Target bitrate (bits/s) for one CAMERA rung at `height`×`fps`, from a
707/// bits-per-pixel model — the camera-content twin of [`screen_bitrate_bps`].
708///
709/// **Calibrated for MOTION VIDEO, and sub-linear in pixels.** A linear-in-pixels
710/// model anchored at 720p30 would price a 320×180 thumbnail at ~57 kbps, well
711/// under what a legible talking head costs; the `pixels^0.70` term instead yields
712/// ~130 kbps at 320×180@15 and ~420 kbps at 640×360@20, which is where three
713/// independent shipped ladders sit:
714///
715/// * LiveKit's presets: `h180` = 320×180 @ **160 kbps**, `h360` = 640×360 @
716///   **500 kbps**, with simulcast defaulting to `h180 + h360 + primary`.
717/// * Daily's published defaults: **80 kbps** / 10 fps / 320×180, **200 kbps** /
718///   15 fps / 640×360, 680 kbps / 30 fps / 1280×720.
719/// * Jitsi's bridge: 180p thumbnails, 360p on-stage.
720///
721/// The same numbers show up as published per-tile budgets: Zoom quotes 2.0 Mbps for
722/// a 25-tile gallery and 4.0 Mbps for 49 (~80 kbps/tile), and LiveKit reports a
723/// 25-person meeting at 4 Mbps (~160 kbps/tile).
724///
725/// Rounded to 10 kbps rather than [`round_rung_bps`]'s 100 kbps: the thumbnail
726/// rung is only ~130 kbps, where a 100 kbps quantum is a 25% distortion.
727pub(crate) fn camera_bitrate_bps(height: u32, fps: u32) -> u64 {
728    let pixels = (width_16x9(height) as f64) * (height as f64);
729    round_bps(
730        pixels.powf(CAM_PIXEL_EXP) * (fps as f64).powf(FPS_EXP) * CAM_BPP_K,
731        10_000,
732    )
733}
734
735/// Round a modelled rung rate to 100 kbps, never below the 100 kbps floor (the
736/// lowest rate worth publishing video at).
737fn round_rung_bps(raw: f64) -> u64 {
738    round_bps(raw, 100_000)
739}
740
741/// Round a modelled rate to `quantum`, never below the 100 kbps floor (the lowest
742/// rate worth publishing video at, and `RateConfig::default().min_bps`).
743fn round_bps(raw: f64, quantum: u64) -> u64 {
744    let q = quantum as f64;
745    let rounded = (raw.max(0.0) / q).round() * q;
746    (rounded as u64).max(100_000)
747}

749/// Build the screen-share rungs for a [`ScreenQuality`], **sans-IO and pure**.
750///
751/// Candidate heights/fps are filtered to the caps, then assembled into one
752/// ascending-bitrate ladder whose *order of climb* encodes the degrade priority:
753/// * [`DegradePriority::Legibility`] climbs resolution at the floor fps, then fps
754///   at the cap resolution — so descending drops fps before resolution. Floor
755///   resolution is 720p (below that text is unreadable) unless the cap is lower.
756/// * [`DegradePriority::Motion`] climbs fps at the floor resolution, then
757///   resolution at the cap fps — so descending drops resolution before fps. Floor
758///   fps is 30 (below that it isn't "motion") unless the cap is lower; floor
759///   resolution is 480p.
760///
761/// The result is sorted ascending by bitrate (the ladder invariant) and
762/// deduplicated, so callers get a clean monotonic ladder for any cap combination.
763pub fn screen_rungs(quality: ScreenQuality) -> Vec<Rung> {
764    let cap_h = quality.max_resolution.height_cap();
765    let (floor_h, floor_fps) = match quality.degrade {
766        DegradePriority::Legibility => (720, 15),
767        DegradePriority::Motion => (480, 30),
768    };
769
770    // Heights ≤ cap and ≥ the priority's floor — but always keep the cap itself,
771    // so a cap below the floor (e.g. legibility @ 480p) still yields one height.
772    let heights: Vec<u32> = LADDER_HEIGHTS
773        .iter()
774        .copied()
775        .filter(|h| *h <= cap_h && (*h >= floor_h || *h >= cap_h))
776        .collect();
777    let heights = if heights.is_empty() {
778        vec![LADDER_HEIGHTS[0].min(cap_h)]
779    } else {
780        heights
781    };
782    // Frame rates ≤ cap and ≥ the priority's floor — always keep at least one.
783    let fpses: Vec<u32> = LADDER_FPS
784        .iter()
785        .copied()
786        .filter(|f| *f <= quality.max_fps && *f >= floor_fps)
787        .collect();
788    let fpses = if fpses.is_empty() {
789        vec![floor_fps.min(quality.max_fps)]
790    } else {
791        fpses
792    };
793
794    let max_h = *heights.last().unwrap();
795    let min_h = *heights.first().unwrap();
796    let max_f = *fpses.last().unwrap();
797    let min_f = *fpses.first().unwrap();
798
799    // (height, fps) levels in climb order per priority.
800    let mut levels: Vec<(u32, u32)> = Vec::new();
801    match quality.degrade {
802        DegradePriority::Legibility => {
803            for &h in &heights {
804                levels.push((h, min_f));
805            }
806            for &f in fpses.iter().skip(1) {
807                levels.push((max_h, f));
808            }
809        }
810        DegradePriority::Motion => {
811            for &f in &fpses {
812                levels.push((min_h, f));
813            }
814            for &h in heights.iter().skip(1) {
815                levels.push((h, max_f));
816            }
817        }
818    }
819
820    let mut rungs: Vec<Rung> = levels
821        .into_iter()
822        .map(|(h, f)| Rung {
823            bitrate_bps: screen_bitrate_bps(h, f),
824            width: width_16x9(h),
825            height: h,
826            fps: f,
827        })
828        .collect();
829    rungs.sort_by_key(|r| r.bitrate_bps);
830    // Bitrate is a pure function of (height, fps), so equal bitrate ⇒ the same
831    // bandwidth target; dedup on it to keep the ladder STRICTLY ascending — the
832    // `QualityLadder` climb/drop hysteresis compares bitrates and `new` documents
833    // an ascending ladder.
834    rungs.dedup_by_key(|r| r.bitrate_bps);
835    rungs
836}

838/// [`screen_rungs`] re-priced for the share's **real capture geometry**.
839///
840/// The rung table is built from nominal 16:9 candidate heights because the bitrate
841/// model needs a reference geometry — but the encoder never upscales past the
842/// source and always follows the source's own aspect ([`crate::adapt`]'s
843/// `fit_encode_dims`). Left unfitted, a 900×500 window share still offers a
844/// "1080p60 = 4 Mbps" top rung: the AIMD climbs to it and hands 4 Mbps to an
845/// encoder pinned at 900×500 — roughly 8× the bits that geometry needs, spent on
846/// the QUIC connection audio shares (and paid again as receive-side packet CPU).
847///
848/// So each rung is re-priced by the pixels it will ACTUALLY encode
849/// (`fit_encode_dims(source, rung.height)`) and its reported geometry becomes that
850/// encode geometry. Rungs that collapse onto the same fitted geometry+fps produce
851/// the same rate and are deduplicated, so a source smaller than the ladder's floor
852/// height simply yields a shorter ladder — the correct outcome, since those rungs
853/// were never distinguishable at that capture size.
854pub fn screen_rungs_for_source(quality: ScreenQuality, source: (u32, u32)) -> Vec<Rung> {
855    let mut rungs: Vec<Rung> = screen_rungs(quality)
856        .into_iter()
857        .map(|r| {
858            let (w, h) = crate::adapt::fit_encode_dims(source, r.height);
859            let nominal = (r.width as f64) * (r.height as f64);
860            let scaled = if nominal > 0.0 {
861                r.bitrate_bps as f64 * ((w as f64) * (h as f64) / nominal)
862            } else {
863                r.bitrate_bps as f64
864            };
865            Rung {
866                bitrate_bps: round_rung_bps(scaled),
867                width: w,
868                height: h,
869                fps: r.fps,
870            }
871        })
872        .collect();
873    // Re-sort: re-pricing is monotonic in pixels but the clamp can reorder a
874    // low-res/high-fps rung against a high-res/low-fps one. Dedup keeps the ladder
875    // STRICTLY ascending, the `QualityLadder::new` precondition.
876    rungs.sort_by_key(|r| r.bitrate_bps);
877    rungs.dedup_by_key(|r| r.bitrate_bps);
878    rungs
879}

1250    /// Closed-loop: model loss as the fraction by which the offered target
1251    /// exceeds the link capacity, feed it back, and assert the controller
1252    /// converges near capacity, then backs off when capacity is cut and recovers
1253    /// when it returns.
1254    #[test]
1255    fn converges_to_capacity_and_reacts_to_changes() {
1256        let mut c = RateController::new(RateConfig {
1257            min_bps: 100_000,
1258            max_bps: 4_000_000,
1259            start_bps: 300_000,
1260            ..Default::default()
1261        });
1262        let mut ladder = QualityLadder::standard();
1263
1264        let observe = |target: u64, capacity: u64| -> f64 {
1265            if target <= capacity {
1266                0.0
1267            } else {
1268                (target - capacity) as f64 / target as f64
1269            }
1270        };
1271
1272        // Capacity ~1.2 Mbps; run for a while.
1273        let mut now = 0i64;
1274        let cap = 1_200_000;
1275        for _ in 0..60 {
1276            let loss = observe(c.target_bps(), cap);
1277            c.on_feedback(loss, now);
1278            ladder.update(c.target_bps());
1279            now += 1_000_000;
1280        }
1281        let t = c.target_bps();
1282        assert!(
1283            t > 900_000 && t < 1_400_000,
1284            "did not converge near cap: {t}"
1285        );
1286        // At ~1.2 Mbps we should be sitting at the 800k (540p) rung.
1287        assert_eq!(
1288            ladder.current(),
1289            Quality::Video(Rung {
1290                bitrate_bps: 800_000,
1291                width: 960,
1292                height: 540,
1293                fps: 30
1294            })
1295        );
1296
1297        // Capacity collapses to 250 kbps → fast back-off.
1298        let cap = 250_000;
1299        for _ in 0..10 {
1300            let loss = observe(c.target_bps(), cap);
1301            c.on_feedback(loss, now);

1303            now += 1_000_000;
1304        }
1305        assert!(c.target_bps() < 400_000, "should have backed off hard");
1306        // Capacity returns → slow recovery climbs back up over time.
1307        let cap = 2_000_000;
1308        for _ in 0..40 {
1309            let loss = observe(c.target_bps(), cap);
1310            c.on_feedback(loss, now);

1799    /// A resume that collapses again before it settles is evidence the link cannot
1800    /// carry rung 0 at all: each failure doubles the dwell the next probe must
1801    /// clear (bounded), so the ladder converges on "stay audio-only, probe rarely"
1802    /// rather than blinking video on and off forever.
1803    #[test]
1804    fn repeated_failed_resumes_back_off_exponentially() {
1805        // Updates spent in AudioOnly before video comes back, feeding an estimate
1806        // that always clears the resume margin.
1807        fn ticks_to_resume(l: &mut QualityLadder) -> u32 {
1808            for i in 1..1000 {
1809                if matches!(l.update(200_000), Quality::Video(_)) {
1810                    return i;
1811                }
1812            }
1813            panic!("never resumed");
1814        }
1815
1816        let mut l = QualityLadder::standard();
1817        let mut seen = Vec::new();
1818        for _ in 0..5 {
1819            assert_eq!(l.update(100_000), Quality::AudioOnly, "collapse");
1820            seen.push(ticks_to_resume(&mut l));
1821        }
1822        assert_eq!(
1823            seen[0], RESUME_DWELL_TICKS,
1824            "first probe uses the base dwell"
1825        );
1826        for w in seen.windows(2) {
1827            assert!(
1828                w[1] >= w[0],
1829                "each failed resume must not shorten the next dwell: {seen:?}"
1830            );
1831        }
1832        let cap = RESUME_DWELL_TICKS << RESUME_BACKOFF_MAX_SHIFT;
1833        assert_eq!(
1834            *seen.last().unwrap(),
1835            cap,
1836            "the backoff saturates rather than growing without bound: {seen:?}"
1837        );
1838    }
