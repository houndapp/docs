1//! # Encoder-rebuild geometry dwell + encode-geometry fitting
2//!
3//! The follow-up to the resolution-adaptation hotfix: the sender's `VideoEncoder`
4//! is opened once at a fixed geometry, so honoring a rung's resolution (not just
5//! its bitrate/fps) means tearing down and rebuilding the encoder when the chosen
6//! rung's geometry changes. That is expensive (a real device open/close) and
7//! visible (forces a fresh IDR — a decoder can't switch resolution mid-GOP, so
8//! every receiver must resync), so unlike bitrate/fps — which the SAME encoder
9//! retargets for free every tick via [`crate::AdaptDecision::target_bps`] /
10//! `AdaptShared::target_fps` — a resolution change must not flap with ordinary
11//! AIMD/delay-BWE noise. [`GeometryDwell`] is the dedicated hysteresis for that
12//! one decision; the IO shell (the capture/encode loop) drives it independently of
13//! [`crate::VideoAdaptation::on_tick`], mirroring the `CodecSwitch`/`LayerHysteresis`
14//! dwell pattern already used for the codec-switch and receive-side layer-switch
15//! decisions (`crates/client/src/call/policy.rs`) — kept here instead so the
16//! decision stays sans-IO and unit-tested in this crate.
17
18/// How long a candidate encode geometry must hold, unchanged, before the encoder
19/// actually rebuilds at it. Without a dwell, ordinary AIMD noise — or the
20/// delay-based estimator pinning low for a beat before recovering — would rebuild
21/// the encoder every few seconds. ~5 ticks at the 1 s adaptation cadence the
22/// client drives [`crate::VideoAdaptation::on_tick`] at.
23pub const GEOMETRY_REBUILD_DWELL_US: i64 = 5_000_000;
24
25/// Make `v` even and at least 2 (encoders need even, non-zero luma extents).
26fn even_min2(v: u32) -> u32 {
27    (v & !1).max(2)
28}
29
30/// Fit a rung's target HEIGHT to the SOURCE capture's own aspect ratio, even-
31/// clamped. The camera and screen quality ladders both store a nominal 16:9
32/// `width` on each [`crate::Rung`] purely for the bitrate model (see [`crate::screen_rungs`]) —
33/// a screen source is not always 16:9 (a portrait phone mirror, a near-square
34/// shared window), so encoding at the rung's raw width would stretch it. Deriving
35/// the width from the source's own aspect instead — the same approach the low
36/// simulcast layer uses for its fixed low-resolution encode — keeps every encode
37/// at the right aspect regardless of what the ladder's rung table says. For the
38/// camera ladder, whose capture geometry and rungs are both already 16:9, this
39/// reproduces the rung's own width exactly.
40///
41/// `target_height` is clamped to never exceed the source's own height: a small
42/// source (a shared window narrower than the ladder's floor rung) must never be
43/// UPSCALED just because a low rung nominally asks for more pixels than the
44/// source has — the pre-adaptation code had the same "never upscale past capture"
45/// invariant (`w.min(capture_dims.width())`), and it still applies now that the
46/// width is aspect-derived instead of taken from the rung directly.
47///
48/// [`crate::ladder::screen_rungs_for_source`] prices its rungs through this
49/// same function, so a rung's bitrate always describes the geometry the encoder
50/// will actually be built at.
51pub fn fit_encode_dims(source: (u32, u32), target_height: u32) -> (u32, u32) {
52    let source_h = source.1.max(1);
53    let target_height = target_height.max(2).min(source_h);
54    let scale = target_height as f64 / source_h as f64;
55    let width = ((source.0.max(1) as f64) * scale).round().max(2.0) as u32;
56    (even_min2(width), even_min2(target_height))
57}
58
59/// Fit a rung's PIXEL BUDGET to the source's own aspect ratio, even-clamped and
60/// never upscaled — the area-based sibling of [`fit_encode_dims`].
61///
62/// Use this where the source aspect is unknown in advance and the rung's *cost* is
63/// what must be held constant: a screen share can be 16:9, a portrait phone mirror,
64/// or a near-square window, and a long-edge budget prices those wildly differently
65/// (a 640-long-edge square is 640×640 = 1.78× the pixels of 640×360 while being
66/// charged the same). Holding area instead keeps the rung's price honest for every
67/// aspect: 16:9 and 9:16 sources land on exactly the same geometry a 640 long edge
68/// gave them, and only the near-square case — the one that was mispriced — moves.
69///
70/// [`fit_encode_dims`] stays the right tool for a FIXED-aspect source (the camera),
71/// where a rung height is the ladder's own unit and area follows from it.
72pub fn fit_encode_area(source: (u32, u32), target_pixels: u64) -> (u32, u32) {
73    let (w, h) = (source.0.max(1) as f64, source.1.max(1) as f64);
74    let source_pixels = w * h;
75    let target = (target_pixels.max(4)) as f64;
76    if source_pixels <= target {
77        return (even_min2(source.0.max(1)), even_min2(source.1.max(1))); // already small
78    }
79    let scale = (target / source_pixels).sqrt();
80    (
81        even_min2((w * scale).round().max(2.0) as u32),
82        even_min2((h * scale).round().max(2.0) as u32),
83    )
84}

86/// What a [`GeometryDwell`] tick is asking for, in whatever unit its ladder
87/// measures rungs in — the dwell fits it to the source with the matching function,
88/// so the hysteresis and the producer's own geometry derivation can never disagree
89/// about what a rung means.
90#[derive(Debug, Clone, Copy, PartialEq, Eq)]
91pub enum RungTarget {
92    /// A 16:9 reference HEIGHT, fitted to a FIXED-aspect source
93    /// ([`fit_encode_dims`]) — the camera, where height is the ladder's own unit.
94    Height(u32),
95    /// A rung's PIXEL BUDGET, fitted to an arbitrary-aspect source
96    /// ([`fit_encode_area`]) — a screen share, which can be 16:9, a portrait phone
97    /// mirror or a near-square window, and where the rung's *price* is what has to
98    /// be held. A near-square window fitted by height is mispriced by up to 1.78×,
99    /// which is the bug `low_screen_dims` exists to fix; the dwell has to fit the
100    /// same way or it would debounce a geometry the producer never builds.
101    Pixels(u64),
102}
103
104/// Dwell-based hysteresis for the encoder-REBUILD geometry decision (pure,
105/// sans-IO — see [`GEOMETRY_REBUILD_DWELL_US`] for why this must not track the
106/// per-tick [`crate::Rung`] directly). [`Self::observe`] feeds the freshly-decided rung
107/// each tick as a [`RungTarget`] (`None` for [`crate::Quality::AudioOnly`] — no video
108/// decision, so the last committed geometry carries through unchanged; encode is
109/// paused anyway) and returns the geometry the encoder should be built at right
110/// now: unchanged unless the SAME candidate has held for
111/// [`GEOMETRY_REBUILD_DWELL_US`] continuously — any intervening DIFFERENT candidate
112/// (a still-settling climb, or genuine flapping) resets the timer, so only a
113/// genuinely stable rung ever triggers a rebuild.
114///
115/// ## The dwell's currency is the TARGET, not the geometry it fits to
116///
117/// Two rungs can fit to the *same* pixels and still be different rungs: a share of a
118/// window already smaller than both rungs' budgets cannot be upscaled onto either
119/// ([`fit_encode_area`] never upscales), so `SCREEN_LOW_RUNG` and `SCREEN_MID_RUNG`
120/// both land on the capture geometry — while differing by 15 vs 30 fps and 200 kbps
121/// vs 1.1 Mbps. Neither of those can be applied to a LIVE encoder: `VideoEncode` has
122/// no frame-rate setter at all (fps is fixed at construction, and on the software
123/// encoders it also fixes the intra period), and `set_bitrate` is a documented no-op
124/// on the software H.264 and AV1 paths. So a rung change with unchanged geometry is
125/// still a REBUILD, and debouncing on the fitted geometry would have made it
126/// invisible — a cap wobbling across the 1.1 Mbps boundary would then have flipped
127/// the encoder's frame rate and rate-control target every tick with no hysteresis
128/// at all, or (with the rung in the rebuild key) rebuilt + IDR'd every tick.
129///
130/// Keying on the target instead means every rung change is debounced by the same
131/// rule, whether or not the geometry moves: [`Self::committed_target`] only ever
132/// moves after a full dwell, so a caller keying its rebuild on
133/// `(committed(), committed_target())` rebuilds at most once per
134/// [`GEOMETRY_REBUILD_DWELL_US`] and not at all under a wobble.
135#[derive(Debug, Clone)]
136pub struct GeometryDwell {
137    source: (u32, u32),
138    committed: (u32, u32),
139    /// The target [`Self::committed`] was fitted from, once one has been observed.
140    /// `None` while the dwell is still holding the geometry it was SEEDED with and
141    /// has been told nothing yet — the first `observe` at that geometry adopts its
142    /// target without a dwell, which is what lets a low encoder open at the rung its
143    /// first coalesced cap bought rather than five seconds later.
144    committed_target: Option<RungTarget>,
145    pending: Option<(RungTarget, i64)>,
146}
147
148impl GeometryDwell {
149    /// `source` is the capture's own geometry — pass the encoder's ACTUAL
150    /// geometry at construction (the capture loop always opens its encoder at
151    /// native/source geometry before any adaptation decision exists), so the
152    /// first tick at the top rung is a no-op rather than a spurious rebuild.
153    pub fn new(source: (u32, u32)) -> Self {
154        let committed = (source.0.max(1), source.1.max(1));
155        Self::with_committed(source, committed)
156    }
157
158    /// Same, for an encoder that is already built BELOW `source` — the low
159    /// simulcast layer, whose encoder opens at whatever rung its first coalesced
160    /// cap bought rather than at capture geometry. Seeding `committed` at the
161    /// source instead would make the first tick a no-op for a full dwell and leave
162    /// the low encoder running at capture resolution for five seconds.
163    ///
164    /// `source` still bounds the ladder: a rung is fitted to the source's aspect
165    /// and never upscaled past it (see [`fit_encode_dims`] / [`fit_encode_area`]),
166    /// so this must be the CAPTURE geometry, not the rung's.
167    pub fn with_committed(source: (u32, u32), committed: (u32, u32)) -> Self {
168        let source = (source.0.max(1), source.1.max(1));
169        Self {
170            source,
171            committed: (even_min2(committed.0), even_min2(committed.1)),
172            committed_target: None,
173            pending: None,
174        }
175    }
176
177    /// Fit a target to the source with the function its ladder measures rungs in.
178    fn fit(&self, target: RungTarget) -> (u32, u32) {
179        match target {
180            RungTarget::Height(h) => fit_encode_dims(self.source, h),
181            RungTarget::Pixels(px) => fit_encode_area(self.source, px),
182        }
183    }
184
185    /// Advance the dwell state one tick. `target` is the rung this tick decided
186    /// (`None` = [`crate::Quality::AudioOnly`]); `now_us`/`dwell_us` mirror the
187    /// `CodecSwitch`/`LayerHysteresis` debounce contract elsewhere in this
188    /// codebase. Returns the committed `(width, height)` the encoder should be
189    /// built at; pair it with [`Self::committed_target`] for the rung half of a
190    /// rebuild key.
191    pub fn observe(
192        &mut self,
193        target: Option<RungTarget>,
194        now_us: i64,
195        dwell_us: i64,
196    ) -> (u32, u32) {
197        let Some(target) = target else {
198            return self.committed;
199        };
200        // Already there — nothing to debounce. Two cases: the same target as last
201        // time, or the very first tick after seeding, which adopts whatever target
202        // the caller seeded the geometry from. Anything else (including a DIFFERENT
203        // target that happens to fit to the same geometry) falls through to the
204        // dwell below: see this type's contract.
205        if self.fit(target) == self.committed && self.committed_target.is_none_or(|t| t == target) {
206            self.committed_target = Some(target);
207            self.pending = None;
208            return self.committed;
209        }
210        match self.pending {
211            Some((p, since)) if p == target && now_us.saturating_sub(since) >= dwell_us => {
212                self.committed = self.fit(target);
213                self.committed_target = Some(target);
214                self.pending = None;
215            }
216            Some((p, _)) if p == target => {}
217            _ => self.pending = Some((target, now_us)),
218        }
219        self.committed
220    }
221
222    /// The currently-committed encode geometry (what the encoder should already
223    /// be built at).
224    pub fn committed(&self) -> (u32, u32) {
225        self.committed
226    }
227
228    /// The rung [`Self::committed`] belongs to — the other half of the rebuild key
229    /// for a caller whose ladder can produce two rungs at one geometry (the screen
230    /// low layer; see this type's contract). `None` only before the first
231    /// [`Self::observe`] with a target.
232    pub fn committed_target(&self) -> Option<RungTarget> {
233        self.committed_target
234    }
235}

478    /// THE wobble contract for the collapsed-geometry case: a coalesced cap
479    /// oscillating across the 1.1 Mbps rung boundary every tick — the shape a
480    /// receiver set flapping between a grid cell and a stage tile produces — must
481    /// actuate NOTHING. Keyed on the fitted geometry this was invisible; keyed on
482    /// the target without a dwell it would rebuild + IDR the low encoder on every
483    /// captured frame.
484    #[test]
485    fn a_wobbling_cap_never_rebuilds_a_collapsed_screen_rung() {
486        use crate::layer::{SCREEN_LOW_RUNG, SCREEN_MID_RUNG};
487        let src = (400u32, 500u32);
488        let mut g =
489            GeometryDwell::with_committed(src, fit_encode_area(src, SCREEN_LOW_RUNG.pixels()));
490        let dwell = GEOMETRY_REBUILD_DWELL_US;
491        let mut now = 0i64;
492        // The activation tick, where the low encoder is BUILT rather than rebuilt.
493        g.observe(
494            Some(RungTarget::Pixels(SCREEN_LOW_RUNG.pixels())),
495            now,
496            dwell,
497        );
498        // The producer's rebuild key, exactly as `screen_tx` forms it.
499        let mut key = (g.committed(), g.committed_target());
500        let mut rebuilds = 0usize;
501        for i in 0..120 {
502            let rung = if i % 2 == 0 {
503                SCREEN_LOW_RUNG
504            } else {
505                SCREEN_MID_RUNG
506            };
507            let dims = g.observe(Some(RungTarget::Pixels(rung.pixels())), now, dwell);
508            if (dims, g.committed_target()) != key {
509                rebuilds += 1;
510                key = (dims, g.committed_target());
511            }
512            now += 100_000; // a 10 fps capture: 12 s of wobble
513        }
514        assert_eq!(
515            rebuilds, 0,
516            "a cap wobbling across the rung boundary must never actuate a rebuild"
517        );
518
519        // And the same cap held steady DOES actuate, exactly once per dwell.
520        let mid = Some(RungTarget::Pixels(SCREEN_MID_RUNG.pixels()));
521        for _ in 0..120 {
522            let dims = g.observe(mid, now, dwell);
523            if (dims, g.committed_target()) != key {
524                rebuilds += 1;
525                key = (dims, g.committed_target());
526            }
527            now += 100_000;
528        }
529        assert_eq!(rebuilds, 1, "one rebuild for one settled rung change");
530    }
