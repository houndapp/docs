1//! # Quality ladder + bitrate models
2//!
3//! [`QualityLadder`] maps an available bitrate to a discrete video [`Rung`] with
4//! hysteresis (**drop fast, climb slow**) and a manual ceiling, falling back to
5//! [`Quality::AudioOnly`] when even the lowest rung won't fit — audio is sacred,
6//! video is the release valve. The screen/camera bitrate models
7//! (`screen_bitrate_bps`/`camera_bitrate_bps`) and the [`ScreenQuality`] rung
8//! construction ([`screen_rungs`]/[`screen_rungs_for_source`]) live here too:
9//! the ladder's shape and its pricing are one calibration.
10
11/// A discrete video quality preset.
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub struct Rung {
14    pub bitrate_bps: u64,
15    pub width: u32,
16    pub height: u32,
17    pub fps: u32,
18}
19
20/// The selected quality for the current conditions.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum Quality {
23    Video(Rung),
24    /// Below the lowest video rung — drop video, keep audio.
25    AudioOnly,
26}
27
28/// Maps available bitrate to a [`Rung`] with hysteresis and a manual ceiling.
29#[derive(Debug)]
30pub struct QualityLadder {
31    /// Rungs sorted ascending by bitrate.
32    rungs: Vec<Rung>,
33    current: usize,
34    /// Whether the last [`Self::update`] decided [`Quality::AudioOnly`] (the
35    /// available bitrate fell below the lowest rung). Kept so [`Self::current`] is
36    /// a faithful projection of the last decision rather than silently reporting
37    /// `Video(rungs[0])` while video is actually dropped.
38    audio_only: bool,
39    /// Manual maximum rung index (inclusive); `None` = no cap.
40    ceiling: Option<usize>,
41    /// Headroom required to climb a rung (e.g. 1.15 = need 15% above its cost).
42    climb_margin: f64,
43    /// Consecutive [`Self::update`] calls that have cleared the resume threshold
44    /// while [`Quality::AudioOnly`]. Reset by any call that does not.
45    resume_streak: u32,
46    /// Doubling exponent on the resume dwell, raised each time a resume collapses
47    /// again before it settled. Bounded by [`RESUME_BACKOFF_MAX_SHIFT`].
48    resume_backoff_shift: u32,
49    /// Consecutive [`Self::update`] calls spent on video since the last resume.
50    video_streak: u32,
51}
52
53/// Headroom above `rungs[0]` required to RESUME video after
54/// [`Quality::AudioOnly`]. Dropping video is deliberately margin-free (drop fast),
55/// but resuming at the bare rung-0 bitrate puts the link *exactly* at the edge that
56/// just failed, so the first real frames re-collapse it — a stable on/off blink.
57/// Mirrors the rung-to-rung `climb_margin`: prove the link can carry the lowest
58/// rung *plus* headroom before putting traffic back on it.
59const RESUME_MARGIN: f64 = 1.15;
60
61/// How many consecutive [`QualityLadder::update`] calls must clear
62/// [`RESUME_MARGIN`] before video resumes. The estimate that drives a resume is
63/// built with no video traffic on the link (see
64/// [`crate::rate::RateController::on_feedback_idle`]), so it is the weakest evidence the
65/// controller ever acts on; requiring it to hold for a few ticks costs a
66/// second-scale delay on a genuine recovery and removes the single-tick flip.
67const RESUME_DWELL_TICKS: u32 = 3;
68
69/// Cap on the resume-dwell doubling. A link whose true capacity sits *below*
70/// `rungs[0]` cannot be fixed by a margin alone: every probe genuinely fits the
71/// estimate, puts real frames on the wire, and is knocked straight back down — a
72/// self-sustaining blink. Each resume that fails to settle doubles the dwell (3 →
73/// 6 → 12 → 24 calls), so the sender converges on "stay audio-only, probe rarely"
74/// instead of blinking, while a link that recovers for real still gets a probe
75/// within the capped window.
76const RESUME_BACKOFF_MAX_SHIFT: u32 = 3;
77
78/// Video ticks after which a resume counts as SETTLED — the link really could
79/// carry rung 0, so the next collapse starts a fresh episode at the base dwell
80/// rather than inheriting the old backoff.
81const RESUME_SETTLED_TICKS: u32 = 30;
82
83impl QualityLadder {
84    /// The default camera ladder: 180p15 → 360p20 → 360p30 → 540p30 → 720p30.
85    pub fn standard() -> Self {
86        Self::new(vec![
87            Rung {
88                bitrate_bps: 150_000,
89                width: 320,
90                height: 180,
91                fps: 15,
92            },
93            Rung {
94                bitrate_bps: 300_000,
95                width: 640,
96                height: 360,
97                fps: 20,
98            },
99            Rung {
100                bitrate_bps: 500_000,
101                width: 640,
102                height: 360,
103                fps: 30,
104            },
105            Rung {
106                bitrate_bps: 800_000,
107                width: 960,
108                height: 540,
109                fps: 30,
110            },
111            Rung {
112                bitrate_bps: 1_500_000,
113                width: 1280,
114                height: 720,
115                fps: 30,
116            },
117        ])
118    }
119
120    /// The default screen-share ladder — [`DegradePriority::Legibility`] capped at
121    /// 1080p60, i.e. `screen_for(ScreenQuality::default())`: 720p15 → 1080p15 →
122    /// 1080p30 → 1080p60 (climb resolution before fps; on the way down, drop fps
123    /// before resolution). Kept as a named constructor for callers that don't
124    /// expose the per-stream control; [`Self::screen_for`] is the configurable one.
125    pub fn screen() -> Self {
126        Self::screen_for(ScreenQuality::default())
127    }
128
129    /// Build a screen-share ladder for a per-stream [`ScreenQuality`] preference
130    /// (degrade priority + resolution/fps caps). See [`screen_rungs`] for the rung
131    /// construction; this just wraps the result. Pair with [`Self::reconfigure`]
132    /// to change the preference on a live stream without losing the bitrate
133    /// position.
134    pub fn screen_for(quality: ScreenQuality) -> Self {
135        Self::new(screen_rungs(quality))
136    }
137
138    /// Swap in a new rung set (e.g. from a mid-stream [`ScreenQuality`] change)
139    /// while preserving the *bitrate position*: the next [`Self::update`] re-picks
140    /// from the live available bitrate, so the only state to carry is "are we
141    /// paused (audio-only)". `current` is clamped into the new ladder; a stale
142    /// ceiling index is dropped (the new ladder's caps are baked into its rungs).
143    pub fn reconfigure(&mut self, rungs: Vec<Rung>) {
144        assert!(!rungs.is_empty(), "ladder needs at least one rung");
145        self.current = self.current.min(rungs.len() - 1);
146        self.ceiling = None;
147        self.rungs = rungs;
148        // The new ladder has its own rung-0 cost, so any resume progress — or
149        // backoff earned against the old floor — is meaningless. `video_streak`
150        // carries through: like `audio_only`, it describes the stream's current
151        // state, not the ladder's shape.
152        self.resume_streak = 0;
153        self.resume_backoff_shift = 0;
154    }
155
156    /// The BWE estimate at which the *top* rung becomes selectable: the top rung's
157    /// bitrate plus the [`Self::update`] climb-margin headroom. The rate controller
158    /// seeds its ceiling from this so it can actually climb to the richest rung on a
159    /// fast link. Without it a flat BWE cap below the top rung × margin would strand
160    /// the high rungs entirely — e.g. a 1440p120 top needs ~20 Mbps, so a 4 Mbps cap
161    /// would top out at 1080p15 and never offer the quality a gaming LAN can carry.
162    pub fn bwe_ceiling_bps(&self) -> u64 {
163        let top = self.rungs.last().map(|r| r.bitrate_bps).unwrap_or(0);
164        (top as f64 * self.climb_margin).ceil() as u64
165    }
166
167    /// `rungs` must be non-empty and sorted ascending by bitrate.
168    pub fn new(rungs: Vec<Rung>) -> Self {
169        assert!(!rungs.is_empty(), "ladder needs at least one rung");
170        Self {
171            rungs,
172            current: 0,
173            audio_only: false,
174            ceiling: None,
175            climb_margin: 1.15,
176            resume_streak: 0,
177            resume_backoff_shift: 0,
178            // A fresh ladder starts on video (see `current`), and that starting
179            // state is presumed settled — the first collapse of a call is a fresh
180            // episode, not a failed resume probe.
181            video_streak: RESUME_SETTLED_TICKS,
182        }
183    }
184
185    /// Set the manual ceiling by rung index (clamped). `None` removes the cap.
186    pub fn set_ceiling(&mut self, idx: Option<usize>) {
187        self.ceiling = idx.map(|i| i.min(self.rungs.len() - 1));
188    }
189
190    /// Pick the quality for the given available bitrate. Drops as many rungs as
191    /// needed immediately, but climbs at most one rung per call and only with
192    /// `climb_margin` headroom (and never above the ceiling).
193    ///
194    /// The [`Quality::AudioOnly`] boundary is **asymmetric**: video drops the
195    /// instant the estimate falls below `rungs[0]`, but resuming needs
196    /// [`RESUME_MARGIN`] of headroom sustained for [`RESUME_DWELL_TICKS`] calls —
197    /// doubled per failed resume (see [`RESUME_BACKOFF_MAX_SHIFT`]).
198    pub fn update(&mut self, available_bps: u64) -> Quality {
199        let max_idx = self
200            .ceiling
201            .unwrap_or(self.rungs.len() - 1)
202            .min(self.rungs.len() - 1);
203
204        if self.audio_only {
205            let resume_at = (self.rungs[0].bitrate_bps as f64 * RESUME_MARGIN).ceil() as u64;
206            if available_bps < resume_at {
207                self.resume_streak = 0;
208                self.current = 0;
209                return Quality::AudioOnly;
210            }
211            self.resume_streak = self.resume_streak.saturating_add(1);
212            if self.resume_streak < (RESUME_DWELL_TICKS << self.resume_backoff_shift) {
213                self.current = 0;
214                return Quality::AudioOnly;
215            }
216        } else if available_bps < self.rungs[0].bitrate_bps {
217            self.resume_backoff_shift = if self.video_streak >= RESUME_SETTLED_TICKS {
218                0 // video held for a long stretch — a fresh episode, not a blink
219            } else {
220                (self.resume_backoff_shift + 1).min(RESUME_BACKOFF_MAX_SHIFT)
221            };
222            self.video_streak = 0;
223            self.current = 0;
224            self.audio_only = true;
225            self.resume_streak = 0;
226            return Quality::AudioOnly;
227        }
228        self.audio_only = false;
229        self.resume_streak = 0;
230        self.video_streak = self.video_streak.saturating_add(1);
231        if self.current > max_idx {
232            self.current = max_idx;
233        }
234        // Drop fast: step down until the current rung fits.
235        while self.current > 0 && self.rungs[self.current].bitrate_bps > available_bps {
236            self.current -= 1;
237        }
238        // Climb slow: a single step, only with headroom and under the ceiling.
239        if self.current < max_idx {
240            let next = self.rungs[self.current + 1];
241            if available_bps as f64 >= next.bitrate_bps as f64 * self.climb_margin {
242                self.current += 1;
243            }
244        }
245        Quality::Video(self.rungs[self.current])
246    }
247
248    /// The quality from the last [`Self::update`] (a faithful projection: reports
249    /// [`Quality::AudioOnly`] if that was the last decision). Before any update,
250    /// reports the lowest video rung (the starting state).
251    pub fn current(&self) -> Quality {
252        if self.audio_only {
253            Quality::AudioOnly
254        } else {
255            Quality::Video(self.rungs[self.current])
256        }
257    }
258}

262/// Which dimension a screen-share ladder sacrifices **first** as bandwidth drops
263/// — a per-stream user choice, switchable on a live stream (not asked at stream
264/// start). Gameplay wants motion; an IDE/slides wants legibility.
265#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
266pub enum DegradePriority {
267    /// Hold frame rate, drop **resolution** first — smooth motion over sharpness
268    /// (gameplay, video). On the way up: climb fps at low res, then climb res.
269    Motion,
270    /// Hold resolution, drop **frame rate** first — readable detail over smooth
271    /// motion (text, IDEs, slides). On the way up: climb res, then climb fps.
272    #[default]
273    Legibility,
274}
275
276/// The user's maximum-resolution cap for a stream (the "Source / 1440p / 1080p /
277/// 720p / 480p" control). `Source` imposes no cap beyond the capture's own
278/// geometry — the session still clamps it to the encoder's native target.
279#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
280pub enum MaxResolution {
281    #[default]
282    Source,
283    P1440,
284    P1080,
285    P720,
286    P480,
287}

289impl MaxResolution {
290    /// Luma-height cap in pixels (`Source` ⇒ effectively unbounded; the caller
291    /// clamps to the real capture size).
292    pub fn height_cap(self) -> u32 {
293        match self {
294            MaxResolution::Source => u32::MAX,
295            MaxResolution::P1440 => 1440,
296            MaxResolution::P1080 => 1080,
297            MaxResolution::P720 => 720,
298            MaxResolution::P480 => 480,
299        }
300    }
301}

303/// One stream's video-quality preference, changeable mid-stream. Drives
304/// [`QualityLadder::screen_for`] / [`QualityLadder::reconfigure`]. The default
305/// is legibility-first, capped at 1080p60.
306#[derive(Debug, Clone, Copy, PartialEq, Eq)]
307pub struct ScreenQuality {
308    pub degrade: DegradePriority,
309    pub max_resolution: MaxResolution,
310    /// Frame-rate cap; the UI offers 30 / 60 / 120.
311    pub max_fps: u32,
312}

314impl Default for ScreenQuality {

321    }

324/// Candidate ladder heights (16:9 luma height), ascending.
325const LADDER_HEIGHTS: [u32; 4] = [480, 720, 1080, 1440];
326/// Candidate ladder frame rates, ascending.
327const LADDER_FPS: [u32; 4] = [15, 30, 60, 120];
328
329/// Make `v` even (encoders want even luma dimensions).
330const fn even(v: u32) -> u32 {
331    v & !1
332}
333
334/// The 16:9 luma width for a height (even). Descriptive: the encoder fits the
335/// *source* aspect, so this is the reference geometry the rung reports.
336const fn width_16x9(height: u32) -> u32 {
337    even(height * 16 / 9)
338}
339
340/// Target bitrate (bits/s) for one screen rung at `height`×`fps`. A single
341/// model — `pixels × fps^0.72 × k`, rounded to 100 kbps — instead of a hand-tuned
342/// table per (res, fps), staying monotonic in both axes (the ladder ordering relies
343/// on it).
344///
345/// **Calibrated for SCREEN content, not camera video.** Screen content (text, UI,
346/// large static regions, sharp intra-friendly edges) compresses several times better
347/// than motion video; a camera-video rate table (1080p30 → ~8 Mbps) oversizes a
348/// screenshare and saturates the shared connection, starving everything else. `k` is
349/// set so 1080p30 ≈ 2.4 Mbps — in line with production-RTC screenshare rates over
350/// codecs less efficient than H.265, and ample for crisp text. Rungs: 720p15≈0.6M,
351/// 1080p15≈1.5M, 1080p30≈2.4M, 1080p60≈4.0M. The BWE ceiling tracks the top rung,
352/// so a fast link reaches it.
353pub(crate) fn screen_bitrate_bps(height: u32, fps: u32) -> u64 {
354    // 0.10 solves 1080p30 → ~2.4 Mbps (screen-content target).
355    const K: f64 = 0.10;
356    let pixels = (width_16x9(height) as f64) * (height as f64);
357    round_rung_bps(pixels * (fps as f64).powf(FPS_EXP) * K)
358}
359
360/// How a rung's rate scales with frame rate. Sub-linear: halving the frame rate
361/// does not halve the bits, because the residual between consecutive frames grows
362/// as they get further apart. Shared by both content models.
363const FPS_EXP: f64 = 0.72;
364
365/// How a CAMERA rung's rate scales with pixel count. Sub-linear for the same
366/// reason the fps term is: a small picture needs proportionally *more* bits per
367/// pixel (the per-macroblock and header overheads amortise worse, and downscaling
368/// concentrates detail), which is why the industry's 180p thumbnails are priced at
369/// ~3× the bits-per-pixel of their 720p sources rather than 1/16th of the bitrate.
370/// Solved from the two anchors below — see [`camera_bitrate_bps`].
371const CAM_PIXEL_EXP: f64 = 0.70;
372
373/// Calibration constant for [`camera_bitrate_bps`], solved so the model reproduces
374/// the shipped SOURCE rung exactly: 1280×720 @ 30 fps → 1500 kbps (`CAM_BITRATE`
375/// in `crates/app/client/src/call/mod.rs`). The whole camera ladder is therefore
376/// anchored on the one rate the client already publishes at.
377const CAM_BPP_K: f64 = 8.6567;
378
379/// Target bitrate (bits/s) for one CAMERA rung at `height`×`fps`, from a
380/// bits-per-pixel model — the camera-content twin of [`screen_bitrate_bps`].
381///
382/// **Calibrated for MOTION VIDEO, and sub-linear in pixels.** A linear-in-pixels
383/// model anchored at 720p30 would price a 320×180 thumbnail at ~57 kbps, well
384/// under what a legible talking head costs; the `pixels^0.70` term instead yields
385/// ~130 kbps at 320×180@15 and ~420 kbps at 640×360@20, which is where three
386/// independent shipped ladders sit:
387///
388/// * LiveKit's presets: `h180` = 320×180 @ **160 kbps**, `h360` = 640×360 @
389///   **500 kbps**, with simulcast defaulting to `h180 + h360 + primary`.
390/// * Daily's published defaults: **80 kbps** / 10 fps / 320×180, **200 kbps** /
391///   15 fps / 640×360, 680 kbps / 30 fps / 1280×720.
392/// * Jitsi's bridge: 180p thumbnails, 360p on-stage.
393///
394/// The same numbers show up as published per-tile budgets: Zoom quotes 2.0 Mbps for
395/// a 25-tile gallery and 4.0 Mbps for 49 (~80 kbps/tile), and LiveKit reports a
396/// 25-person meeting at 4 Mbps (~160 kbps/tile).
397///
398/// Rounded to 10 kbps rather than [`round_rung_bps`]'s 100 kbps: the thumbnail
399/// rung is only ~130 kbps, where a 100 kbps quantum is a 25% distortion.
400pub(crate) fn camera_bitrate_bps(height: u32, fps: u32) -> u64 {
401    let pixels = (width_16x9(height) as f64) * (height as f64);
402    round_bps(
403        pixels.powf(CAM_PIXEL_EXP) * (fps as f64).powf(FPS_EXP) * CAM_BPP_K,
404        10_000,
405    )
406}
407
408/// Round a modelled rung rate to 100 kbps, never below the 100 kbps floor (the
409/// lowest rate worth publishing video at).
410fn round_rung_bps(raw: f64) -> u64 {
411    round_bps(raw, 100_000)
412}
413
414/// Round a modelled rate to `quantum`, never below the 100 kbps floor (the lowest
415/// rate worth publishing video at, and `RateConfig::default().min_bps`).
416fn round_bps(raw: f64, quantum: u64) -> u64 {
417    let q = quantum as f64;
418    let rounded = (raw.max(0.0) / q).round() * q;
419    (rounded as u64).max(100_000)
420}

422/// Build the screen-share rungs for a [`ScreenQuality`], **sans-IO and pure**.
423///
424/// Candidate heights/fps are filtered to the caps, then assembled into one
425/// ascending-bitrate ladder whose *order of climb* encodes the degrade priority:
426/// * [`DegradePriority::Legibility`] climbs resolution at the floor fps, then fps
427///   at the cap resolution — so descending drops fps before resolution. Floor
428///   resolution is 720p (below that text is unreadable) unless the cap is lower.
429/// * [`DegradePriority::Motion`] climbs fps at the floor resolution, then
430///   resolution at the cap fps — so descending drops resolution before fps. Floor
431///   fps is 30 (below that it isn't "motion") unless the cap is lower; floor
432///   resolution is 480p.
433///
434/// The result is sorted ascending by bitrate (the ladder invariant) and
435/// deduplicated, so callers get a clean monotonic ladder for any cap combination.
436pub fn screen_rungs(quality: ScreenQuality) -> Vec<Rung> {
437    let cap_h = quality.max_resolution.height_cap();
438    let (floor_h, floor_fps) = match quality.degrade {
439        DegradePriority::Legibility => (720, 15),
440        DegradePriority::Motion => (480, 30),
441    };
442
443    // Heights ≤ cap and ≥ the priority's floor — but always keep the cap itself,
444    // so a cap below the floor (e.g. legibility @ 480p) still yields one height.
445    let heights: Vec<u32> = LADDER_HEIGHTS
446        .iter()
447        .copied()
448        .filter(|h| *h <= cap_h && (*h >= floor_h || *h >= cap_h))
449        .collect();
450    let heights = if heights.is_empty() {
451        vec![LADDER_HEIGHTS[0].min(cap_h)]
452    } else {
453        heights
454    };
455    // Frame rates ≤ cap and ≥ the priority's floor — always keep at least one.
456    let fpses: Vec<u32> = LADDER_FPS
457        .iter()
458        .copied()
459        .filter(|f| *f <= quality.max_fps && *f >= floor_fps)
460        .collect();
461    let fpses = if fpses.is_empty() {
462        vec![floor_fps.min(quality.max_fps)]
463    } else {
464        fpses
465    };
466
467    let max_h = *heights.last().unwrap();
468    let min_h = *heights.first().unwrap();
469    let max_f = *fpses.last().unwrap();
470    let min_f = *fpses.first().unwrap();
471
472    // (height, fps) levels in climb order per priority.
473    let mut levels: Vec<(u32, u32)> = Vec::new();
474    match quality.degrade {
475        DegradePriority::Legibility => {
476            for &h in &heights {
477                levels.push((h, min_f));
478            }
479            for &f in fpses.iter().skip(1) {
480                levels.push((max_h, f));
481            }
482        }
483        DegradePriority::Motion => {
484            for &f in &fpses {
485                levels.push((min_h, f));
486            }
487            for &h in heights.iter().skip(1) {
488                levels.push((h, max_f));
489            }
490        }
491    }
492
493    let mut rungs: Vec<Rung> = levels
494        .into_iter()
495        .map(|(h, f)| Rung {
496            bitrate_bps: screen_bitrate_bps(h, f),
497            width: width_16x9(h),
498            height: h,
499            fps: f,
500        })
501        .collect();
502    rungs.sort_by_key(|r| r.bitrate_bps);
503    // Bitrate is a pure function of (height, fps), so equal bitrate ⇒ the same
504    // bandwidth target; dedup on it to keep the ladder STRICTLY ascending — the
505    // `QualityLadder` climb/drop hysteresis compares bitrates and `new` documents
506    // an ascending ladder.
507    rungs.dedup_by_key(|r| r.bitrate_bps);
508    rungs
509}

511/// [`screen_rungs`] re-priced for the share's **real capture geometry**.
512///
513/// The rung table is built from nominal 16:9 candidate heights because the bitrate
514/// model needs a reference geometry — but the encoder never upscales past the
515/// source and always follows the source's own aspect ([`crate::geometry`]'s
516/// `fit_encode_dims`). Left unfitted, a 900×500 window share still offers a
517/// "1080p60 = 4 Mbps" top rung: the AIMD climbs to it and hands 4 Mbps to an
518/// encoder pinned at 900×500 — roughly 8× the bits that geometry needs, spent on
519/// the QUIC connection audio shares (and paid again as receive-side packet CPU).
520///
521/// So each rung is re-priced by the pixels it will ACTUALLY encode
522/// (`fit_encode_dims(source, rung.height)`) and its reported geometry becomes that
523/// encode geometry. Rungs that collapse onto the same fitted geometry+fps produce
524/// the same rate and are deduplicated, so a source smaller than the ladder's floor
525/// height simply yields a shorter ladder — the correct outcome, since those rungs
526/// were never distinguishable at that capture size.
527pub fn screen_rungs_for_source(quality: ScreenQuality, source: (u32, u32)) -> Vec<Rung> {
528    let mut rungs: Vec<Rung> = screen_rungs(quality)
529        .into_iter()
530        .map(|r| {
531            let (w, h) = crate::geometry::fit_encode_dims(source, r.height);
532            let nominal = (r.width as f64) * (r.height as f64);
533            let scaled = if nominal > 0.0 {
534                r.bitrate_bps as f64 * ((w as f64) * (h as f64) / nominal)
535            } else {
536                r.bitrate_bps as f64
537            };
538            Rung {
539                bitrate_bps: round_rung_bps(scaled),
540                width: w,
541                height: h,
542                fps: r.fps,
543            }
544        })
545        .collect();
546    // Re-sort: re-pricing is monotonic in pixels but the clamp can reorder a
547    // low-res/high-fps rung against a high-res/low-fps one. Dedup keeps the ladder
548    // STRICTLY ascending, the `QualityLadder::new` precondition.
549    rungs.sort_by_key(|r| r.bitrate_bps);
550    rungs.dedup_by_key(|r| r.bitrate_bps);
551    rungs
552}

837    /// are pinned here, because every large-call bandwidth estimate multiplies
838    /// these numbers.

841        // The anchor: `CAM_BITRATE` (1500 kbps @ 720p30).
842        assert_eq!(camera_bitrate_bps(720, 30), 1_500_000);
843        // Independent confirmation the *shape* is right, not just the anchor:
844        // solved only from 720p30, the model reproduces LiveKit's published
845        // `h180` preset — 320×180 @ 20 fps @ 160 kbps — to the kilobit.
846        assert_eq!(camera_bitrate_bps(180, 20), 160_000);
847        // Thumbnail: LiveKit `h180` = 160 kbps, Daily = 80 kbps @ 10 fps, Jitsi
848        // 180p. 90–160 kbps is the span those three sit in.
849        let thumb = camera_bitrate_bps(180, 15);
850        assert!(
851            (90_000..=160_000).contains(&thumb),

853        );
854        // Low: 640×360@20. LiveKit's `h360` is 500 kbps at 30 fps; Daily ships
855        // 200 kbps at 15. 300–450 kbps is the span at 20 fps.
856        let low = camera_bitrate_bps(360, 20);

858            (300_000..=450_000).contains(&low),
859            "360p20 priced at {low} bps, outside the 300–450 kbps shipped-ladder band"

1072            for i in 1..1000 {
1073                if matches!(l.update(200_000), Quality::Video(_)) {
1074                    return i;
1075                }
1076            }
1077            panic!("never resumed");
1078        }

1088            "first probe uses the base dwell"

1091            assert!(
1092                w[1] >= w[0],
1093                "each failed resume must not shorten the next dwell: {seen:?}"
1094            );
1095        }
1096        let cap = RESUME_DWELL_TICKS << RESUME_BACKOFF_MAX_SHIFT;
1097        assert_eq!(
1098            *seen.last().unwrap(),
1099            cap,
1100            "the backoff saturates rather than growing without bound: {seen:?}"

1144    fn screen_ladder_fitted_to_a_small_window_is_priced_for_it() {
1145        let q = ScreenQuality::default(); // legibility, 1080p cap, 60 fps cap
1146        let nominal = QualityLadder::new(screen_rungs(q));
1147        let fitted = QualityLadder::new(screen_rungs_for_source(q, (900, 500)));
1148
1149        for r in &fitted.rungs {
1150            assert!(
1151                r.height <= 500,
1152                "no rung may exceed the source height: {r:?}"
1153            );
1154            assert!(r.width <= 900, "no rung may exceed the source width: {r:?}");
1155        }
1156        // 900×500 is 0.217× the pixels of 1920×1080 — the ceiling must follow.
1157        assert!(
1158            fitted.bwe_ceiling_bps() * 3 < nominal.bwe_ceiling_bps(),
1159            "fitted ceiling {} must be far below the nominal {}",
1160            fitted.bwe_ceiling_bps(),
1161            nominal.bwe_ceiling_bps()
1162        );
1163        // Still a usable, strictly-ascending ladder (the `new` precondition).
1164        assert!(!fitted.rungs.is_empty());
1165        for w in fitted.rungs.windows(2) {
1166            assert!(w[1].bitrate_bps > w[0].bitrate_bps, "rungs must ascend");
1167        }
1168        // The heights all collapsed onto the source, so only the fps axis survives.
1169        assert_eq!(
1170            fitted.rungs.len(),
1171            3,
1172            "500p at 15/30/60 fps — the resolution rungs are indistinguishable here"
1173        );
1174    }

1176    /// A non-16:9 source gets a proportionate budget: the rung model prices
1177    /// `width_16x9(h) × h`, but a 1052×1080 window encodes ~55% of those pixels for
