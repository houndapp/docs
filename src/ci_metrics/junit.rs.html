1//! nextest's JUnit report → the `ci_e2e_tests` counts.
2//!
3//! This replaces the grep/sed block that lived in ci.yaml's e2e-call metrics
4//! step, and reproduces its arithmetic exactly — the series must stay
5//! continuous with what has already been stored:
6//!
7//! * `tests`/`failures`/`errors` come from the root `<testsuites>` attributes
8//!   (each missing attribute is 0);
9//! * `skipped` is counted from `<skipped` children, because nextest's root tag
10//!   carries no summary `skipped` attribute;
11//! * `failed = failures + errors`, `passed = tests - failed - skipped`, floored
12//!   at 0.
13
14use std::fs;
15use std::path::Path;
16
17use quick_xml::events::{BytesStart, Event};
18use quick_xml::reader::Reader;
19
20pub struct Counts {
21    pub passed: u64,
22    pub failed: u64,
23    pub skipped: u64,
24}
25
26/// Counts from a report on disk, or `None` (with a note on stderr) if it is
27/// missing or unreadable as XML — the caller then emits no `ci_e2e_tests`
28/// series at all, matching the old step's `TESTS_PRESENT` gate.
29pub fn counts_at(path: &Path) -> Option<Counts> {
30    let xml = match fs::read_to_string(path) {
31        Ok(xml) => xml,
32        Err(err) => {
33            eprintln!(
34                "ci-metrics: no junit report at {} ({err}) — skipping ci_e2e_tests",
35                path.display()
36            );
37            return None;
38        }
39    };
40    match counts(&xml) {
41        Ok(counts) => Some(counts),
42        Err(err) => {
43            eprintln!(
44                "ci-metrics: unparseable junit report at {} ({err}) — skipping ci_e2e_tests",
45                path.display()
46            );
47            None
48        }
49    }
50}
51
52fn counts(xml: &str) -> Result<Counts, quick_xml::Error> {
53    let mut reader = Reader::from_str(xml);
54    let (mut tests, mut failures, mut errors, mut skipped) = (0, 0, 0, 0);
55    let mut seen_root = false;
56
57    loop {
58        match reader.read_event()? {
59            Event::Eof => break,
60            // Empty (`<skipped/>`) and Start (`<skipped>…</skipped>`) are the
61            // same element as far as counting goes.
62            Event::Start(tag) | Event::Empty(tag) => match tag.name().as_ref() {
63                b"testsuites" if !seen_root => {
64                    seen_root = true;
65                    tests = attr_u64(&tag, b"tests");
66                    failures = attr_u64(&tag, b"failures");
67                    errors = attr_u64(&tag, b"errors");
68                }
69                b"skipped" => skipped += 1,
70                _ => {}
71            },
72            _ => {}
73        }
74    }
75
76    let failed = failures + errors;
77    Ok(Counts {
78        passed: tests.saturating_sub(failed).saturating_sub(skipped),
79        failed,
80        skipped,
81    })
82}
83
84/// A numeric attribute, or 0 — a report that omits or mangles one is still
85/// worth the counts it does carry.
86fn attr_u64(tag: &BytesStart<'_>, key: &[u8]) -> u64 {
87    tag.attributes()
88        .flatten()
89        .find(|attr| attr.key.as_ref() == key)
90        .and_then(|attr| String::from_utf8(attr.value.into_owned()).ok())
91        .and_then(|value| value.trim().parse().ok())
92        .unwrap_or(0)
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    /// The shape nextest actually writes (see target/nextest/e2e/junit.xml):
100    /// a root summary plus one `<testsuite>` per binary.
101    fn report(tests: u32, failures: u32, errors: u32, skipped_cases: &[&str]) -> String {
102        let cases: String = skipped_cases
103            .iter()
104            .map(|name| {
105                format!(
106                    "<testcase name=\"{name}\" classname=\"c\" time=\"0.1\">\n<skipped message=\"filtered\"/>\n</testcase>\n"
107                )
108            })
109            .collect();
110        format!(
111            "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n\
112             <testsuites name=\"nextest-run\" tests=\"{tests}\" failures=\"{failures}\" errors=\"{errors}\" uuid=\"x\" time=\"0.9\">\n\
113             <testsuite name=\"s\" tests=\"{tests}\" disabled=\"0\" errors=\"{errors}\" failures=\"{failures}\">\n\
114             <testcase name=\"ran\" classname=\"c\" time=\"0.3\">\n</testcase>\n\
115             {cases}</testsuite>\n</testsuites>\n"
116        )
117    }
118
119    #[test]
120    fn a_clean_run_is_all_passed() {
121        let counts = counts(&report(5, 0, 0, &[])).expect("parse");
122        assert_eq!((counts.passed, counts.failed, counts.skipped), (5, 0, 0));
123    }
124
125    #[test]
126    fn failures_and_errors_both_count_as_failed() {
127        let counts = counts(&report(5, 1, 2, &[])).expect("parse");
128        assert_eq!((counts.passed, counts.failed, counts.skipped), (2, 3, 0));
129    }
130
131    #[test]
132    fn skipped_comes_from_the_testcase_children() {
133        let counts = counts(&report(5, 0, 0, &["a", "b"])).expect("parse");
134        assert_eq!((counts.passed, counts.failed, counts.skipped), (3, 0, 2));
135    }
136
137    /// The bash floored `passed` at 0; a report whose attributes disagree with
138    /// its children must not produce a negative (and here, wrapping) count.
139    #[test]
140    fn passed_never_goes_negative() {
141        let counts = counts(&report(1, 1, 1, &["a"])).expect("parse");
142        assert_eq!((counts.passed, counts.failed, counts.skipped), (0, 2, 1));
143    }
144
145    #[test]
146    fn missing_attributes_are_zero() {
147        let counts = counts("<testsuites><testsuite/></testsuites>").expect("parse");
148        assert_eq!((counts.passed, counts.failed, counts.skipped), (0, 0, 0));
149    }
150
151    /// A report cut off mid-write still reports what its root tag carried —
152    /// the same tolerance the grep/sed version had (it only ever looked at the
153    /// `<testsuites` line and the `<skipped` lines).
154    #[test]
155    fn a_truncated_report_still_yields_the_root_counts() {
156        let counts = counts("<testsuites tests=\"5\" failures=\"1\"><testsuite>").expect("parse");
157        assert_eq!((counts.passed, counts.failed, counts.skipped), (4, 1, 0));
158    }
159
160    /// Markup that contradicts itself is an error, and the caller then emits
161    /// no `ci_e2e_tests` series rather than a wrong one.
162    #[test]
163    fn mismatched_markup_is_an_error_not_a_guess() {
164        assert!(counts("<testsuites tests=\"5\"></nope>").is_err());
165    }
166
167    #[test]
168    fn a_missing_file_is_no_counts() {
169        assert!(counts_at(Path::new("/nonexistent/junit.xml")).is_none());
170    }
171}
