1//! `ci-metrics` — what a CI job left on disk, as prometheus exposition.
2//!
3//! Every metrics step in `.github/workflows/` is now one command:
4//!
5//! ```sh
6//! cargo run -q -p ci-metrics -- <subcommand> --push
7//! ```
8//!
9//! Three subcommands, one per producer:
10//!
11//! * `bench` — criterion's own on-disk state (`target/criterion/<bench
12//!   id>/new/estimates.json`, one directory per `bench_function`; the suite
13//!   uses no groups, so ids are flat) becomes one `ci_bench_ns_per_iter`
14//!   series per benchmark. The JSON is read rather than the bencher-format
15//!   stdout because it is what criterion itself reads back for baseline
16//!   comparison, and it survives whatever else the build printed. The figure
17//!   is the **median** point estimate — the same statistic the bencher format
18//!   prints.
19//! * `job` — the job-level trio on its own, plus (with `--junit`) the
20//!   `ci_e2e_tests` counts read out of nextest's JUnit report. This is also
21//!   what the plain trio-only steps (nightly build/smoke) call.
22//! * `visual` — the trio plus the `ci_visual_snapshots` counts from snapdiff's
23//!   run summary.
24//!
25//! Every run ends with the trio the dashboards key off — `ci_job_success`,
26//! `ci_job_duration_seconds`, `ci_run_info` — whose values come from the
27//! environment the workflow step exports (WORKFLOW, JOB, PLATFORM, RUN_ID,
28//! SHA, JOB_SUCCESS, JOB_DURATION).
29//!
30//! The lines always go to stdout, so the job log keeps the record of exactly
31//! what was reported; `--push` additionally POSTs them to the backend's ingest
32//! route (see [`push`]). This tool absorbed both halves of what used to be a
33//! producer script piped into `.github/scripts/push-ci-metrics.sh`, which the
34//! runners' missing jq/python3 had forced into unreviewable grep/sed.
35//!
36//! **Report-only by construction.** A CI job must never fail because its
37//! metrics couldn't be produced, so once the arguments parse, every path exits
38//! 0: a missing criterion directory, an unreadable `estimates.json`, an absent
39//! or malformed `junit.xml`, an unset token, a refused connection, and a value
40//! that can't be a prometheus label are each a stderr warning and a dropped
41//! series, never an abort. (Bad *arguments* still exit non-zero via clap —
42//! that is a broken workflow step, not a broken CI run, and it should be loud.)
43

55#[derive(Parser, Debug)]
56#[command(
57    name = "ci-metrics",
58    about = "CI job results as prometheus text-exposition lines"
59)]
60struct Args {
61    #[command(subcommand)]
62    command: Command,
63}
64
65#[derive(Subcommand, Debug)]
66enum Command {
67    /// Criterion benchmark medians plus the job-level trio.
68    Bench {
69        /// Criterion's results directory. Defaults to `$CARGO_TARGET_DIR` (or
70        /// the workspace `target/`) joined with `criterion`.
71        #[arg(long, value_name = "PATH")]
72        criterion_dir: Option<PathBuf>,
73        #[command(flatten)]
74        push: PushArgs,
75    },
76    /// The job-level trio, optionally with nextest's test counts.
77    Job {
78        /// A nextest JUnit report to read `ci_e2e_tests` counts from.
79        #[arg(long, value_name = "PATH")]
80        junit: Option<PathBuf>,
81        #[command(flatten)]
82        push: PushArgs,
83    },
84    /// snapdiff's snapshot counts plus the job-level trio.
85    Visual {
86        /// snapdiff's run summary (`snapdiff-summary.json`).
87        #[arg(long, value_name = "PATH")]
88        summary: Option<PathBuf>,
89        #[command(flatten)]
90        push: PushArgs,
91    },
92}
93
94#[derive(ClapArgs, Debug)]
95struct PushArgs {
96    /// Also POST the lines to the metrics ingest route (needs CI_METRICS_TOKEN).
97    #[arg(long)]
98    push: bool,
99
100    /// Ingest route to POST to. Only tests should need this.
101    #[arg(long, value_name = "URL", default_value = push::DEFAULT_URL)]
102    url: String,
103}
104
105fn main() {
106    let meta = Meta::from_env();
107    let (body, push) = match Args::parse().command {
108        Command::Bench {
109            criterion_dir,
110            push,
111        } => (render_bench(criterion_dir, &meta), push),
112        Command::Job { junit, push } => (render_job(junit.as_deref(), &meta), push),
113        Command::Visual { summary, push } => (render_visual(summary.as_deref(), &meta), push),
114    };
115
116    // A full disk or a closed pipe is not this tool's problem: report what we
117    // can and leave.
118    let _ = std::io::stdout().lock().write_all(body.as_bytes());
119
120    if push.push {
121        // The token comes from the environment and goes straight into a
122        // header — it is never logged, and never reaches a command line.
123        push::push(&push.url, std::env::var("CI_METRICS_TOKEN").ok(), &body);
124    }
125}

127fn render_bench(dir: Option<PathBuf>, meta: &Meta) -> String {
128    let labels = &meta.labels;
129    let medians = criterion::medians(&criterion::results_dir(dir));
130
131    let mut out = String::new();
132    for (id, ns) in &medians {
133        let _ = writeln!(out, "ci_bench_ns_per_iter{{{labels},bench=\"{id}\"}} {ns}");
134    }
135    // Emitted even at 0, which is the point: criterion changing its on-disk
136    // layout would otherwise look exactly like a green run with no benches.
137    // `ci_bench_series_emitted == 0 && ci_job_success == 1` is the alert.
138    let _ = writeln!(out, "ci_bench_series_emitted{{{labels}}} {}", medians.len());
139    meta.write_trio(&mut out);
140    out
141}
142
143fn render_job(junit: Option<&Path>, meta: &Meta) -> String {
144    let mut out = String::new();
145    meta.write_trio(&mut out);
146
147    if let Some(counts) = junit.and_then(junit::counts_at) {
148        // NOTE: `result` is this series' ONLY label — no workflow/job/platform.
149        // That is what the e2e-call step has always written; labelling it now
150        // would fork the series rather than extend it.
151        let _ = writeln!(out, "ci_e2e_tests{{result=\"passed\"}} {}", counts.passed);
152        let _ = writeln!(out, "ci_e2e_tests{{result=\"failed\"}} {}", counts.failed);
153        let _ = writeln!(out, "ci_e2e_tests{{result=\"skipped\"}} {}", counts.skipped);
154    }
155    out
156}
157
158fn render_visual(summary: Option<&Path>, meta: &Meta) -> String {
159    let mut out = String::new();
160    meta.write_trio(&mut out);
161
162    if let Some(counts) = summary.and_then(visual::counts_at) {
163        // Like ci_e2e_tests, this series predates the tool: its labels are
164        // `platform` and `state` only, never the full job label set.
165        for (state, count) in visual::STATES.iter().zip(counts) {
166            let _ = writeln!(
167                out,
168                "ci_visual_snapshots{{platform=\"{}\",state=\"{state}\"}} {count}",
169                meta.platform
170            );
171        }
172    }
173    out
174}

176/// The label set and job-level values, resolved once from the environment the
177/// workflow step exports.
178struct Meta {
179    /// `workflow="…",job="…",platform="…"` — the prefix every series carries,
180    /// in the order the workflows' metrics steps have always written it.
181    labels: String,
182    /// The platform on its own, for the series that carry only that label.
183    platform: String,
184    success: u64,
185    duration: u64,
186    run_id: String,
187    sha: String,
188}
189
190impl Meta {
191    fn from_env() -> Self {
192        Self::from_lookup(|key| std::env::var(key).ok())
193    }
194
195    /// The environment as a plain lookup, so the defaulting rules can be
196    /// tested without mutating process state (`set_var` is `unsafe` in edition
197    /// 2024, and tests share one process).
198    fn from_lookup(get: impl Fn(&str) -> Option<String>) -> Self {
199        let label = |key: &str, default: &str| {
200            get(key)
201                .filter(|value| !value.is_empty())
202                .and_then(|value| label_value(&value))
203                .unwrap_or_else(|| default.to_owned())
204        };
205        // Unset/empty/garbage is 0, matching the steps' `${VAR:-0}`.
206        let num = |key: &str| {
207            get(key)
208                .and_then(|value| value.trim().parse().ok())
209                .unwrap_or_default()
210        };
211        let platform = label("PLATFORM", "unknown");
212        Self {
213            labels: format!(
214                "workflow=\"{}\",job=\"{}\",platform=\"{platform}\"",
215                label("WORKFLOW", "unknown"),
216                label("JOB", "unknown"),
217            ),
218            platform,
219            success: num("JOB_SUCCESS"),
220            duration: num("JOB_DURATION"),
221            run_id: label("RUN_ID", ""),
222            sha: label("SHA", ""),
223        }
224    }
225
226    fn write_trio(&self, out: &mut String) {
227        let labels = &self.labels;
228        let _ = writeln!(out, "ci_job_success{{{labels}}} {}", self.success);
229        let _ = writeln!(out, "ci_job_duration_seconds{{{labels}}} {}", self.duration);
230        let _ = writeln!(
231            out,
232            "ci_run_info{{{labels},run_id=\"{}\",sha=\"{}\"}} 1",
233            self.run_id, self.sha
234        );
235    }
236}

238/// Escape a string for a prometheus label value, or reject it outright.
239///
240/// Bench ids are `[a-z0-9_]` by construction and the job labels come from the
241/// workflow, but both read as untrusted here: a stray directory name must not
242/// be able to forge label syntax, and a control character would make the
243/// exposition unparseable at the far end.
244fn label_value(raw: &str) -> Option<String> {
245    if raw.is_empty() || raw.chars().any(char::is_control) {
246        return None;
247    }
248    Some(raw.replace('\\', "\\\\").replace('"', "\\\""))
249}

257    fn meta(env: &[(&str, &str)]) -> Meta {
258        let env: HashMap<String, String> = env
259            .iter()
260            .map(|(key, value)| ((*key).to_owned(), (*value).to_owned()))
261            .collect();
262        Meta::from_lookup(|key| env.get(key).cloned())
263    }

265    fn e2e_meta() -> Meta {
266        meta(&[
267            ("WORKFLOW", "ci"),
268            ("JOB", "e2e-call"),
269            ("PLATFORM", "linux-x64"),
270            ("RUN_ID", "123"),
271            ("SHA", "deadbeef"),
272            ("JOB_SUCCESS", "1"),
273            ("JOB_DURATION", "421"),
274        ])
275    }

277    fn trio(meta: &Meta) -> String {
278        let mut out = String::new();
279        meta.write_trio(&mut out);
280        out
