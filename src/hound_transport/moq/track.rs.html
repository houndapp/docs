1//! The deduped per-track publish/subscribe cores, and the multiplexed-session
2//! track handles built directly on them.
3//!
4//! Every publisher-shaped type in this crate ([`crate::MoqPublisher`],
5//! [`MoqTrackPublisher`]) is one [`TrackPublisher`] core — an announced
6//! broadcast plus an open track mapped onto the GOP group/object model — and
7//! every subscriber-shaped type ([`crate::MoqSubscriber`],
8//! [`crate::MoqRelaySubscriber`], [`MoqTrackSubscriber`]) is one
9//! [`TrackSubscriber`] core sharing the [`drain_next_frame`] discipline. The
10//! wrappers differ only in how the track was obtained and in whether they own
11//! their session.
12
13use hound_media::Frame;
14use moq_net::{BroadcastProducer, TrackConsumer, TrackProducer};
15
16use super::MoqError;
17use super::gop::{
18    VideoPull, drain_next_frame, drain_next_frame_stalled, gop_finish, gop_publish_audio_frame,
19    gop_publish_frame,
20};
21
22/// The shared publisher core: one announced broadcast + one open track, mapped
23/// onto the GOP group/object model ([`gop_publish_frame`] /
24/// [`gop_publish_audio_frame`]). Holds its [`BroadcastProducer`] so the announce
25/// persists until drop.
26pub(crate) struct TrackPublisher {
27    _broadcast: BroadcastProducer,
28    track: TrackProducer,
29    /// The currently open group (a GOP for video, a rolled audio window for audio).
30    /// `None` until the first frame opens one.
31    open_group: Option<moq_net::GroupProducer>,
32    /// Sequence of the NEXT group to open (advanced per group roll).
33    next_group_seq: u64,
34    /// Objects written to the open group — drives the audio count-roll
35    /// ([`gop_publish_audio_frame`]). Unused by the video keyframe-driven path.
36    objects_in_group: usize,
37}
38
39impl TrackPublisher {
40    pub(crate) fn new(broadcast: BroadcastProducer, track: TrackProducer) -> Self {
41        Self {
42            _broadcast: broadcast,
43            track,
44            open_group: None,
45            next_group_seq: 0,
46            objects_in_group: 0,
47        }
48    }
49
50    /// Block until at least one subscriber has attached to the track
51    /// (`TrackProducer::used` — the moq-net live-join gate; see the module docs
52    /// on [`crate::moq`]). Deterministic — no `sleep`/yield.
53    pub(crate) async fn wait_subscribed(&self) -> Result<(), MoqError> {
54        self.track
55            .used()
56            .await
57            .map_err(|e| MoqError::Moq(format!("wait_subscribed: {e}")))
58    }
59
60    /// [`Self::wait_subscribed`] as an OWNED future, detached from `&self`, so a
61    /// publish loop can await the subscriber signal concurrently with publishing
62    /// (e.g. one `select!` arm alongside frame forwarding). Resolves `Err` when
63    /// the track closes instead.
64    pub(crate) fn subscribed(
65        &self,
66    ) -> impl std::future::Future<Output = Result<(), MoqError>> + Send + 'static {
67        let track = self.track.clone();
68        async move {
69            track
70                .used()
71                .await
72                .map_err(|e| MoqError::Moq(format!("wait_subscribed: {e}")))
73        }
74    }
75
76    /// Publish one encoded [`Frame`] (keyframe rolls the GOP group; a delta
77    /// appends to the open group) — [`gop_publish_frame`].
78    pub(crate) fn publish_frame(&mut self, frame: &Frame) -> Result<(), MoqError> {
79        gop_publish_frame(
80            &mut self.track,
81            &mut self.open_group,
82            &mut self.next_group_seq,
83            frame,
84        )
85    }
86
87    /// Publish one AUDIO frame as an object in a count-rolled ordered group —
88    /// [`gop_publish_audio_frame`], the ordered-substream model.
89    pub(crate) fn publish_audio_frame(&mut self, frame: &Frame) -> Result<(), MoqError> {
90        gop_publish_audio_frame(
91            &mut self.track,
92            &mut self.open_group,
93            &mut self.next_group_seq,
94            &mut self.objects_in_group,
95            frame,
96        )
97    }
98
99    /// Finish the open group and the track so the subscriber's drain ends
100    /// cleanly ([`gop_finish`]).
101    pub(crate) fn finish(&mut self) -> Result<(), MoqError> {
102        gop_finish(&mut self.track, &mut self.open_group)
103    }
104}

106/// The shared subscriber core: drains groups/objects off one [`TrackConsumer`]
107/// and reassembles [`Frame`]s via [`drain_next_frame`] /
108/// [`drain_next_frame_stalled`].
109pub(crate) struct TrackSubscriber {
110    track: TrackConsumer,
111    /// The group (GOP) currently being drained object-by-object, if any.
112    current: Option<moq_net::GroupConsumer>,
113    /// Stamped onto every reassembled [`Frame`] (the object envelope does not
114    /// carry the sender id; the remote peer's identity is known at subscribe
115    /// time, above this layer).
116    sender: u16,
117    /// Drain in arrival order (audio) vs priority-drop stale groups (video). See
118    /// [`drain_next_frame`].
119    reorder_ok: bool,
120}
121
122impl TrackSubscriber {
123    pub(crate) fn new(track: TrackConsumer, sender: u16, reorder_ok: bool) -> Self {
124        Self {
125            track,
126            current: None,
127            sender,
128            reorder_ok,
129        }
130    }
131
132    /// Receive the next reassembled [`Frame`], or `None` at end-of-track
133    /// ([`drain_next_frame`]).
134    pub(crate) async fn next_frame(&mut self) -> Result<Option<Frame>, MoqError> {
135        drain_next_frame(
136            &mut self.track,
137            &mut self.current,
138            self.sender,
139            self.reorder_ok,
140        )
141        .await
142    }
143
144    /// Stall-bounded pull ([`drain_next_frame_stalled`]); video-only.
145    pub(crate) async fn next_frame_stalled(
146        &mut self,
147        stall: std::time::Duration,
148    ) -> Result<VideoPull, MoqError> {
149        debug_assert!(!self.reorder_ok, "stall-bounded pull is video-only");
150        drain_next_frame_stalled(&mut self.track, &mut self.current, self.sender, stall).await
151    }
152}

154/// One publish track on a multiplexed [`crate::MoqRelaySession`]. Same GOP/object
155/// framing as [`crate::MoqPublisher`] (it shares [`gop_publish_frame`]); the only
156/// difference is the `TrackProducer` comes from a shared session's origin rather
157/// than its own connection. Holds its [`BroadcastProducer`] so the announce
158/// persists until drop.
159pub struct MoqTrackPublisher(pub(crate) TrackPublisher);
160
161impl MoqTrackPublisher {
162    /// Block until at least one subscriber has attached (the moq-net live-join gate;
163    /// see [`crate::MoqPublisher::wait_subscribed`]). Await before the first keyframe.
164    pub async fn wait_subscribed(&self) -> Result<(), MoqError> {
165        self.0.wait_subscribed().await
166    }
167
168    /// [`Self::wait_subscribed`] as an OWNED future, detached from `&self`, so a
169    /// publish loop can await the subscriber signal concurrently with publishing
170    /// (e.g. one `select!` arm alongside frame forwarding). Resolves `Err` when
171    /// the track closes instead.
172    pub fn subscribed(
173        &self,
174    ) -> impl std::future::Future<Output = Result<(), MoqError>> + Send + 'static {
175        self.0.subscribed()
176    }
177
178    /// Publish one encoded [`Frame`] (keyframe rolls the GOP group; a delta appends
179    /// to the open group). See [`crate::MoqPublisher::publish_frame`].
180    pub fn publish_frame(&mut self, frame: &Frame) -> Result<(), MoqError> {
181        self.0.publish_frame(frame)
182    }
183
184    /// Publish one AUDIO frame as an object in a count-rolled ordered group
185    /// ([`gop_publish_audio_frame`]) — the ordered-substream model. Use this (not
186    /// [`Self::publish_frame`]) for mic / screen-audio tracks.
187    pub fn publish_audio_frame(&mut self, frame: &Frame) -> Result<(), MoqError> {
188        self.0.publish_audio_frame(frame)
189    }
190
191    /// Finish the open group and the track so the subscriber's drain ends cleanly.
192    pub fn finish(&mut self) -> Result<(), MoqError> {
193        self.0.finish()
194    }
195}

197/// One subscribe track on a multiplexed [`crate::MoqRelaySession`]. Shares the
198/// priority-drop [`drain_next_frame`] discipline with
199/// [`crate::MoqRelaySubscriber`]; the only difference is the `TrackConsumer` came
200/// from a shared session.
201pub struct MoqTrackSubscriber(pub(crate) TrackSubscriber);
202
203impl MoqTrackSubscriber {
204    /// Receive the next reassembled [`Frame`], or `None` at end-of-track.
205    pub async fn next_frame(&mut self, _now_us: i64) -> Result<Option<Frame>, MoqError> {
206        self.0.next_frame().await
207    }
208
209    /// Stall-bounded pull for VIDEO tracks ([`drain_next_frame_stalled`]):
210    /// a group that delivers nothing within `stall` is abandoned and
211    /// [`VideoPull::Stalled`] returned so the caller can request a keyframe.
212    /// Only valid on a priority-drop subscription (`reorder_ok = false`).
213    pub async fn next_frame_stalled(
214        &mut self,
215        stall: std::time::Duration,
216    ) -> Result<VideoPull, MoqError> {
217        self.0.next_frame_stalled(stall).await
218    }
