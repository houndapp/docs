1//! The MoQ object envelope.
2//!
3//! A MoQ object payload is opaque to the relay, but the *endpoints* still need to
4//! recover the `hound_media::Frame` envelope (which track, which codec, the
5//! keyframe flag, the capture timestamp) that the encoder stamped and the
6//! jitter/playout core consumes. We prepend a tiny fixed descriptor to the object
7//! payload and forward the encoded bytes after it verbatim. The relay never reads it.
8//!
9//! Layout (little-endian), `OBJ_HEADER_LEN` = 16 bytes, then the encoded payload:
10//! ```text
11//!   [0]       track     (TrackKind as u8)
12//!   [1]       codec     (Codec as u8)
13//!   [2]       flags     (bit0 = keyframe)
14//!   [3]       layer     (bits 0..3 = temporal id; bits 3..8 reserved, zero)
15//!   [4..8]    frame_seq (u32)
16//!   [8..16]   timestamp_us (i64)
17//!   [16..]    encoded frame bytes (forwarded verbatim)
18//! ```
19//!
20//! Kept separate from `hound_media::HEADER_LEN` because this is the MoQ-object
21//! envelope, not the datagram `PacketHeader`.
22//!
23//! Byte 3 was the pad byte and now carries the L1T2 temporal id (§2.3 of the
24//! large-call research). It is deliberately a flat per-frame value rather than the
25//! AV1 Dependency Descriptor: the DD is stateful (templates arrive only at
26//! intervals, so a relay must carry template state and survive a missed update)
27//! and has no Rust implementation at all — Pion, webrtc-rs and str0m all lack it.
28//! A masked 3-bit id in a byte we already send is a strictly simpler, already
29//! codec-blind version of RFC 9626's short form, and it costs zero extra bytes.
30//!
31//! It is also the ONLY thing the relay reads out of a hound object
32//! (`services/sfu::derive`), which is why [`object_temporal_id`] is public: the
33//! SFU must never grow its own copy of "where the tid lives".
34
35use bytes::Bytes;
36use hound_media::{Codec, Frame, TrackKind};
37
38use super::MoqError;

40pub(crate) const OBJ_HEADER_LEN: usize = 16;
41
42/// Byte offset of the temporal-layer field in the object envelope.
43pub const OBJ_TEMPORAL_OFFSET: usize = 3;
44
45/// Read the temporal id off an encoded object, without decoding the rest of it —
46/// the relay's derive filter (`services/sfu::derive`) is the only caller, and it
47/// runs per object, so it does exactly this much work: one bounds check and one
48/// mask. `None` for a buffer too short to be an object at all.
49///
50/// Returns the raw layer id; [`hound_media::TemporalLayer::from_id`] classifies
51/// it. Masking (rather than comparing the whole byte) is load-bearing — a future
52/// flag in the reserved bits must not read as a temporal id of 8.
53pub fn object_temporal_id(bytes: &[u8]) -> Option<u8> {
54    (bytes.len() >= OBJ_HEADER_LEN)
55        .then(|| bytes[OBJ_TEMPORAL_OFFSET] & hound_media::TEMPORAL_ID_MASK)
56}
57
58/// Per-object payload bound for AUDIO tracks (defense-in-depth; see
59/// [`MAX_VIDEO_OBJECT`]). A real Opus frame is on the order of tens to low
60/// hundreds of bytes — 256 KiB is orders of magnitude above any legitimate frame,
61/// so this only ever catches a malformed/hostile object, never real traffic.
62const MAX_AUDIO_OBJECT: usize = 256 * 1024;
63
64/// Per-object payload bound for VIDEO tracks (defense-in-depth on top of
65/// moq-net's own transport-level frame cap). Deliberately equals moq-net 0.1.9's
66/// `MAX_FRAME_SIZE` (16 MiB): sizing here is trusted to that cap today, but a
67/// relay/moq-net change should not silently widen what this layer accepts without
68/// the bound being visibly re-examined.
69const MAX_VIDEO_OBJECT: usize = 16 * 1024 * 1024;
70
71pub(crate) fn encode_object(frame: &Frame) -> Bytes {
72    let mut buf = Vec::with_capacity(OBJ_HEADER_LEN + frame.data.len());
73    buf.push(frame.track as u8);
74    buf.push(frame.codec as u8);
75    buf.push(if frame.keyframe { 1 } else { 0 });
76    // Masked, not written raw: the upper 5 bits are reserved and must stay zero,
77    // and an encoder reporting an out-of-range layer must not scribble into them.
78    buf.push(frame.temporal_id & hound_media::TEMPORAL_ID_MASK);
79    buf.extend_from_slice(&frame.frame_seq.to_le_bytes());
80    buf.extend_from_slice(&frame.timestamp_us.to_le_bytes());
81    buf.extend_from_slice(&frame.data);
82    Bytes::from(buf)
83}
84
85pub(crate) fn decode_object(sender: u16, bytes: &[u8]) -> Result<Frame, MoqError> {
86    if bytes.len() < OBJ_HEADER_LEN {
87        return Err(MoqError::UnknownDescriptor);
88    }
89    let track = TrackKind::from_u8(bytes[0]).map_err(|_| MoqError::UnknownDescriptor)?;
90    let codec = Codec::from_u8(bytes[1]).map_err(|_| MoqError::UnknownDescriptor)?;
91    // Defense-in-depth per track kind, on top of moq-net's own MAX_FRAME_SIZE cap
92    // (see the constants' docs): a rejected object is dropped like any other
93    // malformed one — never a session-level error.
94    let max_payload = if track.is_video() {
95        MAX_VIDEO_OBJECT
96    } else {
97        MAX_AUDIO_OBJECT
98    };
99    if bytes.len() - OBJ_HEADER_LEN > max_payload {
100        return Err(MoqError::ObjectTooLarge);
101    }
102    let keyframe = bytes[2] & 1 == 1;
103    let temporal_id = bytes[OBJ_TEMPORAL_OFFSET] & hound_media::TEMPORAL_ID_MASK;
104    let frame_seq = u32::from_le_bytes([bytes[4], bytes[5], bytes[6], bytes[7]]);
105    let timestamp_us = i64::from_le_bytes([
106        bytes[8], bytes[9], bytes[10], bytes[11], bytes[12], bytes[13], bytes[14], bytes[15],
107    ]);
108    Ok(Frame {
109        sender,
110        track,
111        codec,
112        keyframe,
113        frame_seq,
114        timestamp_us,
115        temporal_id,
116        data: bytes[OBJ_HEADER_LEN..].to_vec(),
117    })
118}

120// ─── Object-envelope unit tests (pure, no IO) ───────────────────────────────────
121//
122// These guard the only place the IO shell parses untrusted bytes off the wire:
123// the 16-byte object envelope (`encode_object` / `decode_object`). The relay
124// forwards opaque payloads, so a malformed or malicious envelope must be dropped,
125// never panic the drain task and never be silently mis-decoded into a bogus
126// `Frame`. The drain loop runs on a moq-net background task; a panic here would
127// be a denial-of-service on the whole session.
128
129#[cfg(test)]
130mod envelope_tests {
131    use super::{
132        MAX_AUDIO_OBJECT, MAX_VIDEO_OBJECT, OBJ_HEADER_LEN, OBJ_TEMPORAL_OFFSET, decode_object,
133        encode_object, object_temporal_id,
134    };

138    fn sample(track: TrackKind, codec: Codec, keyframe: bool, data: Vec<u8>) -> Frame {
139        sample_at(track, codec, keyframe, 0, data)

142    fn sample_at(
143        track: TrackKind,
144        codec: Codec,
145        keyframe: bool,
146        temporal_id: u8,
147        data: Vec<u8>,
148    ) -> Frame {

150            sender: 0, // not carried in the envelope; stamped at decode time
151            track,
152            codec,
153            keyframe,
154            frame_seq: 0xDEAD_BEEF,
155            timestamp_us: -1_234_567_890,
156            temporal_id,
157            data,

161    /// Round-trip: every envelope field survives encode→decode byte-for-byte, the
162    /// sender is stamped from the decode argument (not the wire), and the encoded
163    /// payload is forwarded verbatim. This is the forward-no-transcode contract at
164    /// the envelope layer.
165    #[test]
166    fn encode_decode_round_trips_all_fields() {
167        for track in [
168            TrackKind::CamVideo,
169            TrackKind::CamAudio,
170            TrackKind::ScrVideo,
171            TrackKind::ScrAudio,
172        ] {
173            for codec in [Codec::Av1, Codec::H264, Codec::H265, Codec::Opus] {
174                for keyframe in [true, false] {
175                    let payload: Vec<u8> = (0u16..300).map(|b| b as u8).collect();
176                    let frame = sample(track, codec, keyframe, payload.clone());
177                    let bytes = encode_object(&frame);
178                    let decoded = decode_object(99, &bytes).expect("decode");
179                    assert_eq!(decoded.sender, 99, "sender stamped from arg, not wire");
180                    assert_eq!(decoded.track, track);
181                    assert_eq!(decoded.codec, codec);
182                    assert_eq!(decoded.keyframe, keyframe);
183                    assert_eq!(decoded.frame_seq, frame.frame_seq);
184                    assert_eq!(decoded.timestamp_us, frame.timestamp_us);
185                    assert_eq!(decoded.temporal_id, frame.temporal_id);
186                    assert_eq!(decoded.data, payload, "payload forwarded verbatim");
187                }
188            }
189        }
190    }

192    /// THE temporal-SVC wire contract: the layer id survives the envelope, it
193    /// lives in the byte the relay reads, and it never disturbs anything else.
194    ///
195    /// Every field is re-checked at each id rather than just the id itself,
196    /// because the failure this guards against is a temporal write spilling into
197    /// the neighbouring `frame_seq` — which would present as phantom loss on
198    /// every subscription in the call, not as a broken layer.
199    #[test]
200    fn the_temporal_id_rides_the_reserved_byte_and_survives_the_round_trip() {
201        for tid in 0..=hound_media::TEMPORAL_ID_MASK {
202            let payload: Vec<u8> = (0u16..64).map(|b| b as u8).collect();
203            let frame = sample_at(
204                TrackKind::CamVideo,
205                Codec::H264,
206                tid == 0,
207                tid,
208                payload.clone(),
209            );
210            let bytes = encode_object(&frame);
211            // It is in byte 3, which is what the relay's filter indexes.
212            assert_eq!(bytes[OBJ_TEMPORAL_OFFSET], tid);
213            assert_eq!(object_temporal_id(&bytes), Some(tid));
214
215            let decoded = decode_object(7, &bytes).expect("decode");
216            assert_eq!(decoded.temporal_id, tid);
217            assert_eq!(
218                decoded.frame_seq, frame.frame_seq,
219                "tid must not bleed into frame_seq"
220            );
221            assert_eq!(decoded.timestamp_us, frame.timestamp_us);
222            assert_eq!(decoded.keyframe, frame.keyframe);
223            assert_eq!(decoded.data, payload);
224            assert_eq!(
225                hound_media::TemporalLayer::from_id(decoded.temporal_id).discardable(),
226                tid != 0
227            );
228        }
229    }

231    /// The reserved bits are reserved in BOTH directions: an out-of-range id from
232    /// an encoder is masked on the way out (it can never scribble into a future
233    /// flag), and a set reserved bit on the way in is masked off rather than read
234    /// as a temporal id of 8. Both halves are what make the spare bits usable
235    /// later without a wire break.
236    #[test]
237    fn the_reserved_bits_of_the_layer_byte_are_masked_both_ways() {
238        let frame = sample_at(TrackKind::CamVideo, Codec::H264, false, 0xFF, vec![1, 2, 3]);
239        let bytes = encode_object(&frame);
240        assert_eq!(
241            bytes[OBJ_TEMPORAL_OFFSET],
242            hound_media::TEMPORAL_ID_MASK,
243            "an out-of-range id must be masked, not written raw"
244        );
245
246        // A future flag set in a reserved bit reads as its own tid, unchanged.
247        let mut tampered = bytes.to_vec();
248        tampered[OBJ_TEMPORAL_OFFSET] = 0b1111_1000 | 1;
249        assert_eq!(object_temporal_id(&tampered), Some(1));
250        assert_eq!(decode_object(0, &tampered).expect("decode").temporal_id, 1);
251    }

253    /// The relay's accessor is total: any buffer too short to be an object at all
254    /// answers `None` rather than indexing into it. The derive filter runs on a
255    /// moq-net background task, where a panic is a session-wide denial of service.
256    #[test]
257    fn the_relay_accessor_never_indexes_a_short_buffer() {
258        for len in 0..OBJ_HEADER_LEN {
259            assert_eq!(object_temporal_id(&vec![0xFFu8; len]), None, "len {len}");
260        }
261        assert_eq!(object_temporal_id(&[0u8; OBJ_HEADER_LEN]), Some(0));
262    }

264    /// An empty (zero-length) encoded payload still round-trips: the envelope is
265    /// exactly `OBJ_HEADER_LEN` bytes and decodes to an empty `data`.

267    fn empty_payload_round_trips() {

275    /// A buffer shorter than the fixed header is rejected, not indexed into. This
276    /// is the panic-safety guard: every prefix length from 0..OBJ_HEADER_LEN must
277    /// return `UnknownDescriptor`, never slice-out-of-bounds panic.
278    #[test]
279    fn short_buffer_is_rejected_without_panic() {
280        for len in 0..OBJ_HEADER_LEN {
281            let buf = vec![0u8; len];
282            let err = decode_object(0, &buf);
283            assert!(
284                matches!(err, Err(MoqError::UnknownDescriptor)),
285                "len {len} should be UnknownDescriptor, got {err:?}"
286            );
287        }

371    /// The keyframe flag is bit0 only; high bits in the flags byte are ignored
372    /// (forward-compatible) and do not corrupt the keyframe decode.
373    #[test]
374    fn keyframe_flag_reads_only_bit0() {
375        let mut bytes =
376            encode_object(&sample(TrackKind::CamVideo, Codec::Av1, false, vec![9])).to_vec();
377        // Set every high bit but leave bit0 clear → still a delta.
378        bytes[2] = 0b1111_1110;
379        assert!(!decode_object(0, &bytes).unwrap().keyframe);
380        // Set bit0 (plus noise) → keyframe.
381        bytes[2] = 0b1010_1011;
382        assert!(decode_object(0, &bytes).unwrap().keyframe);
383    }
