1//! The shared GOP↔group/object mapping and the drain disciplines.
2//!
3//! The single publish path (keyframe rolls the group; audio rolls by count) and
4//! the single subscribe drain (priority-drop across groups, ordered within one)
5//! that every publisher/subscriber-shaped type in this crate is built on — see
6//! the module docs on [`crate::moq`] for the frame ↔ group/object model.

14/// Map one encoded [`Frame`] onto the MoQ GOP/object model on `track`, advancing
15/// `open_group`/`next_group_seq`. The single GOP-publish code path shared by the
16/// per-track [`crate::MoqPublisher`] and the multiplexed [`crate::MoqTrackPublisher`] — see
17/// [`crate::MoqPublisher::publish_frame`] for the keyframe-rolls-the-group contract.
18pub(crate) fn gop_publish_frame(
19    track: &mut TrackProducer,
20    open_group: &mut Option<moq_net::GroupProducer>,
21    next_group_seq: &mut u64,
22    frame: &Frame,
23) -> Result<(), MoqError> {
24    let object = encode_object(frame);
25    if frame.keyframe {
26        // Roll the GOP boundary: a stale group is dropped *whole* at its boundary
27        // by the subscriber, so a fresh keyframe is never queued behind an older
28        // group's retransmits.
29        if let Some(mut prev) = open_group.take() {
30            prev.finish()
31                .map_err(|e| MoqError::Moq(format!("finish prev group: {e}")))?;
32        }
33        let mut group = track
34            .create_group(Group::from(*next_group_seq))
35            .map_err(|e| MoqError::Moq(format!("create_group: {e}")))?;
36        group
37            .write_frame(object)
38            .map_err(|e| MoqError::Moq(format!("write keyframe object: {e}")))?;
39        *next_group_seq += 1;
40        *open_group = Some(group);
41    } else {
42        let group = open_group.as_mut().ok_or(MoqError::DeltaBeforeKeyframe)?;
43        group
44            .write_frame(object)
45            .map_err(|e| MoqError::Moq(format!("write delta object: {e}")))?;
46    }
47    Ok(())
48}

50/// Audio objects per group before rolling the group boundary (the **ordered audio
51/// substream** model). Audio is sequential and stateful — not GOP-structured like
52/// video — so instead of one group (= one QUIC uni-stream) PER FRAME (which gives ~50
53/// unordered racing streams/s whose cross-stream reordering corrupts playout),
54/// successive Opus frames are appended as objects to ONE group so QUIC delivers them
55/// **in order**. The group is rolled every `AUDIO_GROUP_OBJECTS` frames because a single
56/// never-rolled group is not viable in moq-net 0.1.9 (the 1024-object / 32 MB eviction
57/// cap → fatal `CacheFull`; late-join only attaches at a group boundary; and a lost
58/// object head-of-line-blocks the whole group until retransmit).
59///
60/// The window size is the **HoL blast radius**: under connection congestion (e.g. a
61/// high-bitrate screenshare surge sharing the link) a lost audio group's stream stalls
62/// until QUIC retransmit, so a whole window of audio arrives too late and is dropped —
63/// one audible hole the size of the window. At 20 ms/frame, 4 objects ≈ 80 ms: within
64/// the jitter buffer's absorption (the receiver's reorder buffer mops up the extra
65/// cross-group boundaries), while still cutting stream churn ~4× vs per-frame and
66/// staying far under the eviction cap.
67const AUDIO_GROUP_OBJECTS: usize = 4;
68
69/// Publish one audio frame as an object in a COUNT-ROLLED group (the ordered-substream
70/// model — see [`AUDIO_GROUP_OBJECTS`]). The first frame and every `AUDIO_GROUP_OBJECTS`-th
71/// frame opens a fresh group (new ordered QUIC stream); the rest append to the open
72/// group, so a window of frames rides one in-order stream. The frame's `keyframe` bit is
73/// IGNORED — every Opus frame is independently decodable, so audio grouping is driven by
74/// count, not by a keyframe flag. Shared by [`crate::MoqPublisher`] and [`crate::MoqTrackPublisher`].
75pub(crate) fn gop_publish_audio_frame(
76    track: &mut TrackProducer,
77    open_group: &mut Option<moq_net::GroupProducer>,
78    next_group_seq: &mut u64,
79    objects_in_group: &mut usize,
80    frame: &Frame,
81) -> Result<(), MoqError> {
82    let object = encode_object(frame);
83    // Roll on the first frame (no open group) or once the window is full.
84    if open_group.is_none() || *objects_in_group >= AUDIO_GROUP_OBJECTS {
85        if let Some(mut prev) = open_group.take() {
86            prev.finish()
87                .map_err(|e| MoqError::Moq(format!("finish prev audio group: {e}")))?;
88        }
89        let mut group = track
90            .create_group(Group::from(*next_group_seq))
91            .map_err(|e| MoqError::Moq(format!("create audio group: {e}")))?;
92        group
93            .write_frame(object)
94            .map_err(|e| MoqError::Moq(format!("write audio group-open object: {e}")))?;
95        *next_group_seq += 1;
96        *objects_in_group = 1;
97        *open_group = Some(group);
98    } else {
99        let group = open_group
100            .as_mut()
101            .expect("the roll branch covers the None case");
102        group
103            .write_frame(object)
104            .map_err(|e| MoqError::Moq(format!("write audio object: {e}")))?;
105        *objects_in_group += 1;
106    }
107    Ok(())
108}

110/// Finish the open GOP group (if any) then the track, so the subscriber's drain
111/// terminates cleanly. Shared by [`crate::MoqPublisher`] and [`crate::MoqTrackPublisher`].
112pub(crate) fn gop_finish(
113    track: &mut TrackProducer,
114    open_group: &mut Option<moq_net::GroupProducer>,
115) -> Result<(), MoqError> {
116    if let Some(mut group) = open_group.take() {
117        group
118            .finish()
119            .map_err(|e| MoqError::Moq(format!("finish open group: {e}")))?;
120    }
121    track
122        .finish()
123        .map_err(|e| MoqError::Moq(format!("finish track: {e}")))
124}

126/// The outcome of a stall-bounded video pull ([`crate::MoqTrackSubscriber::next_frame_stalled`]).
127pub enum VideoPull {
128    /// One reassembled frame.
129    Frame(Frame),
130    /// Nothing arrived within the stall bound. If a group was mid-drain it has
131    /// been ABANDONED (its remaining objects are skipped), so the next delivery
132    /// resumes at a fresh group boundary — the caller should request a keyframe
133    /// so the producer rolls one.
134    Stalled,
135    /// End of track.
136    End,
137}

139/// Shared group/object drain used by [`crate::MoqSubscriber`] and
140/// [`crate::MoqRelaySubscriber`] — the only difference between the two is how the
141/// `TrackConsumer` was obtained (direct `Server::accept` vs. `Client::connect`
142/// to the relay); the priority-drop drain discipline is identical.
143///
144/// Drains one group (GOP) at a time via [`TrackConsumer::next_group`]
145/// (priority-drop: a stale group is discarded at its boundary rather than
146/// head-of-line-blocking a newer keyframe); within a group, objects are read in
147/// order. Each object's `OBJ_HEADER_LEN`-byte envelope is decoded back into a
148/// `Frame` and the encoded payload handed out verbatim.
149///
150/// ## Known limitation — within-GOP head-of-line stall under heavy loss
151///
152/// `next_group`'s priority-drop only fires at a *group boundary*: it discards a
153/// stale group instead of head-of-line-blocking a newer keyframe, but only once
154/// the current group has drained. A subscriber already part-way through group K
155/// can still **stall inside K** if an early object of K is lost — QUIC
156/// retransmits it, but moq-net prioritises the *newest* streams, so under
157/// continuous production K's retransmit can be starved behind newer GOPs and the
158/// inner `GroupConsumer::read_frame` blocks. Recovery is then bounded by QUIC
159/// eventually retransmitting K's missing object, which under a sustained,
160/// contiguous downstream loss burst can be slow. Preemptively abandoning the
161/// stalled GOP to jump to the next keyframe does not reliably help: when the
162/// burst also drops the *newer* GOP's opener, `next_group` has nothing newer to
163/// offer either. The fault-injection e2e
164/// (`services/sfu/tests/moq_relay_faults.rs`) characterises the recoverable
165/// regime; see its module docs.
166///
167/// `reorder_ok = true` drains via [`TrackConsumer::recv_group`] (ARRIVAL order, never
168/// discards) instead of [`TrackConsumer::next_group`] (monotonic, drops a group whose
169/// sequence ≤ one already returned). Set it for AUDIO: audio groups are small and
170/// frequent, and on the shared connection a high-bitrate screenshare reorders those
171/// tiny audio streams, so `next_group`'s staleness rule would silently drop reordered
172/// ones — fatal, since a slightly-reordered Opus frame is perfectly playable (the
173/// jitter buffer reorders by timestamp and the decoder skips truly-stale frames). Video
174/// keeps `next_group`: a stale GOP genuinely should be dropped for a newer keyframe.
175pub(crate) async fn drain_next_frame(
176    track: &mut TrackConsumer,
177    current: &mut Option<moq_net::GroupConsumer>,
178    sender: u16,
179    reorder_ok: bool,
180) -> Result<Option<Frame>, MoqError> {
181    loop {
182        // Finish draining the currently open group's objects first.
183        if let Some(group) = current.as_mut() {
184            match group.read_frame().await {
185                Ok(Some(object)) => return decode_object(sender, &object).map(Some),
186                Ok(None) => {
187                    *current = None; // group exhausted; advance
188                }
189                Err(e) => {
190                    // A group-level reset is a NORMAL boundary, not a pipeline
191                    // error: when a new keyframe group appears, the relay's
192                    // per-subscriber forwarding priority-drops the stale group
193                    // by resetting its stream ("remote error: code=0" /
194                    // "cancelled" mid-read). Treating that as fatal would tear
195                    // the session down on every PLI-forced keyframe roll.
196                    // Abandon the group and pull the next one; only
197                    // `next_group` / session-level errors below propagate.
198                    tracing::debug!("moq: group abandoned mid-read (boundary): {e}");
199                    *current = None;
200                }
201            }
202        }
203
204        // No open group: pull the next one. Audio takes EVERY group in arrival order
205        // (`recv_group`); video priority-drops stale groups (`next_group`).
206        let next = if reorder_ok {
207            track.recv_group().await
208        } else {
209            track.next_group().await
210        }
211        .map_err(|e| MoqError::Moq(format!("next_group: {e}")))?;
212        match next {
213            Some(group) => *current = Some(group),
214            None => return Ok(None), // track finished
215        }
216    }
217}

219/// [`drain_next_frame`] bounded by a stall budget, for VIDEO tracks.
220///
221/// With on-demand keyframes a GOP (= one group) can span many seconds, and a
222/// lost object inside the group being drained head-of-line-blocks every frame
223/// behind it until QUIC retransmits — the tile freezes, then fast-forwards.
224/// This pull instead treats "no object within `stall`" as a stuck group:
225/// abandon it (skip its remaining objects) and report [`VideoPull::Stalled`] so
226/// the caller can request a keyframe; the producer's IDR opens a fresh group
227/// that `next_group` jumps to, bounding the freeze at ~`stall` instead of a
228/// retransmit tail.
229///
230/// Cancel-safe: the moq-net consumers are poll-based state machines whose
231/// cursors advance only on `Ready`, so the timed-out future is dropped without
232/// losing a frame — and a mid-object timeout only ever discards the group it
233/// abandons. Never use for audio (audio must drain every group).
234pub(crate) async fn drain_next_frame_stalled(
235    track: &mut TrackConsumer,
236    current: &mut Option<moq_net::GroupConsumer>,
237    sender: u16,
238    stall: std::time::Duration,
239) -> Result<VideoPull, MoqError> {
240    loop {
241        if let Some(group) = current.as_mut() {
242            match tokio::time::timeout(stall, group.read_frame()).await {
243                Ok(Ok(Some(object))) => {
244                    return decode_object(sender, &object).map(VideoPull::Frame);
245                }
246                Ok(Ok(None)) => {
247                    *current = None; // group exhausted; advance
248                }
249                Ok(Err(e)) => {
250                    // Group reset at a keyframe roll — a normal boundary (see
251                    // `drain_next_frame`); advance to the next group.
252                    tracing::debug!("moq: group abandoned mid-read (boundary): {e}");
253                    *current = None;
254                }
255                Err(_) => {
256                    tracing::debug!("moq: group stalled mid-drain — abandoning for next keyframe");
257                    *current = None;
258                    return Ok(VideoPull::Stalled);
259                }
260            }
261            continue;
262        }
263        match tokio::time::timeout(stall, track.next_group()).await {
264            Ok(r) => match r.map_err(|e| MoqError::Moq(format!("next_group: {e}")))? {
265                Some(group) => *current = Some(group),
266                None => return Ok(VideoPull::End),
267            },
268            Err(_) => return Ok(VideoPull::Stalled),
269        }
270    }
