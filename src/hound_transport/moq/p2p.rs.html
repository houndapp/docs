12// ─── Publisher ────────────────────────────────────────────────────────────────

14/// MoQ publisher over an iroh connection. Owns the moq-net session (held alive
15/// for the track's lifetime), the broadcast producer, and the open track
16/// producer. Hand it encoded [`Frame`]s via [`MoqPublisher::publish_frame`]; it
17/// maps each onto the GOP group / object model in §2.
18pub struct MoqPublisher {
19    _session: moq_net::Session,
20    inner: TrackPublisher,
21}
22
23impl MoqPublisher {
24    /// Wrap an established iroh `Connection` as a MoQ session and announce one
25    /// broadcast with one track. Returns once the moq-net client handshake
26    /// completes. (The peer's SUBSCRIBE may not have landed yet — gate the first
27    /// frame on [`Self::wait_subscribed`].)
28    pub async fn connect(
29        conn: Connection,
30        broadcast_path: &str,
31        track_name: &str,
32    ) -> Result<Self, MoqError> {
33        Self::connect_inner(conn, broadcast_path, track_name).await
34    }
35
36    /// Like [`Self::connect`] but for the **MoQ relay** path (`services/sfu::moq_relay`): perform the relay-token auth handshake first,
37    /// then announce. The relay scopes this publisher's announces to the space
38    /// in `relay_token`, so `broadcast_path` must live under `space/{space_id}`
39    /// (the [`crate::moq`] namespace convention) or the relay drops it.
40    ///
41    /// The token is presented on a dedicated bi-stream and the relay's accept
42    /// verdict is awaited *before* the MoQ handshake — so a rejected token fails
43    /// fast with [`MoqError::RelayAuthRejected`] and no broadcast is announced.
44    pub async fn connect_relay(
45        conn: Connection,
46        relay_token: &str,
47        broadcast_path: &str,
48        track_name: &str,
49    ) -> Result<Self, MoqError> {
50        relay_auth_handshake(&conn, relay_token).await?;
51        Self::connect_inner(conn, broadcast_path, track_name).await
52    }
53
54    /// Shared body of [`Self::connect`] / [`Self::connect_relay`]: wrap the conn
55    /// as a raw-QUIC MoQ session, announce one broadcast + track, run the moq-net
56    /// `Client` handshake. The caller has already done any auth handshake.
57    async fn connect_inner(
58        conn: Connection,
59        broadcast_path: &str,
60        track_name: &str,
61    ) -> Result<Self, MoqError> {
62        // Raw-QUIC mode: speak MoQ directly over iroh, no HTTP/3.
63        let wt = web_transport_iroh::Session::raw(conn);
64
65        let origin = Origin::random().produce();
66        let mut broadcast = origin.create_broadcast(broadcast_path).ok_or_else(|| {
67            MoqError::Moq(format!("broadcast path not allowed: {broadcast_path}"))
68        })?;
69        let track = broadcast
70            .create_track(Track::new(track_name))
71            .map_err(|e| MoqError::Moq(format!("create_track: {e}")))?;
72
73        let session = moq_net::Client::new()
74            .with_publish(origin.consume())
75            .connect(wt)
76            .await
77            .map_err(|e| MoqError::Moq(format!("client connect: {e}")))?;
78
79        Ok(Self {
80            _session: session,
81            inner: TrackPublisher::new(broadcast, track),
82        })
83    }
84
85    /// Block until at least one subscriber has attached to the track. Awaiting
86    /// this before producing the first keyframe guarantees the opening group is
87    /// delivered (moq-net live-join gate; see module docs). Deterministic — no
88    /// `sleep`/yield.
89    pub async fn wait_subscribed(&self) -> Result<(), MoqError> {
90        self.inner.wait_subscribed().await
91    }
92
93    /// Map one encoded [`Frame`] onto the MoQ group/object model and publish it.
94    ///
95    /// * **keyframe** → finish the open group (if any) and `create_group` at the
96    ///   next sequence; the keyframe is the first object in that group.
97    /// * **delta** → append an object to the open keyframe group. A delta before
98    ///   any keyframe is [`MoqError::DeltaBeforeKeyframe`].
99    ///
100    /// The encoded payload is forwarded byte-for-byte (an `OBJ_HEADER_LEN`-byte
101    /// `hound_media::Frame` envelope is prepended so the subscriber can rebuild
102    /// the `Frame`; the relay never reads it). Per §2 a GOP is one group: the
103    /// keyframe opens it and deltas ride the same ordered stream until the next
104    /// keyframe rolls the boundary.
105    pub fn publish_frame(&mut self, frame: &Frame) -> Result<(), MoqError> {
106        self.inner.publish_frame(frame)
107    }
108
109    /// Publish one AUDIO frame as an object in a count-rolled ordered group
110    /// (`gop_publish_audio_frame`) — the ordered-substream model that keeps
111    /// successive Opus frames on one in-order QUIC stream instead of one stream/frame.
112    pub fn publish_audio_frame(&mut self, frame: &Frame) -> Result<(), MoqError> {
113        self.inner.publish_audio_frame(frame)
114    }
115
116    /// Finish the open group and signal end-of-track so the subscriber's drain
117    /// loop terminates cleanly. Call when the media flow ends.
118    pub fn finish(&mut self) -> Result<(), MoqError> {
119        self.inner.finish()
120    }
121}

123// ─── Subscriber ──────────────────────────────────────────────────────────────
124
125/// MoQ subscriber over an iroh connection. Accepts the moq-net session, waits
126/// for the publisher's broadcast announce, and subscribes to a track. Drains
127/// groups/objects and reassembles [`Frame`]s for the jitter/playout core.
128pub struct MoqSubscriber {
129    _session: moq_net::Session,
130    inner: TrackSubscriber,
131}
132
133impl MoqSubscriber {
134    /// Accept the MoQ session on this side, wait for the publisher's broadcast
135    /// announce, and subscribe to `track_name`. `sender` is the remote peer id to
136    /// stamp on reassembled frames (identity is resolved above this layer).
137    pub async fn accept(
138        conn: Connection,
139        broadcast_path: &str,
140        track_name: &str,
141        sender: u16,
142    ) -> Result<Self, MoqError> {
143        let wt = web_transport_iroh::Session::raw(conn);
144
145        let origin = Origin::random().produce();
146        let mut announced = origin.consume();
147
148        let session = moq_net::Server::new()
149            .with_consume(origin)
150            .accept(wt)
151            .await
152            .map_err(|e| MoqError::Moq(format!("server accept: {e}")))?;
153
154        // Wait for the publisher's broadcast announce for our path.
155        let broadcast = loop {
156            match announced.announced().await {
157                Some((path, Some(bc))) if path.as_str() == broadcast_path => break bc,
158                Some((_, Some(_))) => continue, // a different broadcast; ignore
159                Some((_, None)) => continue,    // an unannounce; keep waiting
160                None => return Err(MoqError::Moq("origin closed before announce".into())),
161            }
162        };
163
164        let track = broadcast
165            .subscribe_track(&Track::new(track_name))
166            .map_err(|e| MoqError::Moq(format!("subscribe_track: {e}")))?;
167
168        Ok(Self {
169            _session: session,
170            // Test-only path (production audio uses MoqTrackSubscriber); keep the
171            // video-style monotonic drain (`reorder_ok = false`).
172            inner: TrackSubscriber::new(track, sender, false),
173        })
174    }
175
176    /// Receive the next reassembled [`Frame`], or `None` at end-of-track.
177    ///
178    /// Drains one group (GOP) at a time with `TrackConsumer::next_group` — the
179    /// **priority-drop** discipline: a late group (sequence ≤ one already
180    /// returned) is discarded at the boundary instead of head-of-line-blocking a
181    /// newer GOP. Within a group, objects are read in order (QUIC-reliable). Each
182    /// object's `OBJ_HEADER_LEN`-byte envelope is decoded back into a `Frame`
183    /// (track/codec/keyframe/timestamp) and the encoded payload handed out
184    /// verbatim to jitter/playout.
185    ///
186    /// `now_us` is accepted for symmetry with the clock-injected cores and for
187    /// future per-subscriber pacing; today groups are read as they arrive.
188    pub async fn next_frame(&mut self, _now_us: i64) -> Result<Option<Frame>, MoqError> {
189        self.inner.next_frame().await
190    }
191
192    /// Stall-bounded pull (see `drain_next_frame_stalled`); video-only.
193    pub async fn next_frame_stalled(
194        &mut self,
195        stall: std::time::Duration,
196    ) -> Result<VideoPull, MoqError> {
197        self.inner.next_frame_stalled(stall).await
198    }
199}

203/// MoQ subscriber over the **relay** path (`services/sfu::moq_relay`).
204///
205/// Unlike [`MoqSubscriber`] (which is the moq-net *server* for direct
206/// peer-to-peer), this subscriber **dials** the relay: it is a moq-net `Client`,
207/// presents its relay token first, then consumes broadcasts the relay announces
208/// (the relay re-announces every publisher's broadcast to subscribers of the
209/// same space). The drain discipline is identical — it shares
210/// `drain_next_frame`.
211///
212/// The relay scopes this subscriber to the space in `relay_token`, so
213/// `broadcast_path` must live under `space/{space_id}` or the relay never
214/// announces it to this connection.
215pub struct MoqRelaySubscriber {
216    _session: moq_net::Session,
217    inner: TrackSubscriber,
218}
219
220impl MoqRelaySubscriber {
221    /// Dial the relay: relay-token auth handshake, MoQ `Client` handshake
222    /// consuming into a local origin, wait for the relay to announce
223    /// `broadcast_path`, then subscribe to `track_name`. `sender` is the remote
224    /// peer id stamped on reassembled frames.
225    pub async fn connect_relay(
226        conn: Connection,
227        relay_token: &str,
228        broadcast_path: &str,
229        track_name: &str,
230        sender: u16,
231    ) -> Result<Self, MoqError> {
232        relay_auth_handshake(&conn, relay_token).await?;
233
234        let wt = web_transport_iroh::Session::raw(conn);
235
236        // Local origin the relay's announces are inserted INTO (the client
237        // `with_consume` half). We then watch it for `broadcast_path`.
238        let origin = Origin::random().produce();
239        let mut announced = origin.consume();
240
241        let session = moq_net::Client::new()
242            .with_consume(origin)
243            .connect(wt)
244            .await
245            .map_err(|e| MoqError::Moq(format!("relay client connect: {e}")))?;
246
247        // Wait for the relay to announce the publisher's broadcast for our path.
248        let broadcast = loop {
249            match announced.announced().await {
250                Some((path, Some(bc))) if path.as_str() == broadcast_path => break bc,
251                Some((_, Some(_))) => continue, // a different broadcast; ignore
252                Some((_, None)) => continue,    // an unannounce; keep waiting
253                None => {
254                    return Err(MoqError::Moq("relay origin closed before announce".into()));
255                }
256            }
257        };
258
259        let track = broadcast
260            .subscribe_track(&Track::new(track_name))
261            .map_err(|e| MoqError::Moq(format!("subscribe_track: {e}")))?;
262
263        Ok(Self {
264            _session: session,
265            // Test-only path (production audio uses MoqTrackSubscriber); keep the
266            // video-style monotonic drain (`reorder_ok = false`).
267            inner: TrackSubscriber::new(track, sender, false),
268        })
269    }
270
271    /// Receive the next reassembled [`Frame`], or `None` at end-of-track. See
272    /// `drain_next_frame` for the priority-drop drain discipline.
273    pub async fn next_frame(&mut self, _now_us: i64) -> Result<Option<Frame>, MoqError> {
274        self.inner.next_frame().await
275    }
