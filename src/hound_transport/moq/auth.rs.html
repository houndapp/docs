1//! Relay-token handshake (client side).
2//!
3//! When connecting to the MoQ relay (`services/sfu::moq_relay`) the
4//! client must present its relay token BEFORE the MoQ handshake. The relay reads
5//! the token off the first bi-stream, verifies the backend's HMAC scheme, and
6//! replies one byte (1 = ok, 0 = reject). The client gates its MoQ handshake on
7//! reading the ok byte, so the relay's token bi-stream is always accepted before
8//! moq-net opens its own control stream (QUIC accepts bi-streams in open order).
9//!
10//! Wire (client → relay): u16 LE length + token bytes. (relay → client): 1 byte.
11//!
12//! Direct peer-to-peer pub/sub ([`crate::MoqPublisher::connect`] /
13//! [`crate::MoqSubscriber::accept`]) does NOT use this — only the relay path does.
14
15use iroh::endpoint::Connection;
16
17use super::MoqError;
18
19const RELAY_AUTH_OK: u8 = 1;
20
21/// Open the relay-token control bi-stream, send `token`, and await the relay's
22/// one-byte verdict. Returns `Ok(())` only if the relay accepted the token.
23pub(crate) async fn relay_auth_handshake(conn: &Connection, token: &str) -> Result<(), MoqError> {
24    let (mut send, mut recv) = conn
25        .open_bi()
26        .await
27        .map_err(|e| MoqError::Iroh(format!("open token stream: {e}")))?;
28
29    let token_bytes = token.as_bytes();
30    let len = u16::try_from(token_bytes.len())
31        .map_err(|_| MoqError::Moq("relay token too long".into()))?;
32    send.write_all(&len.to_le_bytes())
33        .await
34        .map_err(|e| MoqError::Iroh(format!("write token len: {e}")))?;
35    send.write_all(token_bytes)
36        .await
37        .map_err(|e| MoqError::Iroh(format!("write token: {e}")))?;
38    send.finish()
39        .map_err(|e| MoqError::Iroh(format!("finish token stream: {e}")))?;
40
41    let mut verdict = [0u8; 1];
42    recv.read_exact(&mut verdict)
43        .await
44        .map_err(|e| MoqError::Iroh(format!("read auth verdict: {e}")))?;
45    if verdict[0] == RELAY_AUTH_OK {
46        Ok(())
47    } else {
48        Err(MoqError::RelayAuthRejected)
49    }
50}
