1//! The typed view-models the UI renders — the data plane's shape, owned here so
2//! there is one source of truth (no drift between a Dart model and a Rust model).
3//! These mirror the backend contract (`ui/src/lib/openapi/*`,
4//! `services/backend/src/api/*`) but are *client* view-models: only the fields the
5//! UI needs, in client-friendly forms. They are plain serde types so the FRB layer
6//! mirrors them into Dart and reducers can snapshot them.
7

49/// One chat message. `author` keys into the roster for name/avatar (the UI joins);
50/// `edited_at_ms` is `Some` only if the message was edited.
51#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
52pub struct Message {
53    pub id: Uuid,
54    /// The space (channel) this message belongs to. Redundant with the
55    /// enclosing list's own key everywhere a `Message` arrives already scoped
56    /// to a space (the message list, realtime events) — carried on the
57    /// message itself because [`crate::app::HoundClient::fetch_message`] has
58    /// no other way to learn it: a `<hound:message/…>` reference resolves a
59    /// message id alone, with no enclosing space context.
60    pub space_id: Uuid,
61    pub author: Uuid,
62    pub contents: String,
63    /// `contents` parsed by `hound-markup` into the tree the view layer
64    /// renders. Parsed once at ingestion (every construction site in
65    /// `rest/wire.rs` and the realtime event conversion) so the view layer
66    /// never re-parses; server and client run the same crate, so this is
67    /// exactly the tree the server extracted references from.
68    ///
69    /// `#[serde(skip)]`: `hound_markup`'s AST is plain owned data built for the
70    /// FRB boundary, not serde — and nothing decodes a `Message` itself via
71    /// serde (every wire shape is a private DTO in `rest::wire` that maps
72    /// in via `into_model`, always re-deriving `body` from `contents`), so
73    /// there is no round-trip this default-on-skip could silently lose.
74    #[serde(skip)]
75    pub body: hound_markup::Document,
76    /// Server send time (unix ms). Used for grouping + display ordering.
77    pub sent_at_ms: i64,
78    pub edited_at_ms: Option<i64>,
79    pub attachments: Vec<Attachment>,
80    /// Link previews for the URLs in `contents`, in the order they appear.
81    ///
82    /// Empty until the server has unfurled them, which happens *after* the
83    /// message is delivered — so a message with a link arrives with none, and a
84    /// `MessageEmbeds` event fills them in a moment later. The UI must treat an
85    /// empty list as "not yet", not as "never".
86    ///
87    /// Every media URL inside is **relative to the API base** (see
88    /// [`hound_types::EmbedMedia::url`]); the view layer resolves it. Nothing
89    /// here points at a third-party host.
90    pub embeds: Vec<LinkEmbed>,
91}

112/// Authentication state the UI gates routing on. A snapshot, pushed on every
113/// transition (signed-out ⇄ signed-in / token-refreshed).
114#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
115pub enum AuthState {
116    /// No valid session — show login/signup.
117    SignedOut,
118    /// A valid session exists for this user.
119    SignedIn { user_id: Uuid },
120}
121
122impl AuthState {
123    pub fn user_id(&self) -> Option<Uuid> {
124        match self {
125            Self::SignedIn { user_id } => Some(*user_id),
126            Self::SignedOut => None,
127        }
128    }
129    pub fn is_signed_in(&self) -> bool {
130        matches!(self, Self::SignedIn { .. })
131    }
132}

134/// Result of an auth flow that can end more than one way: a Steam sign-in
135/// (in-Steam ticket or browser OpenID) or a browser-deferred ceremony wait.
136#[derive(Clone, Debug, PartialEq)]
137pub enum AuthFlowOutcome {
138    /// A session was established — signed in.
139    SignedIn(AuthState),
140    /// Unknown Steam ID: a link token is parked on the client and the UI must
141    /// show the create-or-link screen ("Continue as `persona`" / "I already
142    /// have an account").
143    SteamUnknown { persona: String },
144    /// The browser email-link ceremony completed: an email + passkey were
145    /// added to the already-signed-in account. No session change.
146    EmailLinked,
147    /// Unknown email (magic-link sign-in): a link token is parked on the
148    /// client and the UI must show the create-or-link screen ("Continue as
149    /// `email`").
150    EmailUnknown { email: String },
151}

153/// A page of results from a paginated list endpoint.
154///
155/// **Wire note:** the backend's pagination envelope is
156/// `{ contents, current, next, prev, total }` (see `services/backend/src/api/data.rs`).
157/// This view-model keeps a client-friendly shape; the wire→view mapping lives in
158/// [`crate::rest`] (`WirePage<T>`), so the cursor quirk does not leak here.
159#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
160pub struct Page<T> {
161    pub items: Vec<T>,
162    /// 0-indexed current page number.
163    pub page: u32,
164    /// Next page number to request (== `page` when on the last page).
165    pub next_page: u32,
166    /// Previous page number (== `page` when on the first page).
167    pub prev_page: u32,
168    /// Total number of items across all pages.
169    pub total: u64,
170}

172/// What the client must present to the SFU + relay to join a voice channel, plus the
173/// roster snapshot at join time. Built from the backend `VoiceJoinResponse`.
174#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
175pub struct VoiceJoin {
176    /// The joining user's own merged community profile (name + avatar).
177    pub profile: Member,
178    /// iroh relay URL — never hardcoded client-side; always taken from the server.
179    pub relay_url: String,
180    /// Short-lived (15 min TTL) HMAC token proving this user may connect to the SFU.
181    /// Format: `{user_id}:{space_id}:{expiry_unix_secs}:{hex_mac}`.
182    pub relay_token: String,
183    /// Other participants already in the channel at join time (render before media).
184    pub peers: Vec<VoicePeer>,
185    /// iroh node id (hex public key) of the SFU. `None` when the backend has no SFU
186    /// configured (dev) — clients skip the direct QUIC connection in that case.
187    pub sfu_node_id: Option<String>,
188}

202/// Which messages a search may look at.
203///
204/// The default is the community the user is looking at; deleting the scope pill
205/// widens it to every community they are a member of. `in:` inside the query
206/// narrows *within* the scope — it never widens it, and the backend structurally
207/// excludes anything unreadable either way.
208#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
209pub enum SearchScope {
210    Community(Uuid),
211    Everywhere,
212}
213
214/// How a page of hits is ordered.
215///
216/// `Relevant` is only meaningful with at least one positive text term for the
217/// index to score against ([`hound_search::Query::has_positive_text`]), which is
218/// why the UI disables it otherwise.
219#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
220pub enum SearchSort {
221    Newest,
222    Relevant,
223}
224
225/// One `POST /search` call.
226///
227/// `query` is [`hound_search::Query`] itself, not a re-declaration: client and
228/// backend run the same parser crate, so the filter the composer highlighted is
229/// byte-for-byte the filter the SQL applies.
230#[derive(Clone, Debug, PartialEq)]
231pub struct SearchRequest {
232    pub scope: SearchScope,
233    pub query: hound_search::Query,
234    pub sort: SearchSort,
235    /// Opaque keyset cursor from a previous page's
236    /// [`SearchPage::next_cursor`]; `None` asks for the first page.
237    pub cursor: Option<String>,
238    /// Hits per page, 1..=50 (the backend rejects anything else as a 400).
239    pub limit: u32,
240    /// Minutes east of UTC. `before:`/`after:`/`during:`/`today` resolve to
241    /// *the reader's* midnight, so the offset travels with the request — the
242    /// server never reads its own clock for a date filter.
243    pub utc_offset_minutes: i32,
244}

246/// One search hit: the message, the community it lives in, its relevance score,
247/// and the author's display identity.
248///
249/// `author` is joined here rather than left to the roster because a hit can come
250/// from a community whose members the client has never loaded — the wire hoists
251/// the profiles into one `users` array and this is where they land back on their
252/// messages. An author the server did not ship falls back to a bare
253/// [`Member`] (no name, no avatar), which the UI already renders as a short id.
254#[derive(Clone, Debug, PartialEq)]
255pub struct SearchHit {
256    pub message: Message,
257    pub community_id: Uuid,
258    /// Higher is better. `0.0` when the query had nothing to score against.
259    pub score: f64,
260    pub author: Member,
261}
262
263/// A page of search hits. There is no total: the cursor is a keyset, so "how
264/// many more" is a question the backend deliberately cannot answer cheaply.
265#[derive(Clone, Debug, Default, PartialEq)]
266pub struct SearchPage {
267    pub hits: Vec<SearchHit>,
268    /// Cursor for the next page, or `None` at the end of the results.
269    pub next_cursor: Option<String>,
270}
