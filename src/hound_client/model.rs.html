1//! The typed view-models the UI renders — the data plane's shape, owned here so
2//! there is one source of truth (no drift between a Dart model and a Rust model).
3//! These mirror the backend contract (`ui/src/lib/openapi/*`,
4//! `services/backend/src/api/*`) but are *client* view-models: only the fields the
5//! UI needs, in client-friendly forms. They are plain serde types so the FRB layer
6//! mirrors them into Dart and reducers can snapshot them.

12/// A closed enum whose variants ARE wire tokens — one spelling, written once.
13///
14/// The token table is the single source of truth: `as_wire`/`from_wire` and the
15/// serde impls are all generated from it by [`wire_tokens!`], so a spelling
16/// cannot drift between the hand-written mapping and a `rename_all` attribute
17/// (they did, and only one of the two was on the path the backend reads).
18pub trait WireToken: Copy + Sized + 'static {
19    /// Every variant, in wire order.
20    const ALL: &'static [Self];
21    /// What this kind of token is, for the log line an unknown one earns.
22    const KIND: &'static str;
23
24    /// The backend's token for this variant.
25    fn as_wire(self) -> &'static str;
26
27    /// Parse a wire token. `None` for anything this build does not know —
28    /// callers DROP those rather than failing, because a newer backend is
29    /// allowed to send tokens an older client cannot render.
30    fn from_wire(token: &str) -> Option<Self>;
31}
32
33/// Declare a [`WireToken`] enum from its token table.
34///
35/// Generates the enum, its `ALL`/`as_wire`/`from_wire`, and `Serialize` +
36/// `Deserialize` **in terms of those** — so the serde spelling and the REST
37/// layer's spelling are the same string by construction rather than by two
38/// people keeping two lists aligned.
39macro_rules! wire_tokens {
40    (
41        $(#[$enum_meta:meta])*
42        $name:ident($kind:literal) {
43            $($(#[$meta:meta])* $variant:ident = $token:literal),* $(,)?
44        }
45    ) => {
46        $(#[$enum_meta])*
47        #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
48        pub enum $name {
49            $($(#[$meta])* $variant,)*
50        }
51
52        impl WireToken for $name {
53            const ALL: &'static [Self] = &[$(Self::$variant,)*];
54            const KIND: &'static str = $kind;
55
56            fn as_wire(self) -> &'static str {
57                match self {
58                    $(Self::$variant => $token,)*
59                }
60            }
61
62            fn from_wire(token: &str) -> Option<Self> {
63                match token {
64                    $($token => Some(Self::$variant),)*
65                    _ => None,
66                }
67            }
68        }
69
70        impl $name {
71            /// Every variant, in wire order (the inherent form of
72            /// [`WireToken::ALL`], so callers need no trait import).
73            pub const ALL: &'static [Self] = <Self as WireToken>::ALL;
74
75            /// The backend's token for this variant.
76            pub fn as_wire(self) -> &'static str {
77                <Self as WireToken>::as_wire(self)
78            }
79
80            /// Parse a wire token; `None` for one this build does not know.
81            pub fn from_wire(token: &str) -> Option<Self> {
82                <Self as WireToken>::from_wire(token)
83            }
84        }
85
86        impl Serialize for $name {
87            fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
88                s.serialize_str(self.as_wire())
89            }
90        }
91
92        impl<'de> Deserialize<'de> for $name {
93            fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
94                let token = String::deserialize(d)?;
95                Self::from_wire(&token)
96                    .ok_or_else(|| serde::de::Error::unknown_variant(&token, &[$($token,)*]))
97            }
98        }
99    };
100}

139impl Permission {
140    /// The permissions a **channel overwrite** may allow or deny
141    /// ([`SpaceOverwrites`]), in wire order.
142    ///
143    /// A strict subset of [`ALL`](Self::ALL): the four that are meaningful
144    /// per-channel. The management permissions are community-wide by
145    /// construction — "may rename channels" is not a fact about one channel —
146    /// and the backend refuses an overwrite naming one, so the editor must only
147    /// offer these.
148    pub const OVERWRITABLE: &'static [Self] = &[
149        Self::ViewChannels,
150        Self::SendMessages,
151        Self::ReadHistory,
152        Self::ManageMessages,
153    ];

155    /// Whether this permission may be overwritten per-channel (see
156    /// [`OVERWRITABLE`](Self::OVERWRITABLE)).
157    pub fn is_overwritable(self) -> bool {
158        Self::OVERWRITABLE.contains(&self)
159    }
160}

214/// Something changed about a community's membership or roles that no snapshot
215/// stream carries.
216///
217/// Roles, member pages and the ban list are *fetched*, not reduced: they are
218/// paginated or settings-only and rarely looked at, so keeping a live reducer
219/// for them would cost every client a copy of data almost nobody has open.
220/// Instead the realtime plane's role/member/ban events become one of these, and
221/// whoever is showing that data refetches. The signal names the community
222/// only — deliberately: the consumer re-reads the page it is on, which is
223/// correct whatever the change was.
224/// **`seq` makes two otherwise identical signals different values.** Every
225/// signal carries a process-monotonic sequence number, stamped where it is
226/// published. Without it, "roles changed in community X" twice in a row is the
227/// same value twice, and a latest-wins consumer — Riverpod's `AsyncValue`
228/// compares states for equality and drops an equal one — sees the second edit
229/// as nothing happening at all. Nobody reads the number; its only job is to
230/// make the values unequal, and its only rule is that it never repeats.
231#[derive(Clone, Copy, Debug, PartialEq, Eq)]
232pub enum CommunitySignal {
233    /// A role was created/renamed/deleted, or its permissions or members changed.
234    RolesChanged { seq: u64, community_id: Uuid },
235    /// Someone joined, left, or was kicked.
236    MembersChanged { seq: u64, community_id: Uuid },
237    /// Someone was banned or unbanned.
238    ///
239    /// Separate from [`MembersChanged`](Self::MembersChanged) even though a ban
240    /// also removes a member: the two lists are different pages, and a moderator
241    /// looking at the ban list must not have to guess which signal covers it.
242    /// A ban emits both — the member list lost a row, and the ban list gained one.
243    BansChanged { seq: u64, community_id: Uuid },
244    /// Signals were LOST — a subscriber fell behind the bounded channel — so
245    /// every fetched page is now of unknown freshness.
246    ///
247    /// Names no community because it cannot: what was dropped is exactly what
248    /// nobody can enumerate. Consumers treat it as naming all of them and
249    /// re-read; it is rare by construction (the backlog is sized for a burst of
250    /// edits) and re-reading a handful of pages is cheaper than showing one
251    /// that is silently stale for the rest of the session.
252    Resync { seq: u64 },
253}
254
255impl CommunitySignal {
256    /// The community this signal is about, or `None` for a
257    /// [`Resync`](Self::Resync) — which is about all of them.
258    pub fn community_id(self) -> Option<Uuid> {
259        match self {
260            Self::RolesChanged { community_id, .. }
261            | Self::MembersChanged { community_id, .. }
262            | Self::BansChanged { community_id, .. } => Some(community_id),
263            Self::Resync { .. } => None,
264        }
265    }
266
267    /// This signal's sequence number — see the type docs.
268    pub fn seq(self) -> u64 {
269        match self {
270            Self::RolesChanged { seq, .. }
271            | Self::MembersChanged { seq, .. }
272            | Self::BansChanged { seq, .. }
273            | Self::Resync { seq } => seq,
274        }
275    }
276}

292/// One invite link to a community, as its organisers see it in the invite list.
293///
294/// `code` is the shareable token (8 Crockford-base32 characters — see
295/// [`parse_invite_code`](crate::parse_invite_code)) and `url` is the
296/// server-minted link that carries it; the UI copies `url` and shows `code`.
297///
298/// **[`state`](Self::state) is the only thing to gate on.** `expires_at`,
299/// `max_uses`/`uses` and `revoked` are the *inputs* the server resolved it
300/// from, kept so the list can explain itself ("3 of 10 uses, expires Friday") —
301/// but re-deriving the verdict from them client-side would put this client's
302/// clock in charge of whether an invite works, and it is not.
303#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
304pub struct Invite {
305    pub id: Uuid,
306    pub community_id: Uuid,
307    /// The shareable token, uppercase.
308    pub code: String,
309    /// The full link the server minted for `code` — what a "Copy link" action copies.
310    pub url: String,
311    /// Who minted it, or `None` once that account is gone — an invite outlives
312    /// its creator, so the list renders "someone" rather than dropping the row.
313    pub created_by: Option<Uuid>,
314    /// Unix ms.
315    pub created_at: i64,
316    /// Unix ms, or `None` for an invite that never expires.
317    pub expires_at: Option<i64>,
318    /// Use cap, or `None` for unlimited.
319    pub max_uses: Option<u32>,
320    pub uses: u32,
321    pub revoked: bool,
322    /// The server's resolved verdict — see the type docs.
323    pub state: InviteState,
324}

498    pub attachments: Vec<Attachment>,
499    /// Link previews for the URLs in `contents`, in the order they appear.

503    /// `MessageEmbeds` event fills them in a moment later. The UI must treat an
504    /// empty list as "not yet", not as "never".
505    ///
506    /// Every media URL inside is **relative to the API base** (see
507    /// [`hound_types::EmbedMedia::url`]); the view layer resolves it. Nothing

602    /// Other participants already in the channel at join time (render before media).
603    pub peers: Vec<VoicePeer>,
604    /// iroh node id (hex public key) of the SFU. `None` when the backend has no SFU
605    /// configured (dev) — clients skip the direct QUIC connection in that case.
606    pub sfu_node_id: Option<String>,
607}

609/// A participant already in a voice channel, as reported at join time.
610#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
611pub struct VoicePeer {
612    pub user_id: Uuid,
613    pub name: Option<String>,
614    pub picture: String,
615    /// The peer's SFU/control-plane peer id; `None` until they confirm their SFU join.
616    pub sfu_peer_id: Option<u16>,
617}

619// ─── Message search ─────────────────────────────────────────────────────────────
620
621/// Which messages a search may look at.
622///
623/// The default is the community the user is looking at; deleting the scope pill
624/// widens it to every community they are a member of. `in:` inside the query
625/// narrows *within* the scope — it never widens it, and the backend structurally
626/// excludes anything unreadable either way.
627#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
628pub enum SearchScope {
629    Community(Uuid),
630    Everywhere,
631}
632
633/// How a page of hits is ordered.
634///
635/// `Relevant` is only meaningful with at least one positive text term for the
636/// index to score against ([`hound_search::Query::has_positive_text`]), which is
637/// why the UI disables it otherwise.
638#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
639pub enum SearchSort {
640    Newest,
641    Relevant,
642}
643
644/// One `POST /search` call.
645///
646/// `query` is [`hound_search::Query`] itself, not a re-declaration: client and
647/// backend run the same parser crate, so the filter the composer highlighted is
648/// byte-for-byte the filter the SQL applies.
649#[derive(Clone, Debug, PartialEq)]
650pub struct SearchRequest {
651    pub scope: SearchScope,
652    pub query: hound_search::Query,
653    pub sort: SearchSort,
654    /// Opaque keyset cursor from a previous page's
655    /// [`SearchPage::next_cursor`]; `None` asks for the first page.
656    pub cursor: Option<String>,
657    /// Hits per page, 1..=50 (the backend rejects anything else as a 400).
658    pub limit: u32,
659    /// Minutes east of UTC. `before:`/`after:`/`during:`/`today` resolve to
660    /// *the reader's* midnight, so the offset travels with the request — the
661    /// server never reads its own clock for a date filter.
662    pub utc_offset_minutes: i32,
663}
664
665/// One search hit: the message, the community it lives in, its relevance score,
666/// and the author's display identity.
667///
668/// `author` is joined here rather than left to the roster because a hit can come
669/// from a community whose members the client has never loaded — the wire hoists
670/// the profiles into one `users` array and this is where they land back on their
671/// messages. An author the server did not ship falls back to a bare
672/// [`Member`] (no name, no avatar), which the UI already renders as a short id.
673#[derive(Clone, Debug, PartialEq)]
674pub struct SearchHit {
675    pub message: Message,
676    pub community_id: Uuid,
677    /// Higher is better. `0.0` when the query had nothing to score against.
678    pub score: f64,
679    pub author: Member,
680}
681
682/// A page of search hits. There is no total: the cursor is a keyset, so "how
683/// many more" is a question the backend deliberately cannot answer cheaply.
684#[derive(Clone, Debug, Default, PartialEq)]
685pub struct SearchPage {
686    pub hits: Vec<SearchHit>,
687    /// Cursor for the next page, or `None` at the end of the results.
688    pub next_cursor: Option<String>,
689}

695    /// The one property [`wire_tokens!`] exists to guarantee: the REST layer's
696    /// spelling and serde's are the SAME string. Asserted through serde, which
697    /// is the half that used to be written separately — and drifted.
698    fn tokens_agree<T>()
699    where
700        T: WireToken + std::fmt::Debug + PartialEq + Serialize + for<'de> Deserialize<'de>,
701    {
702        for &token in T::ALL {
703            let wire = token.as_wire();
704            assert_eq!(
705                T::from_wire(wire),
706                Some(token),
707                "{wire} does not parse back"
708            );
709            assert_eq!(
710                serde_json::to_value(token).unwrap(),
711                serde_json::Value::String(wire.to_owned()),
712                "serde spells {wire} differently",
713            );
714            assert_eq!(
715                serde_json::from_value::<T>(serde_json::json!(wire)).unwrap(),
716                token,
717            );
718        }
719        assert_eq!(
720            T::from_wire("a_token_from_a_newer_backend"),
721            None,
722            "an unknown token parses as nothing, so callers can drop it",
723        );
724    }
725
726    #[test]
727    fn every_wire_token_has_one_spelling() {
728        tokens_agree::<Permission>();
729        tokens_agree::<InstancePermission>();
730        tokens_agree::<InviteState>();
731        tokens_agree::<SpaceKind>();
732    }
733
734    /// The channel-overwrite subset is a SUBSET, and `is_overwritable` agrees
735    /// with it — a community-wide permission in an overwrite table is a row the
736    /// backend refuses.
737    #[test]
738    fn only_channel_scoped_permissions_are_overwritable() {
739        for &p in Permission::ALL {
740            assert_eq!(
741                p.is_overwritable(),
742                Permission::OVERWRITABLE.contains(&p),
743                "{p:?}",
744            );
745        }
746        assert!(!Permission::ManageCommunity.is_overwritable());
747        assert!(Permission::ViewChannels.is_overwritable());
748    }
