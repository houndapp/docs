1//! Auth/session lifecycle: where tokens live, how they persist, and how they refresh.
2//!
3//! The crate must NOT depend on any platform keyring, so persistence is injected via
4//! the [`SessionStore`] trait — the host (Flutter via FRB) supplies a keyring-backed
5//! impl; tests use [`MemorySessionStore`]. [`Session`] owns the live tokens, tracks an
6//! [`AuthState`] snapshot for the reducer, and republishes the access token onto the
7//! shared [`RestClient`] so every request reads the current value.
8//!
9//! The access token is short-lived (45 min), so it is rotated from three directions,
10//! all funnelled through ONE `refresh_gate` mutex — the single-flight guarantee, so a
11//! burst of simultaneous expiries costs exactly one exchange:
12//!
13//! * **Proactive** — a 30 s-tick background task refreshes once the token passes its
14//!   half-life ([`refresh_due`]). Started by every path that installs a live session,
15//!   aborted on sign-out.
16//! * **On demand** — [`RestClient`] calls [`Session::reauth_for`] through the
17//!   [`Reauth`] hook installed here when a request comes back `401 expiredtoken`, then
18//!   retries it once. This is the net under a suspended laptop / a missed tick.
19//! * **App resume** — [`Session::ensure_fresh`] (the facade's
20//!   `HoundClient::ensure_session_fresh`), for a wake-up where neither of the above
21//!   has run yet.
22//!
23//! A refresh whose *refresh* token is rejected is unrecoverable: the session is signed
24//! out. Anything else (offline, a 5xx) is left alone — the next tick or request retries.
25

38/// How often the proactive task re-checks whether the access token is past its
39/// half-life. Deliberately far shorter than the 45 min lifetime it guards: the check
40/// is a WALL-CLOCK comparison, so a short tick self-corrects across a laptop suspend
41/// (a single long timer armed for the half-life either fires hours late or, on some
42/// platforms, not at all — and the token is dead by then).
43const REFRESH_TICK: Duration = Duration::from_secs(30);
44
45/// The window before `exp` in which a refresh is due for a token we can't compute a
46/// half-life for (no readable `iat`). Wide enough to survive a missed tick or two.
47const REFRESH_FALLBACK_LEAD: i64 = 10 * 60;
48
49/// Platform-supplied persistence for the session tokens. The host implements this
50/// over its secure store (keyring/keychain/keystore); `hound-client` never links one.
51///
52/// `Send + Sync` so a `Session` holding `Arc<dyn SessionStore>` can move across tasks.
53pub trait SessionStore: Send + Sync {
54    /// Load the persisted tokens, if a session was saved. `None` ⇒ signed out.
55    fn load(&self) -> Result<Option<StoredTokens>>;
56    /// Persist the access + refresh tokens (called after login + every refresh).
57    fn save(&self, access: &str, refresh: &str) -> Result<()>;
58    /// Drop the persisted session (sign-out / unrecoverable refresh failure).
59    fn clear(&self) -> Result<()>;
60}
61
62/// The persisted token pair.
63#[derive(Clone, Debug, PartialEq, Eq)]
64pub struct StoredTokens {
65    pub access_token: String,
66    pub refresh_token: String,
67}

69/// An in-memory [`SessionStore`] — the default for tests and headless runs with no
70/// platform store. Cheap to clone; clones share one slot.
71#[derive(Clone, Default)]
72pub struct MemorySessionStore {
73    slot: Arc<ArcSwapOption<StoredTokens>>,
74}
75
76impl MemorySessionStore {
77    pub fn new() -> Self {
78        Self::default()
79    }
80}
81
82impl SessionStore for MemorySessionStore {
83    fn load(&self) -> Result<Option<StoredTokens>> {
84        Ok(self.slot.load_full().map(|t| (*t).clone()))
85    }
86    fn save(&self, access: &str, refresh: &str) -> Result<()> {
87        self.slot.store(Some(Arc::new(StoredTokens {
88            access_token: access.to_owned(),
89            refresh_token: refresh.to_owned(),
90        })));
91        Ok(())
92    }
93    fn clear(&self) -> Result<()> {
94        self.slot.store(None);
95        Ok(())
96    }
97}

99/// Owns the live session: the tokens, the [`AuthState`] snapshot, and the refresh
100/// machinery. Shares an `Arc<RestClient>` so installing the access token here makes it
101/// visible to every REST call. Cheap to clone (an `Arc` bump — the same shape as the
102/// backend's `Api`); every clone drives the one session.
103#[derive(Clone)]
104pub struct Session(Arc<SessionInner>);
105
106struct SessionInner {
107    rest: Arc<RestClient>,
108    store: Arc<dyn SessionStore>,
109    /// Current tokens, swapped atomically on login/refresh/clear.
110    tokens: ArcSwapOption<StoredTokens>,
111    /// Current auth snapshot for the reducer (signed-in user id, or signed-out).
112    state: ArcSwapOption<AuthState>,
113    /// Single-flight gate over the whole exchange (refresh call → install → persist).
114    /// Held across the network round-trip on purpose: a second refresher must not
115    /// race a rotation, it must *wait* for it and then find its token already fresh.
116    refresh_gate: tokio::sync::Mutex<()>,
117    /// The proactive refresh task, while a session is live. A `std::sync::Mutex`
118    /// because every use is a swap-and-abort with no await in between.
119    proactive: std::sync::Mutex<Option<JoinHandle<()>>>,
120}
121
122impl Session {
123    /// Build a session bound to `rest` and `store`, and install this session as
124    /// `rest`'s refresh hook. Does not touch the store — call
125    /// [`restore_with`](Self::restore_with) to rehydrate a persisted session at startup.
126    pub fn new(rest: Arc<RestClient>, store: Arc<dyn SessionStore>) -> Self {
127        let inner = Arc::new(SessionInner {
128            rest,
129            store,
130            tokens: ArcSwapOption::empty(),
131            state: ArcSwapOption::empty(),
132            refresh_gate: tokio::sync::Mutex::new(()),
133            proactive: std::sync::Mutex::new(None),
134        });
135        // `Weak`: the session holds the REST client, so a strong hook here would be a
136        // cycle neither side could ever drop.
137        inner
138            .rest
139            .set_reauth(Box::new(SessionReauth(Arc::downgrade(&inner))));
140        Self(inner)
141    }

143    /// The current auth snapshot ([`AuthState::SignedOut`] until a session is set).
144    pub fn auth_state(&self) -> AuthState {
145        self.0
146            .state
147            .load_full()
148            .map(|s| (*s).clone())
149            .unwrap_or(AuthState::SignedOut)
150    }

152    /// Start the credential-store read on the blocking pool, handing back its handle.
153    ///
154    /// The read is a separate step from [`restore_with`](Self::restore_with) because it
155    /// can sit for *minutes*: [`SessionStore::load`] is synchronous, and on macOS the
156    /// Keychain raises a modal "… wants to access …" prompt whenever the stored item's
157    /// ACL doesn't match the running binary's code signature — the everyday case when
158    /// the user switches between a locally built app and a signed nightly (see
159    /// `hound_passkey::storage`). A blocking call is not an await point, so no
160    /// `timeout` can bound it in place; spawning it is what makes the *wait*
161    /// boundable while the read itself runs to completion on a pool thread.
162    ///
163    /// A caller that gives up waiting must keep the handle: dropping it does not
164    /// cancel the read, and the tokens are still worth adopting once the user
165    /// finally answers the prompt.
166    pub fn load_stored(&self) -> JoinHandle<Result<Option<StoredTokens>>> {
167        let store = self.0.store.clone();
168        tokio::task::spawn_blocking(move || store.load())
169    }
170
171    /// Rehydrate a session from tokens already read out of the store (see
172    /// [`load_stored`](Self::load_stored)). If tokens are present they are
173    /// installed onto the REST client and a refresh is performed to obtain a fresh
174    /// access token (the persisted access token may have expired while the app was
175    /// closed). On refresh failure the stored session is cleared and we end signed
176    /// out. Returns the resulting [`AuthState`].
177    pub async fn restore_with(&self, stored: Option<StoredTokens>) -> Result<AuthState> {
178        let Some(stored) = stored else {
179            tracing::info!("session restore: no stored session — starting signed out");
180            self.set_signed_out();
181            return Ok(AuthState::SignedOut);
182        };
183        tracing::info!("session restore: stored tokens found — refreshing");
184        // Install what we have so the immediate refresh can authenticate, then rotate.
185        self.install(&stored);
186        match self.refresh().await {
187            // The refresh rotated the access token but a bare refresh response carries
188            // no user id; recover it from the fresh token's `sub` claim so we end up
189            // genuinely SignedIn (not the SignedOut default).
190            Ok(()) => match self.current_user_id() {
191                Some(user_id) => {
192                    tracing::info!(%user_id, "session restore: refreshed — signed in");
193                    self.set_signed_in(user_id);
194                    self.spawn_proactive_refresh();
195                    Ok(AuthState::SignedIn { user_id })
196                }
197                // A valid token we can't read a subject from is unusable to the UI —
198                // treat it as a dead session rather than a half-signed-in limbo.
199                None => {
200                    tracing::warn!("session restore: refreshed token has no subject — signing out");
201                    self.sign_out().await?;
202                    Ok(AuthState::SignedOut)
203                }
204            },
205            Err(e) => {
206                // Refresh token rejected/expired — drop the dead session.
207                tracing::warn!(error = %e, "session restore: refresh failed — signing out");
208                self.sign_out().await?;
209                Ok(AuthState::SignedOut)
210            }
211        }
212    }

214    /// The signed-in user id decoded from the current access token's `sub` claim, if a
215    /// token is installed and parseable.
216    fn current_user_id(&self) -> Option<Uuid> {
217        let tokens = self.0.tokens.load_full()?;
218        user_id_from_jwt(&tokens.access_token)
219    }

221    /// Adopt a freshly established session from a login/register `finish` call:
222    /// install the tokens, persist them, set the signed-in state.
223    pub fn adopt_login(&self, login: LoginSession) -> Result<AuthState> {
224        let tokens = StoredTokens {
225            access_token: login.access_token,
226            refresh_token: login.refresh_token,
227        };
228        self.install(&tokens);
229        self.persist(&tokens);
230        let state = AuthState::SignedIn {
231            user_id: login.user_id,
232        };
233        self.0.state.store(Some(Arc::new(state.clone())));
234        self.spawn_proactive_refresh();
235        Ok(state)
236    }

238    /// Adopt a bare token pair (the browser-deferred login flow: the loopback
239    /// callback carries only the tokens, no user id / profile). The user id comes
240    /// from the fresh access token's `sub` claim — a pair we can't read a subject
241    /// from is unusable to the UI, so it is rejected rather than half-adopted.
242    pub fn adopt_tokens(&self, access_token: &str, refresh_token: &str) -> Result<AuthState> {
243        let user_id = user_id_from_jwt(access_token).ok_or_else(|| {
244            HoundError::auth(
245                AuthCode::TokenNoSubject,
246                "browser login token has no subject",
247            )
248        })?;
249        let tokens = StoredTokens {
250            access_token: access_token.to_owned(),
251            refresh_token: refresh_token.to_owned(),
252        };
253        self.install(&tokens);
254        self.persist(&tokens);
255        let state = AuthState::SignedIn { user_id };
256        self.0.state.store(Some(Arc::new(state.clone())));
257        self.spawn_proactive_refresh();
258        Ok(state)
259    }

261    /// Exchange the current refresh token for a fresh access + rotated refresh token,
262    /// republish the access token onto the REST client, and persist. Errors (e.g. a
263    /// rejected refresh token) propagate so the caller can route to sign-out — a
264    /// *store* failure does not (see [`persist`](Self::persist)): a live session is
265    /// not worth killing over a local keychain fault.
266    ///
267    /// Unconditional: it exchanges whatever refresh token is installed. Callers that
268    /// only want a refresh *if one is due* want [`ensure_fresh`](Self::ensure_fresh).
269    pub async fn refresh(&self) -> Result<()> {
270        let _gate = self.0.refresh_gate.lock().await;
271        self.refresh_locked().await
272    }
273
274    /// The on-401 entry point (see [`Reauth`]): rotate the tokens *unless* someone
275    /// already did. `stale_access` is the token the failing request carried — if the
276    /// cell no longer holds it, a concurrent refresher won the race and the caller's
277    /// retry will pick the fresh token up, so this is a no-op with no network cost.
278    /// That check inside the gate is what collapses a burst of simultaneous 401s into
279    /// exactly one exchange.
280    pub(crate) async fn reauth_for(&self, stale_access: &str) -> Result<()> {
281        let _gate = self.0.refresh_gate.lock().await;
282        // A signed-out session (no tokens at all) also lands here: nothing to rotate,
283        // and the caller's retry will fail on its own terms rather than ours.
284        let still_stale = self
285            .0
286            .tokens
287            .load_full()
288            .is_some_and(|t| t.access_token == stale_access);
289        if !still_stale {
290            return Ok(());
291        }
292        self.refresh_locked().await
293    }
294
295    /// Refresh now if — and only if — the current access token is past its half-life
296    /// ([`refresh_due`]). The app-resume hook: a device that slept through the
297    /// proactive task's ticks gets a live token before the UI issues its first
298    /// request. No-op when signed out or when the token is still young.
299    pub async fn ensure_fresh(&self) -> Result<()> {
300        let Some(tokens) = self.0.tokens.load_full() else {
301            return Ok(());
302        };
303        if !access_token_due(&tokens.access_token) {
304            return Ok(());
305        }
306        self.reauth_for(&tokens.access_token).await
307    }

309    /// The exchange itself, with the single-flight gate ALREADY held. Never call it
310    /// without the gate, and never take the gate inside it — both callers hold it.
311    async fn refresh_locked(&self) -> Result<()> {
312        let refresh_token = self
313            .0
314            .tokens
315            .load_full()
316            .map(|t| t.refresh_token.clone())
317            .ok_or_else(|| {
318                HoundError::auth(AuthCode::NoRefreshToken, "no refresh token to refresh with")
319            })?;
320
321        let TokenRefresh {
322            access_token,
323            refresh_token: new_refresh,
324            ..
325        } = self.0.rest.refresh_token(&refresh_token).await?;
326
327        let tokens = StoredTokens {
328            access_token,
329            refresh_token: new_refresh,
330        };
331        self.install(&tokens);
332        self.persist(&tokens);
333        // A successful refresh implies the user the refresh token was minted for is
334        // still valid; preserve the signed-in user id if we already had one. (We do
335        // not learn the user id from the refresh response, so keep the existing one.)
336        Ok(())
337    }

339    /// (Re)start the proactive refresh task for the live session. Idempotent: an
340    /// existing task is aborted and replaced, so re-adopting a session never leaves
341    /// two tasks refreshing the same tokens.
342    fn spawn_proactive_refresh(&self) {
343        // `adopt_login`/`adopt_tokens` are sync and callable off a runtime (tests, a
344        // host thread), where `spawn` would panic. The on-401 path covers that case.
345        let Ok(runtime) = tokio::runtime::Handle::try_current() else {
346            tracing::debug!("session: no tokio runtime — proactive refresh not started");
347            return;
348        };
349        // `Weak`: the task must never be the reason a dropped session stays alive.
350        let weak = Arc::downgrade(&self.0);
351        let handle = runtime.spawn(async move {
352            let mut ticker = tokio::time::interval(REFRESH_TICK);
353            // `Delay`, not the default `Burst`: after a suspend the missed ticks carry
354            // no information (the predicate reads the wall clock, not the tick count),
355            // so replaying them would only spin. One tick on wake is enough.
356            ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
357            loop {
358                ticker.tick().await;
359                let Some(inner) = weak.upgrade() else { return };
360                let session = Session(inner);
361                let Some(tokens) = session.0.tokens.load_full() else {
362                    // Signed out — the session outlives us, the task does not.
363                    return;
364                };
365                if !access_token_due(&tokens.access_token) {
366                    continue;
367                }
368                match session.reauth_for(&tokens.access_token).await {
369                    Ok(()) => tracing::info!("session: access token refreshed proactively"),
370                    // The refresh token itself is dead; nothing here can recover it.
371                    // Same policy as `restore_with`: drop the session, end the task.
372                    // (`sign_out` aborts *this* task — harmless: it has no await
373                    // point left to be cancelled at, so it still runs to completion.)
374                    Err(e @ HoundError::Auth { .. }) => {
375                        tracing::warn!(error = %e, "session: proactive refresh rejected — signing out");
376                        let _ = session.sign_out().await;
377                        return;
378                    }
379                    // Offline / 5xx — the token is still whatever it was; try again
380                    // on the next tick (and the 401 path is under us either way).
381                    Err(e) => {
382                        tracing::warn!(error = %e, "session: proactive refresh failed — retrying");
383                    }
384                }
385            }
386        });
387        if let Some(prev) = self.0.proactive.lock().unwrap().replace(handle) {
388            prev.abort();
389        }
390    }

392    /// Sign out: stop the proactive task, clear the REST token, the in-memory tokens,
393    /// the persisted store, and the auth state. Idempotent.
394    pub async fn sign_out(&self) -> Result<()> {
395        if let Some(task) = self.0.proactive.lock().unwrap().take() {
396            task.abort();
397        }
398        self.0.rest.clear_token();
399        self.0.tokens.store(None);
400        self.0.store.clear()?;
401        self.set_signed_out();
402        Ok(())
403    }

405    /// Persist a token pair — **non-fatally**. A store failure is reported and
406    /// swallowed: the login/refresh itself succeeded and the in-memory session is
407    /// fully usable, so propagating here would sign the user out of a session they
408    /// just established (and, from `refresh`, would route a live session to
409    /// sign-out over a purely local storage fault).
410    ///
411    /// The known cause on macOS is a keychain item whose ACL is bound to a
412    /// differently-signed hound build — a locally built app and a signed nightly on
413    /// one machine (see [`load_stored`](Self::load_stored) and
414    /// `hound_passkey::storage`). The cost is that the session does not survive a
415    /// restart; the `error!` is what makes that visible (the Sentry bridge promotes
416    /// it), since these are one-shot calls and cannot storm.
417    fn persist(&self, tokens: &StoredTokens) {
418        if let Err(e) = self
419            .0
420            .store
421            .save(&tokens.access_token, &tokens.refresh_token)
422        {
423            tracing::error!(
424                error = %e,
425                "session: could not persist tokens to the platform credential store — \
426                 continuing with the in-memory session (it will NOT survive a restart). \
427                 Usually a keychain ACL bound to a differently-signed hound build \
428                 (e.g. a dev build and a nightly on one machine)"
429            );
430        }
431    }
432
433    /// Install a token pair onto self + the REST client (no persistence side effect).
434    fn install(&self, tokens: &StoredTokens) {
435        self.0.rest.set_token(tokens.access_token.clone());
436        self.0.tokens.store(Some(Arc::new(tokens.clone())));
437    }

452/// The [`Reauth`] hook the [`RestClient`] calls on a `401 expiredtoken`. Holds a
453/// `Weak` so the hook the REST client owns can't keep the session it belongs to alive.
454struct SessionReauth(Weak<SessionInner>);
455
456impl Reauth for SessionReauth {
457    fn reauth(&self, stale_access: String) -> futures::future::BoxFuture<'static, Result<()>> {
458        let weak = self.0.clone();
459        Box::pin(async move {
460            let inner = weak
461                .upgrade()
462                .ok_or_else(|| HoundError::auth(AuthCode::SignedOut, "signed out"))?;
463            Session(inner).reauth_for(&stale_access).await
464        })
465    }
466}

468/// The claims the client reads out of an access token. All optional: a token we
469/// can't fully parse still has to be usable (the server is the authority on
470/// validity), so each field independently degrades to `None`.
471#[derive(Debug, Default, PartialEq, Eq)]
472struct TokenClaims {
473    /// The user the token was minted for.
474    sub: Option<Uuid>,
475    /// Issue time, unix seconds — the anchor for the half-life.
476    iat: Option<i64>,
477    /// Expiry, unix seconds.
478    exp: Option<i64>,
479}
480
481/// Decode a JWT's claims **without verifying the signature**. The client trusts a
482/// token the server just minted; signature verification is the server's job on every
483/// request, not the client's — we only need to read who the token is for and how long
484/// it lives. Returns `None` if the token isn't a well-formed JWT.
485/// (Backend claim shape: `services/backend/src/api/auth/jwt.rs`.)
486fn peek_claims(token: &str) -> Option<TokenClaims> {
487    // header.payload.signature — the claims are the middle, base64url (no padding).
488    let payload_b64 = token.split('.').nth(1)?;
489    let payload = base64::engine::general_purpose::URL_SAFE_NO_PAD
490        .decode(payload_b64)
491        .ok()?;
492    let claims: serde_json::Value = serde_json::from_slice(&payload).ok()?;
493    Some(TokenClaims {
494        sub: claims
495            .get("sub")
496            .and_then(|v| v.as_str())
497            .and_then(|s| Uuid::parse_str(s).ok()),
498        iat: claims.get("iat").and_then(|v| v.as_i64()),
499        exp: claims.get("exp").and_then(|v| v.as_i64()),
500    })
501}
502
503/// The `sub` (user id) claim of an access token, if it carries a parseable one.
504fn user_id_from_jwt(token: &str) -> Option<Uuid> {
505    peek_claims(token)?.sub
506}
507
508/// Whether a token with these claims should be refreshed at `now_unix` — pure, so the
509/// whole policy is unit-testable without a clock.
510///
511/// The rule is the *half-life*: refresh once half the token's own lifetime has passed,
512/// which needs no knowledge of the backend's 45-minute constant and scales if it
513/// changes. Degradations, in order:
514/// * no usable `iat` (missing, or not before `exp` — a clock-skewed or malformed
515///   token) ⇒ fall back to a fixed [`REFRESH_FALLBACK_LEAD`] window before `exp`;
516/// * no readable `exp` ⇒ never due. We cannot reason about the token at all, and the
517///   on-401 retry is the net that catches it.
518fn refresh_due(claims: &TokenClaims, now_unix: i64) -> bool {
519    let Some(exp) = claims.exp else {
520        return false;
521    };
522    match claims.iat {
523        Some(iat) if iat < exp => now_unix >= iat + (exp - iat) / 2,
524        _ => now_unix >= exp - REFRESH_FALLBACK_LEAD,
525    }
526}
527
528/// [`refresh_due`] against the current wall clock, for a raw token. An unparseable
529/// token is never due (same reasoning as a missing `exp`).
530fn access_token_due(token: &str) -> bool {
531    let now = std::time::SystemTime::now()
532        .duration_since(std::time::UNIX_EPOCH)
533        .map(|d| d.as_secs() as i64)
534        .unwrap_or(0);
535    peek_claims(token).is_some_and(|claims| refresh_due(&claims, now))
536}

839    /// A JWT-shaped token whose payload carries the claims the client peeks at.
840    /// Nothing in these tests verifies a signature — the backend is the authority on
841    /// validity, and here the fake backend simply compares strings.
842    fn token(label: &str, lifetime_secs: i64) -> String {
843        let now = std::time::SystemTime::now()
844            .duration_since(std::time::UNIX_EPOCH)
845            .unwrap()
846            .as_secs() as i64;
847        let payload = serde_json::json!({
848            "sub": Uuid::now_v7(),
849            "iat": now,
850            "exp": now + lifetime_secs,
851            "typ": "access",
852            "label": label,
853        })
854        .to_string();
855        let b64 = base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(payload);
856        format!("eyJhbGciOiJIUzI1NiJ9.{b64}.{label}")
857    }

859    /// The backend's auth behaviour, reduced to what the retry path depends on: one
860    /// protected endpoint that rejects any bearer but the current access token, and
861    /// the refresh endpoint that mints it.
862    struct FakeBackend {
863        /// The only access token `/profile/global` accepts.
864        fresh: String,
865        /// The `code` the protected route reports for any other token.
866        reject_code: &'static str,
867        /// Whether the refresh endpoint rejects the refresh token in turn.
868        refresh_rejects: bool,
869        /// Refresh handler latency — long enough that concurrent callers genuinely
870        /// overlap, which is what makes the single-flight assertion mean something.
871        refresh_delay: Duration,
872        refreshes: AtomicUsize,
873        /// The bearer tokens the protected route saw, in arrival order.
874        seen: Mutex<Vec<String>>,
875    }
876
877    impl FakeBackend {
878        fn new(fresh: &str) -> Self {
879            Self {
880                fresh: fresh.to_owned(),
881                reject_code: "expiredtoken",
882                refresh_rejects: false,
883                refresh_delay: Duration::ZERO,
884                refreshes: AtomicUsize::new(0),
885                seen: Mutex::new(Vec::new()),
886            }
887        }
888
889        fn refreshes(&self) -> usize {
890            self.refreshes.load(Ordering::Relaxed)
891        }
892
893        fn seen(&self) -> Vec<String> {
894            self.seen.lock().unwrap().clone()
895        }
896    }
897
898    fn bearer(headers: &HeaderMap) -> String {
899        headers
900            .get(axum::http::header::AUTHORIZATION)
901            .and_then(|v| v.to_str().ok())
902            .and_then(|v| v.strip_prefix("Bearer "))
903            .unwrap_or_default()
904            .to_owned()
905    }
906
907    /// Serve `backend` on a loopback port and return its base URL.
908    async fn serve(backend: Arc<FakeBackend>) -> String {
909        async fn profile(
910            State(b): State<Arc<FakeBackend>>,
911            headers: HeaderMap,
912        ) -> (StatusCode, axum::Json<serde_json::Value>) {
913            let token = bearer(&headers);
914            b.seen.lock().unwrap().push(token.clone());
915            if token == b.fresh {
916                let body = serde_json::json!({ "user_id": Uuid::now_v7(), "name": "Aoife" });
917                return (StatusCode::OK, axum::Json(body));
918            }
919            let body = serde_json::json!({ "error": "token has expired", "code": b.reject_code });
920            (StatusCode::UNAUTHORIZED, axum::Json(body))
921        }
922
923        async fn refresh(
924            State(b): State<Arc<FakeBackend>>,
925        ) -> (StatusCode, axum::Json<serde_json::Value>) {
926            b.refreshes.fetch_add(1, Ordering::Relaxed);
927            tokio::time::sleep(b.refresh_delay).await;
928            if b.refresh_rejects {
929                let body =
930                    serde_json::json!({ "error": "token has expired", "code": "expiredtoken" });
931                return (StatusCode::UNAUTHORIZED, axum::Json(body));
932            }
933            let body = serde_json::json!({
934                "access_token": b.fresh,
935                "refresh_token": "refresh-2",
936                "expires_in": 2700,
937            });
938            (StatusCode::OK, axum::Json(body))
939        }
940
941        let app = axum::Router::new()
942            .route("/profile/global", axum::routing::get(profile))
943            .route("/auth/token/refresh", axum::routing::post(refresh))
944            .with_state(backend);
945        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
946        let addr = listener.local_addr().unwrap();
947        tokio::spawn(async move {
948            axum::serve(listener, app).await.unwrap();
949        });
950        format!("http://{addr}")
951    }

956        let rest = Arc::new(RestClient::new(&base).unwrap());

958        session.adopt_tokens(access, "refresh-1").unwrap();
959        (session, rest)

962    /// The whole point: a request that dies on an expired access token refreshes once
963    /// and is replayed with the NEW token — invisibly to the caller.
964    #[tokio::test(flavor = "multi_thread")]
965    async fn an_expired_401_refreshes_once_and_retries_with_the_new_token() {
966        let stale = token("stale", 2700);
967        let fresh = token("fresh", 2700);
968        let backend = Arc::new(FakeBackend::new(&fresh));
969        let (session, rest) = signed_in(backend.clone(), &stale).await;
970
971        let profile = rest.get_profile().await.expect("the retry should succeed");
972        assert_eq!(profile.name.as_deref(), Some("Aoife"));
973        assert_eq!(backend.refreshes(), 1, "exactly one refresh exchange");
974        assert_eq!(
975            backend.seen(),
976            vec![stale, fresh.clone()],
977            "the retry must carry the rotated token, not the dead one",
978        );
979        // The rotation is installed everywhere, not just on the one retried request.
980        assert_eq!(rest.token().as_deref(), Some(fresh.as_str()));
981        assert_eq!(
982            session.0.tokens.load_full().unwrap().refresh_token,
983            "refresh-2",
984            "the refresh token rotates with it",
985        );

1018    /// Single-flight: a burst of requests all meeting the same expiry costs ONE
1019    /// exchange. Without the gate each would rotate the tokens in turn, and every
1020    /// rotation but the last would invalidate the others' freshly minted refresh token.
1021    #[tokio::test(flavor = "multi_thread")]
1022    async fn concurrent_expired_requests_share_one_refresh() {
1023        let mut backend = FakeBackend::new(&token("fresh", 2700));
1024        // Hold the exchange open long enough that the second caller is guaranteed to
1025        // arrive while the first is still mid-flight.
1026        backend.refresh_delay = Duration::from_millis(100);
1027        let backend = Arc::new(backend);
1028        let (_session, rest) = signed_in(backend.clone(), &token("stale", 2700)).await;
1029
1030        let (a, b) = tokio::join!(rest.get_profile(), rest.get_profile());
1031        assert!(a.is_ok() && b.is_ok(), "both requests should be retried");
1032        assert_eq!(backend.refreshes(), 1, "the second caller reused the first");
1033        assert_eq!(backend.seen().len(), 4, "two attempts each, no more");
1034    }
