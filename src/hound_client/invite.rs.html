1//! Turning whatever a user pasted into an invite code.
2//!
3//! One pure function, with no network and no state — but it belongs in the core
4//! rather than in a UI, because the answer must be the same on every platform
5//! and it is the one place the code's shape is written down.
6
7/// The alphabet an invite code is drawn from: Crockford base32, which is the
8/// digits and the uppercase letters MINUS `I`, `L`, `O` and `U` — the four that
9/// a human transcribing a code confuses with `1`, `1`, `0`, and a word nobody
10/// wants generated.
11const ALPHABET: &str = "0123456789ABCDEFGHJKMNPQRSTVWXYZ";
12
13/// How many characters a code has. Fixed: a code is generated, never chosen.
14const CODE_LEN: usize = 8;
15
16/// The path segment that marks the code in a link, in any of the shapes a link
17/// can arrive as (`https://hound.chat/invite/…`, a self-hosted instance's own
18/// host, or the `hound://invite/…` deep link the desktop app registers).
19const INVITE_SEGMENT: &str = "invite/";
20
21/// Extract the invite code from `input`, uppercased, or `None` if there is no
22/// code in it.
23///
24/// Accepts all three forms a code reaches us as — a bare code typed or pasted
25/// (`b3k9mn2p`), an invite URL from **any** host
26/// (`https://example.test/invite/B3K9MN2P`, trailing slash, query and fragment
27/// all tolerated), and the `hound://invite/B3K9MN2P` deep link — because the
28/// person pasting has no reason to know which one they have.
29///
30/// Deliberately strict about the code itself: exactly [`CODE_LEN`] characters
31/// from [`ALPHABET`], case-folded up. `I`/`L`/`O`/`U` are REJECTED rather than
32/// mapped onto the glyphs they resemble — a code containing one was never minted by
33/// this server, so accepting it would turn a typo into a confident 404 instead
34/// of an immediate "that isn't an invite code".
35///
36/// Total: never panics, never allocates on the rejection path.
37pub fn parse_invite_code(input: &str) -> Option<String> {
38    // A link's query/fragment is never part of the code (`?utm_source=…`, the
39    // `#` a chat client sometimes appends), and neither is surrounding space.
40    let trimmed = input.trim();
41    let candidate = trimmed.split(['?', '#']).next().unwrap_or(trimmed).trim();
42
43    // Anything after the LAST `invite/` is the code; a bare code has no such
44    // segment and is taken whole. `rsplit_once` (not `split_once`) so a host
45    // that itself contains the word — `invite.example.com/invite/CODE` — still
46    // resolves to the final segment.
47    let tail = candidate
48        .rsplit_once(INVITE_SEGMENT)
49        .map_or(candidate, |(_, tail)| tail);
50
51    // A trailing slash is normal in a copied link; an INTERIOR one means we are
52    // still looking at a path (a bare URL with no `invite/` segment), not a code.
53    let token = tail.trim_end_matches('/');
54    if token.contains('/') {
55        return None;
56    }
57    normalise(token)
58}
59
60/// Case-fold `token` up and accept it only if it is a whole, well-formed code.
61fn normalise(token: &str) -> Option<String> {
62    if token.chars().count() != CODE_LEN {
63        return None;
64    }
65    let upper = token.to_ascii_uppercase();
66    upper.chars().all(|c| ALPHABET.contains(c)).then_some(upper)
67}

69#[cfg(test)]
70mod tests {
71    use super::parse_invite_code;
72
73    /// The three forms, all landing on the same code — the whole point of the
74    /// function. Case is folded up, because a code is displayed uppercase and
75    /// typed however.
76    #[test]
77    fn every_accepted_form_yields_the_same_code() {
78        for input in [
79            "B3K9MN2P",
80            "b3k9mn2p",
81            "  b3k9mn2p  ",
82            "https://hound.chat/invite/B3K9MN2P",
83            "https://hound.chat/invite/b3k9mn2p",
84            // Any host: a self-hosted instance mints its own links.
85            "https://chat.example.test/invite/B3K9MN2P",
86            "http://localhost:8080/invite/B3K9MN2P",
87            // A trailing slash, a query and a fragment are all link noise.
88            "https://hound.chat/invite/B3K9MN2P/",
89            "https://hound.chat/invite/B3K9MN2P?ref=twitter",
90            "https://hound.chat/invite/B3K9MN2P#top",
91            // The desktop deep link.
92            "hound://invite/B3K9MN2P",
93            "hound://invite/b3k9mn2p",
94        ] {
95            assert_eq!(
96                parse_invite_code(input).as_deref(),
97                Some("B3K9MN2P"),
98                "{input:?} should parse",
99            );
100        }
101    }
102
103    /// The excluded letters are refused, not silently mapped onto `1`/`0`: a
104    /// code with one in it was never minted, so this is a typo the user can fix
105    /// now rather than a 404 they cannot explain.
106    #[test]
107    fn the_confusable_letters_are_rejected() {
108        for input in ["B3K9MN2I", "B3K9MN2L", "B3K9MN2O", "B3K9MN2U"] {
109            assert_eq!(parse_invite_code(input), None, "{input:?}");
110        }
111    }
112
113    /// Wrong length, empty, non-alphanumeric, and a URL that is not an invite
114    /// link at all.
115    #[test]
116    fn malformed_input_is_rejected() {
117        for input in [
118            "",
119            "   ",
120            "B3K9MN2",                                  // too short
121            "B3K9MN2PQ",                                // too long
122            "B3K9-MN2",                                 // punctuation is not in the alphabet
123            "b3k9 mn2p",                                // a space inside is not either
124            "https://hound.chat/B3K9MN2P",              // no `invite/` segment ⇒ still a path
125            "https://hound.chat/invite/",               // the segment with nothing after it
126            "https://hound.chat/invite/B3K9MN2P/extra", // a deeper path is not a code
127        ] {
128            assert_eq!(parse_invite_code(input), None, "{input:?}");
129        }
130    }
131
132    /// A host that itself contains the segment word resolves to the LAST one —
133    /// the path's — not the first.
134    #[test]
135    fn the_last_invite_segment_wins() {
136        assert_eq!(
137            parse_invite_code("https://invite/invite/B3K9MN2P").as_deref(),
138            Some("B3K9MN2P"),
139        );
140    }
141}
