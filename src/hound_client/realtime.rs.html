1//! The realtime plane: the IO shell around [`hound_realtime::ClientSession`].
2//!
3//! All protocol logic — the `Hello`→`Welcome` handshake, the desired-vs-issued
4//! subscription reconciliation that survives a reconnect, the frame→event
5//! mapping — lives in the sans-IO reducer. This module only moves bytes: it
6//! picks a transport, drives one [`MessagePipe`], and translates the reducer's
7//! [`Action`]s and [`Event`]s into wire frames and a typed
8//! [`RealtimeEvent`] stream.
9//!
10//! ## Two legs, one loop
11//!
12//! Per attempt the shell asks the backend where the realtime plane is
13//! (`GET /realtime/config`) and tries **WebTransport** first when a URL is
14//! advertised, falling back to a plain **WebSocket** in the SAME attempt. Both
15//! legs are [`MessagePipe`]s, so [`run_session`] is written once and
16//! instantiated twice. The credential travels in an `Authorization: Bearer`
17//! header on both legs — never in a URL, which every proxy and access log
18//! between here and the backend would keep.
19//!
20//! ## Where an auth verdict may come from
21//!
22//! Only two places: a WebSocket handshake refused **401/403**, and an `Auth`
23//! error from `GET /realtime/config` (which is trustworthy because it has
24//! already survived the REST layer's refresh-and-retry — see
25//! [`RestClient`]'s 401 hook). Everything else is retried, **including a
26//! WebTransport [`WtError::Rejected`]**: wtransport reports `SessionRejected`
27//! for any non-2xx answer to the CONNECT *and* for a CONNECT stream stopped
28//! mid-handshake, which is what a pod rollout and a misrouted host both look
29//! like. Treating that as an answer signed users out over a deploy; it is now a
30//! fallback to the WebSocket leg in the same attempt.
31//!
32//! ## The supervisor
33//!
34//! [`RealtimeClient::connect`] tries the first session inline (so a reachable
35//! backend is welcomed before the caller proceeds) and then hands it to a
36//! supervisor task that owns it. A first attempt that FAILS is not an error to
37//! the caller: it is logged and the supervisor retries, because a sign-in must
38//! not fail on a realtime plane that is merely down. The supervisor holds the
39//! [`ClientSession`] — no lock anywhere, because exactly one task touches it —
40//! and reconnects forever under the CALL plane's ladder
41//! ([`crate::call::reconnect::Backoff`]: 500ms→15s ±20% jitter, re-armed only by
42//! a session that lived ≥30s). The invariants it exists to hold:
43//!
44//! - **Never strand a stale JWT.** Every attempt re-reads the token out of
45//!   session state (through [`RestClient`], whose 401 hook rotates it), so a
46//!   reconnect after an hour offline presents a live credential rather than the
47//!   one captured at sign-in.
48//! - **Only an auth refusal is fatal.** A WS 401/403 or an `Auth` error from the
49//!   config fetch ends the supervisor and emits [`RealtimeEvent::AuthExpired`].
50//! - **A version mismatch is terminal too**, but is NOT an auth outcome: the
51//!   build cannot talk to this server, so the supervisor stops and emits
52//!   [`RealtimeEvent::VersionMismatch`] rather than storming the backend with a
53//!   handshake it will refuse identically forever.

81/// How long one WebTransport dial may take before the shell gives up and falls
82/// back to the WebSocket leg *in the same attempt*. Short on purpose: a blocked
83/// UDP path (a corporate firewall, a NAT that drops QUIC) fails by timing out,
84/// and the user must not wait out a transport-level timeout to reach a plane
85/// the WebSocket serves fine.
86const WT_CONNECT_TIMEOUT: Duration = Duration::from_secs(3);
87
88/// How long opening the WebTransport control stream may take. A session that
89/// handshakes and then never gives us a stream is indistinguishable from a wedge
90/// and would otherwise park this attempt forever — the same reason the backend
91/// bounds its `accept`.
92const WT_STREAM_TIMEOUT: Duration = Duration::from_secs(5);
93
94/// How long the WebSocket leg's dial (TCP + TLS + upgrade) may take. Mirrors
95/// [`RestClient`]'s connect timeout and its rationale verbatim: *a half-dead
96/// server (TCP accepted, upstream never answers — a churning ingress does
97/// exactly this) hangs every caller forever*, and here "every caller" is the
98/// supervisor's ladder, which would stop retrying while it waits.
99const WS_CONNECT_TIMEOUT: Duration = Duration::from_secs(10);
100
101/// Idle period after which the WebSocket leg sends a Ping. A quiet realtime
102/// connection sends nothing for minutes and NAT/proxy idle timeouts kill it; the
103/// WebTransport leg needs no equivalent because QUIC keep-alive is configured on
104/// the endpoint ([`ClientOptions::keep_alive`]).
105const WS_KEEPALIVE: Duration = Duration::from_secs(20);
106
107/// How long the WebSocket leg tolerates hearing *nothing at all* — not one
108/// frame, not one Pong — before declaring the peer gone.
109///
110/// Two keepalive periods, so a single lost probe is not fatal. This is the only
111/// thing that detects a blackholed TCP connection: writes into a dead path keep
112/// succeeding (they land in the kernel's send buffer), so a session with nothing
113/// to say would otherwise sit "connected" until the OS gave up, which can be
114/// hours. QUIC reaps its own, hence no equivalent on the WebTransport leg.
115const WS_SILENCE_LIMIT: Duration = Duration::from_secs(40);
116
117/// How long a session may sit unwelcomed after the transport is up. The backend
118/// closes a connection that has not said `Hello` within 10s; ours must be at
119/// least that (we are the ones who send it) plus room for the round trip, and
120/// anything past it is a peer that accepted a socket and then stopped
121/// progressing.
122const WELCOME_DEADLINE: Duration = Duration::from_secs(15);
123
124/// Consecutive WebTransport dial failures after which the shell stops trying it.
125/// On a network that blocks UDP outright, every reconnect otherwise pays
126/// [`WT_CONNECT_TIMEOUT`] before reaching the leg that works.
127const WT_FAILURE_STREAK: u32 = 2;
128
129/// How long the WebTransport leg stays skipped once the streak trips. A client
130/// moves between networks without restarting, so the skip must expire.
131const WT_RETRY_AFTER: Duration = Duration::from_secs(600);
132
133/// How long [`RealtimeClient::subscribe_space`] waits for the server to confirm
134/// that it joined this connection to the space's room.
135///
136/// Mirrors the app layer's `REALTIME_WELCOME_GRACE` in both value and doctrine:
137/// the wait exists so that opening a space *means* the space's events are
138/// arriving, and expiring it is a warning, never a failure — a slow realtime
139/// plane must not fail the open of a space whose history already loaded. Long
140/// enough to cover a reconnect (the supervisor re-issues subscriptions on the
141/// next `Welcome`, and the confirmation follows it), short enough that a plane
142/// which is simply down does not hold a UI open for longer than a person waits.
143const SUBSCRIBE_ACK_DEADLINE: Duration = Duration::from_secs(5);
144
145/// Delay before the one in-session retry an [`ErrorCode::Unavailable`] earns.
146/// The server could not *reach* the state to decide (a database blip), so the
147/// subscription is neither issued nor refused — asking again shortly is the
148/// whole recovery.
149const UNAVAILABLE_RETRY: Duration = Duration::from_secs(5);
150
151/// A decoded server-side realtime event. The reducers fold these into
152/// view-models.
153///
154/// One-to-one with [`hound_realtime::Event`] plus [`RealtimeEvent::AuthExpired`]
155/// (no protocol frame can produce it — it is the shell's verdict that the
156/// session credential is dead) and [`RealtimeEvent::VersionMismatch`] (split out
157/// of `Error` because it, alone among the error codes, ends the plane). Kept as
158/// its own enum rather than a re-export exactly so those variants exist and so
159/// the app layer's `match` arms don't have to name a protocol type.
160#[derive(Clone, Debug, PartialEq)]
161pub enum RealtimeEvent {
162    /// The server joined this connection to a space's room; its events are
163    /// arriving from here on.
164    Subscribed { space_id: Uuid },
165    /// A new message in a subscribed space.
166    MessageCreated(MessageEvent),
167    /// An edit to an existing message.
168    MessageUpdated(MessageEvent),
169    /// A new space in a joined community.
170    SpaceCreated(SpaceEvent),
171    /// A space was deleted.
172    SpaceRemoved { space_id: Uuid },
173    /// A user joined a voice channel (SFU peer id known).
174    VoiceJoined(VoiceJoinEvent),
175    /// A user left a voice channel.
176    VoiceLeft(VoiceLeaveEvent),
177    /// A participant toggled screenshare.
178    VoiceScreenShare(VoiceScreenShareEvent),
179    /// The full voice roster snapshot, pushed with every `Welcome` and on
180    /// server-side resync.
181    VoiceState(Vec<VoiceJoinEvent>),
182    /// The server refused something (an unauthorized subscribe, an unavailable
183    /// one). Surfaced, never swallowed, so contract drift is visible in the
184    /// logs.
185    Error { code: ErrorCode, message: String },
186    /// This build and the server cannot agree a protocol version. The plane will
187    /// NOT reconnect — retrying re-runs the same refused handshake — but the
188    /// SESSION is fine: this is never a reason to sign anyone out.
189    VersionMismatch { message: String },
190    /// The handshake completed: the session is live and its subscriptions are
191    /// re-issued.
192    ///
193    /// `reconnect` distinguishes the first connect of a plane from every later
194    /// one. Only the latter has a gap behind it: subscriptions resync, but the
195    /// message events fanned out while this plane had no connection are gone for
196    /// good, so the app layer repairs them with a history re-fetch (see
197    /// `HoundClient::arm_resync`). Never set on the first connect, where the
198    /// caller's own `open_space` load is the history.
199    Connected { reconnect: bool },
200    /// The transport went away. The supervisor is already reconnecting; desired
201    /// subscriptions survive and are re-issued on the next `Welcome`.
202    Disconnected,
203    /// The session credential is dead — the plane will NOT reconnect. The app
204    /// layer reacts as it does to any dead session (sign out).
205    AuthExpired,
206}

208impl From<Event> for RealtimeEvent {
209    fn from(event: Event) -> Self {
210        match event {
211            Event::Connected { reconnect } => Self::Connected { reconnect },
212            Event::Disconnected => Self::Disconnected,
213            Event::Subscribed { space_id } => Self::Subscribed { space_id },
214            Event::MessageCreated(m) => Self::MessageCreated(m),
215            Event::MessageUpdated(m) => Self::MessageUpdated(m),
216            Event::SpaceCreated(s) => Self::SpaceCreated(s),
217            Event::SpaceRemoved { space_id } => Self::SpaceRemoved { space_id },
218            Event::VoiceJoined(v) => Self::VoiceJoined(v),
219            Event::VoiceLeft(v) => Self::VoiceLeft(v),
220            Event::VoiceState(v) => Self::VoiceState(v),
221            Event::Screenshare(s) => Self::VoiceScreenShare(s),
222            // The one error code that ends the plane gets its own variant, so no
223            // caller has to know which codes are terminal.
224            Event::Error {
225                code: ErrorCode::VersionMismatch,
226                message,
227            } => Self::VersionMismatch { message },
228            Event::Error { code, message } => Self::Error { code, message },
229        }
230    }
231}

233/// What the public API asks the supervisor to do. The supervisor owns the
234/// [`ClientSession`], so every caller-side intent crosses this channel rather
235/// than touching shared state.
236///
237/// Not `PartialEq` (and not `Clone`): a subscribe carries the caller's
238/// one-shot answer channel, which is a capability, not a value.
239#[derive(Debug)]
240enum Command {
241    Subscribe {
242        space_id: Uuid,
243        /// Signalled once the server has confirmed the room join. Dropping it
244        /// (an unsubscribe, or a supervisor that stopped) releases the caller
245        /// immediately rather than making it wait out its deadline.
246        ack: Option<oneshot::Sender<()>>,
247    },
248    Unsubscribe(Uuid),
249    Screenshare {
250        space_id: Uuid,
251        enabled: bool,
252    },
253    /// Say goodbye and end the supervisor.
254    Shutdown,
255}
256
257/// Callers waiting for a space's subscribe confirmation. Owned by
258/// [`supervise`], not by one session: a subscribe taken while the plane is down
259/// (or one whose connection dies before the answer) is answered by the NEXT
260/// session's confirmation, since the reducer re-issues the whole desired set on
261/// every `Welcome`.
262type Pending = HashMap<Uuid, Vec<oneshot::Sender<()>>>;
263
264/// Release everyone waiting on `space_id`.
265fn confirm(pending: &mut Pending, space_id: Uuid) {
266    for ack in pending.remove(&space_id).unwrap_or_default() {
267        let _ = ack.send(());
268    }
269}
270
271/// Release everyone waiting on anything — for a server that does not confirm
272/// subscribes at all, where waiting is waiting forever.
273fn confirm_all(pending: &mut Pending) {
274    for (_, acks) in pending.drain() {
275        for ack in acks {
276            let _ = ack.send(());
277        }
278    }
279}

281/// A live realtime plane. Cheap to hold: it is a channel to the supervisor task
282/// plus that task's handle. Dropping it aborts the supervisor; [`disconnect`]
283/// ends it gracefully.
284///
285/// [`disconnect`]: Self::disconnect
286pub struct RealtimeClient {
287    commands: mpsc::UnboundedSender<Command>,
288    supervisor: JoinHandle<()>,
289}
290
291impl RealtimeClient {
292    /// Bring the realtime plane up against `rest`'s backend, returning the live
293    /// client plus the receiver end of the typed event stream.
294    ///
295    /// Infallible by design. The first connection is attempted inline so a
296    /// reachable backend is welcomed before the caller continues, but a failure
297    /// is only logged: the supervisor owns every attempt after it, and a plane
298    /// that cannot connect yet must not fail the sign-in (or the session
299    /// restore) that asked for it. Outcomes reach the caller on the event
300    /// stream — [`RealtimeEvent::Connected`] / [`RealtimeEvent::Disconnected`],
301    /// or [`RealtimeEvent::AuthExpired`] once a credential is definitively
302    /// refused.
303    pub async fn connect(rest: Arc<RestClient>) -> (Self, mpsc::UnboundedReceiver<RealtimeEvent>) {
304        Self::start(RestDialer(rest), WtGate::new()).await
305    }
306
307    /// [`Self::connect`] over any [`Dial`] — the seam the shell's own tests
308    /// drive, so the plane's public behaviour (notably: what
309    /// [`Self::subscribe_space`] waits for) is exercised without a socket.
310    async fn start<D: Dial + 'static>(
311        dialer: D,
312        gate: WtGate,
313    ) -> (Self, mpsc::UnboundedReceiver<RealtimeEvent>) {
314        let leg = match connect_leg(&dialer, &gate).await {
315            Ok(leg) => Some(leg),
316            Err(err) => {
317                tracing::warn!(%err, "realtime: first connect attempt failed — the supervisor will keep trying");
318                None
319            }
320        };
321        let (events_tx, events_rx) = mpsc::unbounded_channel();
322        let (commands, command_rx) = mpsc::unbounded_channel();
323        let supervisor = tokio::spawn(async move {
324            supervise(dialer, gate, events_tx, command_rx, leg).await;
325        });
326        (
327            Self {
328                commands,
329                supervisor,
330            },
331            events_rx,
332        )
333    }
334
335    /// Receive a space's message events, returning once the server has
336    /// **confirmed** that this connection is in the space's room.
337    ///
338    /// The waiting is the point. Queueing the subscribe returns in microseconds,
339    /// but the room join is several hops away — a task hop onto the supervisor,
340    /// the frame, the network, the backend's session task, and its two
341    /// authorization lookups — and a message posted into the space during that
342    /// window is fanned out to a room this connection is not in yet. Nothing
343    /// re-fetches it, so it is lost outright; under load (a busy client, a
344    /// contended backend) the window stretches, which is exactly when a caller
345    /// that opened a space and immediately expected its traffic saw nothing.
346    ///
347    /// Idempotent, and safe before the session is live: the reducer records the
348    /// desire and issues it on the next `Welcome` — including the one after a
349    /// reconnect — and this call waits across that too.
350    ///
351    /// Bounded and non-fatal: a server that does not confirm subscriptions is
352    /// not waited for at all, and a confirmation that never arrives inside
353    /// [`SUBSCRIBE_ACK_DEADLINE`] is a warning rather than an error, because the
354    /// subscription remains desired and the space is still usable.
355    pub async fn subscribe_space(&self, space_id: Uuid) -> Result<()> {
356        let (ack, confirmed) = oneshot::channel();
357        self.send(Command::Subscribe {
358            space_id,
359            ack: Some(ack),
360        })?;
361        match tokio::time::timeout(SUBSCRIBE_ACK_DEADLINE, confirmed).await {
362            Ok(Ok(())) => {}
363            // The sender was dropped: the plane stopped, or the space was
364            // unsubscribed under us. Either way there is nothing left to wait
365            // for, and the desire (if any) survives in the reducer.
366            Ok(Err(_)) => tracing::debug!(%space_id, "realtime: subscribe wait released early"),
367            Err(_) => tracing::warn!(
368                %space_id,
369                deadline = ?SUBSCRIBE_ACK_DEADLINE,
370                "realtime: no subscribe confirmation within the deadline — continuing; the \
371                 subscription stays desired and is re-issued on the next welcome"
372            ),
373        }
374        Ok(())
375    }
376
377    /// Stop receiving a space's message events.
378    pub async fn unsubscribe_space(&self, space_id: Uuid) -> Result<()> {
379        self.send(Command::Unsubscribe(space_id))
380    }
381
382    /// Announce a screenshare toggle; the backend re-broadcasts it to the
383    /// community. Dropped while the session is down — it is a momentary state
384    /// change, and the next `Welcome`'s roster snapshot is authoritative anyway.
385    pub async fn set_screenshare(&self, space_id: Uuid, enabled: bool) -> Result<()> {
386        self.send(Command::Screenshare { space_id, enabled })
387    }
388
389    /// Say goodbye and tear the plane down. Bounded: a wedged transport must not
390    /// hold up a sign-out, so an unresponsive supervisor is aborted instead (by
391    /// this value's own `Drop`, on the way out).
392    pub async fn disconnect(mut self) -> Result<()> {
393        let _ = self.commands.send(Command::Shutdown);
394        if tokio::time::timeout(SHUTDOWN_GRACE, &mut self.supervisor)
395            .await
396            .is_err()
397        {
398            tracing::warn!("realtime: supervisor did not stop in time — aborting");
399        }
400        Ok(())
401    }
402
403    /// Queue a command. A closed channel means the supervisor already gave up
404    /// (a fatal auth outcome or a version mismatch), which is a dead plane, not
405    /// a transient blip.
406    fn send(&self, command: Command) -> Result<()> {
407        self.commands.send(command).map_err(|_| {
408            HoundError::realtime(RealtimeCode::NotRunning, "realtime plane is not running")
409        })
410    }
411}

413impl Drop for RealtimeClient {
414    fn drop(&mut self) {
415        // A dropped handle must not leave a task holding a socket open. Harmless
416        // after `disconnect` (the task has already finished).
417        self.supervisor.abort();
418    }
419}

421/// How long [`RealtimeClient::disconnect`] waits for the goodbye to flush.
422const SHUTDOWN_GRACE: Duration = Duration::from_secs(2);

426/// Own the session for the plane's lifetime: run it, reconnect it, and stop on a
427/// shutdown, a fatal auth outcome, or a version mismatch. `initial` is the
428/// connection [`RealtimeClient::connect`] already established, if it managed to.
429async fn supervise<D: Dial>(
430    dialer: D,
431    gate: WtGate,
432    events: mpsc::UnboundedSender<RealtimeEvent>,
433    mut commands: mpsc::UnboundedReceiver<Command>,
434    initial: Option<D::Pipe>,
435) {
436    let mut session = ClientSession::new();
437    let mut backoff = Backoff::new();
438    let mut leg = initial;
439    // Outlives every session: a subscribe taken while down, or one whose
440    // connection died before the answer, is confirmed by the next `Welcome`'s
441    // re-issue. Dropped when this task returns, which releases every waiter
442    // rather than making each wait out its own deadline against a dead plane.
443    let mut pending = Pending::new();
444
445    loop {
446        // Establish, unless we are holding a connection already. Commands are
447        // serviced throughout: while down they only move the reducer's desired
448        // set, which is exactly what must survive an outage.
449        let pipe = match leg.take() {
450            Some(pipe) => pipe,
451            None => {
452                let delay = backoff.next_delay(rng_jitter);
453                tracing::info!(
454                    attempt = backoff.attempt(),
455                    delay_ms = delay.as_millis() as u64,
456                    "realtime: reconnect backoff"
457                );
458                // `biased` here (unlike in `run_session`): a shutdown must win
459                // over both the sleep and the dial, or a sign-out would wait out
460                // a 15s backoff rung and then a 13s dial.
461                tokio::select! {
462                    biased;
463                    _ = drain_offline(&mut commands, &mut session, &mut pending) => return,
464                    _ = tokio::time::sleep(delay) => {}
465                }
466                let attempt = tokio::select! {
467                    biased;
468                    _ = drain_offline(&mut commands, &mut session, &mut pending) => return,
469                    result = connect_leg(&dialer, &gate) => result,
470                };
471                match attempt {
472                    Ok(pipe) => pipe,
473                    Err(err) => match classify(&err) {
474                        Disposition::Fatal => {
475                            tracing::error!(%err, "realtime: session refused — giving up");
476                            let _ = events.send(RealtimeEvent::AuthExpired);
477                            return;
478                        }
479                        Disposition::Retryable => {
480                            tracing::warn!(%err, "realtime: connect attempt failed (retrying)");
481                            continue;
482                        }
483                    },
484                }
485            }
486        };
487
488        // Live. `run_session` returns when the pipe dies or a shutdown lands.
489        let started = Instant::now();
490        let stopped = run_session(pipe, &mut session, &events, &mut commands, &mut pending).await;
491        // Read BEFORE `disconnected()` clears it: a session that reached
492        // `Welcome` proves this network path works end to end, which is what
493        // re-arms the WebTransport probe.
494        if session.is_live() {
495            gate.note_welcomed_session();
496        }
497        for event in session.disconnected() {
498            let _ = events.send(event.into());
499        }
500        // Only a session that PROVED durable re-arms the ladder — see
501        // `Backoff::on_session_death`. A server that accepts and immediately
502        // closes must not pin us on the 500ms rung.
503        backoff.on_session_death(started.elapsed());
504        match stopped {
505            Stopped::Shutdown => return,
506            // Reconnecting would re-run a handshake the server refuses
507            // identically forever; the event already told the app.
508            Stopped::VersionMismatch => {
509                tracing::error!("realtime: protocol version mismatch — the plane is stopping");
510                return;
511            }
512            Stopped::Transport => {}
513        }
514    }
515}

517/// Apply commands to the reducer while there is no live transport, forever.
518/// Resolves ONLY when the plane is shutting down (an explicit `Shutdown` or the
519/// last [`RealtimeClient`] dropped), which is what makes it usable as the losing
520/// arm of a `select!`: the actions it would produce are all empty while down, so
521/// dropping the future mid-poll loses nothing.
522async fn drain_offline(
523    commands: &mut mpsc::UnboundedReceiver<Command>,
524    session: &mut ClientSession,
525    pending: &mut Pending,
526) {
527    while let Some(command) = commands.recv().await {
528        if matches!(command, Command::Shutdown) {
529            return;
530        }
531        // Down: `apply` records desire and yields no frames to drop. A subscribe
532        // taken here parks its waiter, which the next `Welcome`'s re-issue and
533        // its confirmation release.
534        let _ = apply(session, command, pending);
535    }
536}

538/// Why [`run_session`] returned.
539#[derive(Debug, Clone, Copy, PartialEq, Eq)]
540enum Stopped {
541    /// The transport ended (peer closed, read error, a send failed, or a
542    /// deadline this shell owns expired).
543    Transport,
544    /// The caller asked the plane to stop.
545    Shutdown,
546    /// The server and this build cannot agree a protocol version. Terminal, and
547    /// deliberately not an auth outcome.
548    VersionMismatch,
549}

551/// Drive one established pipe: handshake, then pump frames and commands until
552/// something ends it.
553///
554/// Mirrors the call plane's reader discipline — a frame that fails to serialize
555/// is skipped, never fatal, and an unparseable inbound frame is skipped by
556/// [`MessagePipeExt::recv_json`] (that is the protocol's additive-compatibility
557/// rule). Generic over the pipe so both legs and the test fake share one loop.
558///
559/// Four deadlines run alongside the two data arms, all absolute (they are NOT
560/// re-armed by an unrelated wakeup, which is exactly the bug a per-iteration
561/// `sleep` would have): the keepalive probe, the inbound-silence limit, the
562/// `Welcome` deadline, and the one retry an `Unavailable` answer earns.
563async fn run_session<P: Probe>(
564    mut pipe: P,
565    session: &mut ClientSession,
566    events: &mpsc::UnboundedSender<RealtimeEvent>,
567    commands: &mut mpsc::UnboundedReceiver<Command>,
568    pending: &mut Pending,
569) -> Stopped {
570    // The transport is up: the reducer answers with `Hello`, and nothing else
571    // may be sent until the `Welcome` comes back.
572    if !send_all(&mut pipe, session.connected()).await {
573        return Stopped::Transport;
574    }
575
576    // Read once: the timers must not borrow the pipe, which the frame arm needs
577    // mutably.
578    let idle = pipe.idle_period();
579    let silence = pipe.silence_limit();
580    let mut next_probe = deadline(idle);
581    let mut next_silence_check = deadline(silence);
582    let mut inbound_mark = pipe.inbound_count();
583    let mut welcome_by = deadline(Some(WELCOME_DEADLINE));
584    let mut retry_subscriptions_at: Option<Instant> = None;
585    let mut retried_subscriptions = false;
586
587    let stopped = loop {
588        // NOT `biased`: with a fixed order the inbound arm is always polled
589        // first, so a server flooding events starves the command, keepalive and
590        // deadline arms — a shutdown would never be seen and the liveness timers
591        // would never fire. Random order costs nothing and cannot starve.
592        tokio::select! {
593            frame = pipe.recv_json::<ServerMessage>() => {
594                let Some(msg) = frame else { break Stopped::Transport };
595                let reaction = session.handle(msg);
596                let mismatch = reaction.events.iter().any(|event| {
597                    matches!(event, Event::Error { code: ErrorCode::VersionMismatch, .. })
598                });
599                let unavailable = reaction.events.iter().any(|event| {
600                    matches!(event, Event::Error { code: ErrorCode::Unavailable, .. })
601                });
602                for event in reaction.events {
603                    // A confirmed subscribe releases whoever asked for it. Done
604                    // here rather than in the reducer because the waiters are a
605                    // property of this shell's callers, not of the protocol.
606                    if let Event::Subscribed { space_id } = event {
607                        confirm(pending, space_id);
608                    }
609                    let _ = events.send(event.into());
610                }
611                if mismatch {
612                    break Stopped::VersionMismatch;
613                }
614                // Welcomed: the deadline has done its job.
615                if session.is_live() {
616                    welcome_by = None;
617                    // A server that does not confirm subscribes will never
618                    // answer these, and the `Welcome` is the first moment we
619                    // know that. Waiting them out would stall every open of a
620                    // space against an older backend.
621                    if !session.confirms_subscriptions() {
622                        confirm_all(pending);
623                    }
624                }
625                // The server decided nothing, so the subscription is neither
626                // issued nor refused. Exactly one retry per session — a backend
627                // stuck answering `Unavailable` must not earn a subscribe storm
628                // every 5s — and anything past that is the next `Welcome`'s
629                // reconciliation, which re-issues the whole desired set anyway.
630                if unavailable && !retried_subscriptions {
631                    retried_subscriptions = true;
632                    retry_subscriptions_at = deadline(Some(UNAVAILABLE_RETRY));
633                }
634                if !send_all(&mut pipe, reaction.actions).await {
635                    break Stopped::Transport;
636                }
637            }
638            command = commands.recv() => {
639                // `None` = every `RealtimeClient` handle dropped without a
640                // `disconnect`; treat it as one.
641                let Some(command) = command else { break Stopped::Shutdown };
642                if matches!(command, Command::Shutdown) {
643                    let _ = send_all(&mut pipe, session.bye()).await;
644                    break Stopped::Shutdown;
645                }
646                if !send_all(&mut pipe, apply(session, command, pending)).await {
647                    break Stopped::Transport;
648                }
649            }
650            () = tick_at(next_probe) => {
651                if let Err(error) = pipe.ping().await {
652                    tracing::debug!(%error, "realtime: keepalive failed");
653                    break Stopped::Transport;
654                }
655                next_probe = deadline(idle);
656            }
657            () = tick_at(next_silence_check) => {
658                // Writes into a blackholed TCP path keep succeeding, so silence
659                // is the only evidence the peer is gone.
660                let seen = pipe.inbound_count();
661                if seen == inbound_mark {
662                    tracing::warn!(
663                        limit = ?silence,
664                        "realtime: nothing inbound (not even a pong) — treating the transport as dead"
665                    );
666                    break Stopped::Transport;
667                }
668                inbound_mark = seen;
669                next_silence_check = deadline(silence);
670            }
671            () = tick_at(welcome_by) => {
672                tracing::warn!(
673                    deadline = ?WELCOME_DEADLINE,
674                    "realtime: no Welcome after the handshake — reconnecting"
675                );
676                break Stopped::Transport;
677            }
678            () = tick_at(retry_subscriptions_at) => {
679                retry_subscriptions_at = None;
680                if !send_all(&mut pipe, session.reissue_subscriptions()).await {
681                    break Stopped::Transport;
682                }
683            }
684        }
685    };
686    pipe.close().await;
687    stopped
688}

690/// Apply one command to the reducer, parking or releasing its waiter.
691fn apply(session: &mut ClientSession, command: Command, pending: &mut Pending) -> Vec<Action> {
692    match command {
693        Command::Subscribe { space_id, ack } => {
694            let actions = session.subscribe(space_id);
695            if let Some(ack) = ack {
696                // Two ways there is nothing to wait for: this connection is
697                // already in the room (a repeat open of the same space), or its
698                // server does not confirm subscribes at all. Otherwise park the
699                // waiter until the confirmation lands — possibly on a later
700                // connection, since the reducer re-issues the desired set on
701                // every `Welcome`.
702                if session.is_subscribed(space_id)
703                    || (session.is_live() && !session.confirms_subscriptions())
704                {
705                    let _ = ack.send(());
706                } else {
707                    pending.entry(space_id).or_default().push(ack);
708                }
709            }
710            actions
711        }
712        Command::Unsubscribe(space_id) => {
713            // Nobody is waiting to receive a space they just left; dropping the
714            // senders releases them now instead of at their deadline.
715            pending.remove(&space_id);
716            session.unsubscribe(space_id)
717        }
718        Command::Screenshare { space_id, enabled } => session.screenshare(space_id, enabled),
719        // Handled by the caller: it also ends the loop.
720        Command::Shutdown => session.bye(),
721    }
722}

724/// Put every action on the wire. `false` means the pipe is dead. A frame that
725/// fails to ENCODE is skipped (it cannot be fixed by reconnecting, and killing
726/// the session over one malformed frame would be a self-inflicted outage);
727/// anything else is a transport failure.
728async fn send_all<P: MessagePipe>(pipe: &mut P, actions: Vec<Action>) -> bool {
729    for Action::Send(msg) in actions {
730        match pipe.send_json(&msg).await {
731            Ok(()) => {}
732            Err(PipeError::Encode(error)) => {
733                tracing::error!(%error, ?msg, "realtime: skipping unencodable frame");
734            }
735            Err(error) => {
736                tracing::debug!(%error, "realtime: send failed — transport is gone");
737                return false;
738            }
739        }
740    }
741    true
742}

744/// An absolute deadline `period` from now, or `None` for a timer that never
745/// fires.
746fn deadline(period: Option<Duration>) -> Option<Instant> {
747    period.map(|period| Instant::now() + period)
748}

750/// A `select!` arm that fires at `at` — or never, when there is no deadline.
751async fn tick_at(at: Option<Instant>) {
752    match at {
753        Some(at) => tokio::time::sleep_until(at).await,
754        None => std::future::pending().await,
755    }

760/// Where one attempt's legs come from: the config lookup, the credential, and
761/// the two dials.
762///
763/// A trait rather than three free functions so [`connect_leg`]'s *decisions* —
764/// which leg, in which order, and what a failure at each step means — are
765/// testable without a socket, and so the supervisor itself can be driven against
766/// scripted outcomes.
767trait Dial: Send + Sync {
768    /// An established connection over either leg.
769    type Pipe: Probe + Send;
770
771    /// `GET /realtime/config`.
772    fn config(&self) -> impl std::future::Future<Output = Result<RealtimeConfig>> + Send;
773
774    /// The current session credential.
775    fn token(&self) -> Result<String>;
776
777    /// Dial the advertised WebTransport endpoint.
778    fn wt(
779        &self,
780        url: &str,
781        token: &str,
782        cert_hash: Option<&str>,
783    ) -> impl std::future::Future<Output = Result<Self::Pipe>> + Send;
784
785    /// Dial the WebSocket leg (derived from the REST base URL).
786    fn ws(&self, token: &str) -> impl std::future::Future<Output = Result<Self::Pipe>> + Send;
787}
788
789/// One attempt: fetch the config, then WebTransport (if advertised, allowed, and
790/// working), else the WebSocket.
791///
792/// The config fetch is what re-mints the credential: it goes through
793/// [`RestClient`], whose 401 hook rotates the session and republishes the access
794/// token, so the token read straight after is current by construction. Reading a
795/// captured token instead is the stale-JWT strand this ordering exists to
796/// prevent.
797///
798/// A config fetch that FAILS does not fail the attempt: the WebSocket leg is
799/// derived from the REST base URL and needs nothing the config carries, so a
800/// backend blip in that one endpoint must not cost the whole plane. Only an
801/// `Auth` error propagates — that one is an answer (it has already survived the
802/// REST layer's refresh-and-retry).
803async fn connect_leg<D: Dial>(dialer: &D, gate: &WtGate) -> Result<D::Pipe> {
804    let config = match dialer.config().await {
805        Ok(config) => Some(config),
806        Err(err @ HoundError::Auth { .. }) => return Err(err),
807        Err(err) => {
808            tracing::warn!(%err, "realtime: config fetch failed — going straight to the websocket leg");
809            None
810        }
811    };
812    let token = dialer.token()?;
813
814    let advertised = config
815        .as_ref()
816        .and_then(|config| config.webtransport_url.as_deref())
817        .filter(|url| !url.is_empty());
818    if let Some(url) = advertised {
819        if gate.wt_allowed() {
820            let cert_hash = config
821                .as_ref()
822                .and_then(|config| config.cert_hash.as_deref());
823            match dialer.wt(url, &token, cert_hash).await {
824                Ok(pipe) => {
825                    gate.note_wt_success();
826                    return Ok(pipe);
827                }
828                // EVERY WebTransport failure falls back, refusals included — see
829                // the module docs on where an auth verdict may come from.
830                Err(err) => {
831                    gate.note_wt_failure();
832                    tracing::warn!(%err, "realtime: webtransport unavailable — falling back to websocket");
833                }
834            }
835        } else {
836            tracing::debug!("realtime: skipping webtransport (recent dials all failed)");
837        }
838    }
839    dialer.ws(&token).await
840}

842/// Whether the WebTransport leg is worth dialling at all.
843///
844/// On a network that blocks UDP the WT dial cannot succeed, and paying
845/// [`WT_CONNECT_TIMEOUT`] before every single reconnect is a self-inflicted 3s
846/// outage on top of whatever caused the reconnect. After
847/// [`WT_FAILURE_STREAK`] failures in a row the leg is skipped until either a
848/// session actually reached `Welcome` (proof the path changed) or
849/// [`WT_RETRY_AFTER`] elapses — a laptop moves between networks without
850/// restarting, so this can never be permanent.
851///
852/// Scoped to one plane rather than the process: a fresh plane (a new sign-in)
853/// deserves a fresh probe, and process-global state would be untestable in
854/// parallel.
855struct WtGate {
856    state: std::sync::Mutex<GateState>,
857}
858
859#[derive(Default)]
860struct GateState {
861    failures: u32,
862    /// When the streak last tripped the skip, for the expiry.
863    blocked_at: Option<Instant>,
864}
865
866impl WtGate {
867    fn new() -> Self {
868        Self {
869            state: std::sync::Mutex::new(GateState::default()),
870        }
871    }
872
873    fn wt_allowed(&self) -> bool {
874        let state = self.state.lock().expect("wt gate mutex");
875        match state.blocked_at {
876            Some(blocked_at) => Instant::now().duration_since(blocked_at) >= WT_RETRY_AFTER,
877            None => state.failures < WT_FAILURE_STREAK,
878        }
879    }
880
881    fn note_wt_failure(&self) {
882        let mut state = self.state.lock().expect("wt gate mutex");
883        state.failures = state.failures.saturating_add(1);
884        if state.failures >= WT_FAILURE_STREAK {
885            // Re-stamped on every failure, so a dial let through by the expiry
886            // that fails again restarts the cooldown rather than retrying hard.
887            state.blocked_at = Some(Instant::now());
888        }
889    }
890
891    fn note_wt_success(&self) {
892        *self.state.lock().expect("wt gate mutex") = GateState::default();
893    }
894
895    /// A session reached `Welcome` — whichever leg carried it, this network path
896    /// works, so re-probe WebTransport on the next attempt.
897    fn note_welcomed_session(&self) {
898        *self.state.lock().expect("wt gate mutex") = GateState::default();
899    }
900}

949/// A WebTransport control stream, holding the endpoint and connection it rides.
950///
951/// Both are kept alive deliberately: dropping the `Endpoint` (or the
952/// `Connection`) tears the stream down under us — the same lifetime trap the
953/// call plane's iroh endpoint has.
954struct WtPipe {
955    stream: FramedBi,
956    _connection: hound_wt::Connection,
957    _client: WtClient,
958}

960/// The URL the WebTransport leg dials: exactly what the backend advertised, with
961/// no query of our own. The credential rides an `Authorization` header instead —
962/// a `:path` lands in access logs and proxy traces.
963fn wt_dial_url(advertised: &str) -> &str {
964    advertised.trim_end_matches('/')
965}

967/// Dial the advertised WebTransport endpoint and open the control stream.
968///
969/// DNS pins are supplied through a [`PinnedResolver`] closure read at LOOKUP
970/// time, so a harness that installs overrides after this process started still
971/// has them honoured while the URL keeps the real hostname (TLS stays honest).
972async fn connect_wt(url: &str, token: &str, cert_hash: Option<&str>) -> Result<WtPipe> {
973    let trust = match cert_hash {
974        Some(hash) => Trust::CertHash(hound_wt::CertHash::parse(hash).map_err(|e| {
975            HoundError::realtime(
976                RealtimeCode::BadCertHash,
977                format!("bad realtime cert hash: {e}"),
978            )
979        })?),
980        None => Trust::NativeRoots,
981    };
982    let mut options = ClientOptions::new(trust);
983    options.resolver = PinnedResolver::from_fn(|| dns_overrides().to_vec());
984
985    let client = WtClient::new(options).map_err(wt_error)?;
986    let url = wt_dial_url(url);
987    let authorization = format!("Bearer {token}");
988
989    let connection = tokio::time::timeout(
990        WT_CONNECT_TIMEOUT,
991        client.connect_with_headers(url, &[("Authorization", &authorization)]),
992    )
993    .await
994    .map_err(|_| {
995        HoundError::network(
996            NetCode::WebTransportTimeout,
997            "webtransport connect timed out",
998        )
999    })?
1000    .map_err(wt_error)?;
1001    // The client opens the control stream, mirroring the browser API (and what
1002    // the backend's accept loop waits for). Bounded: a session that never gives
1003    // us a stream must not park the attempt.
1004    let stream = FramedBi::open_within(&connection, WT_STREAM_TIMEOUT)
1005        .await
1006        .map_err(wt_error)?;
1007    Ok(WtPipe {
1008        stream,
1009        _connection: connection,
1010        _client: client,
1011    })
1012}

1014/// Map a WebTransport failure onto the shared error taxonomy — always as a
1015/// retryable transport failure, `Rejected` included.
1016///
1017/// `Rejected` is wtransport's verdict for *any* non-2xx answer to the CONNECT
1018/// and for a CONNECT stream stopped before it is answered, so a pod rollout, a
1019/// misrouted host and a genuine refusal are indistinguishable here. Calling that
1020/// an auth failure signed users out over a deploy; auth verdicts now come only
1021/// from the WebSocket handshake's status and from `/realtime/config`.
1022fn wt_error(error: WtError) -> HoundError {
1023    HoundError::network(NetCode::WebTransport, redact(&error.to_string()))
1024}

1076    async fn send(&mut self, frame: String) -> std::result::Result<(), PipeError> {
1077        use futures::SinkExt;
1078        self.socket
1079            .send(tokio_tungstenite::tungstenite::Message::Text(frame.into()))
1080            .await
1081            .map_err(|e| PipeError::Transport(e.to_string()))
1082    }

1085        use futures::StreamExt;
1086        use tokio_tungstenite::tungstenite::Message;
1087        loop {
1088            let message = self.socket.next().await?;
1089            if message.is_ok() {
1090                self.inbound += 1;
1091            }

1093                Ok(Message::Text(text)) => Some(Ok(text.to_string())),
1094                Ok(Message::Close(_)) => None,
1095                // Pongs answer our own Pings; tungstenite answers the server's
1096                // Pings itself. Binary is not part of this protocol.
1097                Ok(_) => continue,
1098                Err(error) => Some(Err(PipeError::Transport(error.to_string()))),
1099            };
1100        }
1101    }

1105    }

1205/// Build the realtime WebSocket URL from the REST base — the same scheme
1206/// rewrite the call plane's `control_ws_url` does, against the route the backend
1207/// registers (`GET /realtime/ws`). No credential: it goes in a header.
1208fn realtime_ws_url(base_url: &str) -> Result<String> {
1209    let base = base_url.trim_end_matches('/');
1210    let ws_base = if let Some(rest) = base.strip_prefix("https://") {
1211        format!("wss://{rest}")
1212    } else if let Some(rest) = base.strip_prefix("http://") {
1213        format!("ws://{rest}")
1214    } else if base.starts_with("ws://") || base.starts_with("wss://") {
1215        base.to_owned()
1216    } else {
1217        return Err(HoundError::realtime(
1218            RealtimeCode::BadBaseScheme,
1219            format!("api_base has no http(s) scheme: {base_url}"),
1220        ));
1221    };
1222    Ok(format!("{ws_base}/realtime/ws"))
1223}

1235/// Open the WebSocket leg, honouring the process's DNS pins.
1236///
1237/// A pinned host is dialled by opening the TCP socket OURSELVES against the
1238/// pinned address and only then running the TLS + WebSocket handshake against
1239/// the request's original hostname — so SNI and certificate validation still
1240/// prove we reached the intended deployment, which an IP-in-the-URL rewrite
1241/// would not. An unpinned host takes `connect_async`'s ordinary path.
1242///
1243/// `timeout` bounds the WHOLE dial (TCP, TLS, upgrade); it is a parameter only
1244/// so tests can pin it.
1245async fn connect_ws(base_url: &str, token: &str, timeout: Duration) -> Result<WsPipe> {
1246    use tokio_tungstenite::tungstenite::client::IntoClientRequest;
1247    use tokio_tungstenite::tungstenite::http::header::AUTHORIZATION;
1248
1249    let url = realtime_ws_url(base_url)?;
1250    let mut request = url.into_client_request().map_err(|e| {
1251        HoundError::realtime(
1252            RealtimeCode::BadWsUrl,
1253            format!("bad realtime ws url: {}", redact(&e.to_string())),
1254        )
1255    })?;
1256    let authorization = format!("Bearer {token}").parse().map_err(|_| {
1257        HoundError::auth(
1258            AuthCode::BadHeaderValue,
1259            "session token is not a valid header value",
1260        )
1261    })?;
1262    request.headers_mut().insert(AUTHORIZATION, authorization);
1263    let (host, port) = ws_authority(request.uri())?;
1264
1265    let dial = async {
1266        match pinned_addr(dns_overrides(), &host, port) {
1267            Some(addr) => {
1268                tracing::debug!(%host, %addr, "realtime: dialling a pinned address");
1269                let tcp = tokio::net::TcpStream::connect(addr).await.map_err(|e| {
1270                    HoundError::network(
1271                        NetCode::RealtimeWsTcp,
1272                        format!("realtime ws tcp connect: {e}"),
1273                    )
1274                })?;
1275                tokio_tungstenite::client_async_tls_with_config(
1276                    request,
1277                    tcp,
1278                    Some(ws_config()),
1279                    None,
1280                )
1281                .await
1282            }
1283            None => {
1284                tokio_tungstenite::connect_async_with_config(request, Some(ws_config()), false)
1285                    .await
1286            }
1287        }
1288        .map_err(ws_error)
1289    };
1290
1291    let (socket, _response) = tokio::time::timeout(timeout, dial).await.map_err(|_| {
1292        HoundError::network(
1293            NetCode::RealtimeWsTimeout,
1294            format!("realtime ws connect timed out after {timeout:?}"),
1295        )
1296    })??;
1297    Ok(WsPipe { socket, inbound: 0 })
1298}

1300/// Map a WebSocket handshake failure onto the shared error taxonomy, preserving
1301/// an HTTP status so [`classify`] can call 401/403 fatal.

1315/// Scrub a `token=…` query value out of anything headed for a log.
1316///
1317/// This client no longer puts a credential in a URL, but the URLs it dials are
1318/// partly server-supplied (`/realtime/config`'s `webtransport_url`), and a
1319/// transport error quotes the URL it failed on. Cheap insurance that a
1320/// mis-configured deployment cannot leak one through our logs.
1321fn redact(text: &str) -> String {
1322    const KEY: &str = "token=";
1323    let mut out = String::with_capacity(text.len());
1324    let mut rest = text;
1325    while let Some(at) = rest.find(KEY) {
1326        out.push_str(&rest[..at + KEY.len()]);
1327        out.push_str("REDACTED");
1328        let tail = &rest[at + KEY.len()..];
1329        let end = tail
1330            .find(['&', ' ', '"', '\'', ',', ')'])
1331            .unwrap_or(tail.len());
1332        rest = &tail[end..];
1333    }
1334    out.push_str(rest);
1335    out
1336}

1338/// The host and port a WebSocket request dials (`wss` ⇒ 443, `ws` ⇒ 80).
1339fn ws_authority(uri: &tokio_tungstenite::tungstenite::http::Uri) -> Result<(String, u16)> {
1340    let host = uri.host().ok_or_else(|| {
1341        HoundError::realtime(
1342            RealtimeCode::WsUrlNoHost,
1343            format!("realtime ws url has no host: {uri}"),
1344        )
1345    })?;
1346    let port = uri.port_u16().or_else(|| match uri.scheme_str() {
1347        Some("wss") => Some(443),
1348        Some("ws") => Some(80),
1349        _ => None,
1350    });
1351    let port = port.ok_or_else(|| {
1352        HoundError::realtime(
1353            RealtimeCode::WsUrlNoPort,
1354            format!("realtime ws url has no port: {uri}"),
1355        )
1356    })?;
1357    Ok((host.to_owned(), port))
1358}

1360/// Pick the address to TCP-connect for `host:port` from a `curl --resolve`-style
1361/// pin table. Deliberately identical to [`PinnedResolver`]'s rule so both legs
1362/// resolve the same way: an exact `(host, port)` pin wins, otherwise a pin for
1363/// the host alone is re-pointed at the port actually being dialled.
1364fn pinned_addr(pins: &[(String, SocketAddr)], host: &str, port: u16) -> Option<SocketAddr> {
1365    pins.iter()
1366        .find(|(name, addr)| name == host && addr.port() == port)
1367        .or_else(|| pins.iter().find(|(name, _)| name == host))
1368        .map(|(_, addr)| SocketAddr::new(addr.ip(), port))
1369}

1574    /// Run one session over a fake wire and collect everything it produced. The
1575    /// command sender stays alive for the session's whole life, so the loop ends
1576    /// only the way the wire (or an explicit `Shutdown`) says it should.
1577    async fn drive(
1578        wire: &Arc<Mutex<Wire>>,
1579        session: &mut ClientSession,
1580        commands: Vec<Command>,
1581    ) -> (Stopped, Vec<RealtimeEvent>) {
1582        let (events_tx, mut events_rx) = mpsc::unbounded_channel();
1583        let (command_tx, mut command_rx) = mpsc::unbounded_channel();
1584        for command in commands {
1585            command_tx.send(command).unwrap();
1586        }

1589            session,

1603    /// A session running in its own task, plus the ends a test drives it by.
1604    ///
1605    /// Needed wherever a command has to be ordered against the handshake:
1606    /// [`run_session`]'s arms are deliberately unbiased (see the `select!`
1607    /// comment), so a command queued before the session starts may be serviced
1608    /// before or after the `Welcome`. Waiting for `Connected` and only then
1609    /// sending is the sequencing the loop actually guarantees — and commands
1610    /// share one channel, so they are FIFO among themselves.
1611    struct Running {
1612        commands: mpsc::UnboundedSender<Command>,
1613        events: mpsc::UnboundedReceiver<RealtimeEvent>,
1614        /// Everything already taken off the stream, so waiting for one event
1615        /// doesn't lose the ones before it.
1616        seen: Vec<RealtimeEvent>,
1617        task: JoinHandle<(Stopped, ClientSession)>,
1618    }

1620    fn start(wire: Arc<Mutex<Wire>>, session: ClientSession) -> Running {
1621        let (events_tx, events) = mpsc::unbounded_channel();
1622        let (commands, mut command_rx) = mpsc::unbounded_channel();
1623        let task = tokio::spawn(async move {
1624            let mut session = session;
1625            let stopped = run_session(
1626                FakePipe(wire),
1627                &mut session,
1628                &events_tx,
1629                &mut command_rx,
1630                &mut Pending::new(),
1631            )
1632            .await;
1633            (stopped, session)
1634        });
1635        Running {
1636            commands,
1637            events,
1638            seen: Vec::new(),
1639            task,
1640        }
1641    }

1643    impl Running {
1644        /// Consume events until one matches, keeping every one on the way.
1645        async fn wait_for(&mut self, mut matches: impl FnMut(&RealtimeEvent) -> bool) {
1646            loop {
1647                let event = self
1648                    .events
1649                    .recv()
1650                    .await
1651                    .expect("the session ended without the expected event");
1652                let hit = matches(&event);
1653                self.seen.push(event);
1654                if hit {
1655                    return;
1656                }
1657            }
1658        }

1660        /// Await the handshake, so what follows is ordered after it.
1661        async fn connected(&mut self) {
1662            self.wait_for(|event| matches!(event, RealtimeEvent::Connected { .. }))
1663                .await;
1664        }

1666        fn send(&self, command: Command) {
1667            self.commands.send(command).expect("the session is running");
1668        }

1670        /// Stop the session and collect everything it produced.
1671        async fn stop(mut self) -> (Stopped, ClientSession, Vec<RealtimeEvent>) {
1672            self.send(Command::Shutdown);
1673            let (stopped, session) = self.task.await.expect("the session task panicked");
1674            while let Ok(event) = self.events.try_recv() {
1675                self.seen.push(event);
1676            }
1677            (stopped, session, self.seen)
1678        }

1817    /// every event it emitted, or `None` if it was still running at `within`
1818    /// (which is how "the ladder keeps retrying" is asserted).
1819    async fn supervise_until(
1820        dialer: FakeDialer,
1821        commands: Vec<Command>,
1822        within: Duration,
1823    ) -> (Option<()>, Vec<RealtimeEvent>) {
1824        let (events_tx, mut events_rx) = mpsc::unbounded_channel();
1825        let (command_tx, command_rx) = mpsc::unbounded_channel();
1826        for command in commands {
1827            command_tx.send(command).unwrap();
1828        }
1829        let stopped = tokio::time::timeout(
1830            within,
1831            supervise(dialer, WtGate::new(), events_tx, command_rx, None),
1832        )
1833        .await
1834        .ok();
1835        drop(command_tx);
1836        let mut events = Vec::new();
1837        while let Ok(event) = events_rx.try_recv() {
1838            events.push(event);
1839        }
1840        (stopped, events)
1841    }
