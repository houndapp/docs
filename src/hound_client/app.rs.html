1//! The orchestration facade: ties the REST data plane, the session lifecycle, and
2//! the realtime (`/community`) plane into one observable, UI-free app core.
3//!
4//! [`HoundClient`] is what the FRB layer (and any future native UI) drives. It owns
5//! the pieces and an [`AppState`] of `tokio::sync::watch` channels — one per
6//! observable snapshot, latest-wins, which is exactly what a UI wants (it never
7//! needs the history, only the current view). REST loads *seed* the snapshots; a
8//! single background task consumes the typed
9//! [`RealtimeEvent`](crate::realtime::RealtimeEvent) stream, folds each
10//! event through the pure reducers ([`CommunityTree`]/[`RosterView`]/[`MessageList`]),
11//! and republishes the mutated snapshot onto the matching `watch` sender.
12//!
13//! Commands are async and return [`Result<_, HoundError>`]; observable getters
14//! return [`watch::Receiver`]s the binding subscribes. No FFI and no UI types live
15//! here — the crate stays standalone-buildable.
16

42/// How long startup waits on the credential store before giving up on a *fast*
43/// restore. Sized for a read that only hits the store (milliseconds); anything past
44/// it means the OS is asking the user something — see [`Session::load_stored`] — and
45/// no startup path may wait on a human.
46const KEYRING_READ_GRACE: Duration = Duration::from_secs(3);
47
48/// Map a Steam-client ticket failure onto the auth registry, one code per cause:
49/// "not under Steam", "Steam is offline", "the ticket callback never came" and "the
50/// SDK itself failed" are four different things to tell a user (and four different
51/// things to triage), so they must not share a Sentry bucket. The upstream display
52/// text stays as the message.
53fn steam_auth_error(e: hound_steam::SteamClientError) -> HoundError {
54    use hound_steam::SteamClientError as E;
55    let code = match e {
56        E::NotAvailable => AuthCode::SteamNotAvailable,
57        E::NotConnected => AuthCode::SteamNotConnected,
58        E::Timeout => AuthCode::SteamTicketTimeout,
59        E::Steamworks(_) => AuthCode::SteamworksFailure,
60    };
61    HoundError::auth(code, e.to_string())
62}

64/// How long [`HoundClient::new_bounded`] waits for the local Steam client before
65/// starting without it. Sized for a handshake with an already-running process; past
66/// it, Steam is booting (or wedged) and no startup path may keep waiting.
67const STEAM_INIT_BOUND: Duration = Duration::from_secs(6);
68
69/// [`SteamHandle::init`] moved off the async worker and given a deadline. `None`
70/// for "no app id", "not a Steam launch", and "Steam did not answer in time"
71/// alike — all three mean the same thing to every caller: no Steam sign-in.
72async fn bounded_steam_init(steam_app_id: Option<u32>) -> Option<SteamHandle> {
73    let app_id = steam_app_id?;
74    let init = tokio::task::spawn_blocking(move || SteamHandle::init(app_id));
75    match tokio::time::timeout(STEAM_INIT_BOUND, init).await {
76        Ok(Ok(handle)) => handle,
77        Ok(Err(e)) => {
78            tracing::warn!("steam: init task failed ({e}) — starting without steam");
79            None
80        }
81        Err(_) => {
82            tracing::warn!(
83                "steam: the client did not answer within {STEAM_INIT_BOUND:?} — starting without steam"
84            );
85            None
86        }
87    }
88}

90/// How long a realtime connect waits for the server's `Welcome` before letting
91/// the caller proceed anyway. See `HoundClient::await_realtime_live`.
92const REALTIME_WELCOME_GRACE: Duration = Duration::from_secs(5);

94/// How many messages a space's history window holds — the page the UI loads on
95/// open, and the same page a reconnect re-fetches to reconcile against.
96///
97/// One constant on purpose: the re-fetched window is authoritative for the range
98/// it covers (see [`MessageList::reconcile_window`]), so fetching a *smaller*
99/// window than the one on screen would silently narrow what a reconnect can
100/// repair.
101const MESSAGE_WINDOW: u32 = 50;
102
103/// How long the post-reconnect re-fetch waits for the server to confirm it
104/// re-joined a space's room before going ahead without the confirmation.
105///
106/// The re-fetch wants to run *after* the room join, so that a message posted
107/// between the history snapshot and the join is delivered as an event rather
108/// than missed by both — the same window [`RealtimeClient::subscribe_space`]
109/// exists to close on open. But a server that does not confirm subscriptions at
110/// all (an older backend) would never release it, and a resync that never
111/// happens is worse than one that races, so the wait is bounded and expiring it
112/// is not an error.
113const RESYNC_SUBSCRIBE_GRACE: Duration = Duration::from_secs(5);
114
115/// How many community signals a slow subscriber may fall behind before it starts
116/// missing them. Sized for a burst of role edits, not for buffering an absent
117/// reader: a missed signal costs one stale page until the next edit, so the
118/// bound is deliberately small.
119const COMMUNITY_SIGNAL_BACKLOG: usize = 64;
120
121/// The observable app state: one `watch` channel per snapshot the UI renders. Each
122/// holds the latest value (a new subscriber sees it immediately) and notifies on
123/// change. Per-space message lists are created lazily on [`HoundClient::open_space`].
124struct AppState {
125    auth: watch::Sender<AuthState>,
126    community_tree: watch::Sender<CommunityTree>,
127    roster: watch::Sender<RosterView>,
128    /// space id → that space's message-list channel (lazily created on open).
129    messages: Mutex<HashMap<Uuid, watch::Sender<MessageList>>>,
130    /// space id → whether that space's list is showing a DETACHED history
131    /// window (an around-page centred on a search hit) rather than the live
132    /// tail. Lazily created alongside the message channel; see
133    /// [`HoundClient::open_space_around`] for the state machine.
134    detached: Mutex<HashMap<Uuid, watch::Sender<bool>>>,
135    /// Spaces whose history still needs re-fetching after a reconnect.
136    ///
137    /// Armed as a set by the reconnect itself and drained one space at a time by
138    /// that space's `Subscribed` confirmation (or by the grace timer). Being a
139    /// set is what makes the resync fire exactly once per space per reconnect,
140    /// however many confirmations, timers and reconnects overlap.
141    resync_pending: Mutex<HashSet<Uuid>>,
142    /// Serializes post-reconnect history re-fetches.
143    ///
144    /// Every space is re-confirmed at once after a `Welcome`, so without this a
145    /// client that has opened a dozen spaces answers one reconnect with a dozen
146    /// simultaneous history requests — and every client on a backend that just
147    /// restarted reconnects at once. One permit turns that into a queue: the
148    /// space the user is looking at is repaired just as fast, and the rest
149    /// trickle.
150    resync_gate: Semaphore,
151    /// Call-state observables, driven by the live [`CallSession`]:
152    /// the roster/track view (from the control plane's `SessionView`), the
153    /// connection state, and the mic + per-peer speaking levels.
154    call_view: watch::Sender<CallView>,
155    connection: watch::Sender<ConnectionState>,
156    levels: watch::Sender<LevelsView>,
157    /// 1 Hz RTC debug-stats snapshot (the data behind the debug panel + per-tile
158    /// overlay). Driven by the call's `spawn_emitter`; default/empty between calls.
159    stats: watch::Sender<crate::stats::StatsSnapshot>,
160    /// Whether the realtime plane is currently welcomed. Latest-wins like every
161    /// other snapshot here; the supervisor heals an outage on its own, so this is
162    /// a status indicator (and a test seam), never a trigger for the app to act.
163    realtime_connected: watch::Sender<bool>,
164    /// "Refetch the roles / the member page of this community" — the one
165    /// observable here that is a BROADCAST rather than a `watch`.
166    ///
167    /// Roles and member pages are fetched on demand (paginated, rarely open), so
168    /// there is no snapshot to republish; what a consumer needs is the *edge*,
169    /// and a latest-wins channel cannot deliver two edges to a subscriber that
170    /// blinked. Bounded: a subscriber too slow to keep up misses signals rather
171    /// than growing the queue, and missing one only costs a stale page until the
172    /// next — never memory. Nothing awaits a receiver, so a send with no
173    /// subscribers is the expected quiet case.
174    community_signals: tokio::sync::broadcast::Sender<crate::model::CommunitySignal>,
175    /// Stamps every [`CommunitySignal`](crate::model::CommunitySignal) with a
176    /// number no other signal has. Two identical edits are then two distinct
177    /// values, which is what a latest-wins consumer needs to see the second one
178    /// at all — see the type's docs.
179    signal_seq: std::sync::atomic::AtomicU64,
180    /// Which realtime plane the reducer task is allowed to speak for.
181    ///
182    /// `HoundClient::connect` stamps a fresh generation and the task it spawns
183    /// carries it. A plane is replaced on every sign-in (and on a re-restore),
184    /// and the OLD task can still hold queued events — including an
185    /// `AuthExpired` from the credential that was just replaced, which would
186    /// sign the user straight back out of the session they just established, or
187    /// a `Connected` that would flip the new plane's liveness to a lie. Events
188    /// stamped with a superseded generation are dropped at apply.
189    realtime_epoch: std::sync::atomic::AtomicU64,
190    /// Live `peer_id → user_id` map for the current call. Seeded by the call at join
191    /// (from the REST voice roster + self) and kept current by the realtime voice
192    /// events here; the call's control reader joins it into each `CallView` so tiles
193    /// show real identities. Cleared on call join/leave so it never leaks across calls.
194    call_identities: PeerIdentities,
195}

220/// The headless application core. Cheap to share via the returned `Arc<Self>`; every

257    /// unknown email, awaiting the user's create-or-link decision. Set by
258    /// [`magic_login_wait`](Self::magic_login_wait) /
259    /// [`magic_login_code`](Self::magic_login_code) on `Unknown`; consumed by
260    /// [`magic_register`](Self::magic_register).
261    pending_magic_link: Mutex<Option<String>>,
262    /// A pending email magic-link login (the bound loopback listener), if the
263    /// UI has begun one. At most one at a time; replaced by a re-begin,
264    /// consumed by wait, dropped by cancel.
265    magic_login: Mutex<Option<hound_browser_auth::MagicLogin>>,
266    /// Wakes a blocked [`magic_login_wait`](Self::magic_login_wait) when the UI

273    /// Which capture backend the next [`call_join`](Self::call_join) opens —

278    #[cfg(feature = "synthetic")]
279    capture_source: Mutex<crate::call::CaptureSource>,
280    /// Live "stop talking" switch for the synthetic mic source (see

282    /// every join, so toggling it reaches the running call and survives reconnects.
283    /// `synthetic`-feature-only.
284    #[cfg(feature = "synthetic")]
285    synthetic_silence: crate::call::SyntheticSilence,

289    /// Build a client bound to `base_url`, persisting tokens through `store`. Does
290    /// not touch the network — call [`restore`](Self::restore) or a login command.
291    /// `steam_app_id` opts into Steamworks detection (Steam-store builds only);
292    /// init failure just means "not launched through Steam".
293    pub fn new(
294        base_url: &str,
295        store: Arc<dyn SessionStore>,
296        steam_app_id: Option<u32>,
297    ) -> Result<Arc<Self>> {
298        Self::with_steam(base_url, store, steam_app_id.and_then(SteamHandle::init))
299    }

301    /// [`new`](Self::new) with the Steamworks handshake taken off the caller's
302    /// thread and time-boxed to [`STEAM_INIT_BOUND`] — the constructor a UI host
303    /// must use.
304    ///
305    /// `steamworks::Client::init_app` is a *blocking* IPC handshake with the local
306    /// Steam process and carries no deadline of its own: on a cold machine where
307    /// Steam is still booting it holds for as long as Steam takes. This runs before
308    /// the app's first frame, so that is a black screen rather than a slow launch
309    /// (which is exactly how it presented — a bootstrap watchdog firing in
310    /// `init_app` with no frame ever rendered).
311    ///
312    /// A timed-out init is ABANDONED, not cancelled — the SDK call has no cancel —
313    /// so the blocking thread may still finish and immediately drop its handle. The
314    /// process then simply runs without Steam: `steam_available` is false, the
315    /// silent sign-in is skipped, and the login screen stands.
316    pub async fn new_bounded(
317        base_url: &str,
318        store: Arc<dyn SessionStore>,
319        steam_app_id: Option<u32>,
320    ) -> Result<Arc<Self>> {
321        Self::with_steam(base_url, store, bounded_steam_init(steam_app_id).await)
322    }

445    /// Load the communities the signed-in user has JOINED and their spaces,
446    /// seeding the community-tree snapshot. Idempotent (re-run on reconnect to
447    /// re-seed).
448    /// The page is a SNAPSHOT of membership, so a community absent from it is
449    /// one this user is no longer in (left from another device, deleted,
450    /// kicked) and it leaves the tree — but only if it was there BEFORE the
451    /// fetch, which is what `known` records (see
452    /// [`CommunityTree::apply_membership`](crate::reducer::CommunityTree::apply_membership)).
453    /// Every fetch happens before the single write, and the fold is one
454    /// `send_modify` over the live value, so a command that folded its own
455    /// answer while this load was in flight survives it.
456    pub async fn load_communities(&self) -> Result<()> {
457        let known = self.state.community_tree.borrow().community_ids();
458        let communities = self.rest.all_communities().await?;
459        let mut spaces = Vec::with_capacity(communities.len());
460        for community in &communities {
461            spaces.push((community.id, self.rest.all_spaces(community.id).await?));
462        }
463        // `send_modify`, not `send_replace` of a snapshot: it publishes with no
464        // subscribers (a headless caller that subscribes after signing in must
465        // still see this load) AND it reads the value as it is NOW.
466        self.state.community_tree.send_modify(|tree| {
467            tree.apply_membership(&known, communities);
468            for (community_id, list) in spaces {
469                tree.set_spaces(community_id, list);
470            }
471        });
472        Ok(())
473    }

475    /// Open a space: load its message history (seeding the per-space `MessageList`
476    /// watch and merging the authors into the roster), then subscribe to its
477    /// realtime room so subsequent `message:*` events flow into the same channel.
478    ///
479    /// Returning means **subscribed**, not merely "subscribe requested": the
480    /// subscribe is awaited to the server's confirmation (see
481    /// [`RealtimeClient::subscribe_space`], which bounds and never fails on
482    /// it). A caller may therefore act on the space — send into it, expect its
483    /// traffic — knowing events are already being delivered here.
484    ///
485    /// One window remains and is deliberately out of scope: a message posted
486    /// *between* the history page above and that confirmation is in neither, and
487    /// nothing re-fetches it. (Its bigger sibling — everything missed while the
488    /// plane was DOWN — is repaired by `resync_space`.)
489    ///
490    /// Re-opening a space that is showing a detached around-window
491    /// ([`open_space_around`](Self::open_space_around)) re-attaches it: page 0
492    /// *replaces* that window rather than merging into it, because the two are
493    /// slices of different parts of the log and merging them would splice a
494    /// hole. An ordinary open still merges, as it always did.
495    pub async fn open_space(&self, space_id: Uuid) -> Result<()> {
496        let page = self.rest.list_messages(space_id, 0, MESSAGE_WINDOW).await?;
497
498        let sender = self.message_sender(space_id).await;
499        if self.take_detached(space_id).await {
500            let window = page.messages.items;
501            sender.send_if_modified(|list| list.replace(window));
502        } else {
503            sender.send_modify(|list| list.apply_page(page.messages.items));
504        }
505
506        if !page.members.is_empty() {
507            self.state
508                .roster
509                .send_modify(|roster| roster.apply_members(page.members));
510        }
511
512        if let Some(rt) = self.realtime.lock().await.as_ref() {
513            rt.subscribe_space(space_id).await?;
514        }
515        Ok(())
516    }
517
518    /// Send a message. The realtime `message:create` echo seeds the list (the
519    /// reducer dedupes by id), so we don't optimistically insert here.
520    pub async fn send_message(&self, space_id: Uuid, contents: &str) -> Result<()> {
521        self.rest.post_message(space_id, contents, &[]).await?;
522        Ok(())
523    }

525    /// Edit a message. The realtime `message:update` echo applies the change.
526    pub async fn edit_message(
527        &self,
528        space_id: Uuid,
529        message_id: Uuid,
530        contents: &str,
531    ) -> Result<()> {
532        self.rest
533            .edit_message(space_id, message_id, contents)
534            .await?;
535        Ok(())
536    }

543    }
