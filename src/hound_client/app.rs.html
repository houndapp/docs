1//! The orchestration facade: ties the REST data plane, the session lifecycle, and
2//! the realtime (`/community`) plane into one observable, UI-free app core.
3//!
4//! [`HoundClient`] is what the FRB layer (and any future native UI) drives. It owns
5//! the pieces and an [`AppState`] of `tokio::sync::watch` channels — one per
6//! observable snapshot, latest-wins, which is exactly what a UI wants (it never
7//! needs the history, only the current view). REST loads *seed* the snapshots; a
8//! single background task consumes the typed
9//! [`RealtimeEvent`](crate::realtime::RealtimeEvent) stream, folds each
10//! event through the pure reducers ([`CommunityTree`]/[`RosterView`]/[`MessageList`]),
11//! and republishes the mutated snapshot onto the matching `watch` sender.
12//!
13//! Commands are async and return [`Result<_, HoundError>`]; observable getters
14//! return [`watch::Receiver`]s the binding subscribes. No FFI and no UI types live
15//! here — the crate stays standalone-buildable.
16

41/// How long startup waits on the credential store before giving up on a *fast*
42/// restore. Sized for a read that only hits the store (milliseconds); anything past
43/// it means the OS is asking the user something — see [`Session::load_stored`] — and
44/// no startup path may wait on a human.
45const KEYRING_READ_GRACE: Duration = Duration::from_secs(3);
46
47/// Map a Steam-client ticket failure onto the auth registry, one code per cause:
48/// "not under Steam", "Steam is offline", "the ticket callback never came" and "the
49/// SDK itself failed" are four different things to tell a user (and four different
50/// things to triage), so they must not share a Sentry bucket. The upstream display
51/// text stays as the message.
52fn steam_auth_error(e: hound_steam::SteamClientError) -> HoundError {
53    use hound_steam::SteamClientError as E;
54    let code = match e {
55        E::NotAvailable => AuthCode::SteamNotAvailable,
56        E::NotConnected => AuthCode::SteamNotConnected,
57        E::Timeout => AuthCode::SteamTicketTimeout,
58        E::Steamworks(_) => AuthCode::SteamworksFailure,
59    };
60    HoundError::auth(code, e.to_string())
61}

63/// How long [`HoundClient::new_bounded`] waits for the local Steam client before
64/// starting without it. Sized for a handshake with an already-running process; past
65/// it, Steam is booting (or wedged) and no startup path may keep waiting.
66const STEAM_INIT_BOUND: Duration = Duration::from_secs(6);
67
68/// [`SteamHandle::init`] moved off the async worker and given a deadline. `None`
69/// for "no app id", "not a Steam launch", and "Steam did not answer in time"
70/// alike — all three mean the same thing to every caller: no Steam sign-in.
71async fn bounded_steam_init(steam_app_id: Option<u32>) -> Option<SteamHandle> {
72    let app_id = steam_app_id?;
73    let init = tokio::task::spawn_blocking(move || SteamHandle::init(app_id));
74    match tokio::time::timeout(STEAM_INIT_BOUND, init).await {
75        Ok(Ok(handle)) => handle,
76        Ok(Err(e)) => {
77            tracing::warn!("steam: init task failed ({e}) — starting without steam");
78            None
79        }
80        Err(_) => {
81            tracing::warn!(
82                "steam: the client did not answer within {STEAM_INIT_BOUND:?} — starting without steam"
83            );
84            None
85        }
86    }
87}

89/// How long a realtime connect waits for the server's `Welcome` before letting
90/// the caller proceed anyway. See `HoundClient::await_realtime_live`.
91const REALTIME_WELCOME_GRACE: Duration = Duration::from_secs(5);

93/// How many messages a space's history window holds — the page the UI loads on
94/// open, and the same page a reconnect re-fetches to reconcile against.
95///
96/// One constant on purpose: the re-fetched window is authoritative for the range
97/// it covers (see [`MessageList::reconcile_window`]), so fetching a *smaller*
98/// window than the one on screen would silently narrow what a reconnect can
99/// repair.
100const MESSAGE_WINDOW: u32 = 50;
101
102/// How long the post-reconnect re-fetch waits for the server to confirm it
103/// re-joined a space's room before going ahead without the confirmation.
104///
105/// The re-fetch wants to run *after* the room join, so that a message posted
106/// between the history snapshot and the join is delivered as an event rather
107/// than missed by both — the same window [`RealtimeClient::subscribe_space`]
108/// exists to close on open. But a server that does not confirm subscriptions at
109/// all (an older backend) would never release it, and a resync that never
110/// happens is worse than one that races, so the wait is bounded and expiring it
111/// is not an error.
112const RESYNC_SUBSCRIBE_GRACE: Duration = Duration::from_secs(5);
113
114/// The observable app state: one `watch` channel per snapshot the UI renders. Each
115/// holds the latest value (a new subscriber sees it immediately) and notifies on
116/// change. Per-space message lists are created lazily on [`HoundClient::open_space`].
117struct AppState {
118    auth: watch::Sender<AuthState>,
119    community_tree: watch::Sender<CommunityTree>,
120    roster: watch::Sender<RosterView>,
121    /// space id → that space's message-list channel (lazily created on open).
122    messages: Mutex<HashMap<Uuid, watch::Sender<MessageList>>>,
123    /// space id → whether that space's list is showing a DETACHED history
124    /// window (an around-page centred on a search hit) rather than the live
125    /// tail. Lazily created alongside the message channel; see
126    /// [`HoundClient::open_space_around`] for the state machine.
127    detached: Mutex<HashMap<Uuid, watch::Sender<bool>>>,
128    /// Spaces whose history still needs re-fetching after a reconnect.
129    ///
130    /// Armed as a set by the reconnect itself and drained one space at a time by
131    /// that space's `Subscribed` confirmation (or by the grace timer). Being a
132    /// set is what makes the resync fire exactly once per space per reconnect,
133    /// however many confirmations, timers and reconnects overlap.
134    resync_pending: Mutex<HashSet<Uuid>>,
135    /// Serializes post-reconnect history re-fetches.
136    ///
137    /// Every space is re-confirmed at once after a `Welcome`, so without this a
138    /// client that has opened a dozen spaces answers one reconnect with a dozen
139    /// simultaneous history requests — and every client on a backend that just
140    /// restarted reconnects at once. One permit turns that into a queue: the
141    /// space the user is looking at is repaired just as fast, and the rest
142    /// trickle.
143    resync_gate: Semaphore,
144    /// Call-state observables, driven by the live [`CallSession`]:
145    /// the roster/track view (from the control plane's `SessionView`), the
146    /// connection state, and the mic + per-peer speaking levels.
147    call_view: watch::Sender<CallView>,
148    connection: watch::Sender<ConnectionState>,
149    levels: watch::Sender<LevelsView>,
150    /// 1 Hz RTC debug-stats snapshot (the data behind the debug panel + per-tile
151    /// overlay). Driven by the call's `spawn_emitter`; default/empty between calls.
152    stats: watch::Sender<crate::stats::StatsSnapshot>,
153    /// Whether the realtime plane is currently welcomed. Latest-wins like every
154    /// other snapshot here; the supervisor heals an outage on its own, so this is
155    /// a status indicator (and a test seam), never a trigger for the app to act.
156    realtime_connected: watch::Sender<bool>,
157    /// Which realtime plane the reducer task is allowed to speak for.
158    ///
159    /// `HoundClient::connect` stamps a fresh generation and the task it spawns
160    /// carries it. A plane is replaced on every sign-in (and on a re-restore),
161    /// and the OLD task can still hold queued events — including an
162    /// `AuthExpired` from the credential that was just replaced, which would
163    /// sign the user straight back out of the session they just established, or
164    /// a `Connected` that would flip the new plane's liveness to a lie. Events
165    /// stamped with a superseded generation are dropped at apply.
166    realtime_epoch: std::sync::atomic::AtomicU64,
167    /// Live `peer_id → user_id` map for the current call. Seeded by the call at join
168    /// (from the REST voice roster + self) and kept current by the realtime voice
169    /// events here; the call's control reader joins it into each `CallView` so tiles
170    /// show real identities. Cleared on call join/leave so it never leaks across calls.
171    call_identities: PeerIdentities,
172}

174impl AppState {
175    fn new() -> Self {
176        Self {
177            auth: watch::channel(AuthState::SignedOut).0,
178            community_tree: watch::channel(CommunityTree::new()).0,
179            roster: watch::channel(RosterView::new()).0,
180            messages: Mutex::new(HashMap::new()),
181            detached: Mutex::new(HashMap::new()),
182            resync_pending: Mutex::new(HashSet::new()),
183            resync_gate: Semaphore::new(1),
184            call_view: watch::channel(CallView::default()).0,
185            connection: watch::channel(ConnectionState::default()).0,
186            levels: watch::channel(LevelsView::default()).0,
187            stats: watch::channel(crate::stats::StatsSnapshot::default()).0,
188            realtime_connected: watch::channel(false).0,
189            realtime_epoch: std::sync::atomic::AtomicU64::new(0),
190            call_identities: PeerIdentities::default(),
191        }
192    }
193}

195/// The headless application core. Cheap to share via the returned `Arc<Self>`; every
196/// command takes `&self`. Holds the REST client, the session, the observable state,
197/// and the live realtime connection (replaced on each (re)connect).
198pub struct HoundClient {
199    rest: Arc<RestClient>,
200    /// The base URL (kept so the call orchestration can derive the control WS URL).
201    base_url: String,
202    session: Session,
203    state: AppState,
204    /// The live realtime connection, if connected. Swapped on connect; dropped (and
205    /// its background task ended) on disconnect/logout.
206    realtime: Mutex<Option<RealtimeClient>>,
207    /// The live voice/video call, if joined. At most one at a time.
208    call: Mutex<Option<CallSession>>,
209    /// The settings screen's mic test, while it is running. Mutually exclusive
210    /// with `call` by construction — [`Self::mic_monitor_start`] refuses while a
211    /// call holds the input device, and [`Self::call_join`] stops any monitor
212    /// before it opens one.
213    mic_monitor: Mutex<Option<crate::call::MicMonitor>>,
214    /// The Steamworks connection when launched through Steam (`None` otherwise,
215    /// including all non-Steam builds). Init success *is* the launch detection.
216    steam: Option<hound_steam::SteamHandle>,
217    /// A pending Steam `link_token`: proof of Steam-account ownership awaiting
218    /// the user's create-or-link decision (or a settings link). Set when a
219    /// Steam sign-in meets an unknown Steam ID; consumed by
220    /// [`steam_register`](Self::steam_register) /
221    /// [`link_steam_pending`](Self::link_steam_pending), auto-consumed by a
222    /// successful sign-in, cleared by [`steam_dismiss`](Self::steam_dismiss).
223    pending_steam_link: Mutex<Option<String>>,
224    /// A pending browser-deferred login (the bound loopback listener), if the UI
225    /// has begun one. At most one at a time; replaced by a re-begin, consumed by
226    /// wait, dropped by cancel.
227    browser_login: Mutex<Option<hound_browser_auth::BrowserLogin>>,
228    /// Wakes a blocked [`browser_login_wait`](Self::browser_login_wait) when the
229    /// UI cancels (the wait has already consumed the listener by then).
230    browser_login_cancelled: tokio::sync::Notify,
231    /// A pending email `link_token`: proof that a magic-link sign-in met an
232    /// unknown email, awaiting the user's create-or-link decision. Set by
233    /// [`magic_login_wait`](Self::magic_login_wait) /
234    /// [`magic_login_code`](Self::magic_login_code) on `Unknown`; consumed by
235    /// [`magic_register`](Self::magic_register).
236    pending_magic_link: Mutex<Option<String>>,
237    /// A pending email magic-link login (the bound loopback listener), if the
238    /// UI has begun one. At most one at a time; replaced by a re-begin,
239    /// consumed by wait, dropped by cancel.
240    magic_login: Mutex<Option<hound_browser_auth::MagicLogin>>,
241    /// Wakes a blocked [`magic_login_wait`](Self::magic_login_wait) when the UI
242    /// cancels (the wait has already consumed the listener by then).
243    magic_login_cancelled: tokio::sync::Notify,
244    /// The decode→render sink the call hands decoded video to (keyed by peer). The
245    /// FRB layer installs one whose body calls `video_render::push_frame(...)`;
246    /// until then a [`NullVideoSink`] keeps the call running (audio + state) headless.
247    video_sink: Mutex<Arc<dyn VideoSink>>,
248    /// Which capture backend the next [`call_join`](Self::call_join) opens —
249    /// real hardware, or the engine's camera-free/mic-free synthetic test
250    /// sources (a headless-bot / test seam; see
251    /// [`set_capture_source`](Self::set_capture_source)). `synthetic`-feature-only:
252    /// a non-synthetic build carries no field, no method, zero cost.
253    #[cfg(feature = "synthetic")]
254    capture_source: Mutex<crate::call::CaptureSource>,
255    /// Live "stop talking" switch for the synthetic mic source (see
256    /// [`set_source_silent`](Self::set_source_silent)). Built once and cloned into
257    /// every join, so toggling it reaches the running call and survives reconnects.
258    /// `synthetic`-feature-only.
259    #[cfg(feature = "synthetic")]
260    synthetic_silence: crate::call::SyntheticSilence,
261}

264    /// Build a client bound to `base_url`, persisting tokens through `store`. Does
265    /// not touch the network — call [`restore`](Self::restore) or a login command.
266    /// `steam_app_id` opts into Steamworks detection (Steam-store builds only);
267    /// init failure just means "not launched through Steam".
268    pub fn new(
269        base_url: &str,
270        store: Arc<dyn SessionStore>,
271        steam_app_id: Option<u32>,
272    ) -> Result<Arc<Self>> {
273        Self::with_steam(base_url, store, steam_app_id.and_then(SteamHandle::init))
274    }

276    /// [`new`](Self::new) with the Steamworks handshake taken off the caller's
277    /// thread and time-boxed to [`STEAM_INIT_BOUND`] — the constructor a UI host
278    /// must use.
279    ///
280    /// `steamworks::Client::init_app` is a *blocking* IPC handshake with the local
281    /// Steam process and carries no deadline of its own: on a cold machine where
282    /// Steam is still booting it holds for as long as Steam takes. This runs before
283    /// the app's first frame, so that is a black screen rather than a slow launch
284    /// (which is exactly how it presented — a bootstrap watchdog firing in
285    /// `init_app` with no frame ever rendered).
286    ///
287    /// A timed-out init is ABANDONED, not cancelled — the SDK call has no cancel —
288    /// so the blocking thread may still finish and immediately drop its handle. The
289    /// process then simply runs without Steam: `steam_available` is false, the
290    /// silent sign-in is skipped, and the login screen stands.
291    pub async fn new_bounded(
292        base_url: &str,
293        store: Arc<dyn SessionStore>,
294        steam_app_id: Option<u32>,
295    ) -> Result<Arc<Self>> {
296        Self::with_steam(base_url, store, bounded_steam_init(steam_app_id).await)
297    }

299    /// The common constructor body: everything here is cheap and non-blocking, so
300    /// the two public constructors differ only in how they obtained `steam`.
301    fn with_steam(
302        base_url: &str,
303        store: Arc<dyn SessionStore>,
304        steam: Option<SteamHandle>,
305    ) -> Result<Arc<Self>> {
306        let rest = Arc::new(RestClient::new(base_url)?);
307        let session = Session::new(rest.clone(), store);
308        Ok(Arc::new(Self {
309            rest,
310            base_url: base_url.to_owned(),
311            session,
312            state: AppState::new(),
313            realtime: Mutex::new(None),
314            call: Mutex::new(None),
315            mic_monitor: Mutex::new(None),
316            steam,
317            pending_steam_link: Mutex::new(None),
318            browser_login: Mutex::new(None),
319            browser_login_cancelled: tokio::sync::Notify::new(),
320            pending_magic_link: Mutex::new(None),
321            magic_login: Mutex::new(None),
322            magic_login_cancelled: tokio::sync::Notify::new(),
323            video_sink: Mutex::new(Arc::new(NullVideoSink)),
324            #[cfg(feature = "synthetic")]
325            capture_source: Mutex::new(crate::call::CaptureSource::default()),
326            #[cfg(feature = "synthetic")]
327            synthetic_silence: crate::call::SyntheticSilence::default(),
328        }))

350    /// Make the synthetic mic source emit digital SILENCE (or resume its speech-like

354    /// The seam a scenario needs to exercise the silence path end to end: the

420    /// Load the signed-in user's communities and their spaces, seeding the
421    /// community-tree snapshot. Idempotent (re-run on reconnect to re-seed).
422    pub async fn load_communities(&self) -> Result<()> {
423        let page = self.rest.list_communities(0, 100).await?;
424        let mut tree = self.state.community_tree.borrow().clone();
425        tree.apply_communities(page.items.clone());
426        for community in &page.items {
427            let spaces = self.rest.list_spaces(community.id, 0, 100).await?;
428            tree.set_spaces(community.id, spaces.items);
429        }
430        // `send_replace`: see the comment in `logout` — a caller that hasn't
431        // subscribed to `watch_community_tree()` yet (e.g. a headless client that
432        // calls it only after sign-in, like a bot) must still see this load, not a
433        // silently-dropped no-op. Was a plain `send`; this is the exact bug that
434        // made a freshly-registered headless caller read back zero communities
435        // despite `list_communities` returning the just-created one.
436        self.state.community_tree.send_replace(tree);
437        Ok(())
438    }

440    /// Open a space: load its message history (seeding the per-space `MessageList`
441    /// watch and merging the authors into the roster), then subscribe to its
442    /// realtime room so subsequent `message:*` events flow into the same channel.
443    ///
444    /// Returning means **subscribed**, not merely "subscribe requested": the
445    /// subscribe is awaited to the server's confirmation (see
446    /// [`RealtimeClient::subscribe_space`], which bounds and never fails on
447    /// it). A caller may therefore act on the space — send into it, expect its
448    /// traffic — knowing events are already being delivered here.
449    ///
450    /// One window remains and is deliberately out of scope: a message posted
451    /// *between* the history page above and that confirmation is in neither, and
452    /// nothing re-fetches it. (Its bigger sibling — everything missed while the
453    /// plane was DOWN — is repaired by `resync_space`.)
454    ///
455    /// Re-opening a space that is showing a detached around-window
456    /// ([`open_space_around`](Self::open_space_around)) re-attaches it: page 0
457    /// *replaces* that window rather than merging into it, because the two are
458    /// slices of different parts of the log and merging them would splice a
459    /// hole. An ordinary open still merges, as it always did.
460    pub async fn open_space(&self, space_id: Uuid) -> Result<()> {
461        let page = self.rest.list_messages(space_id, 0, MESSAGE_WINDOW).await?;
462
463        let sender = self.message_sender(space_id).await;
464        if self.take_detached(space_id).await {
465            let window = page.messages.items;
466            sender.send_if_modified(|list| list.replace(window));
467        } else {
468            sender.send_modify(|list| list.apply_page(page.messages.items));
469        }
470
471        if !page.members.is_empty() {
472            self.state
473                .roster
474                .send_modify(|roster| roster.apply_members(page.members));
475        }
476
477        if let Some(rt) = self.realtime.lock().await.as_ref() {
478            rt.subscribe_space(space_id).await?;
479        }
480        Ok(())
481    }

503    /// Fetch a single message by id — e.g. resolving a `<hound:message/…>`
504    /// reference preview. Plain request/response, unlike [`Self::watch_messages`]:
505    /// no snapshot, no caching, one round trip per call.
506    pub async fn fetch_message(&self, message_id: Uuid) -> Result<Message> {
507        self.rest.get_message(message_id).await
508    }
