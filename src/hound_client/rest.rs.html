1//! The REST data plane: a thin, typed wrapper over the backend HTTP API. Holds the
2//! session bearer token in a lock-free cell (read on every request, republished by
3//! the auth refresh task) and maps transport/status failures onto [`HoundError`].
4//!
5//! **Refresh seam:** every verb funnels through one request core that retries a
6//! request ONCE when the backend answers `401 expiredtoken` and a [`Reauth`] hook is
7//! installed ([`crate::session::Session`] installs one). The hook rotates the tokens
8//! and republishes the access token here, so the rebuilt request carries the fresh
9//! one. A REFUSED refresh, any other auth code, or a second failure surfaces as
10//! [`HoundError::Auth`], which is therefore a genuinely dead session — but a refresh
11//! that merely *failed* (offline, 5xx) surfaces as its own retryable error, because
12//! it proves nothing about the session and callers treat `Auth` as terminal.
13//!
14//! **Wire mapping discipline:** responses are decoded against private wire DTOs
15//! (the backend's serde, snake_case) and mapped into the *client* view-models in
16//! [`crate::model`] via `into_model()`/`From`. Wire quirks — the pagination envelope
17//! (`contents`/`current`/`next`/`prev`), the nested community `spaces`, the
18//! `messages`+`users` split, `/profile/global`, `/community/members?id=` — are
19//! confined to this module. The exact shapes live in
20//! `services/backend/src/api/{auth,community,space,profile,data}.rs`.
21//!
22//! WebAuthn note: register/login `start`/`finish` carry the ceremony payloads as the
23//! TYPED `hound-passkey` wire types
24//! ([`CreationChallengeResponse`](hound_passkey::proto::CreationChallengeResponse) /
25//! [`RequestChallengeResponse`](hound_passkey::proto::RequestChallengeResponse) from
26//! the server, [`RegisterPublicKeyCredential`] /
27//! [`PublicKeyCredential`] from the authenticator) and (de)serialise the request /
28//! response bodies through `hound_passkey::dto` — the SAME vocabulary the backend
29//! uses, so the wire stays byte-identical. The client still does not interpret the
30//! ceremony content; it threads the typed value between the authenticator and the
31//! backend.

68/// HTTP client bound to one backend base URL, carrying the optional session token.
69/// Cheap to clone-share via `Arc<RestClient>`; the token cell is interior-mutable.
70pub struct RestClient {
71    /// Built on first request, not at construction: `reqwest::Client::new`
72    /// runs macOS system-proxy discovery (SCDynamicStore), which stalls ~2 s
73    /// in unbundled binaries (dev runs, every per-test process). Deferring it
74    /// keeps constructors instant and moves the one-time cost onto the first
75    /// actual HTTP call, off the caller's startup path.
76    http: std::sync::OnceLock<Client>,
77    base: Url,
78    /// Current bearer token (`None` when signed out). Swapped atomically by auth.
79    token: ArcSwapOption<String>,
80    /// The installed refresh hook, if any. `None` (a client built without a
81    /// [`Session`](crate::session::Session)) simply never retries an expired 401.
82    /// `Box`ed because arc-swap's `RefCnt` covers only `Arc<T: Sized>`, so the trait
83    /// object needs the extra indirection to live in the cell.
84    reauth: ArcSwapOption<Box<dyn Reauth>>,
85}

87/// The refresh seam. The REST layer knows only "ask whoever owns the session to
88/// rotate the access token that was just rejected"; [`crate::session::Session`]
89/// supplies the implementation (and the single-flight guarantee) in its constructor.
90///
91/// Returns a boxed future rather than being an `async fn`: the hook is held as
92/// `dyn Reauth`, and async-fn-in-trait is not dyn-compatible.
93pub trait Reauth: Send + Sync {
94    /// Rotate the session tokens, given the access token whose rejection triggered
95    /// this. Implementations MUST be single-flight over `stale_access` — several
96    /// in-flight requests hit the same expiry at once and only one exchange may run.
97    fn reauth(&self, stale_access: String) -> futures::future::BoxFuture<'static, Result<()>>;
98}

100/// Which auth failure the backend reported in a 401 body. Only [`AuthCode::Expired`]
101/// is worth a refresh-and-retry; the others describe a session that is dead however
102/// often we re-present it. (Backend: `services/backend/src/api/auth/jwt.rs`.)
103#[derive(Debug, Clone, Copy, PartialEq, Eq)]
104enum AuthCode {
105    Expired,
106    Invalid,
107    Revoked,
108    InvalidType,
109    /// Any body we can't read a known code out of — including a non-JSON one (a
110    /// proxy's HTML error page), which is why parsing must never be fatal here.
111    Unknown,
112}
113
114impl AuthCode {
115    fn from_body(body: &str) -> Self {
116        /// The backend's error body: `{ "error": "…", "code": "…" }`, where `code`
117        /// is the lowercased `AuthError` variant name.
118        #[derive(serde::Deserialize)]
119        struct WireAuthError {
120            code: String,
121        }
122
123        let Ok(wire) = serde_json::from_str::<WireAuthError>(body) else {
124            return Self::Unknown;
125        };
126        match wire.code.as_str() {
127            "expiredtoken" => Self::Expired,
128            "invalidtoken" => Self::Invalid,
129            "revokedtoken" => Self::Revoked,
130            "invalidtokentype" => Self::InvalidType,
131            _ => Self::Unknown,
132        }
133    }
134}

136/// Re-label a `501` from a Steam auth route as [`SteamLoginDisabled`].
137///
138/// The backend answers 501 on every `/auth/steam/*` route when the instance has no
139/// Steam credentials (`ApiError::SteamDisabled`) — an instance-configuration fact,
140/// not a request failure, and the one Steam outcome the UI must never treat as
141/// retryable. As a bare `HoundError::Http` it reached the UI (and Sentry) as
142/// `HTTP501` with whatever body the deployment happened to send, which on older
143/// backends is nothing at all; this gives it the registry code the caller branches
144/// on regardless of body.
145///
146/// [`SteamLoginDisabled`]: crate::error::AuthCode::SteamLoginDisabled
147fn steam_disabled(e: HoundError) -> HoundError {
148    match e {
149        HoundError::Http { status: 501, .. } => HoundError::auth(
150            crate::error::AuthCode::SteamLoginDisabled,
151            "steam sign-in is not enabled on this server",
152        ),
153        other => other,
154    }
155}

157/// Cap on a route label. Bounded because the label becomes a Sentry TAG, and an
158/// unbounded tag is a grouping key that never groups. Every real route is far shorter;
159/// the cap exists so a pathological path can't be the one that proves otherwise.
160const MAX_ENDPOINT_LEN: usize = 64;
161
162/// The `METHOD /route` label a request is named by in errors and telemetry.
163///
164/// Ids are elided (`/space/{id}/message`, never the uuid) for two independent
165/// reasons: a tag with a uuid in it has one value per request and groups nothing, and
166/// an id in a tag is user data leaving the device in a field nobody audits. The query
167/// string goes the same way — its values are the most likely to be identifying, and
168/// the path alone already says which endpoint answered.
169fn endpoint(method: &str, path: &str) -> String {
170    let path = path.split(['?', '#']).next().unwrap_or(path);
171    let mut label = String::with_capacity(method.len() + path.len() + 2);
172    label.push_str(method);
173    label.push(' ');
174    for segment in path.split('/') {
175        label.push('/');
176        if Uuid::parse_str(segment).is_ok() {
177            label.push_str("{id}");
178        } else if !segment.is_empty() && segment.bytes().all(|b| b.is_ascii_digit()) {
179            label.push_str("{n}");
180        } else {
181            label.push_str(segment);
182        }
183        if label.len() > MAX_ENDPOINT_LEN {
184            let mut cut = MAX_ENDPOINT_LEN;
185            while !label.is_char_boundary(cut) {
186                cut -= 1;
187            }
188            label.truncate(cut);
189            break;
190        }
191    }
192    label
193}

199    /// Build a client for `base_url` (e.g. `http://localhost:8080` /
200    /// `https://api.hound.chat`). Errors only on a malformed base URL.
201    pub fn new(base_url: &str) -> Result<Self> {
202        // A trailing slash makes `Url::join` treat the base as a directory, so a
203        // relative path like `community` resolves under it rather than replacing
204        // the last segment. Normalise here so callers pass clean relative paths.
205        let base_url = if base_url.ends_with('/') {
206            base_url.to_owned()
207        } else {
208            format!("{base_url}/")
209        };
210        let base = Url::parse(&base_url).map_err(|e| {
211            HoundError::internal(
212                InternalCode::InvalidBaseUrl,
213                format!("invalid base url {base_url:?}: {e}"),
214            )
215        })?;
216        Ok(Self {
217            http: std::sync::OnceLock::new(),
218            base,
219            token: ArcSwapOption::empty(),
220            reauth: ArcSwapOption::empty(),
221        })
222    }

224    /// The lazily-built HTTP client (see the field docs for why lazy).
225    fn http(&self) -> &Client {
226        self.http.get_or_init(|| {
227            // Bounded, or a half-dead server (TCP accepted, upstream never
228            // answers — a churning ingress does exactly this) hangs every
229            // caller forever. `restore()` runs before the app's first frame,
230            // so an unbounded hang here is a permanent black screen the
231            // crash reporter can't see (nothing crashed, nothing hung — the
232            // isolate is just awaiting). read_timeout rather than a total
233            // `timeout` so long attachment transfers aren't capped; it fires
234            // only when the socket goes idle.
235            let builder = Client::builder()
236                .connect_timeout(std::time::Duration::from_secs(10))
237                .read_timeout(std::time::Duration::from_secs(20));
238            // Test-harness DNS pinning (remote-target e2e runs) — a no-op
239            // unless `hound_client::dns::set_overrides` ran first.
240            #[cfg(feature = "dns-override")]
241            let builder = crate::dns::apply(builder);
242            builder.build().expect("building reqwest client")
243        })
244    }

284    /// The request core every verb helper funnels through: attach the current bearer
285    /// token, send, and — on a `401 expiredtoken` with a [`Reauth`] hook installed —
286    /// rotate the session and resend ONCE. At most one retry per call, so a backend
287    /// that answers `expiredtoken` to a freshly minted token cannot loop us.
288    ///
289    /// `build` is a closure, not a `RequestBuilder`, for two reasons: a builder is
290    /// single-use (consumed by `send`), and re-invoking it re-reads the token cell,
291    /// which is what makes the retry carry the *rotated* token rather than the dead
292    /// one. It must therefore be a pure re-build — no side effects.
293    async fn send(
294        &self,
295        endpoint: &str,
296        build: impl Fn() -> reqwest::RequestBuilder,
297    ) -> Result<reqwest::Response> {
298        let stale = self.token.load_full();
299        let resp = self.send_once(&build).await?;
300        if resp.status() != reqwest::StatusCode::UNAUTHORIZED {
301            return Ok(resp);
302        }
303        // No session token ⇒ nothing to rotate (an unauthenticated endpoint answering
304        // 401 is the caller's error); no hook ⇒ nobody owns the session.
305        let (Some(stale), Some(hook)) = (stale, self.reauth.load_full()) else {
306            return Ok(resp);
307        };
308        // Reading the body is what tells expiry apart from a dead session; the
309        // response is consumed either way, so from here we build the error ourselves.
310        let body = resp.text().await.unwrap_or_default();
311        if AuthCode::from_body(&body) != AuthCode::Expired {
312            return Err(HoundError::from_endpoint(401, endpoint, &body));
313        }
314        if let Err(err) = hook.reauth(stale.as_str().to_owned()).await {
315            return Err(Self::refresh_failure(err, endpoint, &body));
316        }
317        self.send_once(&build).await
318    }

320    /// What the caller is told when the refresh triggered by an expired 401 did
321    /// not itself succeed.
322    ///
323    /// The distinction is whether we LEARNED anything about the session. A
324    /// refresh the backend REFUSED (auth-class) says the refresh token is dead
325    /// too, so the original 401 is the honest answer and the caller may route to
326    /// sign-out. A refresh that never got an answer — offline, DNS, a 5xx, a
327    /// database outage behind the auth service — says nothing at all about the
328    /// session, and reporting it as a 401 turned every such blip into a
329    /// sign-out (the realtime supervisor and `Session`'s proactive refresh both
330    /// classify `Auth` as fatal). Those propagate as themselves, which
331    /// [`classify`](crate::call::reconnect::classify) calls retryable — the same
332    /// verdict the proactive-refresh task already reaches for a non-auth failure.
333    fn refresh_failure(err: HoundError, endpoint: &str, rejection: &str) -> HoundError {
334        use crate::call::reconnect::{Disposition, classify};
335        match classify(&err) {
336            Disposition::Fatal => {
337                tracing::warn!(%err, "rest: the session refresh was REFUSED — the session is dead");
338                HoundError::from_endpoint(401, endpoint, rejection)
339            }
340            Disposition::Retryable => {
341                tracing::warn!(%err, "rest: the session refresh failed transiently — surfacing it as retryable");
342                err
343            }
344        }
345    }

347    /// One build → attach the *current* token → send. Split out so the retry reads
348    /// the token cell again after the hook rotated it.
349    async fn send_once(
350        &self,
351        build: &impl Fn() -> reqwest::RequestBuilder,
352    ) -> Result<reqwest::Response> {
353        let mut req = build();
354        if let Some(tok) = self.token.load_full() {
355            req = req.bearer_auth(tok.as_str());
356        }
357        Ok(req.send().await?)
358    }

437    /// OPTIONS `path` (token attached), returning the response headers.
438    ///
439    /// The only verb here whose answer lives entirely in the headers — a discovery
440    /// probe answers `204 No Content` — so unlike its siblings it hands the map
441    /// back instead of a decoded body.
442    async fn options_headers(&self, path: &str) -> Result<reqwest::header::HeaderMap> {
443        let url = self.url(path)?;
444        let at = endpoint("OPTIONS", path);
445        let resp = self
446            .send(&at, || {
447                self.http().request(reqwest::Method::OPTIONS, url.clone())
448            })
449            .await?;
450        let status = resp.status();
451        if !status.is_success() {
452            let body = resp.text().await.unwrap_or_default();
453            return Err(HoundError::from_endpoint(status.as_u16(), &at, &body));
454        }

469    /// Shared response handling: map non-success status onto a typed error
470    /// (reading the body for detail, named by the route that produced it), else
471    /// decode the JSON body into `T`.
472    async fn decode<T: DeserializeOwned>(at: &str, resp: reqwest::Response) -> Result<T> {
473        let status = resp.status();
474        if !status.is_success() {
475            let body = resp.text().await.unwrap_or_default();
476            return Err(HoundError::from_endpoint(status.as_u16(), at, &body));
477        }
478        let bytes = resp.bytes().await?;
479        serde_json::from_slice(&bytes).map_err(HoundError::from)
480    }

482    /// Like [`decode`] but for endpoints whose success body is empty/ignored.
483    async fn decode_unit(at: &str, resp: reqwest::Response) -> Result<()> {
484        let status = resp.status();
485        if !status.is_success() {
486            let body = resp.text().await.unwrap_or_default();
487            return Err(HoundError::from_endpoint(status.as_u16(), at, &body));
488        }
489        Ok(())
490    }

556    /// `POST /auth/token/refresh` — exchanges a refresh token (sent as a `Bearer`
557    /// header) for a fresh access token + rotated refresh token. The backend reads
558    /// the refresh token from the `Authorization` header, not the body.
559    ///
560    /// Built outside [`send`](Self::send) deliberately: it authenticates with the
561    /// *refresh* token, and keeping it off the retry path is what makes recursion
562    /// impossible — a 401 here is the end of the session, not another refresh.
563    pub async fn refresh_token(&self, refresh_token: &str) -> Result<TokenRefresh> {
564        let req = self
565            .http()
566            .post(self.url("auth/token/refresh")?)
567            .bearer_auth(refresh_token);

953    /// `OPTIONS /upload/{avatar,attachment}` — the image types **this deployment**
954    /// will accept for that upload, read off the `Accept-Post` response header
955    /// (the IANA-registered header a resource uses to advertise what it takes).
956    ///
957    /// Asked rather than assumed because the answer moves with the server: JPEG XL
958    /// support arrives by backend deploy, not by client release, and a client that
959    /// hard-codes the list either offers a type the server rejects or hides one it
960    /// accepts.
961    ///
962    /// **An unadvertised list is empty, not an error.** A backend that answers the
963    /// probe without the header (an older one) has told us nothing, and "nothing" is
964    /// for the caller to interpret — it falls back to its own built-in set rather
965    /// than blocking an upload on a missing advertisement. A transport failure or a
966    /// non-2xx status still surfaces as [`HoundError`], because those *are* failures.
967    pub async fn upload_accept_post(&self, purpose: UploadPurpose) -> Result<Vec<Mime>> {
968        let headers = self.options_headers(purpose.path()).await?;
969        Ok(headers
970            .get(ACCEPT_POST)
971            .and_then(|value| value.to_str().ok())
972            .map(parse_accept_post)
973            .unwrap_or_default())
974    }

1035        let addr = listener.local_addr().unwrap();
1036        tokio::spawn(async move {

1038            while let Ok((mut socket, _)) = listener.accept().await {

1040                let _ = socket.read(&mut buf).await;

1042                    .write_all(
