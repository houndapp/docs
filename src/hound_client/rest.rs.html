1//! The REST data plane: a thin, typed wrapper over the backend HTTP API. Holds the
2//! session bearer token in a lock-free cell (read on every request, republished by
3//! the auth refresh task) and maps transport/status failures onto [`HoundError`].
4//!
5//! **Refresh seam:** every verb funnels through one request core that retries a
6//! request ONCE when the backend answers `401 expiredtoken` and a [`Reauth`] hook is
7//! installed ([`crate::session::Session`] installs one). The hook rotates the tokens
8//! and republishes the access token here, so the rebuilt request carries the fresh
9//! one. A REFUSED refresh, any other auth code, or a second failure surfaces as
10//! [`HoundError::Auth`], which is therefore a genuinely dead session — but a refresh
11//! that merely *failed* (offline, 5xx) surfaces as its own retryable error, because
12//! it proves nothing about the session and callers treat `Auth` as terminal.
13//!
14//! **Wire mapping discipline:** responses are decoded against private wire DTOs
15//! (the backend's serde, snake_case) and mapped into the *client* view-models in
16//! [`crate::model`] via `into_model()`/`From`. Wire quirks — the pagination envelope
17//! (`contents`/`current`/`next`/`prev`), the nested community `spaces`, the
18//! `messages`+`users` split, `/profile/global`, `/community/members?id=` — are
19//! confined to this module. The exact shapes live in
20//! `services/backend/src/api/{auth,community,space,profile,data}.rs`.
21//!
22//! WebAuthn note: register/login `start`/`finish` carry the ceremony payloads as the
23//! TYPED `hound-passkey` wire types
24//! ([`CreationChallengeResponse`](hound_passkey::proto::CreationChallengeResponse) /
25//! [`RequestChallengeResponse`](hound_passkey::proto::RequestChallengeResponse) from
26//! the server, [`RegisterPublicKeyCredential`] /
27//! [`PublicKeyCredential`] from the authenticator) and (de)serialise the request /
28//! response bodies through `hound_passkey::dto` — the SAME vocabulary the backend
29//! uses, so the wire stays byte-identical. The client still does not interpret the
30//! ceremony content; it threads the typed value between the authenticator and the
31//! backend.

71/// HTTP client bound to one backend base URL, carrying the optional session token.
72/// Cheap to clone-share via `Arc<RestClient>`; the token cell is interior-mutable.
73pub struct RestClient {
74    /// Built on first request, not at construction: `reqwest::Client::new`
75    /// runs macOS system-proxy discovery (SCDynamicStore), which stalls ~2 s
76    /// in unbundled binaries (dev runs, every per-test process). Deferring it
77    /// keeps constructors instant and moves the one-time cost onto the first
78    /// actual HTTP call, off the caller's startup path.
79    http: std::sync::OnceLock<Client>,
80    base: Url,
81    /// Current bearer token (`None` when signed out). Swapped atomically by auth.
82    token: ArcSwapOption<String>,
83    /// The installed refresh hook, if any. `None` (a client built without a
84    /// [`Session`](crate::session::Session)) simply never retries an expired 401.
85    /// `Box`ed because arc-swap's `RefCnt` covers only `Arc<T: Sized>`, so the trait
86    /// object needs the extra indirection to live in the cell.
87    reauth: ArcSwapOption<Box<dyn Reauth>>,

90/// The refresh seam. The REST layer knows only "ask whoever owns the session to
91/// rotate the access token that was just rejected"; [`crate::session::Session`]
92/// supplies the implementation (and the single-flight guarantee) in its constructor.
93///
94/// Returns a boxed future rather than being an `async fn`: the hook is held as
95/// `dyn Reauth`, and async-fn-in-trait is not dyn-compatible.
96pub trait Reauth: Send + Sync {
97    /// Rotate the session tokens, given the access token whose rejection triggered
98    /// this. Implementations MUST be single-flight over `stale_access` — several
99    /// in-flight requests hit the same expiry at once and only one exchange may run.
100    fn reauth(&self, stale_access: String) -> futures::future::BoxFuture<'static, Result<()>>;
101}

103/// Which auth failure the backend reported in a 401 body. Only [`AuthCode::Expired`]
104/// is worth a refresh-and-retry; the others describe a session that is dead however
105/// often we re-present it. (Backend: `services/backend/src/api/auth/jwt.rs`.)
106#[derive(Debug, Clone, Copy, PartialEq, Eq)]
107enum AuthCode {
108    Expired,
109    Invalid,
110    Revoked,
111    InvalidType,
112    /// Any body we can't read a known code out of — including a non-JSON one (a
113    /// proxy's HTML error page), which is why parsing must never be fatal here.
114    Unknown,
115}
116
117impl AuthCode {
118    fn from_body(body: &str) -> Self {
119        /// The backend's error body: `{ "error": "…", "code": "…" }`, where `code`
120        /// is the lowercased `AuthError` variant name.
121        #[derive(serde::Deserialize)]
122        struct WireAuthError {
123            code: String,
124        }
125
126        let Ok(wire) = serde_json::from_str::<WireAuthError>(body) else {
127            return Self::Unknown;
128        };
129        match wire.code.as_str() {
130            "expiredtoken" => Self::Expired,
131            "invalidtoken" => Self::Invalid,
132            "revokedtoken" => Self::Revoked,
133            "invalidtokentype" => Self::InvalidType,
134            _ => Self::Unknown,
135        }
136    }
137}

139/// Re-label a `501` from a Steam auth route as [`SteamLoginDisabled`].
140///
141/// The backend answers 501 on every `/auth/steam/*` route when the instance has no
142/// Steam credentials (`ApiError::SteamDisabled`) — an instance-configuration fact,
143/// not a request failure, and the one Steam outcome the UI must never treat as
144/// retryable. As a bare `HoundError::Http` it reached the UI (and Sentry) as
145/// `HTTP501` with whatever body the deployment happened to send, which on older
146/// backends is nothing at all; this gives it the registry code the caller branches
147/// on regardless of body.
148///
149/// [`SteamLoginDisabled`]: crate::error::AuthCode::SteamLoginDisabled
150fn steam_disabled(e: HoundError) -> HoundError {
151    match e {
152        HoundError::Http { status: 501, .. } => HoundError::auth(
153            crate::error::AuthCode::SteamLoginDisabled,
154            "steam sign-in is not enabled on this server",
155        ),
156        other => other,
157    }
158}

160/// Build the error for a non-success response.
161///
162/// One place, on the path EVERY verb's decode takes, so no call site can forget
163/// to map a refusal (the previous per-call `map_err` was opt-in, and the routes
164/// that forgot it showed developer prose to users).
165///
166/// A `400`/`403` whose body names a refusal this build knows becomes
167/// [`HoundError::Refused`] — the status, the backend's canonical code, and the
168/// sentence the user reads. Everything else keeps the generic route-prefixed
169/// shape, so the server's own detail still reaches triage.
170fn http_error(status: u16, at: &str, body: &str) -> HoundError {
171    if matches!(status, 400 | 403)
172        && let Some((code, copy)) = refusal(body)
173    {
174        return HoundError::refused(status, code, copy);
175    }
176    HoundError::from_endpoint(status, at, body)
177}

179/// The canonical code and the user-facing sentence for a refusal body, or
180/// `None` when it names none this build knows.
181///
182/// **Matched exactly, on the `code` field alone.** The body is parsed here —
183/// before [`HoundError::from_endpoint`] has composed the route label and the
184/// server's prose into one string — so the only thing compared is the machine
185/// code, folded to bare lowercase letters and digits (`EmptySpaceName`,
186/// `empty_space_name` and `emptyspacename` are one code; `cannot_ban_self` is
187/// not `banned` with extra letters). A body carrying no `code` field falls
188/// through: prose is not a code, and treating it as one made an unrelated
189/// sentence mentioning a token answer for it.
190fn refusal(body: &str) -> Option<(&'static str, &'static str)> {
191    #[derive(serde::Deserialize)]
192    struct WireCode {
193        code: String,
194    }
195
196    fn fold(s: &str) -> String {
197        s.chars()
198            .filter(char::is_ascii_alphanumeric)
199            .map(|c| c.to_ascii_lowercase())
200            .collect()
201    }
202
203    let wire: WireCode = serde_json::from_str(body).ok()?;
204    // The table: folded wire code → (canonical code, the sentence). The
205    // canonical spelling is what travels on the error as its grouping key, so a
206    // backend that rewords its own prose never moves the key.
207    let mapped = match fold(&wire.code).as_str() {
208        "emptycommunityname" => ("empty_community_name", "A community needs a name."),
209        "communitynametoolong" => (
210            "community_name_too_long",
211            "That community name is too long.",
212        ),
213        "communitydescriptiontoolong" => (
214            "community_description_too_long",
215            "That description is too long.",
216        ),
217        "invaliddefaultspace" => (
218            "invalid_default_space",
219            "The default channel must be a text channel in this community.",
220        ),
221        "emptyspacename" => ("empty_space_name", "A channel needs a name."),
222        "spacenametoolong" => ("space_name_too_long", "That channel name is too long."),
223        "defaultspace" => (
224            "default_space",
225            "The default channel cannot be deleted — pick a different default first.",
226        ),
227        "lastorganiser" => (
228            "last_organiser",
229            "You are the last organiser — make someone else an organiser first.",
230        ),
231        "emptyrolename" => ("empty_role_name", "A role needs a name."),
232        "rolenametoolong" => ("role_name_too_long", "That role name is too long."),
233        // Invites and bans.
234        "invitenotactive" => ("invite_not_active", "This invite is no longer valid."),
235        "inviteexpirytoolong" => (
236            "invite_expiry_too_long",
237            "Invites can last at most 30 days.",
238        ),
239        "invalidmaxuses" => ("invalid_max_uses", "Max uses must be between 1 and 1000."),
240        "cannotbanself" => ("cannot_ban_self", "You cannot ban yourself."),
241        "cannotkickself" => (
242            "cannot_kick_self",
243            "You cannot remove yourself — leave the community instead.",
244        ),
245        "banreasontoolong" => ("ban_reason_too_long", "That ban note is too long."),
246        // Channel overwrites.
247        "duplicateoverwrite" => (
248            "duplicate_overwrite",
249            "That role or member already has an entry for this permission.",
250        ),
251        // The role guard and the ban wall: 403s that name their own reason,
252        // which is the whole message — nothing else in the stack can
253        // reconstruct "you may not edit a role that outranks you" from a status.
254        "banned" => ("banned", "You are banned from this community."),
255        "roleaboveyou" => (
256            "role_above_you",
257            "You can only manage roles whose permissions you hold.",
258        ),
259        "permissionnotheld" => (
260            "permission_not_held",
261            "You cannot grant a permission you do not hold.",
262        ),
263        "notamember" => ("not_a_member", "They are not a member of this community."),
264        _ => return None,
265    };
266    Some(mapped)
267}

269/// Cap on a route label. Bounded because the label becomes a Sentry TAG, and an
270/// unbounded tag is a grouping key that never groups. Every real route is far shorter;
271/// the cap exists so a pathological path can't be the one that proves otherwise.
272const MAX_ENDPOINT_LEN: usize = 64;
273
274/// The largest `per_page` the backend honours. It **clamps** rather than
275/// refuses, so asking for more is not an error — it is a short page that looks
276/// like the last one, which is how a list silently loses its tail.
277pub const PAGE_MAX: u32 = 80;
278
279/// Read every page of a paginated route into one list.
280///
281/// The backend marks the last page by pointing `next_page` back at the current
282/// one; `total` is the second stop, so a server that always advances cannot
283/// spin this forever. For callers whose answer must be COMPLETE — the sidebar's
284/// communities and channels — rather than one screenful.
285async fn drain<T, F>(mut fetch: impl FnMut(u32, u32) -> F) -> Result<Vec<T>>
286where
287    F: std::future::Future<Output = Result<Page<T>>>,
288{
289    let mut all = Vec::new();
290    let mut page = 0;
291    loop {
292        let this = fetch(page, PAGE_MAX).await?;
293        let last = this.next_page <= this.page;
294        all.extend(this.items);
295        if last || all.len() as u64 >= this.total {
296            return Ok(all);
297        }
298        page = this.next_page;
299    }
300}

302/// The `METHOD /route` label a request is named by in errors and telemetry.
303///
304/// Ids are elided (`/space/{id}/message`, never the uuid) for two independent
305/// reasons: a tag with a uuid in it has one value per request and groups nothing, and
306/// an id in a tag is user data leaving the device in a field nobody audits. The query
307/// string goes the same way — its values are the most likely to be identifying, and
308/// the path alone already says which endpoint answered.
309///
310/// An **invite code** is elided too, by position: the segment right after a
311/// literal `invite` is `{code}`. It is an id in every sense that matters here —
312/// unbounded as a tag, and a secret good enough to join a community with — but
313/// it is neither a uuid nor a number, so it would otherwise ride out verbatim.
314/// Positional rather than shape-matching so no route segment that happens to
315/// look like a code is ever swallowed.
316fn endpoint(method: &str, path: &str) -> String {
317    let path = path.split(['?', '#']).next().unwrap_or(path);
318    let mut label = String::with_capacity(method.len() + path.len() + 2);
319    label.push_str(method);
320    label.push(' ');
321    let mut after_invite = false;
322    for segment in path.split('/') {
323        label.push('/');
324        if Uuid::parse_str(segment).is_ok() {
325            label.push_str("{id}");
326        } else if !segment.is_empty() && segment.bytes().all(|b| b.is_ascii_digit()) {
327            label.push_str("{n}");
328        } else if after_invite && !segment.is_empty() {
329            label.push_str("{code}");
330        } else {
331            label.push_str(segment);
332        }
333        after_invite = segment == "invite";
334        if label.len() > MAX_ENDPOINT_LEN {
335            let mut cut = MAX_ENDPOINT_LEN;
336            while !label.is_char_boundary(cut) {
337                cut -= 1;
338            }
339            label.truncate(cut);
340            break;
341        }
342    }
343    label
344}

350    /// Build a client for `base_url` (e.g. `http://localhost:8080` /
351    /// `https://api.hound.chat`). Errors only on a malformed base URL.
352    pub fn new(base_url: &str) -> Result<Self> {
353        // A trailing slash makes `Url::join` treat the base as a directory, so a
354        // relative path like `community` resolves under it rather than replacing
355        // the last segment. Normalise here so callers pass clean relative paths.
356        let base_url = if base_url.ends_with('/') {
357            base_url.to_owned()
358        } else {
359            format!("{base_url}/")
360        };
361        let base = Url::parse(&base_url).map_err(|e| {
362            HoundError::internal(
363                InternalCode::InvalidBaseUrl,
364                format!("invalid base url {base_url:?}: {e}"),
365            )
366        })?;
367        Ok(Self {
368            http: std::sync::OnceLock::new(),
369            base,
370            token: ArcSwapOption::empty(),
371            reauth: ArcSwapOption::empty(),
372        })
373    }

375    /// The lazily-built HTTP client (see the field docs for why lazy).
376    fn http(&self) -> &Client {
377        self.http.get_or_init(|| {
378            // Bounded, or a half-dead server (TCP accepted, upstream never
379            // answers — a churning ingress does exactly this) hangs every
380            // caller forever. `restore()` runs before the app's first frame,
381            // so an unbounded hang here is a permanent black screen the
382            // crash reporter can't see (nothing crashed, nothing hung — the
383            // isolate is just awaiting). read_timeout rather than a total
384            // `timeout` so long attachment transfers aren't capped; it fires
385            // only when the socket goes idle.
386            let builder = Client::builder()
387                .connect_timeout(std::time::Duration::from_secs(10))
388                .read_timeout(std::time::Duration::from_secs(20));
389            // Test-harness DNS pinning (remote-target e2e runs) — a no-op
390            // unless `hound_client::dns::set_overrides` ran first.
391            #[cfg(feature = "dns-override")]
392            let builder = crate::dns::apply(builder);
393            builder.build().expect("building reqwest client")
394        })
395    }

435    /// The request core every verb helper funnels through: attach the current bearer
436    /// token, send, and — on a `401 expiredtoken` with a [`Reauth`] hook installed —
437    /// rotate the session and resend ONCE. At most one retry per call, so a backend
438    /// that answers `expiredtoken` to a freshly minted token cannot loop us.
439    ///
440    /// `build` is a closure, not a `RequestBuilder`, for two reasons: a builder is
441    /// single-use (consumed by `send`), and re-invoking it re-reads the token cell,
442    /// which is what makes the retry carry the *rotated* token rather than the dead
443    /// one. It must therefore be a pure re-build — no side effects.
444    async fn send(
445        &self,
446        endpoint: &str,
447        build: impl Fn() -> reqwest::RequestBuilder,
448    ) -> Result<reqwest::Response> {
449        let stale = self.token.load_full();
450        let resp = self.send_once(&build).await?;
451        if resp.status() != reqwest::StatusCode::UNAUTHORIZED {
452            return Ok(resp);
453        }
454        // No session token ⇒ nothing to rotate (an unauthenticated endpoint answering
455        // 401 is the caller's error); no hook ⇒ nobody owns the session.
456        let (Some(stale), Some(hook)) = (stale, self.reauth.load_full()) else {
457            return Ok(resp);
458        };
459        // Reading the body is what tells expiry apart from a dead session; the
460        // response is consumed either way, so from here we build the error ourselves.
461        let body = resp.text().await.unwrap_or_default();
462        if AuthCode::from_body(&body) != AuthCode::Expired {
463            return Err(HoundError::from_endpoint(401, endpoint, &body));
464        }
465        if let Err(err) = hook.reauth(stale.as_str().to_owned()).await {
466            return Err(Self::refresh_failure(err, endpoint, &body));
467        }
468        self.send_once(&build).await
469    }

471    /// What the caller is told when the refresh triggered by an expired 401 did
472    /// not itself succeed.
473    ///
474    /// The distinction is whether we LEARNED anything about the session. A
475    /// refresh the backend REFUSED (auth-class) says the refresh token is dead
476    /// too, so the original 401 is the honest answer and the caller may route to
477    /// sign-out. A refresh that never got an answer — offline, DNS, a 5xx, a
478    /// database outage behind the auth service — says nothing at all about the
479    /// session, and reporting it as a 401 turned every such blip into a
480    /// sign-out (the realtime supervisor and `Session`'s proactive refresh both
481    /// classify `Auth` as fatal). Those propagate as themselves, which
482    /// [`classify`](crate::call::reconnect::classify) calls retryable — the same
483    /// verdict the proactive-refresh task already reaches for a non-auth failure.
484    fn refresh_failure(err: HoundError, endpoint: &str, rejection: &str) -> HoundError {
485        use crate::call::reconnect::{Disposition, classify};
486        match classify(&err) {
487            Disposition::Fatal => {
488                tracing::warn!(%err, "rest: the session refresh was REFUSED — the session is dead");
489                HoundError::from_endpoint(401, endpoint, rejection)
490            }
491            Disposition::Retryable => {
492                tracing::warn!(%err, "rest: the session refresh failed transiently — surfacing it as retryable");
493                err
494            }
495        }
496    }

498    /// One build → attach the *current* token → send. Split out so the retry reads
499    /// the token cell again after the hook rotated it.
500    async fn send_once(
501        &self,
502        build: &impl Fn() -> reqwest::RequestBuilder,
503    ) -> Result<reqwest::Response> {
504        let mut req = build();
505        if let Some(tok) = self.token.load_full() {
506            req = req.bearer_auth(tok.as_str());
507        }
508        Ok(req.send().await?)
509    }

600    /// POST one file as a `multipart/form-data` body with a single `file` part
601    /// (the part name the upload routes look for), decoding the JSON response.
602    ///
603    /// The part carries an explicit `Content-Type`: the server fast-paths on the
604    /// DECLARED type and then verifies the bytes' magic against it, so a part
605    /// with no type is rejected as unsupported however valid the image is.
606    /// `bytes` is moved, not borrowed — the retry seam rebuilds the request, so
607    /// the body must be re-creatable from an owned buffer.
608    ///
609    /// `mime` is a parsed [`Mime`], not a string: a media type this layer would
610    /// have to re-validate (and re-fail on) is a caller's argument error, and
611    /// taking the type makes it unrepresentable here.
612    async fn post_file<T: DeserializeOwned>(
613        &self,
614        path: &str,
615        bytes: Vec<u8>,
616        mime: &Mime,
617    ) -> Result<T> {
618        let url = self.url(path)?;
619        let at = endpoint("POST", path);
620        let build = || {
621            // Rebuilt per attempt: a `Form` is consumed by the request it is
622            // attached to, and the 401-refresh path sends twice.
623            let part = reqwest::multipart::Part::bytes(bytes.clone())
624                .file_name("upload")
625                .mime_str(mime.as_ref())
626                .expect("a parsed Mime is a valid media type");
627            self.http()
628                .post(url.clone())
629                .multipart(reqwest::multipart::Form::new().part("file", part))
630        };
631        Self::decode(&at, self.send(&at, build).await?).await
632    }

678    /// Shared response handling: map non-success status onto a typed error
679    /// (reading the body for detail, named by the route that produced it), else
680    /// decode the JSON body into `T`.
681    async fn decode<T: DeserializeOwned>(at: &str, resp: reqwest::Response) -> Result<T> {
682        let status = resp.status();
683        if !status.is_success() {
684            let body = resp.text().await.unwrap_or_default();
685            return Err(http_error(status.as_u16(), at, &body));
686        }
687        let bytes = resp.bytes().await?;
688        serde_json::from_slice(&bytes).map_err(HoundError::from)
689    }
