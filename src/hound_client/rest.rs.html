1//! The REST data plane: a thin, typed wrapper over the backend HTTP API. Holds the
2//! session bearer token in a lock-free cell (read on every request, republished by
3//! the auth refresh task) and maps transport/status failures onto [`HoundError`].
4//!
5//! **Refresh seam:** every verb funnels through one request core that retries a
6//! request ONCE when the backend answers `401 expiredtoken` and a [`Reauth`] hook is
7//! installed ([`crate::session::Session`] installs one). The hook rotates the tokens
8//! and republishes the access token here, so the rebuilt request carries the fresh
9//! one. A REFUSED refresh, any other auth code, or a second failure surfaces as
10//! [`HoundError::Auth`], which is therefore a genuinely dead session — but a refresh
11//! that merely *failed* (offline, 5xx) surfaces as its own retryable error, because
12//! it proves nothing about the session and callers treat `Auth` as terminal.
13//!
14//! **Wire mapping discipline:** responses are decoded against private wire DTOs
15//! (the backend's serde, snake_case) and mapped into the *client* view-models in
16//! [`crate::model`] via `into_model()`/`From`. Wire quirks — the pagination envelope
17//! (`contents`/`current`/`next`/`prev`), the nested community `spaces`, the
18//! `messages`+`users` split, `/profile/global`, `/community/members?id=` — are
19//! confined to this module. The exact shapes live in
20//! `services/backend/src/api/{auth,community,space,profile,data}.rs`.
21//!
22//! WebAuthn note: register/login `start`/`finish` carry the ceremony payloads as the
23//! TYPED `hound-passkey` wire types ([`CreationChallengeResponse`] /
24//! [`RequestChallengeResponse`] from the server, [`RegisterPublicKeyCredential`] /
25//! [`PublicKeyCredential`] from the authenticator) and (de)serialise the request /
26//! response bodies through `hound_passkey::dto` — the SAME vocabulary the backend
27//! uses, so the wire stays byte-identical. The client still does not interpret the
28//! ceremony content; it threads the typed value between the authenticator and the
29//! backend.

54/// HTTP client bound to one backend base URL, carrying the optional session token.
55/// Cheap to clone-share via `Arc<RestClient>`; the token cell is interior-mutable.
56pub struct RestClient {
57    /// Built on first request, not at construction: `reqwest::Client::new`
58    /// runs macOS system-proxy discovery (SCDynamicStore), which stalls ~2 s
59    /// in unbundled binaries (dev runs, every per-test process). Deferring it
60    /// keeps constructors instant and moves the one-time cost onto the first
61    /// actual HTTP call, off the caller's startup path.
62    http: std::sync::OnceLock<Client>,
63    base: Url,
64    /// Current bearer token (`None` when signed out). Swapped atomically by auth.
65    token: ArcSwapOption<String>,
66    /// The installed refresh hook, if any. `None` (a client built without a
67    /// [`Session`](crate::session::Session)) simply never retries an expired 401.
68    /// `Box`ed because arc-swap's `RefCnt` covers only `Arc<T: Sized>`, so the trait
69    /// object needs the extra indirection to live in the cell.
70    reauth: ArcSwapOption<Box<dyn Reauth>>,
71}

73/// The refresh seam. The REST layer knows only "ask whoever owns the session to
74/// rotate the access token that was just rejected"; [`crate::session::Session`]
75/// supplies the implementation (and the single-flight guarantee) in its constructor.
76///
77/// Returns a boxed future rather than being an `async fn`: the hook is held as
78/// `dyn Reauth`, and async-fn-in-trait is not dyn-compatible.
79pub trait Reauth: Send + Sync {
80    /// Rotate the session tokens, given the access token whose rejection triggered
81    /// this. Implementations MUST be single-flight over `stale_access` — several
82    /// in-flight requests hit the same expiry at once and only one exchange may run.
83    fn reauth(&self, stale_access: String) -> futures::future::BoxFuture<'static, Result<()>>;
84}

86/// Which auth failure the backend reported in a 401 body. Only [`AuthCode::Expired`]
87/// is worth a refresh-and-retry; the others describe a session that is dead however
88/// often we re-present it. (Backend: `services/backend/src/api/auth/jwt.rs`.)
89#[derive(Debug, Clone, Copy, PartialEq, Eq)]
90enum AuthCode {
91    Expired,
92    Invalid,
93    Revoked,
94    InvalidType,
95    /// Any body we can't read a known code out of — including a non-JSON one (a
96    /// proxy's HTML error page), which is why parsing must never be fatal here.
97    Unknown,
98}
99
100impl AuthCode {
101    fn from_body(body: &str) -> Self {
102        /// The backend's error body: `{ "error": "…", "code": "…" }`, where `code`
103        /// is the lowercased `AuthError` variant name.
104        #[derive(serde::Deserialize)]
105        struct WireAuthError {
106            code: String,
107        }
108
109        let Ok(wire) = serde_json::from_str::<WireAuthError>(body) else {
110            return Self::Unknown;
111        };
112        match wire.code.as_str() {
113            "expiredtoken" => Self::Expired,
114            "invalidtoken" => Self::Invalid,
115            "revokedtoken" => Self::Revoked,
116            "invalidtokentype" => Self::InvalidType,
117            _ => Self::Unknown,
118        }
119    }
120}

122/// Re-label a `501` from a Steam auth route as [`SteamLoginDisabled`].
123///
124/// The backend answers 501 on every `/auth/steam/*` route when the instance has no
125/// Steam credentials (`ApiError::SteamDisabled`) — an instance-configuration fact,
126/// not a request failure, and the one Steam outcome the UI must never treat as
127/// retryable. As a bare `HoundError::Http` it reached the UI (and Sentry) as
128/// `HTTP501` with whatever body the deployment happened to send, which on older
129/// backends is nothing at all; this gives it the registry code the caller branches
130/// on regardless of body.
131///
132/// [`SteamLoginDisabled`]: crate::error::AuthCode::SteamLoginDisabled
133fn steam_disabled(e: HoundError) -> HoundError {
134    match e {
135        HoundError::Http { status: 501, .. } => HoundError::auth(
136            crate::error::AuthCode::SteamLoginDisabled,
137            "steam sign-in is not enabled on this server",
138        ),
139        other => other,
140    }
141}

143/// Cap on a route label. Bounded because the label becomes a Sentry TAG, and an
144/// unbounded tag is a grouping key that never groups. Every real route is far shorter;
145/// the cap exists so a pathological path can't be the one that proves otherwise.
146const MAX_ENDPOINT_LEN: usize = 64;
147
148/// The `METHOD /route` label a request is named by in errors and telemetry.
149///
150/// Ids are elided (`/space/{id}/message`, never the uuid) for two independent
151/// reasons: a tag with a uuid in it has one value per request and groups nothing, and
152/// an id in a tag is user data leaving the device in a field nobody audits. The query
153/// string goes the same way — its values are the most likely to be identifying, and
154/// the path alone already says which endpoint answered.
155fn endpoint(method: &str, path: &str) -> String {
156    let path = path.split(['?', '#']).next().unwrap_or(path);
157    let mut label = String::with_capacity(method.len() + path.len() + 2);
158    label.push_str(method);
159    label.push(' ');
160    for segment in path.split('/') {
161        label.push('/');
162        if Uuid::parse_str(segment).is_ok() {
163            label.push_str("{id}");
164        } else if !segment.is_empty() && segment.bytes().all(|b| b.is_ascii_digit()) {
165            label.push_str("{n}");
166        } else {
167            label.push_str(segment);
168        }
169        if label.len() > MAX_ENDPOINT_LEN {
170            let mut cut = MAX_ENDPOINT_LEN;
171            while !label.is_char_boundary(cut) {
172                cut -= 1;
173            }
174            label.truncate(cut);
175            break;
176        }
177    }
178    label
179}

182    /// Build a client for `base_url` (e.g. `http://localhost:8080` /
183    /// `https://api.hound.chat`). Errors only on a malformed base URL.
184    pub fn new(base_url: &str) -> Result<Self> {
185        // A trailing slash makes `Url::join` treat the base as a directory, so a
186        // relative path like `community` resolves under it rather than replacing
187        // the last segment. Normalise here so callers pass clean relative paths.
188        let base_url = if base_url.ends_with('/') {
189            base_url.to_owned()
190        } else {
191            format!("{base_url}/")
192        };
193        let base = Url::parse(&base_url).map_err(|e| {
194            HoundError::internal(
195                InternalCode::InvalidBaseUrl,
196                format!("invalid base url {base_url:?}: {e}"),
197            )
198        })?;
199        Ok(Self {
200            http: std::sync::OnceLock::new(),
201            base,
202            token: ArcSwapOption::empty(),
203            reauth: ArcSwapOption::empty(),
204        })
205    }

207    /// The lazily-built HTTP client (see the field docs for why lazy).
208    fn http(&self) -> &Client {
209        self.http.get_or_init(|| {
210            // Bounded, or a half-dead server (TCP accepted, upstream never
211            // answers — a churning ingress does exactly this) hangs every
212            // caller forever. `restore()` runs before the app's first frame,
213            // so an unbounded hang here is a permanent black screen the
214            // crash reporter can't see (nothing crashed, nothing hung — the
215            // isolate is just awaiting). read_timeout rather than a total
216            // `timeout` so long attachment transfers aren't capped; it fires
217            // only when the socket goes idle.
218            let builder = Client::builder()
219                .connect_timeout(std::time::Duration::from_secs(10))
220                .read_timeout(std::time::Duration::from_secs(20));
221            // Test-harness DNS pinning (remote-target e2e runs) — a no-op
222            // unless `hound_client::dns::set_overrides` ran first.
223            #[cfg(feature = "dns-override")]
224            let builder = crate::dns::apply(builder);
225            builder.build().expect("building reqwest client")
226        })
227    }

267    /// The request core every verb helper funnels through: attach the current bearer
268    /// token, send, and — on a `401 expiredtoken` with a [`Reauth`] hook installed —
269    /// rotate the session and resend ONCE. At most one retry per call, so a backend
270    /// that answers `expiredtoken` to a freshly minted token cannot loop us.
271    ///
272    /// `build` is a closure, not a `RequestBuilder`, for two reasons: a builder is
273    /// single-use (consumed by `send`), and re-invoking it re-reads the token cell,
274    /// which is what makes the retry carry the *rotated* token rather than the dead
275    /// one. It must therefore be a pure re-build — no side effects.
276    async fn send(
277        &self,
278        endpoint: &str,
279        build: impl Fn() -> reqwest::RequestBuilder,
280    ) -> Result<reqwest::Response> {
281        let stale = self.token.load_full();
282        let resp = self.send_once(&build).await?;
283        if resp.status() != reqwest::StatusCode::UNAUTHORIZED {
284            return Ok(resp);
285        }
286        // No session token ⇒ nothing to rotate (an unauthenticated endpoint answering
287        // 401 is the caller's error); no hook ⇒ nobody owns the session.
288        let (Some(stale), Some(hook)) = (stale, self.reauth.load_full()) else {
289            return Ok(resp);
290        };
291        // Reading the body is what tells expiry apart from a dead session; the
292        // response is consumed either way, so from here we build the error ourselves.
293        let body = resp.text().await.unwrap_or_default();
294        if AuthCode::from_body(&body) != AuthCode::Expired {
295            return Err(HoundError::from_endpoint(401, endpoint, &body));
296        }
297        if let Err(err) = hook.reauth(stale.as_str().to_owned()).await {
298            return Err(Self::refresh_failure(err, endpoint, &body));
299        }
300        self.send_once(&build).await
301    }

303    /// What the caller is told when the refresh triggered by an expired 401 did
304    /// not itself succeed.
305    ///
306    /// The distinction is whether we LEARNED anything about the session. A
307    /// refresh the backend REFUSED (auth-class) says the refresh token is dead
308    /// too, so the original 401 is the honest answer and the caller may route to
309    /// sign-out. A refresh that never got an answer — offline, DNS, a 5xx, a
310    /// database outage behind the auth service — says nothing at all about the
311    /// session, and reporting it as a 401 turned every such blip into a
312    /// sign-out (the realtime supervisor and `Session`'s proactive refresh both
313    /// classify `Auth` as fatal). Those propagate as themselves, which
314    /// [`classify`](crate::call::reconnect::classify) calls retryable — the same
315    /// verdict the proactive-refresh task already reaches for a non-auth failure.
316    fn refresh_failure(err: HoundError, endpoint: &str, rejection: &str) -> HoundError {
317        use crate::call::reconnect::{Disposition, classify};
318        match classify(&err) {
319            Disposition::Fatal => {
320                tracing::warn!(%err, "rest: the session refresh was REFUSED — the session is dead");
321                HoundError::from_endpoint(401, endpoint, rejection)
322            }
323            Disposition::Retryable => {
324                tracing::warn!(%err, "rest: the session refresh failed transiently — surfacing it as retryable");
325                err
326            }
327        }
328    }

330    /// One build → attach the *current* token → send. Split out so the retry reads
331    /// the token cell again after the hook rotated it.
332    async fn send_once(
333        &self,
334        build: &impl Fn() -> reqwest::RequestBuilder,
335    ) -> Result<reqwest::Response> {
336        let mut req = build();
337        if let Some(tok) = self.token.load_full() {
338            req = req.bearer_auth(tok.as_str());
339        }
340        Ok(req.send().await?)
341    }

431    /// Shared response handling: map non-success status onto a typed error
432    /// (reading the body for detail, named by the route that produced it), else
433    /// decode the JSON body into `T`.
434    async fn decode<T: DeserializeOwned>(at: &str, resp: reqwest::Response) -> Result<T> {
435        let status = resp.status();
436        if !status.is_success() {
437            let body = resp.text().await.unwrap_or_default();
438            return Err(HoundError::from_endpoint(status.as_u16(), at, &body));
439        }
440        let bytes = resp.bytes().await?;
441        serde_json::from_slice(&bytes).map_err(HoundError::from)
442    }
443
444    /// Like [`decode`] but for endpoints whose success body is empty/ignored.
445    async fn decode_unit(at: &str, resp: reqwest::Response) -> Result<()> {
446        let status = resp.status();
447        if !status.is_success() {
448            let body = resp.text().await.unwrap_or_default();
449            return Err(HoundError::from_endpoint(status.as_u16(), at, &body));
450        }
451        Ok(())
452    }

518    /// `POST /auth/token/refresh` — exchanges a refresh token (sent as a `Bearer`
519    /// header) for a fresh access token + rotated refresh token. The backend reads
520    /// the refresh token from the `Authorization` header, not the body.
521    ///
522    /// Built outside [`send`](Self::send) deliberately: it authenticates with the
523    /// *refresh* token, and keeping it off the retry path is what makes recursion
524    /// impossible — a 401 here is the end of the session, not another refresh.
525    pub async fn refresh_token(&self, refresh_token: &str) -> Result<TokenRefresh> {
526        let req = self
527            .http()
528            .post(self.url("auth/token/refresh")?)
529            .bearer_auth(refresh_token);
530        let w: hound_types::RefreshTokenResponse =
531            Self::decode(&endpoint("POST", "auth/token/refresh"), req.send().await?).await?;
532        Ok(TokenRefresh {
533            access_token: w.access_token,
534            refresh_token: w.refresh_token,
535            access_token_expires_in: w.expires_in,
536        })
537    }

1029/// The backend pagination envelope. View-models use [`Page`] (`items`/`page`/…).
1030#[derive(serde::Deserialize)]
1031struct WirePage<T> {
1032    contents: Vec<T>,
1033    current: u64,
1034    next: u64,
1035    prev: u64,
1036    total: u64,
1037}
1038
1039impl<W> WirePage<W> {
1040    /// Map each wire item into its view-model and convert the envelope.
1041    fn map_model<M>(self) -> Page<M>
1042    where
1043        W: IntoModel<M>,
1044    {
1045        Page {
1046            items: self
1047                .contents
1048                .into_iter()
1049                .map(IntoModel::into_model)
1050                .collect(),
1051            page: self.current as u32,
1052            next_page: self.next as u32,
1053            prev_page: self.prev as u32,
1054            total: self.total,
1055        }
1056    }
1057}

1351/// Extract the unix-ms timestamp embedded in a UUIDv7. The backend keys all ids with
1352/// `Uuid::now_v7()`, whose high 48 bits are the creation time in ms — so a message's
1353/// id *is* its send time, sparing us a separate field the backend never sends.
1354/// Returns 0 for non-v7 uuids (the timestamp bits are still read but meaningless).
1355fn uuid_v7_millis(id: Uuid) -> i64 {
1356    let bytes = id.as_bytes();
1357    let ms = ((bytes[0] as u64) << 40)
1358        | ((bytes[1] as u64) << 32)
1359        | ((bytes[2] as u64) << 24)
1360        | ((bytes[3] as u64) << 16)
1361        | ((bytes[4] as u64) << 8)
1362        | (bytes[5] as u64);
1363    ms as i64
1364}
