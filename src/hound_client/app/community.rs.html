1//! Community + channel management commands: the settings surfaces' write path.
2//!
3//! Two kinds of command live here, and the difference is what the caller may
4//! assume when one returns:
5//!
6//! * **Tree-folding commands** (everything that changes a community or a space)
7//!   fold their own response into the community-tree snapshot before returning,
8//!   with `send_modify`. The realtime echo will say the same thing a moment
9//!   later and the reducers are idempotent, so this is not a source of truth —
10//!   it is what makes the editor's own client update at the speed of the
11//!   request instead of the speed of the socket. Nothing here is *optimistic*:
12//!   the fold happens after the server agreed.
13//! * **Fetch commands** (members, roles, bans, invites, channel overwrites,
14//!   instance) return a value and touch no snapshot. Those datasets are
15//!   paginated or settings-only and rarely open, so they are pulled on demand
16//!   and refreshed off [`CommunitySignal`](crate::model::CommunitySignal)
17//!   rather than reduced. Invites and overwrites have no signal at all —
18//!   nothing on the wire announces them — so whoever wrote one re-reads it.
19//!
20//! Joining is the one case that is both: `join_invite` and `create_community`
21//! fold the community they return AND fetch its channels, because the sidebar
22//! row would otherwise expand to nothing until the next full load.
23

37    /// Subscribe to "refetch the roles / members of this community" edges.
38    ///
39    /// A `broadcast` receiver, not a `watch`: these are events, not a snapshot,
40    /// and a new subscriber has nothing to catch up on (it is about to fetch
41    /// anyway). A receiver that falls behind is told it lagged and keeps going —
42    /// one missed signal costs a stale page, never correctness, because the next
43    /// fetch reads the truth.
44    pub fn watch_community_signals(&self) -> tokio::sync::broadcast::Receiver<CommunitySignal> {
45        self.state.community_signals.subscribe()
46    }
47
48    /// Publish one signal, stamped with its sequence number.
49    ///
50    /// The caller passes the variant's constructor rather than a built value,
51    /// because the number is not theirs to choose: it comes from here, and it
52    /// is the whole reason two identical edits are two distinct signals (see
53    /// [`CommunitySignal`]). Deliberately ignores "no subscribers": nothing in
54    /// the app needs these, only whichever settings page happens to be open.
55    pub(super) fn signal_community(&self, signal: impl FnOnce(u64) -> CommunitySignal) {
56        let _ = self
57            .state
58            .community_signals
59            .send(signal(self.next_signal_seq()));
60    }
61
62    /// A number no other signal carries. Wrapping is not a concern at one per
63    /// edit; consumers compare signals for equality and never read it.
64    fn next_signal_seq(&self) -> u64 {
65        self.state
66            .signal_seq
67            .fetch_add(1, std::sync::atomic::Ordering::Relaxed)
68    }
69
70    /// A [`CommunitySignal::Resync`] for a consumer that just found out it
71    /// MISSED signals (the bridge's stream, when its receiver lagged) — stamped
72    /// here so it is ordered among the ones it is standing in for.
73    pub fn community_resync_signal(&self) -> CommunitySignal {
74        CommunitySignal::Resync {
75            seq: self.next_signal_seq(),
76        }
77    }

92    /// Change a community's profile (name / description / landing channel /
93    /// icon). See [`CommunityPatch`] for the omit-vs-clear contract.
94    pub async fn update_community(&self, id: Uuid, patch: CommunityPatch) -> Result<Community> {
95        let community = self.rest.update_community(id, &patch).await?;

99            true

104    /// Set the community icon: upload the image, then point the community at it.
105    ///
106    /// Two round trips because they are two facts — the stored file and the
107    /// community's choice of it. `mime` must parse as a media type AND be one
108    /// this deployment advertises (see
109    /// [`supported_upload_mime_types`](HoundClient::supported_upload_mime_types));
110    /// the server verifies the bytes against it.
111    pub async fn set_community_icon(
112        &self,
113        id: Uuid,
114        bytes: Vec<u8>,
115        mime: &str,
116    ) -> Result<Community> {
117        // Parsed at the boundary, so nothing below here can be handed a media
118        // type that is not one.
119        let mime: mime::Mime = mime.parse().map_err(|e| {
120            HoundError::invalid_state(
121                StateCode::InvalidArgument,
122                format!("invalid upload media type {mime:?}: {e}"),
123            )
124        })?;
125        let upload_id = self.rest.upload_community_icon(id, bytes, &mime).await?;
126        self.update_community(
127            id,
128            CommunityPatch {
129                avatar_upload_id: Some(Some(upload_id)),
130                ..CommunityPatch::default()
131            },
132        )
133        .await
134    }

155    /// Reset a community to its default channels and roles.
156    ///
157    /// Refetched afterwards rather than reasoned about: a reset replaces the
158    /// whole space set server-side, so the only honest local answer is to ask
159    /// what is there now. A failed refetch is not a failed reset — the realtime
160    /// events and the next load still converge — so it is logged, not returned.
161    pub async fn reset_community(&self, id: Uuid) -> Result<()> {
162        self.rest.reset_community(id).await?;
163        if let Err(err) = self.refetch(id).await {
164            tracing::warn!(%id, %err, "community: reset succeeded but the refetch did not");
165        }
166        Ok(())
167    }

417    /// Fold a community, WITH its channels, into the tree in one publish.
418    ///
419    /// The single "this community is ours now" path — a create, a join, a
420    /// post-reset refetch. The space list is fetched rather than assumed
421    /// because the create/join responses carry the community alone, and a
422    /// sidebar row that expands to nothing is what the user would otherwise see
423    /// until the next `load_communities`.
424    ///
425    /// The space fetch is best-effort — a failure there is not a failed join,
426    /// and the community still lands (the realtime plane and the next load both
427    /// fill the channels in). Both writes happen in ONE `send_modify` so
428    /// subscribers never observe the half-state, and the community goes in
429    /// FIRST because the tree refuses channels for a community it does not
430    /// hold.
431    async fn adopt(&self, community: Community) -> Result<Community> {
432        let id = community.id;
433        let spaces = match self.rest.all_spaces(id).await {
434            Ok(spaces) => Some(spaces),
435            Err(err) => {
436                tracing::warn!(%id, %err, "community: adopted, but its channels did not load");
437                None
438            }
439        };
440        let folded = community.clone();
441        self.mutate(|tree| {
442            tree.upsert_community(folded);
443            if let Some(spaces) = spaces {
444                tree.set_spaces(id, spaces);
445            }
446            true
447        });
448        Ok(community)
449    }
450
451    /// Apply one mutation to the community tree, publishing only if it changed
452    /// something. The single write path, so "how a fold reaches subscribers" is
453    /// one decision rather than one per command.
454    fn mutate(&self, change: impl FnOnce(&mut crate::reducer::CommunityTree) -> bool) {
455        self.state.community_tree.send_if_modified(change);
456    }
457}

459/// Turn whatever the caller passed into a bare invite code, or refuse it here.
460///
461/// The commands take the raw string rather than a pre-parsed code so that a UI
462/// that hands over a pasted URL (or forgets to call
463/// [`parse_invite_code`](crate::parse_invite_code)) still works. But input the
464/// parser REJECTS never becomes a request: it would be spliced into the URL
465/// path — a pasted paragraph, a `../` segment, anything at all — to ask the
466/// server about something that cannot be a code. The answer is known without
467/// asking, so it is given here, with the same `404` shape (and ready-to-show
468/// copy) the server's own miss carries.
469fn invite_code(input: &str) -> Result<String> {
470    crate::parse_invite_code(input).ok_or_else(|| {
471        HoundError::refused(
472            404,
473            "invite_not_found",
474            "That does not look like an invite.",
475        )
476    })
477}

485    /// A backend that answers by path prefix — `(prefix, status, body)`, first
486    /// match wins — and `500`s anything unlisted.
487    ///
488    /// Raw HTTP because this crate has no server framework, and the fixture is
489    /// one response per request. What it buys is the half of a two-request
490    /// command that FAILS, which is the only way to pin what a command does
491    /// with the part it is allowed to lose.
492    async fn scripted(routes: &'static [(&'static str, u16, &'static str)]) -> String {
493        use tokio::io::{AsyncReadExt, AsyncWriteExt};
494
495        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
496        let addr = listener.local_addr().unwrap();
497        tokio::spawn(async move {
498            while let Ok((mut socket, _)) = listener.accept().await {
499                let mut buf = [0u8; 2048];
500                let n = socket.read(&mut buf).await.unwrap_or(0);
501                let head = String::from_utf8_lossy(&buf[..n]).into_owned();
502                // "GET /community/… HTTP/1.1" → "community/…".
503                let path = head
504                    .split_whitespace()
505                    .nth(1)
506                    .unwrap_or("/")
507                    .trim_start_matches('/')
508                    .to_owned();
509                let (status, body) = routes
510                    .iter()
511                    .find(|(prefix, _, _)| path.starts_with(prefix))
512                    .map(|(_, status, body)| (*status, *body))
513                    .unwrap_or((500, r#"{"error":"unscripted"}"#));
514                let _ = socket
515                    .write_all(
516                        format!(
517                            "HTTP/1.1 {status} OK\r\nContent-Type: application/json\r\n\
518                             Content-Length: {}\r\nConnection: close\r\n\r\n{body}",
519                            body.len()
520                        )
521                        .as_bytes(),
522                    )
523                    .await;
524            }
525        });
526        format!("http://{addr}")
527    }

532        HoundClient::new(&base, store, None).unwrap()
533    }
