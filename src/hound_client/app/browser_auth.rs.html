12/// Map a failed browser-callback wait onto the auth registry. An `error=` the
13/// auth page sent is a verdict, not a transport fault: the two Steam codes get
14/// their own [`AuthCode`] with the code itself as the message (the UI matches
15/// on it); anything else stays a generic wait failure.
16fn browser_wait_error(e: hound_browser_auth::BrowserAuthError) -> HoundError {
17    use hound_browser_auth::BrowserAuthError as E;
18    match e {
19        E::Rejected(code) => match code.as_str() {
20            "steam_cancelled" => HoundError::auth(AuthCode::SteamBrowserCancelled, code),
21            "steam_failed" => HoundError::auth(AuthCode::SteamBrowserFailed, code),
22            _ => HoundError::auth(AuthCode::BrowserCeremonyWait, code),
23        },
24        other => HoundError::auth(AuthCode::BrowserCeremonyWait, other.to_string()),
25    }
26}

29    /// Begin the browser-deferred email-link ceremony (platforms with no
30    /// system passkey provider): mint the scoped ceremony token, bind the
31    /// loopback callback, and return the auth-page URL. The wait ends in
32    /// [`AuthFlowOutcome::EmailLinked`].
33    pub async fn browser_link_email_begin(&self, auth_url: &str) -> Result<String> {
34        let token = self.rest.link_email_token().await?;
35        let login = hound_browser_auth::BrowserLogin::begin_email_link(auth_url, &token)
36            .await
37            .map_err(|e| HoundError::auth(AuthCode::BrowserCeremonyBegin, e.to_string()))?;
38        let url = login.url().to_owned();
39        *self.browser_login.lock().await = Some(login);
40        Ok(url)
41    }

45    /// Begin a browser-deferred login (same flow as `houndctl auth login`): bind
46    /// the loopback callback listener and return the hosted auth-page URL for the
47    /// UI to open + display. Replaces any previously pending flow. `signup` picks
48    /// the auth page's initial view (the user can still switch there).
49    pub async fn browser_login_begin(&self, auth_url: &str, signup: bool) -> Result<String> {
50        let mode = if signup {
51            hound_browser_auth::Mode::SignUp
52        } else {
53            hound_browser_auth::Mode::SignIn
54        };
55        self.browser_begin(auth_url, mode).await
56    }

83    /// Await the browser callback for the flow begun by
84    /// [`browser_login_begin`](Self::browser_login_begin) /
85    /// [`browser_steam_begin`](Self::browser_steam_begin). A token callback is
86    /// adopted as a session (auth published, realtime up); a Steam link-token
87    /// callback parks the token and returns [`AuthFlowOutcome::SteamUnknown`] for
88    /// the create-or-link screen (or the settings link). Errors if no flow is
89    /// pending, on timeout, or when
90    /// [`browser_login_cancel`](Self::browser_login_cancel) interrupts the wait.
91    pub async fn browser_login_wait(self: &Arc<Self>) -> Result<AuthFlowOutcome> {
92        /// Generous next to houndctl's 120 s: the UI shows a cancel button, so a
93        /// slow ceremony (finding the phone, cross-device passkey) shouldn't
94        /// time the flow out from under the user.
95        const BROWSER_LOGIN_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(300);
96
97        let login = self.browser_login.lock().await.take().ok_or_else(|| {
98            HoundError::invalid_state(
99                StateCode::NoBrowserLoginPending,
100                "no browser login in progress",
101            )
102        })?;
103
104        let callback = tokio::select! {
105            r = login.wait_callback(BROWSER_LOGIN_TIMEOUT) => r.map_err(browser_wait_error)?,
106            _ = self.browser_login_cancelled.notified() => {
107                return Err(HoundError::auth(
108                    AuthCode::BrowserLoginCancelled,
109                    "browser login cancelled",
110                ));
111            }
112        };
113
114        match callback {
115            hound_browser_auth::CallbackResult::Tokens(tokens) => {
116                let state = self
117                    .session
118                    .adopt_tokens(&tokens.access_token, &tokens.refresh_token)?;
119                Ok(AuthFlowOutcome::SignedIn(self.finish_sign_in(state).await?))
120            }
121            hound_browser_auth::CallbackResult::SteamLink {
122                link_token,
123                persona,
124                avatar_url,
125            } => {
126                self.park_steam_link(link_token).await;
127                Ok(AuthFlowOutcome::SteamUnknown {
128                    persona: persona.unwrap_or_default(),
129                    avatar_url,
130                })
131            }
132            hound_browser_auth::CallbackResult::Linked => Ok(AuthFlowOutcome::EmailLinked),
133            // Only ever produced by a `MagicLogin` wait, never a `BrowserLogin` one.
134            hound_browser_auth::CallbackResult::MagicUnknown { .. } => Err(HoundError::internal(
135                InternalCode::UnexpectedCallback,
136                "unexpected magic-link callback on a browser login wait",
137            )),
138        }
139    }

141    /// Cancel a pending browser login: drop an un-waited listener and wake a
142    /// blocked [`browser_login_wait`](Self::browser_login_wait). Idempotent.
143    pub async fn browser_login_cancel(&self) {
144        self.browser_login.lock().await.take();
145        self.browser_login_cancelled.notify_waiters();
146    }

150    /// Begin an email magic-link login: bind the loopback listener and ask the
151    /// backend to email `email` a sign-in link (carrying our port/state) plus a
152    /// bare code for manual entry. Follow with
153    /// [`magic_login_wait`](Self::magic_login_wait) (same-device link click)
154    /// and/or [`magic_login_code`](Self::magic_login_code) (typed code) —
155    /// whichever completes first wins; cancel the other with
156    /// [`magic_login_cancel`](Self::magic_login_cancel).
157    pub async fn magic_login_begin(&self, email: &str) -> Result<()> {
158        let login = hound_browser_auth::MagicLogin::begin()
159            .await
160            .map_err(|e| HoundError::auth(AuthCode::MagicLinkBegin, e.to_string()))?;
161        self.rest
162            .magic_start(email, login.port(), login.state())
163            .await?;
164        *self.magic_login.lock().await = Some(login);
165        Ok(())
166    }

168    /// Await the loopback callback for the flow begun by
169    /// [`magic_login_begin`](Self::magic_login_begin) — the emailed link
170    /// clicked on this device. A token callback is adopted as a session (auth
171    /// published, realtime up); an unknown-email callback parks the link token
172    /// and returns [`AuthFlowOutcome::EmailUnknown`] for the create-or-link
173    /// screen. Errors if no flow is pending, on timeout, or when
174    /// [`magic_login_cancel`](Self::magic_login_cancel) interrupts the wait.
175    pub async fn magic_login_wait(self: &Arc<Self>) -> Result<AuthFlowOutcome> {
176        /// Matches the browser-login wait's generosity — the UI shows a cancel
177        /// button, so a slow inbox check shouldn't time the flow out.
178        const MAGIC_LOGIN_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(300);
179
180        let login = self.magic_login.lock().await.take().ok_or_else(|| {
181            HoundError::invalid_state(
182                StateCode::NoMagicLoginPending,
183                "no magic-link login in progress",
184            )
185        })?;
186
187        let callback = tokio::select! {
188            r = login.wait(MAGIC_LOGIN_TIMEOUT) => {
189                r.map_err(|e| HoundError::auth(AuthCode::MagicLinkWait, e.to_string()))?
190            }
191            _ = self.magic_login_cancelled.notified() => {
192                return Err(HoundError::auth(
193                    AuthCode::MagicLinkCancelled,
194                    "magic-link login cancelled",
195                ));
196            }
197        };
198        self.adopt_magic_callback(callback).await
199    }

201    /// Verify a manually-typed sign-in code directly over REST (no loopback
202    /// involved — works even for a link opened on another device). Same
203    /// outcome shape as [`magic_login_wait`](Self::magic_login_wait).
204    pub async fn magic_login_code(self: &Arc<Self>, code: &str) -> Result<AuthFlowOutcome> {
205        match self.rest.magic_verify(code).await? {
206            crate::rest::MagicLogin::SignedIn(login) => Ok(AuthFlowOutcome::SignedIn(
207                self.adopt_and_connect(login).await?,
208            )),
209            crate::rest::MagicLogin::Unknown { link_token, email } => {
210                *self.pending_magic_link.lock().await = Some(link_token);
211                Ok(AuthFlowOutcome::EmailUnknown { email })
212            }
213        }
214    }

216    /// Shared tail of [`magic_login_wait`](Self::magic_login_wait): adopt a
217    /// token callback as a session, or park an unknown-email link token.
218    async fn adopt_magic_callback(
219        self: &Arc<Self>,
220        callback: hound_browser_auth::CallbackResult,
221    ) -> Result<AuthFlowOutcome> {
222        match callback {
223            hound_browser_auth::CallbackResult::Tokens(tokens) => {
224                let state = self
225                    .session
226                    .adopt_tokens(&tokens.access_token, &tokens.refresh_token)?;
227                Ok(AuthFlowOutcome::SignedIn(self.finish_sign_in(state).await?))
228            }
229            hound_browser_auth::CallbackResult::MagicUnknown { link_token, email } => {
230                *self.pending_magic_link.lock().await = Some(link_token);
231                Ok(AuthFlowOutcome::EmailUnknown { email })
232            }
233            other => Err(HoundError::internal(
234                InternalCode::UnexpectedCallback,
235                format!("unexpected magic-link callback: {other:?}"),
236            )),
237        }
238    }

240    /// "Continue as `email`": create a new account for the pending email
241    /// create-or-link decision, with `display_name` as the profile display
242    /// name.
243    pub async fn magic_register(self: &Arc<Self>, display_name: &str) -> Result<AuthState> {
244        let token = self.pending_magic_link.lock().await.take().ok_or_else(|| {
245            HoundError::invalid_state(
246                StateCode::NoEmailLinkPending,
247                "no email create-or-link decision pending",
248            )
249        })?;
250        let login = self.rest.magic_register(&token, display_name).await?;
251        self.adopt_and_connect(login).await
252    }

261    /// Sign out: tear down realtime, clear the session + tokens, publish signed-out,
262    /// and reset the observable snapshots (so a re-login starts from a clean view).
263    pub async fn logout(self: &Arc<Self>) -> Result<()> {
264        if let Some(rt) = self.realtime.lock().await.take() {
265            let _ = rt.disconnect().await;
266        }
267        self.session.sign_out().await?;
268        self.publish_auth(AuthState::SignedOut);
269        // A link verdict belongs to the session that produced it.
270        self.steam_autolink.lock().await.take();
271        // `send_replace`, not `send`: `watch::Sender::send` silently drops the value
272        // (returns `Err`, leaves the cell untouched) when there are zero live
273        // receivers — fine for a periodic/self-healing publish (like the 1 Hz stats
274        // emitter), wrong here, where this is the only write and a caller that
275        // subscribes lazily (any non-UI client — a headless bot, a fresh REST-driven
276        // consumer) must still see the reset state on its first `watch_*` call.
277        self.state.community_tree.send_replace(CommunityTree::new());
278        self.state.roster.send_replace(RosterView::new());
279        self.state.messages.lock().await.clear();
280        // The detached flags describe the lists that just went away.
281        self.state.detached.lock().await.clear();
282        // Nothing left to resync into, and the next sign-in's plane is a fresh
283        // session whose first `Welcome` is a connect, not a reconnect.
284        self.state.resync_pending.lock().await.clear();
285        Ok(())
286    }

288    /// Adopt a freshly established session and bring the realtime plane up.
289    pub(super) async fn adopt_and_connect(
290        self: &Arc<Self>,
291        login: crate::rest::LoginSession,
292    ) -> Result<AuthState> {
293        let state = self.session.adopt_login(login)?;
294        self.finish_sign_in(state).await
295    }

297    /// The tail every fresh sign-in shares once its tokens are adopted: publish
298    /// the new auth state, start the session, then honour a pending Steam link.
299    async fn finish_sign_in(self: &Arc<Self>, state: AuthState) -> Result<AuthState> {
300        self.publish_auth(state.clone());
301        self.start_session().await?;
302        // Never fails the sign-in: a conflict or an expired token is the
303        // link's verdict, held for the UI's `link_steam_pending`.
304        if let Some(verdict) = self.autolink_pending_steam().await {
305            *self.steam_autolink.lock().await = Some(verdict);
306        }
307        Ok(state)
308    }

310    /// Start everything a signed-in session runs on. The one path a restore and
311    /// every fresh sign-in converge on, so no leg can be missed by one of them.
312    ///
313    /// The own-profile fetch is spawned, not awaited: a restore is on the way to
314    /// the first frame, and the roster entry it produces is not worth a round
315    /// trip of that. It runs alongside the two legs below instead.
316    pub(super) async fn start_session(self: &Arc<Self>) -> Result<()> {
317        self.spawn_self_profile_fetch();
318        // Sub-phases, not merely awaited: a restore runs this on the path to the
319        // first frame and both legs are network round-trips. Dart's log bridge
320        // turns these into Sentry breadcrumbs, so a slow or wedged start names the
321        // leg instead of leaving the bootstrap watchdog to say only "restore".
322        tracing::info!("session start: connecting the realtime plane");
323        self.connect().await?;
324        tracing::info!("session start: loading the community tree");
325        self.load_communities().await
326    }
