4/// The `call:stats` event payload — one structured snapshot of everything the call
5/// can observe, emitted at 1 Hz. **This is the wire contract the RTC-debug panel
6/// consumes verbatim.**
7///
8/// Conventions for every field (so the UI can render generically and new fields
9/// appear automatically): keys are snake_case and self-describing, with units in
10/// the name — `_ms` (milliseconds), `_kbps` (kilobits/sec), `_pct` (percent),
11/// `_fps` (frames/sec). Cumulative counts have no suffix and are monotonic totals
12/// for the current session. Rates (`*_fps`, `*_kbps`) are windowed over the last
13/// ~1 s (delta / elapsed). Gauges (`rtt_ms`, `level`, geometry, `target_kbps`,
14/// `quality_rung`) are last-observed values.
15#[derive(Debug, Clone, Default, serde::Serialize)]
16pub struct StatsSnapshot {
17    pub transport: TransportSnapshot,
18    pub outbound_audio: OutboundAudioSnapshot,
19    pub outbound_video: OutboundVideoSnapshot,
20    /// Screen-share send section (mirrors `outbound_video`). Additive — present
21    /// only as a new field; `enabled=false` when not sharing.
22    pub outbound_screen: OutboundScreenSnapshot,
23    /// One entry per remote peer with a live inbound pipeline, sorted by peer id.
24    pub inbound: Vec<InboundSnapshot>,
25    /// Receiver-wide inbound video totals — the two figures the per-stream rows
26    /// above cannot answer between them. See [`InboundTotalsSnapshot`].
27    pub inbound_totals: InboundTotalsSnapshot,
28    /// Recent discrete audio-glitch events (newest first), over the last few seconds —
29    /// the timestamped, cause-attributed glitch log the audio debug report renders.
30    /// Populated by the emitter from the mix bus's glitch ring; empty when healthy.
31    pub glitches: Vec<GlitchSnapshot>,
32}

34/// What the whole RECEIVER is spending on inbound video, across every peer and both
35/// video sources — the two numbers the receive-side total-budget allocator
36/// (`hound_media::allocate`) is a controller for, and which no per-stream row can
37/// answer: a call is affordable or not in aggregate, and each individual stream can
38/// look perfectly healthy while their sum does not fit.
39///
40/// Both are folded from the per-stream rows rather than counted independently, so
41/// they cannot disagree with the section they summarise.
42#[derive(Debug, Clone, Default, serde::Serialize)]
43pub struct InboundTotalsSnapshot {
44    /// Sum of every inbound video stream's receive rate (camera + screen, all peers)
45    /// over the last ~1 s window — what the budget is a budget OF.
46    pub video_recv_kbps: f64,
47    /// Concurrent LIVE video decoder instances. The other half of the receiver's
48    /// cost: decode is CPU and one OS thread per stream, and a receiver can be
49    /// decode-bound long before it is bandwidth-bound.
50    ///
51    /// This is the **assertion gauge** for `call::policy::DecodeBudget`: the
52    /// cardinality it caps is exactly this fold — one subscribed video stream is
53    /// one pipeline, one decode thread and one decoder object — so `video_decoders`
54    /// must never exceed the live budget. It counts PIPELINES, not OS decode
55    /// sessions: a backend that opened two sessions for one stream, or none at all
56    /// because the stream has yet to deliver a keyframe, is invisible here, and the
57    /// budget is a policy over pipelines precisely because that is the number this
58    /// client controls.
59    pub video_decoders: usize,
60}

62/// One discrete audio-glitch event for the report's glitch log: how long ago it
63/// happened, which peer (if attributed), the source kind, the cause, and a magnitude.
64#[derive(Debug, Clone, Default, serde::Serialize)]
65pub struct GlitchSnapshot {
66    /// How long before the snapshot this glitch occurred (ms) — renders as "t-0.4s".
67    pub ago_ms: f64,
68    /// The peer it's attributed to, or `None` for a bus-global event (a late tick).
69    pub peer: Option<u32>,
70    /// Source kind: `"mic"` / `"screen_audio"` / `"tick"`.
71    pub label: String,
72    /// Cause: `"underflow"` (jitter buffer ran dry) or `"late_tick"` (RT feeder slip).
73    pub kind: String,
74    /// Magnitude (ms) where meaningful — e.g. how late a `late_tick` was; else 0.
75    pub detail_ms: f64,
76}

78/// Transport section: connection lifecycle + the relay/SFU identity this session
79/// landed on.
80#[derive(Debug, Clone, Default, serde::Serialize)]
81pub struct TransportSnapshot {
82    /// `"connected" | "reconnecting" | "failed"`.
83    pub state: &'static str,
84    /// How many times the supervisor re-established this call.
85    pub reconnect_attempts: u32,
86    /// Last measured round-trip time over the relay path (ms).
87    pub rtt_ms: f64,
88    /// The relay URL this session dialed.
89    pub relay_url: String,
90    /// Short prefix of the SFU node id (enough to disambiguate which relay).
91    pub sfu_node_id_short: String,
92    /// Our control-plane peer id for this session.
93    pub self_peer_id: u32,
94    /// Output-ring underrun SAMPLES this window — the device RT callback played
95    /// silence because the mix didn't keep its ring fed (a device-side / tick-pacing
96    /// gap affecting ALL audio). 0 in steady state; > 0 is audible playout stutter.
97    pub playout_output_underruns: u64,
98    /// Per-peer mix-slot STARVATION events this window — a peer's due audio wasn't in
99    /// its slot when the 10 ms mix tick drained it (the pump releases on network
100    /// arrival, the bus drains on a steady clock). 0 in steady state; > 0 localizes
101    /// stutter to the per-peer pump→mix cadence rather than the device.
102    pub playout_mix_starves: u64,
103    /// Output-ring overrun SAMPLES this window — the mix bus produced faster than the
104    /// device drained, so the sink discarded samples that were never played and output
105    /// latency is pinned at the ring depth. The opposite drift direction to
106    /// `playout_output_underruns`; 0 while the closed-loop cushion holds.
107    pub playout_output_overruns: u64,
108    /// Demoted output-stream faults this window (cpal `Xrun`/`RealtimeDenied` sent to
109    /// the settle funnel rather than killing the stream). 0 in steady state; a
110    /// sustained positive rate means the device is faulting continuously WITHOUT a
111    /// reopen firing (the storm keeps re-arming the 2 s settle deadline) — the
112    /// otherwise-invisible case this counter exists for.
113    pub playout_soft_faults: u64,
114    /// Mix-tick lateness p50 (ms) this window — the steady RT-feeder cadence error.
115    /// ~0 when healthy.
116    pub tick_late_p50_ms: f64,
117    /// Mix-tick lateness p95 (ms) this window — the RT-feeder tail. A high p95 means
118    /// the bus thread is being descheduled (CPU/encode contention) and dragging the
119    /// whole playout with it — the device-side cause of `playout_output_underruns`.
120    pub tick_late_p95_ms: f64,
121}

123/// Outbound microphone section.
124#[derive(Debug, Clone, Default, serde::Serialize)]
125pub struct OutboundAudioSnapshot {
126    pub frames_captured: u64,
127    pub frames_sent: u64,
128    /// Send bitrate over the last ~1 s window.
129    pub send_kbps: f64,
130    /// Current self speaking level (0..1).
131    pub level: f32,
132    /// Whether the APM (AEC/NS/AGC) is active.
133    pub apm_active: bool,
134    /// Whether the local mic is muted.
135    pub muted: bool,
136    /// Mic capture-ring overflow SAMPLES this window — the capture worker fell behind
137    /// the device callback and the ring dropped audio before it was ever encoded. The
138    /// far end hears a chopped word; nothing on the RECEIVE side can see it (assembled
139    /// frames stay timestamp-contiguous over the hole), so this is the only signal.
140    pub capture_overflow_samples: u64,
141    /// Encoded frames withheld by publisher-side silence suppression since the
142    /// establish (`call::policy::SilenceGate`). Read next to `frames_sent`: a send rate
143    /// that has collapsed from ~50/s to ~2–4/s with this climbing is a quiet room
144    /// working as designed, not a broken capture path. (2/s is the bare keepalive
145    /// cadence; a real mic's noise floor adds its periodic comfort-noise refresh on
146    /// top, measured at ~2.4/s combined.)
147    pub frames_suppressed: u64,
148    /// Whether suppression is withholding frames right now.
149    pub suppressed: bool,
150    /// Mic RMS BEFORE capture processing, dBFS (-100 = silence). With
151    /// `processed_dbfs` this splits "the mic is quiet" from "the processing made
152    /// it quiet" without a repro.
153    pub raw_dbfs: f32,
154    /// Mic RMS AFTER capture processing — what the encoder (and the far end) gets.
155    pub processed_dbfs: f32,
156    /// OS input-volume controller (P5): whether a platform provider resolved
157    /// for the open mic device. `false` = digital-only gain (Linux today, or a
158    /// device with no settable property) — expected, not a fault.
159    pub os_volume_available: bool,
160    /// Last OS-read applied input volume (0..1) — the OS mic slider's current
161    /// position as far as this call can tell.
162    pub os_volume_applied: f32,
163    /// The controller's live recommendation (0..1). Differs from
164    /// `os_volume_applied` when AGC is off/muted (the write is gated) or the
165    /// write is still inside its ~1 s rate limit.
166    pub os_volume_recommended: f32,
167    /// Frames with a full-scale sample after processing. AGC2's limiter should
168    /// hold this near 0; growth means the gain chain is clipping speech.
169    pub clipped_frames: u64,
170}

198    pub target_kbps: f64,

202    pub codec: String,
203    /// Worst receiver-reported downstream loss this last interval (percent).
204    pub remote_loss_pct: f64,

206    pub publish_drops: u64,
207    /// PLIs received for our cam track.
208    pub pli_received: u64,
209    /// Whether a receiver is attached, i.e. the source publish gate has OPENED.
210    /// `false` = dark by demand: the capture thread keeps capturing (self-preview)
211    /// but deliberately encodes nothing, so `encoded_fps == 0` against a healthy
212    /// `captured_fps` is the intended state, not a dead encoder.
213    pub publish_live: bool,
214    /// Per-window twins of the two counters above (see the note on
215    /// [`InboundVideoSnapshot`]): what happened in the LAST second, which is what a
216    /// per-tick diagnosis can act on.
217    pub publish_drops_window: u64,
218    pub pli_received_window: u64,
219}

297    /// Jitter-buffer drops this window because it was FULL (backed-up / clock skew).
298    pub overflow_drops: u64,
299    /// Mix-tick UNDERFLOWS this window (the per-peer audible gap).
300    pub underflow: u64,
301    /// Frames inferred LOST this window (sequence gap; each concealed with PLC).
302    pub lost: u64,
303    /// Frames that arrived out of ORDER this window.
304    pub reordered: u64,
305    /// Audio wire codec — always Opus on the inbound path.
306    pub codec: String,

309#[derive(Debug, Clone, Default, serde::Serialize)]
310pub struct InboundVideoSnapshot {
311    pub frames_received: u64,
312    /// ARRIVAL rate over the last ~1 s window. The counterpart to `decoded_fps`:
313    /// together they split a dead tile into "nothing is reaching us" (both 0) and
314    /// "frames arrive but don't decode" (arriving ≫ decoding).
315    pub received_fps: f64,
316    /// Decode rate over the last ~1 s window.
317    pub decoded_fps: f64,
318    /// Whether a decode pipeline is live for this stream right now. `false` means
319    /// the counters below are the stale totals of a track that is no longer
320    /// subscribed — NOT a stall (see
321    /// [`PeerStats::set_video_subscribed`](crate::stats::PeerStats::set_video_subscribed)).
322    pub subscribed: bool,
323    /// Receive bitrate over the last ~1 s window.
324    pub recv_kbps: f64,
325    pub width: u32,
326    pub height: u32,
327    /// Average decode time per frame over the last window.
328    pub decode_ms_avg: f64,
329    /// p95 of one `VideoDecoder::decode` call this window — the decode tail.
330    pub decode_p95_ms: f64,
331    /// How long a decoded frame waited in the playout (jitter) buffer before
332    /// release — the actual receive buffering latency. p50 is the steady cushion,
333    /// p95 the tail.
334    pub jitter_dwell_p50_ms: f64,
335    pub jitter_dwell_p95_ms: f64,
336    pub frames_lost: u64,
337    pub keyframes_requested: u64,
338    /// Decoded frames dropped by the receive-side playout buffer.
339    pub playout_dropped: u64,
340    /// Wire codec the decoder is producing (`"H264"` / `"H265"` / `"AV1"`); empty
341    /// until the first frame decodes.
342    pub codec: String,
343    // ── Per-window twins of the cumulative counters above ────────────────────
344    //
345    // The totals are session-cumulative (the panel renders them as counters), which
346    // makes them useless to a per-tick rules engine: one lost frame at second 3
347    // would keep "loss" latched for the rest of the call. These carry the same
348    // events as a THIS-WINDOW delta, exactly like the inbound-audio section already
349    // does, so [`crate::video_report`] can judge the last second on its own.
350    /// Frames inferred lost this window (`frame_seq` gaps).
351    pub frames_lost_window: u64,
352    /// Recovery keyframe requests (PLIs) this receiver SENT this window — the nag
353    /// that accompanies a stalled or undecodable stream.
354    pub keyframes_requested_window: u64,
355    /// Decoded frames the playout buffer dropped this window (superseded / stale).
356    pub playout_dropped_window: u64,
357    /// Decoder (re)builds this window. > 1 is churn: each rebuild costs the stream
358    /// its reference chain, so it decodes nothing until the next keyframe.
359    pub decoder_builds_window: u64,
360    /// The decoder could not be CONSTRUCTED for the wire codec the sender is using.
361    /// A latched gauge, not a window delta: the decode loop's retry window makes the
362    /// attempts rarer than this 1 Hz window, while the condition itself persists
363    /// until the sender changes codec. Frames arriving with none decoding is
364    /// otherwise indistinguishable from a broken reference chain, which is what
365    /// [`crate::video_report`] used to call it.
366    pub decoder_build_failed: bool,
367    /// The failure is a platform capability miss (this box has no decoder for that
368    /// codec at all), not a transient one — the sender must downgrade.
369    pub decoder_build_unsupported: bool,
370    /// Cumulative decoder-construction failures, and this window's delta.
371    pub decoder_build_failures: u64,
372    pub decoder_build_failures_window: u64,
373    /// The wire codec whose decoder would not build. Distinct from `codec`, which
374    /// names what the live decoder produces and is empty exactly when this matters.
375    pub decoder_failed_codec: String,
376    /// The decode path in use, as the engine labels it (`"VideoToolbox (hardware)"`,
377    /// `"openh264 (software)"`, `"dav1d (software)"`); empty until the first decoder
378    /// is built. Tracks a MID-STREAM arm switch, so it is the live answer rather
379    /// than what was selected at build.
380    pub decoder_backend: String,
381    /// Whether that path is hardware-backed. `false` with a non-empty
382    /// `decoder_backend` means this stream is burning CPU to decode — which shares
383    /// cores with the RT audio mix thread, and is why the engine's decode-session
384    /// budget tightens while it is true.
385    pub decoder_hardware: bool,
386    /// Cumulative hardware→software decode fallbacks for this stream, and this
387    /// window's delta. A non-zero rate across a fleet is a hardware-decode
388    /// regression (a platform update refusing decode sessions); a non-zero total on
389    /// one client is that client running out of hardware sessions.
390    pub decoder_sw_fallbacks: u64,
391    pub decoder_sw_fallbacks_window: u64,
392}
