1//! # RTC debug stats (`call:stats`)
2//!
3//! The shared counter/gauge registry + 1 Hz emitter behind the `call:stats`
4//! snapshot — the data source for the in-app RTC debug panel. One structured
5//! snapshot per second of everything the live call can observe, grouped by
6//! section (transport / outbound audio / outbound video / per-peer inbound).
7//!
8//! ## Design: cheap hot paths, rates computed off the hot path
9//!
10//! Every media loop (capture / encode / decode / playback / publish / subscribe)
11//! only ever does a single **relaxed atomic increment** to record a stat — no
12//! locks, no allocation, no clock read. The registry is a bag of `AtomicU64` /
13//! `AtomicU32` counters (monotonic totals) and gauges (last-written values),
14//! grouped into per-section structs and shared as one `Arc<CallStats>` across the
15//! pipelines exactly the way the call's `AdaptShared` seam is.
16//!
17//! **Windowed rates** (fps / kbps) are NOT measured on the hot path. The emitter
18//! samples the monotonic counters once per second and divides the delta since the
19//! previous sample by the elapsed wall time — so a frame-per-second or
20//! kilobit-per-second figure costs the hot path nothing but one `fetch_add`.
21//!
22//! ## Extending it (the whole point)
23//!
24//! Adding a stat is two edits: bump one atomic from the producing loop, and add
25//! one field to the matching `*Snapshot` struct (a counter total, or a rate
26//! derived in `StatsEmitter::sample`). The UI renders the snapshot generically,
27//! so a new self-describing field (snake_case, units in the name — `_ms`,
28//! `_kbps`, `_pct`, `_fps`) appears in the panel automatically.
29//!
30//! ## The payload contract (consumed verbatim by the UI)
31//!
32//! The serialized [`StatsSnapshot`] is the wire contract. See its doc comment and
33//! each section struct for the exact field list and units. This is intentionally
34//! separate from `call:levels` (10 Hz, hot speaking meter) — do not merge them.
35

58/// Emit rate for `call:stats`: 1 Hz. A debug panel does not need to update faster
59/// than once a second, and the per-second cadence makes the windowed rates

93/// Render one engine [`AudioGlitch`](hound_engine::audio::mix::AudioGlitch) into the
94/// serialized [`GlitchSnapshot`], aging it against the bus's current capture time.
95fn render_glitch(g: &hound_engine::audio::mix::AudioGlitch, now_us: i64) -> GlitchSnapshot {
96    use hound_engine::audio::mix::{GlitchKind, NO_PEER_TAG};
97    GlitchSnapshot {
98        ago_ms: ((now_us - g.at_us).max(0) as f64) / 1000.0,
99        peer: (g.peer_tag != NO_PEER_TAG).then_some(g.peer_tag),
100        label: g.label.to_string(),
101        kind: match g.kind {
102            GlitchKind::Underflow => "underflow",
103            GlitchKind::LateTick => "late_tick",
104            GlitchKind::DeviceReopen => GLITCH_KIND_DEVICE_REOPEN,
105        }
106        .to_string(),
107        detail_ms: (g.detail_us.max(0) as f64) / 1000.0,
108    }
109}

111/// Coarse transport state, stored as a `u8` gauge so the supervisor can write it
112/// from any task without a lock. Mirrors the `call:connection` lifecycle but is
113/// folded into the one-snapshot-per-second debug event as a string.
114#[derive(Debug, Clone, Copy, PartialEq, Eq)]
115pub enum TransportState {
116    /// A live session is up.
117    Connected,
118    /// A layer failed and the supervisor is retrying.
119    Reconnecting,
120    /// The supervisor gave up (fatal error).
121    Failed,
122}
123
124impl TransportState {
125    /// The wire string the UI renders. Matches the `call:connection` vocabulary.
126    fn as_str(self) -> &'static str {
127        match self {
128            TransportState::Connected => "connected",
129            TransportState::Reconnecting => "reconnecting",
130            TransportState::Failed => "failed",
131        }
132    }
133
134    /// Encode for the `AtomicU8` gauge.
135    fn to_u8(self) -> u8 {
136        match self {
137            TransportState::Connected => 0,
138            TransportState::Reconnecting => 1,
139            TransportState::Failed => 2,
140        }
141    }
142
143    /// Decode from the `AtomicU8` gauge (unknown values read as `Connected` — the
144    /// gauge only ever holds values this module writes).
145    fn from_u8(v: u8) -> Self {
146        match v {
147            1 => TransportState::Reconnecting,
148            2 => TransportState::Failed,
149            _ => TransportState::Connected,
150        }
151    }
152}

154/// Transport-section counters/gauges. `reconnect_attempts` is the only field that
155/// must survive a reconnect (it counts them), so the whole [`CallStats`] registry
156/// is created once per call and shared into every establish; [`CallStats::reset_session`]
157/// clears the per-establish fields but leaves this counter alone.
158#[derive(Default)]
159struct TransportStats {
160    /// Coarse connection state (see [`TransportState`]). Gauge.
161    state: AtomicU8,
162    /// How many times the supervisor has re-established the session this call.
163    /// Survives a reconnect (it counts them).
164    reconnect_attempts: AtomicU32,
165    /// Last RTT sample in microseconds (written by the adaptation tick each second
166    /// from `conn.rtt`). `0` means "not yet sampled / no path". Gauge.
167    rtt_us: AtomicU64,
168    /// Our control-plane peer id for the current session. Gauge (re-set each
169    /// establish — a reconnect mints a fresh peer id).
170    self_peer_id: AtomicU32,
171    /// The relay URL + short SFU node id, set once per establish. A `Mutex<String>`
172    /// rather than an atomic because they are strings; written once on establish,
173    /// read once per second by the emitter — no contention.
174    relay_url: Mutex<String>,
175    sfu_node_id_short: Mutex<String>,
176    /// CUMULATIVE playout-health counters mirrored from the mix bus each emit tick
177    /// ([`CallStats::set_playout_health`]); the snapshot diffs them to a per-window
178    /// rate. `playout_output_underruns` = output samples the device RT callback had
179    /// to silence-pad (device-side gap); `playout_mix_starves` = per-peer mix slots
180    /// that came up short at a 10 ms tick (per-peer gap). Both 0 in steady state;
181    /// rising = audio stutter, localized (device path vs per-peer pump cadence).
182    playout_output_underruns: AtomicU64,
183    playout_mix_starves: AtomicU64,
184    /// CUMULATIVE output-ring OVERRUN samples (the opposite drift direction: the mix
185    /// bus outran the device, so the sink discarded samples that were never played and
186    /// output latency sits pinned at the ring depth). The regulated cushion holds this
187    /// at 0; rising means the depth feedback lost control.
188    playout_output_overruns: AtomicU64,
189    /// CUMULATIVE demoted output-stream faults (cpal `Xrun`/`RealtimeDenied` routed to
190    /// the settle/re-sync funnel instead of an immediate reopen). A sustained storm
191    /// keeps re-arming the settle deadline and never reopens — correct, but invisible
192    /// without this counter.
193    playout_soft_faults: AtomicU64,
194    /// Mix-tick lateness percentiles (µs) for the last window, mirrored from the bus
195    /// each emit. The RT-health signal: ~0 when the bus thread keeps its 10 ms
196    /// cadence, high when it is descheduled (CPU/encode contention) — which drains the
197    /// device ring and stutters ALL audio at once. Gauges.
198    tick_late_p50_us: AtomicU32,
199    tick_late_p95_us: AtomicU32,
200}

202/// Outbound-audio counters/gauges. Totals are monotonic; the emitter derives
203/// `send_kbps` from the `sent_bytes` delta.
204#[derive(Default)]
205struct OutboundAudioStats {
206    /// PCM frames pulled from the mic device (post-APM) since the call started.
207    frames_captured: AtomicU64,
208    /// Opus frames successfully published to the relay.
209    frames_sent: AtomicU64,
210    /// Opus payload bytes published (drives `send_kbps`).
211    sent_bytes: AtomicU64,
212    /// Last self speaking level (0..1), written by the mic loop. Gauge.
213    level_milli: AtomicU32,
214    /// Whether the APM (AEC/NS/AGC) is live this session. Gauge.
215    apm_active: AtomicBool,
216    /// Whether the local mic is muted. Gauge (written on every mute edge).
217    muted: AtomicBool,
218    /// CUMULATIVE captured samples dropped by the capture ring (the mic worker was
219    /// descheduled past the ring's depth). Accumulated as deltas by the capture loop
220    /// so a device hot-swap — which mints a fresh device counter — keeps adding
221    /// instead of resetting. The only local signal for "the FAR end hears chopped
222    /// words": assembled frames stay timestamp-contiguous over the hole, so every
223    /// receive-side gauge reads healthy.
224    capture_overflow_samples: AtomicU64,
225    /// Encoded frames the publisher's `call::policy::SilenceGate` withheld because the
226    /// mic was silent. The counterpart to `frames_sent`: together they say whether a
227    /// low send rate is suppression working or capture failing.
228    frames_suppressed: AtomicU64,
229    /// Whether suppression is engaged RIGHT NOW. Gauge, so the audio report can
230    /// attribute "this peer is sending nothing" to a quiet room rather than a stall.
231    suppressed: AtomicBool,
232    /// Mic level BEFORE capture processing, deci-dBFS (dBFS × 10, negative), gauge.
233    /// The raw/processed pair splits "the mic is quiet" from "the processing made
234    /// it quiet" in a report — the two causes a far-end loudness complaint has.
235    raw_deci_dbfs: AtomicI32,
236    /// Mic level AFTER capture processing (what the encoder sees), deci-dBFS. Gauge.
237    processed_deci_dbfs: AtomicI32,
238    /// OS input-volume controller (P5): whether a platform provider is
239    /// currently resolved for the open mic device. `false` on Linux (no
240    /// backend yet) or a device with no settable property (aggregate/USB) —
241    /// a normal digital-only degrade, not a fault. Gauge.
242    os_volume_available: AtomicBool,
243    /// Last OS-READ applied input volume (0..1), milli-units via
244    /// `level_to_milli` (same encoding as `level_milli`). Gauge.
245    os_volume_applied_milli: AtomicU32,
246    /// The controller's live recommendation (0..1), same encoding. May differ
247    /// from `os_volume_applied_milli` when AGC is off/muted (the write is
248    /// gated) or the write is still rate-limited. Gauge.
249    os_volume_recommended_milli: AtomicU32,
250    /// Frames with any full-scale sample AFTER processing. AGC2's limiter should
251    /// hold this at ~0; growth is the smoking gun for a gain-chain bug.
252    clipped_frames: AtomicU64,
253}

356/// One remote peer's inbound audio counters/gauges (per [`InboundAudioKind`] stream).
357/// The jitter-buffer + loss fields are what the audio debug report uses to attribute a
358/// glitch to the network / jitter-buffer stage.
359#[derive(Default)]
360struct InboundAudioStats {
361    /// Opus frames received + decoded for this peer.
362    frames_received: AtomicU64,
363    /// Received Opus payload bytes (drives `recv_kbps`).
364    recv_bytes: AtomicU64,
365    /// Packet-loss-concealment frames synthesized (decoder ran with no packet, on a
366    /// detected sequence gap).
367    plc_frames: AtomicU64,
368    /// Frames currently buffered in this peer's playout jitter buffer. Gauge.
369    playout_buffered_frames: AtomicU32,
370    /// Smoothed inter-arrival jitter (µs, RFC 3550) driving the adaptive playout
371    /// depth. Gauge.
372    jitter_us: AtomicU32,
373    /// Last decoded speaking level (0..1). Gauge.
374    level_milli: AtomicU32,
375    /// Current jitter-buffer depth (µs). Gauge.
376    target_delay_us: AtomicU32,
377    /// Depth the estimator recommends (`floor + 4·J`, µs). Gauge. `recommended ≫
378    /// target` = the buffer is too shallow for the link's jitter → underflows.
379    recommended_delay_us: AtomicU32,
380    /// Cumulative jitter-buffer drops because the frame arrived too LATE (network
381    /// late-arrival). Counter.
382    late_drops: AtomicU64,
383    /// Cumulative jitter-buffer drops because it was full (backed-up / clock skew).
384    /// Counter.
385    overflow_drops: AtomicU64,
386    /// Cumulative mix-tick UNDERFLOWS for this stream (primed slot came up short — the
387    /// per-peer audible gap). Mirrored from the bus slot's starve count. Counter.
388    underflow: AtomicU64,
389    /// Cumulative frames inferred LOST from receive-side sequence-gap detection (the
390    /// decode loop fed the decoder PLC for each). Counter.
391    lost: AtomicU64,
392    /// Cumulative frames that arrived out of ORDER (timestamp before the last seen).
393    /// Counter.
394    reordered: AtomicU64,
395}
396
397impl InboundAudioStats {
398    /// One Opus frame of `bytes` received + decoded.
399    fn received(&self, bytes: usize) {
400        self.frames_received.fetch_add(1, Ordering::Relaxed);
401        self.recv_bytes.fetch_add(bytes as u64, Ordering::Relaxed);
402    }
403    fn set_playout_depth(&self, frames: u32) {
404        self.playout_buffered_frames
405            .store(frames, Ordering::Relaxed);
406    }
407    fn set_jitter(&self, jitter_us: u32) {
408        self.jitter_us.store(jitter_us, Ordering::Relaxed);
409    }
410    fn set_level(&self, level: f32) {
411        self.level_milli
412            .store(level_to_milli(level), Ordering::Relaxed);
413    }
414    /// Mirror the jitter-buffer health from a `BusProducer::push` result: the current
415    /// + recommended depth (gauges) and the cumulative drop/underflow totals.
416    fn set_jitterbuf(
417        &self,
418        target_us: u32,
419        recommended_us: u32,
420        late_drops: u64,
421        overflow_drops: u64,
422        underflow: u64,
423    ) {
424        self.target_delay_us.store(target_us, Ordering::Relaxed);
425        self.recommended_delay_us
426            .store(recommended_us, Ordering::Relaxed);
427        self.late_drops.store(late_drops, Ordering::Relaxed);
428        self.overflow_drops.store(overflow_drops, Ordering::Relaxed);
429        self.underflow.store(underflow, Ordering::Relaxed);
430    }
431    /// Record `lost` newly-detected lost frames (sequence gap) + whether this frame
432    /// arrived out of order. Counters.
433    fn record_loss(&self, lost: u64, reordered: bool) {
434        if lost > 0 {
435            self.lost.fetch_add(lost, Ordering::Relaxed);
436            // Each concealed gap drove one PLC decode.
437            self.plc_frames.fetch_add(lost, Ordering::Relaxed);
438        }
439        if reordered {
440            self.reordered.fetch_add(1, Ordering::Relaxed);
441        }
442    }
443}

470    frames_received: AtomicU64,
471    /// Frames successfully decoded to I420.
472    frames_decoded: AtomicU64,
473    /// Received video payload bytes (drives `recv_kbps`).

475    /// Last decoded geometry. Gauges.
476    width: AtomicU32,
477    height: AtomicU32,
478    /// Cumulative decode time + count, in microseconds, for a windowed average.

480    decode_count: AtomicU64,
481    /// Frames inferred lost from `frame_seq` gaps (the receiver-side accounting the
482    /// BWE feedback loop already computes).
483    frames_lost: AtomicU64,
484    /// Keyframe requests (PLIs) this receiver sent for the peer's track.
485    keyframes_requested: AtomicU64,
486    /// Decoded frames dropped by the receive-side playout buffer (superseded /
487    /// stale — never rendered).
488    playout_dropped: AtomicU64,
489    /// Whether a decode pipeline is LIVE for this stream (set at pipeline start,
490    /// cleared on every exit path — including an abort, which drops the future).
491    /// Gauge. Without it a torn-down stream (the peer stopped their camera) and a
492    /// WEDGED one are indistinguishable: both simply stop counting frames while the
493    /// totals sit stale, so [`crate::video_report`] gates its stall rule on this.
494    subscribed: AtomicBool,
495    /// Decoder builds + rebuilds for this stream. The decode thread rebuilds
496    /// whenever the sender's per-packet codec byte changes (and after a fatal
497    /// decode); repeated rebuilds inside one window are churn, and a rebuilding
498    /// decoder produces no frames.
499    decoder_builds: AtomicU64,
500    /// The decoder could not be CONSTRUCTED for the wire codec the sender is using.
501    /// A gauge, latched until a build succeeds, because the condition is persistent
502    /// while the retry window suppresses the attempts that would otherwise re-count
503    /// it: a per-window counter reads 0 in four windows out of five and would make
504    /// the fault flicker. Cumulative attempts live beside it in
505    /// [`Self::decoder_build_failures`].
506    ///
507    /// Nothing else in this struct can distinguish this state. Frames keep arriving
508    /// and none decode, which is byte-identical to a broken reference chain — so
509    /// without it [`crate::video_report`] attributes a codec the platform cannot
510    /// decode to a wedged downlink or the keyframe gate.
511    decoder_build_failed: AtomicBool,
512    /// The failure is the platform's typed capability miss
513    /// (`CaptureError::Unsupported`) rather than a transient one: no retry, no
514    /// keyframe and no reconnect can fix it — only the SENDER switching codec.
515    decoder_build_unsupported: AtomicBool,
516    /// Cumulative decoder-construction failures (attempts, not frames — the decode
517    /// loop's retry window bounds them to one per codec per `RETRY_AFTER`).
518    decoder_build_failures: AtomicU64,
519    /// The wire codec of the frames whose decoder would not build (`"H264"` /
520    /// `"H265"` / `"AV1"`). Distinct from [`Self::codec`], which names the codec the
521    /// LIVE decoder is producing and so is empty exactly when this matters.
522    failed_codec: Mutex<String>,
523    /// The decode path actually in use, as the engine labels it — e.g.
524    /// `"VideoToolbox (hardware)"`, `"openh264 (software)"`, `"dav1d (software)"`.
525    /// Gauge, re-read whenever the decoder is built AND on every decode failure,
526    /// because the engine's composite decoders can switch arms MID-STREAM (a
527    /// refused hardware decode session degrades that stream onto the software
528    /// floor). See [`PeerStats::set_video_decoder`].
529    decoder_backend: Mutex<String>,
530    /// Whether that path is hardware-backed. The gauge that makes a fleet-wide
531    /// hardware-decode regression legible: without it a client whose VideoToolbox /
532    /// MFT sessions are all being refused looks perfectly healthy — it IS decoding,
533    /// just on the CPU, and the only other symptom is audio starting to stutter for
534    /// reasons nothing attributes to decode.
535    decoder_hardware: AtomicBool,
536    /// Cumulative hardware→software decode fallbacks for this stream. Distinct from
537    /// [`Self::decoder_builds`]: a rebuild is routine (a sender codec switch), a
538    /// fallback means the platform refused a decode session. > 0 with
539    /// `decoder_hardware == false` is the signature.
540    decoder_sw_fallbacks: AtomicU64,
541    /// Per-stage RECEIVE-side video latency, as windowed percentile histograms (the
542    /// tail, not the mean — a p95 spike is what stutters). Drained + cleared by the
543    /// 1 Hz emitter. A `Mutex` is free here: recorded at most once per decoded /
544    /// released frame (≤60 Hz), never on an audio-sample hot path.
545    ///  * `decode_lat` — wall time of one `VideoDecoder::decode` call.
546    ///  * `dwell_lat` — how long a decoded frame waited in the playout (jitter)
547    ///    buffer before release: the actual receive buffering latency.
548    decode_lat: Mutex<LatencyHistogram>,
549    dwell_lat: Mutex<LatencyHistogram>,
550    /// Wire codec the decoder is currently producing for this track (`"H264"` /
551    /// `"H265"` / `"AV1"`), set by the decode loop — and re-set if the sender
552    /// switches codec mid-stream. Empty until the first frame decodes.
553    codec: Mutex<String>,
554}

1430        }
