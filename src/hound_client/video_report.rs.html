1//! # Agent-readable video-quality debug report
2//!
3//! The video twin of [`crate::audio_report`]: a pure function — [`video_report`] —
4//! that turns one [`StatsSnapshot`] into a compact, deterministic, paste-able text
5//! block that **pre-attributes** a degraded tile to a pipeline stage. A reader (human
6//! or agent) sees a *cause*, not a wall of counters: a `diagnosis:` headline names the
7//! most likely fault with the numbers that justify it, and a per-stage table flags
8//! every signal `ok` / `HIGH` / `BAD`.
9//!
10//! ## The pipeline stages (any can freeze a tile)
11//! ```text
12//! SEND     source ─▶ encoder ─▶ publish channel ─▶ uplink ─▶ relay
13//! RECEIVE  relay ─▶ downlink ─▶ keyframe gate ─▶ decoder ─▶ playout ─▶ render
14//! ```
15//! One client observes BOTH halves: its own send path (invisible to every receive-side
16//! gauge — a dead camera or a jammed uplink reads perfectly healthy locally, which is
17//! exactly why the send rules come first) and, per peer and per stream (`cam` vs
18//! `screen`), everything from the downlink rightward.
19//!
20//! ## Why the split matters
21//! Video fails in ways audio does not. A single lost frame breaks the reference chain
22//! until the next keyframe, so "arriving but not decoding" is a distinct state from
23//! "not arriving"; the encoder is a rung-driven consumer, so "encoded ≪ captured" is
24//! only a fault relative to `min(capture rate, rung fps)`; and a screen share
25//! deliberately captures NOTHING while its content is static. The rules below encode
26//! those asymmetries so the report never cries wolf at a healthy static share, a BWE
27//! pause, or a track the peer simply stopped publishing.
28//!
29//! ## Windowed, never cumulative
30//! Every rule reads per-window signals (`*_fps`, `*_window`) so the verdict describes
31//! the LAST second. The cumulative totals in the snapshot are for the panel; a rules
32//! engine fed cumulative counters latches on the first bad frame of the call and never
33//! recovers.
34//!
35//! Pure + deterministic ⇒ unit-tested per fault class, which is what keeps the
36//! agent-facing format honest.

42// ── Verdict thresholds (named so they are tunable + asserted in tests) ─────────────
43
44/// The rung label the adaptation tick writes when congestion has paused video
45/// entirely. Every capture/encode rule is inert on a paused track — the encoder is
46/// *supposed* to be silent — and the pause itself is the [`Diagnosis`].
47const AUDIO_ONLY_RUNG: &str = "audio_only";
48
49/// Frames/sec below which a stream counts as delivering NOTHING (send or receive).
50/// Deliberately under 1 fps: a static screen share legitimately re-emits its cached
51/// frame at the 1 s static-refresh cadence, and a report that called that "stalled"
52/// would be wrong on every idle share.
53const STALL_FPS: f64 = 0.5;
54
55/// Capture rate below this fraction of the rung's negotiated fps is a struggling
56/// source (thermal throttle, a contended camera, a compositor starving the capturer).
57/// Half the negotiated rate is the point at which motion visibly judders rather than
58/// merely looking soft.
59const CAPTURE_KEEPUP_RATIO: f64 = 0.5;
60
61/// Encoder output below this fraction of what it was FED (`min(capture, rung fps)`)
62/// means access units are being dropped rather than paced — the encoder is failing or
63/// rejecting frames, not merely rate-limiting.
64const ENCODE_KEEPUP_RATIO: f64 = 0.5;
65
66/// Minimum expected encode rate before the keep-up ratio is judged at all. Below a
67/// handful of frames per second the denominator is too small to distinguish a drop
68/// from ordinary pacing jitter, and a 1 fps static share would trip on rounding.
69const MEANINGFUL_FPS: f64 = 5.0;
70
71/// p95 of one HW-encode call (ms) above which the encoder is the bottleneck: over half
72/// a 30 fps frame budget (33 ms) spent inside a single encode leaves no headroom for
73/// capture + publish, and on a shared CPU it competes with the RT audio path.
74const ENCODE_P95_HIGH_MS: f64 = 20.0;
75
76/// p95 dwell (ms) on the bounded publish channel above which the uplink — not the
77/// encoder — is the constraint. Three 30 fps frame times: past that the channel is
78/// running at its depth and the next burst drops.
79const PUBLISH_QUEUE_HIGH_MS: f64 = 100.0;
80
81/// Worst receiver-reported downstream loss (percent) that counts as a lossy uplink.
82/// Video has no PLC: at this rate the reference chain breaks faster than keyframes can
83/// repair it, which is also roughly where the congestion controller starts backing off.
84const REMOTE_LOSS_HIGH_PCT: f64 = 5.0;
85
86/// Keyframe requests from remote receivers, per window, that count as a PLI storm. The
87/// producer coalesces duplicates through its own throttle, so two DISTINCT requests in
88/// one second means multiple receivers independently lost their reference.
89const PLI_STORM_WINDOW: u64 = 2;
90
91/// Receive-side loss (percent of the frames that should have arrived) above which the
92/// downlink is the fault. Lower than the audio threshold on purpose: a concealed Opus
93/// frame is a blip, a lost video frame corrupts every dependent frame after it.
94const RECV_LOSS_HIGH_PCT: f64 = 3.0;
95
96/// Recovery keyframe requests this receiver SENT in one window that count as a nag
97/// storm. The pipeline throttles its own requests to one per 2 s, so reaching two in a
98/// single window means the stream is breaking as fast as it is being repaired.
99const KEYFRAME_NAG_STORM: u64 = 2;
100
101/// Decoder (re)builds in one window that count as churn. One is a normal (re)start;
102/// two means the sender's codec byte is flapping or decode keeps failing, and each
103/// rebuild throws away the reference chain.
104const DECODER_CHURN_WINDOW: u64 = 2;
105
106/// p95 of one `VideoDecoder::decode` call (ms) above which decode is the bottleneck.
107/// Same budget argument as [`ENCODE_P95_HIGH_MS`] — and a software fallback (AV1 with
108/// no hardware block) blowing past it also starves the audio fill path.
109const DECODE_P95_HIGH_MS: f64 = 20.0;
110
111/// Decode rate below this fraction of the ARRIVAL rate: frames reach the decoder and
112/// do not come out the other side.
113const DECODE_KEEPUP_RATIO: f64 = 0.5;
114
115/// Decoded frames the playout buffer may discard per window before it counts as a
116/// fault. The buffer legitimately supersedes a stale frame around a pause/resume or a
117/// layer switch; three in one second is the render stage throwing away work the
118/// decoder already paid for.
119const PLAYOUT_DROP_WINDOW: u64 = 3;
120
121/// p95 playout dwell (ms) above which frames are queueing instead of being paced. The
122/// receive playout stage is a SHALLOW presentation pacer (~40 ms target), so a p95
123/// several times that is a backlog building toward a collapse-drop.
124const DWELL_HIGH_MS: f64 = 150.0;
125
126/// A per-row verdict in the stage table.
127#[derive(Clone, Copy, PartialEq, Eq)]
128enum Verdict {
129    Ok,
130    /// Notable but not yet the likely culprit (e.g. an elevated encode tail).
131    High,
132    /// A real fault contributing to the degraded video.
133    Bad,
134}
135
136impl Verdict {
137    fn tag(self) -> &'static str {
138        match self {
139            Verdict::Ok => "ok",
140            Verdict::High => "HIGH",
141            Verdict::Bad => "BAD",
142        }
143    }
144
145    /// `Bad` when the fault is present, else `Ok` — the shape most rows want.
146    fn bad_if(cond: bool) -> Self {
147        if cond { Verdict::Bad } else { Verdict::Ok }
148    }
149}

151// ── Send side: one normalized view over the camera + screen sections ───────────────
152
153/// The camera and screen send sections carry the same signals under two struct types
154/// (screen has no `pli_received`, camera no `audio_send_kbps`). Normalizing them into
155/// one view is what lets every send rule be written once and applied to both — and it
156/// is where the ONE real difference between them lives ([`SendView::static_dedup`]).
157struct SendView<'a> {
158    label: &'static str,
159    enabled: bool,
160    captured_fps: f64,
161    encoded_fps: f64,
162    send_kbps: f64,
163    encode_p95_ms: f64,
164    publish_queue_p95_ms: f64,
165    publish_drops: u64,
166    /// Keyframe requests received from remote receivers this window (camera only —
167    /// the screen section has no PLI counter, so it reads 0 and its rule is inert).
168    plis: u64,
169    remote_loss_pct: f64,
170    target_kbps: f64,
171    rung: &'a str,
172    codec: &'a str,
173    width: u32,
174    height: u32,
175    /// Whether a receiver is attached. While DARK the capture thread deliberately
176    /// encodes nothing — capture keeps running so self-preview never freezes — so an
177    /// encoded rate of 0 against a full-rate capture is the DESIGNED state. Every
178    /// capture/encode rule is gated on this exactly the way it is gated on the pause.
179    publish_live: bool,
180    /// Screen capture only counts REAL frames: a static share stops capturing entirely
181    /// while the encoder keeps re-emitting the cached frame at ~1 Hz. So a stalled
182    /// capture is only a fault here if the encode side went quiet too, and the
183    /// capture-vs-rung keep-up rule cannot apply at all.
184    static_dedup: bool,
185}

187impl<'a> SendView<'a> {
188    fn camera(s: &'a crate::stats::OutboundVideoSnapshot) -> Self {
189        Self {
190            label: "camera",
191            enabled: s.enabled,
192            captured_fps: s.captured_fps,
193            encoded_fps: s.encoded_fps,
194            send_kbps: s.send_kbps,
195            encode_p95_ms: s.encode_p95_ms,
196            publish_queue_p95_ms: s.publish_queue_p95_ms,
197            publish_drops: s.publish_drops_window,
198            plis: s.pli_received_window,
199            remote_loss_pct: s.remote_loss_pct,
200            target_kbps: s.target_kbps,
201            rung: &s.quality_rung,
202            codec: &s.codec,
203            width: s.width,
204            height: s.height,
205            publish_live: s.publish_live,
206            static_dedup: false,
207        }
208    }
209
210    fn screen(s: &'a crate::stats::OutboundScreenSnapshot) -> Self {
211        Self {
212            label: "screen",
213            enabled: s.enabled,
214            captured_fps: s.captured_fps,
215            encoded_fps: s.encoded_fps,
216            send_kbps: s.send_kbps,
217            encode_p95_ms: s.encode_p95_ms,
218            publish_queue_p95_ms: s.publish_queue_p95_ms,
219            publish_drops: s.publish_drops_window,
220            plis: 0,
221            remote_loss_pct: s.remote_loss_pct,
222            target_kbps: s.target_kbps,
223            rung: &s.quality_rung,
224            codec: &s.codec,
225            width: s.width,
226            height: s.height,
227            publish_live: s.publish_live,
228            static_dedup: true,
229        }
230    }
231
232    /// Congestion has paused this track's video. Every capture/encode rule is inert
233    /// while paused — silence is the intended behaviour, not a fault.
234    fn paused(&self) -> bool {
235        self.rung == AUDIO_ONLY_RUNG
236    }
237
238    /// Publishing with nobody subscribed. The publish loop's live gate is still shut,
239    /// so the capture thread skips encode entirely (nobody encodes into the void)
240    /// while capture keeps running for the self-preview tile. A HEALTHY idle state —
241    /// the report says so on an info line rather than diagnosing it.
242    fn dark(&self) -> bool {
243        self.enabled && !self.publish_live
244    }
245
246    /// Whether this track's capture/encode stages can be judged at all. The two idle
247    /// states are indistinguishable from a dead pipeline on the counters alone —
248    /// full-rate capture, zero encode — so both have to gate the rules, or a lone
249    /// participant with nobody watching is diagnosed with a dead encoder every second
250    /// (and the sustained-degradation latch WARNs about it once per dark period).
251    fn encoding_expected(&self) -> bool {
252        self.enabled && self.publish_live && !self.paused()
253    }
254
255    /// The rung's negotiated frame rate, parsed from the adaptation tick's label
256    /// (`"{w}x{h}@{fps}"`). `None` before the first tick (empty label) or while
257    /// paused — the keep-up rules then simply don't apply.
258    fn rung_fps(&self) -> Option<f64> {
259        self.rung
260            .rsplit_once('@')
261            .and_then(|(_, fps)| fps.parse::<f64>().ok())
262            .filter(|f| *f > 0.0)
263    }
264
265    /// What the encoder SHOULD be emitting: it can't encode frames capture never
266    /// produced, and it deliberately won't exceed the rung's fps cap.
267    fn expected_encode_fps(&self) -> f64 {
268        match self.rung_fps() {
269            Some(cap) => self.captured_fps.min(cap),
270            None => self.captured_fps,
271        }
272    }
273
274    /// The source has stopped delivering. For a screen share the encode side must be
275    /// quiet too, otherwise this is just static content (see [`Self::static_dedup`]).
276    fn capture_stalled(&self) -> bool {
277        self.encoding_expected()
278            && self.captured_fps < STALL_FPS
279            && (!self.static_dedup || self.encoded_fps < STALL_FPS)
280    }
281
282    /// The source is delivering, but well under the rate that was negotiated.
283    fn capture_slow(&self) -> bool {
284        self.encoding_expected()
285            && !self.static_dedup
286            && !self.capture_stalled()
287            && self
288                .rung_fps()
289                .is_some_and(|f| self.captured_fps < CAPTURE_KEEPUP_RATIO * f)
290    }
291
292    /// Frames go in, access units don't come out.
293    fn encode_stalled(&self) -> bool {
294        let expected = self.expected_encode_fps();
295        self.encoding_expected()
296            && expected >= MEANINGFUL_FPS
297            && self.encoded_fps < ENCODE_KEEPUP_RATIO * expected
298    }
299
300    /// One encode call is eating the frame budget.
301    fn encode_slow(&self) -> bool {
302        self.encoding_expected() && self.encode_p95_ms >= ENCODE_P95_HIGH_MS
303    }
304
305    /// Encoded AUs can't get onto the wire fast enough.
306    fn publish_backpressure(&self) -> bool {
307        self.enabled
308            && (self.publish_drops > 0 || self.publish_queue_p95_ms >= PUBLISH_QUEUE_HIGH_MS)
309    }
310
311    /// Receivers are losing what we send (loss reports, or a storm of keyframe begging).
312    fn uplink_loss(&self) -> bool {
313        self.enabled
314            && (self.remote_loss_pct >= REMOTE_LOSS_HIGH_PCT || self.plis >= PLI_STORM_WINDOW)
315    }
316}

326// ── Receive side ───────────────────────────────────────────────────────────────────

328/// A peer's two inbound video streams (camera + screen share), labelled.
329fn streams(p: &InboundSnapshot) -> [(&'static str, &InboundVideoSnapshot); 2] {
330    [("cam", &p.video), ("screen", &p.screen_video)]
331}
332
333/// Whether this stream can be judged at all: a decode pipeline is live AND it has
334/// delivered at least one frame. Both halves matter. Without `subscribed`, a peer who
335/// simply turned their camera off leaves stale totals that read exactly like a wedge;
336/// without `frames_received`, a subscription still waiting on its first bytes would be
337/// diagnosed before it has ever had a chance to deliver.
338fn live(v: &InboundVideoSnapshot) -> bool {
339    v.subscribed && v.frames_received > 0
340}
341
342/// Whether the stream is worth a table block (live, or with totals worth showing).
343fn video_active(v: &InboundVideoSnapshot) -> bool {
344    v.subscribed || v.frames_received > 0
345}
346
347/// Receive loss for one stream as a percent of what should have arrived this window.
348/// `received_fps` is frames-per-second over a ~1 s emit window, so it doubles as this
349/// window's arrival count (see the `call:stats` 1 Hz contract).
350fn recv_loss_pct(v: &InboundVideoSnapshot) -> f64 {
351    let denom = v.received_fps + v.frames_lost_window as f64;
352    if denom <= 0.0 {
353        0.0
354    } else {
355        v.frames_lost_window as f64 / denom * 100.0
356    }
357}
358
359/// Nothing is arriving on a subscription that is still live — the classic wedge (a
360/// zombie broadcast, a stalled group, a producer that stopped publishing).
361fn recv_stalled(v: &InboundVideoSnapshot) -> bool {
362    live(v) && v.received_fps < STALL_FPS
363}
364
365/// Frames arrive but none decode, and the receiver is asking for a keyframe: the
366/// keyframe gate is dropping deltas because there is no decodable reference.
367fn waiting_keyframe(v: &InboundVideoSnapshot) -> bool {
368    live(v)
369        && v.received_fps >= STALL_FPS
370        && v.decoded_fps < STALL_FPS
371        && v.keyframes_requested_window > 0
372}
373
374/// The decoder is being rebuilt repeatedly (codec flap / repeated decode failure).
375fn decoder_churn(v: &InboundVideoSnapshot) -> bool {
376    live(v) && v.decoder_builds_window >= DECODER_CHURN_WINDOW
377}
378
379/// The downlink is lossy, or the stream keeps breaking faster than it can be repaired.
380fn recv_loss(v: &InboundVideoSnapshot) -> bool {
381    live(v)
382        && (recv_loss_pct(v) >= RECV_LOSS_HIGH_PCT
383            || v.keyframes_requested_window >= KEYFRAME_NAG_STORM)
384}
385
386/// One decode call is eating the frame budget (software fallback / CPU contention).
387fn decode_slow(v: &InboundVideoSnapshot) -> bool {
388    live(v) && v.decode_p95_ms >= DECODE_P95_HIGH_MS
389}
390
391/// Frames arrive faster than they decode, with no keyframe gate to explain it.
392fn decode_behind(v: &InboundVideoSnapshot) -> bool {
393    live(v)
394        && v.received_fps >= MEANINGFUL_FPS
395        && v.decoded_fps < DECODE_KEEPUP_RATIO * v.received_fps
396}
397
398/// Decoded frames are being thrown away at the playout/render stage.
399fn render_drops(v: &InboundVideoSnapshot) -> bool {
400    live(v)
401        && (v.playout_dropped_window >= PLAYOUT_DROP_WINDOW
402            || v.jitter_dwell_p95_ms >= DWELL_HIGH_MS)
403}

405// ── Diagnosis: signals → one most-likely cause ─────────────────────────────────────

407/// One diagnosis: the pipeline stage at fault, and the headline naming it.
408///
409/// Two forms because they have different jobs (mirrors
410/// [`crate::audio_report::Diagnosis`]). [`headline`](Self::headline) is for a
411/// human/agent reading the report — it embeds the live counters that justify the
412/// verdict, which makes every occurrence a unique string. [`stage`](Self::stage) is
413/// the same verdict as a stable token, so it can be a Sentry tag or a metric label
414/// without exploding cardinality.
415#[derive(Debug, Clone, PartialEq, Eq)]
416pub struct Diagnosis {
417    /// Stable, low-cardinality stage token — one of the values listed on
418    /// [`diagnose_video_detail`]. Safe to group by.
419    pub stage: &'static str,
420    /// The full sentence, counters and all.
421    pub headline: String,
422}
423
424impl Diagnosis {
425    fn new(stage: &'static str, headline: String) -> Self {
426        Self { stage, headline }
427    }
428}

430/// The single most-likely cause of degraded video. Rules are ordered by severity, and
431/// within that by causality — a rule that MANUFACTURES another's signature is checked
432/// first, so the report names the cause instead of its consequence (a wedged downlink
433/// is not "decode falling behind"; a paused rung is not "the encoder died").
434///
435/// Tier 1, dead streams: `send-capture-stalled`, `send-encode-stalled`, `recv-stalled`,
436/// `recv-waiting-keyframe`, `decode-churn`.
437/// Tier 2, uplink congestion: `uplink-publish-backpressure`, `uplink-bwe-floor`,
438/// `uplink-loss`.
439/// Tier 3, soft degradation: `send-capture-slow`, `send-encode-slow`, `recv-loss`,
440/// `decode-slow`, `decode-behind`, `render-drops`.
441/// Otherwise `healthy`.
442pub fn diagnose_video_detail(snap: &StatsSnapshot) -> Diagnosis {
443    let send = send_views(snap);
444
445    // ── Tier 1: something is dead ────────────────────────────────────────────────
446
447    // 1) The source stopped. Nothing downstream of it can be judged — an idle encoder
448    //    and a silent uplink are consequences — so it is checked first.
449    for s in &send {
450        if s.capture_stalled() {
451            return Diagnosis::new(
452                "send-capture-stalled",
453                format!(
454                    "{} capture stalled ({:.1} fps from the source, rung {}) — the device/source \
455                     stopped delivering; encode ({:.1} fps) and uplink ({:.0}kbps) are idle \
456                     because of it, and no receive-side gauge can see this",
457                    s.label,
458                    s.captured_fps,
459                    rung_or_unknown(s.rung),
460                    s.encoded_fps,
461                    s.send_kbps
462                ),
463            );
464        }
465    }
466
467    // 2) Capture is feeding the encoder and access units are not coming out. Above the
468    //    publish rules: an encoder producing nothing cannot back a publish channel up.
469    for s in &send {
470        if s.encode_stalled() {
471            return Diagnosis::new(
472                "send-encode-stalled",
473                format!(
474                    "{} encoder dropping frames ({:.1} encoded of {:.1} expected fps, capture \
475                     {:.1} fps at rung {}) — encode p95 {:.1}ms, {}x{} geometry; the far end sees \
476                     a frozen tile",
477                    s.label,
478                    s.encoded_fps,
479                    s.expected_encode_fps(),
480                    s.captured_fps,
481                    rung_or_unknown(s.rung),
482                    s.encode_p95_ms,
483                    s.width,
484                    s.height
485                ),
486            );
487        }
488    }
489
490    // 3) A live subscription delivering nothing: the wedge. Outranks every decode rule
491    //    — with no arrivals there is nothing to decode.
492    for p in &snap.inbound {
493        for (label, v) in streams(p) {
494            if recv_stalled(v) {
495                return Diagnosis::new(
496                    "recv-stalled",
497                    format!(
498                        "downlink stalled (peer {} {label}): {:.1} fps / {:.0}kbps arriving on a \
499                         live subscription ({} frames received all session) — a wedged group, a \
500                         zombie broadcast, or a producer that went silent; {} keyframe request(s) \
501                         sent this window",
502                        p.peer_id,
503                        v.received_fps,
504                        v.recv_kbps,
505                        v.frames_received,
506                        v.keyframes_requested_window
507                    ),
508                );
509            }
510        }
511    }
512
513    // 4) Arriving but undecodable. Above `decode-behind`, whose exact signature
514    //    (received ≫ decoded) this manufactures — the frames are dropped BEFORE the
515    //    decoder, at the keyframe gate, so the decoder is not the fault.
516    for p in &snap.inbound {
517        for (label, v) in streams(p) {
518            if waiting_keyframe(v) {
519                return Diagnosis::new(
520                    "recv-waiting-keyframe",
521                    format!(
522                        "waiting for a keyframe (peer {} {label}): {:.1} fps arriving, {:.1} fps \
523                         decoding — deltas are dropped at the keyframe gate with no decodable \
524                         reference; {} request(s) sent this window, {} frame(s) lost",
525                        p.peer_id,
526                        v.received_fps,
527                        v.decoded_fps,
528                        v.keyframes_requested_window,
529                        v.frames_lost_window
530                    ),
531                );
532            }
533        }
534    }
535
536    // 5) Decoder thrash. Same argument as (4): a decoder being rebuilt produces nothing,
537    //    so its throughput collapse is not a decode-speed problem.
538    for p in &snap.inbound {
539        for (label, v) in streams(p) {
540            if decoder_churn(v) {
541                return Diagnosis::new(
542                    "decode-churn",
543                    format!(
544                        "decoder rebuild churn (peer {} {label}): {} rebuilds this window (codec \
545                         {}) — each one throws away the reference chain, so only {:.1} of {:.1} \
546                         fps decode",
547                        p.peer_id,
548                        v.decoder_builds_window,
549                        codec_or_unknown(&v.codec),
550                        v.decoded_fps,
551                        v.received_fps
552                    ),
553                );
554            }
555        }
556    }
557
558    // ── Tier 2: the uplink is the constraint ─────────────────────────────────────
559
560    // 6) The acute symptom: encoded AUs waiting on (or dropped at) the publish channel.
561    //    Above the BWE floor, which is the controller's measured RESPONSE to this.
562    for s in &send {
563        if s.publish_backpressure() {
564            return Diagnosis::new(
565                "uplink-publish-backpressure",
566                format!(
567                    "{} uplink backpressure: {} AU(s) dropped at the publish channel this window, \
568                     queue p95 {:.0}ms (sending {:.0}kbps against a {:.0}kbps target) — the wire \
569                     can't take what the encoder produces",
570                    s.label, s.publish_drops, s.publish_queue_p95_ms, s.send_kbps, s.target_kbps
571                ),
572            );
573        }
574    }
575
576    // 7) The controller gave up on video for this track. Reported as its own stage
577    //    rather than left to be inferred from a silent encoder.
578    for s in &send {
579        if s.enabled && s.paused() {
580            return Diagnosis::new(
581                "uplink-bwe-floor",
582                format!(
583                    "{} video paused by congestion control (rung {AUDIO_ONLY_RUNG}, target \
584                     {:.0}kbps, remote loss {:.1}%) — the uplink can't carry video, so the \
585                     encoder is idle by design",
586                    s.label, s.target_kbps, s.remote_loss_pct
587                ),
588            );
589        }
590    }
591
592    // 8) Receivers are losing what we send.
593    for s in &send {
594        if s.uplink_loss() {
595            return Diagnosis::new(
596                "uplink-loss",
597                format!(
598                    "{} uplink loss: receivers report {:.1}% downstream loss and sent {} keyframe \
599                     request(s) this window (sending {:.0}kbps at rung {}) — our video is breaking \
600                     on the way out",
601                    s.label,
602                    s.remote_loss_pct,
603                    s.plis,
604                    s.send_kbps,
605                    rung_or_unknown(s.rung)
606                ),
607            );
608        }
609    }
610
611    // ── Tier 3: soft degradation ─────────────────────────────────────────────────
612
613    // 9) The source is slow rather than dead.
614    for s in &send {
615        if s.capture_slow() {
616            return Diagnosis::new(
617                "send-capture-slow",
618                format!(
619                    "{} capture below its negotiated rate ({:.1} fps vs rung {}) — the source is \
620                     struggling (thermal / contended device), so the far end sees juddery motion \
621                     no matter how healthy the uplink is",
622                    s.label,
623                    s.captured_fps,
624                    rung_or_unknown(s.rung)
625                ),
626            );
627        }
628    }
629
630    // 10) Encode tail eating the frame budget (still producing, just expensively).
631    for s in &send {
632        if s.encode_slow() {
633            return Diagnosis::new(
634                "send-encode-slow",
635                format!(
636                    "{} encode tail high (p95 {:.1}ms at {}x{}, {:.1} fps out) — one encode call \
637                     eats most of the frame budget; CPU/thermal pressure, or a software encoder \
638                     where hardware was expected",
639                    s.label, s.encode_p95_ms, s.width, s.height, s.encoded_fps
640                ),
641            );
642        }
643    }
644
645    // 11) Downlink loss / repair churn on a stream that is still decoding.
646    for p in &snap.inbound {
647        for (label, v) in streams(p) {
648            if recv_loss(v) {
649                return Diagnosis::new(
650                    "recv-loss",
651                    format!(
652                        "downlink loss (peer {} {label}): {:.1}% of arrivals lost ({} frame(s)) \
653                         with {} keyframe request(s) this window — every gap corrupts frames until \
654                         the next keyframe lands",
655                        p.peer_id,
656                        recv_loss_pct(v),
657                        v.frames_lost_window,
658                        v.keyframes_requested_window
659                    ),
660                );
661            }
662        }
663    }
664
665    // 12) Decode is expensive (usually a software fallback).
666    for p in &snap.inbound {
667        for (label, v) in streams(p) {
668            if decode_slow(v) {
669                return Diagnosis::new(
670                    "decode-slow",
671                    format!(
672                        "decode tail high (peer {} {label}): p95 {:.1}ms / avg {:.1}ms for {} at \
673                         {}x{} — a software decode path or CPU contention; it also competes with \
674                         the audio fill pump",
675                        p.peer_id,
676                        v.decode_p95_ms,
677                        v.decode_ms_avg,
678                        codec_or_unknown(&v.codec),
679                        v.width,
680                        v.height
681                    ),
682                );
683            }
684        }
685    }
686
687    // 13) Decode falling behind arrivals with none of the above to explain it.
688    for p in &snap.inbound {
689        for (label, v) in streams(p) {
690            if decode_behind(v) {
691                return Diagnosis::new(
692                    "decode-behind",
693                    format!(
694                        "decode behind the wire (peer {} {label}): {:.1} fps decoded of {:.1} fps \
695                         arriving (p95 {:.1}ms) — frames reach the decoder and don't come out",
696                        p.peer_id, v.decoded_fps, v.received_fps, v.decode_p95_ms
697                    ),
698                );
699            }
700        }
701    }
702
703    // 14) The last stage: decoded work discarded before it was ever shown.
704    for p in &snap.inbound {
705        for (label, v) in streams(p) {
706            if render_drops(v) {
707                return Diagnosis::new(
708                    "render-drops",
709                    format!(
710                        "playout dropping decoded frames (peer {} {label}): {} dropped this \
711                         window, dwell p50/p95 {:.0}/{:.0}ms — frames queue past their playout \
712                         time and are superseded before they render",
713                        p.peer_id,
714                        v.playout_dropped_window,
715                        v.jitter_dwell_p50_ms,
716                        v.jitter_dwell_p95_ms
717                    ),
718                );
719            }
720        }
721    }
722
723    Diagnosis::new(
724        "healthy",
725        "healthy — no video faults in this window".to_string(),
726    )
727}

746/// Render the agent-readable video-quality report for one snapshot. Pure +
747/// deterministic. See the module docs for the format and rationale.
748pub fn video_report(snap: &StatsSnapshot) -> String {
749    let t = &snap.transport;
750    let mut out = String::with_capacity(1024);
751
752    let _ = writeln!(
753        out,
754        "VIDEO HEALTH — self peer {} · {} · ~1s window",
755        t.self_peer_id, t.state
756    );
757    let _ = writeln!(out, "diagnosis: {}", diagnose_video(snap));
758    let _ = writeln!(out, "transport  rtt={:.0}ms  relay={}", t.rtt_ms, {
759        if t.sfu_node_id_short.is_empty() {
760            "-"
761        } else {
762            &t.sfu_node_id_short
763        }
764    });
765
766    // Send side: our own capture → encode → publish → uplink.
767    for s in &send_views(snap) {
768        let _ = writeln!(out);
769        if !s.enabled {
770            let _ = writeln!(out, "send {} · (not publishing)", s.label);
771            continue;
772        }
773        write_send_stage(&mut out, s);
774    }
775
776    // Receive side: per peer, per stream.
777    let mut any_peer = false;
778    for p in &snap.inbound {
779        for (label, v) in streams(p) {
780            if !video_active(v) {
781                continue;
782            }
783            any_peer = true;
784            let _ = writeln!(out);
785            let _ = writeln!(
786                out,
787                "peer {} · {label}{}",
788                p.peer_id,
789                if v.subscribed { "" } else { " (unsubscribed)" }
790            );
791            write_recv_stage(&mut out, v);
792        }
793    }
794    if !any_peer {
795        let _ = writeln!(out);
796        let _ = writeln!(out, "(no inbound video streams this window)");
797    }
798
799    out
800}

802/// Write the per-track SEND stage rows (capture / encode / publish / uplink).
803fn write_send_stage(out: &mut String, s: &SendView<'_>) {
804    let _ = writeln!(
805        out,
806        "send {} · rung {} · {:.0}kbps target · {} · {}x{}{}{}",
807        s.label,
808        rung_or_unknown(s.rung),
809        s.target_kbps,
810        codec_or_unknown(s.codec),
811        s.width,
812        s.height,
813        if s.paused() { " · PAUSED" } else { "" },
814        if s.dark() { " · DARK" } else { "" }
815    );
816    // Dark is a healthy state, not a verdict — but "0 fps encoded" beside a full-rate
817    // capture is alarming enough that it has to be explained where it is read, or the
818    // next person to open this report re-derives the same false alarm the rules avoid.
819    if s.dark() {
820        let _ = writeln!(
821            out,
822            "  info      dark by demand — nobody is subscribed, so encode is gated off; \
823             capture keeps running for self-preview (healthy)"
824        );
825    }
826    // The screen note is CONDITIONAL on the verdict beside it. "Static content is not
827    // re-captured" is a REASSURANCE — it exists to stop a reader treating a healthy
828    // idle share's 0 fps as a fault — and printing it next to a capture the headline
829    // has just called stalled says the opposite of the diagnosis. A stalled share gets
830    // the fact instead: the source is delivering nothing, and (unlike a static share)
831    // the encoder is not covering for it.
832    let capture_note = match (s.static_dedup, s.capture_stalled()) {
833        (true, true) => "(no frames from the source)",
834        (true, false) => "(screen: static content is not re-captured)",
835        (false, _) => "",
836    };
837    let _ = writeln!(
838        out,
839        "  capture   fps            {:<14} [{}]   {}",
840        format!("{:.1}", s.captured_fps),
841        Verdict::bad_if(s.capture_stalled() || s.capture_slow()).tag(),
842        capture_note
843    );
844    let encode_v = if s.encode_stalled() {
845        Verdict::Bad
846    } else if s.encode_slow() {
847        Verdict::High
848    } else {
849        Verdict::Ok
850    };
851    // What the encoder is measured against — `min(capture, rung fps)`, except on a
852    // static screen share, where capture legitimately reads 0 while the re-emit keeps
853    // publishing (printing "expected 0.0 fps" beside a working encoder reads as a bug).
854    //
855    // The re-emit claim needs BOTH halves of its own evidence: a quiet capture says
856    // only that nothing changed on screen, and the encode rate is the only thing that
857    // says the cached frame is actually still going out. With both stages silent there
858    // is no re-emit to describe — the share is stalled, the headline says so, and this
859    // column stays empty rather than reassuring the reader about work that stopped.
860    let expectation = if s.static_dedup && s.captured_fps < STALL_FPS {
861        if s.encoded_fps >= STALL_FPS {
862            "static share: re-emitting the cached frame".to_owned()
863        } else {
864            String::new()
865        }
866    } else {
867        format!("expected {:.1} fps", s.expected_encode_fps())
868    };
869    let _ = writeln!(
870        out,
871        "  encode    fps / p95      {:<14} [{}]   {expectation}",
872        format!("{:.1} / {:.1}ms", s.encoded_fps, s.encode_p95_ms),
873        encode_v.tag(),
874    );
875    let _ = writeln!(
876        out,
877        "  publish   queue p95      {:<14} [{}]   drops/win {}",
878        format!("{:.0}ms", s.publish_queue_p95_ms),
879        Verdict::bad_if(s.publish_backpressure()).tag(),
880        s.publish_drops
881    );
882    let _ = writeln!(
883        out,
884        "  uplink    remote loss    {:<14} [{}]   PLIs/win {}  sending {:.0}kbps",
885        format!("{:.1}%", s.remote_loss_pct),
886        Verdict::bad_if(s.uplink_loss()).tag(),
887        s.plis,
888        s.send_kbps
889    );
890}

892/// Write the per-stream RECEIVE stage rows (network / decode / render).
893fn write_recv_stage(out: &mut String, v: &InboundVideoSnapshot) {
894    let _ = writeln!(
895        out,
896        "  network   recv fps       {:<14} [{}]   {:.0}kbps{}",
897        format!("{:.1}", v.received_fps),
898        Verdict::bad_if(recv_stalled(v)).tag(),
899        v.recv_kbps,
900        if recv_stalled(v) { "  ← stalled" } else { "" }
901    );
902    let _ = writeln!(
903        out,
904        "            lost/win       {:<14} [{}]   keyframe reqs/win {}",
905        format!("{} ({:.1}%)", v.frames_lost_window, recv_loss_pct(v)),
906        Verdict::bad_if(recv_loss(v)).tag(),
907        v.keyframes_requested_window
908    );
909    let decode_v = if waiting_keyframe(v) || decoder_churn(v) || decode_behind(v) {
910        Verdict::Bad
911    } else if decode_slow(v) {
912        Verdict::High
913    } else {
914        Verdict::Ok
915    };
916    let _ = writeln!(
917        out,
918        "  decode    fps / p95      {:<14} [{}]   {} {}x{}{}",
919        format!("{:.1} / {:.1}ms", v.decoded_fps, v.decode_p95_ms),
920        decode_v.tag(),
921        codec_or_unknown(&v.codec),
922        v.width,
923        v.height,
924        if waiting_keyframe(v) {
925            "  ← awaiting keyframe"
926        } else {
927            ""
928        }
929    );
930    let _ = writeln!(
931        out,
932        "            rebuilds/win   {:<14} [{}]",
933        v.decoder_builds_window,
934        Verdict::bad_if(decoder_churn(v)).tag()
935    );
936    let _ = writeln!(
937        out,
938        "  render    dwell p50/p95  {:<14} [{}]   playout drops/win {}",
939        format!(
940            "{:.0}/{:.0}ms",
941            v.jitter_dwell_p50_ms, v.jitter_dwell_p95_ms
942        ),
943        Verdict::bad_if(render_drops(v)).tag(),
944        v.playout_dropped_window
945    );
946}
