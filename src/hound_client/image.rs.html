1//! Still-image decoding for formats the platform does not know.
2//!
3//! Flutter renders a remote image through Skia's own codecs (JPEG/PNG/GIF/WebP);
4//! **JPEG XL** is not among them, so a `.jxl` avatar or attachment is decoded
5//! here and handed to Dart as raw pixels. The output shape is deliberately the
6//! same `{width, height, rgba}` triple the share picker's
7//! [`ScreenThumbnail`](crate::call::ScreenThumbnail) already uses, so the Dart
8//! side reuses the existing `decodeImageFromPixels` path rather than growing a
9//! second pixel convention.
10//!
11//! Scope is narrow on purpose: JPEG XL only, one still frame, 8-bit sRGB.
12//! Everything the platform already decodes stays on the platform path — the Dart
13//! provider sniffs the bytes and only calls in here for JXL.
14//!
15//! **Untrusted input.** These bytes come off the network from another user's
16//! upload, so every failure is a typed [`HoundError::Decode`] and never a panic,
17//! and the canvas size is checked *before* the pixel buffer is allocated (a few
18//! hundred bytes of JXL can declare a gigapixel canvas).
19
20use jxl_oxide::{EnumColourEncoding, ExtraChannelType, JxlImage, Render, RenderingIntent};
21
22use crate::error::{DecodeCode, HoundError, Result};
23
24/// A decoded still image: **straight** (unassociated) 8-bit sRGB RGBA,
25/// `width * height * 4` bytes, row-major, no row padding.
26///
27/// Straight, not premultiplied, is the honest general-purpose contract — but note
28/// that Flutter's raw-pixel entry points (`decodeImageFromPixels`,
29/// `ImageDescriptor.raw`) interpret `rgba8888` as *premultiplied*, so the Dart
30/// provider multiplies the colour through before handing the buffer over.
31#[derive(Debug, Clone)]
32pub struct DecodedImage {
33    pub width: u32,
34    pub height: u32,
35    pub rgba: Vec<u8>,
36}
37
38/// A bare JPEG XL codestream starts with this two-byte signature.
39const JXL_CODESTREAM_SIGNATURE: [u8; 2] = [0xFF, 0x0A];
40
41/// A containerised (ISOBMFF) JPEG XL file starts with the 12-byte JXL signature box.
42const JXL_CONTAINER_SIGNATURE: [u8; 12] = [
43    0x00, 0x00, 0x00, 0x0C, 0x4A, 0x58, 0x4C, 0x20, 0x0D, 0x0A, 0x87, 0x0A,
44];
45
46/// The most pixels [`decode_image`] will allocate a buffer for (64 MP — well past
47/// any avatar or chat attachment, well short of the 2^30-per-side the format
48/// allows). Refusing here is what keeps a tiny malicious file from turning into a
49/// multi-gigabyte allocation.
50const MAX_PIXELS: usize = 64 << 20;
51
52/// Whether `bytes` begins with either JPEG XL signature — the bare codestream
53/// (`FF 0A`) or the ISOBMFF container's signature box.
54///
55/// A sniff, not a validation: a file may carry the signature and still be corrupt,
56/// which [`decode_image`] reports. Mirrored by the Dart provider, which uses it to
57/// decide whether an image needs this decoder at all.
58pub fn is_jpeg_xl(bytes: &[u8]) -> bool {
59    bytes.starts_with(&JXL_CODESTREAM_SIGNATURE) || bytes.starts_with(&JXL_CONTAINER_SIGNATURE)
60}
61
62/// Decode a JPEG XL image to straight-alpha 8-bit sRGB RGBA.
63///
64/// For an **animated** JXL only the first keyframe is rendered: this feeds avatars
65/// and inline previews, which are stills.
66///
67/// Errors (never panics) on: bytes that are not JPEG XL (`DECODE004`), a canvas
68/// past the decode budget (`DECODE006`), and anything the decoder rejects —
69/// truncation, corruption, an unsupported feature (`DECODE005`).
70pub fn decode_image(bytes: &[u8]) -> Result<DecodedImage> {
71    if !is_jpeg_xl(bytes) {
72        let head = &bytes[..bytes.len().min(4)];
73        return Err(HoundError::decode(
74            DecodeCode::NotJpegXl,
75            format!(
76                "not a JPEG XL image: {} byte(s) starting {head:02x?}",
77                bytes.len()
78            ),
79        ));
80    }
81
82    let mut image = JxlImage::builder()
83        .read(bytes)
84        .map_err(|e| HoundError::decode(DecodeCode::JpegXl, format!("jpeg xl header: {e}")))?;
85
86    // Ask for sRGB output rather than the image's own encoding, so the caller gets
87    // one predictable colour space. Anything carrying an ICC profile needs a CMS to
88    // get there; `Moxcms` is the pure-Rust one (see the Cargo.toml note).
89    image.set_cms(jxl_oxide::Moxcms);
90    image.request_color_encoding(EnumColourEncoding::srgb(RenderingIntent::Relative));
91
92    // Refuse an oversized canvas from the HEADER, before a single plane is rendered.
93    budgeted_pixels(image.width(), image.height())?;
94
95    // Keyframe 0: the still, or the first frame of an animation.
96    let render = image
97        .render_frame(0)
98        .map_err(|e| HoundError::decode(DecodeCode::JpegXl, format!("jpeg xl render: {e}")))?;
99
100    // `stream()` interleaves the colour channels plus alpha (last), with the image's
101    // EXIF orientation already applied — so its geometry, not the header's, is what
102    // the output buffer must be sized against (a 90° orientation swaps the two).
103    let mut stream = render.stream();
104    let (width, height, channels) = (stream.width(), stream.height(), stream.channels());
105    let pixels = budgeted_pixels(width, height)?;
106
107    let mut interleaved = vec![0u8; pixels * channels as usize];
108    stream.write_to_buffer(&mut interleaved);
109
110    let rgba = to_rgba8(&interleaved, pixels, channels, alpha_is_associated(&render))?;
111    Ok(DecodedImage {
112        width,
113        height,
114        rgba,
115    })
116}
117
118/// `width * height`, refused if it overflows or exceeds [`MAX_PIXELS`]. Called on
119/// the header geometry (before rendering) and again on the rendered frame's.
120fn budgeted_pixels(width: u32, height: u32) -> Result<usize> {
121    (width as usize)
122        .checked_mul(height as usize)
123        .filter(|n| *n <= MAX_PIXELS)
124        .ok_or_else(|| {
125            HoundError::decode(
126                DecodeCode::ImageTooLarge,
127                format!("jpeg xl {width}x{height} exceeds the {MAX_PIXELS}-pixel decode budget"),
128            )
129        })
130}
131
132/// Whether the frame's alpha channel is *associated* (i.e. the colour channels are
133/// stored already multiplied by it). jxl-oxide streams channels as stored, and
134/// [`DecodedImage`] promises straight alpha, so this decides whether the colour has
135/// to be divided back out.
136fn alpha_is_associated(render: &Render) -> bool {
137    let (extra, _) = render.extra_channels();
138    extra.iter().find(|ec| ec.is_alpha()).is_some_and(|ec| {
139        matches!(
140            ec.ty(),
141            ExtraChannelType::Alpha {
142                alpha_associated: true
143            }
144        )
145    })
146}

148/// Widen an interleaved 1/2/3/4-channel frame to RGBA8888, straightening
149/// premultiplied colour on the way when `associated_alpha` says it is stored that
150/// way.
151///
152/// The sRGB request in [`decode_image`] is what bounds the channel count: greyscale
153/// and CMYK are already converted by the time the stream is built, so what arrives
154/// is grey, grey+A, RGB, or RGBA.
155fn to_rgba8(src: &[u8], pixels: usize, channels: u32, associated_alpha: bool) -> Result<Vec<u8>> {
156    let mut out = Vec::with_capacity(pixels * 4);
157    match channels {
158        1 => src
159            .iter()
160            .for_each(|&v| out.extend_from_slice(&[v, v, v, u8::MAX])),
161        2 => src.chunks_exact(2).for_each(|p| {
162            out.extend_from_slice(&unassociate([p[0], p[0], p[0], p[1]], associated_alpha))
163        }),
164        3 => src
165            .chunks_exact(3)
166            .for_each(|p| out.extend_from_slice(&[p[0], p[1], p[2], u8::MAX])),
167        4 => src.chunks_exact(4).for_each(|p| {
168            out.extend_from_slice(&unassociate([p[0], p[1], p[2], p[3]], associated_alpha))
169        }),
170        n => {
171            return Err(HoundError::decode(
172                DecodeCode::JpegXl,
173                format!("jpeg xl produced {n} channel(s) after the sRGB conversion"),
174            ));
175        }
176    }
177    Ok(out)
178}

180/// Divide premultiplied colour back out of one pixel (a no-op unless the source
181/// said its alpha is associated). Fully transparent stays black — there is no
182/// colour to recover, and any value would do.
183fn unassociate(px: [u8; 4], associated_alpha: bool) -> [u8; 4] {
184    let [r, g, b, a] = px;
185    if !associated_alpha || a == u8::MAX {
186        return px;
187    }
188    if a == 0 {
189        return [0, 0, 0, 0];
190    }
191    let straighten = |c: u8| ((c as u32 * 255 + a as u32 / 2) / a as u32).min(255) as u8;
192    [straighten(r), straighten(g), straighten(b), a]
193}

199    /// 4×4 lossless JXL, bare codestream: pixel (x, y) is [`PALETTE`]`[(x+y) % 4]`.
200    /// Encoded with libjxl 0.11.1 — `cjxl 4x4.png 4x4-rgb-codestream.jxl -d 0 -e 7`
201    /// — from an RGBA PNG of that pattern. (macOS `sips` cannot regenerate these:
202    /// it lists `public.jpeg-xl` as readable but *not* writable.)
203    ///
204    /// Both fixtures are plain git blobs, and must stay that way: `.gitattributes`
205    /// routes image extensions to git-lfs, and an lfs POINTER here would fail
206    /// `include_bytes!` on any clone that skipped `git lfs pull`. Do not add `*.jxl`
207    /// to that list.
208    const CODESTREAM_4X4: &[u8] = include_bytes!("../tests/fixtures/4x4-rgb-codestream.jxl");
209    /// The same image in the ISOBMFF container (`cjxl … --container=1`), so the
210    /// second signature is covered by a real file rather than a hand-built header.
211    const CONTAINER_4X4: &[u8] = include_bytes!("../tests/fixtures/4x4-rgb-container.jxl");
212
213    /// The four colours the fixture's rows cycle through, as straight RGBA.
214    const PALETTE: [[u8; 4]; 4] = [
215        [255, 0, 0, 255],
216        [0, 255, 0, 255],
217        [0, 0, 255, 255],
218        [255, 255, 0, 255],
219    ];
220
221    fn pixel(img: &DecodedImage, x: u32, y: u32) -> [u8; 4] {
222        let i = ((y * img.width + x) * 4) as usize;
223        img.rgba[i..i + 4].try_into().expect("4 bytes per pixel")
224    }

226    #[test]
227    fn both_signatures_sniff_as_jpeg_xl() {
228        assert!(is_jpeg_xl(CODESTREAM_4X4));
229        assert!(is_jpeg_xl(CONTAINER_4X4));
230        // A PNG, a JPEG, and a prefix too short to hold either signature.
231        assert!(!is_jpeg_xl(b"\x89PNG\r\n\x1a\n"));
232        assert!(!is_jpeg_xl(b"\xff\xd8\xff\xe0"));
233        assert!(!is_jpeg_xl(&[0xFF]));
234        assert!(!is_jpeg_xl(&[]));
235    }

237    /// The whole point: a real JXL decodes to the geometry and the pixels it was
238    /// encoded from. Lossless, so exact equality is the right assertion.
239    #[test]
240    fn a_lossless_codestream_decodes_to_its_original_pixels() {
241        let img = decode_image(CODESTREAM_4X4).expect("the fixture decodes");
242        assert_eq!((img.width, img.height), (4, 4));
243        assert_eq!(img.rgba.len(), 4 * 4 * 4);
244        for y in 0..4 {
245            for x in 0..4 {
246                assert_eq!(
247                    pixel(&img, x, y),
248                    PALETTE[((x + y) % 4) as usize],
249                    "pixel ({x},{y})"
250                );
251            }
252        }
253    }

255    /// The container form is the same image — so the container signature is not
256    /// merely sniffed, it is decoded through.
257    #[test]
258    fn the_container_form_decodes_to_the_same_image() {
259        let bare = decode_image(CODESTREAM_4X4).expect("the codestream fixture decodes");
260        let boxed = decode_image(CONTAINER_4X4).expect("the container fixture decodes");
261        assert_eq!((bare.width, bare.height), (boxed.width, boxed.height));
262        assert_eq!(bare.rgba, boxed.rgba);
263    }

307    #[test]
308    fn straightening_only_touches_premultiplied_pixels() {
309        // Not associated: passed through untouched, whatever the alpha says.
310        assert_eq!(unassociate([128, 64, 32, 128], false), [128, 64, 32, 128]);
311        // Associated: colour divided back out (half-transparent grey → full grey).
312        assert_eq!(unassociate([64, 64, 64, 128], true), [128, 128, 128, 128]);
313        // Fully transparent has no colour to recover.
314        assert_eq!(unassociate([9, 9, 9, 0], true), [0, 0, 0, 0]);
315        // Opaque is its own straight value under either flag.
316        assert_eq!(unassociate([1, 2, 3, 255], true), [1, 2, 3, 255]);
317    }

319    /// Greyscale and RGB widen to opaque RGBA; the 2/4-channel forms keep alpha.
320    #[test]
321    fn channel_widening_covers_every_shape_the_stream_can_produce() {
322        assert_eq!(
323            to_rgba8(&[7, 9], 2, 1, false).unwrap(),
324            [7, 7, 7, 255, 9, 9, 9, 255]
325        );
326        assert_eq!(to_rgba8(&[7, 128], 1, 2, false).unwrap(), [7, 7, 7, 128]);
327        assert_eq!(to_rgba8(&[1, 2, 3], 1, 3, false).unwrap(), [1, 2, 3, 255]);
328        assert_eq!(to_rgba8(&[1, 2, 3, 4], 1, 4, false).unwrap(), [1, 2, 3, 4]);
329        assert!(to_rgba8(&[0; 5], 1, 5, false).is_err());
330    }
