13// ─── Screen sources (share picker) ──────────────────────────────────────────────

15/// A shareable screen source for the picker — a display, a window, or a camera
16/// streamed as a screen. UI-friendly + FFI-flat: the typed engine selection token
17/// is encoded into [`id`](Self::id) (see [`encode_screen_source`]) so the UI round-
18/// trips an opaque string back into [`CallSession::screen_share_start`].
19#[derive(Clone, Debug, PartialEq, Eq)]
20pub struct ScreenSourceView {
21    /// Opaque round-trip token (`"{kind}:{native_id}"`); pass back to share-start.
22    pub id: String,
23    /// Human-readable label (window title / display description).
24    pub title: String,
25    /// Owning application for a window; `None` for a display or camera.
26    pub app_name: Option<String>,
27    /// A whole-display source (vs a single window / camera) — for picker grouping.
28    pub is_display: bool,
29}
30
31/// Enumerate the shareable screen sources (displays + windows). UI-free + needs no
32/// active call, so it lives as a free function the facade/FRB call directly. Blocking
33/// (round-trips ScreenCaptureKit's async enumeration); a Screen-Recording TCC denial
34/// or an unsupported platform surfaces as a typed [`HoundError::Media`].
35pub fn list_screen_sources() -> Result<Vec<ScreenSourceView>> {
36    let sources = Engine::new().screen_sources().map_err(|e| {
37        HoundError::media(
38            MediaCode::ScreenSourceEnumeration,
39            format!("enumerate screen sources: {e}"),
40        )
41    })?;
42    Ok(sources
43        .into_iter()
44        .map(|s| ScreenSourceView {
45            id: encode_screen_source(&s.source),
46            title: s.title,
47            app_name: s.app_name,
48            is_display: matches!(s.source, hound_engine::types::ScreenSource::Display(_)),
49        })
50        .collect())
51}

53// ─── Screen-share failure surface ───────────────────────────────────────────────
54
55/// A screen-share failure the user must be TOLD about, split into the two things a
56/// surface needs: the stable machine [`code`](Self::code) (what a support log is
57/// grepped for and what Sentry groups on) and the human [`message`](Self::message).
58/// FFI-flat (two `String`s) so the FRB layer mirrors it with no reshape.
59///
60/// Only TERMINAL failures land here — the ones after which no share is running and
61/// nothing will retry: the give-up on a source that opens but never delivers
62/// ([`MediaCode::ScreenCaptureNoFrames`]), a restart whose re-open failed, and a
63/// post-reconnect republish that failed. A transient fault the supervisor is about to
64/// restart is not a failure the user can act on, and surfacing one would make the
65/// banner flicker on every window resize.
66#[derive(Clone, Debug, PartialEq, Eq)]
67pub struct ScreenShareError {
68    /// The stable error code (`MEDIA024`, …) — see `hound_client::error`.
69    pub code: String,
70    /// Human-readable detail, WITHOUT the code prefix (`HoundError::message`).
71    pub message: String,
72}
73
74impl ScreenShareError {
75    /// Split a typed [`HoundError`] into the surfaced pair.
76    pub(super) fn of(err: &HoundError) -> Self {
77        Self {
78            code: err.code().into_owned(),
79            message: err.message().to_owned(),
80        }
81    }
82}

84/// The process-wide screen-share error surface: `Some` after a terminal failure,
85/// `None` once a share starts cleanly or the user stops one.
86///
87/// A process-global (the [`call_runtime`] / `adapt::coordinator` pattern) rather than
88/// a per-call channel, for the two reasons the failure itself has: a terminal error
89/// must OUTLIVE the share that produced it (the pipeline is gone by the time anyone
90/// renders the banner) and be readable with no call in progress at all — a subscriber
91/// that had to hold a live `CallSession` could not see the error that ended one. One
92/// call runs at a time, so a single cell is the whole state.
93pub(crate) fn screen_error_channel() -> &'static watch::Sender<Option<ScreenShareError>> {
94    static CH: std::sync::OnceLock<watch::Sender<Option<ScreenShareError>>> =
95        std::sync::OnceLock::new();
96    CH.get_or_init(|| watch::channel(None).0)
97}
98
99/// Subscribe to the screen-share error surface (the FRB `watch_screen_error` stream).
100/// Needs no active call; the current value is delivered on subscribe.
101pub fn watch_screen_error() -> watch::Receiver<Option<ScreenShareError>> {
102    screen_error_channel().subscribe()
103}

105/// A share-picker preview image: opaque RGBA8888, `width * height * 4` bytes.
106#[derive(Debug, Clone)]
107pub struct ScreenThumbnail {
108    pub width: u32,
109    pub height: u32,
110    pub rgba: Vec<u8>,
111}
112
113/// One-shot preview thumbnail of a screen source for the picker, aspect-fit within
114/// (`max_w`, `max_h`). Best-effort and per-platform (ScreenCaptureKit on macOS, a WGC
115/// one-shot on Windows; unsupported on Linux, where capturing would raise the portal
116/// dialog): a capture failure — unsupported platform, a permission/portal denial, an
117/// undeliverable source — surfaces as a typed [`HoundError::Media`] the picker treats
118/// as "no thumbnail" (title-only card).
119/// Blocking (drives a one-shot engine capture); the facade runs it off-runtime.
120pub fn screen_thumbnail(source_id: String, max_w: u32, max_h: u32) -> Result<ScreenThumbnail> {
121    let source = parse_screen_source(&source_id)?;
122    let t = Engine::new()
123        .screen_thumbnail(&source, max_w, max_h)
124        .map_err(|e| {
125            HoundError::media(MediaCode::ScreenThumbnail, format!("screen thumbnail: {e}"))
126        })?;
127    Ok(ScreenThumbnail {
128        width: t.width,
129        height: t.height,
130        rgba: t.rgba,
131    })
132}

134/// Encode the engine's typed selection token into the opaque `"{kind}:{id}"` string
135/// the UI round-trips. Inverse of [`parse_screen_source`].
136pub(super) fn encode_screen_source(source: &hound_engine::types::ScreenSource) -> String {
137    use hound_engine::types::ScreenSource;
138    match source {
139        ScreenSource::Display(id) => format!("display:{}", id.as_str()),
140        ScreenSource::Window(id) => format!("window:{}", id.as_str()),
141        ScreenSource::Camera(id) => format!("camera:{}", id.as_str()),
142    }
143}
144
145/// Parse the opaque `"{kind}:{id}"` picker token back into the engine's typed
146/// [`ScreenSource`](hound_engine::types::ScreenSource). Inverse of
147/// [`encode_screen_source`].
148pub(super) fn parse_screen_source(token: &str) -> Result<hound_engine::types::ScreenSource> {
149    use hound_engine::types::{CameraId, DisplayId, ScreenSource, WindowId};
150    let (kind, id) = token.split_once(':').ok_or_else(|| {
151        HoundError::media(
152            MediaCode::MalformedScreenToken,
153            format!("malformed screen source token {token:?}"),
154        )
155    })?;
156    match kind {
157        "display" => Ok(ScreenSource::Display(DisplayId::new(id))),
158        "window" => Ok(ScreenSource::Window(WindowId::new(id))),
159        "camera" => Ok(ScreenSource::Camera(CameraId::new(id))),
160        other => Err(HoundError::media(
161            MediaCode::UnknownScreenSourceKind,
162            format!("unknown screen source kind {other:?} (want display|window|camera)"),
163        )),
164    }
165}

167// ─── Media devices (mic / camera / speaker picker) ──────────────────────────────
168
169/// One enumerated media device for a picker row. FFI-flat: `id` is the opaque
170/// selection token to pass back to `call_select_device` for this kind (for audio it
171/// equals the cpal device name; for a camera the AVFoundation `uniqueID`), and
172/// `is_default` flags the host's current default for that kind.
173#[derive(Clone, Debug, PartialEq, Eq)]
174pub struct DeviceView {
175    pub id: String,
176    pub name: String,
177    pub is_default: bool,
178}
179
180/// The available media devices grouped by kind — the data the Devices panel renders.
181/// A kind whose platform backend is absent simply comes back empty (a capability
182/// miss, not an error).
183#[derive(Clone, Debug, Default, PartialEq, Eq)]
184pub struct DeviceListView {
185    pub microphones: Vec<DeviceView>,
186    pub cameras: Vec<DeviceView>,
187    pub speakers: Vec<DeviceView>,
188}
189
190/// Enumerate the available microphones, cameras, and speakers for the picker. UI-free
191/// and needs no active call. The engine enumeration is synchronous and may briefly
192/// touch the host, so the facade runs it on a blocking thread. Best-effort.
193pub fn list_devices() -> DeviceListView {
194    let eng = Engine::new();
195    DeviceListView {
196        microphones: eng.microphones().into_iter().map(device_view).collect(),
197        cameras: eng.cameras().into_iter().map(device_view).collect(),
198        speakers: eng.speakers().into_iter().map(device_view).collect(),
199    }
200}
201
202/// Map a typed engine [`DeviceDescriptor`](hound_engine::api::DeviceDescriptor) onto
203/// the FFI-flat [`DeviceView`] (id as a plain string). The typed ids keep device
204/// kinds apart inside the engine; the wire stays stringly so one
205/// `call_select_device(kind, id)` round-trips all three kinds.
206fn device_view<Id: hound_engine::types::DeviceId>(
207    d: hound_engine::api::DeviceDescriptor<Id>,
208) -> DeviceView {
209    DeviceView {
210        id: d.id.as_str().to_owned(),
211        name: d.name,
212        is_default: d.is_default,
213    }
