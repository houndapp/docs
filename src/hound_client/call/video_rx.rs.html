1//! Inbound video: the async pull half (receive-side keyframe gate; the nag task
2//! lives with the subscribe loop) feeds a blocking decode thread that owns the
3//! `!Send` `VideoDecoder` and rebuilds it whenever the sender's per-packet codec
4//! byte changes — the sender's choice drives decode. Decoded frames pass through a
5//! shallow playout stage ([`VIDEO_PLAYOUT_DELAY_US`]) that replays the sender's
6//! frame spacing, then reach the sink surface-first (zero-copy GPU handle) with
7//! materialized I420 as the universal fallback; [`push_preview`] is the
8//! borrow-based capture-side twin the publish loops use for self-preview.
9//!
10//! The playout stage is presentation pacing only — it does NOT anchor video against
11//! the audio plane's own (adaptive, deeper) jitter depth, so the two planes are
12//! still independently scheduled.
13

16/// Push one CAPTURED frame to the render sink for a self-preview tile, preferring
17/// the ZERO-COPY surface path: a surface-backed frame (SCK / AVFoundation NV12
18/// IOSurface) is handed to the renderer to sample directly — no NV12→I420 CPU
19/// conversion, no ~1.4–3 MB per-frame alloc+copy, no re-upload (~90 MB/s of
20/// conversion at a 1080p 30 fps share). Falls back to materialized I420
21/// for non-surface frames (scaled camera frames, other
22/// platforms) or a sink without a surface renderer. Borrows the frame — the
23/// caller still owns it for encode. (The REMOTE decode path keeps its own
24/// ownership-optimized version of this pattern: it consumes the frame, so its
25/// no-surface case can MOVE the bytes instead of copying.)
26pub(super) fn push_preview(sink: &dyn VideoSink, key: &str, frame: &hound_engine::types::RawFrame) {
27    let zero_copy =
28        RenderSurfaceHandle::for_frame(frame).is_some_and(|handle| sink.push_surface(key, handle));
29    if !zero_copy && let Some(i420) = frame.as_i420() {
30        sink.push_frame(key, i420.width(), i420.height(), i420.into_owned().data);
31    }
32}

34/// How long the video pull tolerates delivering NOTHING before treating the
35/// subscription as stalled. Above the static-share re-emit cadence
36/// ([`SCREEN_STATIC_REFRESH`], 1 s) so a healthy static share never trips it; a
37/// mid-GOP loss stall or a silent producer does within seconds instead of a
38/// retransmit tail.
39const VIDEO_STALL: Duration = Duration::from_secs(3);
40
41/// How long a fresh video subscription may sit with nothing DECODED before the
42/// pipeline ends itself so the reconcile re-subscribes (restart-ladder paced).
43/// Past the nag window (5 × 2 s): a producer that ignored every re-request is
44/// wedged or its broadcast is a zombie from a same-path replacement, and only a
45/// fresh subscribe — a new announce cursor + control-plane `Subscribe` — heals
46/// those; waiting on the 60 s safety-net IDR is a frozen tile.
47///
48/// The deadline is armed by a DECODED frame, not by a keyframe reaching the gate:
49/// a keyframe in a codec this platform has no decoder for arrives perfectly and
50/// decodes never, so satisfying the latch pre-decode disarms every recovery this
51/// pipeline has and leaves the tile black for the rest of the call. Restarting
52/// against a sender still stuck on that codec is churn, but the reconcile's
53/// [`RestartPacer`] backs it off exponentially, and bounded churn is the price of
54/// ever recovering at all.
55const FIRST_KEYFRAME_DEADLINE: Duration = Duration::from_secs(12);

57/// Holds an inbound stream's `subscribed` stats gauge true for the lifetime of its

77/// Video decode pipeline. The MoQ subscriber pull is async (must run on the
78/// runtime), but the engine `VideoDecoder` (VideoToolbox/dav1d) is `!Send`, so it
79/// CANNOT live across an `.await`. The async half therefore only PULLS frames —
80/// applying the receive-side keyframe gate so we never feed a delta before the
81/// first keyframe — and forwards them to a dedicated blocking decode thread that
82/// owns the `!Send` decoder and pushes decoded I420 to the sink keyed by peer.
83///
84/// The pull is stall-bounded ([`VIDEO_STALL`]): a stuck group is abandoned and a
85/// throttled `RequestKeyframe` sent, so the producer rolls a fresh group and the
86/// tile resumes at live content instead of freezing behind a retransmit tail and
87/// then fast-forwarding. A subscription that never DECODES a frame ends itself at
88/// [`FIRST_KEYFRAME_DEADLINE`] (the reap → paced-resubscribe path).
89///
90/// `first_decode` is the pipeline's delivery latch, shared with the reconcile (it
91/// picks the restart pace) and the subscribe-time keyframe nag (it stops on it).
92/// The DECODE half sets it, not the pull half: arriving is not delivering, and the
93/// two diverge exactly when the sender's codec is one this platform cannot decode.
94#[allow(clippy::too_many_arguments)]
95pub(super) async fn video_pipeline(
96    mut subscriber: MoqTrackSubscriber,
97    peer: PeerId,
98    source: Source,
99    sink: Arc<dyn VideoSink>,
100    stop: Arc<AtomicBool>,
101    peer_stats: Arc<crate::stats::PeerStats>,
102    first_decode: Arc<AtomicBool>,
103    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
104    track_id: u32,
105    downlink_cell: Arc<DownlinkCell>,
106    source_bps: u64,
107    seq_stride: u32,
108) {
109    use hound_transport::VideoPull;
110
111    // The TILE's key, shared with the reconcile loop's release path: both simulcast
112    // layers of this tile land here, so a layer switch never changes it.
113    let key = video_key(peer, source);
114    // Camera vs screen feed SEPARATE inbound-video stat buckets (so a peer sharing
115    // both does not report a summed fps/bitrate).
116    let vkind = inbound_video_kind(source);
117    // RTC-debug: mark this stream's decode pipeline LIVE for as long as this future
118    // exists. A guard (not a pair of calls) because the subscribe loop stops a
119    // pipeline by aborting the task — tokio drops the future, which runs this Drop,
120    // where a trailing statement would simply never execute. The video report needs
121    // the distinction: subscribed-but-dry is a wedge, unsubscribed-and-dry is a track
122    // the peer stopped publishing.
123    let _subscribed = SubscribedGauge::arm(peer_stats.clone(), vkind);
124
125    // Drop-on-full channel from the async pull half to the blocking decode thread.
126    let (frame_tx, frame_rx) = std::sync::mpsc::sync_channel::<WireFrame>(8);
127    let decode_stop = stop.clone();
128    let decode_stats = peer_stats.clone();
129    // Decode → pull back-channel. The decode thread owns the `!Send` decoder and
130    // has no control-plane handle, so a decode failure (stale reference, wedged
131    // session) is raised here; the pull half — the only half holding the keyframe
132    // gate and the PLI throttle — acts on it. Without this the tile freezes until
133    // the sender's next NATURAL IDR (4 s camera, up to 60 s screenshare) because
134    // neither pull-side recovery fires: frames keep arriving (no stall) and a
135    // fast-failing decode never backs the channel up.
136    let resync_needed = Arc::new(AtomicBool::new(false));
137    let decode_resync = resync_needed.clone();
138    let decode_delivered = first_decode.clone();
139    let decode_thread = std::thread::Builder::new()
140        .name(format!("call-video-decode:{}", peer.0))
141        .spawn(move || {
142            video_decode_loop(
143                frame_rx,
144                key,
145                vkind,
146                sink,
147                decode_stop,
148                decode_stats,
149                decode_resync,
150                decode_delivered,
151            )
152        })
153        .ok();
154
155    let started = Instant::now();
156    let mut pli = PliThrottle::default();
157    // One throttled recovery keyframe request — stall, backpressure drop and
158    // decode resync all need exactly this.
159    let ask_keyframe = |pli: &mut PliThrottle| {
160        if pli.fire(Instant::now()) {
161            let _ = control_outbound.send(ClientMessage::RequestKeyframe {
162                track: hound_control::TrackId(track_id),
163            });
164            peer_stats.video_keyframe_requested(vkind);
165        }
166    };
167    // The bandwidth half of demand-driven simulcast: folds each arriving frame's
168    // wire `frame_seq` into a windowed delivery ratio against `source_bps`, with
169    // the source/4 · source/2 hysteresis (see `hound_media::downlink`'s docs). A
170    // fresh instance per pipeline — the reset-on-restart contract falls out of
171    // that for free, matching the fresh `downlink_cell` this pipeline was given.
172    //
173    // `seq_stride` is how many wire sequence numbers one EXPECTED arrival spans:
174    // 1 on an ordinary track, 2 on the relay's derived `…_t0` track, where the
175    // enhancement layer's sequence numbers are missing by design. `source_bps`
176    // stays the FULL track's price either way — that is the currency the
177    // source/4 · source/2 thresholds are written in, and a clean base layer must
178    // read as clearing the top of that band, not as half a link.
179    let mut downlink_demand = hound_media::DownlinkDemand::new(hound_media::DownlinkConfig {
180        expected_seq_stride: seq_stride,
181        ..hound_media::DownlinkConfig::default()
182    });
183
184    let mut awaiting_keyframe = true;
185    // A decode-triggered resync is outstanding (distinct from the pre-first-keyframe
186    // wait, which the subscribe-time nag owns): keep re-asking while deltas arrive.
187    let mut resync_pending = false;
188    while !stop.load(Ordering::Relaxed) {
189        // Checked per iteration, not only on the stall tick: a stream that ARRIVES
190        // but never decodes (an undecodable wire codec) never stalls, so a
191        // stall-only check would never evaluate the one case the deadline exists to
192        // escape. Every `continue` below returns here, so the stall path is covered
193        // by the same test.
194        if !first_decode.load(Ordering::Relaxed) && started.elapsed() >= FIRST_KEYFRAME_DEADLINE {
195            tracing::warn!(
196                peer = peer.0,
197                track = track_id,
198                "video: nothing decoded within the deadline — ending pipeline for a fresh subscribe"
199            );
200            break;
201        }
202        let frame = match subscriber.next_frame_stalled(VIDEO_STALL).await {
203            Ok(VideoPull::Frame(f)) => f,
204            Ok(VideoPull::Stalled) => {
205                // A total downlink stall is exactly when this receiver most needs the
206                // low layer, and `observe_frame` can only decide from an arrival — so
207                // drive the decider from the stall tick too (same window boundary, so
208                // it can neither shorten a window nor double-evaluate one).
209                let now_us = started.elapsed().as_micros().min(i64::MAX as u128) as i64;
210                downlink_cell.store(downlink_demand.on_tick(now_us, source_bps));
211                // Whatever the abandoned group still owed is gone — re-latch the
212                // gate and ask the producer to roll a fresh, decodable group.
213                awaiting_keyframe = true;
214                ask_keyframe(&mut pli);
215                continue;
216            }
217            Ok(VideoPull::End) | Err(_) => break,
218        };
219        // RTC-debug: one inbound encoded AU received for this peer's camera/screen video.
220        peer_stats.video_received(vkind, frame.data.len());
221        // Downlink demand: fold this arrival's wire `frame_seq` into the windowed
222        // delivery-ratio decider, independent of the keyframe gate below — a frame
223        // that arrives but gets dropped pre-decode (awaiting a keyframe) still
224        // reached us over the wire, so it is NOT downlink loss. `now_us` mirrors
225        // the elapsed-since-`started` convention the decode-throughput log below
226        // already uses for this same pipeline.
227        let now_us = started.elapsed().as_micros().min(i64::MAX as u128) as i64;
228        let req = downlink_demand.observe_frame(frame.frame_seq, now_us, source_bps);
229        downlink_cell.store(req);
230        // RTC-debug: the same `frame_seq` continuity the decider folds is the only
231        // receiver-side loss signal this path has — surface its running total.
232        peer_stats.set_video_frames_lost(vkind, downlink_demand.lost_estimate());
233        // Decode failed on the other half: its reference is gone (or its session is
234        // wedged), so every further delta is undecodable. Re-latch the gate — cheap
235        // delta skips instead of feeding a decoder that can't use them — and ask for
236        // a fresh reference.
237        if resync_needed.swap(false, Ordering::Relaxed) {
238            awaiting_keyframe = true;
239            resync_pending = true;
240            ask_keyframe(&mut pli);
241        }
242        // Keyframe gate: drop deltas until the first keyframe so the decode thread
243        // never decodes against a missing reference. Passing it is NOT delivery —
244        // `first_decode` is set by the decode half, on a frame that came out the
245        // other side.
246        if awaiting_keyframe {
247            if !frame.keyframe {
248                // Keep nagging while a resync is outstanding: a single request can
249                // land inside the producer's own keyframe throttle and be coalesced
250                // away, and nothing else re-fires once the stream is flowing.
251                if resync_pending {
252                    ask_keyframe(&mut pli);
253                }
254                continue;
255            }
256            awaiting_keyframe = false;
257            resync_pending = false;
258        }
259        // Backpressure → drop. A dropped frame breaks decode continuity for the
260        // rest of its GOP, so re-latch the gate (cheap delta skips instead of
261        // corrupt decodes) and request a keyframe to resync.
262        if frame_tx.try_send(frame).is_err() {
263            // If the decode thread is gone, end the pull.
264            if decode_thread.as_ref().is_none_or(|h| h.is_finished()) {
265                break;
266            }
267            awaiting_keyframe = true;
268            ask_keyframe(&mut pli);
269        }
270    }
271    drop(frame_tx); // ends the decode loop
272    if let Some(h) = decode_thread {
273        let _ = h.join();
274    }
275}

277/// How long the decode loop withholds retrying platform-decoder construction for a
278/// codec that just failed to build. Without this, the per-packet wire codec byte is
279/// **sender-controlled** — a peer sending AV1 to a box with no AV1 decode makes every
280/// single frame re-attempt `Engine::new().video_decoder(..)` (a VideoToolbox/VAAPI
281/// session setup) and `warn!`, forever. That is a remotely triggerable retry storm
282/// (CPU burn + log spam) for the lifetime of the subscription.
283const RETRY_AFTER: Duration = Duration::from_secs(5);
284
285/// Whether a just-recorded decoder-construction failure should be logged. Only the
286/// FIRST failure of a streak is worth a `warn!`; every suppressed retry inside
287/// [`RETRY_AFTER`] would just repeat it forever.
288#[derive(Debug, Clone, Copy, PartialEq, Eq)]
289enum RetryLog {
290    /// First failure for this codec (or the first after a different codec/success) —
291    /// log it.
292    WarnFirstFailure,
293    /// A retry still inside the window — stay quiet.
294    Suppressed,
295}
296
297/// Pure decoder-(re)build retry policy (see [`RETRY_AFTER`]). Consulted before every
298/// `Engine::new().video_decoder(..)` call for a codec whose current decoder doesn't
299/// match. Takes `now` as a parameter so it is unit-testable without a live clock.
300#[derive(Default)]
301struct DecoderRetry {
302    /// The codec that last failed to construct, and when.
303    failed: Option<(WireCodec, Instant)>,
304}
305
306impl DecoderRetry {
307    /// Should a construction attempt for `codec` be skipped right now? Only true
308    /// within [`RETRY_AFTER`] of a failure for the SAME codec — a different incoming
309    /// codec byte is unrelated and always gets its own first attempt immediately.
310    fn blocked(&self, codec: WireCodec, now: Instant) -> bool {
311        matches!(self.failed, Some((c, at)) if c == codec && now.duration_since(at) < RETRY_AFTER)
312    }
313
314    /// Record a construction failure for `codec` at `now`. Returns whether to log it
315    /// — [`RetryLog::WarnFirstFailure`] only the first time this codec fails in a row.
316    fn on_failure(&mut self, codec: WireCodec, now: Instant) -> RetryLog {
317        let first = !matches!(self.failed, Some((c, _)) if c == codec);
318        self.failed = Some((codec, now));
319        if first {
320            RetryLog::WarnFirstFailure
321        } else {
322            RetryLog::Suppressed
323        }
324    }
325
326    /// Clear the block — call on a successful build.
327    fn clear(&mut self) {
328        self.failed = None;
329    }
330}

332/// What the decode loop should do about one `decode` failure — see [`DecodeFailures`].
333#[derive(Debug, Clone, Copy, PartialEq, Eq)]
334struct DecodeAction {
335    /// `warn!` this one: the first of a streak, then at most one per [`RETRY_AFTER`].
336    log: bool,
337    /// Tear the decoder down so the next frame rebuilds it.
338    rebuild: bool,
339}
340
341/// Pure consecutive-decode-failure policy. A decode error is NOT a construction
342/// error: the usual cause is a missing reference, which the decoders heal from on
343/// the next keyframe by themselves (VideoToolbox rebuilds its session on new
344/// SPS/PPS, openh264 absorbs in-band parameter sets) — so the loop KEEPS the
345/// decoder and only signals the pull half to request one. Tearing down per failure
346/// instead re-ran `Engine::new().video_decoder(..)` (a full VAAPI/openh264 setup)
347/// and an `info!` at the arrival frame rate — the very storm [`RETRY_AFTER`] exists
348/// to bound, reached through the decode door.
349///
350/// Only a streak that OUTLIVES a full [`RETRY_AFTER`] window is treated as a wedged
351/// decoder (a VT session invalidated across sleep keeps identical SPS/PPS, so
352/// nothing triggers its own rebuild) and torn down — at most once per window, so a
353/// sender-controlled corrupt stream still can't drive a rebuild storm. Takes `now`
354/// as a parameter so it is unit-testable without a live clock.
355#[derive(Default)]
356struct DecodeFailures {
357    /// Start of the current failure streak (`None` between streaks).
358    since: Option<Instant>,
359    /// Last failure logged, and last teardown — two independent [`RETRY_AFTER`] paces.
360    logged: Option<Instant>,
361    torn_down: Option<Instant>,
362}
363
364impl DecodeFailures {
365    /// Record a decode failure at `now` and decide what the loop does about it.
366    fn on_failure(&mut self, now: Instant) -> DecodeAction {
367        let started = *self.since.get_or_insert(now);
368        let log = self
369            .logged
370            .is_none_or(|t| now.duration_since(t) >= RETRY_AFTER);
371        if log {
372            self.logged = Some(now);
373        }
374        let rebuild = now.duration_since(started) >= RETRY_AFTER
375            && self
376                .torn_down
377                .is_none_or(|t| now.duration_since(t) >= RETRY_AFTER);
378        if rebuild {
379            self.torn_down = Some(now);
380        }
381        DecodeAction { log, rebuild }
382    }
383
384    /// A frame decoded — the streak (if any) is over.
385    fn on_success(&mut self) {
386        *self = Self::default();
387    }
388}

392/// How long a decoded frame waits in the playout buffer before it reaches the
393/// renderer. Deliberately SHALLOW — between one and two 30 fps frame intervals —
394/// because this stage buys *presentation pacing*, not loss recovery: frames arrive
395/// and decode in bursts (a group flush, a retransmit tail, a slow software decode),
396/// and rendering them the instant they pop out plays a nominally-30 fps tile as
397/// judder. Scheduling each frame against the sender's capture clock replays the
398/// sender's spacing instead.
399///
400/// It is NOT lip-sync. Anchoring video against the audio plane's own (adaptive,
401/// far deeper) jitter depth is a cross-plane problem this stage deliberately does
402/// not attempt — see the module docs.
403const VIDEO_PLAYOUT_DELAY_US: i64 = 40_000;
404
405/// Bound on decoded frames held for playout. A decoded frame can hold a GPU
406/// surface lease (a VideoToolbox IOSurface, a D3D11 texture-pool slot), so the
407/// depth must stay well inside a decoder's pool: 8 is several times the
408/// steady-state occupancy (1–2 frames at the shallow target) and only reachable
409/// when the pump is already hundreds of ms behind, where drop-oldest is the right
410/// answer for video anyway.
411const VIDEO_PLAYOUT_MAX_FRAMES: usize = 8;
412
413/// Consecutive strictly-advancing wire timestamps that turn playout scheduling on
414/// (or, in the other direction, non-advancing ones that latch it off) — see
415/// [`PlayoutMode`]. Three frames is ~100 ms of stream: long enough not to judge a
416/// sender on one duplicate stamp, short enough that pacing engages immediately.
417const VIDEO_STAMP_RUN: u32 = 3;
418
419/// Whether this sender's wire timestamps can schedule playout at all.
420#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
421enum PlayoutMode {
422    /// Not enough evidence yet — render on arrival, so a stream is never delayed
423    /// by the probe itself.
424    #[default]
425    Probing,
426    /// Timestamps advance: schedule through the playout buffer.
427    Buffered,
428    /// Latched off. Some encoders stamp every chunk `TimestampUs::ZERO` (the MFT
429    /// and rav1e paths do), and a buffer keyed on a constant timestamp drops every
430    /// frame after the first as late — a frozen tile. Render on arrival, forever.
431    PassThrough,
432}
433
434/// Pure wire-timestamp usability policy behind [`PlayoutMode`]: it only ever reads
435/// the timestamps a sender puts on the wire, so it is unit-testable without a
436/// decoder, a clock, or a sink.
437#[derive(Debug, Default)]
438struct PlayoutStamps {
439    mode: PlayoutMode,
440    last_ts: Option<i64>,
441    /// Current run of advancing / non-advancing timestamps (each resets the other).
442    advancing: u32,
443    stale: u32,
444}
445
446impl PlayoutStamps {
447    /// Fold one arriving frame's wire timestamp in and report how its decoded
448    /// output should be delivered. [`PlayoutMode::PassThrough`] is terminal — a
449    /// sender that cannot stamp will not start.
450    fn observe(&mut self, ts_us: i64) -> PlayoutMode {
451        if self.mode == PlayoutMode::PassThrough {
452            return self.mode;
453        }
454        match self.last_ts {
455            Some(last) if ts_us > last => {
456                self.stale = 0;
457                self.advancing = self.advancing.saturating_add(1);
458                if self.advancing >= VIDEO_STAMP_RUN {
459                    self.mode = PlayoutMode::Buffered;
460                }
461            }
462            Some(_) => {
463                self.advancing = 0;
464                self.stale = self.stale.saturating_add(1);
465                if self.stale >= VIDEO_STAMP_RUN {
466                    self.mode = PlayoutMode::PassThrough;
467                }
468            }
469            None => {}
470        }
471        // A non-advancing stamp must not move the baseline backwards (a stamp of 0
472        // after a real one would otherwise re-arm the advancing run every frame).
473        self.last_ts = Some(self.last_ts.map_or(ts_us, |last| last.max(ts_us)));
474        self.mode
475    }
476}

478/// A decoded frame waiting for its playout slot. Carries the SENDER's capture-clock
479/// timestamp (what the buffer schedules against — the decoders do not thread a
480/// timestamp onto their output, and the wire frame is the only place it exists) and
481/// the local time it was queued, for the dwell stat. The frame itself is moved in
482/// and out, never copied.
483///
484/// Pairing the decoder's output with the timestamp of the frame just fed assumes
485/// one-in-one-out decode (true for every backend here — no B-frames on the wire). A
486/// decoder that reordered would shift every stamp by a CONSTANT, which the buffer
487/// absorbs in its anchor; only spacing matters to this stage.
488struct PlayoutVideo {
489    ts_us: i64,
490    queued_us: i64,
491    frame: hound_engine::types::RawFrame,
492}
493
494impl hound_media::Timestamped for PlayoutVideo {
495    fn timestamp_us(&self) -> i64 {
496        self.ts_us
497    }
498}

500/// Hand one DECODED frame to the render sink, surface-first (zero-copy GPU handle)
501/// with materialized I420 as the universal fallback. Consumes the frame, so the
502/// no-surface case MOVES the decoder's conversion buffer instead of copying ~1.4 MB
503/// per frame per peer ([`push_preview`] is the borrow-based capture-side twin).
504fn push_decoded(sink: &dyn VideoSink, key: &str, raw: hound_engine::types::RawFrame) {
505    if RenderSurfaceHandle::for_frame(&raw).is_some_and(|handle| sink.push_surface(key, handle)) {
506        return;
507    }
508    let (w, h) = (raw.width(), raw.height());
509    let data = if raw.surface.is_none() && raw.format == hound_engine::types::PixelFormat::I420 {
510        raw.data
511    } else {
512        match raw.as_i420() {
513            Some(c) => c.into_owned().data,
514            None => return,
515        }
516    };
517    sink.push_frame(key, w, h, data);
518}

520/// The blocking video decode loop: owns the `!Send` `VideoDecoder`, rebuilds it
521/// when the sender's codec changes (the sender's choice drives decode), and pushes
522/// each decoded frame to the sink through a shallow playout stage
523/// ([`VIDEO_PLAYOUT_DELAY_US`]) that replays the sender's frame spacing. A decode
524/// failure — and a decoder that will not BUILD — raises `resync_needed`, this
525/// thread's only route to the control plane (see [`DecodeFailures`]).
526///
527/// `first_decode` is the pipeline's delivery latch, set here on the first frame that
528/// actually comes out of the decoder; see [`video_pipeline`] for why the pull half
529/// cannot set it.
530#[allow(clippy::too_many_arguments)]
531fn video_decode_loop(
532    frame_rx: std::sync::mpsc::Receiver<WireFrame>,
533    key: String,
534    vkind: InboundVideoKind,
535    sink: Arc<dyn VideoSink>,
536    stop: Arc<AtomicBool>,
537    peer_stats: Arc<crate::stats::PeerStats>,
538    resync_needed: Arc<AtomicBool>,
539    first_decode: Arc<AtomicBool>,
540) {
541    use hound_engine::types::VideoCodec;
542
543    let mut decoder: Option<VideoDecoder> = None;
544    let mut decoder_codec: Option<WireCodec> = None;
545    let mut retry = DecoderRetry::default();
546    let mut decode_failures = DecodeFailures::default();
547    // The current decode path + codec (captured at build) and a ~5 s throughput
548    // window, so the log shows decoded geometry + ACTUAL decode fps + mean decode
549    // cost — a crawling software decode is visible at a glance.
550    let mut current_backend: &'static str = "?";
551    let mut current_codec = VideoCodec::H264;
552    let mut win_start: Option<std::time::Instant> = None;
553    let mut win_frames: u64 = 0;
554    let mut win_decode_us: u64 = 0;
555
556    // Playout stage: decoded frames are scheduled a shallow `VIDEO_PLAYOUT_DELAY_US`
557    // ahead on the sender's capture timeline and released newest-due-first. `clock`
558    // is this thread's local timeline; the buffer maps the sender's stamps onto it
559    // on first push. Recovery is untouched — this sits entirely BELOW decode, so
560    // the pull half's keyframe gate, PLI throttle, stall and resync paths never see
561    // it, and the channel is drained exactly as fast as before.
562    let clock = CaptureClock::start();
563    let mut playout: hound_media::PlayoutBuffer<PlayoutVideo> =
564        hound_media::PlayoutBuffer::with_capacity(VIDEO_PLAYOUT_DELAY_US, VIDEO_PLAYOUT_MAX_FRAMES);
565    let mut stamps = PlayoutStamps::default();
566    let mut reported_drops = 0u64;
567
568    loop {
569        // Wait for the next encoded frame — but never past the moment a buffered
570        // frame comes due, so a sparse producer (a static share re-emitting at ~1 fps)
571        // is still released at the target instead of being held for its successor.
572        let arrived = match playout.next_due_us() {
573            Some(due) => {
574                let wait = Duration::from_micros(due.saturating_sub(clock.now()).max(0) as u64);
575                match frame_rx.recv_timeout(wait) {
576                    Ok(f) => Some(f),
577                    Err(std::sync::mpsc::RecvTimeoutError::Timeout) => None,
578                    Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => break,
579                }
580            }
581            None => match frame_rx.recv() {
582                Ok(f) => Some(f),
583                Err(_) => break,
584            },
585        };
586        if stop.load(Ordering::Relaxed) {
587            break;
588        }
589        // Release BEFORE decoding what just arrived: that frame belongs to a later
590        // playout slot, so its decode must not delay a frame that is due now. A
591        // burst collapses to the freshest — an overdrawn intermediate is pure judder.
592        let now_us = clock.now();
593        if let Some(item) = playout.pop_latest_due(now_us) {
594            peer_stats.video_dwell(vkind, now_us - item.queued_us);
595            push_decoded(sink.as_ref(), &key, item.frame);
596        }
597        // Playout-drop total: superseded (the collapse policy above), plus the
598        // buffer's own late/overflow drops. A gauge — only written when it moves.
599        let dropped = playout.superseded_drops() + playout.late_drops() + playout.overflow_drops();
600        if dropped != reported_drops {
601            reported_drops = dropped;
602            peer_stats.set_video_playout_dropped_total(vkind, dropped);
603        }
604        let Some(frame) = arrived else {
605            continue; // woke only to release
606        };
607        // (Re)build the decoder if the sender's codec changed.
608        if decoder_codec != Some(frame.codec) {
609            let now = std::time::Instant::now();
610            if retry.blocked(frame.codec, now) {
611                // Still inside the retry window for this codec's last failure — skip
612                // the expensive platform-decoder construction and stay silent (see
613                // `RETRY_AFTER`).
614                tracing::trace!(codec = ?frame.codec, "video: decoder build suppressed (retry window)");
615                continue;
616            }
617            let vc = match frame.codec {
618                WireCodec::H264 => VideoCodec::H264,
619                WireCodec::H265 => VideoCodec::H265,
620                WireCodec::Av1 => VideoCodec::Av1,
621                WireCodec::Opus => continue, // not a video codec; ignore
622            };
623            match Engine::new().video_decoder(vc) {
624                Ok(d) => {
625                    retry.clear();
626                    // Surface the actual decode path + codec. A software fallback
627                    // (e.g. AV1 with no HW block) is CPU-heavy and can starve the
628                    // audio fill path, so make HW-vs-SW visible in the call log and
629                    // in the per-stream stats. Re-read on the decode-FAILURE arm
630                    // below as well as here, because the engine's composite decoders
631                    // switch arms mid-stream (a refused hardware decode session
632                    // degrades that stream onto openh264/dav1d) — so this is the
633                    // live answer, not the one chosen at build.
634                    current_backend = d.backend();
635                    peer_stats.set_video_decoder(vkind, current_backend, d.is_hardware());
636                    current_codec = vc;
637                    win_start = None;
638                    win_frames = 0;
639                    win_decode_us = 0;
640                    tracing::info!(
641                        target: "video_decode",
642                        key = %key,
643                        codec = ?vc,
644                        backend = current_backend,
645                        "video: decode path selected"
646                    );
647                    decoder = Some(d);
648                    decoder_codec = Some(frame.codec);
649                    // Surface the live decode codec in the RTC-debug panel (also
650                    // tracks a mid-stream sender codec switch), and count the build:
651                    // repeated builds inside one window are churn, and each one costs
652                    // the stream its reference chain (the video report reads this).
653                    peer_stats.set_inbound_video_codec(vkind, video_codec_to_ctl(vc).label());
654                    peer_stats.video_decoder_built(vkind);
655                }
656                // A decoder that will not BUILD is a black tile whose cause is
657                // invisible in every other counter: frames arrive, none decode, which
658                // reads exactly like a broken reference chain. So the failure is
659                // recorded as its own stat — `video_report` attributes it to codec
660                // negotiation instead of to a wedged downlink — and it raises
661                // `resync_needed`, which is what turns a publisher's codec downgrade
662                // into a keyframe within a frame or two instead of at the sender's
663                // next natural IDR (up to 60 s on a share). Both are rate-bound by
664                // `RETRY_AFTER` above (this arm is unreachable inside the window) and
665                // by the pull half's `PliThrottle`.
666                //
667                // Apple decode-session exhaustion does NOT land here, despite an
668                // earlier comment on this arm saying so: VideoToolbox's open is
669                // infallible (the session is built lazily on the first keyframe), so
670                // that failure arrives on the DECODE-failure arm below instead. The
671                // engine now degrades it there — H.264 switches to the SW openh264
672                // floor mid-stream and AV1 to the SW dav1d floor (`hound_engine`'s
673                // `apple::video_codec::AppleVideoDecoder`), which is why the
674                // switching decode error is deliberately still returned: this loop's
675                // `resync_needed` is what fetches the keyframe the software arm needs
676                // to start. HEVC keeps no software arm anywhere, and the preventive
677                // cap — `DecodeBudget`, seeded and retargeted from
678                // `Engine::decode_session_budget()` — is still the first line of
679                // defence for all three.
680                Err(e) => {
681                    // `Unsupported` is the platform's typed capability miss: no retry,
682                    // keyframe or reconnect can fix it — only the sender downgrading.
683                    // Every other error is transient and the retry is the cure, so the
684                    // report must not tell the reader to go change the sender's codec.
685                    let unsupported =
686                        matches!(e, hound_engine::error::CaptureError::Unsupported { .. });
687                    peer_stats.video_decoder_build_failed(
688                        vkind,
689                        wire_codec_to_ctl(frame.codec).label(),
690                        unsupported,
691                    );
692                    resync_needed.store(true, Ordering::Relaxed);
693                    match retry.on_failure(frame.codec, now) {
694                        RetryLog::WarnFirstFailure => {
695                            tracing::warn!(%e, codec = ?frame.codec, unsupported, "video: failed to build decoder");
696                        }
697                        RetryLog::Suppressed => {
698                            tracing::trace!(%e, codec = ?frame.codec, "video: decoder build still failing");
699                        }
700                    }
701                    continue;
702                }
703            }
704        }
705        let Some(dec) = decoder.as_mut() else {
706            continue;
707        };
708        let decode_start = std::time::Instant::now();
709        match dec.decode(&frame.data) {
710            Ok(Some(raw)) => {
711                decode_failures.on_success();
712                first_decode.store(true, Ordering::Relaxed);
713                let decode_us = decode_start.elapsed().as_micros().min(i64::MAX as u128) as i64;
714                let (w, h) = (raw.width(), raw.height());
715                // Every decoder backend emits an OWNED I420 `RawFrame` — each `decode`
716                // call allocates a fresh conversion buffer (vt_decoder / vaapi / sw all
717                // output I420, asserted in their tests). So the frame is MOVED from here
718                // to the sink (through the playout buffer, which only ever reads its
719                // timestamp): the decoder's single conversion alloc stays the only one
720                // on this path, and a GPU-resident frame keeps its zero-copy surface.
721                match stamps.observe(frame.timestamp_us) {
722                    // Schedule it — the pump at the top of the loop releases it at its
723                    // playout time.
724                    PlayoutMode::Buffered => {
725                        let queued_us = clock.now();
726                        playout.push(
727                            queued_us,
728                            PlayoutVideo {
729                                ts_us: frame.timestamp_us,
730                                queued_us,
731                                frame: raw,
732                            },
733                        );
734                    }
735                    // Still probing: render on arrival so the probe itself costs nothing.
736                    PlayoutMode::Probing => push_decoded(sink.as_ref(), &key, raw),
737                    // Latched off (possibly just now): hand over anything still queued —
738                    // newest wins — so no frame is stranded, then render on arrival.
739                    PlayoutMode::PassThrough => {
740                        if let Some(item) = playout.pop_latest_due(i64::MAX) {
741                            push_decoded(sink.as_ref(), &key, item.frame);
742                        }
743                        push_decoded(sink.as_ref(), &key, raw);
744                    }
745                }
746                // RTC-debug: one decoded frame (geometry + decode-time tail).
747                peer_stats.video_decoded(vkind, w, h, decode_us);
748
749                // ~5 s decode-throughput sample: decoded geometry + ACTUAL fps +
750                // mean decode cost on the current path (HW/SW visible above).
751                let now = std::time::Instant::now();
752                let start = *win_start.get_or_insert(now);
753                win_frames += 1;
754                win_decode_us += decode_us.max(0) as u64;
755                let elapsed = now.duration_since(start);
756                if elapsed >= std::time::Duration::from_secs(5) {
757                    let secs = elapsed.as_secs_f64().max(0.001);
758                    let fps = win_frames as f64 / secs;
759                    let mean_ms = (win_decode_us as f64 / win_frames as f64) / 1000.0;
760                    tracing::info!(
761                        target: "video_decode",
762                        key = %key,
763                        codec = ?current_codec,
764                        backend = current_backend,
765                        width = w,
766                        height = h,
767                        fps = format!("{fps:.1}"),
768                        decode_ms = format!("{mean_ms:.1}"),
769                        "video: decode throughput"
770                    );
771                    win_start = Some(now);
772                    win_frames = 0;
773                    win_decode_us = 0;
774                }
775            }
776            Ok(None) => {} // decoder buffering; keep feeding
777            Err(e) => {
778                // Ask the pull half for a fresh reference — this thread has no
779                // control-plane handle, and nothing else notices a decode failure
780                // (frames keep arriving, so no stall; failing is fast, so no
781                // backpressure). Decoder teardown is paced, not per-frame — see
782                // `DecodeFailures`.
783                let act = decode_failures.on_failure(std::time::Instant::now());
784                // The engine's HW→SW switch happens ON this failure and RETURNS the
785                // triggering error (deliberately — it is what raises `resync_needed`
786                // below, which fetches the keyframe the freshly-installed software
787                // decoder needs to start). So the decode path may have changed since
788                // the last line: re-read it, which both keeps the log honest and
789                // bumps the per-stream fallback counter that makes a fleet-wide
790                // hardware-decode regression visible.
791                current_backend = dec.backend();
792                peer_stats.set_video_decoder(vkind, current_backend, dec.is_hardware());
793                if act.log {
794                    tracing::warn!(
795                        %e,
796                        key = %key,
797                        codec = ?current_codec,
798                        backend = current_backend,
799                        rebuild = act.rebuild,
800                        "video: decode failed — requesting a keyframe"
801                    );
802                }
803                if act.rebuild {
804                    decoder = None;
805                    decoder_codec = None;
806                }
807                resync_needed.store(true, Ordering::Relaxed);
808            }
809        }
810    }
811}
