1//! Inbound video: the async pull half (receive-side keyframe gate; the nag task
2//! lives with the subscribe loop) feeds a blocking decode thread that owns the
3//! `!Send` `VideoDecoder` and rebuilds it whenever the sender's per-packet codec
4//! byte changes — the sender's choice drives decode. Decoded frames pass through a
5//! shallow playout stage ([`VIDEO_PLAYOUT_DELAY_US`]) that replays the sender's
6//! frame spacing, then reach the sink surface-first (zero-copy GPU handle) with
7//! materialized I420 as the universal fallback; [`push_preview`] is the
8//! borrow-based capture-side twin the publish loops use for self-preview.
9//!
10//! The playout stage is presentation pacing only — it does NOT anchor video against
11//! the audio plane's own (adaptive, deeper) jitter depth, so the two planes are
12//! still independently scheduled.
13

16/// Push one CAPTURED frame to the render sink for a self-preview tile, preferring
17/// the ZERO-COPY surface path: a surface-backed frame (SCK / AVFoundation NV12
18/// IOSurface) is handed to the renderer to sample directly — no NV12→I420 CPU
19/// conversion, no ~1.4–3 MB per-frame alloc+copy, no re-upload (~90 MB/s of
20/// conversion at a 1080p 30 fps share). Falls back to materialized I420
21/// for non-surface frames (scaled camera frames, other
22/// platforms) or a sink without a surface renderer. Borrows the frame — the
23/// caller still owns it for encode. (The REMOTE decode path keeps its own
24/// ownership-optimized version of this pattern: it consumes the frame, so its
25/// no-surface case can MOVE the bytes instead of copying.)
26pub(super) fn push_preview(sink: &dyn VideoSink, key: &str, frame: &hound_engine::types::RawFrame) {
27    let zero_copy =
28        RenderSurfaceHandle::for_frame(frame).is_some_and(|handle| sink.push_surface(key, handle));
29    if !zero_copy && let Some(i420) = frame.as_i420() {
30        sink.push_frame(key, i420.width(), i420.height(), i420.into_owned().data);
31    }
32}

34/// How long the video pull tolerates delivering NOTHING before treating the
35/// subscription as stalled. Above the static-share re-emit cadence
36/// ([`SCREEN_STATIC_REFRESH`], 1 s) so a healthy static share never trips it; a
37/// mid-GOP loss stall or a silent producer does within seconds instead of a
38/// retransmit tail.
39const VIDEO_STALL: Duration = Duration::from_secs(3);
40
41/// How long a fresh video subscription may sit with NO keyframe before the
42/// pipeline ends itself so the reconcile re-subscribes (restart-ladder paced).
43/// Past the nag window (5 × 2 s): a producer that ignored every re-request is
44/// wedged or its broadcast is a zombie from a same-path replacement, and only a
45/// fresh subscribe — a new announce cursor + control-plane `Subscribe` — heals
46/// those; waiting on the 60 s safety-net IDR is a frozen tile.
47const FIRST_KEYFRAME_DEADLINE: Duration = Duration::from_secs(12);

49/// Holds an inbound stream's `subscribed` stats gauge true for the lifetime of its

60    }
61}

69/// Video decode pipeline. The MoQ subscriber pull is async (must run on the
70/// runtime), but the engine `VideoDecoder` (VideoToolbox/dav1d) is `!Send`, so it
71/// CANNOT live across an `.await`. The async half therefore only PULLS frames —
72/// applying the receive-side keyframe gate so we never feed a delta before the
73/// first keyframe — and forwards them to a dedicated blocking decode thread that
74/// owns the `!Send` decoder and pushes decoded I420 to the sink keyed by peer.
75///
76/// The pull is stall-bounded ([`VIDEO_STALL`]): a stuck group is abandoned and a
77/// throttled `RequestKeyframe` sent, so the producer rolls a fresh group and the
78/// tile resumes at live content instead of freezing behind a retransmit tail and
79/// then fast-forwarding. A subscription that never yields its first keyframe
80/// ends itself at [`FIRST_KEYFRAME_DEADLINE`] (the reap → paced-resubscribe path).
81#[allow(clippy::too_many_arguments)]
82pub(super) async fn video_pipeline(
83    mut subscriber: MoqTrackSubscriber,
84    peer: PeerId,
85    source: Source,
86    sink: Arc<dyn VideoSink>,
87    stop: Arc<AtomicBool>,
88    peer_stats: Arc<crate::stats::PeerStats>,
89    got_keyframe: Arc<AtomicBool>,
90    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
91    track_id: u32,
92    downlink_cell: Arc<DownlinkCell>,
93    source_bps: u64,
94) {
95    use hound_transport::VideoPull;
96
97    // The TILE's key, shared with the reconcile loop's release path: both simulcast
98    // layers of this tile land here, so a layer switch never changes it.
99    let key = video_key(peer, source);
100    // Camera vs screen feed SEPARATE inbound-video stat buckets (so a peer sharing
101    // both does not report a summed fps/bitrate).
102    let vkind = inbound_video_kind(source);
103    // RTC-debug: mark this stream's decode pipeline LIVE for as long as this future
104    // exists. A guard (not a pair of calls) because the subscribe loop stops a
105    // pipeline by aborting the task — tokio drops the future, which runs this Drop,
106    // where a trailing statement would simply never execute. The video report needs
107    // the distinction: subscribed-but-dry is a wedge, unsubscribed-and-dry is a track
108    // the peer stopped publishing.
109    let _subscribed = SubscribedGauge::arm(peer_stats.clone(), vkind);
110
111    // Drop-on-full channel from the async pull half to the blocking decode thread.
112    let (frame_tx, frame_rx) = std::sync::mpsc::sync_channel::<WireFrame>(8);
113    let decode_stop = stop.clone();
114    let decode_stats = peer_stats.clone();
115    // Decode → pull back-channel. The decode thread owns the `!Send` decoder and
116    // has no control-plane handle, so a decode failure (stale reference, wedged
117    // session) is raised here; the pull half — the only half holding the keyframe
118    // gate and the PLI throttle — acts on it. Without this the tile freezes until
119    // the sender's next NATURAL IDR (4 s camera, up to 60 s screenshare) because
120    // neither pull-side recovery fires: frames keep arriving (no stall) and a
121    // fast-failing decode never backs the channel up.
122    let resync_needed = Arc::new(AtomicBool::new(false));
123    let decode_resync = resync_needed.clone();
124    let decode_thread = std::thread::Builder::new()
125        .name(format!("call-video-decode:{}", peer.0))
126        .spawn(move || {
127            video_decode_loop(
128                frame_rx,
129                key,
130                vkind,
131                sink,
132                decode_stop,
133                decode_stats,
134                decode_resync,
135            )
136        })
137        .ok();
138
139    let started = Instant::now();
140    let mut pli = PliThrottle::default();
141    // One throttled recovery keyframe request — stall, backpressure drop and
142    // decode resync all need exactly this.
143    let ask_keyframe = |pli: &mut PliThrottle| {
144        if pli.fire(Instant::now()) {
145            let _ = control_outbound.send(ClientMessage::RequestKeyframe {
146                track: hound_control::TrackId(track_id),
147            });
148            peer_stats.video_keyframe_requested(vkind);
149        }
150    };
151    // The bandwidth half of demand-driven simulcast: folds each arriving frame's
152    // wire `frame_seq` into a windowed delivery ratio against `source_bps`, with
153    // the source/4 · source/2 hysteresis (see `hound_media::downlink`'s docs). A
154    // fresh instance per pipeline — the reset-on-restart contract falls out of
155    // that for free, matching the fresh `downlink_cell` this pipeline was given.
156    let mut downlink_demand =
157        hound_media::DownlinkDemand::new(hound_media::DownlinkConfig::default());
158
159    let mut awaiting_keyframe = true;
160    // A decode-triggered resync is outstanding (distinct from the pre-first-keyframe
161    // wait, which the subscribe-time nag owns): keep re-asking while deltas arrive.
162    let mut resync_pending = false;
163    while !stop.load(Ordering::Relaxed) {
164        let frame = match subscriber.next_frame_stalled(VIDEO_STALL).await {
165            Ok(VideoPull::Frame(f)) => f,
166            Ok(VideoPull::Stalled) => {
167                // A total downlink stall is exactly when this receiver most needs the
168                // low layer, and `observe_frame` can only decide from an arrival — so
169                // drive the decider from the stall tick too (same window boundary, so
170                // it can neither shorten a window nor double-evaluate one).
171                let now_us = started.elapsed().as_micros().min(i64::MAX as u128) as i64;
172                downlink_cell.store(downlink_demand.on_tick(now_us, source_bps));
173                if !got_keyframe.load(Ordering::Relaxed)
174                    && started.elapsed() >= FIRST_KEYFRAME_DEADLINE
175                {
176                    tracing::warn!(
177                        peer = peer.0,
178                        track = track_id,
179                        "video: no keyframe within the deadline — ending pipeline for a fresh subscribe"
180                    );
181                    break;
182                }
183                // Whatever the abandoned group still owed is gone — re-latch the
184                // gate and ask the producer to roll a fresh, decodable group.
185                awaiting_keyframe = true;
186                ask_keyframe(&mut pli);
187                continue;
188            }
189            Ok(VideoPull::End) | Err(_) => break,
190        };
191        // RTC-debug: one inbound encoded AU received for this peer's camera/screen video.
192        peer_stats.video_received(vkind, frame.data.len());
193        // Downlink demand: fold this arrival's wire `frame_seq` into the windowed
194        // delivery-ratio decider, independent of the keyframe gate below — a frame
195        // that arrives but gets dropped pre-decode (awaiting a keyframe) still
196        // reached us over the wire, so it is NOT downlink loss. `now_us` mirrors
197        // the elapsed-since-`started` convention the decode-throughput log below
198        // already uses for this same pipeline.
199        let now_us = started.elapsed().as_micros().min(i64::MAX as u128) as i64;
200        let req = downlink_demand.observe_frame(frame.frame_seq, now_us, source_bps);
201        downlink_cell.store(req);
202        // RTC-debug: the same `frame_seq` continuity the decider folds is the only
203        // receiver-side loss signal this path has — surface its running total.
204        peer_stats.set_video_frames_lost(vkind, downlink_demand.lost_estimate());
205        // Decode failed on the other half: its reference is gone (or its session is
206        // wedged), so every further delta is undecodable. Re-latch the gate — cheap
207        // delta skips instead of feeding a decoder that can't use them — and ask for
208        // a fresh reference.
209        if resync_needed.swap(false, Ordering::Relaxed) {
210            awaiting_keyframe = true;
211            resync_pending = true;
212            ask_keyframe(&mut pli);
213        }
214        // Keyframe gate: drop deltas until the first keyframe so the decode thread
215        // never decodes against a missing reference. Passing the gate also stops
216        // the awaiting-keyframe re-request nag (see `start_pipeline`).
217        if awaiting_keyframe {
218            if !frame.keyframe {
219                // Keep nagging while a resync is outstanding: a single request can
220                // land inside the producer's own keyframe throttle and be coalesced
221                // away, and nothing else re-fires once the stream is flowing.
222                if resync_pending {
223                    ask_keyframe(&mut pli);
224                }
225                continue;
226            }
227            awaiting_keyframe = false;
228            resync_pending = false;
229            got_keyframe.store(true, Ordering::Relaxed);
230        }
231        // Backpressure → drop. A dropped frame breaks decode continuity for the
232        // rest of its GOP, so re-latch the gate (cheap delta skips instead of
233        // corrupt decodes) and request a keyframe to resync.
234        if frame_tx.try_send(frame).is_err() {
235            // If the decode thread is gone, end the pull.
236            if decode_thread.as_ref().is_none_or(|h| h.is_finished()) {
237                break;
238            }
239            awaiting_keyframe = true;
240            ask_keyframe(&mut pli);
241        }
242    }
243    drop(frame_tx); // ends the decode loop
244    if let Some(h) = decode_thread {
245        let _ = h.join();
246    }
247}

249/// How long the decode loop withholds retrying platform-decoder construction for a
250/// codec that just failed to build. Without this, the per-packet wire codec byte is
251/// **sender-controlled** — a peer sending AV1 to a box with no AV1 decode makes every
252/// single frame re-attempt `Engine::new().video_decoder(..)` (a VideoToolbox/VAAPI
253/// session setup) and `warn!`, forever. That is a remotely triggerable retry storm
254/// (CPU burn + log spam) for the lifetime of the subscription.
255const RETRY_AFTER: Duration = Duration::from_secs(5);
256
257/// Whether a just-recorded decoder-construction failure should be logged. Only the
258/// FIRST failure of a streak is worth a `warn!`; every suppressed retry inside
259/// [`RETRY_AFTER`] would just repeat it forever.
260#[derive(Debug, Clone, Copy, PartialEq, Eq)]
261enum RetryLog {
262    /// First failure for this codec (or the first after a different codec/success) —
263    /// log it.
264    WarnFirstFailure,
265    /// A retry still inside the window — stay quiet.
266    Suppressed,
267}
268
269/// Pure decoder-(re)build retry policy (see [`RETRY_AFTER`]). Consulted before every
270/// `Engine::new().video_decoder(..)` call for a codec whose current decoder doesn't
271/// match. Takes `now` as a parameter so it is unit-testable without a live clock.
272#[derive(Default)]
273struct DecoderRetry {
274    /// The codec that last failed to construct, and when.
275    failed: Option<(WireCodec, Instant)>,
276}
277
278impl DecoderRetry {
279    /// Should a construction attempt for `codec` be skipped right now? Only true
280    /// within [`RETRY_AFTER`] of a failure for the SAME codec — a different incoming
281    /// codec byte is unrelated and always gets its own first attempt immediately.
282    fn blocked(&self, codec: WireCodec, now: Instant) -> bool {
283        matches!(self.failed, Some((c, at)) if c == codec && now.duration_since(at) < RETRY_AFTER)
284    }
285
286    /// Record a construction failure for `codec` at `now`. Returns whether to log it
287    /// — [`RetryLog::WarnFirstFailure`] only the first time this codec fails in a row.
288    fn on_failure(&mut self, codec: WireCodec, now: Instant) -> RetryLog {
289        let first = !matches!(self.failed, Some((c, _)) if c == codec);
290        self.failed = Some((codec, now));
291        if first {
292            RetryLog::WarnFirstFailure
293        } else {
294            RetryLog::Suppressed
295        }
296    }
297
298    /// Clear the block — call on a successful build.
299    fn clear(&mut self) {
300        self.failed = None;
301    }
302}

304/// What the decode loop should do about one `decode` failure — see [`DecodeFailures`].
305#[derive(Debug, Clone, Copy, PartialEq, Eq)]
306struct DecodeAction {
307    /// `warn!` this one: the first of a streak, then at most one per [`RETRY_AFTER`].
308    log: bool,
309    /// Tear the decoder down so the next frame rebuilds it.
310    rebuild: bool,
311}
312
313/// Pure consecutive-decode-failure policy. A decode error is NOT a construction
314/// error: the usual cause is a missing reference, which the decoders heal from on
315/// the next keyframe by themselves (VideoToolbox rebuilds its session on new
316/// SPS/PPS, openh264 absorbs in-band parameter sets) — so the loop KEEPS the
317/// decoder and only signals the pull half to request one. Tearing down per failure
318/// instead re-ran `Engine::new().video_decoder(..)` (a full VAAPI/openh264 setup)
319/// and an `info!` at the arrival frame rate — the very storm [`RETRY_AFTER`] exists
320/// to bound, reached through the decode door.
321///
322/// Only a streak that OUTLIVES a full [`RETRY_AFTER`] window is treated as a wedged
323/// decoder (a VT session invalidated across sleep keeps identical SPS/PPS, so
324/// nothing triggers its own rebuild) and torn down — at most once per window, so a
325/// sender-controlled corrupt stream still can't drive a rebuild storm. Takes `now`
326/// as a parameter so it is unit-testable without a live clock.
327#[derive(Default)]
328struct DecodeFailures {
329    /// Start of the current failure streak (`None` between streaks).
330    since: Option<Instant>,
331    /// Last failure logged, and last teardown — two independent [`RETRY_AFTER`] paces.
332    logged: Option<Instant>,
333    torn_down: Option<Instant>,
334}
335
336impl DecodeFailures {
337    /// Record a decode failure at `now` and decide what the loop does about it.
338    fn on_failure(&mut self, now: Instant) -> DecodeAction {
339        let started = *self.since.get_or_insert(now);
340        let log = self
341            .logged
342            .is_none_or(|t| now.duration_since(t) >= RETRY_AFTER);
343        if log {
344            self.logged = Some(now);
345        }
346        let rebuild = now.duration_since(started) >= RETRY_AFTER
347            && self
348                .torn_down
349                .is_none_or(|t| now.duration_since(t) >= RETRY_AFTER);
350        if rebuild {
351            self.torn_down = Some(now);
352        }
353        DecodeAction { log, rebuild }
354    }
355
356    /// A frame decoded — the streak (if any) is over.
357    fn on_success(&mut self) {
358        *self = Self::default();
359    }
360}

364/// How long a decoded frame waits in the playout buffer before it reaches the
365/// renderer. Deliberately SHALLOW — between one and two 30 fps frame intervals —
366/// because this stage buys *presentation pacing*, not loss recovery: frames arrive
367/// and decode in bursts (a group flush, a retransmit tail, a slow software decode),
368/// and rendering them the instant they pop out plays a nominally-30 fps tile as
369/// judder. Scheduling each frame against the sender's capture clock replays the
370/// sender's spacing instead.
371///
372/// It is NOT lip-sync. Anchoring video against the audio plane's own (adaptive,
373/// far deeper) jitter depth is a cross-plane problem this stage deliberately does
374/// not attempt — see the module docs.
375const VIDEO_PLAYOUT_DELAY_US: i64 = 40_000;
376
377/// Bound on decoded frames held for playout. A decoded frame can hold a GPU
378/// surface lease (a VideoToolbox IOSurface, a D3D11 texture-pool slot), so the
379/// depth must stay well inside a decoder's pool: 8 is several times the
380/// steady-state occupancy (1–2 frames at the shallow target) and only reachable
381/// when the pump is already hundreds of ms behind, where drop-oldest is the right
382/// answer for video anyway.
383const VIDEO_PLAYOUT_MAX_FRAMES: usize = 8;
384
385/// Consecutive strictly-advancing wire timestamps that turn playout scheduling on
386/// (or, in the other direction, non-advancing ones that latch it off) — see
387/// [`PlayoutMode`]. Three frames is ~100 ms of stream: long enough not to judge a
388/// sender on one duplicate stamp, short enough that pacing engages immediately.
389const VIDEO_STAMP_RUN: u32 = 3;
390
391/// Whether this sender's wire timestamps can schedule playout at all.
392#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
393enum PlayoutMode {
394    /// Not enough evidence yet — render on arrival, so a stream is never delayed
395    /// by the probe itself.
396    #[default]
397    Probing,
398    /// Timestamps advance: schedule through the playout buffer.
399    Buffered,
400    /// Latched off. Some encoders stamp every chunk `TimestampUs::ZERO` (the MFT
401    /// and rav1e paths do), and a buffer keyed on a constant timestamp drops every
402    /// frame after the first as late — a frozen tile. Render on arrival, forever.
403    PassThrough,
404}
405
406/// Pure wire-timestamp usability policy behind [`PlayoutMode`]: it only ever reads
407/// the timestamps a sender puts on the wire, so it is unit-testable without a
408/// decoder, a clock, or a sink.
409#[derive(Debug, Default)]
410struct PlayoutStamps {
411    mode: PlayoutMode,
412    last_ts: Option<i64>,
413    /// Current run of advancing / non-advancing timestamps (each resets the other).
414    advancing: u32,
415    stale: u32,
416}
417
418impl PlayoutStamps {
419    /// Fold one arriving frame's wire timestamp in and report how its decoded
420    /// output should be delivered. [`PlayoutMode::PassThrough`] is terminal — a
421    /// sender that cannot stamp will not start.
422    fn observe(&mut self, ts_us: i64) -> PlayoutMode {
423        if self.mode == PlayoutMode::PassThrough {
424            return self.mode;
425        }
426        match self.last_ts {
427            Some(last) if ts_us > last => {
428                self.stale = 0;
429                self.advancing = self.advancing.saturating_add(1);
430                if self.advancing >= VIDEO_STAMP_RUN {
431                    self.mode = PlayoutMode::Buffered;
432                }
433            }
434            Some(_) => {
435                self.advancing = 0;
436                self.stale = self.stale.saturating_add(1);
437                if self.stale >= VIDEO_STAMP_RUN {
438                    self.mode = PlayoutMode::PassThrough;
439                }
440            }
441            None => {}
442        }
443        // A non-advancing stamp must not move the baseline backwards (a stamp of 0
444        // after a real one would otherwise re-arm the advancing run every frame).
445        self.last_ts = Some(self.last_ts.map_or(ts_us, |last| last.max(ts_us)));
446        self.mode
447    }
448}

450/// A decoded frame waiting for its playout slot. Carries the SENDER's capture-clock
451/// timestamp (what the buffer schedules against — the decoders do not thread a
452/// timestamp onto their output, and the wire frame is the only place it exists) and
453/// the local time it was queued, for the dwell stat. The frame itself is moved in
454/// and out, never copied.
455///
456/// Pairing the decoder's output with the timestamp of the frame just fed assumes
457/// one-in-one-out decode (true for every backend here — no B-frames on the wire). A
458/// decoder that reordered would shift every stamp by a CONSTANT, which the buffer
459/// absorbs in its anchor; only spacing matters to this stage.
460struct PlayoutVideo {
461    ts_us: i64,
462    queued_us: i64,
463    frame: hound_engine::types::RawFrame,
464}
465
466impl hound_media::Timestamped for PlayoutVideo {
467    fn timestamp_us(&self) -> i64 {
468        self.ts_us
469    }
470}

472/// Hand one DECODED frame to the render sink, surface-first (zero-copy GPU handle)
473/// with materialized I420 as the universal fallback. Consumes the frame, so the
474/// no-surface case MOVES the decoder's conversion buffer instead of copying ~1.4 MB
475/// per frame per peer ([`push_preview`] is the borrow-based capture-side twin).
476fn push_decoded(sink: &dyn VideoSink, key: &str, raw: hound_engine::types::RawFrame) {
477    if RenderSurfaceHandle::for_frame(&raw).is_some_and(|handle| sink.push_surface(key, handle)) {
478        return;
479    }
480    let (w, h) = (raw.width(), raw.height());
481    let data = if raw.surface.is_none() && raw.format == hound_engine::types::PixelFormat::I420 {
482        raw.data
483    } else {
484        match raw.as_i420() {
485            Some(c) => c.into_owned().data,
486            None => return,
487        }
488    };
489    sink.push_frame(key, w, h, data);
490}

492/// The blocking video decode loop: owns the `!Send` `VideoDecoder`, rebuilds it
493/// when the sender's codec changes (the sender's choice drives decode), and pushes
494/// each decoded frame to the sink through a shallow playout stage
495/// ([`VIDEO_PLAYOUT_DELAY_US`]) that replays the sender's frame spacing. A decode
496/// failure raises `resync_needed` — this thread's only route to the control plane
497/// (see [`DecodeFailures`]).
498fn video_decode_loop(
499    frame_rx: std::sync::mpsc::Receiver<WireFrame>,
500    key: String,
501    vkind: InboundVideoKind,
502    sink: Arc<dyn VideoSink>,
503    stop: Arc<AtomicBool>,
504    peer_stats: Arc<crate::stats::PeerStats>,
505    resync_needed: Arc<AtomicBool>,
506) {
507    use hound_engine::types::VideoCodec;
508
509    let mut decoder: Option<VideoDecoder> = None;
510    let mut decoder_codec: Option<WireCodec> = None;
511    let mut retry = DecoderRetry::default();
512    let mut decode_failures = DecodeFailures::default();
513    // The current decode path + codec (captured at build) and a ~5 s throughput
514    // window, so the log shows decoded geometry + ACTUAL decode fps + mean decode
515    // cost — a crawling software decode is visible at a glance.
516    let mut current_backend: &'static str = "?";
517    let mut current_codec = VideoCodec::H264;
518    let mut win_start: Option<std::time::Instant> = None;
519    let mut win_frames: u64 = 0;
520    let mut win_decode_us: u64 = 0;
521
522    // Playout stage: decoded frames are scheduled a shallow `VIDEO_PLAYOUT_DELAY_US`
523    // ahead on the sender's capture timeline and released newest-due-first. `clock`
524    // is this thread's local timeline; the buffer maps the sender's stamps onto it
525    // on first push. Recovery is untouched — this sits entirely BELOW decode, so
526    // the pull half's keyframe gate, PLI throttle, stall and resync paths never see
527    // it, and the channel is drained exactly as fast as before.
528    let clock = CaptureClock::start();
529    let mut playout: hound_media::PlayoutBuffer<PlayoutVideo> =
530        hound_media::PlayoutBuffer::with_capacity(VIDEO_PLAYOUT_DELAY_US, VIDEO_PLAYOUT_MAX_FRAMES);
531    let mut stamps = PlayoutStamps::default();
532    let mut reported_drops = 0u64;
533
534    loop {
535        // Wait for the next encoded frame — but never past the moment a buffered
536        // frame comes due, so a sparse producer (a static share re-emitting at ~1 fps)
537        // is still released at the target instead of being held for its successor.
538        let arrived = match playout.next_due_us() {
539            Some(due) => {
540                let wait = Duration::from_micros(due.saturating_sub(clock.now()).max(0) as u64);
541                match frame_rx.recv_timeout(wait) {
542                    Ok(f) => Some(f),
543                    Err(std::sync::mpsc::RecvTimeoutError::Timeout) => None,
544                    Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => break,
545                }
546            }
547            None => match frame_rx.recv() {
548                Ok(f) => Some(f),
549                Err(_) => break,
550            },
551        };
552        if stop.load(Ordering::Relaxed) {
553            break;
554        }
555        // Release BEFORE decoding what just arrived: that frame belongs to a later
556        // playout slot, so its decode must not delay a frame that is due now. A
557        // burst collapses to the freshest — an overdrawn intermediate is pure judder.
558        let now_us = clock.now();
559        if let Some(item) = playout.pop_latest_due(now_us) {
560            peer_stats.video_dwell(vkind, now_us - item.queued_us);
561            push_decoded(sink.as_ref(), &key, item.frame);
562        }
563        // Playout-drop total: superseded (the collapse policy above), plus the
564        // buffer's own late/overflow drops. A gauge — only written when it moves.
565        let dropped = playout.superseded_drops() + playout.late_drops() + playout.overflow_drops();
566        if dropped != reported_drops {
567            reported_drops = dropped;
568            peer_stats.set_video_playout_dropped_total(vkind, dropped);
569        }
570        let Some(frame) = arrived else {
571            continue; // woke only to release
572        };
573        // (Re)build the decoder if the sender's codec changed.
574        if decoder_codec != Some(frame.codec) {
575            let now = std::time::Instant::now();
576            if retry.blocked(frame.codec, now) {
577                // Still inside the retry window for this codec's last failure — skip
578                // the expensive platform-decoder construction and stay silent (see
579                // `RETRY_AFTER`).
580                tracing::trace!(codec = ?frame.codec, "video: decoder build suppressed (retry window)");
581                continue;
582            }
583            let vc = match frame.codec {
584                WireCodec::H264 => VideoCodec::H264,
585                WireCodec::H265 => VideoCodec::H265,
586                WireCodec::Av1 => VideoCodec::Av1,
587                WireCodec::Opus => continue, // not a video codec; ignore
588            };
589            match Engine::new().video_decoder(vc) {
590                Ok(d) => {
591                    retry.clear();
592                    // Surface the actual decode path + codec. A software fallback
593                    // (e.g. AV1 with no HW block) is CPU-heavy and can starve the
594                    // audio fill path, so make HW-vs-SW visible in the call log.
595                    current_backend = d.backend();
596                    current_codec = vc;
597                    win_start = None;
598                    win_frames = 0;
599                    win_decode_us = 0;
600                    tracing::info!(
601                        target: "video_decode",
602                        key = %key,
603                        codec = ?vc,
604                        backend = current_backend,
605                        "video: decode path selected"
606                    );
607                    decoder = Some(d);
608                    decoder_codec = Some(frame.codec);
609                    // Surface the live decode codec in the RTC-debug panel (also
610                    // tracks a mid-stream sender codec switch), and count the build:
611                    // repeated builds inside one window are churn, and each one costs
612                    // the stream its reference chain (the video report reads this).
613                    peer_stats.set_inbound_video_codec(vkind, video_codec_to_ctl(vc).label());
614                    peer_stats.video_decoder_built(vkind);
615                }
616                // A decoder that will not BUILD is, today, a silently black tile: the
617                // loop logs once and re-attempts every `RETRY_AFTER` for as long as
618                // the subscription lives, and nothing reaches the UI — no event, no
619                // placeholder, no "this stream can't be played here". Deliberate for
620                // the codec case it was written for (an unsupported wire codec is the
621                // sender's problem to fix and the retry is how it heals), and a known
622                // gap for the RESOURCE case: on Apple this is also where decode-session
623                // exhaustion lands, because VideoToolbox has no software fallback
624                // behind it (`hound_engine`'s `platform::HW_DECODE_SESSIONS`). The
625                // protection for that is upstream and preventive — `DecodeBudget` caps
626                // how many sessions this receiver ever asks for — not a recovery here.
627                // Two fallbacks that would make it a recovery are deferred, not
628                // rejected: openh264 SW H.264 on Apple, and dav1d for AV1.
629                Err(e) => {
630                    match retry.on_failure(frame.codec, now) {
631                        RetryLog::WarnFirstFailure => {
632                            tracing::warn!(%e, codec = ?frame.codec, "video: failed to build decoder");
633                        }
634                        RetryLog::Suppressed => {
635                            tracing::trace!(%e, codec = ?frame.codec, "video: decoder build still failing");
636                        }
637                    }
638                    continue;
639                }
640            }
641        }
642        let Some(dec) = decoder.as_mut() else {
643            continue;
644        };
645        let decode_start = std::time::Instant::now();
646        match dec.decode(&frame.data) {
647            Ok(Some(raw)) => {
648                decode_failures.on_success();
649                let decode_us = decode_start.elapsed().as_micros().min(i64::MAX as u128) as i64;
650                let (w, h) = (raw.width(), raw.height());
651                // Every decoder backend emits an OWNED I420 `RawFrame` — each `decode`
652                // call allocates a fresh conversion buffer (vt_decoder / vaapi / sw all
653                // output I420, asserted in their tests). So the frame is MOVED from here
654                // to the sink (through the playout buffer, which only ever reads its
655                // timestamp): the decoder's single conversion alloc stays the only one
656                // on this path, and a GPU-resident frame keeps its zero-copy surface.
657                match stamps.observe(frame.timestamp_us) {
658                    // Schedule it — the pump at the top of the loop releases it at its
659                    // playout time.
660                    PlayoutMode::Buffered => {
661                        let queued_us = clock.now();
662                        playout.push(
663                            queued_us,
664                            PlayoutVideo {
665                                ts_us: frame.timestamp_us,
666                                queued_us,
667                                frame: raw,
668                            },
669                        );
670                    }
671                    // Still probing: render on arrival so the probe itself costs nothing.
672                    PlayoutMode::Probing => push_decoded(sink.as_ref(), &key, raw),
673                    // Latched off (possibly just now): hand over anything still queued —
674                    // newest wins — so no frame is stranded, then render on arrival.
675                    PlayoutMode::PassThrough => {
676                        if let Some(item) = playout.pop_latest_due(i64::MAX) {
677                            push_decoded(sink.as_ref(), &key, item.frame);
678                        }
679                        push_decoded(sink.as_ref(), &key, raw);
680                    }
681                }
682                // RTC-debug: one decoded frame (geometry + decode-time tail).
683                peer_stats.video_decoded(vkind, w, h, decode_us);
684
685                // ~5 s decode-throughput sample: decoded geometry + ACTUAL fps +
686                // mean decode cost on the current path (HW/SW visible above).
687                let now = std::time::Instant::now();
688                let start = *win_start.get_or_insert(now);
689                win_frames += 1;
690                win_decode_us += decode_us.max(0) as u64;
691                let elapsed = now.duration_since(start);
692                if elapsed >= std::time::Duration::from_secs(5) {
693                    let secs = elapsed.as_secs_f64().max(0.001);
694                    let fps = win_frames as f64 / secs;
695                    let mean_ms = (win_decode_us as f64 / win_frames as f64) / 1000.0;
696                    tracing::info!(
697                        target: "video_decode",
698                        key = %key,
699                        codec = ?current_codec,
700                        backend = current_backend,
701                        width = w,
702                        height = h,
703                        fps = format!("{fps:.1}"),
704                        decode_ms = format!("{mean_ms:.1}"),
705                        "video: decode throughput"
706                    );
707                    win_start = Some(now);
708                    win_frames = 0;
709                    win_decode_us = 0;
710                }
711            }
712            Ok(None) => {} // decoder buffering; keep feeding
713            Err(e) => {
714                // Ask the pull half for a fresh reference — this thread has no
715                // control-plane handle, and nothing else notices a decode failure
716                // (frames keep arriving, so no stall; failing is fast, so no
717                // backpressure). Decoder teardown is paced, not per-frame — see
718                // `DecodeFailures`.
719                let act = decode_failures.on_failure(std::time::Instant::now());
720                if act.log {
721                    tracing::warn!(
722                        %e,
723                        key = %key,
724                        codec = ?current_codec,
725                        backend = current_backend,
726                        rebuild = act.rebuild,
727                        "video: decode failed — requesting a keyframe"
728                    );
729                }
730                if act.rebuild {
731                    decoder = None;
732                    decoder_codec = None;
733                }
734                resync_needed.store(true, Ordering::Relaxed);
735            }
736        }
737    }
738}

742// Both guard the same storm described on `RETRY_AFTER` — the wire codec byte and
