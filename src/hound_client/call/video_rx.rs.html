1//! Inbound video: the async pull half (receive-side keyframe gate; the nag task
2//! lives with the subscribe loop) feeds a blocking decode thread that owns the
3//! `!Send` `VideoDecoder` and rebuilds it whenever the sender's per-packet codec
4//! byte changes — the sender's choice drives decode. Decoded frames pass through a
5//! shallow playout stage ([`VIDEO_PLAYOUT_DELAY_US`]) that replays the sender's
6//! frame spacing, then reach the sink surface-first (zero-copy GPU handle) with
7//! materialized I420 as the universal fallback; [`push_preview`] is the
8//! borrow-based capture-side twin the publish loops use for self-preview.
9//!
10//! The playout stage is presentation pacing only — it does NOT anchor video against
11//! the audio plane's own (adaptive, deeper) jitter depth, so the two planes are
12//! still independently scheduled.
13

16/// Push one CAPTURED frame to the render sink for a self-preview tile, preferring
17/// the ZERO-COPY surface path: a surface-backed frame (SCK / AVFoundation NV12
18/// IOSurface) is handed to the renderer to sample directly — no NV12→I420 CPU
19/// conversion, no ~1.4–3 MB per-frame alloc+copy, no re-upload (~90 MB/s of
20/// conversion at a 1080p 30 fps share). Falls back to materialized I420
21/// for non-surface frames (scaled camera frames, other
22/// platforms) or a sink without a surface renderer. Borrows the frame — the
23/// caller still owns it for encode. (The REMOTE decode path keeps its own
24/// ownership-optimized version of this pattern: it consumes the frame, so its
25/// no-surface case can MOVE the bytes instead of copying.)
26pub(super) fn push_preview(sink: &dyn VideoSink, key: &str, frame: &hound_engine::types::RawFrame) {
27    let zero_copy =
28        RenderSurfaceHandle::for_frame(frame).is_some_and(|handle| sink.push_surface(key, handle));
29    if !zero_copy && let Some(i420) = frame.as_i420() {
30        sink.push_frame(key, i420.width(), i420.height(), i420.into_owned().data);
31    }
32}

34/// How long the video pull tolerates delivering NOTHING before treating the
35/// subscription as stalled. Above the static-share re-emit cadence
36/// ([`SCREEN_STATIC_REFRESH`], 1 s) so a healthy static share never trips it; a
37/// mid-GOP loss stall or a silent producer does within seconds instead of a
38/// retransmit tail.
39const VIDEO_STALL: Duration = Duration::from_secs(3);
40
41/// How long a fresh video subscription may sit with NO keyframe before the
42/// pipeline ends itself so the reconcile re-subscribes (restart-ladder paced).
43/// Past the nag window (5 × 2 s): a producer that ignored every re-request is
44/// wedged or its broadcast is a zombie from a same-path replacement, and only a
45/// fresh subscribe — a new announce cursor + control-plane `Subscribe` — heals
46/// those; waiting on the 60 s safety-net IDR is a frozen tile.
47const FIRST_KEYFRAME_DEADLINE: Duration = Duration::from_secs(12);

49/// Video decode pipeline. The MoQ subscriber pull is async (must run on the
50/// runtime), but the engine `VideoDecoder` (VideoToolbox/dav1d) is `!Send`, so it
51/// CANNOT live across an `.await`. The async half therefore only PULLS frames —
52/// applying the receive-side keyframe gate so we never feed a delta before the
53/// first keyframe — and forwards them to a dedicated blocking decode thread that
54/// owns the `!Send` decoder and pushes decoded I420 to the sink keyed by peer.
55///
56/// The pull is stall-bounded ([`VIDEO_STALL`]): a stuck group is abandoned and a
57/// throttled `RequestKeyframe` sent, so the producer rolls a fresh group and the
58/// tile resumes at live content instead of freezing behind a retransmit tail and
59/// then fast-forwarding. A subscription that never yields its first keyframe
60/// ends itself at [`FIRST_KEYFRAME_DEADLINE`] (the reap → paced-resubscribe path).
61#[allow(clippy::too_many_arguments)]
62pub(super) async fn video_pipeline(
63    mut subscriber: MoqTrackSubscriber,
64    peer: PeerId,
65    source: Source,
66    sink: Arc<dyn VideoSink>,
67    stop: Arc<AtomicBool>,
68    peer_stats: Arc<crate::stats::PeerStats>,
69    got_keyframe: Arc<AtomicBool>,
70    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
71    track_id: u32,
72    downlink_cell: Arc<DownlinkCell>,
73    source_bps: u64,
74) {
75    use hound_transport::VideoPull;
76
77    let key = match source {
78        Source::Screen => screen_key(peer),
79        _ => cam_key(peer),
80    };
81    // Camera vs screen feed SEPARATE inbound-video stat buckets (so a peer sharing
82    // both does not report a summed fps/bitrate).
83    let vkind = inbound_video_kind(source);
84
85    // Drop-on-full channel from the async pull half to the blocking decode thread.
86    let (frame_tx, frame_rx) = std::sync::mpsc::sync_channel::<WireFrame>(8);
87    let decode_stop = stop.clone();
88    let decode_stats = peer_stats.clone();
89    // Decode → pull back-channel. The decode thread owns the `!Send` decoder and
90    // has no control-plane handle, so a decode failure (stale reference, wedged
91    // session) is raised here; the pull half — the only half holding the keyframe
92    // gate and the PLI throttle — acts on it. Without this the tile freezes until
93    // the sender's next NATURAL IDR (4 s camera, up to 60 s screenshare) because
94    // neither pull-side recovery fires: frames keep arriving (no stall) and a
95    // fast-failing decode never backs the channel up.
96    let resync_needed = Arc::new(AtomicBool::new(false));
97    let decode_resync = resync_needed.clone();
98    let decode_thread = std::thread::Builder::new()
99        .name(format!("call-video-decode:{}", peer.0))
100        .spawn(move || {
101            video_decode_loop(
102                frame_rx,
103                key,
104                vkind,
105                sink,
106                decode_stop,
107                decode_stats,
108                decode_resync,
109            )
110        })
111        .ok();
112
113    let started = Instant::now();
114    let mut pli = PliThrottle::default();
115    // One throttled recovery keyframe request — stall, backpressure drop and
116    // decode resync all need exactly this.
117    let ask_keyframe = |pli: &mut PliThrottle| {
118        if pli.fire(Instant::now()) {
119            let _ = control_outbound.send(ClientMessage::RequestKeyframe {
120                track: hound_control::TrackId(track_id),
121            });
122            peer_stats.video_keyframe_requested(vkind);
123        }
124    };
125    // The bandwidth half of demand-driven simulcast: folds each arriving frame's
126    // wire `frame_seq` into a windowed delivery ratio against `source_bps`, with
127    // the source/4 · source/2 hysteresis (see `hound_media::downlink`'s docs). A
128    // fresh instance per pipeline — the reset-on-restart contract falls out of
129    // that for free, matching the fresh `downlink_cell` this pipeline was given.
130    let mut downlink_demand =
131        hound_media::DownlinkDemand::new(hound_media::DownlinkConfig::default());
132
133    let mut awaiting_keyframe = true;
134    // A decode-triggered resync is outstanding (distinct from the pre-first-keyframe
135    // wait, which the subscribe-time nag owns): keep re-asking while deltas arrive.
136    let mut resync_pending = false;
137    while !stop.load(Ordering::Relaxed) {
138        let frame = match subscriber.next_frame_stalled(VIDEO_STALL).await {
139            Ok(VideoPull::Frame(f)) => f,
140            Ok(VideoPull::Stalled) => {
141                // A total downlink stall is exactly when this receiver most needs the
142                // low layer, and `observe_frame` can only decide from an arrival — so
143                // drive the decider from the stall tick too (same window boundary, so
144                // it can neither shorten a window nor double-evaluate one).
145                let now_us = started.elapsed().as_micros().min(i64::MAX as u128) as i64;
146                downlink_cell.store(downlink_demand.on_tick(now_us, source_bps));
147                if !got_keyframe.load(Ordering::Relaxed)
148                    && started.elapsed() >= FIRST_KEYFRAME_DEADLINE
149                {
150                    tracing::warn!(
151                        peer = peer.0,
152                        track = track_id,
153                        "video: no keyframe within the deadline — ending pipeline for a fresh subscribe"
154                    );
155                    break;
156                }
157                // Whatever the abandoned group still owed is gone — re-latch the
158                // gate and ask the producer to roll a fresh, decodable group.
159                awaiting_keyframe = true;
160                ask_keyframe(&mut pli);
161                continue;
162            }
163            Ok(VideoPull::End) | Err(_) => break,
164        };
165        // RTC-debug: one inbound encoded AU received for this peer's camera/screen video.
166        peer_stats.video_received(vkind, frame.data.len());
167        // Downlink demand: fold this arrival's wire `frame_seq` into the windowed
168        // delivery-ratio decider, independent of the keyframe gate below — a frame
169        // that arrives but gets dropped pre-decode (awaiting a keyframe) still
170        // reached us over the wire, so it is NOT downlink loss. `now_us` mirrors
171        // the elapsed-since-`started` convention the decode-throughput log below
172        // already uses for this same pipeline.
173        let now_us = started.elapsed().as_micros().min(i64::MAX as u128) as i64;
174        let req = downlink_demand.observe_frame(frame.frame_seq, now_us, source_bps);
175        downlink_cell.store(req);
176        // RTC-debug: the same `frame_seq` continuity the decider folds is the only
177        // receiver-side loss signal this path has — surface its running total.
178        peer_stats.set_video_frames_lost(vkind, downlink_demand.lost_estimate());
179        // Decode failed on the other half: its reference is gone (or its session is
180        // wedged), so every further delta is undecodable. Re-latch the gate — cheap
181        // delta skips instead of feeding a decoder that can't use them — and ask for
182        // a fresh reference.
183        if resync_needed.swap(false, Ordering::Relaxed) {
184            awaiting_keyframe = true;
185            resync_pending = true;
186            ask_keyframe(&mut pli);
187        }
188        // Keyframe gate: drop deltas until the first keyframe so the decode thread
189        // never decodes against a missing reference. Passing the gate also stops
190        // the awaiting-keyframe re-request nag (see `start_pipeline`).
191        if awaiting_keyframe {
192            if !frame.keyframe {
193                // Keep nagging while a resync is outstanding: a single request can
194                // land inside the producer's own keyframe throttle and be coalesced
195                // away, and nothing else re-fires once the stream is flowing.
196                if resync_pending {
197                    ask_keyframe(&mut pli);
198                }
199                continue;
200            }
201            awaiting_keyframe = false;
202            resync_pending = false;
203            got_keyframe.store(true, Ordering::Relaxed);
204        }
205        // Backpressure → drop. A dropped frame breaks decode continuity for the
206        // rest of its GOP, so re-latch the gate (cheap delta skips instead of
207        // corrupt decodes) and request a keyframe to resync.
208        if frame_tx.try_send(frame).is_err() {
209            // If the decode thread is gone, end the pull.
210            if decode_thread.as_ref().is_none_or(|h| h.is_finished()) {
211                break;
212            }
213            awaiting_keyframe = true;
214            ask_keyframe(&mut pli);
215        }
216    }
217    drop(frame_tx); // ends the decode loop
218    if let Some(h) = decode_thread {
219        let _ = h.join();
220    }
221}

223/// How long the decode loop withholds retrying platform-decoder construction for a
224/// codec that just failed to build. Without this, the per-packet wire codec byte is
225/// **sender-controlled** — a peer sending AV1 to a box with no AV1 decode makes every
226/// single frame re-attempt `Engine::new().video_decoder(..)` (a VideoToolbox/VAAPI
227/// session setup) and `warn!`, forever. That is a remotely triggerable retry storm
228/// (CPU burn + log spam) for the lifetime of the subscription.
229const RETRY_AFTER: Duration = Duration::from_secs(5);
230
231/// Whether a just-recorded decoder-construction failure should be logged. Only the
232/// FIRST failure of a streak is worth a `warn!`; every suppressed retry inside
233/// [`RETRY_AFTER`] would just repeat it forever.
234#[derive(Debug, Clone, Copy, PartialEq, Eq)]
235enum RetryLog {
236    /// First failure for this codec (or the first after a different codec/success) —
237    /// log it.
238    WarnFirstFailure,
239    /// A retry still inside the window — stay quiet.
240    Suppressed,
241}
242
243/// Pure decoder-(re)build retry policy (see [`RETRY_AFTER`]). Consulted before every
244/// `Engine::new().video_decoder(..)` call for a codec whose current decoder doesn't
245/// match. Takes `now` as a parameter so it is unit-testable without a live clock.
246#[derive(Default)]
247struct DecoderRetry {
248    /// The codec that last failed to construct, and when.
249    failed: Option<(WireCodec, Instant)>,
250}
251
252impl DecoderRetry {
253    /// Should a construction attempt for `codec` be skipped right now? Only true
254    /// within [`RETRY_AFTER`] of a failure for the SAME codec — a different incoming
255    /// codec byte is unrelated and always gets its own first attempt immediately.
256    fn blocked(&self, codec: WireCodec, now: Instant) -> bool {
257        matches!(self.failed, Some((c, at)) if c == codec && now.duration_since(at) < RETRY_AFTER)
258    }
259
260    /// Record a construction failure for `codec` at `now`. Returns whether to log it
261    /// — [`RetryLog::WarnFirstFailure`] only the first time this codec fails in a row.
262    fn on_failure(&mut self, codec: WireCodec, now: Instant) -> RetryLog {
263        let first = !matches!(self.failed, Some((c, _)) if c == codec);
264        self.failed = Some((codec, now));
265        if first {
266            RetryLog::WarnFirstFailure
267        } else {
268            RetryLog::Suppressed
269        }
270    }
271
272    /// Clear the block — call on a successful build.
273    fn clear(&mut self) {
274        self.failed = None;
275    }
276}

278/// What the decode loop should do about one `decode` failure — see [`DecodeFailures`].
279#[derive(Debug, Clone, Copy, PartialEq, Eq)]
280struct DecodeAction {
281    /// `warn!` this one: the first of a streak, then at most one per [`RETRY_AFTER`].
282    log: bool,
283    /// Tear the decoder down so the next frame rebuilds it.
284    rebuild: bool,
285}
286
287/// Pure consecutive-decode-failure policy. A decode error is NOT a construction
288/// error: the usual cause is a missing reference, which the decoders heal from on
289/// the next keyframe by themselves (VideoToolbox rebuilds its session on new
290/// SPS/PPS, openh264 absorbs in-band parameter sets) — so the loop KEEPS the
291/// decoder and only signals the pull half to request one. Tearing down per failure
292/// instead re-ran `Engine::new().video_decoder(..)` (a full VAAPI/openh264 setup)
293/// and an `info!` at the arrival frame rate — the very storm [`RETRY_AFTER`] exists
294/// to bound, reached through the decode door.
295///
296/// Only a streak that OUTLIVES a full [`RETRY_AFTER`] window is treated as a wedged
297/// decoder (a VT session invalidated across sleep keeps identical SPS/PPS, so
298/// nothing triggers its own rebuild) and torn down — at most once per window, so a
299/// sender-controlled corrupt stream still can't drive a rebuild storm. Takes `now`
300/// as a parameter so it is unit-testable without a live clock.
301#[derive(Default)]
302struct DecodeFailures {
303    /// Start of the current failure streak (`None` between streaks).
304    since: Option<Instant>,
305    /// Last failure logged, and last teardown — two independent [`RETRY_AFTER`] paces.
306    logged: Option<Instant>,
307    torn_down: Option<Instant>,
308}
309
310impl DecodeFailures {
311    /// Record a decode failure at `now` and decide what the loop does about it.
312    fn on_failure(&mut self, now: Instant) -> DecodeAction {
313        let started = *self.since.get_or_insert(now);
314        let log = self
315            .logged
316            .is_none_or(|t| now.duration_since(t) >= RETRY_AFTER);
317        if log {
318            self.logged = Some(now);
319        }
320        let rebuild = now.duration_since(started) >= RETRY_AFTER
321            && self
322                .torn_down
323                .is_none_or(|t| now.duration_since(t) >= RETRY_AFTER);
324        if rebuild {
325            self.torn_down = Some(now);
326        }
327        DecodeAction { log, rebuild }
328    }
329
330    /// A frame decoded — the streak (if any) is over.
331    fn on_success(&mut self) {
332        *self = Self::default();
333    }
334}

338/// How long a decoded frame waits in the playout buffer before it reaches the
339/// renderer. Deliberately SHALLOW — between one and two 30 fps frame intervals —
340/// because this stage buys *presentation pacing*, not loss recovery: frames arrive
341/// and decode in bursts (a group flush, a retransmit tail, a slow software decode),
342/// and rendering them the instant they pop out plays a nominally-30 fps tile as
343/// judder. Scheduling each frame against the sender's capture clock replays the
344/// sender's spacing instead.
345///
346/// It is NOT lip-sync. Anchoring video against the audio plane's own (adaptive,
347/// far deeper) jitter depth is a cross-plane problem this stage deliberately does
348/// not attempt — see the module docs.
349const VIDEO_PLAYOUT_DELAY_US: i64 = 40_000;
350
351/// Bound on decoded frames held for playout. A decoded frame can hold a GPU
352/// surface lease (a VideoToolbox IOSurface, a D3D11 texture-pool slot), so the
353/// depth must stay well inside a decoder's pool: 8 is several times the
354/// steady-state occupancy (1–2 frames at the shallow target) and only reachable
355/// when the pump is already hundreds of ms behind, where drop-oldest is the right
356/// answer for video anyway.
357const VIDEO_PLAYOUT_MAX_FRAMES: usize = 8;
358
359/// Consecutive strictly-advancing wire timestamps that turn playout scheduling on
360/// (or, in the other direction, non-advancing ones that latch it off) — see
361/// [`PlayoutMode`]. Three frames is ~100 ms of stream: long enough not to judge a
362/// sender on one duplicate stamp, short enough that pacing engages immediately.
363const VIDEO_STAMP_RUN: u32 = 3;
364
365/// Whether this sender's wire timestamps can schedule playout at all.
366#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
367enum PlayoutMode {
368    /// Not enough evidence yet — render on arrival, so a stream is never delayed
369    /// by the probe itself.
370    #[default]
371    Probing,
372    /// Timestamps advance: schedule through the playout buffer.
373    Buffered,
374    /// Latched off. Some encoders stamp every chunk `TimestampUs::ZERO` (the MFT
375    /// and rav1e paths do), and a buffer keyed on a constant timestamp drops every
376    /// frame after the first as late — a frozen tile. Render on arrival, forever.
377    PassThrough,
378}
379
380/// Pure wire-timestamp usability policy behind [`PlayoutMode`]: it only ever reads
381/// the timestamps a sender puts on the wire, so it is unit-testable without a
382/// decoder, a clock, or a sink.
383#[derive(Debug, Default)]
384struct PlayoutStamps {
385    mode: PlayoutMode,
386    last_ts: Option<i64>,
387    /// Current run of advancing / non-advancing timestamps (each resets the other).
388    advancing: u32,
389    stale: u32,
390}
391
392impl PlayoutStamps {
393    /// Fold one arriving frame's wire timestamp in and report how its decoded
394    /// output should be delivered. [`PlayoutMode::PassThrough`] is terminal — a
395    /// sender that cannot stamp will not start.
396    fn observe(&mut self, ts_us: i64) -> PlayoutMode {
397        if self.mode == PlayoutMode::PassThrough {
398            return self.mode;
399        }
400        match self.last_ts {
401            Some(last) if ts_us > last => {
402                self.stale = 0;
403                self.advancing = self.advancing.saturating_add(1);
404                if self.advancing >= VIDEO_STAMP_RUN {
405                    self.mode = PlayoutMode::Buffered;
406                }
407            }
408            Some(_) => {
409                self.advancing = 0;
410                self.stale = self.stale.saturating_add(1);
411                if self.stale >= VIDEO_STAMP_RUN {
412                    self.mode = PlayoutMode::PassThrough;
413                }
414            }
415            None => {}
416        }
417        // A non-advancing stamp must not move the baseline backwards (a stamp of 0
418        // after a real one would otherwise re-arm the advancing run every frame).
419        self.last_ts = Some(self.last_ts.map_or(ts_us, |last| last.max(ts_us)));
420        self.mode
421    }
422}

424/// A decoded frame waiting for its playout slot. Carries the SENDER's capture-clock
425/// timestamp (what the buffer schedules against — the decoders do not thread a
426/// timestamp onto their output, and the wire frame is the only place it exists) and
427/// the local time it was queued, for the dwell stat. The frame itself is moved in
428/// and out, never copied.
429///
430/// Pairing the decoder's output with the timestamp of the frame just fed assumes
431/// one-in-one-out decode (true for every backend here — no B-frames on the wire). A
432/// decoder that reordered would shift every stamp by a CONSTANT, which the buffer
433/// absorbs in its anchor; only spacing matters to this stage.
434struct PlayoutVideo {
435    ts_us: i64,
436    queued_us: i64,
437    frame: hound_engine::types::RawFrame,
438}
439
440impl hound_media::Timestamped for PlayoutVideo {
441    fn timestamp_us(&self) -> i64 {
442        self.ts_us
443    }
444}

446/// Hand one DECODED frame to the render sink, surface-first (zero-copy GPU handle)
447/// with materialized I420 as the universal fallback. Consumes the frame, so the
448/// no-surface case MOVES the decoder's conversion buffer instead of copying ~1.4 MB
449/// per frame per peer ([`push_preview`] is the borrow-based capture-side twin).
450fn push_decoded(sink: &dyn VideoSink, key: &str, raw: hound_engine::types::RawFrame) {
451    if RenderSurfaceHandle::for_frame(&raw).is_some_and(|handle| sink.push_surface(key, handle)) {
452        return;
453    }
454    let (w, h) = (raw.width(), raw.height());
455    let data = if raw.surface.is_none() && raw.format == hound_engine::types::PixelFormat::I420 {
456        raw.data
457    } else {
458        match raw.as_i420() {
459            Some(c) => c.into_owned().data,
460            None => return,
461        }
462    };
463    sink.push_frame(key, w, h, data);
464}

466/// The blocking video decode loop: owns the `!Send` `VideoDecoder`, rebuilds it
467/// when the sender's codec changes (the sender's choice drives decode), and pushes
468/// each decoded frame to the sink through a shallow playout stage
469/// ([`VIDEO_PLAYOUT_DELAY_US`]) that replays the sender's frame spacing. A decode
470/// failure raises `resync_needed` — this thread's only route to the control plane
471/// (see [`DecodeFailures`]).
472fn video_decode_loop(
473    frame_rx: std::sync::mpsc::Receiver<WireFrame>,
474    key: String,
475    vkind: InboundVideoKind,
476    sink: Arc<dyn VideoSink>,
477    stop: Arc<AtomicBool>,
478    peer_stats: Arc<crate::stats::PeerStats>,
479    resync_needed: Arc<AtomicBool>,
480) {
481    use hound_engine::types::VideoCodec;
482
483    let mut decoder: Option<VideoDecoder> = None;
484    let mut decoder_codec: Option<WireCodec> = None;
485    let mut retry = DecoderRetry::default();
486    let mut decode_failures = DecodeFailures::default();
487    // The current decode path + codec (captured at build) and a ~5 s throughput
488    // window, so the log shows decoded geometry + ACTUAL decode fps + mean decode
489    // cost — a crawling software decode is visible at a glance.
490    let mut current_backend: &'static str = "?";
491    let mut current_codec = VideoCodec::H264;
492    let mut win_start: Option<std::time::Instant> = None;
493    let mut win_frames: u64 = 0;
494    let mut win_decode_us: u64 = 0;
495
496    // Playout stage: decoded frames are scheduled a shallow `VIDEO_PLAYOUT_DELAY_US`
497    // ahead on the sender's capture timeline and released newest-due-first. `clock`
498    // is this thread's local timeline; the buffer maps the sender's stamps onto it
499    // on first push. Recovery is untouched — this sits entirely BELOW decode, so
500    // the pull half's keyframe gate, PLI throttle, stall and resync paths never see
501    // it, and the channel is drained exactly as fast as before.
502    let clock = CaptureClock::start();
503    let mut playout: hound_media::PlayoutBuffer<PlayoutVideo> =
504        hound_media::PlayoutBuffer::with_capacity(VIDEO_PLAYOUT_DELAY_US, VIDEO_PLAYOUT_MAX_FRAMES);
505    let mut stamps = PlayoutStamps::default();
506    let mut reported_drops = 0u64;
507
508    loop {
509        // Wait for the next encoded frame — but never past the moment a buffered
510        // frame comes due, so a sparse producer (a static share re-emitting at ~1 fps)
511        // is still released at the target instead of being held for its successor.
512        let arrived = match playout.next_due_us() {
513            Some(due) => {
514                let wait = Duration::from_micros(due.saturating_sub(clock.now()).max(0) as u64);
515                match frame_rx.recv_timeout(wait) {
516                    Ok(f) => Some(f),
517                    Err(std::sync::mpsc::RecvTimeoutError::Timeout) => None,
518                    Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => break,
519                }
520            }
521            None => match frame_rx.recv() {
522                Ok(f) => Some(f),
523                Err(_) => break,
524            },
525        };
526        if stop.load(Ordering::Relaxed) {
527            break;
528        }
529        // Release BEFORE decoding what just arrived: that frame belongs to a later
530        // playout slot, so its decode must not delay a frame that is due now. A
531        // burst collapses to the freshest — an overdrawn intermediate is pure judder.
532        let now_us = clock.now();
533        if let Some(item) = playout.pop_latest_due(now_us) {
534            peer_stats.video_dwell(vkind, now_us - item.queued_us);
535            push_decoded(sink.as_ref(), &key, item.frame);
536        }
537        // Playout-drop total: superseded (the collapse policy above), plus the
538        // buffer's own late/overflow drops. A gauge — only written when it moves.
539        let dropped = playout.superseded_drops() + playout.late_drops() + playout.overflow_drops();
540        if dropped != reported_drops {
541            reported_drops = dropped;
542            peer_stats.set_video_playout_dropped_total(vkind, dropped);
543        }
544        let Some(frame) = arrived else {
545            continue; // woke only to release
546        };
547        // (Re)build the decoder if the sender's codec changed.
548        if decoder_codec != Some(frame.codec) {
549            let now = std::time::Instant::now();
550            if retry.blocked(frame.codec, now) {
551                // Still inside the retry window for this codec's last failure — skip
552                // the expensive platform-decoder construction and stay silent (see
553                // `RETRY_AFTER`).
554                tracing::trace!(codec = ?frame.codec, "video: decoder build suppressed (retry window)");
555                continue;
556            }
557            let vc = match frame.codec {
558                WireCodec::H264 => VideoCodec::H264,
559                WireCodec::H265 => VideoCodec::H265,
560                WireCodec::Av1 => VideoCodec::Av1,
561                WireCodec::Opus => continue, // not a video codec; ignore
562            };
563            match Engine::new().video_decoder(vc) {
564                Ok(d) => {
565                    retry.clear();
566                    // Surface the actual decode path + codec. A software fallback
567                    // (e.g. AV1 with no HW block) is CPU-heavy and can starve the
568                    // audio fill path, so make HW-vs-SW visible in the call log.
569                    current_backend = d.backend();
570                    current_codec = vc;
571                    win_start = None;
572                    win_frames = 0;
573                    win_decode_us = 0;
574                    tracing::info!(
575                        target: "video_decode",
576                        key = %key,
577                        codec = ?vc,
578                        backend = current_backend,
579                        "video: decode path selected"
580                    );
581                    decoder = Some(d);
582                    decoder_codec = Some(frame.codec);
583                    // Surface the live decode codec in the RTC-debug panel (also
584                    // tracks a mid-stream sender codec switch).
585                    peer_stats.set_inbound_video_codec(vkind, video_codec_to_ctl(vc).label());
586                }
587                Err(e) => {
588                    match retry.on_failure(frame.codec, now) {
589                        RetryLog::WarnFirstFailure => {
590                            tracing::warn!(%e, codec = ?frame.codec, "video: failed to build decoder");
591                        }
592                        RetryLog::Suppressed => {
593                            tracing::trace!(%e, codec = ?frame.codec, "video: decoder build still failing");
594                        }
595                    }
596                    continue;
597                }
598            }
599        }
600        let Some(dec) = decoder.as_mut() else {
601            continue;
602        };
603        let decode_start = std::time::Instant::now();
604        match dec.decode(&frame.data) {
605            Ok(Some(raw)) => {
606                decode_failures.on_success();
607                let decode_us = decode_start.elapsed().as_micros().min(i64::MAX as u128) as i64;
608                let (w, h) = (raw.width(), raw.height());
609                // Every decoder backend emits an OWNED I420 `RawFrame` — each `decode`
610                // call allocates a fresh conversion buffer (vt_decoder / vaapi / sw all
611                // output I420, asserted in their tests). So the frame is MOVED from here
612                // to the sink (through the playout buffer, which only ever reads its
613                // timestamp): the decoder's single conversion alloc stays the only one
614                // on this path, and a GPU-resident frame keeps its zero-copy surface.
615                match stamps.observe(frame.timestamp_us) {
616                    // Schedule it — the pump at the top of the loop releases it at its
617                    // playout time.
618                    PlayoutMode::Buffered => {
619                        let queued_us = clock.now();
620                        playout.push(
621                            queued_us,
622                            PlayoutVideo {
623                                ts_us: frame.timestamp_us,
624                                queued_us,
625                                frame: raw,
626                            },
627                        );
628                    }
629                    // Still probing: render on arrival so the probe itself costs nothing.
630                    PlayoutMode::Probing => push_decoded(sink.as_ref(), &key, raw),
631                    // Latched off (possibly just now): hand over anything still queued —
632                    // newest wins — so no frame is stranded, then render on arrival.
633                    PlayoutMode::PassThrough => {
634                        if let Some(item) = playout.pop_latest_due(i64::MAX) {
635                            push_decoded(sink.as_ref(), &key, item.frame);
636                        }
637                        push_decoded(sink.as_ref(), &key, raw);
638                    }
639                }
640                // RTC-debug: one decoded frame (geometry + decode-time tail).
641                peer_stats.video_decoded(vkind, w, h, decode_us);
642
643                // ~5 s decode-throughput sample: decoded geometry + ACTUAL fps +
644                // mean decode cost on the current path (HW/SW visible above).
645                let now = std::time::Instant::now();
646                let start = *win_start.get_or_insert(now);
647                win_frames += 1;
648                win_decode_us += decode_us.max(0) as u64;
649                let elapsed = now.duration_since(start);
650                if elapsed >= std::time::Duration::from_secs(5) {
651                    let secs = elapsed.as_secs_f64().max(0.001);
652                    let fps = win_frames as f64 / secs;
653                    let mean_ms = (win_decode_us as f64 / win_frames as f64) / 1000.0;
654                    tracing::info!(
655                        target: "video_decode",
656                        key = %key,
657                        codec = ?current_codec,
658                        backend = current_backend,
659                        width = w,
660                        height = h,
661                        fps = format!("{fps:.1}"),
662                        decode_ms = format!("{mean_ms:.1}"),
663                        "video: decode throughput"
664                    );
665                    win_start = Some(now);
666                    win_frames = 0;
667                    win_decode_us = 0;
668                }
669            }
670            Ok(None) => {} // decoder buffering; keep feeding
671            Err(e) => {
672                // Ask the pull half for a fresh reference — this thread has no
673                // control-plane handle, and nothing else notices a decode failure
674                // (frames keep arriving, so no stall; failing is fast, so no
675                // backpressure). Decoder teardown is paced, not per-frame — see
676                // `DecodeFailures`.
677                let act = decode_failures.on_failure(std::time::Instant::now());
678                if act.log {
679                    tracing::warn!(
680                        %e,
681                        key = %key,
682                        codec = ?current_codec,
683                        backend = current_backend,
684                        rebuild = act.rebuild,
685                        "video: decode failed — requesting a keyframe"
686                    );
687                }
688                if act.rebuild {
689                    decoder = None;
690                    decoder_codec = None;
691                }
692                resync_needed.store(true, Ordering::Relaxed);
693            }
694        }
695    }
696}

700// Both guard the same storm described on `RETRY_AFTER` — the wire codec byte and

702// undecodable stream may make the decode loop retry expensive platform-decoder
703// construction (and warn) on every single frame.
