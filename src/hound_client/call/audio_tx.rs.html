1//! Outbound audio: the mic capture loop + Opus publish, and the screen APP-AUDIO
2//! leg (its `!Send` capture thread + publish task). Encode + publish run on the
3//! RT audio runtime ("audio is sacred": a screenshare encode can't stall the
4//! mic), and audio frames publish via the ordered-substream path (count-rolled
5//! groups) so successive Opus frames ride one in-order QUIC stream — never one
6//! stream per frame.

10// ─── P5: the OS input-volume controller ─────────────────────────────────────────
11//
12// Chrome/Discord don't just apply digital gain — they drive the OS mic slider so
13// a mic sitting at 20% gets *louder at the source*. sonora's AGC2 computes the
14// recommendation (`Apm::recommended_input_volume`, wired into `apm.rs`'s
15// `process_capture` round trip); this section is the platform round trip that
16// acts on it: read the OS level, feed it in, and — gated on the AGC toggle (user
17// decision: coupled, no separate setting), not muted, and rate-limited — write
18// the recommendation back.
19//
20// All HAL/COM I/O (`hound_engine::api::InputVolumeHandle::{get,set}` and the
21// `resolve` that builds one) happens on a dedicated background thread
22// ([`volume_actor_loop`]), never the mic capture thread: [`OsInputVolume`] is the
23// atomic mailbox between them, mirroring how the capture thread already hands
24// PCM to the async publish task over a channel rather than doing its own I/O.
25
26/// Rate limit for OS input-volume WRITES: at most one per this interval even if
27/// the controller's recommendation changes every frame. A real HAL/COM round
28/// trip per 20 ms frame would be needless syscall churn for a value that only
29/// needs to track speech level on a many-seconds timescale.
30const OS_VOLUME_WRITE_INTERVAL: Duration = Duration::from_secs(1);
31
32/// How often the volume actor polls its device-change mailbox (and, with no
33/// pending write, re-reads the OS level to catch an out-of-band change — the
34/// user manually moving the OS mixer slider mid-call).
35const OS_VOLUME_POLL_INTERVAL: Duration = Duration::from_millis(200);
36
37/// Floor between the actor's own periodic re-reads when there is no write to
38/// service. Independent of [`OS_VOLUME_WRITE_INTERVAL`] — this is the READ side
39/// (catching external changes), not the write throttle.
40const OS_VOLUME_PERIODIC_READ_INTERVAL: Duration = Duration::from_secs(1);
41
42/// The write-side decision for the OS input-volume round trip: given the
43/// controller's live recommendation and the gate state, decide whether to push
44/// a write now. Pure (no I/O, no `Apm`, no platform handle) so the AGC-off/
45/// mute/rate-limit/unavailable-provider rules are unit-testable without real
46/// hardware. Owned by [`mic_capture_loop`]; consulted once per captured frame.
47struct VolumeWriteGate {
48    last_written: Option<f32>,
49    last_write_at: Option<Instant>,
50}
51
52impl VolumeWriteGate {
53    fn new() -> Self {
54        Self {
55            last_written: None,
56            last_write_at: None,
57        }
58    }
59
60    /// `Some(level)` = write `level` now (and adopt it as the new baseline for
61    /// future change-detection); `None` = do nothing this frame. Every gate must
62    /// hold: a resolved provider, the AGC toggle ON, the mic not muted, AND
63    /// (the recommendation changed since the last write OR no write has
64    /// happened yet) AND the rate limit has elapsed.
65    fn should_write(
66        &mut self,
67        recommended: f32,
68        provider_available: bool,
69        agc_enabled: bool,
70        muted: bool,
71        now: Instant,
72    ) -> Option<f32> {
73        if !provider_available || !agc_enabled || muted {
74            return None;
75        }
76        let changed = self.last_written != Some(recommended);
77        let rate_ok = self
78            .last_write_at
79            .is_none_or(|at| now.duration_since(at) >= OS_VOLUME_WRITE_INTERVAL);
80        if changed && rate_ok {
81            self.last_written = Some(recommended);
82            self.last_write_at = Some(now);
83            Some(recommended)
84        } else {
85            None
86        }
87    }
88}

90/// The capture-loop-facing handle onto the OS input-volume round trip: an
91/// atomic mailbox to a dedicated background thread ([`volume_actor_loop`]) that
92/// owns the resolved platform handle and does every HAL/COM call. The capture
93/// loop only ever touches atomics here — never blocks, never does device I/O
94/// itself.
95struct OsInputVolume {
96    /// Last OS-READ applied volume (`f32::to_bits`), refreshed by the actor on
97    /// resolve, after a write, and on its periodic re-read. Read every captured
98    /// frame — a plain atomic load.
99    applied_bits: Arc<AtomicU32>,
100    /// Whether a platform provider is currently resolved for the open device.
101    /// Gates whether the capture loop even attempts a write.
102    available: Arc<AtomicBool>,
103    /// The capture loop's latest write REQUEST (`f32::to_bits`); "last one
104    /// wins" — no queue. `has_request` distinguishes a fresh request from the
105    /// actor still holding the value it already wrote.
106    requested_bits: Arc<AtomicU32>,
107    has_request: Arc<AtomicBool>,
108    /// Device-changed notices (first open, hot-swap, reopen): `Some(name)` to
109    /// drop any cached handle and re-resolve; `None` when the loop has no
110    /// device open at all. The actor owns the receiving end. Unbounded — sends
111    /// are rare (once per open) and must never block the capture thread.
112    device_tx: std::sync::mpsc::Sender<Option<String>>,
113}
114
115impl OsInputVolume {
116    /// Spawn the volume actor thread and return the capture-loop-facing handle.
117    fn start() -> Self {
118        // 1.0 (fully open) matches `Apm`'s own `FALLBACK_INPUT_VOLUME` default —
119        // before the first resolve/read lands, the round trip behaves exactly as
120        // if this wiring did not exist yet.
121        let applied_bits = Arc::new(AtomicU32::new(1.0f32.to_bits()));
122        let available = Arc::new(AtomicBool::new(false));
123        let requested_bits = Arc::new(AtomicU32::new(0));
124        let has_request = Arc::new(AtomicBool::new(false));
125        let (device_tx, device_rx) = std::sync::mpsc::channel::<Option<String>>();
126
127        let (a, av, r, hr) = (
128            applied_bits.clone(),
129            available.clone(),
130            requested_bits.clone(),
131            has_request.clone(),
132        );
133        std::thread::Builder::new()
134            .name("call-mic-volume".into())
135            .spawn(move || volume_actor_loop(device_rx, a, av, r, hr))
136            .ok();
137
138        Self {
139            applied_bits,
140            available,
141            requested_bits,
142            has_request,
143            device_tx,
144        }
145    }
146
147    /// The device this loop has open changed (first open, hot-swap, reopen):
148    /// drop the actor's cached handle and re-resolve for `name` (`None` = no
149    /// device — the actor parks with `available = false` until the next
150    /// notice). Non-blocking.
151    fn device_changed(&self, name: Option<String>) {
152        let _ = self.device_tx.send(name);
153    }
154
155    /// Last OS-read applied volume (0.0..=1.0). Cheap; never blocks.
156    fn applied(&self) -> f32 {
157        f32::from_bits(self.applied_bits.load(Ordering::Relaxed))
158    }
159
160    /// Whether a platform provider is resolved for the current device.
161    fn is_available(&self) -> bool {
162        self.available.load(Ordering::Relaxed)
163    }
164
165    /// Ask the actor to write `level` back to the OS at its own pace
166    /// (best-effort, "last one wins"). Cheap; never blocks.
167    fn request_write(&self, level: f32) {
168        self.requested_bits
169            .store(level.to_bits(), Ordering::Relaxed);
170        self.has_request.store(true, Ordering::Release);
171    }
172}

174/// The volume actor's body: owns the resolved [`InputVolumeHandle`] and does
175/// every HAL/COM call, off the mic capture thread. Exits when `device_rx`'s
176/// sender drops ([`OsInputVolume`] — and so this actor — lives exactly as long
177/// as one [`mic_capture_loop`] call).
178fn volume_actor_loop(
179    device_rx: std::sync::mpsc::Receiver<Option<String>>,
180    applied_bits: Arc<AtomicU32>,
181    available: Arc<AtomicBool>,
182    requested_bits: Arc<AtomicU32>,
183    has_request: Arc<AtomicBool>,
184) {
185    let mut provider: Option<InputVolumeHandle> = None;
186    let mut last_read = Instant::now();
187    loop {
188        match device_rx.recv_timeout(OS_VOLUME_POLL_INTERVAL) {
189            Ok(name) => {
190                provider = name.and_then(|n| Engine::new().resolve_input_volume(&n));
191                available.store(provider.is_some(), Ordering::Relaxed);
192                if let Some(v) = provider.as_ref().and_then(InputVolumeHandle::get) {
193                    applied_bits.store(v.to_bits(), Ordering::Relaxed);
194                }
195                last_read = Instant::now();
196            }
197            Err(std::sync::mpsc::RecvTimeoutError::Timeout) => {}
198            // The capture loop returned (its `OsInputVolume` — and `device_tx`
199            // with it — dropped): nothing left to serve.
200            Err(std::sync::mpsc::RecvTimeoutError::Disconnected) => break,
201        }
202        let Some(p) = provider.as_ref() else { continue };
203        if has_request.swap(false, Ordering::AcqRel) {
204            let level = f32::from_bits(requested_bits.load(Ordering::Relaxed));
205            if p.set(level) {
206                // Trust a fresh read over the value asked for — the OS may
207                // quantize/clamp to something slightly different.
208                let refreshed = p.get().unwrap_or(level);
209                applied_bits.store(refreshed.to_bits(), Ordering::Relaxed);
210            }
211            last_read = Instant::now();
212        } else if last_read.elapsed() >= OS_VOLUME_PERIODIC_READ_INTERVAL {
213            if let Some(v) = p.get() {
214                applied_bits.store(v.to_bits(), Ordering::Relaxed);
215            }
216            last_read = Instant::now();
217        }
218    }
219}

223/// Spawn the mic publish task. A blocking capture thread pulls PCM from the engine
224/// `Microphone` (hot-swappable via `device_rx`); the async half Opus-encodes and
225/// publishes each frame on the relay session's `mic` track.
226///
227/// Muting gates the encoder at BOTH ends of the PCM channel — capture-side (frames
228/// are never queued) and publish-side (anything already queued is discarded before
229/// encode) — and zeroes the reported level. Both are needed: the channel holds 8
230/// frames, so a capture-only gate lets ~160 ms of pre-mute audio keep publishing
231/// after the user hits mute.
232#[allow(clippy::too_many_arguments)]
233pub(super) fn spawn_mic_publish(
234    relay: Arc<MoqRelaySession>,
235    space_id: Uuid,
236    self_user_id: Uuid,
237    self_id: PeerId,
238    clock: CaptureClock,
239    apm: Option<Arc<Apm>>,
240    device_rx: watch::Receiver<DeviceChoice<MicId>>,
241    mute: Arc<MuteFlags>,
242    voice: Arc<VoiceProcessing>,
243    levels: watch::Sender<LevelsView>,
244    stop: Arc<AtomicBool>,
245    stats: Arc<CallStats>,
246    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
247    #[cfg(feature = "synthetic")] synthetic_silence: SyntheticSilence,
248) -> JoinHandle<()> {
249    let (pcm_tx, mut pcm_rx) = tokio::sync::mpsc::channel::<hound_engine::types::PcmFrame>(8);
250
251    // Blocking capture thread (the cpal Microphone is !Send — own it on one thread).
252    let capture_stop = stop.clone();
253    let stats_capture = stats.clone();
254    // The publish task's own view of the mute flag — see the gate in its loop below.
255    let mute_publish = mute.clone();
256    std::thread::Builder::new()
257        .name("call-mic-capture".into())
258        .spawn(move || {
259            mic_capture_loop(
260                capture_stop,
261                clock,
262                apm,
263                device_rx,
264                pcm_tx,
265                mute,
266                voice,
267                levels,
268                stats_capture,
269                #[cfg(feature = "synthetic")]
270                capture_source,
271                #[cfg(feature = "synthetic")]
272                synthetic_silence,
273            )
274        })
275        .ok();
276
277    let path = broadcast_path(space_id, self_user_id, self_id, MIC_TRACK);
278    let sender = self_id.0 as u16;
279    // Outbound voice is sacred too: encode + publish on the RT audio runtime so a
280    // screenshare encode can't stall the mic and make us stutter for everyone else.
281    audio_runtime().spawn(async move {
282        let mut encoder = match OpusEncoder::new(MIC_BITRATE, OpusProfile::Voice) {
283            Ok(e) => e,
284            Err(e) => {
285                tracing::error!(%e, "mic: failed to build Opus encoder");
286                return;
287            }
288        };
289        let mut frame_seq = 0u32;
290        // Publisher-side silence suppression — the policy (and the whole rationale for
291        // suppressing here rather than unsubscribing at the receiver) lives on
292        // `SilenceGate`. Never engages for a headless/e2e client: the synthetic capture
293        // source is VAD-active by construction, so its frames never come out DTX-sized.
294        let mut silence = SilenceGate::default();
295        let mut publisher = match relay.publish_track(&path, MIC_TRACK, AUDIO_PRIORITY) {
296            Ok(p) => p,
297            Err(e) => {
298                tracing::error!(%e, "mic: failed to publish track");
299                return;
300            }
301        };
302        if publisher.wait_subscribed().await.is_err() {
303            return;
304        }
305        while let Some(pcm) = pcm_rx.recv().await {
306            // Mute quiesces the encoder HERE as well as at capture. The capture loop
307            // already drops muted frames before `pcm_tx`, but that alone leaves a
308            // straggler window: up to the channel's 8 frames (~160 ms) are already
309            // queued when the flag flips, and without this check every one of them is
310            // still encoded and published — audio the user believes they muted.
311            //
312            // This is the same gate the screen-audio leg has always had (see
313            // `spawn_screen_audio_publish`); the mic simply never grew one. Checking
314            // at BOTH ends makes the quiesce synchronous with the flag rather than
315            // with the drain: the moment mute is set, nothing further is encoded, and
316            // `frame_seq` does not advance for a frame that was never sent. Receivers
317            // read the resulting gap as a quiet source (their starve metric is
318            // DTX-aware), and unmuting resumes instantly on this same track.
319            if mute_publish.mic.load(Ordering::Relaxed) {
320                // Same invariant as `silence_mic_meter`, on the wire gauge instead of
321                // the level: **the suppression gauge must read false whenever this loop
322                // is not delivering.** It is only ever written from an encoded frame, so
323                // a mute that lands while the room is quiet would otherwise leave
324                // `suppressed=true` latched for the whole mute — and the audio report
325                // would then attribute a muted (or dead) mic to a quiet room, which is
326                // precisely the inversion this fact exists to prevent.
327                stats.set_audio_suppressed(false);
328                continue;
329            }
330            let Ok(packet) = encoder.encode(&pcm) else {
331                // The third non-delivering path, same invariant as the mute gate above:
332                // a wedged encoder must not leave the gauge latched either.
333                stats.set_audio_suppressed(false);
334                continue;
335            };
336            let bytes = packet.data.len();
337            // Silence suppression. The encoder's own DTX verdict (a 1–2 byte payload)
338            // is the input; timestamps and the encoder's state are untouched, so the
339            // first voiced frame publishes on this same warm subscription with no
340            // added latency.
341            let publish = silence.publish(bytes, Instant::now());
342            stats.set_audio_suppressed(silence.suppressed());
343            if !publish {
344                stats.audio_frame_suppressed();
345                continue;
346            }
347            let frame = WireFrame {
348                sender,
349                track: TrackKind::CamAudio,
350                codec: WireCodec::Opus,
351                // Grouping is count-driven for audio (ordered substream), not keyframe-
352                // driven — every Opus frame is independently decodable, so the bit is moot.
353                keyframe: false,
354                frame_seq,
355                timestamp_us: packet.timestamp_us.micros(),
356                data: packet.data,
357            };
358            frame_seq = frame_seq.wrapping_add(1);
359            if let Err(e) = publish_audio_one(&mut publisher, &frame) {
360                tracing::warn!(%e, "mic: publish frame failed");
361            } else {
362                // RTC-debug: one Opus mic frame published (drives outbound send_kbps).
363                stats.audio_sent(bytes);
364            }
365        }
366        // The capture channel closed: capture died, or the call is tearing down. Either
367        // way nothing further is encoded, so the gauge must not stay latched — a dead
368        // capture path reading `suppressed` is the report blaming a quiet room for a
369        // broken mic. (`silence_mic_meter` holds the identical invariant for the level.)
370        stats.set_audio_suppressed(false);
371    })
372}

374/// Publish one AUDIO frame via the ordered-substream path (count-rolled groups), so
375/// successive Opus frames ride one in-order QUIC stream instead of one stream per frame
376/// (the source of the cross-stream reordering). Use for mic + screen-audio.
377fn publish_audio_one(
378    publisher: &mut MoqTrackPublisher,
379    frame: &WireFrame,
380) -> std::result::Result<(), String> {
381    publisher
382        .publish_audio_frame(frame)
383        .map_err(|e| e.to_string())
384}

386/// Open the mic per `capture_source`: real hardware, or (behind `synthetic`)
387/// the engine's mic-free test tone. The one seam every (re)open site in
388/// [`mic_capture_loop`] routes through, so a headless build never needs its
389/// own copy of the branch.
390///
391/// The error is PROPAGATED (it used to be `.ok()`-swallowed): a TCC-denied mic is
392/// a typed [`AudioError::PermissionDenied`] the loop reports and paces off, not an
393/// indistinguishable `None` that spins silently forever.
394fn open_mic(
395    choice: &DeviceChoice<MicId>,
396    clock: CaptureClock,
397    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
398    #[cfg(feature = "synthetic")] synthetic_silence: &SyntheticSilence,
399) -> std::result::Result<hound_engine::api::Microphone, hound_engine::error::AudioError> {
400    #[cfg(feature = "synthetic")]
401    if capture_source != CaptureSource::Default {
402        // The handle is shared, so a reopen (device fault, hot-swap) rejoins the same
403        // live switch rather than resetting a bot to "talking" mid-scenario.
404        return Engine::new().open_synthetic_microphone(clock, synthetic_silence.clone());
405    }
406    Engine::new().open_microphone(choice, clock)
407}

409/// Force the mic meter to silence.
410///
411/// THE invariant, and it is a privacy one: **the meter must read 0 whenever capture
412/// is not delivering.** The speaking reporter (`call::speaking`) samples this very
413/// watch and re-asserts whatever band it reads every 600 ms for as long as it is
414/// non-zero — so a level left latched by a dead BT mic or a mute that raced a
415/// capture stall does not merely mis-draw a bar, it makes its owner the call's
416/// permanent dominant speaker and leaks their pre-mute level to every peer. Every
417/// non-delivering path (mute, open failure, capture fault, no device) calls this.
418fn silence_mic_meter(levels: &watch::Sender<LevelsView>, stats: &CallStats) {
419    levels.send_modify(|l| l.mic = 0.0);
420    stats.set_self_level(0.0);
421}

423/// Full-scale RMS (0..1) to dBFS, for the mic-stage gauges. `-inf` on digital
424/// silence is fine — the stats layer clamps it to its -100 dBFS floor.
425fn dbfs(rms: f32) -> f32 {
426    20.0 * rms.log10()
427}

429/// The mic capture loop: pull PCM, process (AEC/NS/AGC), gate on mute, meter the
430/// level, forward to the encoder. Hot-swaps the device when `device_rx` changes;
431/// re-opens on error.
432#[allow(clippy::too_many_arguments)]
433fn mic_capture_loop(
434    stop: Arc<AtomicBool>,
435    clock: CaptureClock,
436    apm: Option<Arc<Apm>>,
437    mut device_rx: watch::Receiver<DeviceChoice<MicId>>,
438    pcm_tx: tokio::sync::mpsc::Sender<hound_engine::types::PcmFrame>,
439    mute: Arc<MuteFlags>,
440    voice: Arc<VoiceProcessing>,
441    levels: watch::Sender<LevelsView>,
442    stats: Arc<CallStats>,
443    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
444    #[cfg(feature = "synthetic")] synthetic_silence: SyntheticSilence,
445) {
446    use hound_engine::api::level_from_rms;
447
448    // P5: the OS input-volume round trip lives for exactly this loop's lifetime
449    // (dropped — and its actor thread wound down — when the loop returns).
450    let os_volume = OsInputVolume::start();
451    let mut volume_gate = VolumeWriteGate::new();
452
453    let mut choice = device_rx.borrow_and_update().clone();
454    // The device is opened from ONE place (top of the loop), so every open — first,
455    // hot-swap, warmup, post-fault — reports and paces through the same policy.
456    let mut mic: Option<hound_engine::api::Microphone> = None;
457    let mut warmup = MicRouteWarmup { due: None };
458    // Cleared for the reopen the warmup itself triggers: one warmup per open cycle
459    // (re-arming there would reopen the device every ROUTE_WARMUP_DELAY forever).
460    let mut arm_warmup = true;
461    // Open-failure reporting + retry cadence. A denied mic used to be swallowed by
462    // `.ok()` (silent 200 ms spin, nothing in the logs or Sentry); now it reports
463    // once per transition and retries slowly, since the grant can still arrive.
464    let mut open_retry = OpenRetry::default();
465    // How long the mic has been UNAVAILABLE, and whether that has outlived recovery.
466    //
467    // The engine's cpal input error callback only warns when a stream dies — at that
468    // point the reopen below has not been tried, and it fixes nearly every fault
469    // (unplug, suspend/resume, an automatic reroute). What deserves an event is a mic
470    // that stays gone: `OpenRetry` alone reports a transient class exactly once at
471    // `warn`, so a permanently vanished device was otherwise invisible in Sentry
472    // while the far end heard nothing.
473    let mut mic_outage = hound_engine::audio::DeviceOutage::default();
474    let mut silence = MicSilenceWatch::default();
475    // Last capture-ring overflow total seen from the CURRENT device. The counter is
476    // per-open, so a value below the last read means a reopen/hot-swap minted a fresh
477    // one — count it whole rather than under-reporting until it passes the old total.
478    let mut last_overflow: u64 = 0;
479    // Mute-edge memory for the APM's capture-output-used hint (`None` until the
480    // first frame so the initial state is always asserted).
481    let mut apm_output_used: Option<bool> = None;
482
483    while !stop.load(Ordering::Relaxed) {
484        if device_rx.has_changed().unwrap_or(false) {
485            choice = device_rx.borrow_and_update().clone();
486            drop(mic.take());
487            // Disarm the OLD device's pending warmup: were it left due, it could
488            // fire in the gap before the reopen below and burn the NEW device's
489            // one warmup cycle on a device it never belonged to.
490            warmup = MicRouteWarmup { due: None };
491            arm_warmup = true;
492        }
493        // BT/HFP route warmup: one reopen after the SCO link has had time to
494        // settle, mirroring the playout side's warmup in the mix bus.
495        if warmup.fire(Instant::now()) {
496            tracing::info!("voice: warmup reopen of non-48k (BT/HFP) mic — silent-route fix");
497            drop(mic.take());
498            arm_warmup = false;
499        }
500        if mic.is_none() {
501            match open_mic(
502                &choice,
503                clock,
504                #[cfg(feature = "synthetic")]
505                capture_source,
506                #[cfg(feature = "synthetic")]
507                &synthetic_silence,
508            ) {
509                Ok(m) => {
510                    // The rate is the one fact that identifies which route we actually
511                    // landed on after a fault/hot-swap reopen (48 k built-in vs 24 k HFP)
512                    // — and it is what the capture resampler is pinned to for this open.
513                    // Every open used to be silent, so a mid-call device change left no
514                    // trace at all.
515                    tracing::debug!(
516                        track = "mic",
517                        device_rate = m.device_rate(),
518                        "mic: capture device opened"
519                    );
520                    if arm_warmup {
521                        warmup = MicRouteWarmup::arm(m.device_rate(), Instant::now());
522                    }
523                    arm_warmup = true;
524                    // P5: every open (first, hot-swap, warmup, post-fault) is a new
525                    // device — drop the volume actor's cached provider and re-resolve
526                    // against whichever name this open actually landed on.
527                    os_volume.device_changed(m.opened_name().map(str::to_owned));
528                    mic = Some(m);
529                    open_retry.opened();
530                    if let Some(o) = mic_outage.recovered(Instant::now()) {
531                        tracing::info!(
532                            track = "mic",
533                            down_ms = o.elapsed.as_millis() as u64,
534                            failed_opens = o.failed_attempts,
535                            "mic: capture recovered after {:?} — the far end can hear \
536                             this user again",
537                            o.elapsed
538                        );
539                    }
540                }
541                Err(e) => {
542                    // Classify, then report ONCE per transition into this failure
543                    // class — the Sentry bridge promotes `error!` to an event, so a
544                    // sticky denial must not log per tick. Keep looping either way:
545                    // the device (or the grant) can still show up.
546                    let failure = OpenFailure::of_audio(&e);
547                    let (log, wait) = open_retry.failed(failure);
548                    if log {
549                        match failure {
550                            OpenFailure::PermissionDenied => tracing::error!(
551                                track = "mic",
552                                %e,
553                                "microphone permission denied — enable hound in System Settings → \
554                                 Privacy & Security → Microphone"
555                            ),
556                            OpenFailure::Other => tracing::warn!(
557                                track = "mic",
558                                %e,
559                                "mic: microphone open failed — retrying"
560                            ),
561                        }
562                    }
563                    // A transient class only becomes an EVENT once the retry ladder
564                    // has demonstrably failed for the whole escalation window — the
565                    // reopen fixes an unplug/suspend/reroute far inside it. Denial is
566                    // excluded: it is already reported above, actionably and once.
567                    let now = Instant::now();
568                    mic_outage.attempt_failed(now);
569                    if failure == OpenFailure::Other
570                        && let Some(o) = mic_outage.escalate(now)
571                    {
572                        tracing::error!(
573                            track = "mic",
574                            down_ms = o.elapsed.as_millis() as u64,
575                            failed_opens = o.failed_attempts,
576                            %e,
577                            "mic: no capture device for {:?} across {} failed opens — the \
578                             far end has heard nothing since",
579                            o.elapsed,
580                            o.failed_attempts
581                        );
582                    }
583                    // Nothing is being captured, so nothing may still be metered —
584                    // and no device to drive the OS volume actor against either.
585                    os_volume.device_changed(None);
586                    silence_mic_meter(&levels, &stats);
587                    std::thread::sleep(wait);
588                    continue;
589                }
590            }
591        }
592        let Some(cap) = mic.as_mut() else {
593            silence_mic_meter(&levels, &stats);
594            continue;
595        };
596        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| cap.next_frame()));
597        match pulled {
598            Ok(Ok(mut frame)) => {
599                // RTC-debug: one PCM frame captured from the mic device, plus any
600                // samples the capture ring dropped since the last frame — the ONLY
601                // local signal that the far end is hearing chopped words.
602                stats.audio_captured();
603                let overflow = cap.capture_overflow_samples();
604                stats.capture_overflowed(if overflow >= last_overflow {
605                    overflow - last_overflow
606                } else {
607                    overflow
608                });
609                last_overflow = overflow;
610                // The dead-route detector reads the RAW frame — it keys on
611                // bit-exact device zeros, which the APM's dither/HPF would erase.
612                if silence.observe(&frame.samples) {
613                    tracing::warn!(
614                        device_rate = cap.device_rate(),
615                        "voice: mic delivering PURE SILENCE (~3 s of bit-exact zeros) — \
616                         dead capture route (BT/SCO not live, permission, or a silent \
617                         virtual device)"
618                    );
619                }
620                let muted = mute.mic.load(Ordering::Relaxed);
621                // Capture processing runs on EVERY frame, muted included, and
622                // before the mute gate: AEC3's render swap-queue is drained by
623                // capture calls and its delay tracking (and AGC2's speech-level
624                // estimate) need the 10 ms cadence — skipping during mute means
625                // an echo burst and wrong gain on every unmute. Muted frames are
626                // still dropped below, so nothing processed ever leaves. The
627                // output-used hint (on mute edges) freezes AGC2's adaptation so a
628                // long mute over keyboard clatter cannot retune the speech gain.
629                if let Some(apm) = apm.as_ref() {
630                    if apm_output_used != Some(!muted) {
631                        apm.set_capture_output_used(!muted);
632                        apm_output_used = Some(!muted);
633                    }
634                    // P5: feed the last OS-read applied level BEFORE processing (the
635                    // round trip's "apply" half — see `apm.rs`'s `process_capture` doc),
636                    // unconditionally and every frame, same as the mute hint above: this
637                    // must keep flowing even while AGC is off or the mic is muted, so the
638                    // controller's next opinion (once re-enabled/unmuted) starts from
639                    // ground truth rather than a stale reading.
640                    apm.set_applied_input_volume(os_volume.applied());
641                    let raw_rms = frame.rms();
642                    apm.process_capture(&mut frame);
643                    let clipped = frame
644                        .samples
645                        .iter()
646                        .any(|&s| s == i16::MIN || s == i16::MAX);
647                    stats.set_mic_stage_levels(dbfs(raw_rms), dbfs(frame.rms()), clipped);
648                    // Re-assert liveness per frame: the APM is reported active at
649                    // establish, but its panic guard can retire it mid-call (a
650                    // panicking process call poisons the lock and every later frame
651                    // passes through untouched). Without this the audio report would
652                    // keep claiming AEC/NS/AGC are running on a raw mic.
653                    stats.set_apm_active(!apm.is_degraded());
654
655                    // P5's "apply" half: the controller's fresh recommendation, gated on
656                    // a resolved provider + the AGC toggle (coupled, no separate setting)
657                    // + not muted + the ~1/s rate limit. `VolumeWriteGate` is pure and
658                    // unit-tested (`call::audio_tx::tests`); this call site is the only
659                    // place its inputs come from real state.
660                    let recommended = apm.recommended_input_volume();
661                    if let Some(level) = volume_gate.should_write(
662                        recommended,
663                        os_volume.is_available(),
664                        voice.agc_enabled(),
665                        muted,
666                        Instant::now(),
667                    ) {
668                        os_volume.request_write(level);
669                    }
670                    stats.set_os_input_volume(
671                        os_volume.is_available(),
672                        os_volume.applied(),
673                        recommended,
674                    );
675                }
676                if muted {
677                    silence_mic_meter(&levels, &stats);
678                    continue;
679                }
680                // Metered POST-processing: the bar (and dominant-speaker ranking)
681                // reflect what peers actually hear, gain included.
682                let level = level_from_rms(frame.rms());
683                levels.send_modify(|l| l.mic = level);
684                stats.set_self_level(level);
685                let _ = pcm_tx.try_send(frame);
686            }
687            _ => {
688                // The capture ring stopped (the cpal input error callback fired — device
689                // gone, config invalidated, or an automatic reroute) or `next_frame`
690                // panicked. Rebuilding is REQUIRED: the device rate and the resampler are
691                // captured at open, so a stream that silently rerouted to a 24 kHz HFP mic
692                // would keep resampling as if it were still 48 k. Announce it — this path
693                // used to drop the device and sleep with nothing in the log, so a mid-call
694                // mic route change was invisible.
695                tracing::info!(
696                    track = "mic",
697                    device_rate = mic.as_ref().map(|m| m.device_rate()).unwrap_or(0),
698                    "mic: capture stream ended — reopening the device"
699                );
700                // Start the outage clock at the DEATH, not at the first failed
701                // reopen, so the escalation window measures how long the user has
702                // actually been inaudible.
703                mic_outage.down(Instant::now());
704                drop(mic.take());
705                // The route is gone; the last level it produced must not outlive it,
706                // and neither should the OS volume actor's now-stale provider.
707                os_volume.device_changed(None);
708                silence_mic_meter(&levels, &stats);
709                std::thread::sleep(std::time::Duration::from_millis(200));
710            }
711        }
712    }
713}

715/// The `!Send` screen APP-AUDIO capture thread: block on the engine `ScreenAudio`
716/// (the mic-thread analogue — no APM, no levels, no hot-swap) and forward each 20 ms
717/// PCM frame to the publish task. A `next_frame` error/panic is the leg's clean end
718/// (its parent screen stream stopped); drop the leg on this owning thread and exit.
719pub(super) fn screen_audio_capture_loop(
720    stop: Arc<AtomicBool>,
721    mut audio: hound_engine::api::ScreenAudio,
722    pcm_tx: tokio::sync::mpsc::Sender<hound_engine::types::PcmFrame>,
723) {
724    let mut captured: u64 = 0;
725    while !stop.load(Ordering::Relaxed) {
726        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| audio.next_frame()));
727        match pulled {
728            Ok(Ok(pcm)) => {
729                captured += 1;
730                if captured == 1 || captured.is_multiple_of(500) {
731                    tracing::info!(captured, "screen: captured app-audio frames");
732                }
733                let _ = pcm_tx.try_send(pcm); // drop on backpressure
734            }
735            Ok(Err(e)) => {
736                tracing::warn!(%e, captured, "screen: app-audio capture ended (error)");
737                break;
738            }
739            Err(_panic) => {
740                tracing::warn!(captured, "screen: app-audio capture panicked");
741                break;
742            }
743        }
744    }
745    tracing::info!(captured, "screen: app-audio capture loop stopped");
746    drop(audio);
747}

749/// The async screen AUDIO publish task: the mic publish path MINUS APM / levels /
750/// hot-swap (app audio is not a mic). Opus-encodes each forwarded PCM frame and
751/// publishes it on the `screen_audio` track; the receiver plays it on the shared
752/// mix bus exactly like a mic.
753pub(super) fn spawn_screen_audio_publish(
754    relay: Arc<MoqRelaySession>,
755    path: String,
756    sender: u16,
757    mut pcm_rx: tokio::sync::mpsc::Receiver<hound_engine::types::PcmFrame>,
758    stats: Arc<CallStats>,
759    mute: Arc<MuteFlags>,
760) -> JoinHandle<()> {
761    // App-audio is audio: encode + publish on the RT audio runtime, off the video pool.
762    audio_runtime().spawn(async move {
763        let mut encoder = match OpusEncoder::new(SCREEN_AUDIO_BITRATE, OpusProfile::Music) {
764            Ok(e) => e,
765            Err(e) => {
766                tracing::error!(%e, "screen: failed to build app-audio Opus encoder");
767                return;
768            }
769        };
770        let mut publisher = match relay.publish_track(&path, SCREEN_AUDIO_TRACK, AUDIO_PRIORITY) {
771            Ok(p) => p,
772            Err(e) => {
773                tracing::error!(%e, "screen: failed to publish audio track");
774                return;
775            }
776        };
777        if publisher.wait_subscribed().await.is_err() {
778            return;
779        }
780        tracing::info!("screen: app-audio track subscribed — publishing");
781        let mut frame_seq = 0u32;
782        let mut sent: u64 = 0;
783        // The same suppression gate as the mic, and today a permanent no-op: the MUSIC
784        // profile deliberately runs without DTX (`OpusProfile::dtx`), so a shared
785        // track's silence is still coded at full rate and never reaches the DTX size
786        // band. It is wired anyway so the two audio publish legs cannot drift — the day
787        // the music profile's silence-boundary listening check lands and DTX flips on,
788        // app audio inherits suppression with no further change.
789        let mut silence = SilenceGate::default();
790        while let Some(pcm) = pcm_rx.recv().await {
791            // The broadcaster's "share audio" toggle: drop BEFORE encode, so a muted
792            // share costs zero CPU and zero wire data. Receivers read the gap as a
793            // quiet source (their starve metric is DTX-aware), and un-muting resumes
794            // instantly on this same track.
795            if mute.screen_audio.load(Ordering::Relaxed) {
796                continue;
797            }
798            let Ok(packet) = encoder.encode(&pcm) else {
799                continue;
800            };
801            let bytes = packet.data.len();
802            if !silence.publish(bytes, Instant::now()) {
803                continue;
804            }
805            let frame = WireFrame {
806                sender,
807                track: TrackKind::ScrAudio,
808                codec: WireCodec::Opus,
809                // Count-driven grouping (ordered substream); keyframe bit is moot for audio.
810                keyframe: false,
811                frame_seq,
812                timestamp_us: packet.timestamp_us.micros(),
813                data: packet.data,
814            };
815            frame_seq = frame_seq.wrapping_add(1);
816            if let Err(e) = publish_audio_one(&mut publisher, &frame) {
817                tracing::warn!(%e, "screen: publish audio frame failed");
818            } else {
819                // RTC-debug: one screen app-audio frame published.
820                stats.screen_audio_sent(bytes);
821                sent += 1;
822                if sent == 1 || sent.is_multiple_of(500) {
823                    tracing::info!(sent, "screen: published app-audio frames");
824                }
825            }
826        }
827        tracing::info!(sent, "screen: app-audio publish ended");
828    })
829}

835    /// AGC-off inertness: no write, no matter how much the recommendation

899            None,
