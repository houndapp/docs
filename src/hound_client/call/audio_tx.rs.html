1//! Outbound audio: the mic capture loop + Opus publish, and the screen APP-AUDIO
2//! leg (its `!Send` capture thread + publish task). Encode + publish run on the
3//! RT audio runtime ("audio is sacred": a screenshare encode can't stall the
4//! mic), and audio frames publish via the ordered-substream path (count-rolled
5//! groups) so successive Opus frames ride one in-order QUIC stream — never one
6//! stream per frame.

12/// Spawn the mic publish task. A blocking capture thread pulls PCM from the engine
13/// `Microphone` (hot-swappable via `device_rx`); the async half Opus-encodes and
14/// publishes each frame on the relay session's `mic` track.
15///
16/// Muting gates the encoder at BOTH ends of the PCM channel — capture-side (frames
17/// are never queued) and publish-side (anything already queued is discarded before
18/// encode) — and zeroes the reported level. Both are needed: the channel holds 8
19/// frames, so a capture-only gate lets ~160 ms of pre-mute audio keep publishing
20/// after the user hits mute.
21#[allow(clippy::too_many_arguments)]
22pub(super) fn spawn_mic_publish(
23    relay: Arc<MoqRelaySession>,
24    space_id: Uuid,
25    self_user_id: Uuid,
26    self_id: PeerId,
27    clock: CaptureClock,
28    apm: Option<Arc<Apm>>,
29    device_rx: watch::Receiver<DeviceChoice<MicId>>,
30    mute: Arc<MuteFlags>,
31    levels: watch::Sender<LevelsView>,
32    stop: Arc<AtomicBool>,
33    stats: Arc<CallStats>,
34    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
35    #[cfg(feature = "synthetic")] synthetic_silence: SyntheticSilence,
36) -> JoinHandle<()> {
37    let (pcm_tx, mut pcm_rx) = tokio::sync::mpsc::channel::<hound_engine::types::PcmFrame>(8);
38
39    // Blocking capture thread (the cpal Microphone is !Send — own it on one thread).
40    let capture_stop = stop.clone();
41    let stats_capture = stats.clone();
42    // The publish task's own view of the mute flag — see the gate in its loop below.
43    let mute_publish = mute.clone();
44    std::thread::Builder::new()
45        .name("call-mic-capture".into())
46        .spawn(move || {
47            mic_capture_loop(
48                capture_stop,
49                clock,
50                apm,
51                device_rx,
52                pcm_tx,
53                mute,
54                levels,
55                stats_capture,
56                #[cfg(feature = "synthetic")]
57                capture_source,
58                #[cfg(feature = "synthetic")]
59                synthetic_silence,
60            )
61        })
62        .ok();
63
64    let path = broadcast_path(space_id, self_user_id, self_id, MIC_TRACK);
65    let sender = self_id.0 as u16;
66    // Outbound voice is sacred too: encode + publish on the RT audio runtime so a
67    // screenshare encode can't stall the mic and make us stutter for everyone else.
68    audio_runtime().spawn(async move {
69        let mut encoder = match OpusEncoder::new(MIC_BITRATE, OpusProfile::Voice) {
70            Ok(e) => e,
71            Err(e) => {
72                tracing::error!(%e, "mic: failed to build Opus encoder");
73                return;
74            }
75        };
76        let mut frame_seq = 0u32;
77        // Publisher-side silence suppression — the policy (and the whole rationale for
78        // suppressing here rather than unsubscribing at the receiver) lives on
79        // `SilenceGate`. Never engages for a headless/e2e client: the synthetic capture
80        // source is VAD-active by construction, so its frames never come out DTX-sized.
81        let mut silence = SilenceGate::default();
82        let mut publisher = match relay.publish_track(&path, MIC_TRACK, AUDIO_PRIORITY) {
83            Ok(p) => p,
84            Err(e) => {
85                tracing::error!(%e, "mic: failed to publish track");
86                return;
87            }
88        };
89        if publisher.wait_subscribed().await.is_err() {
90            return;
91        }
92        while let Some(pcm) = pcm_rx.recv().await {
93            // Mute quiesces the encoder HERE as well as at capture. The capture loop
94            // already drops muted frames before `pcm_tx`, but that alone leaves a
95            // straggler window: up to the channel's 8 frames (~160 ms) are already
96            // queued when the flag flips, and without this check every one of them is
97            // still encoded and published — audio the user believes they muted.
98            //
99            // This is the same gate the screen-audio leg has always had (see
100            // `spawn_screen_audio_publish`); the mic simply never grew one. Checking
101            // at BOTH ends makes the quiesce synchronous with the flag rather than
102            // with the drain: the moment mute is set, nothing further is encoded, and
103            // `frame_seq` does not advance for a frame that was never sent. Receivers
104            // read the resulting gap as a quiet source (their starve metric is
105            // DTX-aware), and unmuting resumes instantly on this same track.
106            if mute_publish.mic.load(Ordering::Relaxed) {
107                // Same invariant as `silence_mic_meter`, on the wire gauge instead of
108                // the level: **the suppression gauge must read false whenever this loop
109                // is not delivering.** It is only ever written from an encoded frame, so
110                // a mute that lands while the room is quiet would otherwise leave
111                // `suppressed=true` latched for the whole mute — and the audio report
112                // would then attribute a muted (or dead) mic to a quiet room, which is
113                // precisely the inversion this fact exists to prevent.
114                stats.set_audio_suppressed(false);
115                continue;
116            }
117            let Ok(packet) = encoder.encode(&pcm) else {
118                // The third non-delivering path, same invariant as the mute gate above:
119                // a wedged encoder must not leave the gauge latched either.
120                stats.set_audio_suppressed(false);
121                continue;
122            };
123            let bytes = packet.data.len();
124            // Silence suppression. The encoder's own DTX verdict (a 1–2 byte payload)
125            // is the input; timestamps and the encoder's state are untouched, so the
126            // first voiced frame publishes on this same warm subscription with no
127            // added latency.
128            let publish = silence.publish(bytes, Instant::now());
129            stats.set_audio_suppressed(silence.suppressed());
130            if !publish {
131                stats.audio_frame_suppressed();
132                continue;
133            }
134            let frame = WireFrame {
135                sender,
136                track: TrackKind::CamAudio,
137                codec: WireCodec::Opus,
138                // Grouping is count-driven for audio (ordered substream), not keyframe-
139                // driven — every Opus frame is independently decodable, so the bit is moot.
140                keyframe: false,
141                frame_seq,
142                timestamp_us: packet.timestamp_us.micros(),
143                data: packet.data,
144            };
145            frame_seq = frame_seq.wrapping_add(1);
146            if let Err(e) = publish_audio_one(&mut publisher, &frame) {
147                tracing::warn!(%e, "mic: publish frame failed");
148            } else {
149                // RTC-debug: one Opus mic frame published (drives outbound send_kbps).
150                stats.audio_sent(bytes);
151            }
152        }
153        // The capture channel closed: capture died, or the call is tearing down. Either
154        // way nothing further is encoded, so the gauge must not stay latched — a dead
155        // capture path reading `suppressed` is the report blaming a quiet room for a
156        // broken mic. (`silence_mic_meter` holds the identical invariant for the level.)
157        stats.set_audio_suppressed(false);
158    })
159}

161/// Publish one AUDIO frame via the ordered-substream path (count-rolled groups), so
162/// successive Opus frames ride one in-order QUIC stream instead of one stream per frame
163/// (the source of the cross-stream reordering). Use for mic + screen-audio.
164fn publish_audio_one(
165    publisher: &mut MoqTrackPublisher,
166    frame: &WireFrame,
167) -> std::result::Result<(), String> {
168    publisher
169        .publish_audio_frame(frame)
170        .map_err(|e| e.to_string())
171}

173/// Open the mic per `capture_source`: real hardware, or (behind `synthetic`)
174/// the engine's mic-free test tone. The one seam every (re)open site in
175/// [`mic_capture_loop`] routes through, so a headless build never needs its
176/// own copy of the branch.
177///
178/// The error is PROPAGATED (it used to be `.ok()`-swallowed): a TCC-denied mic is
179/// a typed [`AudioError::PermissionDenied`] the loop reports and paces off, not an
180/// indistinguishable `None` that spins silently forever.
181fn open_mic(
182    choice: &DeviceChoice<MicId>,
183    clock: CaptureClock,
184    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
185    #[cfg(feature = "synthetic")] synthetic_silence: &SyntheticSilence,
186) -> std::result::Result<hound_engine::api::Microphone, hound_engine::error::AudioError> {
187    #[cfg(feature = "synthetic")]
188    if capture_source != CaptureSource::Default {
189        // The handle is shared, so a reopen (device fault, hot-swap) rejoins the same
190        // live switch rather than resetting a bot to "talking" mid-scenario.
191        return Engine::new().open_synthetic_microphone(clock, synthetic_silence.clone());
192    }
193    Engine::new().open_microphone(choice, clock)
194}

196/// Force the mic meter to silence.
197///
198/// THE invariant, and it is a privacy one: **the meter must read 0 whenever capture
199/// is not delivering.** The speaking reporter (`call::speaking`) samples this very
200/// watch and re-asserts whatever band it reads every 600 ms for as long as it is
201/// non-zero — so a level left latched by a dead BT mic or a mute that raced a
202/// capture stall does not merely mis-draw a bar, it makes its owner the call's
203/// permanent dominant speaker and leaks their pre-mute level to every peer. Every
204/// non-delivering path (mute, open failure, capture fault, no device) calls this.
205fn silence_mic_meter(levels: &watch::Sender<LevelsView>, stats: &CallStats) {
206    levels.send_modify(|l| l.mic = 0.0);
207    stats.set_self_level(0.0);
208}

210/// Full-scale RMS (0..1) to dBFS, for the mic-stage gauges. `-inf` on digital
211/// silence is fine — the stats layer clamps it to its -100 dBFS floor.
212fn dbfs(rms: f32) -> f32 {
213    20.0 * rms.log10()
214}

216/// The mic capture loop: pull PCM, process (AEC/NS/AGC), gate on mute, meter the
217/// level, forward to the encoder. Hot-swaps the device when `device_rx` changes;
218/// re-opens on error.
219#[allow(clippy::too_many_arguments)]
220fn mic_capture_loop(
221    stop: Arc<AtomicBool>,
222    clock: CaptureClock,
223    apm: Option<Arc<Apm>>,
224    mut device_rx: watch::Receiver<DeviceChoice<MicId>>,
225    pcm_tx: tokio::sync::mpsc::Sender<hound_engine::types::PcmFrame>,
226    mute: Arc<MuteFlags>,
227    levels: watch::Sender<LevelsView>,
228    stats: Arc<CallStats>,
229    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
230    #[cfg(feature = "synthetic")] synthetic_silence: SyntheticSilence,
231) {
232    use hound_engine::api::level_from_rms;
233
234    let mut choice = device_rx.borrow_and_update().clone();
235    // The device is opened from ONE place (top of the loop), so every open — first,
236    // hot-swap, warmup, post-fault — reports and paces through the same policy.
237    let mut mic: Option<hound_engine::api::Microphone> = None;
238    let mut warmup = MicRouteWarmup { due: None };
239    // Cleared for the reopen the warmup itself triggers: one warmup per open cycle
240    // (re-arming there would reopen the device every ROUTE_WARMUP_DELAY forever).
241    let mut arm_warmup = true;
242    // Open-failure reporting + retry cadence. A denied mic used to be swallowed by
243    // `.ok()` (silent 200 ms spin, nothing in the logs or Sentry); now it reports
244    // once per transition and retries slowly, since the grant can still arrive.
245    let mut open_retry = OpenRetry::default();
246    let mut silence = MicSilenceWatch::default();
247    // Last capture-ring overflow total seen from the CURRENT device. The counter is
248    // per-open, so a value below the last read means a reopen/hot-swap minted a fresh
249    // one — count it whole rather than under-reporting until it passes the old total.
250    let mut last_overflow: u64 = 0;
251    // Mute-edge memory for the APM's capture-output-used hint (`None` until the
252    // first frame so the initial state is always asserted).
253    let mut apm_output_used: Option<bool> = None;
254
255    while !stop.load(Ordering::Relaxed) {
256        if device_rx.has_changed().unwrap_or(false) {
257            choice = device_rx.borrow_and_update().clone();
258            drop(mic.take());
259            // Disarm the OLD device's pending warmup: were it left due, it could
260            // fire in the gap before the reopen below and burn the NEW device's
261            // one warmup cycle on a device it never belonged to.
262            warmup = MicRouteWarmup { due: None };
263            arm_warmup = true;
264        }
265        // BT/HFP route warmup: one reopen after the SCO link has had time to
266        // settle, mirroring the playout side's warmup in the mix bus.
267        if warmup.fire(Instant::now()) {
268            tracing::info!("voice: warmup reopen of non-48k (BT/HFP) mic — silent-route fix");
269            drop(mic.take());
270            arm_warmup = false;
271        }
272        if mic.is_none() {
273            match open_mic(
274                &choice,
275                clock,
276                #[cfg(feature = "synthetic")]
277                capture_source,
278                #[cfg(feature = "synthetic")]
279                &synthetic_silence,
280            ) {
281                Ok(m) => {
282                    // The rate is the one fact that identifies which route we actually
283                    // landed on after a fault/hot-swap reopen (48 k built-in vs 24 k HFP)
284                    // — and it is what the capture resampler is pinned to for this open.
285                    // Every open used to be silent, so a mid-call device change left no
286                    // trace at all.
287                    tracing::debug!(
288                        track = "mic",
289                        device_rate = m.device_rate(),
290                        "mic: capture device opened"
291                    );
292                    if arm_warmup {
293                        warmup = MicRouteWarmup::arm(m.device_rate(), Instant::now());
294                    }
295                    arm_warmup = true;
296                    mic = Some(m);
297                    open_retry.opened();
298                }
299                Err(e) => {
300                    // Classify, then report ONCE per transition into this failure
301                    // class — the Sentry bridge promotes `error!` to an event, so a
302                    // sticky denial must not log per tick. Keep looping either way:
303                    // the device (or the grant) can still show up.
304                    let failure = OpenFailure::of_audio(&e);
305                    let (log, wait) = open_retry.failed(failure);
306                    if log {
307                        match failure {
308                            OpenFailure::PermissionDenied => tracing::error!(
309                                track = "mic",
310                                %e,
311                                "microphone permission denied — enable hound in System Settings → \
312                                 Privacy & Security → Microphone"
313                            ),
314                            OpenFailure::Other => tracing::warn!(
315                                track = "mic",
316                                %e,
317                                "mic: microphone open failed — retrying"
318                            ),
319                        }
320                    }
321                    // Nothing is being captured, so nothing may still be metered.
322                    silence_mic_meter(&levels, &stats);
323                    std::thread::sleep(wait);
324                    continue;
325                }
326            }
327        }
328        let Some(cap) = mic.as_mut() else {
329            silence_mic_meter(&levels, &stats);
330            continue;
331        };
332        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| cap.next_frame()));
333        match pulled {
334            Ok(Ok(mut frame)) => {
335                // RTC-debug: one PCM frame captured from the mic device, plus any
336                // samples the capture ring dropped since the last frame — the ONLY
337                // local signal that the far end is hearing chopped words.
338                stats.audio_captured();
339                let overflow = cap.capture_overflow_samples();
340                stats.capture_overflowed(if overflow >= last_overflow {
341                    overflow - last_overflow
342                } else {
343                    overflow
344                });
345                last_overflow = overflow;
346                // The dead-route detector reads the RAW frame — it keys on
347                // bit-exact device zeros, which the APM's dither/HPF would erase.
348                if silence.observe(&frame.samples) {
349                    tracing::warn!(
350                        device_rate = cap.device_rate(),
351                        "voice: mic delivering PURE SILENCE (~3 s of bit-exact zeros) — \
352                         dead capture route (BT/SCO not live, permission, or a silent \
353                         virtual device)"
354                    );
355                }
356                let muted = mute.mic.load(Ordering::Relaxed);
357                // Capture processing runs on EVERY frame, muted included, and
358                // before the mute gate: AEC3's render swap-queue is drained by
359                // capture calls and its delay tracking (and AGC2's speech-level
360                // estimate) need the 10 ms cadence — skipping during mute means
361                // an echo burst and wrong gain on every unmute. Muted frames are
362                // still dropped below, so nothing processed ever leaves. The
363                // output-used hint (on mute edges) freezes AGC2's adaptation so a
364                // long mute over keyboard clatter cannot retune the speech gain.
365                if let Some(apm) = apm.as_ref() {
366                    if apm_output_used != Some(!muted) {
367                        apm.set_capture_output_used(!muted);
368                        apm_output_used = Some(!muted);
369                    }
370                    let raw_rms = frame.rms();
371                    apm.process_capture(&mut frame);
372                    let clipped = frame
373                        .samples
374                        .iter()
375                        .any(|&s| s == i16::MIN || s == i16::MAX);
376                    stats.set_mic_stage_levels(dbfs(raw_rms), dbfs(frame.rms()), clipped);
377                    // Re-assert liveness per frame: the APM is reported active at
378                    // establish, but its panic guard can retire it mid-call (a
379                    // panicking process call poisons the lock and every later frame
380                    // passes through untouched). Without this the audio report would
381                    // keep claiming AEC/NS/AGC are running on a raw mic.
382                    stats.set_apm_active(!apm.is_degraded());
383                }
384                if muted {
385                    silence_mic_meter(&levels, &stats);
386                    continue;
387                }
388                // Metered POST-processing: the bar (and dominant-speaker ranking)
389                // reflect what peers actually hear, gain included.
390                let level = level_from_rms(frame.rms());
391                levels.send_modify(|l| l.mic = level);
392                stats.set_self_level(level);
393                let _ = pcm_tx.try_send(frame);
394            }
395            _ => {
396                // The capture ring stopped (the cpal input error callback fired — device
397                // gone, config invalidated, or an automatic reroute) or `next_frame`
398                // panicked. Rebuilding is REQUIRED: the device rate and the resampler are
399                // captured at open, so a stream that silently rerouted to a 24 kHz HFP mic
400                // would keep resampling as if it were still 48 k. Announce it — this path
401                // used to drop the device and sleep with nothing in the log, so a mid-call
402                // mic route change was invisible.
403                tracing::info!(
404                    track = "mic",
405                    device_rate = mic.as_ref().map(|m| m.device_rate()).unwrap_or(0),
406                    "mic: capture stream ended — reopening the device"
407                );
408                drop(mic.take());
409                // The route is gone; the last level it produced must not outlive it.
410                silence_mic_meter(&levels, &stats);
411                std::thread::sleep(std::time::Duration::from_millis(200));
412            }
413        }
414    }
415}

417/// The `!Send` screen APP-AUDIO capture thread: block on the engine `ScreenAudio`
418/// (the mic-thread analogue — no APM, no levels, no hot-swap) and forward each 20 ms
419/// PCM frame to the publish task. A `next_frame` error/panic is the leg's clean end
420/// (its parent screen stream stopped); drop the leg on this owning thread and exit.
421pub(super) fn screen_audio_capture_loop(
422    stop: Arc<AtomicBool>,
423    mut audio: hound_engine::api::ScreenAudio,
424    pcm_tx: tokio::sync::mpsc::Sender<hound_engine::types::PcmFrame>,
425) {
426    let mut captured: u64 = 0;
427    while !stop.load(Ordering::Relaxed) {
428        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| audio.next_frame()));
429        match pulled {
430            Ok(Ok(pcm)) => {
431                captured += 1;
432                if captured == 1 || captured.is_multiple_of(500) {
433                    tracing::info!(captured, "screen: captured app-audio frames");
434                }
435                let _ = pcm_tx.try_send(pcm); // drop on backpressure
436            }
437            Ok(Err(e)) => {
438                tracing::warn!(%e, captured, "screen: app-audio capture ended (error)");
439                break;
440            }
441            Err(_panic) => {
442                tracing::warn!(captured, "screen: app-audio capture panicked");
443                break;
444            }
445        }
446    }
447    tracing::info!(captured, "screen: app-audio capture loop stopped");
448    drop(audio);
449}

451/// The async screen AUDIO publish task: the mic publish path MINUS APM / levels /
452/// hot-swap (app audio is not a mic). Opus-encodes each forwarded PCM frame and
453/// publishes it on the `screen_audio` track; the receiver plays it on the shared
454/// mix bus exactly like a mic.
455pub(super) fn spawn_screen_audio_publish(
456    relay: Arc<MoqRelaySession>,
457    path: String,
458    sender: u16,
459    mut pcm_rx: tokio::sync::mpsc::Receiver<hound_engine::types::PcmFrame>,
460    stats: Arc<CallStats>,
461    mute: Arc<MuteFlags>,
462) -> JoinHandle<()> {
463    // App-audio is audio: encode + publish on the RT audio runtime, off the video pool.
464    audio_runtime().spawn(async move {
465        let mut encoder = match OpusEncoder::new(SCREEN_AUDIO_BITRATE, OpusProfile::Music) {
466            Ok(e) => e,
467            Err(e) => {
468                tracing::error!(%e, "screen: failed to build app-audio Opus encoder");
469                return;
470            }
471        };
472        let mut publisher = match relay.publish_track(&path, SCREEN_AUDIO_TRACK, AUDIO_PRIORITY) {
473            Ok(p) => p,
474            Err(e) => {
475                tracing::error!(%e, "screen: failed to publish audio track");
476                return;
477            }
478        };
479        if publisher.wait_subscribed().await.is_err() {
480            return;
481        }
482        tracing::info!("screen: app-audio track subscribed — publishing");
483        let mut frame_seq = 0u32;
484        let mut sent: u64 = 0;
485        // The same suppression gate as the mic, and today a permanent no-op: the MUSIC
486        // profile deliberately runs without DTX (`OpusProfile::dtx`), so a shared
487        // track's silence is still coded at full rate and never reaches the DTX size
488        // band. It is wired anyway so the two audio publish legs cannot drift — the day
489        // the music profile's silence-boundary listening check lands and DTX flips on,
490        // app audio inherits suppression with no further change.
491        let mut silence = SilenceGate::default();
492        while let Some(pcm) = pcm_rx.recv().await {
493            // The broadcaster's "share audio" toggle: drop BEFORE encode, so a muted
494            // share costs zero CPU and zero wire data. Receivers read the gap as a
495            // quiet source (their starve metric is DTX-aware), and un-muting resumes
496            // instantly on this same track.
497            if mute.screen_audio.load(Ordering::Relaxed) {
498                continue;
499            }
500            let Ok(packet) = encoder.encode(&pcm) else {
501                continue;
502            };
503            let bytes = packet.data.len();
504            if !silence.publish(bytes, Instant::now()) {
505                continue;
506            }
507            let frame = WireFrame {
508                sender,
509                track: TrackKind::ScrAudio,
510                codec: WireCodec::Opus,
511                // Count-driven grouping (ordered substream); keyframe bit is moot for audio.
512                keyframe: false,
513                frame_seq,
514                timestamp_us: packet.timestamp_us.micros(),
515                data: packet.data,
516            };
517            frame_seq = frame_seq.wrapping_add(1);
518            if let Err(e) = publish_audio_one(&mut publisher, &frame) {
519                tracing::warn!(%e, "screen: publish audio frame failed");
520            } else {
521                // RTC-debug: one screen app-audio frame published.
522                stats.screen_audio_sent(bytes);
523                sent += 1;
524                if sent == 1 || sent.is_multiple_of(500) {
525                    tracing::info!(sent, "screen: published app-audio frames");
526                }
527            }
528        }
529        tracing::info!(sent, "screen: app-audio publish ended");
530    })
