1//! Outbound audio: the mic capture loop + Opus publish, and the screen APP-AUDIO
2//! leg (its `!Send` capture thread + publish task). Encode + publish run on the
3//! RT audio runtime ("audio is sacred": a screenshare encode can't stall the
4//! mic), and audio frames publish via the ordered-substream path (count-rolled
5//! groups) so successive Opus frames ride one in-order QUIC stream — never one

12/// Spawn the mic publish task. A blocking capture thread pulls PCM from the engine
13/// `Microphone` (hot-swappable via `device_rx`); the async half Opus-encodes and
14/// publishes each frame on the relay session's `mic` track.
15///
16/// Muting gates the encoder at BOTH ends of the PCM channel — capture-side (frames
17/// are never queued) and publish-side (anything already queued is discarded before
18/// encode) — and zeroes the reported level. Both are needed: the channel holds 8
19/// frames, so a capture-only gate lets ~160 ms of pre-mute audio keep publishing
20/// after the user hits mute.
21#[allow(clippy::too_many_arguments)]
22pub(super) fn spawn_mic_publish(
23    relay: Arc<MoqRelaySession>,
24    space_id: Uuid,
25    self_user_id: Uuid,
26    self_id: PeerId,
27    clock: CaptureClock,
28    device_rx: watch::Receiver<DeviceChoice<MicId>>,
29    mute: Arc<MuteFlags>,
30    levels: watch::Sender<LevelsView>,
31    stop: Arc<AtomicBool>,
32    stats: Arc<CallStats>,
33    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
34) -> JoinHandle<()> {
35    let (pcm_tx, mut pcm_rx) = tokio::sync::mpsc::channel::<hound_engine::types::PcmFrame>(8);
36
37    // Blocking capture thread (the cpal Microphone is !Send — own it on one thread).
38    let capture_stop = stop.clone();
39    let stats_capture = stats.clone();
40    // The publish task's own view of the mute flag — see the gate in its loop below.
41    let mute_publish = mute.clone();
42    std::thread::Builder::new()
43        .name("call-mic-capture".into())
44        .spawn(move || {
45            mic_capture_loop(
46                capture_stop,
47                clock,
48                device_rx,
49                pcm_tx,
50                mute,
51                levels,
52                stats_capture,
53                #[cfg(feature = "synthetic")]
54                capture_source,
55            )
56        })
57        .ok();
58
59    let path = broadcast_path(space_id, self_user_id, self_id, MIC_TRACK);
60    let sender = self_id.0 as u16;
61    // Outbound voice is sacred too: encode + publish on the RT audio runtime so a
62    // screenshare encode can't stall the mic and make us stutter for everyone else.
63    audio_runtime().spawn(async move {
64        let mut encoder = match OpusEncoder::new(Bitrate::kbps(24), OpusProfile::Voice) {
65            Ok(e) => e,
66            Err(e) => {
67                tracing::error!(%e, "mic: failed to build Opus encoder");
68                return;
69            }
70        };
71        let mut frame_seq = 0u32;
72        let mut publisher = match relay.publish_track(&path, MIC_TRACK, AUDIO_PRIORITY) {
73            Ok(p) => p,
74            Err(e) => {
75                tracing::error!(%e, "mic: failed to publish track");
76                return;
77            }
78        };
79        if publisher.wait_subscribed().await.is_err() {
80            return;
81        }
82        while let Some(pcm) = pcm_rx.recv().await {
83            // Mute quiesces the encoder HERE as well as at capture. The capture loop
84            // already drops muted frames before `pcm_tx`, but that alone leaves a
85            // straggler window: up to the channel's 8 frames (~160 ms) are already
86            // queued when the flag flips, and without this check every one of them is
87            // still encoded and published — audio the user believes they muted.
88            //
89            // This is the same gate the screen-audio leg has always had (see
90            // `spawn_screen_audio_publish`); the mic simply never grew one. Checking
91            // at BOTH ends makes the quiesce synchronous with the flag rather than
92            // with the drain: the moment mute is set, nothing further is encoded, and
93            // `frame_seq` does not advance for a frame that was never sent. Receivers
94            // read the resulting gap as a quiet source (their starve metric is
95            // DTX-aware), and unmuting resumes instantly on this same track.
96            if mute_publish.mic.load(Ordering::Relaxed) {
97                continue;
98            }
99            let Ok(packet) = encoder.encode(&pcm) else {
100                continue;
101            };
102            let bytes = packet.data.len();
103            let frame = WireFrame {
104                sender,
105                track: TrackKind::CamAudio,
106                codec: WireCodec::Opus,
107                // Grouping is count-driven for audio (ordered substream), not keyframe-
108                // driven — every Opus frame is independently decodable, so the bit is moot.
109                keyframe: false,
110                frame_seq,
111                timestamp_us: packet.timestamp_us.micros(),
112                data: packet.data,
113            };
114            frame_seq = frame_seq.wrapping_add(1);
115            if let Err(e) = publish_audio_one(&mut publisher, &frame) {
116                tracing::warn!(%e, "mic: publish frame failed");
117            } else {
118                // RTC-debug: one Opus mic frame published (drives outbound send_kbps).
119                stats.audio_sent(bytes);
120            }
121        }
122    })
123}

125/// Publish one AUDIO frame via the ordered-substream path (count-rolled groups), so
126/// successive Opus frames ride one in-order QUIC stream instead of one stream per frame
127/// (the source of the cross-stream reordering). Use for mic + screen-audio.
128fn publish_audio_one(
129    publisher: &mut MoqTrackPublisher,
130    frame: &WireFrame,
131) -> std::result::Result<(), String> {
132    publisher
133        .publish_audio_frame(frame)
134        .map_err(|e| e.to_string())
135}

137/// Open the mic per `capture_source`: real hardware, or (behind `synthetic`)
138/// the engine's mic-free test tone. The one seam every (re)open site in
139/// [`mic_capture_loop`] routes through, so a headless build never needs its
140/// own copy of the branch.
141///
142/// The error is PROPAGATED (it used to be `.ok()`-swallowed): a TCC-denied mic is
143/// a typed [`AudioError::PermissionDenied`] the loop reports and paces off, not an
144/// indistinguishable `None` that spins silently forever.
145fn open_mic(
146    choice: &DeviceChoice<MicId>,
147    clock: CaptureClock,
148    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
149) -> std::result::Result<hound_engine::api::Microphone, hound_engine::error::AudioError> {
150    #[cfg(feature = "synthetic")]
151    if capture_source != CaptureSource::Default {
152        return Engine::new().open_synthetic_microphone(clock);
153    }
154    Engine::new().open_microphone(choice, clock)
155}

157/// Force the mic meter to silence.
158///
159/// THE invariant, and it is a privacy one: **the meter must read 0 whenever capture
160/// is not delivering.** The speaking reporter (`call::speaking`) samples this very
161/// watch and re-asserts whatever band it reads every 600 ms for as long as it is
162/// non-zero — so a level left latched by a dead BT mic or a mute that raced a
163/// capture stall does not merely mis-draw a bar, it makes its owner the call's
164/// permanent dominant speaker and leaks their pre-mute level to every peer. Every
165/// non-delivering path (mute, open failure, capture fault, no device) calls this.
166fn silence_mic_meter(levels: &watch::Sender<LevelsView>, stats: &CallStats) {
167    levels.send_modify(|l| l.mic = 0.0);
168    stats.set_self_level(0.0);
169}

171/// The mic capture loop: pull PCM, gate on mute, meter the level, forward to the
172/// encoder. Hot-swaps the device when `device_rx` changes; re-opens on error.
173#[allow(clippy::too_many_arguments)]
174fn mic_capture_loop(
175    stop: Arc<AtomicBool>,
176    clock: CaptureClock,
177    mut device_rx: watch::Receiver<DeviceChoice<MicId>>,
178    pcm_tx: tokio::sync::mpsc::Sender<hound_engine::types::PcmFrame>,
179    mute: Arc<MuteFlags>,
180    levels: watch::Sender<LevelsView>,
181    stats: Arc<CallStats>,
182    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
183) {
184    use hound_engine::api::level_from_rms;
185
186    let mut choice = device_rx.borrow_and_update().clone();
187    // The device is opened from ONE place (top of the loop), so every open — first,
188    // hot-swap, warmup, post-fault — reports and paces through the same policy.
189    let mut mic: Option<hound_engine::api::Microphone> = None;
190    let mut warmup = MicRouteWarmup { due: None };
191    // Cleared for the reopen the warmup itself triggers: one warmup per open cycle
192    // (re-arming there would reopen the device every ROUTE_WARMUP_DELAY forever).
193    let mut arm_warmup = true;
194    // Open-failure reporting + retry cadence. A denied mic used to be swallowed by
195    // `.ok()` (silent 200 ms spin, nothing in the logs or Sentry); now it reports
196    // once per transition and retries slowly, since the grant can still arrive.
197    let mut open_retry = OpenRetry::default();
198    let mut silence = MicSilenceWatch::default();
199    // Last capture-ring overflow total seen from the CURRENT device. The counter is
200    // per-open, so a value below the last read means a reopen/hot-swap minted a fresh
201    // one — count it whole rather than under-reporting until it passes the old total.
202    let mut last_overflow: u64 = 0;
203
204    while !stop.load(Ordering::Relaxed) {
205        if device_rx.has_changed().unwrap_or(false) {
206            choice = device_rx.borrow_and_update().clone();
207            drop(mic.take());
208            // Disarm the OLD device's pending warmup: were it left due, it could
209            // fire in the gap before the reopen below and burn the NEW device's
210            // one warmup cycle on a device it never belonged to.
211            warmup = MicRouteWarmup { due: None };
212            arm_warmup = true;
213        }
214        // BT/HFP route warmup: one reopen after the SCO link has had time to
215        // settle, mirroring the playout side's warmup in the mix bus.
216        if warmup.fire(Instant::now()) {
217            tracing::info!("voice: warmup reopen of non-48k (BT/HFP) mic — silent-route fix");
218            drop(mic.take());
219            arm_warmup = false;
220        }
221        if mic.is_none() {
222            match open_mic(
223                &choice,
224                clock,
225                #[cfg(feature = "synthetic")]
226                capture_source,
227            ) {
228                Ok(m) => {
229                    // The rate is the one fact that identifies which route we actually
230                    // landed on after a fault/hot-swap reopen (48 k built-in vs 24 k HFP)
231                    // — and it is what the capture resampler is pinned to for this open.
232                    // Every open used to be silent, so a mid-call device change left no
233                    // trace at all.
234                    tracing::debug!(
235                        track = "mic",
236                        device_rate = m.device_rate(),
237                        "mic: capture device opened"
238                    );
239                    if arm_warmup {
240                        warmup = MicRouteWarmup::arm(m.device_rate(), Instant::now());
241                    }
242                    arm_warmup = true;
243                    mic = Some(m);
244                    open_retry.opened();
245                }
246                Err(e) => {
247                    // Classify, then report ONCE per transition into this failure
248                    // class — the Sentry bridge promotes `error!` to an event, so a
249                    // sticky denial must not log per tick. Keep looping either way:
250                    // the device (or the grant) can still show up.
251                    let failure = OpenFailure::of_audio(&e);
252                    let (log, wait) = open_retry.failed(failure);
253                    if log {
254                        match failure {
255                            OpenFailure::PermissionDenied => tracing::error!(
256                                track = "mic",
257                                %e,
258                                "microphone permission denied — enable hound in System Settings → \
259                                 Privacy & Security → Microphone"
260                            ),
261                            OpenFailure::Other => tracing::warn!(
262                                track = "mic",
263                                %e,
264                                "mic: microphone open failed — retrying"
265                            ),
266                        }
267                    }
268                    // Nothing is being captured, so nothing may still be metered.
269                    silence_mic_meter(&levels, &stats);
270                    std::thread::sleep(wait);
271                    continue;
272                }
273            }
274        }
275        let Some(cap) = mic.as_mut() else {
276            silence_mic_meter(&levels, &stats);
277            continue;
278        };
279        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| cap.next_frame()));
280        match pulled {
281            Ok(Ok(frame)) => {
282                // RTC-debug: one PCM frame captured from the mic device, plus any
283                // samples the capture ring dropped since the last frame — the ONLY
284                // local signal that the far end is hearing chopped words.
285                stats.audio_captured();
286                let overflow = cap.capture_overflow_samples();
287                stats.capture_overflowed(if overflow >= last_overflow {
288                    overflow - last_overflow
289                } else {
290                    overflow
291                });
292                last_overflow = overflow;
293                if silence.observe(&frame.samples) {
294                    tracing::warn!(
295                        device_rate = cap.device_rate(),
296                        "voice: mic delivering PURE SILENCE (~3 s of bit-exact zeros) — \
297                         dead capture route (BT/SCO not live, permission, or a silent \
298                         virtual device)"
299                    );
300                }
301                if mute.mic.load(Ordering::Relaxed) {
302                    silence_mic_meter(&levels, &stats);
303                    continue;
304                }
305                let level = level_from_rms(frame.rms());
306                levels.send_modify(|l| l.mic = level);
307                stats.set_self_level(level);
308                let _ = pcm_tx.try_send(frame);
309            }
310            _ => {
311                // The capture ring stopped (the cpal input error callback fired — device
312                // gone, config invalidated, or an automatic reroute) or `next_frame`
313                // panicked. Rebuilding is REQUIRED: the device rate and the resampler are
314                // captured at open, so a stream that silently rerouted to a 24 kHz HFP mic
315                // would keep resampling as if it were still 48 k. Announce it — this path
316                // used to drop the device and sleep with nothing in the log, so a mid-call
317                // mic route change was invisible.
318                tracing::info!(
319                    track = "mic",
320                    device_rate = mic.as_ref().map(|m| m.device_rate()).unwrap_or(0),
321                    "mic: capture stream ended — reopening the device"
322                );
323                drop(mic.take());
324                // The route is gone; the last level it produced must not outlive it.
325                silence_mic_meter(&levels, &stats);
326                std::thread::sleep(std::time::Duration::from_millis(200));
327            }
328        }
329    }
330}

332/// The `!Send` screen APP-AUDIO capture thread: block on the engine `ScreenAudio`
333/// (the mic-thread analogue — no APM, no levels, no hot-swap) and forward each 20 ms
334/// PCM frame to the publish task. A `next_frame` error/panic is the leg's clean end
335/// (its parent screen stream stopped); drop the leg on this owning thread and exit.
336pub(super) fn screen_audio_capture_loop(
337    stop: Arc<AtomicBool>,
338    mut audio: hound_engine::api::ScreenAudio,
339    pcm_tx: tokio::sync::mpsc::Sender<hound_engine::types::PcmFrame>,
340) {
341    let mut captured: u64 = 0;
342    while !stop.load(Ordering::Relaxed) {
343        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| audio.next_frame()));
344        match pulled {
345            Ok(Ok(pcm)) => {
346                captured += 1;
347                if captured == 1 || captured.is_multiple_of(500) {
348                    tracing::info!(captured, "screen: captured app-audio frames");
349                }
350                let _ = pcm_tx.try_send(pcm); // drop on backpressure
351            }
352            Ok(Err(e)) => {
353                tracing::warn!(%e, captured, "screen: app-audio capture ended (error)");
354                break;
355            }
356            Err(_panic) => {
357                tracing::warn!(captured, "screen: app-audio capture panicked");
358                break;
359            }
360        }
361    }
362    tracing::info!(captured, "screen: app-audio capture loop stopped");
363    drop(audio);
364}

366/// The async screen AUDIO publish task: the mic publish path MINUS APM / levels /
367/// hot-swap (app audio is not a mic). Opus-encodes each forwarded PCM frame and
368/// publishes it on the `screen_audio` track; the receiver plays it on the shared
369/// mix bus exactly like a mic.
370pub(super) fn spawn_screen_audio_publish(
371    relay: Arc<MoqRelaySession>,
372    path: String,
373    sender: u16,
374    mut pcm_rx: tokio::sync::mpsc::Receiver<hound_engine::types::PcmFrame>,
375    stats: Arc<CallStats>,
376    mute: Arc<MuteFlags>,
377) -> JoinHandle<()> {
378    // App-audio is audio: encode + publish on the RT audio runtime, off the video pool.
379    audio_runtime().spawn(async move {
380        let mut encoder = match OpusEncoder::new(SCREEN_AUDIO_BITRATE, OpusProfile::Music) {
381            Ok(e) => e,
382            Err(e) => {
383                tracing::error!(%e, "screen: failed to build app-audio Opus encoder");
384                return;
385            }
386        };
387        let mut publisher = match relay.publish_track(&path, SCREEN_AUDIO_TRACK, AUDIO_PRIORITY) {
388            Ok(p) => p,
389            Err(e) => {
390                tracing::error!(%e, "screen: failed to publish audio track");
391                return;
392            }
393        };
394        if publisher.wait_subscribed().await.is_err() {
395            return;
396        }
397        tracing::info!("screen: app-audio track subscribed — publishing");
398        let mut frame_seq = 0u32;
399        let mut sent: u64 = 0;
400        while let Some(pcm) = pcm_rx.recv().await {
401            // The broadcaster's "share audio" toggle: drop BEFORE encode, so a muted
402            // share costs zero CPU and zero wire data. Receivers read the gap as a
403            // quiet source (their starve metric is DTX-aware), and un-muting resumes
404            // instantly on this same track.
405            if mute.screen_audio.load(Ordering::Relaxed) {
406                continue;
407            }
408            let Ok(packet) = encoder.encode(&pcm) else {
409                continue;
410            };
411            let bytes = packet.data.len();
412            let frame = WireFrame {
413                sender,
414                track: TrackKind::ScrAudio,
415                codec: WireCodec::Opus,
416                // Count-driven grouping (ordered substream); keyframe bit is moot for audio.
417                keyframe: false,
418                frame_seq,
419                timestamp_us: packet.timestamp_us.micros(),
420                data: packet.data,
421            };
422            frame_seq = frame_seq.wrapping_add(1);
423            if let Err(e) = publish_audio_one(&mut publisher, &frame) {
424                tracing::warn!(%e, "screen: publish audio frame failed");
425            } else {
426                // RTC-debug: one screen app-audio frame published.
427                stats.screen_audio_sent(bytes);
428                sent += 1;
429                if sent == 1 || sent.is_multiple_of(500) {
430                    tracing::info!(sent, "screen: published app-audio frames");
431                }
432            }
433        }
434        tracing::info!(sent, "screen: app-audio publish ended");
435    })
