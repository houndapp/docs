1//! The dominant-speaker plumbing: the shared [`SpeakerRanking`] every part of one
2//! establish reads, and the task that reports THIS client's own level.
3//!
4//! The policy itself is in `hound_media::speaking` (pure, virtual-clock, tested
5//! there). This file is only the IO shell around it:
6//!
7//! * [`SharedSpeakers`] — the ranker behind a mutex plus the monotonic epoch that
8//!   turns wall time into the `now_us` the core wants. Written by the control
9//!   reader (one report per speaking peer per 300 ms), read by the subscribe
10//!   reconcile (twice a second). Per-establish, like the roster: peer ids are
11//!   assigned per `Welcome`, so a ranking that outlived a reconnect would promote
12//!   whoever inherits a departed speaker's id.
13//! * [`spawn_speaking_report`] — samples our own mic meter on the same 300 ms grid
14//!   and puts a coarse subband on the control plane, suppressed while silent.
15
16use super::*;
17
18use hound_media::speaking::{REPORT_INTERVAL_US, SpeakerRanking, SpeakerTiers};
19
20/// The per-establish ranking, shared between the control reader and the subscribe
21/// loop. A plain `Mutex`: contention is a handful of 300 ms reports against one
22/// 2 Hz reader, and neither side holds it across an await.
23#[derive(Clone)]
24pub(super) struct SharedSpeakers(Arc<Inner>);
25
26struct Inner {
27    /// This establish's time origin. The core is clock-free by design (its tests
28    /// drive a virtual `now_us`); the shell owns the one real clock, and both
29    /// writers and readers derive `now_us` from THIS epoch so the sample grid is
30    /// consistent no matter which task advances it.
31    epoch: Instant,
32    ranking: std::sync::Mutex<SpeakerRanking>,
33}
34
35impl SharedSpeakers {
36    pub(super) fn new() -> Self {
37        Self(Arc::new(Inner {
38            epoch: Instant::now(),
39            ranking: std::sync::Mutex::new(SpeakerRanking::new()),
40        }))
41    }
42
43    fn now_us(&self) -> i64 {
44        self.0.epoch.elapsed().as_micros() as i64
45    }
46
47    /// Recover from a poisoned mutex rather than propagate a panic: this guards
48    /// pure bookkeeping, and a panic elsewhere must not take the subscribe loop —
49    /// and with it every inbound stream — down with it. Same rule as `TilePolicies`.
50    fn lock(&self) -> std::sync::MutexGuard<'_, SpeakerRanking> {
51        self.0.ranking.lock().unwrap_or_else(|e| e.into_inner())
52    }
53
54    /// A peer reported its level (the control reader's hook).
55    pub(super) fn observe(&self, peer: PeerId, level: u8) {
56        let now = self.now_us();
57        self.observe_at(peer, level, now);
58    }
59
60    /// [`Self::observe`] with the clock supplied — the seam that keeps the shell as
61    /// deterministic as the core it wraps.
62    fn observe_at(&self, peer: PeerId, level: u8, now_us: i64) {
63        self.lock().observe(peer.0, level, now_us);
64    }
65
66    /// Advance the decision grid and snapshot the tiers — the subscribe loop's one
67    /// call per reconcile pass. `keep` prunes departed peers in the same breath, so
68    /// a leave can never strand a promotion (the roster is the only liveness signal
69    /// here, exactly as it is for tile demand).
70    pub(super) fn decide(&self, keep: impl Fn(u32) -> bool) -> SpeakerTiers {
71        let now = self.now_us();
72        self.decide_at(keep, now)
73    }
74
75    /// [`Self::decide`] with the clock supplied.
76    fn decide_at(&self, keep: impl Fn(u32) -> bool, now_us: i64) -> SpeakerTiers {
77        let mut ranking = self.lock();
78        ranking.retain(keep);
79        if ranking.decide(now_us)
80            && let Some(dominant) = ranking.tiers().dominant()
81        {
82            tracing::info!(peer = dominant, "call: dominant speaker changed");
83        }
84        ranking.tiers()
85    }
86}

88/// How long a non-silent report may go unrepeated before the receiver's ranker
89/// would read the peer as gone quiet. The emitter only sends on a *change*, so a
90/// peer talking steadily at one band would otherwise fall off the grid; re-asserting
91/// well inside [`hound_media::speaking::IDLE_US`] keeps a steady speaker steady
92/// without paying the full 3 Hz for an unchanging level.
93///
94/// Expressed in TICKS, because the naive `IDLE_US / 2` does not survive the grid: a
95/// re-assert can only land on a [`REPORT_INTERVAL_US`] tick, so the real period is
96/// this value rounded UP to whole ticks, and the ticker runs
97/// `MissedTickBehavior::Delay` — a late tick pushes the whole grid out rather than
98/// being chased. `IDLE_US / 2` = 1.0 s therefore lands at 1.2 s and leaves only
99/// 0.8 s of slack under the receiver's 2 s decay, which one scheduler hiccup on a
100/// loaded encode thread eats whole (a steady talker then decays mid-sentence and
101/// loses the floor it is currently holding). **Two ticks = 600 ms** leaves 1.4 s,
102/// for one extra ~40-byte frame per talking peer per 1.2 s.
103const SPEAKING_REASSERT: Duration = Duration::from_micros(2 * REPORT_INTERVAL_US as u64);
104
105/// The emitter's decision, as a pure policy so the wire cost is a unit test rather
106/// than a claim: **send on change, re-assert a non-silent band before the receiver
107/// would time it out, and otherwise say nothing at all.**
108///
109/// The asymmetry is the point. A change must be reported at once — dominance is
110/// evidence about *now* — while an unchanged band is only worth repeating because
111/// the receiver's ranker decays a peer it has not heard from
112/// ([`hound_media::speaking::IDLE_US`]). Silence needs neither: one trailing zero
113/// says it, and the same decay covers a report that never arrives.
114#[derive(Debug, Default)]
115struct ReportGate {
116    last: Option<u8>,
117    sent_at: Option<Instant>,
118}
119
120impl ReportGate {
121    /// Should this sample go on the wire? Records it if so.
122    fn should_send(&mut self, band: u8, now: Instant) -> bool {
123        let stale = band > 0
124            && self
125                .sent_at
126                .is_none_or(|t| now.duration_since(t) >= SPEAKING_REASSERT);
127        if self.last == Some(band) && !stale {
128            return false;
129        }
130        self.last = Some(band);
131        self.sent_at = Some(now);
132        true
133    }
134}

136/// Report this client's own speaking level on the control plane.
137///
138/// Cadence: one sample per [`REPORT_INTERVAL_US`] (Jitsi's 300 ms decision
139/// interval), but a frame is only put on the wire when the coarse band **changes**,
140/// or to re-assert a non-silent band before [`SPEAKING_REASSERT`]. So:
141///
142/// * a quiet participant sends one trailing zero when they stop and then nothing at
143///   all — a 32-person call where nobody speaks costs the control plane zero;
144/// * a talking participant sends a few frames a second (each ~40 bytes of JSON),
145///   which against the ~130 kbps their thumbnail rung already costs is noise.
146///
147/// The level itself is free: the mic capture loop already computes it for the level
148/// meter (`level_from_rms`) and publishes it on the same `levels` watch the UI
149/// reads, and it is forced to zero while muted — so a muted peer reports silence
150/// without this task knowing what mute is.
151pub(super) fn spawn_speaking_report(
152    mut levels: watch::Receiver<LevelsView>,
153    control_outbound: mpsc::UnboundedSender<ClientMessage>,
154    stop: Arc<AtomicBool>,
155) -> JoinHandle<()> {
156    call_runtime().spawn(async move {
157        let period = Duration::from_micros(REPORT_INTERVAL_US as u64);
158        let mut ticker = tokio::time::interval(period);
159        // A late tick must not be chased with a burst — the cadence IS the anti-flap
160        // budget, and a burst of stale levels would be a burst of stale evidence.
161        ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Delay);
162        let mut gate = ReportGate::default();
163        while !stop.load(Ordering::Relaxed) {
164            ticker.tick().await;
165            let band = hound_media::speaking_subband(levels.borrow_and_update().mic);
166            if !gate.should_send(band, Instant::now()) {
167                continue;
168            }
169            if control_outbound
170                .send(ClientMessage::SpeakingLevel { level: band })
171                .is_err()
172            {
173                break; // control plane gone — the establish is tearing down
174            }
175        }
176    })
177}

179#[cfg(test)]
180mod tests {
181    use super::*;
182    use hound_media::SpeakerTier;
183    use hound_media::speaking::{IDLE_US, SPEECH_SUBBAND, SUBBANDS};
184
185    /// The shell's job in one test: reports go in over the shared handle, a
186    /// ranking comes out, and the roster bounds its lifetime. (The ranking policy
187    /// itself is `hound_media::speaking`'s to prove; this is the plumbing.)
188    #[test]
189    fn reports_rank_a_speaker_over_the_shared_handle() {
190        let speakers = SharedSpeakers::new();
191        let mut now = 0;
192        // ~12 s of one peer talking and one below the speech floor, on the grid.
193        for _ in 0..40 {
194            speakers.observe_at(PeerId(1), SUBBANDS - 1, now);
195            speakers.observe_at(PeerId(2), SPEECH_SUBBAND - 1, now);
196            now += REPORT_INTERVAL_US;
197            speakers.decide_at(|_| true, now);
198        }
199        let tiers = speakers.decide_at(|_| true, now);
200        assert_eq!(tiers.dominant(), Some(1));
201        assert_eq!(tiers.tier(2), SpeakerTier::Silent);
202
203        // A departed peer takes its rank with it — the promotion must not be
204        // inherited by whoever gets peer id 1 on the next establish.
205        let tiers = speakers.decide_at(|p| p != 1, now);
206        assert_eq!(tiers.dominant(), None);
207    }
208
209    /// The wire cost, which is the whole justification for putting a 3 Hz signal on
210    /// the control plane: silence costs one frame and then nothing, a change costs
211    /// one frame, and a steady band costs one frame per re-assert — never one per
212    /// 300 ms tick.
213    #[test]
214    fn silence_is_free_and_a_steady_band_only_re_asserts() {
215        let mut gate = ReportGate::default();
216        let t0 = Instant::now();
217        let tick = |n: u32| t0 + Duration::from_micros(REPORT_INTERVAL_US as u64 * n as u64);
218
219        // A silent call: the first sample states the band, the rest say nothing.
220        assert!(gate.should_send(0, tick(0)));
221        for n in 1..100 {
222            assert!(!gate.should_send(0, tick(n)), "silence must cost nothing");
223        }
224        // Speech: reported the moment it changes.
225        assert!(gate.should_send(9, tick(100)));
226        // …then held: one re-assert per window, not one per tick, and always
227        // inside the window the receiver's ranker times a peer out on.
228        let mut sent = 0;
229        for n in 101..200 {
230            sent += u32::from(gate.should_send(9, tick(n)));
231        }
232        // A re-assert can only land on a sample tick, so its period is the window
233        // rounded UP to whole ticks — 2 of them, i.e. one frame per 600 ms held.
234        let per_reassert = SPEAKING_REASSERT
235            .as_micros()
236            .div_ceil(REPORT_INTERVAL_US as u128);
237        let expected = (99 / per_reassert) as u32;
238        assert_eq!(sent, expected, "a steady band must only re-assert");
239        // THE margin that keeps a steady talker ranked. The worst-case gap between
240        // two re-asserts is the tick-rounded period, and `MissedTickBehavior::Delay`
241        // means a stalled tick pushes it out further still — so the slack under the
242        // receiver's decay is what a scheduler hiccup gets to eat. Half the idle
243        // window is not slack, it is a coin flip.
244        let worst_gap = per_reassert as i64 * REPORT_INTERVAL_US;
245        assert!(
246            IDLE_US - worst_gap >= worst_gap,
247            "a re-assert every {worst_gap}µs leaves only {}µs under the receiver's \
248             {IDLE_US}µs decay — less than one whole period of slack",
249            IDLE_US - worst_gap
250        );
251        // Falling quiet is a change, so it is reported at once — that single
252        // trailing zero is what makes the following silence readable.
253        assert!(gate.should_send(0, tick(200)));
254        assert!(!gate.should_send(0, tick(201)));
255    }
