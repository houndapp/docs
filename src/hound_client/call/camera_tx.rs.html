1//! Outbound camera: the `!Send` capture+encode thread (device hot-swap,
2//! self-preview tee, adaptive encode) and the async publish task running the 1 s
3//! BWE adaptation tick. The capture side reads its decisions off
4//! [`adapt::AdaptShared`]; the codec is negotiated against the roster and
5//! re-driven mid-call by the re-negotiator. Self-preview keeps running even
6//! while the ladder pauses video (AudioOnly), so the local tile never freezes.
7

10// ─── Publish: camera (capture → H.264 → MoQ + self-preview) ─────────────────────
11
12/// Spawn the camera publish task. When `enabled`, a blocking thread opens the
13/// engine `Camera`, pushes each frame's I420 to the self-preview sink (`"self"`),
14/// H.264-encodes it, and queues the access units for the async publish half. When
15/// disabled, capture stops (the device is released). Self-preview works even if
16/// the publish/subscribe never gets a remote peer.
17///
18/// Sender-side BWE adaptation: a 1 s tick folds `rtt_conn`'s
19/// RTT + the worst receiver-reported loss (from the per-track feedback listener) into
20/// a target the capture/encode loop applies (bitrate retarget, fps gate, downscale to
21/// the rung, AudioOnly pause), and the keyframe listener arms forced IDRs on PLI. The
22/// adaptation is per-establish — rebuilt fresh on each reconnect (correct: the rate
23/// controller starts clean against the new connection).
24#[allow(clippy::too_many_arguments)]
25pub(super) fn spawn_cam_publish(
26    relay: Arc<MoqRelaySession>,
27    space_id: Uuid,
28    self_user_id: Uuid,
29    self_id: PeerId,
30    clock: CaptureClock,
31    device_rx: watch::Receiver<DeviceChoice<CameraId>>,
32    enabled: Arc<AtomicBool>,
33    sink: Arc<dyn VideoSink>,
34    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
35    stop: Arc<AtomicBool>,
36    rtt_conn: iroh::endpoint::Connection,
37    view_rx: watch::Receiver<SessionView>,
38    keyframe_rx: broadcast::Receiver<TrackId>,
39    feedback_rx: broadcast::Receiver<MediaFeedback>,
40    quality_rx: broadcast::Receiver<QualityReport>,
41    stats: Arc<CallStats>,
42    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
43) -> JoinHandle<()> {
44    let (enc_tx, mut enc_rx) = tokio::sync::mpsc::channel::<WireFrame>(8);
45
46    // The shared adaptation seam between the async tick (below) and the capture
47    // thread. The keyframe/feedback listeners write into it too; the encode loop reads.
48    let adapt = Arc::new(AdaptShared::default());
49
50    // Negotiate the camera codec against the live roster (HW-encode ∩ all-peers
51    // decode) and seed the slot BEFORE the capture thread starts so its first encoder
52    // is built in the right codec. `encode_caps` is probed once and reused by the
53    // re-negotiator below.
54    //
55    // No `openh264::settle()` here (this fn is sync): the control plane already
56    // awaited it during `connect`, which every call passes through before any
57    // publish starts, so by now the first-run download has either landed or hit
58    // its ceiling.
59    let encode_caps = Engine::new().encode_caps();
60    let codec = negotiate_codec(encode_caps, &view_rx.borrow());
61    adapt
62        .target_codec
63        .store(codec_rank(codec), Ordering::Relaxed);
64
65    // Demand-driven simulcast control for the camera source + the LOW-layer channel
66    // the capture thread feeds. `low_active` is the capture thread's gate; it stays
67    // `None` (low path inert) until a receiver demands the low layer AND actuation is
68    // on. Created here so it can be handed to the capture thread below.
69    let layers = Arc::new(LayerControl::new(CAM_BITRATE.bps() as u64));
70    let low_active = layers.low_active.subscribe();
71    let (low_vid_tx, low_vid_rx) = tokio::sync::mpsc::channel::<WireFrame>(8);
72    // A low-track PLI sets this; the capture thread forces a low IDR so a receiver
73    // subscribing to the low layer resyncs promptly (not on the slow safety-net IDR).
74    let low_keyframe = Arc::new(AtomicBool::new(false));
75    // The LOW encoder's live codec (`codec_rank`), written by the capture thread on
76    // every low build and read by the publish task when it announces the `_l0` decl.
77    // Shared rather than captured because the announce lands on the first low frame,
78    // which can be long after a mid-call codec downgrade — see
79    // `spawn_low_video_publish`.
80    let low_codec = Arc::new(AtomicU8::new(codec_rank(codec)));
81
82    let capture_stop = stop.clone();
83    let adapt_for_capture = adapt.clone();
84    let stats_capture = stats.clone();
85    let low_keyframe_capture = low_keyframe.clone();
86    let low_codec_capture = low_codec.clone();
87    std::thread::Builder::new()
88        .name("call-cam-capture-encode".into())
89        .spawn(move || {
90            cam_capture_encode_loop(
91                capture_stop,
92                clock,
93                device_rx,
94                enabled,
95                sink,
96                enc_tx,
97                self_id.0 as u16,
98                adapt_for_capture,
99                stats_capture,
100                low_active,
101                low_vid_tx,
102                low_keyframe_capture,
103                low_codec_capture,
104                #[cfg(feature = "synthetic")]
105                capture_source,
106            )
107        })
108        .ok();
109
110    // Producer-side PLI relay + receiver→producer BWE feedback, scoped to OUR camera
111    // track. Their tasks ride the cam publish task (returned), so they tear down with
112    // it on leave/reconnect.
113    let keyframe_task = spawn_keyframe_listener(
114        keyframe_rx,
115        view_rx.clone(),
116        self_id,
117        Source::Camera,
118        adapt.clone(),
119        Some(low_keyframe.clone()),
120    );
121    let feedback_task = spawn_feedback_listener(
122        feedback_rx,
123        view_rx.clone(),
124        self_id,
125        Source::Camera,
126        adapt.clone(),
127        layers.clone(),
128    );
129    // The low-publish task's own view of the gate: it withdraws the `_l0` decl from
130    // the roster when the layer deactivates (subscribed before `layers` is moved).
131    let low_active_decl = layers.low_active.subscribe();
132    // Demand-driven simulcast: aggregate receiver quality requests for the camera
133    // track into the layer planner (drives the low layer's `low_active` gate). Dark
134    // until `SIMULCAST_ACTUATE`.
135    let quality_task =
136        spawn_quality_listener(quality_rx, view_rx.clone(), self_id, Source::Camera, layers);
137    // LOW-layer publish: drains the capture thread's low encode and publishes it on
138    // the `cam_l0` track, announcing it on the first frame (so it's invisible to
139    // receivers until actually producing) and withdrawing the decl again when the
140    // layer is torn down (so no receiver picks a layer that stopped producing).
141    let low_publish_task = spawn_low_video_publish(
142        relay.clone(),
143        broadcast_path(space_id, self_user_id, self_id, "cam_l0"),
144        "cam_l0",
145        Source::Camera,
146        low_codec,
147        low_vid_rx,
148        control_outbound.clone(),
149        view_rx.clone(),
150        self_id,
151        low_active_decl,
152    );
153    // Re-negotiate the codec as the roster changes mid-call (a less-capable peer
154    // joins → downgrade so they get video; leaves → climb back).
155    let renegotiate_task =
156        spawn_codec_renegotiator(view_rx, encode_caps, adapt.clone(), codec, "cam");
157
158    let path = broadcast_path(space_id, self_user_id, self_id, CAM_TRACK);
159    call_runtime().spawn(async move {
160        // Abort the listeners when this task ends (clean return or `leave`-time abort).
161        let _keyframe_task = AbortOnDrop(keyframe_task);
162        let _feedback_task = AbortOnDrop(feedback_task);
163        let _quality_task = AbortOnDrop(quality_task);
164        let _low_publish_task = AbortOnDrop(low_publish_task);
165        let _renegotiate_task = AbortOnDrop(renegotiate_task);
166
167        let mut publisher = match relay.publish_track(&path, CAM_TRACK, VIDEO_PRIORITY) {
168            Ok(p) => p,
169            Err(e) => {
170                tracing::error!(%e, "cam: failed to publish track");
171                return;
172            }
173        };
174        // RTC-debug: the camera track is announced in the negotiated codec. The
175        // `enabled` gauge is NOT set here — publishing a track says nothing about
176        // whether a camera is actually running (you can join with video off, or turn
177        // it off mid-call). The capture loop owns that gauge; see
178        // `cam_capture_encode_loop`.
179        stats.set_video_codec(video_codec_to_ctl(codec).label());
180
181        // Sender-side adaptation core (sans-IO), seeded at the encoder's static launch
182        // bitrate. The 1 s tick folds backpressure + RTT + remote loss into a decision
183        // the capture thread reads off `adapt`; ticks run from the start, forwarding
184        // waits on the live gate (see `drive_video_publish`). Per-track, but NOT
185        // independent of the screen share: `drive_video_publish` registers this
186        // publisher with the connection-level `SendCoordinator`, so while both
187        // publish the two ladders draw from one budget instead of both claiming the
188        // whole uplink.
189        let mut adaptation = VideoAdaptation::standard(CAM_BITRATE.bps() as u64);
190        // The encoder-rebuild geometry dwell, seeded at the capture's own native
191        // geometry (CAM_DIMS) — exactly what the capture thread's encoder is already
192        // built at before any tick runs, so a healthy call at the top rung never
193        // rebuilds. See `hound_media::GeometryDwell`.
194        let mut geometry = hound_media::GeometryDwell::new((CAM_DIMS.width(), CAM_DIMS.height()));
195        drive_video_publish(
196            &mut publisher,
197            &mut enc_rx,
198            &adapt,
199            &mut adaptation,
200            &mut geometry,
201            &rtt_conn,
202            &stats,
203            false,
204            "cam",
205            |_| {},
206        )
207        .await;
208    })
209}

211/// Open the camera per `capture_source`: real hardware at `CAM_DIMS`@`CAM_FPS`,
212/// or (behind `synthetic`) the engine's camera-free test pattern. The one seam
213/// `cam_capture_encode_loop`'s (re)open routes through.
214fn open_cam(
215    choice: &DeviceChoice<CameraId>,
216    clock: CaptureClock,
217    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
218) -> std::result::Result<hound_engine::api::Camera, hound_engine::error::CaptureError> {
219    #[cfg(feature = "synthetic")]
220    if capture_source != CaptureSource::Default {
221        let testcard = capture_source == CaptureSource::SyntheticTestcard;
222        return Engine::new().open_synthetic_camera(CAM_DIMS, CAM_FPS, clock, testcard);
223    }
224    Engine::new().open_camera(choice, CAM_DIMS, CAM_FPS, clock)
225}

227/// The camera capture+encode loop. Mirrors `crates/frb/src/camera.rs` for the
228/// self-preview push, and additionally H.264-encodes for publish. Honors the
229/// `enabled` flag (released device when off) and hot-swaps on `device_rx` change.
230/// Reads `adapt` each frame: retargets the encoder bitrate, downsamples to the rung
231/// fps, downscales to the rung geometry, serves a forced IDR, and skips encode/publish
232/// while paused (AudioOnly under severe congestion) or while the source publish is
233/// still dark (no subscriber — [`AdaptShared::source_encode_open`]). Self-preview keeps
234/// running in both cases, so the local tile never freezes.
235#[allow(clippy::too_many_arguments)]
236fn cam_capture_encode_loop(
237    stop: Arc<AtomicBool>,
238    clock: CaptureClock,
239    mut device_rx: watch::Receiver<DeviceChoice<CameraId>>,
240    enabled: Arc<AtomicBool>,
241    sink: Arc<dyn VideoSink>,
242    enc_tx: tokio::sync::mpsc::Sender<WireFrame>,
243    sender: u16,
244    adapt: Arc<AdaptShared>,
245    stats: Arc<CallStats>,
246    // Demand-driven LOW simulcast layer: `Some(bps)` ⇒ also downscale + encode this
247    // capture at the rung that cap buys (`hound_media::low_rung_for_cap`) and forward
248    // it on `low_vid_tx` (the `cam_l0` track). Stays `None` (inert, source path
249    // byte-identical) until a receiver demands it AND `SIMULCAST_ACTUATE` is on.
250    low_active: watch::Receiver<Option<u64>>,
251    low_vid_tx: tokio::sync::mpsc::Sender<WireFrame>,
252    low_keyframe: Arc<AtomicBool>,
253    // The low encoder's live codec (`codec_rank`) for the `cam_l0` publish task's decl
254    // announce — written here on every low build, since the ladder may land a rung
255    // below the negotiated target.
256    low_codec: Arc<AtomicU8>,
257    #[cfg(feature = "synthetic")] capture_source: CaptureSource,
258) {
259    let mut choice = device_rx.borrow_and_update().clone();
260    let mut camera = None;
261    let mut encoder: Option<hound_engine::api::VideoEncoder> = None;
262    let mut frame_seq = 0u32;
263    // The negotiated codec target the live encoder was BUILT for; the mid-call
264    // rebuild keys off this changing, not off the encoder's actual codec, so a target
265    // the device can't build doesn't re-trigger every frame.
266    let mut last_target = codec_from_rank(adapt.target_codec.load(Ordering::Relaxed));
267    // Throttles the retry of a rebuild that found no usable encoder. `last_target` is
268    // committed only on a successful open, so the pending switch survives a failure —
269    // this is what stops the surviving switch from re-attempting per frame.
270    let mut rebuild_retry = RebuildRetry::default();
271    // The geometry the live `encoder` was actually built at — starts at CAM_DIMS
272    // (native capture) and only moves when the adaptation's dwelled rung geometry
273    // (`adapt.encoder_w/h`) settles on something different. Passed to
274    // `encode_and_publish_adaptive` so it always scales the frame to match whatever
275    // the encoder really is.
276    let mut enc_dims = CAM_DIMS;
277    // The encoder bitrate we last applied, so a steady target costs no syscall.
278    let mut applied_bps = CAM_BITRATE.bps();
279    // Downsamples capture (CAM_FPS) to the chosen rung's fps.
280    let mut fps_gate = hound_media::FrameRateGate::new();
281    let mut fail_streak = EncodeFailStreak::default();
282    // Open-failure reporting + retry cadence. A denied camera used to be discarded
283    // here (silent 300 ms spin, nothing in the logs or Sentry); now it reports once
284    // per transition and retries slowly.
285    let mut open_retry = OpenRetry::default();
286
287    // LOW simulcast layer state. Built lazily the first frame `low_active` is
288    // `Some`, torn down when it goes `None`, rebuilt on a codec switch (tracks the
289    // same `target_codec` as the source) or on a RUNG change.
290    //
291    // The rung — geometry AND frame rate — is derived from the coalesced cap
292    // (`hound_media::low_rung_for_cap`), not pinned at 640×360@30: a call full of
293    // grid thumbnails now runs this encoder at 320×180@15. `low_geometry` is the
294    // same dwell the source path uses, seeded at the rung the first cap buys
295    // (`with_committed`) so activation is immediate while a LATER rung change still
296    // has to hold for `GEOMETRY_REBUILD_DWELL_US` before the encoder is rebuilt —
297    // a receiver resizing tiles must not churn the encoder, and an encoder whose
298    // geometry moves out from under the frames it is fed is the exact failure the
299    // dwell exists to prevent.
300    let mut low_encoder: Option<hound_engine::api::VideoEncoder> = None;
301    let mut low_last_target: Option<VideoCodec> = None;
302    let mut low_applied_bps = 0u32;
303    let mut low_frame_seq = 0u32;
304    let mut low_fps_gate = hound_media::FrameRateGate::new();
305    let mut low_fail_streak = EncodeFailStreak::default();
306    let mut low_geometry: Option<hound_media::GeometryDwell> = None;
307    // The rung the live `low_encoder` was actually built at (geometry + fps).
308    let mut low_rung: Option<(Dimensions, hound_media::VideoRung)> = None;
309    let low_adapt = Arc::new(AdaptShared::default());
310
311    while !stop.load(Ordering::Relaxed) {
312        // Camera disabled: release the device and idle.
313        if !enabled.load(Ordering::Relaxed) {
314            if camera.is_some() {
315                drop(camera.take());
316                drop(encoder.take());
317            }
318            // Tell the RTC-debug snapshot the camera pipeline is DOWN. Without this
319            // the gauge stays latched from whenever the camera last ran, and since
320            // `AdaptShared::live` is latched too, turning video off mid-call would
321            // leave the snapshot reading enabled + publish_live + 0 fps captured —
322            // which is precisely `video_report`'s send-capture-stalled signature. It
323            // would then diagnose a dead camera every window, WARN once the streak
324            // latches, and pollute the Sentry `video_stage` tag for the rest of the
325            // call, all for a user who simply turned their camera off.
326            stats.set_video_enabled(false);
327            std::thread::sleep(std::time::Duration::from_millis(100));
328            continue;
329        }
330
331        if device_rx.has_changed().unwrap_or(false) {
332            choice = device_rx.borrow_and_update().clone();
333            drop(camera.take());
334            drop(encoder.take());
335        }
336
337        // Open (or re-open) the camera + encoder.
338        if camera.is_none() {
339            match open_cam(
340                &choice,
341                clock,
342                #[cfg(feature = "synthetic")]
343                capture_source,
344            ) {
345                Ok(c) => {
346                    camera = Some(c);
347                    open_retry.opened();
348                }
349                Err(e) => {
350                    // Classify, then report ONCE per transition into this failure
351                    // class — the Sentry bridge promotes `error!` to an event, so a
352                    // sticky denial must not log per tick. Keep looping either way:
353                    // the camera (or the grant) can still show up.
354                    let failure = OpenFailure::of_capture(&e);
355                    let (log, wait) = open_retry.failed(failure);
356                    if log {
357                        match failure {
358                            OpenFailure::PermissionDenied => tracing::error!(
359                                track = "cam",
360                                %e,
361                                "camera permission denied — enable hound in System Settings → \
362                                 Privacy & Security → Camera"
363                            ),
364                            OpenFailure::Other => tracing::warn!(
365                                track = "cam",
366                                %e,
367                                "cam: camera open failed — retrying"
368                            ),
369                        }
370                    }
371                    std::thread::sleep(wait);
372                    continue;
373                }
374            }
375            // Build in the negotiated codec (HW-encode ∩ all-peers decode), seeded by
376            // `spawn_cam_publish` and re-driven by the mid-call re-negotiator,
377            // descending the ladder if it won't open.
378            let want = codec_from_rank(adapt.target_codec.load(Ordering::Relaxed));
379            // The SOURCE camera encoder asks for L1T2 — see
380            // `open_temporal_encoder_laddered`. The low encoder below does not.
381            encoder = open_temporal_encoder_laddered(
382                want,
383                CAM_DIMS,
384                CAM_FPS,
385                CAM_BITRATE,
386                CAM_KEYFRAME_CADENCE,
387            );
388            last_target = want;
389            if let Some(enc) = encoder.as_ref() {
390                use hound_engine::backend::VideoEncode;
391                tracing::info!(
392                    target: "video_encode",
393                    track = "cam",
394                    codec = ?enc.codec(),
395                    backend = enc.backend(),
396                    width = CAM_DIMS.width(),
397                    height = CAM_DIMS.height(),
398                    "video: encode path selected"
399                );
400            }
401            // Re-baseline so the first frame re-applies the live BWE target, and
402            // open at a fresh IDR for any subscriber that attached meanwhile. A
403            // (re)open always starts at native geometry — the geometry dwell only
404            // degrades it once a fresh adaptation decision settles.
405            applied_bps = CAM_BITRATE.bps();
406            enc_dims = CAM_DIMS;
407            adapt.force_keyframe.store(true, Ordering::Relaxed);
408            // The camera pipeline is UP: a device is open and about to deliver
409            // frames. This is the only place the gauge goes true, so it always means
410            // "capture is running" — the thing every capture/encode rule in
411            // `video_report` is judged against.
412            stats.set_video_enabled(true);
413        }
414
415        let Some(cap) = camera.as_mut() else {
416            continue;
417        };
418        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| cap.next_frame()));
419        let frame = match pulled {
420            Ok(Ok(f)) => f,
421            _ => {
422                drop(camera.take());
423                drop(encoder.take());
424                std::thread::sleep(std::time::Duration::from_millis(200));
425                continue;
426            }
427        };
428
429        // RTC-debug: one raw camera frame pulled from the device.
430        stats.video_captured();
431
432        // Self-preview keyed "self" (runs even while paused, so the local preview
433        // never freezes mid-call). Surface-first: a surface-backed camera frame is
434        // rendered zero-copy; a scaled/I420 frame falls back to bytes. The capture
435        // geometry gauge is what the self tile sizes its texture from (stable,
436        // unlike the BWE-driven encode rung) — kept fresh per frame so a camera
437        // hot-swap with a different geometry follows automatically.
438        stats.set_video_capture_dims(frame.dims.width(), frame.dims.height());
439        push_preview(sink.as_ref(), SELF_KEY, &frame);
440
441        // AudioOnly: the ladder dropped video under severe congestion. Keep pulling the
442        // device (resume is instant) but skip encode + publish entirely.
443        if adapt.paused.load(Ordering::Relaxed) {
444            continue;
445        }
446
447        // Subscriber-gated SOURCE encode. While the publish loop's live gate is still
448        // dark (nobody subscribed, no control-plane demand) every encoded frame was
449        // discarded downstream — a whole H.264/AV1 encode per captured frame for
450        // nothing. Skip the encode (and the mid-call rebuild that only serves it);
451        // capture, self-preview and the LOW rung below are untouched. Going live arms
452        // `force_keyframe`, so the first frame encoded after the transition is an IDR.
453        if adapt.source_encode_open() {
454            // Mid-call codec switch (a peer joined/left mid-call) AND/OR mid-call
455            // resolution-rung change (the adaptation tick's DWELLED geometry —
456            // `adapt.encoder_w/h`, see `hound_media::GeometryDwell` — settled on
457            // something different from what's built): rebuild the encoder to match
458            // either, at the live bitrate. `0` on the geometry atomics means "no
459            // adaptation decision yet" (before the first tick), so fall back to the
460            // current `enc_dims` rather than a bogus 0×0 rebuild. Either change forces
461            // a fresh IDR — a decoder can't switch codec OR resolution mid-GOP, so
462            // every receiver must resync.
463            //
464            // A failed open keeps the current encoder and leaves the switch PENDING
465            // (`last_target`/`enc_dims` move only on success), so it is retried on the
466            // `RebuildRetry` cadence. Committing the target up-front instead makes one
467            // transient failure permanent: a downgrade forced by a peer who cannot
468            // decode the live codec would never be re-attempted, and that peer sits at
469            // 0 fps for the rest of the call with nothing left to renegotiate.
470            let want_codec = codec_from_rank(adapt.target_codec.load(Ordering::Relaxed));
471            let (want_w, want_h) = (
472                adapt.encoder_w.load(Ordering::Relaxed),
473                adapt.encoder_h.load(Ordering::Relaxed),
474            );
475            let want_dims = if want_w > 0 && want_h > 0 {
476                Dimensions::even_clamped(want_w, want_h)
477            } else {
478                enc_dims
479            };
480            // `encoder.is_none()` is a rebuild trigger in its own right (the low path's
481            // self-heal): the device open above only runs while the CAMERA is closed, so
482            // an encoder that failed to open beside a camera that opened fine has no
483            // other route back — the pipeline would capture and publish nothing for the
484            // rest of the call.
485            let rebuild_due =
486                encoder.is_none() || want_codec != last_target || want_dims != enc_dims;
487            if rebuild_due && rebuild_retry.ready(Instant::now()) {
488                if let Some(e) = open_temporal_encoder_laddered(
489                    want_codec,
490                    want_dims,
491                    CAM_FPS,
492                    Bitrate::from_bps(applied_bps),
493                    CAM_KEYFRAME_CADENCE,
494                ) {
495                    tracing::info!(
496                        track = "cam",
497                        codec = ?want_codec,
498                        from = %enc_dims,
499                        to = %want_dims,
500                        "call: rebuilding encoder mid-call (codec/geometry)"
501                    );
502                    encoder = Some(e);
503                    last_target = want_codec;
504                    enc_dims = want_dims;
505                    rebuild_retry.succeeded();
506                    adapt.force_keyframe.store(true, Ordering::Relaxed);
507                } else {
508                    rebuild_retry.failed(Instant::now());
509                    tracing::warn!(track = "cam", codec = ?want_codec, "call: mid-call encoder rebuild found no usable encoder; keeping current and retrying");
510                }
511            }
512
513            if let Some(enc) = encoder.as_mut() {
514                encode_and_publish_adaptive(
515                    &frame,
516                    enc,
517                    &adapt,
518                    &mut applied_bps,
519                    &mut fps_gate,
520                    enc_dims,
521                    |wire| enc_tx.try_send(wire).is_ok(),
522                    &mut frame_seq,
523                    sender,
524                    TrackKind::CamVideo,
525                    &stats,
526                    &mut fail_streak,
527                );
528            }
529        }
530
531        // LOW simulcast layer: when a constrained / small-tile receiver demands it,
532        // downscale this same capture frame and encode + forward it on the cam_l0
533        // track. Inert while `low_active` is `None` (dark until SIMULCAST_ACTUATE),
534        // so the source path above is byte-identical to a no-simulcast build.
535        match *low_active.borrow() {
536            Some(low_bps) => {
537                low_adapt.target_bps.store(low_bps, Ordering::Relaxed);
538                // A low-track PLI (new/recovering low subscriber) → force a low IDR.
539                if low_keyframe.swap(false, Ordering::Relaxed) {
540                    low_adapt.force_keyframe.store(true, Ordering::Relaxed);
541                }
542                // What this cap buys, and the geometry it settles on. The dwell is
543                // created on activation seeded at the first rung (so the encoder
544                // opens there at once) and thereafter debounces every change.
545                let want_rung =
546                    hound_media::low_rung_for_cap(hound_media::VideoSource::Camera, low_bps);
547                let fit = |h| {
548                    let (w, h) =
549                        hound_media::fit_encode_dims((CAM_DIMS.width(), CAM_DIMS.height()), h);
550                    Dimensions::even_clamped(w, h)
551                };
552                let dwell = low_geometry.get_or_insert_with(|| {
553                    hound_media::GeometryDwell::with_committed(
554                        (CAM_DIMS.width(), CAM_DIMS.height()),
555                        {
556                            let d = fit(want_rung.height);
557                            (d.width(), d.height())
558                        },
559                    )
560                });
561                let (settled_w, settled_h) = dwell.observe(
562                    // HEIGHT: a camera's aspect is fixed, so a rung height is the
563                    // ladder's own unit and area follows from it.
564                    Some(hound_media::RungTarget::Height(want_rung.height)),
565                    frame.timestamp_us.micros(),
566                    hound_media::GEOMETRY_REBUILD_DWELL_US,
567                );
568                let low_dims = Dimensions::even_clamped(settled_w, settled_h);
569                // The rung the settled geometry belongs to, read off the dwell's own
570                // committed TARGET rather than inferred from the geometry: while a
571                // change is still dwelling this is the rung the encoder is BUILT at,
572                // so its fps gate stays coherent with its geometry. (Inferring it
573                // from the geometry breaks wherever two rungs fit to the same pixels
574                // — a capture already smaller than both — see `GeometryDwell`.)
575                let settled_rung = if dwell.committed_target()
576                    == Some(hound_media::RungTarget::Height(want_rung.height))
577                {
578                    want_rung
579                } else {
580                    low_rung.map(|(_, r)| r).unwrap_or(want_rung)
581                };
582                low_adapt
583                    .target_fps
584                    .store(settled_rung.fps, Ordering::Relaxed);
585
586                // (Re)build the low encoder on a codec switch OR a settled rung
587                // change. Either forces a fresh IDR: no decoder can follow a codec
588                // or a resolution change mid-GOP.
589                let low_want = codec_from_rank(adapt.target_codec.load(Ordering::Relaxed));
590                let built = low_rung;
591                if low_encoder.is_none()
592                    || low_last_target != Some(low_want)
593                    || built != Some((low_dims, settled_rung))
594                {
595                    low_last_target = Some(low_want);
596                    low_encoder = open_encoder_laddered(
597                        low_want,
598                        low_dims,
599                        Fps::new(settled_rung.fps),
600                        Bitrate::from_bps(low_bps.min(u32::MAX as u64) as u32),
601                        CAM_KEYFRAME_CADENCE,
602                    );
603                    if let Some(enc) = low_encoder.as_ref() {
604                        use hound_engine::backend::VideoEncode;
605                        // What the ladder actually built, which the `_l0` decl must
606                        // name — the roster gate rejects a decl for a codec the
607                        // receivers can't decode, and a rejected decl means no
608                        // `TrackInfo` and so no subscribers at all.
609                        low_codec.store(codec_rank(enc.codec()), Ordering::Relaxed);
610                        if built.is_some_and(|(d, _)| d != low_dims) {
611                            tracing::info!(
612                                track = "cam_l0",
613                                cap_bps = low_bps,
614                                height = settled_rung.height,
615                                fps = settled_rung.fps,
616                                "simulcast[send]: low rung re-priced — rebuilding the low encoder"
617                            );
618                        }
619                        low_rung = Some((low_dims, settled_rung));
620                        low_adapt.force_keyframe.store(true, Ordering::Relaxed);
621                        // Re-baseline the per-encoder state a rebuild invalidates.
622                        // `low_applied_bps` tracks what was pushed into the encoder
623                        // that just went away: a codec ladder rung may scale the
624                        // target it actually applies (AV1/HEVC take a fraction of the
625                        // H.264 rate), so leaving it latched would make the first
626                        // `set_bitrate` a no-op and strand the new encoder at its
627                        // construction rate. The fps gate is re-armed for the same
628                        // reason — its cadence belongs to the rung that just ended,
629                        // and a stale one would swallow the first frames of the new
630                        // rung, including the IDR just forced.
631                        low_applied_bps = 0;
632                        low_fps_gate = hound_media::FrameRateGate::new();
633                    }
634                }
635                if let Some(low_enc) = low_encoder.as_mut()
636                    && let Some(low_frame) = low_rung_frame(&frame, low_enc, low_dims)
637                {
638                    encode_and_publish_adaptive(
639                        &low_frame,
640                        low_enc,
641                        &low_adapt,
642                        &mut low_applied_bps,
643                        &mut low_fps_gate,
644                        low_dims,
645                        |wire| low_vid_tx.try_send(wire).is_ok(),
646                        &mut low_frame_seq,
647                        sender,
648                        TrackKind::CamVideoLow,
649                        &stats,
650                        &mut low_fail_streak,
651                    );
652                }
653            }
654            // Demand gone — release the low encoder (resumes instantly on re-demand).
655            // The dwell goes with it: the next activation re-seeds at whatever rung
656            // the fresh cap buys rather than inheriting a stale committed geometry.
657            None => {
658                low_encoder = None;
659                low_geometry = None;
660                low_rung = None;
661            }
662        }
663    }
664    // Teardown (leave / reconnect / stop): the camera pipeline is gone. Mirrors
665    // `set_screen_disabled` on the share path, so a torn-down camera never leaves a
666    // latched `enabled` behind for the diagnosis to judge.
667    stats.set_video_enabled(false);
668}
