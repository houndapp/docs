1//! The control-plane IO shell: [`ControlClient`] drives the `/control` WebSocket
2//! (applies each `ServerMessage` to the `SessionView` reducer, publishes
3//! `CallView` snapshots, exposes the outbound intent queue, and fans the
4//! keyframe/feedback/quality signals out to the per-track listeners), plus
5//! [`dial_relay`] — the iroh endpoint + QUIC connection the MoQ session rides.
6//! Invariant: the returned [`GracefulEndpoint`] must outlive the connection — the
7//! SFU is relay-only reachable, so dropping the endpoint wedges every stream after
8//! the handshake. Its drop is also what closes the endpoint, so every exit path
9//! (bail, abort, teardown) closes the socket gracefully rather than aborting it.

15/// The Rust side of the `/control` WebSocket: connects, drives a [`SessionView`],
16/// publishes a [`CallView`] snapshot on every change, and exposes an outbound
17/// intent channel.
18pub(super) struct ControlClient {
19    pub(super) outbound: mpsc::UnboundedSender<ClientMessage>,
20    pub(super) view_rx: watch::Receiver<SessionView>,
21    /// Fan-out of incoming `KeyframeRequested` PLIs (the producer side of the PLI
22    /// relay). The reader forwards every requested `TrackId` here; the per-track
23    /// keyframe listener filters to OUR cam/screen track and arms a forced IDR. A
24    /// `broadcast` so cam + screen listeners each observe it independently.
25    pub(super) keyframe_tx: broadcast::Sender<TrackId>,
26    /// Fan-out of incoming `MediaFeedbackReceived` reception reports (the producer
27    /// side of the receiver→producer BWE loop). The reader forwards every report
28    /// here; the per-track feedback listener folds the loss into OUR track's
29    /// adaptation. A `broadcast` for the same multi-consumer reason.
30    pub(super) feedback_tx: broadcast::Sender<MediaFeedback>,
31    /// Fan-out of incoming `QualityRequested` standing demands (the producer side of
32    /// demand-driven simulcast). The reader forwards every request here; the per-track
33    /// quality listener filters to OUR cam/screen track and drives the layer planner.
34    pub(super) quality_tx: broadcast::Sender<QualityReport>,
35    /// Resolves when the reader loop exits (server closed / read error / `view_tx`
36    /// dropped). The supervisor's [`LiveSession::monitor`] selects on this; the
37    /// reader never pushes a connection-state transition itself — the supervisor
38    /// owns those.
39    pub(super) closed: oneshot::Receiver<()>,
40}

42impl ControlClient {
43    pub(super) async fn connect(
44        base_url: &str,
45        space_id: Uuid,
46        token: &str,
47        channels: CallChannels,
48        identities: PeerIdentities,
49        // The receive-side hidden-tile set. The reader is a READER: it rebuilds the
50        // whole `CallView` per roster frame, so it has to re-join this in or a decode
51        // cut's flags would be dropped on the floor by the next roster change. The
52        // subscribe loop owns the writes. See [`HiddenTiles`].
53        hidden: HiddenTiles,
54        // This establish's dominant-speaker ranking. The reader is its only
55        // writer; the subscribe loop reads a snapshot per reconcile pass.
56        speakers: SharedSpeakers,
57    ) -> Result<Self> {
58        use futures::{SinkExt, StreamExt};
59        use tokio_tungstenite::tungstenite::Message;
60        use tokio_tungstenite::tungstenite::client::IntoClientRequest;
61        use tokio_tungstenite::tungstenite::http::header::AUTHORIZATION;
62
63        // The credential rides a header, never the URL — a query-string token is
64        // logged by every proxy in the path (same rule as the realtime leg).
65        let url = control_ws_url(base_url, space_id)?;
66        let mut request = url.into_client_request().map_err(|e| {
67            HoundError::realtime(
68                RealtimeCode::ControlWsUrl,
69                format!("bad control WS url: {e}"),
70            )
71        })?;
72        let authorization = format!("Bearer {token}").parse().map_err(|_| {
73            HoundError::auth(
74                AuthCode::BadHeaderValue,
75                "session token is not a valid header value",
76            )
77        })?;
78        request.headers_mut().insert(AUTHORIZATION, authorization);
79        let (ws, _resp) = tokio_tungstenite::connect_async(request)
80            .await
81            .map_err(|e| {
82                HoundError::realtime(
83                    RealtimeCode::ControlWsConnect,
84                    format!("control WS connect failed: {e}"),
85                )
86            })?;
87        let (mut sink, mut stream) = ws.split();
88
89        let (tx, mut rx) = mpsc::unbounded_channel::<ClientMessage>();
90        // First frame MUST be Hello. Advertise our decode caps so producers only
91        // ever send us a codec we can play — but first give a first-run openh264
92        // download a bounded chance to land, because these caps are snapshotted
93        // for the whole session and there is no mid-call re-announce.
94        crate::openh264::settle().await;
95        let decode_caps = local_decode_caps();
96        tx.send(ClientMessage::Hello {
97            protocol_version: PROTOCOL_VERSION,
98            decode_caps,
99        })
100        .expect("receiver just created");
101
102        let (view_tx, view_rx) = watch::channel(SessionView::new());
103
104        // Writer task: drains outbound, with a periodic Ping keepalive (a quiet
105        // call sends nothing for minutes and NAT/proxy idle timeouts kill the WS).
106        const KEEPALIVE: std::time::Duration = std::time::Duration::from_secs(20);
107        let writer = call_runtime().spawn(async move {
108            loop {
109                let msg = match tokio::time::timeout(KEEPALIVE, rx.recv()).await {
110                    Ok(Some(m)) => m,
111                    Ok(None) => break, // tx dropped — teardown
112                    Err(_) => {
113                        if sink.send(Message::Ping(Vec::new().into())).await.is_err() {
114                            return;
115                        }
116                        continue;
117                    }
118                };
119                let Ok(json) = serde_json::to_string(&msg) else {
120                    continue;
121                };
122                if sink.send(Message::Text(json.into())).await.is_err() {
123                    break;
124                }
125            }
126            let _ = sink.close().await;
127        });
128
129        // The control-closed signal: the reader fires this when its loop exits, and
130        // the supervisor's `LiveSession::monitor` selects on it. The reader does NOT
131        // push a connection-state transition itself — the supervisor owns those (a WS
132        // drop must heal, not surface `Failed`, except on a fatal re-establish error).
133        let (closed_tx, closed_rx) = oneshot::channel::<()>();
134
135        // Producer-side BWE seams: fan `KeyframeRequested` PLIs and
136        // `MediaFeedbackReceived` reception reports out of the reader to the per-track
137        // adaptation listeners (each `.subscribe()`s a fresh receiver per establish).
138        // A small bounded buffer is plenty — a lag just coalesces into one keyframe /
139        // is superseded by the next per-interval report. Returned so the publish loops
140        // subscribe; the reader holds clones to forward onto.
141        let (keyframe_tx, _keyframe_rx) = broadcast::channel::<TrackId>(16);
142        let (feedback_tx, _feedback_rx) = broadcast::channel::<MediaFeedback>(32);
143        let (quality_tx, _quality_rx) = broadcast::channel::<QualityReport>(32);
144        let keyframe_tx_reader = keyframe_tx.clone();
145        let feedback_tx_reader = feedback_tx.clone();
146        let quality_tx_reader = quality_tx.clone();
147
148        // Reader loop: apply each ServerMessage to the view, publish CallView, forward
149        // roster updates, AND fan keyframe/feedback signals to the adaptation
150        // listeners. On close/error, signal `closed` and exit.
151        let view_out = channels.view.clone();
152        call_runtime().spawn(async move {
153            let _writer = writer;
154            let mut view = SessionView::new();
155            while let Some(frame) = stream.next().await {
156                match frame {
157                    Ok(Message::Text(text)) => {
158                        let Ok(msg) = serde_json::from_str::<ServerMessage>(&text) else {
159                            continue;
160                        };
161                        // The backend picked a new best SFU node (rollout, or this
162                        // node started draining). Reuse the connection-loss path:
163                        // break out so `closed_tx` fires and the supervisor's
164                        // `monitor()` sees `REALTIME009` ("control WebSocket
165                        // dropped") — retryable, so the reconnect loop tears down
166                        // and re-establishes, which re-fetches `voice_join` and
167                        // lands on the new node. No new state machine needed.
168                        if let ServerMessage::SfuMigrate { node_id } = &msg {
169                            tracing::info!(
170                                %node_id,
171                                "call: SFU migrate signal — re-establishing on the new node"
172                            );
173                            break;
174                        }
175                        // Frames the reader consumes ENTIRELY on its own, before the
176                        // view/publish tail below. See [`consume_reader_only`].
177                        if consume_reader_only(&msg, &speakers) {
178                            continue;
179                        }
180                        // Producer-side BWE/keyframe signals: fan to the per-track
181                        // adaptation listeners (which filter to OUR track). `send` errs
182                        // only with no live receivers (no video publishing) — ignore.
183                        match &msg {
184                            ServerMessage::KeyframeRequested { track } => {
185                                let _ = keyframe_tx_reader.send(*track);
186                            }
187                            ServerMessage::MediaFeedbackReceived {
188                                track,
189                                from,
190                                received,
191                                lost,
192                                jitter_us,
193                            } => {
194                                let _ = feedback_tx_reader.send(MediaFeedback {
195                                    track: *track,
196                                    from: *from,
197                                    received: *received,
198                                    lost: *lost,
199                                    jitter_us: *jitter_us,
200                                });
201                            }
202                            ServerMessage::QualityRequested {
203                                track,
204                                from,
205                                cap_bps,
206                                hidden,
207                            } => {
208                                let _ = quality_tx_reader.send(QualityReport {
209                                    track: *track,
210                                    from: *from,
211                                    cap_bps: *cap_bps,
212                                    hidden: *hidden,
213                                });
214                            }
215                            // The server declined our `Publish` (codec outside the
216                            // roster's common decode caps) — no track was added, so
217                            // there's nothing to reconcile; just surface it for
218                            // diagnosis. No UI plumbing: this should be rare (the
219                            // client's own negotiation already targets the common
220                            // set) and is not actionable by the user.
221                            ServerMessage::PublishRejected { source, kind, codec } => {
222                                tracing::warn!(
223                                    peer = source.0,
224                                    ?kind,
225                                    ?codec,
226                                    "call: server rejected our publish (codec outside roster's common decode caps)"
227                                );
228                            }
229                            _ => {}
230                        }
231                        view.apply(&msg);
232                        let _ = view_tx.send_replace(view.clone());
233                        // Join the control-plane peers against the latest identity
234                        // snapshot so the published view carries display identities.
235                        // `send_replace`: a headless caller may not have subscribed
236                        // `watch_call` yet (plain `send` drops the publish with zero
237                        // receivers and the last roster change would be lost forever).
238                        view_out.send_replace(CallView::from_view_with(
239                            &view,
240                            &identities.load(),
241                            &hidden.load(),
242                        ));
243                    }
244                    // Distinguish the two ways the control WS ends, so a recurring
245                    // drop is diagnosable: a server-sent Close (with its reason) vs a
246                    // read/transport error (network blip, reset). Either way the
247                    // supervisor heals; this is purely for visibility.
248                    Ok(Message::Close(frame)) => {
249                        tracing::warn!(?frame, "call: control WS closed by server");
250                        break;
251                    }
252                    Err(e) => {
253                        tracing::warn!(error = %e, "call: control WS read error");
254                        break;
255                    }
256                    Ok(_) => {}
257                }
258            }
259            // The control plane dropped — wake the supervisor's monitor so it heals
260            // the whole session (a fresh PeerId; the roster fully resyncs).
261            let _ = closed_tx.send(());
262        });
263
264        Ok(Self {
265            outbound: tx,
266            view_rx,
267            keyframe_tx,
268            feedback_tx,
269            quality_tx,
270            closed: closed_rx,
271        })
272    }
273}

275/// Consume the frames the reader handles by itself, returning `true` when the
276/// caller must NOT fall through to the view/publish tail.
277///
278/// That tail is unconditional by construction: `view.apply` is followed by two
279/// `watch::send_replace`s, and `send_replace` notifies whether or not anything
280/// changed. So every frame that reaches it clones the `SessionView`, republishes a
281/// whole `CallView` to the UI, and wakes the subscribe reconcile loop — fine at
282/// roster rate, ruinous at media rate. `SpeakingLevelReported` arrives ~3 Hz per
283/// talking peer (10–33 Hz in a lively call) and carries no roster fact at all, so
284/// it stops here.
285fn consume_reader_only(msg: &ServerMessage, speakers: &SharedSpeakers) -> bool {
286    match msg {
287        ServerMessage::SpeakingLevelReported { peer, level } => {
288            speakers.observe(*peer, *level);
289            true
290        }
291        _ => false,
292    }
293}

295/// This device's video decode capabilities, advertised at `Hello`. The engine
296/// reports HW/SW support. H.264 is NOT unconditional: off Apple, SW H.264 is
297/// Cisco's downloaded binary, so a first run whose download has not landed (or a
298/// user who switched the codec off) advertises `h264: false` — deliberately, so
299/// peers never send a stream this client cannot decode. `Engine::decode_caps`
300/// waits briefly for an in-flight first download before answering; a blob that
301/// lands after that is not re-announced mid-session (see `hound_client::openh264`).
302fn local_decode_caps() -> hound_control::DecodeCaps {
303    let caps = Engine::new().decode_caps();
304    hound_control::DecodeCaps {
305        av1: caps.av1,
306        h265: caps.h265,
307        h264: caps.h264,
308    }
309}

311/// Build the `ws(s)://.../space/{id}/control` URL from the HTTP base. It carries
312/// no credential: the token goes in `Authorization: Bearer` at dial time.
313fn control_ws_url(base_url: &str, space_id: Uuid) -> Result<String> {
314    let base = base_url.trim_end_matches('/');
315    let ws_base = if let Some(rest) = base.strip_prefix("https://") {
316        format!("wss://{rest}")
317    } else if let Some(rest) = base.strip_prefix("http://") {
318        format!("ws://{rest}")
319    } else if base.starts_with("ws://") || base.starts_with("wss://") {
320        base.to_owned()
321    } else {
322        return Err(HoundError::media(
323            MediaCode::ControlBaseScheme,
324            format!("api_base has no http(s) scheme: {base_url}"),
325        ));
326    };
327    Ok(format!("{ws_base}/space/{space_id}/control"))
328}

330/// Wait until the control reducer has a `self_id` (the server's `Welcome`).
331pub(super) async fn wait_for_self_id(view_rx: &mut watch::Receiver<SessionView>) -> Result<PeerId> {
332    loop {
333        if let Some(id) = view_rx.borrow().self_id() {
334            return Ok(id);
335        }
336        view_rx.changed().await.map_err(|_| {
337            HoundError::media(
338                MediaCode::ControlClosedBeforeWelcome,
339                "control closed before Welcome",
340            )
341        })?;
342    }
343}

347/// Dial the MoQ relay (its node id + relay URL) on a fresh iroh endpoint, returning
348/// the QUIC connection the MoQ session rides. The endpoint is per-call with an
349/// ephemeral key (no persisted device identity yet) — fine for the call's lifetime;
350/// sharing it across calls is a later optimization.
351/// Returns BOTH the endpoint and the connection: the caller MUST keep the
352/// [`GracefulEndpoint`] alive for the connection's lifetime. The SFU is reachable
353/// only via the relay (it has no client-routable direct address), so all packets —
354/// including every stream after the handshake — ride the endpoint's magicsock/relay
355/// actor. Dropping the endpoint stops that actor, so the QUIC handshake still
356/// completes but the very next stream silently never transmits (the relay's
357/// `accept_bi` then times out). Holding only the `Connection` is not enough.
358///
359/// The [`GracefulEndpoint`] wrapper is what makes every exit path close the socket
360/// rather than abort it — the `?` bails below and throughout [`establish_live`], and
361/// a supervisor task aborted mid-establish by `leave`/drop. It detaches the close
362/// onto the call runtime, so no teardown or reconnect ever waits on a dead socket.
363pub(super) async fn dial_relay(
364    relay_url: &str,
365    sfu_node_id: &str,
366) -> Result<(GracefulEndpoint, iroh::endpoint::Connection)> {
367    use iroh::endpoint::presets;
368    use iroh::endpoint::{QuicTransportConfig, VarInt};
369    use iroh::{Endpoint, EndpointAddr, EndpointId, RelayMap, RelayMode, RelayUrl};
370
371    let relay_url: RelayUrl = relay_url
372        .parse()
373        .map_err(|e| HoundError::media(MediaCode::BadRelayUrl, format!("bad relay url: {e}")))?;
374    let relay_id: EndpointId = sfu_node_id.parse().map_err(|_| {
375        HoundError::media(
376            MediaCode::BadSfuNodeId,
377            format!("unparseable sfu node id: {sfu_node_id:?}"),
378        )
379    })?;
380
381    // Flow-control windows sized for the heaviest video track (a ~2.5 Mbps 1080p
382    // screenshare, or a fast link's 4 Mbps 1080p60 top rung):
383    // a too-small receive window forces the sender to stop-and-wait for window
384    // updates, and every such update is another inbound packet → another wakeup
385    // on a receive path that has no UDP GRO on macOS to coalesce them. Hold well
386    // over a high-RTT BDP per stream, and the sum across the call's concurrent
387    // moq streams (mic + cam + screen + screen-audio) at the connection level.
388    // MTU is left at iroh's conservative default — PMTUD raises it, and a high
389    // initial MTU risks black-holing on constrained paths. `max_concurrent_uni`
390    // gives moq's per-group streams headroom so a screenshare burst never stalls
391    // on the stream-count limit.
392    let transport = QuicTransportConfig::builder()
393        .stream_receive_window(VarInt::from_u32(4 * 1024 * 1024))
394        .receive_window(VarInt::from_u32(32 * 1024 * 1024))
395        .max_concurrent_uni_streams(VarInt::from_u32(1024))
396        .build();
397
398    let endpoint = Endpoint::builder(presets::N0)
399        .relay_mode(RelayMode::Custom(RelayMap::from(relay_url.clone())))
400        .transport_config(transport)
401        .bind()
402        .await
403        .map_err(|e| HoundError::media(MediaCode::IrohBind, format!("bind iroh endpoint: {e}")))?;
404
405    // Guard it before the dial: a failed `connect` must still close the endpoint.
406    // Pinned to the call runtime (not `Handle::current`) like every other spawn in
407    // this module — and unlike a task-local handle, it is alive for the process, so
408    // a drop that lands after the runtime that dialled is gone still closes.
409    let endpoint = GracefulEndpoint::on_runtime(endpoint, call_runtime().handle().clone());
410    let addr = EndpointAddr::new(relay_id).with_relay_url(relay_url);
411    let conn = endpoint
412        .connect(addr, hound_transport::ALPN)
413        .await
414        .map_err(|e| HoundError::media(MediaCode::RelayDial, format!("dial MoQ relay: {e}")))?;
415    Ok((endpoint, conn))
416}

418#[cfg(test)]
419mod tests {
420    use super::control_ws_url;

422    /// THE token-in-logs property for the call plane: the dialled control URL is
423    /// the base with the scheme rewritten and the route appended — and no query
424    /// at all, because the credential moved to an `Authorization` header.
425    #[test]
426    fn the_control_url_carries_no_credential() {
427        let space_id = uuid::Uuid::nil();
428        for base in [
429            "https://api.hound.chat/",
430            "http://localhost:8080",
431            "wss://api.hound.chat",
432        ] {
433            let url = control_ws_url(base, space_id).unwrap();
434            assert!(!url.contains("token="), "{url} leaks a credential");
435            assert!(!url.contains('?'), "{url} must have no query at all");
436        }
437        assert_eq!(
438            control_ws_url("https://api.hound.chat/", space_id).unwrap(),
439            format!("wss://api.hound.chat/space/{space_id}/control"),
440        );
441        assert_eq!(
442            control_ws_url("http://localhost:8080", space_id).unwrap(),
443            format!("ws://localhost:8080/space/{space_id}/control"),
444        );
445        assert!(control_ws_url("api.hound.chat", space_id).is_err());
446    }

448    /// THE watch-churn property: a speaking level is consumed by the reader and
449    /// never reaches the view/publish tail, while every roster-bearing frame does.
450    /// If this inverts, a lively 32-person call republishes the whole `CallView`
451    /// and wakes the subscribe reconcile ~10–33 times a second for a signal that
452    /// changes no roster fact — the reason the level is a ranker input and not
453    /// session state (`hound_control::SessionView::apply` pins the other half).
454    #[test]
455    fn a_speaking_level_never_reaches_the_view_tail() {
456        use super::{SharedSpeakers, consume_reader_only};
457        use hound_control::proto::{ErrorCode, PeerId, ServerMessage, TrackId};
458
459        let speakers = SharedSpeakers::new();
460        assert!(consume_reader_only(
461            &ServerMessage::SpeakingLevelReported {
462                peer: PeerId(1),
463                level: 9,
464            },
465            &speakers
466        ));
467        // (That it lands in the RANKING rather than on the floor is
468        // `speaking::reports_rank_a_speaker_over_the_shared_handle`'s to pin.)
469
470        // Everything else still falls through to the view/publish tail.
471        for msg in [
472            ServerMessage::PeerLeft { peer: PeerId(1) },
473            ServerMessage::TrackUnpublished { track: TrackId(1) },
474            ServerMessage::KeyframeRequested { track: TrackId(1) },
475            ServerMessage::Error {
476                code: ErrorCode::NotFound,
477                message: "nope".into(),
478            },
479        ] {
480            assert!(
481                !consume_reader_only(&msg, &speakers),
482                "{msg:?} must reach the view tail"
483            );
484        }
485    }
