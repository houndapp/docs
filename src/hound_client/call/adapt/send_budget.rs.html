1//! Connection-level coordinated send budget.
2//!
3//! The MoQ multiplex puts every published track on ONE QUIC connection, but each
4//! video publisher runs its own rate controller that sees only its own receivers'
5//! loss/RTT — never the connection's uplink saturation. Camera + a heavy screen share
6//! therefore sum their ladders past the wire: publish channels drop, bufferbloat
7//! builds, and our OWN downlink receive starves. This is the IO shell for
8//! `hound_media`'s coordinator core: each publish loop registers a slot, reports its
9//! UNCAPPED desire + its publish drops once per tick, and takes back its share of a
10//! single AIMD budget. It is inert with one publisher — a lone camera or a lone share
11//! is byte-identical to no coordination.
12
13use std::sync::Arc;
14
15use hound_media::CaptureClock;
16
17/// Carved off the connection budget before the video split: audio is sacred, so it
18/// takes its reserve first and the video tracks divide only what is left. Opus at
19/// the call's profile plus packet/QUIC overhead fits well inside this.
20const AUDIO_RESERVE_BPS: u64 = 128_000;
21
22/// Where the coordinator's AIMD starts, and its ceiling. Deliberately optimistic —
23/// it only ever clamps DOWN from here toward the wire, so a fast uplink never pays
24/// for the coordinator existing (the desires stay under the budget and the split
25/// hands out the full ceiling); a slow one converges within a few seconds of
26/// sustained publish drops.
27const SEND_BUDGET_MAX_BPS: u64 = 12_000_000;
28
29/// The coordinator's floor: below this there is no useful video left to divide (both
30/// ladders have already collapsed toward AudioOnly on their own).
31const SEND_BUDGET_MIN_BPS: u64 = 600_000;
32
33/// Minimum spacing between coordinator advances. The two publish loops tick at
34/// [`ADAPT_TICK_INTERVAL`] on their OWN `CaptureClock`s, so whichever crosses this
35/// gate first drives the advance and the other reads the fresh allocation. Slightly
36/// under 1 s so ordinary tick jitter can't push the effective cadence out to 2 s
37/// (`SendBudget`'s own decrease cooldown still paces the back-off at ~1 Hz).
38const SEND_BUDGET_INTERVAL_US: i64 = 900_000;
39
40/// Which shared-connection video publisher a budget slot belongs to. Exactly two
41/// exist; the coordinator only bites when BOTH are registered *and* have reported.
42#[derive(Debug, Clone, Copy, PartialEq, Eq)]
43pub(in crate::call) enum SendTrack {
44    Camera = 0,
45    Screen = 1,
46}
47
48const SEND_TRACKS: usize = 2;
49
50/// The coordinator's pure state: the AIMD budget plus each slot's registration,
51/// latest desire and current cap. Every method is a `(state, input) -> (state,
52/// output)` transition with `now_us` passed in — the clock read lives one layer up
53/// in [`SendCoordinator`], so the split policy is unit-testable.
54#[derive(Debug)]
55struct CoordState {
56    budget: hound_media::SendBudget,
57    registered: [bool; SEND_TRACKS],
58    /// Each slot's last reported UNCAPPED desire; `None` until it first ticks.
59    desired: [Option<u64>; SEND_TRACKS],
60    caps: [u64; SEND_TRACKS],
61    /// Bumped on every registration of that slot. A [`SendSlot`] only deregisters
62    /// when its own generation is still the live one, so a restarting publisher
63    /// (a share stopping and immediately restarting) can't have its predecessor's
64    /// late drop knock the new registration out.
65    generation: [u64; SEND_TRACKS],
66    /// Whether either track dropped publish AUs since the last advance — the one
67    /// connection-level congestion signal a sender sees without new transport
68    /// surface. Consumed (and cleared) by each advance.
69    over_capacity: bool,
70    last_advance_us: Option<i64>,
71}
72
73impl CoordState {
74    fn fresh() -> Self {
75        Self {
76            budget: hound_media::SendBudget::new(
77                SEND_BUDGET_MAX_BPS,
78                SEND_BUDGET_MIN_BPS,
79                SEND_BUDGET_MAX_BPS,
80            ),
81            registered: [false; SEND_TRACKS],
82            desired: [None; SEND_TRACKS],
83            caps: [u64::MAX; SEND_TRACKS],
84            generation: [0; SEND_TRACKS],
85            over_capacity: false,
86            last_advance_us: None,
87        }
88    }
89
90    /// Both publishers are live AND each has reported a desire at least once. Until
91    /// then the split has nothing honest to divide, so nobody is capped.
92    fn coordinating(&self) -> bool {
93        (0..SEND_TRACKS).all(|i| self.registered[i] && self.desired[i].is_some())
94    }
95
96    /// Any membership change starts the pairing over: a fresh (optimistic) budget and
97    /// no caps, so the surviving publisher is released the moment the other stops.
98    fn membership_changed(&mut self) {
99        self.budget = CoordState::fresh().budget;
100        self.caps = [u64::MAX; SEND_TRACKS];
101        self.over_capacity = false;
102        self.last_advance_us = None;
103    }
104
105    /// Fold one publisher's tick in and return its share. `None` = uncoordinated
106    /// (fewer than two live publishers) — the caller clears its cap and its own
107    /// ladder governs, exactly as before this coordinator existed.
108    fn observe(
109        &mut self,
110        track: SendTrack,
111        desired_bps: u64,
112        dropped: u32,
113        now_us: i64,
114    ) -> Option<u64> {
115        let i = track as usize;
116        self.desired[i] = Some(desired_bps);
117        self.over_capacity |= dropped > 0;
118        if !self.coordinating() {
119            return None;
120        }
121        let due = self
122            .last_advance_us
123            .is_none_or(|prev| now_us.saturating_sub(prev) >= SEND_BUDGET_INTERVAL_US);
124        if due {
125            let total = self.budget.observe(self.over_capacity, now_us);
126            self.over_capacity = false;
127            self.last_advance_us = Some(now_us);
128            let video_budget = total.saturating_sub(AUDIO_RESERVE_BPS);
129            let desired: Vec<u64> = self.desired.iter().map(|d| d.unwrap_or(0)).collect();
130            for (cap, share) in self
131                .caps
132                .iter_mut()
133                .zip(hound_media::allocate_send_budget(video_budget, &desired))
134            {
135                *cap = share;
136            }
137        }
138        Some(self.caps[i])
139    }
140}

142/// The shared-connection send coordinator. One per process (see
143/// [`send_coordinator`]) — a call has exactly one connection, and the RAII
144/// [`SendSlot`]s make it inert between calls.
145pub(in crate::call) struct SendCoordinator {
146    /// The coordinator's OWN timeline. Each publish loop stamps its adaptation ticks
147    /// from its own `CaptureClock`, so their `now_us` values are not comparable;
148    /// the AIMD needs one clock, and this is it (the documented single real-clock
149    /// seam — [`CoordState`] itself takes time as an argument).
150    clock: CaptureClock,
151    state: std::sync::Mutex<CoordState>,
152}
153
154impl SendCoordinator {
155    fn new() -> Self {
156        Self {
157            clock: CaptureClock::start(),
158            state: std::sync::Mutex::new(CoordState::fresh()),
159        }
160    }
161
162    /// Claim this publisher's slot for the life of the returned guard.
163    pub(super) fn register(self: &Arc<Self>, track: SendTrack) -> SendSlot {
164        let mut generation = 0;
165        if let Ok(mut s) = self.state.lock() {
166            let i = track as usize;
167            s.registered[i] = true;
168            s.generation[i] = s.generation[i].wrapping_add(1);
169            generation = s.generation[i];
170            s.membership_changed();
171        }
172        SendSlot {
173            coord: self.clone(),
174            track,
175            generation,
176        }
177    }
178}
179
180/// A publisher's registration in the [`SendCoordinator`], released on drop (so a
181/// publish loop returning — leave, reconnect teardown, a share ending — hands the
182/// wire back to whoever is left within one tick).
183pub(in crate::call) struct SendSlot {
184    coord: Arc<SendCoordinator>,
185    track: SendTrack,
186    /// Which registration of `track` this guard is, so a late drop from a replaced
187    /// publisher never deregisters its successor.
188    generation: u64,
189}
190
191impl SendSlot {
192    /// Report this track's uncapped desire + the publish AUs it dropped this
193    /// interval; get back our share of the connection. `None` = uncoordinated.
194    pub(super) fn observe(&self, desired_bps: u64, dropped: u32) -> Option<u64> {
195        let now_us = self.coord.clock.now();
196        let Ok(mut s) = self.coord.state.lock() else {
197            // A poisoned coordinator degrades to uncoordinated — never to a stale cap.
198            return None;
199        };
200        s.observe(self.track, desired_bps, dropped, now_us)
201    }
202}
203
204impl Drop for SendSlot {
205    fn drop(&mut self) {
206        if let Ok(mut s) = self.coord.state.lock() {
207            let i = self.track as usize;
208            // A publisher that has already been replaced (share restart) must not
209            // deregister the live registration that took its place.
210            if s.generation[i] != self.generation {
211                return;
212            }
213            s.registered[i] = false;
214            s.desired[i] = None;
215            s.membership_changed();
216        }
217    }
218}

220/// The process-wide coordinator the publish loops register with. A `OnceLock` like
221/// [`call_runtime`] — a call has one connection at a time, and the slots deregister
222/// on teardown, so no per-session construction is needed.
223pub(super) fn send_coordinator() -> &'static Arc<SendCoordinator> {
224    static COORD: std::sync::OnceLock<Arc<SendCoordinator>> = std::sync::OnceLock::new();
225    COORD.get_or_init(|| Arc::new(SendCoordinator::new()))
226}

228#[cfg(test)]
229mod send_budget_glue {
230    //! The shared-connection coordinator's split policy. `CoordState` is the pure
231    //! seam (time in, caps out), so drive it directly — the AIMD and the max-min-fair
232    //! allocation themselves are covered by `hound_media`'s own tests.
233
234    use super::*;
235
236    /// Advance one publisher for `ticks` seconds, returning its last share.
237    fn drive(
238        state: &mut CoordState,
239        track: SendTrack,
240        desired: u64,
241        dropped: u32,
242        ticks: usize,
243        now: &mut i64,
244    ) -> Option<u64> {
245        let mut last = None;
246        for _ in 0..ticks {
247            last = state.observe(track, desired, dropped, *now);
248            *now += 1_000_000;
249        }
250        last
251    }
252
253    /// A lone publisher is NEVER capped, however badly it congests the wire: with
254    /// nothing to share the budget with, its own ladder is already the right answer.
255    /// This is the "single-publisher behaviour is unchanged" contract.
256    #[test]
257    fn a_lone_publisher_is_never_capped() {
258        let mut state = CoordState::fresh();
259        state.registered[SendTrack::Screen as usize] = true;
260        let mut now = 0i64;
261        for _ in 0..60 {
262            assert_eq!(
263                state.observe(SendTrack::Screen, 20_000_000, 30, now),
264                None,
265                "one publisher must never be capped by the coordinator"
266            );
267            now += 1_000_000;
268        }
269    }
270
271    /// Both publishers live but the wire comfortably carries both desires: the
272    /// coordinator hands out a ceiling ABOVE each desire, so neither ladder is
273    /// actually limited. A healthy link must not pay for coordination.
274    #[test]
275    fn headroom_does_not_bite() {
276        let mut state = CoordState::fresh();
277        state.registered = [true; SEND_TRACKS];
278        let mut now = 0i64;
279        // The first publisher to report has nothing to be split against yet; the
280        // second completes the pair and drives the first allocation.
281        assert_eq!(state.observe(SendTrack::Camera, 1_500_000, 0, now), None);
282        let screen = state.observe(SendTrack::Screen, 4_000_000, 0, now).unwrap();
283        now += 1_000_000;
284        let cam = state.observe(SendTrack::Camera, 1_500_000, 0, now).unwrap();
285        assert!(
286            cam > 1_500_000 && screen > 4_000_000,
287            "with headroom every track gets a ceiling above its desire: {cam} {screen}"
288        );
289    }
290
291    /// Sustained publish drops walk the shared budget down until it really is the
292    /// binding constraint, and then the two ladders divide it instead of each
293    /// claiming the whole wire: the caps sum to the budget (audio's reserve already
294    /// removed) and the greedier screen share takes the larger share — but never all
295    /// of it, the camera keeps its floor.
296    #[test]
297    fn an_oversubscribed_wire_is_split_and_the_camera_keeps_its_floor() {
298        let mut state = CoordState::fresh();
299        state.registered = [true; SEND_TRACKS];
300        let (cam_want, screen_want) = (1_700_000u64, 20_000_000u64);
301        let mut now = 0i64;
302        let mut cam = None;
303        let mut screen = None;
304        for _ in 0..12 {
305            cam = state.observe(SendTrack::Camera, cam_want, 4, now);
306            screen = state.observe(SendTrack::Screen, screen_want, 6, now);
307            now += 1_000_000;
308        }
309        let (cam, screen) = (cam.unwrap(), screen.unwrap());
310        let total = state.budget.total_bps();
311        assert!(
312            total < SEND_BUDGET_MAX_BPS,
313            "sustained drops must walk the budget down from its optimistic start"
314        );
315        assert!(
316            cam + screen <= total - AUDIO_RESERVE_BPS + 2,
317            "the split must never hand out more than the budget minus audio's \
318             reserve: {cam} + {screen} vs {total}"
319        );
320        assert!(
321            screen > cam,
322            "the greedier publisher takes the larger share: {cam} vs {screen}"
323        );
324        assert!(
325            cam >= hound_media::MIN_VIDEO_BPS,
326            "a greedy screen share must never starve the camera below the floor: {cam}"
327        );
328    }
329
330    /// Coordination is level-triggered on membership: when one publisher stops (its
331    /// slot drops), the survivor is released on its very next tick — no stale cap
332    /// outlives the pairing that justified it.
333    #[test]
334    fn a_departing_publisher_releases_the_survivor() {
335        let mut state = CoordState::fresh();
336        state.registered = [true; SEND_TRACKS];
337        let mut now = 0i64;
338        for _ in 0..30 {
339            state.observe(SendTrack::Camera, 1_700_000, 4, now);
340            state.observe(SendTrack::Screen, 20_000_000, 6, now);
341            now += 1_000_000;
342        }
343        assert!(
344            state
345                .observe(SendTrack::Camera, 1_700_000, 4, now)
346                .is_some(),
347            "both live ⇒ capped"
348        );
349
350        // The screen share ends: its slot deregisters exactly as `SendSlot::drop` does.
351        state.registered[SendTrack::Screen as usize] = false;
352        state.desired[SendTrack::Screen as usize] = None;
353        state.membership_changed();
354
355        now += 1_000_000;
356        assert_eq!(
357            state.observe(SendTrack::Camera, 1_700_000, 4, now),
358            None,
359            "the survivor must be released, not left on the pairing's cap"
360        );
361    }
362
363    /// A slot registered but not yet ticked contributes no phantom demand: the split
364    /// waits for a real desire from both before it binds anyone.
365    #[test]
366    fn a_registered_but_silent_publisher_does_not_cap_the_other() {
367        let mut state = CoordState::fresh();
368        state.registered = [true; SEND_TRACKS];
369        let mut now = 0i64;
370        assert_eq!(
371            drive(&mut state, SendTrack::Screen, 20_000_000, 10, 10, &mut now),
372            None,
373            "no cap until BOTH publishers have reported a desire"
374        );
375    }

377    /// A restarting publisher (a share stopped and immediately restarted) registers
378    /// its replacement before the predecessor's guard drops. The late drop must not
379    /// deregister the live slot — that would silently disable coordination for the
380    /// rest of the call.
381    #[test]
382    fn a_replaced_publisher_does_not_deregister_its_successor() {
383        let coord = Arc::new(SendCoordinator::new());
384        let cam = coord.register(SendTrack::Camera);
385        let old_screen = coord.register(SendTrack::Screen);
386        let new_screen = coord.register(SendTrack::Screen);
387        drop(old_screen); // the aborted task's guard, dropped late
388
389        assert!(
390            cam.observe(1_700_000, 0).is_none(),
391            "the first report of a fresh pairing has nothing to split against yet"
392        );
393        assert!(
394            new_screen.observe(8_000_000, 0).is_some(),
395            "the surviving registration must still count as a live publisher"
396        );
397    }

399    /// The advance is rate-limited on the coordinator's own clock, not on call count:
400    /// two publishers ticking into the same window must not double-step the AIMD (the
401    /// budget would fall twice as fast as the wire warrants).
402    #[test]
403    fn interleaved_ticks_do_not_double_step_the_budget() {
404        let mut paired = CoordState::fresh();
405        paired.registered = [true; SEND_TRACKS];
406        // Seed both desires, then advance once so `last_advance_us` is armed.
407        paired.observe(SendTrack::Camera, 1_700_000, 0, 0);
408        paired.observe(SendTrack::Screen, 8_000_000, 0, 0);
409        let after_first = paired.budget.total_bps();
410
411        // A second tick inside the same window changes nothing.
412        paired.observe(SendTrack::Screen, 8_000_000, 0, 100_000);
413        assert_eq!(
414            paired.budget.total_bps(),
415            after_first,
416            "a second publisher ticking in the same window must not advance the AIMD"
417        );
418    }
