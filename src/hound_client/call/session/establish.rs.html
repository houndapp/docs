2//! voice/join, control WS, relay dial, playout bus, publish/subscribe tasks) —
3//! plus the per-session roster mute reconciler.

34/// Build one full control+MoQ [`LiveSession`]. Each establish re-mints the relay
35/// token by calling `voice/join`, so a token-expiry SFU drop heals via reconnect —
36/// that re-mint is what keeps a call alive past the relay token's TTL. The capture
37/// loops subscribe to `params.devices.*` (so device selection survives a reconnect)
38/// and read `params.mute`/`params.camera_enabled`. Seeds the identity map from the
39/// fresh `voice/join` each time.
40pub(in crate::call) async fn establish_live(params: &ConnectionParams) -> Result<LiveSession> {
41    let rest = &params.rest;
42    let space_id = params.space_id;
43    let channels = &params.channels;
44    let identities = &params.identities;
45    let sink = &params.sink;
46
47    // 1. REST voice/join → relay url/token/peers/sfu node id (re-minted every establish).
48    let join = rest.voice_join(space_id).await?;
49    let Some(sfu_node_id) = join.sfu_node_id.clone() else {
50        // Dev backend with no SFU configured: nothing to dial. A typed error
51        // rather than a silent no-op so the caller knows media won't flow.
52        return Err(HoundError::media(
53            MediaCode::NoSfuConfigured,
54            "voice/join returned no sfu_node_id (no SFU configured)",
55        ));
56    };
57
58    // Seed the peer_id → user_id map from the join roster: peers already in the
59    // channel carry their sfu_peer_id, so their identity is known before the first
60    // control publish. (Self is added once Welcome assigns our peer id below.)
61    {
62        let mut map = HashMap::new();
63        for p in &join.peers {
64            if let Some(pid) = p.sfu_peer_id {
65                map.insert(pid as u32, p.user_id);
66            }
67        }
68        identities.store(Arc::new(map));
69    }
70
71    // 2. Connect the control WS and drive the SessionView → CallView watch.
72    let token = rest.token().ok_or_else(|| {
73        HoundError::auth(
74            AuthCode::NoSessionToken,
75            "cannot connect control plane: no session token",
76        )
77    })?;
78    // The dominant-speaker ranking is per-ESTABLISH: peer ids are assigned fresh
79    // on every `Welcome`, so a ranking that survived a reconnect would hand a
80    // departed speaker's promotion to whoever inherits their id.
81    let speakers = SharedSpeakers::new();
82    let control = ControlClient::connect(
83        &params.base_url,
84        space_id,
85        &token,
86        channels.clone(),
87        identities.clone(),
88        params.hidden_tiles.clone(),
89        speakers.clone(),
90    )
91    .await?;
92    let control_outbound = control.outbound.clone();
93    let announced_decode_caps = control.announced_decode_caps;
94    let control_closed = control.closed;
95    let keyframe_tx = control.keyframe_tx.clone();
96    let feedback_tx = control.feedback_tx.clone();
97    let quality_tx = control.quality_tx.clone();
98    let mut view_rx = control.view_rx.clone();
99
100    // Wait for the server's Welcome to learn our peer id (with a timeout, so a
101    // wedged handshake fails the establish rather than hanging the reconnect ladder).
102    let self_id = tokio::time::timeout(
103        std::time::Duration::from_secs(15),
104        wait_for_self_id(&mut view_rx),
105    )
106    .await
107    .map_err(|_| HoundError::media(MediaCode::WelcomeTimeout, "control Welcome timed out"))??;
108
109    // This client's own authenticated identity — used to build SELF broadcast
110    // paths (`broadcast_path`) under the SFU's peer-namespace convention.
111    let self_user_id = join.profile.user_id;
112
113    // Now that Welcome assigned our (fresh) peer id, add self to the identity map and
114    // refresh the published participants (the reader already sent peers).
115    identities.rcu(|m| {
116        let mut m = HashMap::clone(m);
117        m.insert(self_id.0, self_user_id);
118        m
119    });
120    channels.view.send_modify(|cv| {
121        cv.participants = build_participants(&cv.peers, &identities.load());
122    });
123
124    // 3. Dial the MoQ relay over a fresh iroh endpoint + connection. Keep the
125    //    endpoint (stored on the LiveSession) alive for the whole call — see
126    //    `dial_relay`.
127    let (endpoint, conn) = dial_relay(&join.relay_url, &sfu_node_id).await?;
128    let relay = Arc::new(
129        MoqRelaySession::connect_relay(conn.clone(), &join.relay_token)
130            .await
131            .map_err(|e| {
132                HoundError::media(
133                    MediaCode::RelayHandshake,
134                    format!("MoQ relay handshake: {e}"),
135                )
136            })?,
137    );
138
139    // Shared session receive clock — every capture loop and the playout bus
140    // stamp against the SAME clock so A/V sync and the bus pop schedule align.
141    let clock = CaptureClock::start();
142
143    // 4. Audio infrastructure: one mixed-playout bus for all remote peers. The
144    //    speaker selection is shared across reconnects (the bus subscribes to it).
145    let apm = Engine::new().audio_processor().map(Arc::new);
146    // Re-assert the user's voice-processing settings onto this establish's fresh
147    // APM (it is rebuilt with code defaults each session) and make it the live
148    // target for mid-call settings changes.
149    params.voice.attach(apm.clone());
150    // A synthetic capture source (a headless bot / CI) also gets a speaker-free
151    // playout bus: opening a real DAC is wrong (no hardware) and two real RT mix
152    // engines contending on one host give false audio-starvation verdicts.
153    #[cfg(feature = "synthetic")]
154    let playout_bus = if params.capture_source != CaptureSource::Default {
155        Engine::new().start_synthetic_playout_bus(
156            params.devices.speaker.subscribe(),
157            apm.clone(),
158            clock,
159            PLAYOUT_TARGET_US,
160        )
161    } else {
162        Engine::new().start_playout_bus(
163            params.devices.speaker.subscribe(),
164            apm.clone(),
165            clock,
166            PLAYOUT_TARGET_US,
167        )
168    };
169    #[cfg(not(feature = "synthetic"))]
170    let playout_bus = Engine::new().start_playout_bus(
171        params.devices.speaker.subscribe(),
172        apm.clone(),
173        clock,
174        PLAYOUT_TARGET_US,
175    );
176
177    let capture_stop = Arc::new(AtomicBool::new(false));
178    let mut tasks: Vec<JoinHandle<()>> = Vec::new();
179
180    // 4b. RTC debug stats: clear the per-establish counters (the registry itself
181    //     survives reconnects — it keeps the reconnect count), record this session's
182    //     identity, mark the transport connected, and start the 1 Hz emitter that
183    //     samples the registry into the `watch_stats` observable. Torn down +
184    //     respawned per establish like the other tasks.
185    params.stats.reset_session();
186    // The reset zeroes the per-session gauges, so re-state the cross-reconnect mic
187    // mute onto the fresh session (the flag itself survives; the gauge does not).
188    params
189        .stats
190        .set_muted(params.mute.mic.load(Ordering::Relaxed));
191    // Same shape as the mute re-assert: whether this session's capture is
192    // processed (AEC/NS/AGC) — the field truth behind "you sound quiet" reports.
193    params.stats.set_apm_active(apm.is_some());
194    params
195        .stats
196        .set_session_identity(self_id.0, &join.relay_url, &sfu_node_id);
197    params.stats.set_transport_state(TransportState::Connected);
198    tasks.push(crate::stats::spawn_emitter(
199        params.stats_out.clone(),
200        params.stats.clone(),
201        playout_bus.clone(),
202    ));
203
204    // 5. Publish mic (Opus). Announce on the control plane, then run the
205    //    capture→encode→publish task (subscribes to the SHARED mic device watch).
206    control_outbound
207        .send(ClientMessage::Publish {
208            track: TrackDecl {
209                kind: MediaKind::Audio,
210                source: Source::Microphone,
211                codec: CtlCodec::Opus,
212                low: false,
213            },
214        })
215        .map_err(|_| {
216            HoundError::media(
217                MediaCode::ControlClosedBeforePublish,
218                "control plane closed before mic publish",
219            )
220        })?;
221    tasks.push(spawn_mic_publish(
222        relay.clone(),
223        space_id,
224        self_user_id,
225        self_id,
226        clock,
227        apm,
228        params.devices.mic.subscribe(),
229        params.mute.clone(),
230        params.voice.clone(),
231        channels.levels.clone(),
232        capture_stop.clone(),
233        params.stats.clone(),
234        #[cfg(feature = "synthetic")]
235        params.capture_source,
236        #[cfg(feature = "synthetic")]
237        params.synthetic_silence.clone(),
238    ));
239
240    // 5b. Re-assert the local mute intent onto THIS session's roster (mute is
241    //     per-session control-plane state; the local flags are call-lifetime).
242    tasks.push(spawn_mute_sync(
243        channels.view.subscribe(),
244        params.mute.clone(),
245        control_outbound.clone(),
246    ));
247
248    // 6. Publish camera (H.264) if enabled. Best-effort: a missing camera does not
249    //    fail the establish (audio-only is valid). Only announce the video track when
250    //    the camera is on. The shared `camera_enabled` flag survives a reconnect.
251    if params.camera_enabled.load(Ordering::Relaxed) {
252        control_outbound
253            .send(ClientMessage::Publish {
254                track: TrackDecl {
255                    kind: MediaKind::Video,
256                    source: Source::Camera,
257                    codec: CtlCodec::H264,
258                    low: false,
259                },
260            })
261            .map_err(|_| {
262                HoundError::media(
263                    MediaCode::ControlClosedBeforePublish,
264                    "control plane closed before cam publish",
265                )
266            })?;
267    }
268    tasks.push(spawn_cam_publish(
269        relay.clone(),
270        space_id,
271        self_user_id,
272        self_id,
273        clock,
274        params.devices.camera.subscribe(),
275        params.camera_enabled.clone(),
276        sink.clone(),
277        control_outbound.clone(),
278        capture_stop.clone(),
279        conn.clone(),
280        view_rx.clone(),
281        keyframe_tx.subscribe(),
282        feedback_tx.subscribe(),
283        quality_tx.subscribe(),
284        params.stats.clone(),
285        #[cfg(feature = "synthetic")]
286        params.capture_source,
287    ));
288
289    // 7. Subscribe loop: reconcile remote tracks → decode → sink / mix bus. On a
290    //    reconnect this re-diffs the fresh roster and re-subscribes every remote.
291    tasks.push(spawn_subscribe_loop(
292        view_rx.clone(),
293        self_id,
294        space_id,
295        relay.clone(),
296        control_outbound.clone(),
297        sink.clone(),
298        playout_bus,
299        channels.levels.clone(),
300        clock,
301        params.stats.clone(),
302        params.video_paused.subscribe(),
303        params.tile_quality.clone(),
304        params.tile_policies.clone(),
305        params.hidden_tiles.clone(),
306        params.volumes.clone(),
307        speakers.clone(),
308    ));
309
310    // 7b. Report OUR speaking level so every other participant can rank us. Reads
311    //     the same mic meter the level UI does (zero while muted), on the ranker's
312    //     own 300 ms grid, and goes quiet entirely when we are not talking.
313    tasks.push(spawn_speaking_report(
314        channels.levels.subscribe(),
315        control_outbound.clone(),
316        capture_stop.clone(),
317    ));
318
319    // 7c. Re-announce OUR decode caps if a codec appears (or is turned off) during
320    //     the call — `Hello`'s snapshot would otherwise stand for the whole
321    //     session, leaving a late openh264 blob unusable to every producer.
322    tasks.push(spawn_decode_caps_announcer(
323        control_outbound.clone(),
324        announced_decode_caps,
325    ));
326
327    // 8. voice/confirm with our (fresh) peer id (best-effort) + heartbeat. No
328    //    relay-token refresh task: a reconnect re-mints the token via `voice/join`,
329    //    so a token-expiry SFU drop heals through the supervisor.
330    {
331        let rest = rest.clone();
332        let peer = self_id.0 as u16;
333        call_runtime().spawn(async move {
334            let _ = rest.voice_confirm(space_id, Some(peer)).await;
335        });
336    }
337    tasks.push(spawn_heartbeat(rest.clone(), space_id));
338
339    Ok(LiveSession {
340        relay,
341        rtt_conn: conn.clone(),
342        endpoint,
343        conn,
344        control_outbound,
345        self_id,
346        self_user_id,
347        clock,
348        view_rx,
349        keyframe_tx,
350        feedback_tx,
351        quality_tx,
352        control_closed,
353        tasks,
354        capture_stop,
355    })
356}

358/// Keep the ROSTER's mute flags in step with our local intent for this session's
359/// lifetime. Mute is per-session control-plane state — a fresh establish (and every
360/// track republish) starts un-muted — while [`MuteFlags`] are call-lifetime, so
361/// without this a call that heals while muted goes live to everyone, and every
362/// receiver reads the silence of a track it believes unmuted as a wedged subscription
363/// (the audio stall watchdog re-subscribes it for nothing).
364///
365/// Level-triggered: re-diff the view on each change and send only on a mismatch, so it
366/// is idempotent and settles as soon as the server echoes the change back.
367fn spawn_mute_sync(
368    mut view_rx: watch::Receiver<CallView>,
369    mute: Arc<MuteFlags>,
370    outbound: mpsc::UnboundedSender<ClientMessage>,
371) -> JoinHandle<()> {
372    call_runtime().spawn(async move {
373        loop {
374            // Scoped: the watch borrow must not be held across the await below.
375            {
376                let view = view_rx.borrow_and_update();
377                if let Some(me) = view.self_id {
378                    for (source, want) in [
379                        (Source::Microphone, mute.mic.load(Ordering::Relaxed)),
380                        (Source::Screen, mute.screen_audio.load(Ordering::Relaxed)),
381                    ] {
382                        for t in view.tracks.iter().filter(|t| {
383                            t.peer == me && t.kind == MediaKind::Audio && t.source == source
384                        }) {
385                            if t.muted != want {
386                                let _ = outbound.send(ClientMessage::SetMuted {
387                                    track: hound_control::TrackId(t.id),
388                                    muted: want,
389                                });
390                            }
391                        }
392                    }
393                }
394            }
395            if view_rx.changed().await.is_err() {
396                return; // the call's view sender is gone — the session is over
397            }
398        }
399    })
