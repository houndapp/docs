1//! Inbound audio: pull → reorder by timestamp → Opus decode (bounded PLC on
2//! genuine gaps) → per-peer jitter buffer on the mix bus, plus level metering.
3//! The encoded-frame reorder hold undoes the cross-stream reordering a
4//! screenshare induces on the shared connection BEFORE the stateful decoder;
5//! release is arrival-driven — count-based in the steady state, and drained
6//! outright once the sender goes DTX (see [`release_next`]) — because the moq
7//! group read is not cancel-safe mid-object, so no recv timeout is ever raced
8//! against it. For the same reason
9//! a wedged track can't be detected here: the pipeline publishes an arrival
10//! counter instead, and the subscribe loop's stall watchdog kills it out of band.
11
12use super::*;
13
14/// Frames held in the encoded-audio reorder buffer to undo network reordering before
15/// the (stateful) Opus decode. `recv_group` delivers frames in ARRIVAL order; on the
16/// shared connection a screenshare reorders the tiny per-frame audio streams, so a
17/// frame can arrive after a later one. Decoding in arrival order forces dropping the
18/// late frame and concealing it with PLC — reordered becomes lost, and on a busy
19/// connection that is continuous distortion. Holding a few frames and releasing the oldest by
20/// timestamp restores a clean in-order stream into the decoder; genuine loss still
21/// surfaces as a real gap. 3 frames (~60 ms) absorbs the multi-frame reordering the
22/// screenshare induces; the added latency is recovered by the PCM jitter buffer
23/// shrinking once arrival is smooth. NOT done via a recv timeout — the moq group read
24/// is not cancel-safe mid-object — so release is arrival-driven throughout; see
25/// [`release_next`] for the count rule and the DTX drain that suspends it.
26const AUDIO_REORDER_HOLD: usize = 3;
27
28/// A pulled frame plus **when it came off the wire**.
29///
30/// The arrival instant has to be captured at the pull and carried through the hold,
31/// because the hold is count-based: how long a frame waits in it is a function of when
32/// the NEXT frames arrive, not of the network. Two consumers need the honest number
33/// and would otherwise silently measure the hold instead of the link — the jitter
34/// estimator (differential over `arrivalΔ − tsΔ`; see [`BusProducer::push_at`]) and,
35/// through it, the recommended playout depth.
36struct HeldFrame {
37    /// Local time (on the shared [`CaptureClock`]) the pull completed.
38    arrival_us: i64,
39    frame: WireFrame,
40}
41
42/// Insert `frame` into `buf` kept sorted ascending by timestamp, dropping an exact
43/// duplicate-timestamp frame (a retransmit of something already buffered). Returns
44/// false if dropped. Pure + unit-tested — the reorder correctness hinges on this.
45fn reorder_insert(buf: &mut Vec<HeldFrame>, held: HeldFrame) -> bool {
46    match buf.binary_search_by_key(&held.frame.timestamp_us, |f| f.frame.timestamp_us) {
47        Ok(_) => false,
48        Err(i) => {
49            buf.insert(i, held);
50            true
51        }
52    }
53}

55/// The hold's release decision — the whole of it, so the pull loop below is only
56/// network I/O and the policy can be driven by tests against the real jitter machinery.
57/// Returns the frame to process now, or `None` meaning "pull more".
58///
59/// Three rules, in order:
60///
61/// 1. **The suppression drain.** `drain` is set when a DTX-sized frame arrives, which is
62///    the sender telling us it has stopped transmitting: the next frame is a keepalive
63///    interval (~500 ms) away, so there is nothing left to reorder against and every
64///    frame still held is waiting on a network event that will not come for half a
65///    second. Empty the hold, then clear the flag and resume normally.
66///
67///    Holding through suppression is not merely late, it corrupts the receive-side
68///    estimates. The hold releases by COUNT, so each keepalive would be handed on one
69///    arrival later — ~1.5 s behind the wire — and every push would then land past its
70///    scheduled playout (a run of overdue arrivals → shallow re-anchors), while the
71///    500 ms release cadence would read as transit deviation and pin the recommended
72///    depth at its cap for the whole peak window.
73///
74///    What the drain gives up is reorder protection for the frames it releases early.
75///    That is the right trade and a small one: the frames in question are the tail of
76///    the confirmation window and the keepalives themselves — silence — and past the
77///    first keepalive the stream is 500 ms/frame, which no reordering reaches across.
78///    Speech is never drained early, because a coded frame never arms the flag.
79/// 2. **Steady state.** Release the oldest once more than [`AUDIO_REORDER_HOLD`] are
80///    buffered, so network reordering is undone before the stateful decode.
81/// 3. **End of track.** Drain whatever remains, oldest-first.
82///
83/// All three are arrival-driven, never timed: the moq group read is not cancel-safe
84/// mid-object, so nothing may be raced against `next_frame`.
85fn release_next(buf: &mut Vec<HeldFrame>, drain: &mut bool, ended: bool) -> Option<HeldFrame> {
86    if *drain {
87        if !buf.is_empty() {
88            return Some(buf.remove(0));
89        }
90        *drain = false;
91    }
92    if buf.len() > AUDIO_REORDER_HOLD || (ended && !buf.is_empty()) {
93        return Some(buf.remove(0));
94    }
95    None
96}

98/// Audio decode pipeline: pull frames, **reorder by timestamp**, Opus-decode (PLC on
99/// genuine gaps via the decoder's `None`), push into the per-peer jitter buffer on the
100/// mix bus, and meter levels.
101#[allow(clippy::too_many_arguments)]
102pub(super) async fn audio_pipeline(
103    mut subscriber: MoqTrackSubscriber,
104    peer: PeerId,
105    source: Source,
106    playout_bus: Arc<PlayoutBus>,
107    levels: watch::Sender<LevelsView>,
108    clock: CaptureClock,
109    stop: Arc<AtomicBool>,
110    peer_stats: Arc<crate::stats::PeerStats>,
111    gain: Arc<std::sync::atomic::AtomicU32>,
112    delivered: Arc<AtomicBool>,
113    // Monotone count of frames pulled off the wire, sampled by the subscribe loop's
114    // stall watchdog. Bumped on ARRIVAL (not on decode) so a stream whose frames fail
115    // to decode still reads as live — the watchdog only asks "is the relay still
116    // delivering?", which is the one thing the unbounded pull can wedge on.
117    arrivals: Arc<AtomicU64>,
118) {
119    use hound_engine::types::OpusPacket;
120
121    let Ok(mut decoder) = OpusDecoder::new() else {
122        return;
123    };
124    // The mix bus mixes all slots regardless of label; the label only attributes
125    // starve-stats, so name screen app-audio honestly.
126    let is_screen = source == Source::Screen;
127    let akind = if is_screen {
128        InboundAudioKind::Screen
129    } else {
130        InboundAudioKind::Mic
131    };
132    // Tag the producer with the peer id so a recorded underflow glitch names the peer.
133    let producer =
134        playout_bus.register_tagged(if is_screen { "screen_audio" } else { "mic" }, peer.0);
135    let mut decoded: u64 = 0;
136    // Loss/reorder detection from the sender's capture timestamps. `prev_ts` advances
137    // only on in-order frames; `frame_us` (the Opus frame duration) is learned from the
138    // first decoded frame so the gap math doesn't assume 20 ms. The engine decodes at
139    // 48 kHz mono, so duration = samples · 1e6 / 48000.
140    const ENGINE_RATE_HZ: i64 = 48_000;
141    let mut prev_ts: Option<i64> = None;
142    // Was the frame before this one DTX-sized? A gap that follows one is the sender not
143    // transmitting (suppression/DTX) whatever its size — see `classify_audio_gap`.
144    let mut prev_was_dtx = false;
145    let mut frame_us: i64 = 20_000; // sane default until the first frame refines it
146    let mut learned_frame_us = false;
147    // Cadence diagnostics — a retained debug tool. Distinguishes the THREE causes of an
148    // active-stream gap that the smoothed jitter number hides: brief network loss
149    // (small ts gaps → PLC), sender DTX / capture discontinuity (large ts gaps → NOT
150    // loss, must not be PLC-fabricated), and sender↔receiver clock skew (ts-elapsed vs
151    // local-elapsed divergence → buffer slowly drains). Logged once per window.
152    let mut diag = CadenceDiag::default();
153    // Encoded-frame reorder buffer (sorted by ts): we pull from the network into it so
154    // reordering is undone BEFORE the stateful decode, and `release_next` owns when a
155    // frame comes back out — including the DTX drain `drain_hold` arms.
156    let mut reorder: Vec<HeldFrame> = Vec::with_capacity(AUDIO_REORDER_HOLD + 2);
157    let mut ended = false;
158    let mut drain_hold = false;
159    while !stop.load(Ordering::Relaxed) {
160        // Acquire the next frame to PROCESS, in TIMESTAMP order, refilling from the
161        // network until the hold will give one up. `next_frame` is never cancelled —
162        // only fully-received frames enter the buffer.
163        let held = loop {
164            if let Some(f) = release_next(&mut reorder, &mut drain_hold, ended) {
165                break Some(f);
166            }
167            if ended {
168                break None; // track over and the hold is dry
169            }
170            let now = clock.now();
171            match subscriber.next_frame(now).await {
172                Ok(Some(f)) => {
173                    arrivals.fetch_add(1, Ordering::Relaxed);
174                    // RTC-debug: one inbound Opus frame received (counted on arrival).
175                    peer_stats.audio_received(akind, f.data.len());
176                    drain_hold |= f.data.len() <= DTX_MAX_BYTES;
177                    // Stamped AFTER the await completes: this is the wire instant, and
178                    // the only place it can still be observed.
179                    let arrival_us = clock.now();
180                    reorder_insert(
181                        &mut reorder,
182                        HeldFrame {
183                            arrival_us,
184                            frame: f,
185                        },
186                    );
187                }
188                Ok(None) | Err(_) => ended = true,
189            }
190        };
191        let Some(HeldFrame { arrival_us, frame }) = held else {
192            break;
193        };
194        let ts = frame.timestamp_us;
195        // Local time at processing (after any reorder hold) — for the cadence skew calc.
196        let now = clock.now();
197        // Built up-front so the gap handler below can redeem this packet's in-band
198        // FEC for the frame that preceded it; also the decode input afterwards.
199        let encoded_len = frame.data.len();
200        let packet = OpusPacket {
201            data: frame.data,
202            timestamp_us: TimestampUs::from_micros(ts),
203        };
204
205        // Frames now arrive in timestamp order, so a non-advancing timestamp is a true
206        // duplicate/straggler the reorder buffer couldn't place (rare), not routine
207        // network reordering — skip it to keep the stateful decoder coherent.
208        if let Some(pt) = prev_ts {
209            let gap = ts - pt;
210            if gap <= 0 {
211                peer_stats.record_audio_loss(akind, 0, true);
212                continue;
213            }
214            diag.note_gap(gap);
215            // PLC conceals BRIEF loss only — see [`classify_audio_gap`], which draws
216            // the line. A larger gap is the sender not transmitting (Opus DTX plus the
217            // publisher's silence suppression: nothing but a keepalive every ~500 ms)
218            // or a capture discontinuity — NOT packet loss. Fabricating one
219            // extrapolated PLC frame per 20 ms of gap would mislabel silence as loss (a
220            // silent peer would produce more concealment than real audio), waste CPU,
221            // and — worst — splice garbage PLC into an ACTIVE stream where it
222            // leapfrogs/drops the real frame behind it.
223            match classify_audio_gap(gap, frame_us, prev_was_dtx) {
224                AudioGap::Continuous => {}
225                AudioGap::Conceal(missing) => {
226                    // The arriving packet carries in-band FEC for the frame
227                    // immediately before it, so the LAST missing slot is redeemed
228                    // from real (re-encoded) signal and only the rest fall back to
229                    // PLC extrapolation. A DTX-sized packet carries no FEC —
230                    // libopus would just conceal — so the guard keeps the two
231                    // paths honest in the stats.
232                    let fec = encoded_len > DTX_MAX_BYTES;
233                    let plc_until = if fec { missing - 1 } else { missing };
234                    for i in 1..=plc_until {
235                        let plc_ts = pt + frame_us * i;
236                        if let Ok(plc) = decoder.decode(None, TimestampUs::from_micros(plc_ts)) {
237                            // `push_concealed`, never `push`: this frame never crossed
238                            // the wire, so it has no arrival to fold into the jitter
239                            // estimator — and the only instant available here is the
240                            // pipeline's own `now`, which is exactly what that estimator
241                            // must never see. See [`BusProducer::push_concealed`].
242                            let _ = producer.push_concealed(clock.now(), plc_ts, &plc.samples);
243                        }
244                    }
245                    if fec {
246                        let fec_ts = pt + frame_us * missing;
247                        if let Ok(rec) =
248                            decoder.decode_fec(&packet, TimestampUs::from_micros(fec_ts))
249                        {
250                            let _ = producer.push(clock.now(), fec_ts, &rec.samples);
251                        }
252                    }
253                    peer_stats.record_audio_loss(akind, missing as u64, false);
254                    diag.plc += missing as u64;
255                }
256                AudioGap::Silence => {
257                    // Silence across the gap (correct — there was no audio), no loss
258                    // recorded, and the peer's meter goes dark at once rather than
259                    // holding its last speaking level until the next frame decodes.
260                    // Resync to the new frame (prev_ts advances after decode).
261                    diag.big_gaps += 1;
262                    if !is_screen {
263                        levels.send_modify(|l| {
264                            l.peers.insert(peer.0, 0.0);
265                        });
266                    }
267                    peer_stats.set_audio_level(akind, 0.0);
268                }
269            }
270        }
271
272        let Ok(mut pcm) = decoder.decode(Some(&packet), TimestampUs::from_micros(ts)) else {
273            continue;
274        };
275        delivered.store(true, Ordering::Relaxed);
276        prev_ts = Some(ts);
277        prev_was_dtx = encoded_len <= DTX_MAX_BYTES;
278        if !learned_frame_us && !pcm.samples.is_empty() {
279            frame_us = (pcm.samples.len() as i64 * 1_000_000 / ENGINE_RATE_HZ).max(1);
280            learned_frame_us = true;
281        }
282        // Per-stream playback gain (the tile volume/mute control), applied before the
283        // level meter and the mix push so both reflect what is actually played (a
284        // muted stream's meter reads 0 and its speaking ring stays off). Unity skips
285        // the pass entirely (the common case costs one atomic load).
286        let g = f32::from_bits(gain.load(Ordering::Relaxed));
287        if g != 1.0 {
288            apply_gain(&mut pcm.samples, g);
289        }
290        // Both mic + screen audio record their decoded level for the panel; only the
291        // mic drives the per-peer level meter / speaking ring (both key on peer id).
292        let level = reported_audio_level(encoded_len, pcm.rms());
293        peer_stats.set_audio_level(akind, level);
294        if !is_screen {
295            levels.send_modify(|l| {
296                l.peers.insert(peer.0, level);
297            });
298        }
299        // The mix bus pulls on its own RT clock; we only FILL the jitter buffer.
300        // Surface the post-push jitter-buffer health to the panel/report: depth
301        // (cur vs recommended), late/overflow drops, and this slot's underflow total.
302        // A depth that climbs while silent = buffered-but-never-released (a playout
303        // stall); recommended ≫ target = the buffer is too shallow for the link.
304        // `arrival_us` (the wire instant, carried through the hold) feeds the jitter
305        // estimator; `clock.now()` is when the frame actually became available for
306        // playout. Conflating them makes the reorder hold's own release cadence read as
307        // link jitter — see [`BusProducer::push_at`].
308        let status = producer.push_at(arrival_us, clock.now(), ts, &pcm.samples);
309        peer_stats.set_audio_playout_depth(akind, status.buffered_frames);
310        peer_stats.set_audio_jitter(akind, status.jitter_us.max(0) as u32);
311        peer_stats.set_audio_jitterbuf(
312            akind,
313            status.target_delay_us.max(0) as u32,
314            status.recommended_delay_us.max(0) as u32,
315            status.late_drops,
316            status.overflow_drops,
317            status.underflow_total,
318        );
319        decoded += 1;
320        // Feed the cadence diagnostic and emit a window summary periodically. This is
321        // the tool for attributing an active-stream gap: high `plc` = brief loss;
322        // high `big_gaps`/`max_gap_ms` = sender DTX or a capture stall (a sender-side
323        // problem, not network); large |skew_ppm| = clock drift slowly draining the
324        // buffer (needs resampling/retiming, not a deeper buffer).
325        if let Some((real, plc, big_gaps, max_gap_ms, skew_ppm)) = diag.observe(ts, now) {
326            tracing::info!(
327                ?source,
328                peer = peer.0,
329                real,
330                plc,
331                big_gaps,
332                max_gap_ms,
333                skew_ppm,
334                jitterbuf_ms = status.target_delay_us / 1000,
335                underflow_total = status.underflow_total,
336                "audio: cadence window"
337            );
338        }
339        if decoded == 1 || decoded.is_multiple_of(500) {
340            tracing::info!(
341                ?source,
342                peer = peer.0,
343                decoded,
344                "audio: decoded+mixed frames"
345            );
346        }
347    }
348    if !is_screen {
349        levels.send_modify(|l| {
350            l.peers.remove(&peer.0);
351        });
352    }
353    tracing::info!(?source, peer = peer.0, decoded, "audio: pipeline ended");
354}

364    /// One 20 ms frame at the engine's rate — what `frame_us` is learned as.
365    const FRAME_US: i64 = 20_000;
366    /// A coded-speech payload size, and a DTX one. The gap between them is the
367    /// sender's own speech/no-speech verdict, and the drain rule reads it.
368    const VOICED_BYTES: usize = 62;
369    const DTX_BYTES: usize = 2;
370
371    /// A CODED frame at `ts` (arrival = ts, i.e. a zero-latency link).
372    fn afr(ts: i64) -> HeldFrame {
373        held(ts, ts, VOICED_BYTES)
374    }
375
376    fn held(ts: i64, arrival_us: i64, bytes: usize) -> HeldFrame {
377        HeldFrame {
378            arrival_us,
379            frame: WireFrame {
380                sender: 0,
381                track: TrackKind::CamAudio,
382                codec: WireCodec::Opus,
383                keyframe: true,
384                frame_seq: 0,
385                timestamp_us: ts,
386                data: vec![ts as u8; bytes],
387            },
388        }
389    }
390
391    /// Release everything the hold will currently give up, appending timestamps to
392    /// `out`. The real [`release_next`], looped — never a restatement of its rules.
393    fn drain_available(
394        buf: &mut Vec<HeldFrame>,
395        drain: &mut bool,
396        ended: bool,
397        out: &mut Vec<i64>,
398    ) {
399        while let Some(h) = release_next(buf, drain, ended) {
400            out.push(h.frame.timestamp_us);
401        }
402    }

447    // ─── Receiving from a SUPPRESSING publisher ─────────────────────────────────
448
449    /// The wire timeline of a publisher that talks, goes quiet (suppressed, one
450    /// keepalive per `KEEPALIVE_INTERVAL`) for `quiet` frames' worth of wall time, then
451    /// talks again — `(capture timestamp, payload size)`, since the SIZE is the sender's
452    /// own speech/no-speech verdict and half the classification reads it. Capture
453    /// timestamps throughout: the publisher never invents any.
454    ///
455    /// `resume_lag_frames` offsets the resume from the keepalive cadence. A speaker does
456    /// not resume on the wire's schedule, so the real gap on the resume edge is uniform
457    /// in `(0, KEEPALIVE_INTERVAL]` — and a resume that lands early in that interval is
458    /// the geometry that produced the CI failure this file's fix addresses.
459    fn suppressed_timeline(quiet_frames: i64, resume_lag_frames: i64) -> Vec<(i64, usize)> {
460        let keepalive_us = KEEPALIVE_INTERVAL.as_micros() as i64;
461        // 1 s of speech.
462        let mut wire: Vec<(i64, usize)> = (0..50).map(|i| (i * FRAME_US, VOICED_BYTES)).collect();
463        let quiet_start = 50 * FRAME_US;
464        // The confirmation window still publishes every frame — DTX-sized ones, which is
465        // what arms the gate — and then only keepalives.
466        for i in 0..SUPPRESS_AFTER as i64 {
467            wire.push((quiet_start + i * FRAME_US, DTX_BYTES));
468        }
469        let mut t = quiet_start + SUPPRESS_AFTER as i64 * FRAME_US;
470        let mut last_keepalive = t;
471        let quiet_until = quiet_start + quiet_frames * FRAME_US;
472        while t + keepalive_us < quiet_until {
473            t += keepalive_us;
474            // A keepalive is a REAL captured frame, so its stamp lands on the frame grid.
475            last_keepalive = t - t % FRAME_US;
476            wire.push((last_keepalive, DTX_BYTES));
477        }
478        let resume_at = last_keepalive + resume_lag_frames * FRAME_US;
479        wire.extend((0..50).map(|i| (resume_at + i * FRAME_US, VOICED_BYTES))); // speech again
480        wire
481    }

483    /// Classify every gap in a wire timeline the way the pipeline does — each gap
484    /// against the payload size of the frame BEFORE it. Returns (concealed frames,
485    /// silences, continuous).
486    fn classify_timeline(wire: &[(i64, usize)]) -> (i64, u32, u32) {
487        let (mut concealed, mut silences, mut continuous) = (0i64, 0u32, 0u32);
488        for w in wire.windows(2) {
489            match classify_audio_gap(w[1].0 - w[0].0, FRAME_US, w[0].1 <= DTX_MAX_BYTES) {
490                AudioGap::Continuous => continuous += 1,
491                AudioGap::Conceal(n) => concealed += n,
492                AudioGap::Silence => silences += 1,
493            }
494        }
495        (concealed, silences, continuous)
496    }

498    /// A multi-second suppression gap must classify as SILENCE end to end: not one
499    /// concealed frame, not one frame counted lost. The failure this pins is the
500    /// receiver reading a quiet peer as a lossy one — which would put every silent
501    /// participant in the audio report's `network-loss` rule and make PLC (fabricated
502    /// audio) the dominant thing a silent call decodes.
503    #[test]
504    fn a_suppression_gap_classifies_as_silence_with_no_loss_or_plc() {
505        // Resumed a full keepalive interval after the last one — the ALIGNED case.
506        let wire = suppressed_timeline(500, KEEPALIVE_INTERVAL.as_micros() as i64 / FRAME_US);
507        let (concealed, silences, continuous) = classify_timeline(&wire);
508        assert_eq!(
509            concealed, 0,
510            "a suppressed publisher must never make the receiver conceal or count loss"
511        );
512        assert!(
513            silences >= 19,
514            "each keepalive gap is a silence classification; got {silences}"
515        );
516        assert!(
517            continuous > 90,
518            "the speech either side stays ordinary cadence; got {continuous}"

522    /// **The resume edge is silence WHEREVER it lands**, which the gap size alone cannot
523    /// tell you. A speaker resumes on their own schedule, so the gap between the last
524    /// keepalive and the first word is uniform in `(0, KEEPALIVE_INTERVAL]` — and every
525    /// resume landing inside the PLC band (~20% of them) used to conceal up to
526    /// [`MAX_PLC_FRAMES`] fabricated frames in front of the first word, count them as

530    /// frames on a peer who had merely been quiet.
531    ///
532    /// The aligned case in the test above cannot see this: it resumes exactly one
533    /// keepalive interval on, so the gap is a keepalive's worth and classifies as

548    // ─── The receive seam, driven against the REAL jitter machinery ─────────────
549    //
550    // These are the tests that catch the class the e2e scenario cannot see: the
551    // scenario's resume budget is dominated by ~2 s of 1 Hz stats quantization, so a
552    // 600 ms playout regression hides completely inside it. Here the seam is exact.
553
554    /// A frame on its way into the mix bus. `PlayoutBuffer` only ever reads
555    /// `timestamp_us()`, so this is the whole item.
556    #[derive(Debug)]
557    struct Pcm(i64);
558    impl Timestamped for Pcm {
559        fn timestamp_us(&self) -> i64 {
560            self.0
561        }
562    }
563
564    /// The receive seam: the real hold ([`reorder_insert`] + [`release_next`]) feeding
565    /// the real [`AdaptiveJitter`] + [`PlayoutBuffer`] the mix bus runs, wired the way
566    /// `BusProducer::push_at` wires them — the estimator on the WIRE arrival, the
567    /// buffer on the post-hold instant.
568    struct Seam {
569        buf: Vec<HeldFrame>,
570        drain: bool,
571        playout: PlayoutBuffer<Pcm>,
572        adapt: AdaptiveJitter,
573        /// Post-hold release instants, so a test can measure the added latency.
574        released_at: Vec<(i64, i64)>,
575    }
576
577    impl Seam {
578        fn new() -> Self {
579            Self {
580                buf: Vec::new(),
581                drain: false,
582                playout: PlayoutBuffer::new(PLAYOUT_TARGET_US),
583                adapt: AdaptiveJitter::with_floor(PLAYOUT_TARGET_US),
584                released_at: Vec::new(),
585            }
586        }
587
588        /// One frame off the wire at `arrival_us`, then everything the hold gives up as
589        /// a result — pushed at `arrival_us` too, since the hold does not block.
590        fn arrive(&mut self, ts: i64, arrival_us: i64, bytes: usize) {
591            self.drain |= bytes <= DTX_MAX_BYTES;
592            reorder_insert(&mut self.buf, held(ts, arrival_us, bytes));
593            while let Some(h) = release_next(&mut self.buf, &mut self.drain, false) {
594                // Exactly `BusProducer::push_at(h.arrival_us, now, ts, …)`.
595                self.adapt
596                    .on_frame(&mut self.playout, h.arrival_us, h.frame.timestamp_us);
597                self.playout.push(arrival_us, Pcm(h.frame.timestamp_us));
598                self.released_at.push((h.frame.timestamp_us, arrival_us));
599                let _ = self.playout.pop_due(arrival_us);
600            }
601        }
602    }
603
604    /// A one-way link delay, so arrivals are not trivially equal to timestamps.
605    const TRANSIT_US: i64 = 5_000;
606
607    /// **10 s of suppression must leave the playout depth exactly where speech left
608    /// it.** THE F1 regression, pinned against the real estimator.
609    ///
610    /// Before the drain + wire-arrival stamping, this measured:
611    /// `recommended_delay_us` = 450 000 (the `MAX_ADAPTIVE_DEPTH_US` cap, held for the
612    /// full 90 s peak window) and a run of shallow re-anchors — because each keepalive
613    /// was released one arrival late by the count-based hold, so a 20 ms timestamp step
614    /// was reported against a 500 ms release step. A peer who had merely been quiet came
615    /// back with ~0.6 s of latency stapled to their first word.
616    #[test]
617    fn a_suppressed_window_leaves_the_playout_depth_at_the_floor() {
618        let mut s = Seam::new();
619        // 1 s of speech.
620        for k in 0..50i64 {
621            let ts = k * FRAME_US;
622            s.arrive(ts, ts + TRANSIT_US, VOICED_BYTES);
623        }
624        let speaking = s.adapt.recommended_delay_us();
625        assert_eq!(
626            speaking, PLAYOUT_TARGET_US,
627            "precondition: a clean link idles at the floor"
628        );
629        let base_resyncs = s.playout.shallow_resyncs();
630
631        // 10 s suppressed: the three frames genuinely in the hold when quiet begins,
632        // then one keepalive every 500 ms.
633        let keepalive_us = KEEPALIVE_INTERVAL.as_micros() as i64;
634        let quiet_start = 50 * FRAME_US;
635        let mut t = quiet_start;
636        while t < quiet_start + 10 * 1_000_000 - keepalive_us {
637            t += keepalive_us;
638            let ts = t - t % FRAME_US;
639            s.arrive(ts, t + TRANSIT_US, DTX_BYTES);
640        }
641
642        assert_eq!(
643            s.adapt.recommended_delay_us(),
644            speaking,
645            "10 s of keepalives must leave the recommended depth exactly where the live \
646             stream left it (jitter {} µs)",
647            s.adapt.jitter_us()
648        );
649        assert_eq!(
650            s.playout.shallow_resyncs(),
651            base_resyncs,
652            "a suppression gap must not erode the anchor: the arrivals are on time, \
653             they are simply far apart"
654        );
655        assert_eq!(s.playout.late_drops(), 0, "no keepalive is a straggler");
656    }

658    /// **Silence ONSET must not spike the peak either.** When the sender goes quiet the
659    /// hold is genuinely carrying [`AUDIO_REORDER_HOLD`] real frames; the drain rule
660    /// releases all of them on the first keepalive's arrival. That is the moment three
661    /// frames are handed over at once, and if the estimator saw the release instant
662    /// rather than each frame's own arrival it would book a ~500 ms deviation right
663    /// there — enough on its own to pin the recommendation at the cap for 90 s.
664    #[test]
665    fn the_frames_stranded_in_the_hold_at_silence_onset_do_not_spike_the_peak() {
666        let mut s = Seam::new();
667        for k in 0..50i64 {
668            let ts = k * FRAME_US;
669            s.arrive(ts, ts + TRANSIT_US, VOICED_BYTES);
670        }
671        let held_before = s.buf.len();
672        assert_eq!(
673            held_before, AUDIO_REORDER_HOLD,
674            "precondition: the hold is full of real frames when the room goes quiet"
675        );
676        let peak_before = s.adapt.recommended_delay_us();
677
678        // The first keepalive, 500 ms later. It drains the stranded frames with it.
679        let quiet_start = 50 * FRAME_US;
680        let t = quiet_start + KEEPALIVE_INTERVAL.as_micros() as i64;
681        s.arrive(t - t % FRAME_US, t + TRANSIT_US, DTX_BYTES);
682        assert!(
683            s.buf.is_empty(),
684            "the drain rule empties the hold, it does not top it up"
685        );
686        assert_eq!(
687            s.adapt.recommended_delay_us(),
688            peak_before,
689            "onset must cost the estimate nothing (peak {} µs)",
690            s.adapt.peak_us()
691        );
692    }

694    /// **The first word back plays on the normal schedule.** The resume latency, in
695    /// microseconds rather than the e2e budget's seconds.
696    ///
697    /// Measured post-fix: the first voiced frame is released from the hold 60 ms after
698    /// it arrives (three subsequent arrivals — the ordinary live-stream cost of
699    /// [`AUDIO_REORDER_HOLD`], paid by every frame of every stream) and comes due one
700    /// [`PLAYOUT_TARGET_US`] later. Pre-fix the same measurement was ~610 ms: the depth
701    /// had been dragged to the 450 ms cap by the suppression gaps.
702    #[test]
703    fn the_first_voiced_frame_after_ten_quiet_seconds_plays_on_the_normal_schedule() {
704        let mut s = Seam::new();
705        for k in 0..50i64 {
706            let ts = k * FRAME_US;
707            s.arrive(ts, ts + TRANSIT_US, VOICED_BYTES);
708        }
709        let keepalive_us = KEEPALIVE_INTERVAL.as_micros() as i64;
710        let quiet_start = 50 * FRAME_US;
711        let mut t = quiet_start;
712        while t < quiet_start + 10 * 1_000_000 - keepalive_us {
713            t += keepalive_us;
714            let ts = t - t % FRAME_US;
715            s.arrive(ts, t + TRANSIT_US, DTX_BYTES);
716        }
717
718        // Speech resumes: a live 20 ms stream again.
719        let resume_ts = quiet_start + 10 * 1_000_000;
720        let resume_arrival = resume_ts + TRANSIT_US;
721        s.released_at.clear();
722        for k in 0..10i64 {
723            let ts = resume_ts + k * FRAME_US;
724            s.arrive(ts, ts + TRANSIT_US, VOICED_BYTES);
725        }
726        let (_, released_at) = *s
727            .released_at
728            .iter()
729            .find(|(ts, _)| *ts == resume_ts)
730            .expect("the first voiced frame is released");
731        let hold_cost = released_at - resume_arrival;
732        assert_eq!(
733            hold_cost,
734            AUDIO_REORDER_HOLD as i64 * FRAME_US,
735            "the first voiced frame waits the ordinary hold ({} ms), not longer",
736            AUDIO_REORDER_HOLD as i64 * FRAME_US / 1000
737        );
738        assert_eq!(
739            s.playout.target_delay_us(),
740            PLAYOUT_TARGET_US,
741            "and it plays at the normal 80 ms depth — pre-fix the suppression gaps had \
742             dragged this to the 450 ms adaptive cap (recommended {} µs, peak {} µs)",
743            s.adapt.recommended_delay_us(),
744            s.adapt.peak_us()
745        );
746        assert_eq!(
747            s.playout.late_drops(),
748            0,
749            "nothing was thrown away across the resume"
750        );
751    }

753    /// A suppressing publisher does not advance `frame_seq` for a frame it never sent,
754    /// so its stream has SEQ HOLES. No audio consumer may read one as loss: the reorder
755    /// buffer sorts by timestamp, the gap math is timestamp-driven, and the stall
756    /// watchdog counts arrivals — video's `DownlinkDemand` (the one thing that does fold
757    /// `frame_seq`) has no audio counterpart. This pins that the ordering is unaffected.
758    #[test]
759    fn the_audio_path_reads_nothing_from_frame_seq() {
760        let (mut buf, mut drain, mut released) = (Vec::new(), false, Vec::new());
761        // Timestamps advance on the 20 ms grid; seq jumps as suppression withholds.
762        for (i, seq) in [0u32, 1, 2, 40, 41, 900, 901, 902].into_iter().enumerate() {
763            let mut f = afr(i as i64 * FRAME_US);
764            f.frame.frame_seq = seq;
765            reorder_insert(&mut buf, f);
766            drain_available(&mut buf, &mut drain, false, &mut released);
767        }
768        drain_available(&mut buf, &mut drain, true, &mut released);
769        assert_eq!(
770            released,
771            (0..8).map(|i| i * FRAME_US).collect::<Vec<_>>(),
772            "release order follows timestamps alone; seq holes are invisible here"
773        );
774    }
