1//! The subscribe loop: reconcile remote tracks → decode → sink / mix bus. Diffs
2//! the `SessionView` roster into per-track pipelines ([`start_pipeline`]), picks
3//! nothing / low layer / source per tile demand — and, on the source track, which
4//! TEMPORAL view of it ([`pick_view`]: the whole track, or the relay's derived
5//! half-rate `…_t0`) — debounced through the policy hysteresis, and gated at join
6//! by [`SourceHold`] so an unserved tile waits for the right rung instead of
7//! stampeding the source. It self-heals: every
8//! pipeline carries a [`PipelineEndGuard`] whose Drop reports the death so the
9//! reconcile reaps + re-subscribes it (restart-paced).
10//! Invariant: audio subscribes with arrival-order drain and the smaller moq
11//! priority value — audio must strictly outrank video on the shared connection.

15/// Lock-free single-slot handle for one video pipeline's downlink-driven demand:
16/// the video pull loop ([`video_pipeline`]) writes its latest
17/// [`hound_media::QualityRequest`] here as frames arrive (via
18/// [`hound_media::DownlinkDemand`]); the reconcile loop below reads it every tick.
19/// A plain atomic (not a lock) so the per-frame write on the pull loop never
20/// blocks. `None` (audio, or a pipeline that hasn't decided yet) reads as
21/// [`hound_media::QualityRequest::Source`] — the optimistic default, matching
22/// [`hound_media::DownlinkDemand::new`].
23///
24/// Two consumers, one per half of the receiver's adaptation, and the SAME value
25/// answers both:
26///
27/// * folded per track with that tile's own demand
28///   ([`hound_media::most_constrained`]) — the final clamp, which no budget can lift;
29/// * compared against what the allocator granted that tile
30///   ([`hound_media::downlink_deficient`]) to produce this receiver's DEFICIENT
31///   verdict, which drives the total video budget.
32#[derive(Debug)]
33pub(super) struct DownlinkCell(AtomicU64);
34
35/// Sentinel encoding `QualityRequest::Source` in the packed atomic — a `Capped`
36/// bitrate this large never occurs (an encode rung tops out in the low millions).
37const SOURCE_SENTINEL: u64 = u64::MAX;
38
39impl DownlinkCell {
40    fn new() -> Self {
41        Self(AtomicU64::new(SOURCE_SENTINEL))
42    }
43
44    pub(super) fn store(&self, req: hound_media::QualityRequest) {
45        let v = match req {
46            hound_media::QualityRequest::Source => SOURCE_SENTINEL,
47            // This cell carries the BANDWIDTH half of the demand only; visibility
48            // is the tile half, folded in by `most_constrained` at the call site.
49            // A `Hidden` can never be produced here (see `DownlinkDemand::evaluate`)
50            // and is stored as its cap rather than silently widened to source.
51            hound_media::QualityRequest::Capped(bps) | hound_media::QualityRequest::Hidden(bps) => {
52                bps.min(SOURCE_SENTINEL - 1)
53            }
54        };
55        self.0.store(v, Ordering::Relaxed);
56    }
57
58    fn load(&self) -> hound_media::QualityRequest {
59        match self.0.load(Ordering::Relaxed) {
60            SOURCE_SENTINEL => hound_media::QualityRequest::Source,
61            bps => hound_media::QualityRequest::Capped(bps),
62        }
63    }
64}

66/// Floor wake-up for the reconcile loop. Every OTHER wake source is event-driven
67/// (roster change, pause flip, tile-demand change, pipeline death, a pending
68/// debounced switch) and in a steady call none of them fire for minutes — so the
69/// signals the pipelines fold per frame ([`DownlinkCell`], the audio arrival
70/// counter) would never be observed and a congestion downgrade / a wedged audio
71/// track could not actuate. [`LayerHysteresis`]'s debounce+dwell bounds any churn
72/// a periodic pass could cause, so polling is safe. 500 ms is the reconcile rate
73/// the dwell tests in [`super::policy`] already model.
74const RECONCILE_TICK: Duration = Duration::from_millis(500);
75
76/// How long an inbound AUDIO subscription may deliver NOTHING before the reconcile
77/// treats it as wedged. The audio pull is deliberately unbounded — the moq group
78/// read is not cancel-safe mid-object, so no timeout may ever be raced against
79/// `next_frame` — which means a relay that stops delivering a track WITHOUT
80/// closing it parks the pipeline task forever: its [`PipelineEndGuard`] never
81/// drops, the reconcile never reaps it, and that peer is silent for the rest of
82/// the establish. This is the out-of-band equivalent of video's
83/// `FIRST_KEYFRAME_DEADLINE`.
84///
85/// **What an unmuted sender guarantees.** Not a frame every 20 ms: the voice encoder
86/// runs with DTX and the publisher suppresses the resulting silence
87/// (`super::policy::SilenceGate`), so a quiet-but-unmuted peer legitimately sends
88/// ~2 frames/s. What it *does* guarantee is a real DTX frame at least every
89/// `super::policy::KEEPALIVE_INTERVAL` (500 ms), and that keepalive is precisely what
90/// feeds this watchdog. 8 s is 16 keepalive intervals — far above any
91/// jitter/retransmit tail, and with room for several lost keepalives before a healthy
92/// quiet peer could be accused — while keeping a genuinely wedged peer's voice loss
93/// inside ~10 s. See [`StallWatch`] for why mute is exempt.
94const AUDIO_STALL: Duration = Duration::from_secs(8);

96/// Pure liveness policy for one inbound audio subscription (unit-tested). The
97/// pipeline bumps a monotone arrival counter as frames come off the wire; this
98/// samples it on the reconcile tick and reports a subscription that has not moved
99/// for `stall`.
100///
101/// A `muted` observation RE-ARMS the window instead of accumulating stall time: a
102/// muted mic gates before encode and publishes nothing at all, so silence there is
103/// healthy — accusing it would reap + re-subscribe every muted peer in the call
104/// every window, churning the connection audio shares (an "audio is sacred"
105/// violation of exactly the kind this watchdog exists to prevent).
106#[derive(Debug)]
107struct StallWatch {
108    last_count: u64,
109    /// When the counter last moved (or the window was last re-armed).
110    since: Instant,
111}
112
113impl StallWatch {
114    fn new(now: Instant) -> Self {
115        Self {
116            last_count: 0,
117            since: now,
118        }
119    }
120
121    /// Fold one sample of the arrival counter. Returns true once the subscription
122    /// has delivered nothing for `stall` while the roster reports it unmuted.
123    fn observe(&mut self, count: u64, now: Instant, muted: bool, stall: Duration) -> bool {
124        if muted || count != self.last_count {
125            self.last_count = count;
126            self.since = now;
127            return false;
128        }
129        now.duration_since(self.since) >= stall
130    }
131}

140/// The reconcile select's wait: the soonest pending debounced switch, FLOORED by
141/// [`RECONCILE_TICK`] so the loop always wakes to poll the per-frame signals even
142/// when no event is pending. Pure (takes `now`) so the floor is unit-tested.
143fn reconcile_wait(next_deadline: Option<Instant>, now: Instant) -> Duration {
144    next_deadline
145        .map(|d| d.saturating_duration_since(now).min(RECONCILE_TICK))
146        .unwrap_or(RECONCILE_TICK)
147}

149/// Which track of one (peer, video source) group to subscribe to for the committed
150/// demand `req`. Pure, so the fallbacks are unit-tested. `None` means **subscribe to
151/// nothing** — the reconcile's `gone` sweep then tears any existing pipeline down and
152/// sends `Unsubscribe`, and this peer's AUDIO is untouched (a different track).
153///
154/// `req` is `None` while the UI has not reported this tile even once; `source_ok`
155/// carries the bounded escalation gate ([`SourceHold`]) that decides whether an
156/// unserved tile may take the SOURCE track yet.
157///
158/// The roster is the ONLY liveness signal available here, and the producer keeps it
159/// truthful: it withdraws the `_l0` [`TrackDecl`] whenever it deactivates the low
160/// encoder (see [`super::simulcast::LowDecl`]). So:
161///
162/// - a HIDDEN tile subscribes to nothing at all. Its demand still goes to the
163///   producer (`RequestQuality { hidden: true }`) — a fully-hidden receiver set
164///   *sizes* the shared low layer rather than letting it fall back to source, which
165///   is what keeps the thumbnail rung published and makes an unhide a subscribe
166///   rather than a wait on a producer (Octo's "forward every layer so a switch is
167///   instant", inverted to one relay);
168/// - a capped demand takes the low track when its decl is present; when it is not,
169///   it takes the source only if [`SourceHold`] allows. Subscribing to a decl-less
170///   or torn-down layer yields nothing and costs that tile the full `VIDEO_STALL` →
171///   `FIRST_KEYFRAME_DEADLINE` → restart-ladder walk (minutes of black video);
172/// - a low track WITHOUT its source is never picked at all. The producer's source
173///   encode is never dropped (the low layer is purely additive), so a lone `_l0` is
174///   by construction stale — its owner stopped publishing that video and only the
175///   source's `Unpublish` has reached us so far.
176///
177/// When the low decl (re)appears the reconcile switches to it through the normal
178/// [`LayerHysteresis`] path — source→low is a downgrade, so it is debounced.
179fn pick_layer(
180    req: Option<hound_media::VideoAllocation>,
181    src_id: Option<u32>,
182    low_id: Option<u32>,
183    source_ok: bool,
184) -> Option<u32> {
185    // A low track is only ever picked alongside its source (see above).
186    let low_id = src_id.and(low_id);
187    match req.map(|r| r.demand) {
188        Some(hound_media::QualityRequest::Hidden(_)) => None,
189        Some(hound_media::QualityRequest::Capped(_)) => {
190            low_id.or_else(|| src_id.filter(|_| source_ok))
191        }
192        Some(hound_media::QualityRequest::Source) => src_id,
193        // No report yet: this tile has stated no demand, so it gets the source only
194        // once the hold gives up waiting for one.
195        None => src_id.filter(|_| source_ok),
196    }
197}

199/// Which temporal view of the picked track this tile subscribes to.
200///
201/// The spatial pick above is a control-plane TRACK ID; the temporal view is not —
202/// the relay derives `…_t0` from the publisher's own broadcast and nobody
203/// declares it, so there is no id to pick and nothing to wait for. It is decided
204/// here, purely from the committed allocation, and it applies only when the pick
205/// actually landed on the SOURCE track: there is no base-layer view of `_l0`
206/// (`hound_media::base_view_offered`), so a `Capped` demand that took the low
207/// track — or a `Source` demand the [`SourceHold`] waiver put on the source —
208/// takes it whole.
209fn pick_view(
210    req: Option<hound_media::VideoAllocation>,
211    picked: u32,
212    src_id: Option<u32>,
213) -> hound_media::TemporalView {
214    match req {
215        Some(r) if src_id == Some(picked) => r.view,
216        _ => hound_media::TemporalView::Full,
217    }
218}

220/// How long a video track may go unsubscribed while the receiver waits for the
221/// evidence it needs to pick the RIGHT rung, before it gives up and takes the source.
222///
223/// Two things are waited on, both of which arrive in well under a second in a healthy
224/// call: the UI's first size report for a freshly-rostered tile (a texture bind plus
225/// one layout pass), and the producer's `_l0` decl after our `RequestQuality` (a
226/// control round trip, an encoder build and one capture frame). 3 s is ~10× either,
227/// far inside `FIRST_KEYFRAME_DEADLINE`, and — because the tile is blank for the whole
228/// wait either way — invisible when it is spent.
229const SOURCE_HOLD_GRACE: Duration = Duration::from_secs(3);

231/// The bounded gate on escalating a tile to the SOURCE track. One per
232/// (peer, video source); pure, so the whole policy is unit-tested.
233///
234/// ## What it fixes
235///
236/// `pick_layer` used to answer a capped demand with `low.or(source)`, and a tile with
237/// no report yet defaulted to `Source`. At a 32-peer join neither the low decls nor
238/// the tile reports exist yet, so every receiver subscribed to **31 source tracks**
239/// (~46 Mbps) and only walked back down a debounce later — a self-inflicted
240/// congestion collapse at exactly the moment the call is most fragile. Jitsi ships the
241/// same policy as a default (`initial-max-height-px = 0`: forward nothing until the
242/// receiver states its demand), and MoQT draft-19 names the shape `RENDEZVOUS_TIMEOUT`
243/// — "the duration the subscriber is willing to wait for a publisher to become
244/// available".
245///
246/// ## Why it cannot blank a legitimate tile
247///
248/// Three independent escapes, any one of which allows the source immediately:
249///
250/// 1. **`on_source`** — a tile already showing source video is never torn down to
251///    wait for anything. A shrink or a hide is a *downgrade* decision, made by the
252///    hysteresis, not by this gate;
253/// 2. **`low_seen`** — once the `_l0` decl has been in the roster for this
254///    (peer, source), its later absence is a real teardown (the resume-after-pause
255///    path, where the producer withdrew the decl while we were paused), not a join
256///    race. That case keeps its original instant fallback, guard tests and all;
257/// 3. **the grace itself** — a producer that will never publish a low layer (an old
258///    peer, a failed low encoder, a lost request) is indistinguishable from a slow
259///    one, so the wait is bounded and then **waived for good**: the tile takes the
260///    source and never pays the wait again on this establish. Without the latch a
261///    peer that never publishes low would re-pay it on every resubscribe, and the
262///    resume-after-pause budget would be a grace longer for no benefit.
263///
264/// ## Who still reaches the `demand_known: false` branch
265///
266/// In the shipped app, essentially nobody past the first tile that lays out:
267/// [`TileDemand`] seeds every roster peer's silence to `Hidden` once this receiver
268/// has reported anything at all, so a tile the virtualized grid never BUILDS is
269/// treated as not-rendered rather than as unknown. The branch is kept because it is
270/// the honest answer for a receiver that reports no demand at all — houndbot, the
271/// testkit harness, any future surface that renders video without a layout pass —
272/// which must still get video, and gets it one grace after joining.
273///
274/// ## Interaction with the receive-side budget
275///
276/// The waiver outranks the budget, and deliberately. `hound_media::allocate_video` can
277/// hand a tile the thumbnail rung and this gate can still put it on the SOURCE track a
278/// grace later, because the alternative — holding a capped subscription against a
279/// producer that never published `_l0` — is a permanently blank tile, and P2 already
280/// decided a blank tile is worse than an expensive one. The budget is not undermined by
281/// that: it bounds what this receiver ASKS the producers for, the waiver is a
282/// last-resort escape from asking for something nobody publishes, and the escape shows
283/// up honestly in the controller's own accounting — the apply pass prices the track
284/// actually subscribed to, so a waived tile is billed at the source rate it really
285/// costs and the budget backs off accordingly.
286///
287/// How long a tile whose demand is only INFERRED — this receiver reports, but has
288/// never reported *this* tile — stays unsubscribed before it escalates to the
289/// thumbnail rung. See [`TileDemand`]'s two-tier trust model: a reported hide is
290/// trusted forever; an inferred one is not, because a lost report and a tile the
291/// grid never built look identical from here.
292///
293/// 10 s is long enough that the UI's own recovery (a re-assert on connect or roster
294/// change, `demandNeedsReassert`) always wins the race in a healthy app, so this
295/// never fires for a tile that is genuinely off screen — and short enough that a
296/// user staring at a black tile because their report was swallowed gets video back
297/// before they reach for reload.
298const SEEDED_ESCAPE_GRACE: Duration = Duration::from_secs(10);

300/// Per-(peer, video source) subscription policy state that must **survive a
301/// re-establish**: the bounded source gate ([`SourceHold`]) and the seeded tile's
302/// escape clock. Owned by the session (see [`TilePolicies`]), not by the subscribe
303/// loop, because a reconnect must not re-pay a grace this tile already paid — a 1:1
304/// call whose peer publishes no low layer would otherwise go black for
305/// [`SOURCE_HOLD_GRACE`] on every single reconnect.
306#[derive(Debug, Default)]
307pub(super) struct TilePolicy {
308    hold: SourceHold,
309    /// When this tile's demand first became a SEED with nothing subscribed —
310    /// the [`SEEDED_ESCAPE_GRACE`] clock. Cleared the moment the tile reports.
311    seeded_since: Option<Instant>,
312}

319/// Lock the shared policy map, recovering from a poisoned mutex: the guarded state
320/// is plain policy bookkeeping, and a panic elsewhere must not take the subscribe
321/// loop — and with it every inbound stream — down with it.
322fn lock_policies(
323    policies: &TilePolicies,
324) -> std::sync::MutexGuard<'_, HashMap<(PeerId, Source), TilePolicy>> {
325    policies.lock().unwrap_or_else(|e| e.into_inner())
326}

328impl TilePolicy {
329    /// Fold one reconcile observation of a SEEDED demand (see [`DemandOrigin`]),
330    /// returning the demand to actually act on: the seed itself while the escape
331    /// grace runs, then the thumbnail rung — never the source, which is the
332    /// standing overshoot this whole policy exists to remove.
333    fn escalate_seed(
334        &mut self,
335        seed: hound_media::QualityRequest,
336        now: Instant,
337        grace: Duration,
338    ) -> hound_media::QualityRequest {
339        let since = *self.seeded_since.get_or_insert(now);
340        match seed {
341            hound_media::QualityRequest::Hidden(bps) if now.duration_since(since) >= grace => {
342                hound_media::QualityRequest::Capped(bps)
343            }
344            other => other,
345        }
346    }
347
348    /// This tile reported for itself: the inference — and its escape clock — no
349    /// longer apply.
350    fn reported(&mut self) {
351        self.seeded_since = None;
352    }
353
354    /// A subscription for this tile died on its own and is being reaped. It had a
355    /// pipeline, so the evidence [`SourceHold`] waits for has already been
356    /// collected: the restart must not re-pay the grace on top of the restart
357    /// ladder. `low` marks the death of the `_l0` track specifically, which is
358    /// standing proof the layer exists.
359    fn subscription_reaped(&mut self, low: bool) {
360        self.hold.waived = true;
361        self.hold.since = None;
362        self.hold.low_seen |= low;
363    }
364}

366#[derive(Debug, Default)]
367struct SourceHold {
368    /// Whether the `_l0` decl has EVER been present for this (peer, source).
369    low_seen: bool,
370    /// Whether the grace has already been spent (see escape 3 above).
371    waived: bool,
372    /// When the current wait started; `None` when not waiting.
373    since: Option<Instant>,
374}

376/// One reconcile pass's view of a (peer, video source) group, as [`SourceHold`]
377/// reads it. A struct rather than four bare `bool`s so no call site can transpose
378/// them.
379#[derive(Debug, Clone, Copy)]
380struct HoldInputs {
381    /// The UI has reported this tile's size at least once.
382    demand_known: bool,
383    /// The committed demand wants a below-source rung (`Capped`; `Hidden` does not —
384    /// it wants nothing at all).
385    wants_low: bool,
386    /// The `_l0` decl is in the roster right now.
387    low_present: bool,
388    /// We are already subscribed to this group's SOURCE track.
389    on_source: bool,
390    /// This tile's demand is INFERRED, not reported ([`DemandOrigin::Seeded`]).
391    /// Such a tile may never take the source track: the whole point of seeding is
392    /// that ~19 of 32 peers in a virtualized grid stop paying for a rung nobody
393    /// renders, and letting the gate's own escape hatch hand them the source
394    /// would put every one of them straight back on it. Its escape is
395    /// [`TilePolicy::escalate_seed`] — one thumbnail rung — instead.
396    seeded: bool,
397}
398
399impl SourceHold {
400    /// Fold one reconcile observation; returns whether the SOURCE track may be
401    /// subscribed for this tile right now.
402    fn observe(&mut self, i: HoldInputs, now: Instant, grace: Duration) -> bool {
403        if i.low_present {
404            self.low_seen = true;
405        }
406        if i.seeded {
407            // Not this gate's decision to make, and its timer must not run: a tile
408            // that later REPORTS gets a full, unspent grace.
409            self.since = None;
410            return false;
411        }
412        let waiting = !self.waived
413            && !i.on_source
414            && (!i.demand_known || (i.wants_low && !i.low_present && !self.low_seen));
415        if !waiting {
416            self.since = None;
417            return true;
418        }
419        let since = *self.since.get_or_insert(now);
420        if now.duration_since(since) >= grace {
421            self.waived = true;
422            self.since = None;
423            return true;
424        }
425        false
426    }
427
428    /// When the reconcile must wake to end an ongoing wait, so the fallback actuates
429    /// without depending on the floor tick. `None` when nothing is being waited on.
430    fn pending_deadline(&self, grace: Duration) -> Option<Instant> {
431        self.since.map(|since| since + grace)
432    }
433}

487/// One active remote subscription: the decode task + a stop flag.
488struct Subscription {
489    stop: Arc<AtomicBool>,
490    task: JoinHandle<()>,
491    /// Generation stamp: a reap signal carries the epoch of the pipeline that died,
492    /// so a LATE signal (a stopped pipeline's guard firing after the loop already
493    /// re-subscribed the same track id) can never remove the healthy replacement.
494    epoch: u64,
495    /// When this pipeline started — a death soon after start is paced (see
496    /// [`RestartPacer`]) so a track the relay instantly rejects can't hot-loop.
497    started: Instant,
498    /// Whether the pipeline delivered anything downstream (a decoded audio frame /
499    /// a keyframe past the video gate). A death without delivery climbs the
500    /// restart ladder — the subscription itself is suspect, not just the moment.
501    delivered: Arc<AtomicBool>,
502    /// This pipeline's downlink-driven demand (video only — `None` for audio). A
503    /// FRESH cell per pipeline start, so a resubscribe / layer switch / reap
504    /// naturally resets the demand to `Source` (see [`DownlinkCell`]'s docs).
505    downlink: Option<Arc<DownlinkCell>>,
506    /// Inbound-audio liveness (audio only — `None` for video, which carries its
507    /// own stall/first-keyframe deadlines inside the pull loop). See [`StallWatch`].
508    audio: Option<AudioLiveness>,
509    /// Which tile this subscription serves, and whether it is the low rung —
510    /// video only. The reaper needs it to credit the tile's [`TilePolicy`]: a
511    /// subscription that existed is standing evidence that this tile's rung
512    /// question is already answered, so its restart must not also wait out
513    /// [`SOURCE_HOLD_GRACE`].
514    tile: Option<((PeerId, Source), bool)>,
515    /// Which temporal view of the track this pipeline subscribed to. Part of the
516    /// subscription's IDENTITY, not decoration: a temporal switch keeps the same
517    /// control-plane track id (both views are the same declared track — the relay
518    /// derives one from the other), so without this the reconcile would see the id
519    /// already active and the switch would never actuate.
520    view: hound_media::TemporalView,
521}
522
523impl Subscription {
524    fn stop(self) {
525        self.stop.store(true, Ordering::Relaxed);
526        self.task.abort();
527    }
528}

530/// The render stream key a just-stopped subscription should RELEASE
531/// ([`VideoSink::stream_ended`]), or `None` if it must not be released.
532///
533/// Two things must not release, and they are why this is a function rather than an
534/// unconditional call at the stop site:
535///
536/// * **audio** (`tile: None`) — it feeds the mix bus, not the renderer; and
537/// * **the losing half of a simulcast LAYER SWITCH** — stopping the source track
538///   the instant the low track is subscribed (or the reverse) is one tile changing
539///   rung, not a stream ending. Both tracks feed the SAME `(peer, source)` stream
540///   key, so `successors` (the tiles that still have a video subscription after this
541///   pass) is exactly the discriminator: a switched tile is in it, a torn-down tile
542///   is not. Releasing mid-switch would drop the pending frame the make-before-break
543///   texture swap is waiting to draw into its new surface, which is the one moment
544///   the mailbox is load-bearing for the no-blank contract.
545///
546/// Everything else — the allocator hiding a tile, a paused call, a withdrawn track,
547/// a peer leaving — is a real end and releases.
548fn released_key(
549    tile: Option<((PeerId, Source), bool)>,
550    successors: &HashSet<(PeerId, Source)>,
551) -> Option<String> {
552    let ((peer, source), _low) = tile?;
553    (!successors.contains(&(peer, source))).then(|| video_key(peer, source))
554}

556/// Fires `(track_id, epoch)` on the reap channel when the owning pipeline task ends
557/// — by ANY path: clean end-of-track, a transport error, a panic unwind, or an
558/// abort (tokio drops the future at an await point, running destructors). This is
559/// what lets the subscribe loop REAP a self-terminated pipeline: without it, a
560/// dead track would stay in `active` forever, every reconcile would skip it as
561/// "already subscribed", and its tile would freeze permanently.
562struct PipelineEndGuard {
563    track_id: u32,
564    epoch: u64,
565    tx: tokio::sync::mpsc::UnboundedSender<(u32, u64)>,
566}
567
568impl Drop for PipelineEndGuard {
569    fn drop(&mut self) {
570        let _ = self.tx.send((self.track_id, self.epoch));
571    }
572}

574/// Spawn the subscribe loop: watch the `SessionView`, and for every remote track
575/// start a per-track pipeline (subscribe → decode → sink for video, mix bus for
576/// audio); stop pipelines for tracks that vanish. Sends `Subscribe`/keyframe
577/// intent on the control plane as tracks appear.
578#[allow(clippy::too_many_arguments)]
579pub(super) fn spawn_subscribe_loop(
580    mut view_rx: watch::Receiver<SessionView>,
581    self_id: PeerId,
582    space_id: Uuid,
583    relay: Arc<MoqRelaySession>,
584    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
585    sink: Arc<dyn VideoSink>,
586    playout_bus: Arc<PlayoutBus>,
587    levels: watch::Sender<LevelsView>,
588    clock: CaptureClock,
589    stats: Arc<CallStats>,
590    mut video_paused: watch::Receiver<bool>,
591    // The SENDER, not a receiver: this loop also PRUNES the map when a peer leaves
592    // (a departed peer's demand would otherwise outlive it and be applied to
593    // whoever inherits its peer id on a later establish). The map itself belongs to
594    // the session and survives re-establishes.
595    tile_quality: watch::Sender<TileDemand>,
596    // Per-(peer, source) policy state that outlives this establish: the bounded
597    // source gate and the seeded tile's escape clock. See [`TilePolicy`].
598    tile_policies: TilePolicies,
599    // This loop is the SOLE writer of the hidden-tile set: it is the only thing that
600    // knows which tiles the allocation left without a subscription, and the UI needs
601    // that to render an avatar instead of a black texture. See [`HiddenTiles`].
602    hidden_tiles: HiddenTiles,
603    volumes: Arc<StreamVolumes>,
604    // The dominant-speaker ranking this establish shares with the control reader
605    // (which feeds it every peer's reported level). Read once per reconcile pass.
606    speakers: SharedSpeakers,
607) -> JoinHandle<()> {
608    call_runtime().spawn(async move {
609        let mut tile_changed = tile_quality.subscribe();
610        // Keyed by TrackId.0 so we can diff against the view on each change.
611        let mut active: HashMap<u32, Subscription> = HashMap::new();
612        // The cap we last reported per (peer, source) — `(source track id, cap)` —
613        // so a RequestQuality is sent only when our demand for that peer's video
614        // actually changes, and so a PAUSE can address the withdrawal at the same
615        // track it was reported against.
616        // The last standing request actually put on the wire per (peer, source), so
617        // an unchanged demand re-sends nothing.
618        let mut last_sent: HashMap<(PeerId, Source), SentQuality> = HashMap::new();
619        // Per-(peer, source) hysteresis so a transient tile-resize flap (notably a
620        // rejoin relayout) can't thrash the source⇄low subscription on the shared
621        // audio+video connection. Holds a demanded layer switch until it is durable.
622        let mut hysteresis: HashMap<(PeerId, Source), LayerHysteresis> = HashMap::new();
623        // Self-healing: every pipeline task carries a PipelineEndGuard that reports
624        // its (track_id, epoch) here on ANY exit, so a self-terminated pipeline is
625        // reaped from `active` and the reconcile re-subscribes it (paced).
626        let (ended_tx, mut ended_rx) = tokio::sync::mpsc::unbounded_channel::<(u32, u64)>();
627        let mut next_epoch: u64 = 0;
628        let mut pacer = RestartPacer::default();
629        // This receiver's TOTAL video budget, walked by AIMD against the DEFICIENT
630        // verdict the per-track downlink cells produce. Per-establish state: a
631        // reconnect starts probing from the neutral 4 Mbps again rather than
632        // inheriting a backoff that belonged to a link that no longer exists.
633        let mut video_budget = hound_media::VideoBudget::new(hound_media::BudgetConfig::default());
634        // The receiver's DECODE budget, the CPU-side twin of the bandwidth one above:
635        // how many tiles may hold a decoder at all. It starts at this platform's
636        // decode-session ceiling — the only `#[cfg]`-resolved number in the whole
637        // receive path, and it lives behind the engine's capability seam beside
638        // `decode_caps()` rather than here (this crate carries no platform `cfg`) —
639        // and walks down against RT audio health. Per-establish for the same reason
640        // the bandwidth budget is: a reconnect starts from the platform ceiling again
641        // rather than inheriting a cut earned on a machine state that has moved on.
642        //
643        // Re-read every pass, not snapshotted: the engine's answer MOVES when a
644        // stream degrades from a hardware decode session onto a software one (the
645        // VideoToolbox/MFT session-create fallback), because a software session is
646        // CPU on the mix thread's cores rather than a slot on a fixed-function
647        // block. See `DecodeBudget::retarget_ceiling`.
648        let mut decode_ceiling = Engine::new().decode_session_budget();
649        // The budget controller is sans-IO and takes µs; this is the loop's own
650        // monotone origin for it (`Instant` has no epoch to subtract), and it doubles
651        // as the decode budget's warmup origin — the join storm this establish is
652        // about to cause starts here.
653        let loop_start = Instant::now();
654        let mut decode_budget = DecodeBudget::new(decode_ceiling, loop_start);
655        // What the budget last handed out, so a change is logged once instead of every
656        // pass (the same shape as `last_sent` for the per-tile demand).
657        let mut last_decode_k = decode_ceiling;
658        // The (peer, source) tiles the allocation currently has HIDDEN, republished to
659        // the UI on change so a tile with no subscription renders its owner's avatar
660        // instead of a black rectangle holding an idle texture. See [`HiddenTiles`].
661        // Keyed by the raw peer id, which is the key [`CallView`] carries.
662        let mut hidden_now: HashSet<(u32, Source)> = HashSet::new();
663        // Evidence about the allocation that is currently IN FORCE, produced by the
664        // apply pass below and consumed by the next pass's `observe`. It has to lag by
665        // a tick: the budget decides the allocation, so it cannot also be derived from
666        // it. At a 500 ms tick against a 1 s sustain window that costs nothing.
667        let mut in_force = hound_media::BudgetObservation::default();
668        // Remote peers we have inbound stats rows for, so a departed peer's row is
669        // dropped from the debug snapshot instead of freezing there until the next
670        // re-establish.
671        let mut known_peers: std::collections::HashSet<u32> = std::collections::HashSet::new();
672
673        loop {
674            // Soonest pending debounced layer switch — the loop wakes to actuate it
675            // even if no input changes in the meantime.
676            let mut next_switch_deadline: Option<Instant> = None;
677            // When video is paused (app backgrounded / call surface hidden) drop
678            // every inbound video sub — receiving a high-bitrate screenshare nobody
679            // is watching is wasted CPU + bandwidth. Audio always flows.
680            let drop_video = *video_paused.borrow();
681            let demand = tile_quality.borrow().clone();
682            let now = Instant::now();
683            // Rebuilt from scratch every pass (the allocation is), so a peer that
684            // left, un-paused or was un-hidden cannot leave a stale flag behind.
685            hidden_now.clear();
686
687            // Is this receiver decode-LIMITED? The evidence is the RT audio mix
688            // thread's tick lateness, not any video signal — see
689            // [`DECODE_TICK_LATE_LIMIT_US`] for why video yields to audio here and
690            // never the reverse. The raw tail crosses the seam, not a verdict: the
691            // threshold it is compared against is calibrated per platform inside the
692            // controller (see `DecodeBudget::observe`). While video is PAUSED nothing
693            // is decoding, so the mix thread's schedule is evidence about neither
694            // direction and both of the controller's clocks freeze.
695            let late = if drop_video {
696                TickLate::Unobserved
697            } else {
698                TickLate::Observed(stats.tick_late_p95_us())
699            };
700            // The platform half of the budget, re-read before the dynamic half folds
701            // this pass's evidence: a decoder that fell back to software since the
702            // last pass re-budgets this receiver against the CPU-shaped bound, and
703            // the cut has to reach `observe` on the same pass that learns about it.
704            let platform_ceiling = Engine::new().decode_session_budget();
705            if decode_budget.retarget_ceiling(platform_ceiling) {
706                let (sw_live, sw_total) = Engine::new().sw_decode_sessions();
707                tracing::info!(
708                    ceiling = platform_ceiling,
709                    from = decode_ceiling,
710                    sw_decode_sessions = sw_live,
711                    sw_decode_fallbacks_total = sw_total,
712                    "simulcast[recv]: platform decode ceiling moved (software decode sessions live)"
713                );
714                decode_ceiling = platform_ceiling;
715            }
716            let decode_k = decode_budget.observe(late, now);
717            if decode_k != last_decode_k {
718                tracing::info!(
719                    decode_k,
720                    from = last_decode_k,
721                    ceiling = decode_ceiling,
722                    tick_late_p95_us = stats.tick_late_p95_us(),
723                    "simulcast[recv]: decode budget moved"
724                );
725                last_decode_k = decode_k;
726            }
727            // A CPU-motivated cut must not fund an upgrade. Once the controller has
728            // walked BELOW the platform's static ceiling this receiver has evidenced
729            // that it is short of CPU, and the bitrate freed by hiding tiles would
730            // otherwise buy the survivors the source rung — trading 24 thumbnails for
731            // one 720p decode, which is not the saving that was asked for. So every
732            // gathered tile's demand is folded with `cpu_crisis_cap` below. Screen
733            // shares included, and clamped harder than cameras are — to the 360p15
734            // legibility floor rather than the top of the screen ladder, because
735            // `SCREEN_MID_RUNG` is a 720p30 decode and would let the clamp pass over
736            // the most expensive leg in the call. A 360p share is the lesser evil
737            // against a share whose rung is what is starving the mix thread, and the
738            // screen ladder's legibility ordering keeps that rung readable. At
739            // `k == ceiling` the cut is the platform's static session cap alone (32
740            // tiles on Apple's 16 says nothing about this box's CPU) and upgrades
741            // stay allowed.
742            let suppress_upgrades = decode_budget.cpu_constrained();
743
744            // Pausing drops every video subscription — withdraw the standing quality
745            // demand too. The producer keeps a receiver's request until that peer
746            // LEAVES the roster, so a paused-but-present receiver otherwise pins the
747            // shared low layer alive: a full extra downscale + encode of every frame
748            // for a viewer receiving nothing. `last_sent` is cleared but the
749            // hysteresis is NOT, so a resume re-reports the same committed demand at
750            // once and nothing churns on the way back.
751            if drop_video && !last_sent.is_empty() {
752                for (&(peer, source), &(sid, (cap, _hidden))) in last_sent.iter() {
753                    if cap.is_none() {
754                        continue; // nothing standing at the producer
755                    }
756                    tracing::info!(
757                        peer = peer.0,
758                        ?source,
759                        src_track = sid,
760                        "simulcast[recv]: video paused — withdrawing standing RequestQuality"
761                    );
762                    let _ = control_outbound.send(ClientMessage::RequestQuality {
763                        track: hound_control::TrackId(sid),
764                        cap_bps: None,
765                        hidden: false,
766                    });
767                }
768                last_sent.clear();
769            }
770
771            // Audio stall watchdog + departed-peer stats reap. Both need the roster,
772            // and both must run BEFORE the `desired` snapshot below borrows `active`.
773            {
774                let (audio_muted, present): (HashMap<u32, bool>, std::collections::HashSet<u32>) = {
775                    let view = view_rx.borrow();
776                    (
777                        view.tracks()
778                            .filter(|t| t.kind == MediaKind::Audio)
779                            .map(|t| (t.id.0, t.muted))
780                            .collect(),
781                        view.peers()
782                            .map(|p| p.id.0)
783                            .filter(|&id| id != self_id.0)
784                            .collect(),
785                    )
786                };
787                // A pipeline parked in the unbounded audio pull never drops its end
788                // guard, so the reap + restart ladder that exists for exactly this
789                // never runs. Kill it here; the guard then drives the normal path.
790                // A track absent from the roster reads as muted (don't accuse it —
791                // the `gone` sweep below removes it this same pass).
792                let wedged: Vec<u32> = active
793                    .iter_mut()
794                    .filter_map(|(id, sub)| {
795                        let live = sub.audio.as_mut()?;
796                        let count = live.arrivals.load(Ordering::Relaxed);
797                        let muted = audio_muted.get(id).copied().unwrap_or(true);
798                        live.watch
799                            .observe(count, now, muted, AUDIO_STALL)
800                            .then_some(*id)
801                    })
802                    .collect();
803                for id in wedged {
804                    if let Some(sub) = active.remove(&id) {
805                        let delivered = sub.delivered.load(Ordering::Relaxed);
806                        tracing::warn!(
807                            track = id,
808                            delivered,
809                            stall_s = AUDIO_STALL.as_secs(),
810                            "audio: subscription delivered nothing for the stall window \
811                             — killing the wedged pipeline for a fresh subscribe"
812                        );
813                        pacer.on_death(id, delivered, sub.started.elapsed(), now);
814                        sub.stop();
815                    }
816                }
817                // Inbound stats rows are created lazily per pipeline and otherwise
818                // only cleared on a full re-establish; drop a departed peer's row so
819                // the snapshot stops reporting its frozen gauges for the rest of the call.
820                let departed: Vec<u32> = known_peers.difference(&present).copied().collect();
821                for id in &departed {
822                    stats.remove_peer(*id);
823                    // Their streams are gone for good — let the renderer drop the
824                    // mailboxes (each pins the last decoded frame otherwise).
825                    sink.stream_ended(&cam_key(PeerId(*id)));
826                    sink.stream_ended(&screen_key(PeerId(*id)));
827                }
828                // …and their tile demand, which the session would otherwise carry for
829                // the rest of the call and hand to whoever inherits that peer id on a
830                // later establish. The UI never reports for self, so pruning to the
831                // remote roster cannot drop a live entry.
832                if !departed.is_empty() {
833                    tile_quality
834                        .send_if_modified(|d| d.retain_peers(|peer| present.contains(&peer.0)));
835                }
836                known_peers = present;
837            }
838            // Advance the dominant-speaker grid and snapshot it for this pass,
839            // pruning peers that have left in the same call (a departed peer must
840            // not keep a rung promotion for whoever inherits its id). Taken AFTER
841            // the roster reap above so `known_peers` is this pass's truth, and
842            // outside the borrow below so the lock is never held while reconciling.
843            let speaking = speakers.decide(|peer| known_peers.contains(&peer));
844            // Snapshot the remote tracks we should be subscribed to. Audio is taken
845            // as-is; for each (peer, video source) we pick EITHER the source or the
846            // low track by our demand, and report that demand upstream so the producer
847            // spins up the low layer.
848            let desired: Vec<DesiredTrack> = {
849                let view = view_rx.borrow();
850                let mut out: Vec<DesiredTrack> = Vec::new();
851                for t in view.tracks().filter(|t| {
852                    t.peer != self_id
853                        && t.kind == MediaKind::Audio
854                        && matches!(t.source, Source::Microphone | Source::Screen)
855                }) {
856                    // The broadcast path is built from the peer's AUTHENTICATED
857                    // user_id (control-plane roster), never the realtime-derived
858                    // identity map. An old backend that hasn't stamped it yet
859                    // sends the nil default — skip rather than subscribe under a
860                    // wrong (or nobody's) path.
861                    let Some(user_id) =
862                        view.peer(t.peer).map(|p| p.user_id).filter(|u| !u.is_nil())
863                    else {
864                        tracing::warn!(
865                            peer = t.peer.0,
866                            track = t.id.0,
867                            "subscribe: remote peer has no known user_id (old backend?) — skipping"
868                        );
869                        continue;
870                    };
871                    out.push((
872                        t.id.0,
873                        t.peer,
874                        t.kind,
875                        t.source,
876                        false,
877                        hound_media::TemporalView::Full,
878                        user_id,
879                    ));
880                }
881                if !drop_video {
882                    // Group each peer's video into (source track, low track).
883                    let mut groups: HashMap<(PeerId, Source), (Option<u32>, Option<u32>)> =
884                        HashMap::new();
885                    for t in view.tracks().filter(|t| {
886                        t.peer != self_id
887                            && t.kind == MediaKind::Video
888                            && matches!(t.source, Source::Camera | Source::Screen)
889                    }) {
890                        let e = groups.entry((t.peer, t.source)).or_insert((None, None));
891                        if t.low {
892                            e.1 = Some(t.id.0);
893                        } else {
894                            e.0 = Some(t.id.0);
895                        }
896                    }
897                    // How many remote peers publish video at all — the room-size
898                    // input to the ranking's DOWNGRADE half (a small call can
899                    // afford everyone; see `hound_media::SPEAKER_DEMOTE_MIN_PEERS`).
900                    let video_peers = groups
901                        .keys()
902                        .map(|&(peer, _)| peer)
903                        .collect::<std::collections::HashSet<_>>()
904                        .len();
905
906                    // ── GATHER ──────────────────────────────────────────────────
907                    // Every tile's inputs, collected before ANY of them is committed.
908                    // This pass exists because the rung decision is no longer per
909                    // tile: `hound_media::allocate_video` ranks the whole set against
910                    // one total budget, and deriving each tile's rung inside a single
911                    // loop — which is what this used to be — is exactly how N tiles
912                    // that each fit the link come to overshoot it N-fold.
913                    //
914                    // The gather ORDER is a `HashMap`'s and therefore arbitrary; that
915                    // is safe because the allocator's own priority key is total (it
916                    // ends in the peer id, and a peer's camera and screen are always
917                    // in different classes), so the input order cannot reach a
918                    // decision — pinned by `the_allocation_is_deterministic_and_
919                    // order_independent`.
920                    let mut gathered: Vec<Gathered> = Vec::with_capacity(groups.len());
921                    for (&(peer, source), &(src_id, low_id)) in &groups {
922                        let Some(user_id) =
923                            view.peer(peer).map(|p| p.user_id).filter(|u| !u.is_nil())
924                        else {
925                            tracing::warn!(
926                                peer = peer.0,
927                                ?source,
928                                "subscribe: remote peer has no known user_id (old backend?) — \
929                                 skipping video"
930                            );
931                            continue;
932                        };
933                        // The demand for this tile: its own report, or — once this
934                        // receiver is known to report at all — the roster seed for a
935                        // tile the UI never built, which is `Hidden` (see
936                        // [`TileDemand`]). `None` only for a receiver that has never
937                        // reported anything, and it is then fed to neither the
938                        // allocator (which would be ranking a demand we made up), the
939                        // hysteresis (which would debounce the tile's first real
940                        // report against it) nor the producer: inventing a
941                        // full-quality demand for a tile that has said nothing is half
942                        // of the join transient (see [`SourceHold`]).
943                        let origin = demand.get(peer, source);
944                        let tile_req = origin.map(|(req, origin)| {
945                            let mut policies = lock_policies(&tile_policies);
946                            let policy = policies.entry((peer, source)).or_default();
947                            match origin {
948                                // Trusted forever: a tile that says it is hidden
949                                // mounted and then disposed, and dispose is
950                                // reliable.
951                                DemandOrigin::Reported => {
952                                    policy.reported();
953                                    req
954                                }
955                                // Inferred, so bounded: it escalates to the
956                                // thumbnail rung (never the source) if the
957                                // inference outlives the grace.
958                                DemandOrigin::Seeded => {
959                                    policy.escalate_seed(req, now, SEEDED_ESCAPE_GRACE)
960                                }
961                            }
962                        });
963                        // Ranking modulates the tile's own demand, it does not
964                        // replace it: the dominant speaker's thumbnail buys the
965                        // next rung up, a peer observed silent in a crowded call
966                        // buys one less, and everyone else — including every peer
967                        // we have no speaking evidence about — keeps exactly what
968                        // its tile asked for. It moves a CAP, never a layer axis
969                        // (`speaker_adjusted`'s pinned property), so it can neither
970                        // subscribe, unsubscribe nor layer-switch a tile behind the
971                        // hysteresis's back. Applied BEFORE both the allocator and
972                        // the fold with the downlink demand, so a promotion can never
973                        // lift a receiver above its budget or its link estimate.
974                        let tile_req = tile_req.map(|req| {
975                            hound_media::speaker_adjusted(
976                                video_source(source),
977                                req,
978                                speaking.tier(peer.0),
979                                video_peers,
980                            )
981                        });
982                        let seeded = matches!(origin, Some((_, DemandOrigin::Seeded)));
983                        // This receiver's OWN downlink state for this video, read from
984                        // whichever track (source or low) is CURRENTLY subscribed —
985                        // the pull loop for that pipeline folds each arriving frame's
986                        // `frame_seq` into a `DownlinkDemand` and writes the result
987                        // here (see `DownlinkCell`). Not yet subscribed / audio-only /
988                        // no data yet all read as `Source` with no delivery evidence,
989                        // so a fresh pipeline neither spuriously caps nor drags the
990                        // budget down.
991                        let delivered = [src_id, low_id]
992                            .into_iter()
993                            .flatten()
994                            .find_map(|id| active.get(&id))
995                            .and_then(|s| s.downlink.as_ref())
996                            .map(|c| c.load());
997                        gathered.push(Gathered {
998                            peer,
999                            source,
1000                            user_id,
1001                            src_id,
1002                            low_id,
1003                            tile_req,
1004                            delivered,
1005                            seeded,
1006                        });
1007                    }
1008
1009                    // ── ALLOCATE ────────────────────────────────────────────────
1010                    // One decision for the whole receiver: rank the tiles, walk the
1011                    // total budget by AIMD against whether anything is actually
1012                    // arriving, and hand each tile back a rung no richer than what it
1013                    // asked for whose SUM fits. This is what closes the hole the
1014                    // per-track fold below cannot: a track with no pipeline yet has no
1015                    // delivery signal, so it reads as `Source` and the bandwidth half of
1016                    // the fold constrains nothing at all at a 32-peer join.
1017                    let tiles: Vec<hound_media::TileRequest> = gathered
1018                        .iter()
1019                        .filter_map(|g| {
1020                            let vsource = video_source(g.source);
1021                            g.tile_req.map(|req| hound_media::TileRequest {
1022                                peer: g.peer.0,
1023                                source: vsource,
1024                                // The CPU-crisis clamp, applied to the DEMAND rather
1025                                // than to the allocator: the allocator may only ever
1026                                // lower a demand, so capping the ladder here is
1027                                // exactly "no upgrade past what this receiver can
1028                                // decode" and it keeps `allocate_video` a pure
1029                                // function of its inputs.
1030                                demand: if suppress_upgrades {
1031                                    hound_media::most_constrained(
1032                                        req,
1033                                        hound_media::cpu_crisis_cap(vsource),
1034                                    )
1035                                } else {
1036                                    req
1037                                },
1038                                rank: speaking.rank(g.peer.0),
1039                                tier: speaking.tier(g.peer.0),
1040                                tile_pixels: demand.pixels(g.peer, g.source),
1041                            })
1042                        })
1043                        .collect();
1044                    let now_us = now
1045                        .duration_since(loop_start)
1046                        .as_micros()
1047                        .min(i64::MAX as u128) as i64;
1048                    // The link evidence (`deficient` / `committed_bps`) describes the
1049                    // allocation that is in force; the tile figures describe the roster
1050                    // as it is right now. See `hound_media::BudgetObservation`.
1051                    //
1052                    // Both tile figures are priced over the tiles the decode budget
1053                    // will actually SERVE, using the same `decode_k` the allocation
1054                    // below is given: a tile the decode cut hides subscribes to
1055                    // nothing, so its floor must not seed this controller and its
1056                    // demand must not be probed toward. See `hound_media::served_order`.
1057                    let budget = video_budget.observe(
1058                        hound_media::BudgetObservation {
1059                            ceiling_bps: hound_media::demand_ceiling_bps(&tiles, Some(decode_k)),
1060                            floor_bps: hound_media::floor_cost_bps(&tiles, Some(decode_k)),
1061                            ..in_force
1062                        },
1063                        now_us,
1064                    );
1065                    // Two independent caps, one call: `budget` is what this receiver
1066                    // can afford to RECEIVE, `decode_k` is what it can afford to
1067                    // DECODE. Always a number (never `None`) — see [`DecodeBudget`].
1068                    let mut allocation =
1069                        hound_media::allocate_video(&tiles, budget, Some(decode_k)).into_iter();
1070                    // Re-align to `gathered`: the allocator answers in input order,
1071                    // and its input was the gathered tiles that HAVE a demand.
1072                    let allocated: Vec<Option<hound_media::VideoAllocation>> = gathered
1073                        .iter()
1074                        .map(|g| {
1075                            if g.tile_req.is_some() {
1076                                allocation.next()
1077                            } else {
1078                                None
1079                            }
1080                        })
1081                        .collect();
1082
1083                    // ── APPLY ───────────────────────────────────────────────────
1084                    // What this pass ends up committed to, fed back to the controller
1085                    // on the NEXT pass (see `in_force`): the tracks whose delivery is
1086                    // measurably short of what they were just allocated, and the priced
1087                    // total of the subscriptions actually taken.
1088                    let mut deliveries: Vec<hound_media::TrackDelivery> = Vec::new();
1089                    let mut committed_bps: u64 = 0;
1090                    for (g, alloc) in gathered.iter().zip(allocated) {
1091                        let (peer, source, src_id, low_id) = (g.peer, g.source, g.src_id, g.low_id);
1092                        let vsource = video_source(source);
1093                        // Absent (no pipeline yet) reads as the optimistic `Source`, so
1094                        // a fresh pipeline never spuriously caps.
1095                        let bwe_req = g.delivered.unwrap_or(hound_media::QualityRequest::Source);
1096                        // This track's contribution to the DEFICIENT verdict — only for
1097                        // a track that HAS a pipeline, and only ever measured against
1098                        // what the allocator just granted it. A peer capped by its own
1099                        // bad uplink but delivering the thumbnail it was given says
1100                        // nothing about this receiver's downlink.
1101                        if let (Some(delivered), Some(allocated)) = (g.delivered, alloc) {
1102                            deliveries.push(hound_media::TrackDelivery {
1103                                source: vsource,
1104                                delivered,
1105                                allocated,
1106                            });
1107                        }
1108                        // The EFFECTIVE demand is the more constrained of the
1109                        // allocator's answer and this receiver's own per-track
1110                        // bandwidth demand. The allocator replaces nothing in
1111                        // `DownlinkDemand`: the budget decides what this receiver may
1112                        // ASK for across every tile, the per-track decider observes
1113                        // what one track actually GETS, and the more constrained wins.
1114                        // A budget cannot know a single track is wedged; a per-track
1115                        // observation cannot know 31 healthy tracks are 31 too many.
1116                        //
1117                        // Debounce the off⇄low⇄source decision so a transient
1118                        // tile-resize flap (a rejoin relayout, a fast scroll through a
1119                        // virtualized grid) can't thrash the subscription on the
1120                        // shared audio+video connection — only a DURABLE layer change
1121                        // actuates. Directional: the constants below gate DOWNGRADES;
1122                        // an upgrade (the user expanded a tile, or scrolled one back
1123                        // into view) commits after a short flap-absorbing window and
1124                        // ignores the dwell. A cap-value change on the same layer
1125                        // passes through — which is why the allocator emits RUNGS and
1126                        // not raw bitrates: a budget that drifts by a few kbps lands
1127                        // on the same rung and nothing downstream moves at all.
1128                        //
1129                        // `constrain` (not a bare `most_constrained`) because the
1130                        // allocation now carries a TEMPORAL view alongside its
1131                        // demand: a bandwidth demand that wins the fold moves this
1132                        // tile to a different track, and there is no base-layer
1133                        // view of that one to inherit.
1134                        let req = alloc.map(|allocated| {
1135                            let raw_req = allocated.constrain(bwe_req);
1136                            hysteresis
1137                                .entry((peer, source))
1138                                .or_insert_with(LayerHysteresis::new)
1139                                .observe(raw_req, now, LAYER_SWITCH_DEBOUNCE, LAYER_SWITCH_DWELL)
1140                        });
1141                        // The COMMITTED request, not the allocator's raw answer, is
1142                        // what the UI is told about: while the hysteresis is still
1143                        // holding a cut back the old subscription is live and frames
1144                        // are arriving, and flipping the tile to its avatar then would
1145                        // blank a picture that is still being painted. So the flag
1146                        // tracks exactly the state "this tile subscribes to nothing".
1147                        if req.is_some_and(|r| r.is_hidden()) {
1148                            hidden_now.insert((peer.0, source));
1149                        }
1150                        if let Some(deadline) = hysteresis.get(&(peer, source)).and_then(|h| {
1151                            h.pending_deadline(LAYER_SWITCH_DEBOUNCE, LAYER_SWITCH_DWELL)
1152                        }) {
1153                            next_switch_deadline = Some(
1154                                next_switch_deadline.map_or(deadline, |d: Instant| d.min(deadline)),
1155                            );
1156                        }
1157                        // Report our cap upstream (against the SOURCE track) on change,
1158                        // so the producer activates / tears down the shared low layer.
1159                        // A tile that has reported nothing yet reports nothing upstream
1160                        // either — an absent request leaves the producer's fold to the
1161                        // receivers that have spoken.
1162                        if let (Some(sid), Some(req)) = (src_id, req) {
1163                            // `hidden` rides alongside the cap rather than replacing
1164                            // it: the producer needs the cap to size a layer when
1165                            // EVERY requester is hidden, and needs the flag to keep
1166                            // this one out of the fold when any of them is not.
1167                            // The SPATIAL demand only. The temporal view never
1168                            // travels upstream: the producer publishes one
1169                            // full-rate track whichever view this receiver takes
1170                            // (the relay carves the base layer out of it), so
1171                            // reporting it could only invite the producer to
1172                            // change what it encodes — and a `Capped` demand is
1173                            // folded into the cross-receiver MINIMUM, which is
1174                            // the legibility trap `SCREEN_MID_RUNG` documents.
1175                            let (cap_bps, hidden) = match req.demand {
1176                                hound_media::QualityRequest::Capped(b) => (Some(b), false),
1177                                hound_media::QualityRequest::Hidden(b) => (Some(b), true),
1178                                hound_media::QualityRequest::Source => (None, false),
1179                            };
1180                            if last_sent.get(&(peer, source)).map(|&(_, c)| c)
1181                                != Some((cap_bps, hidden))
1182                            {
1183                                last_sent.insert((peer, source), (sid, (cap_bps, hidden)));
1184                                tracing::info!(
1185                                    peer = peer.0,
1186                                    ?source,
1187                                    ?cap_bps,
1188                                    hidden,
1189                                    src_track = sid,
1190                                    low_present = low_id.is_some(),
1191                                    budget_kbps = budget / 1000,
1192                                    "simulcast[recv]: RequestQuality → producer"
1193                                );
1194                                let _ = control_outbound.send(ClientMessage::RequestQuality {
1195                                    track: hound_control::TrackId(sid),
1196                                    cap_bps,
1197                                    hidden,
1198                                });
1199                            }
1200                        }
1201                        // The bounded escalation gate: while a tile is waiting for its
1202                        // own first report or for the producer's `_l0` decl, it
1203                        // subscribes to NOTHING rather than pulling the source it
1204                        // cannot use. Never blanks a live tile — see [`SourceHold`].
1205                        let (source_ok, hold_deadline) = {
1206                            let mut policies = lock_policies(&tile_policies);
1207                            let hold = &mut policies.entry((peer, source)).or_default().hold;
1208                            let ok = hold.observe(
1209                                HoldInputs {
1210                                    demand_known: req.is_some(),
1211                                    wants_low: matches!(
1212                                        req.map(|r| r.demand),
1213                                        Some(hound_media::QualityRequest::Capped(_))
1214                                    ),
1215                                    low_present: low_id.is_some(),
1216                                    on_source: src_id.is_some_and(|id| active.contains_key(&id)),
1217                                    seeded: g.seeded,
1218                                },
1219                                now,
1220                                SOURCE_HOLD_GRACE,
1221                            );
1222                            (ok, hold.pending_deadline(SOURCE_HOLD_GRACE))
1223                        };
1224                        if let Some(deadline) = hold_deadline {
1225                            next_switch_deadline = Some(
1226                                next_switch_deadline.map_or(deadline, |d: Instant| d.min(deadline)),
1227                            );
1228                        }
1229                        if let Some(id) = pick_layer(req, src_id, low_id, source_ok) {
1230                            let view = pick_view(req, id, src_id);
1231                            // Priced against the track actually taken, not against the
1232                            // request: the `SourceHold` waiver can put a `Capped` tile
1233                            // on the SOURCE when its producer never published `_l0`, and
1234                            // a committed-spend figure that missed that would be the one
1235                            // number the controller must not be able to lie about. The
1236                            // temporal view is part of "the track actually taken" — a
1237                            // `_t0` subscription genuinely costs ~70% of the source.
1238                            committed_bps += match req {
1239                                Some(r) if Some(id) == low_id => {
1240                                    hound_media::request_bps(vsource, r.demand)
1241                                }
1242                                _ if view == hound_media::TemporalView::BaseOnly => {
1243                                    hound_media::base_only_bps(hound_media::source_bps(vsource))
1244                                }
1245                                _ => hound_media::source_bps(vsource),
1246                            };
1247                            out.push((
1248                                id,
1249                                peer,
1250                                MediaKind::Video,
1251                                source,
1252                                Some(id) == low_id,
1253                                view,
1254                                g.user_id,
1255                            ));
1256                        }
1257                    }
1258                    in_force = hound_media::BudgetObservation {
1259                        deficient: hound_media::downlink_deficient(&deliveries),
1260                        committed_bps,
1261                        ..Default::default()
1262                    };
1263                } else {
1264                    // Video is paused: nothing is subscribed, so the evidence about the
1265                    // last allocation is not stale, it is GONE. Carrying a deficiency
1266                    // across a pause would cut the budget on the first pass after the
1267                    // resume for a link nobody was using (`VideoBudget::observe` clears
1268                    // its own clocks over the same gap).
1269                    in_force = hound_media::BudgetObservation::default();
1270                }
1271                out
1272            };
1273            // Tell the UI which tiles have no subscription, so each renders its
1274            // owner's avatar rather than a texture nobody is feeding. Every pass, not
1275            // only on change: `publish` compares before it notifies, and re-patching
1276            // repairs a `CallView` the control reader rebuilt from a snapshot taken
1277            // just before this pass's store. A PAUSED call publishes the empty set —
1278            // nothing is hidden by policy there, the whole surface is away.
1279            hidden_tiles.publish(&hidden_now);
1280
1281            // Stop pipelines for tracks no longer present (or video we just paused).
1282            let desired_ids: std::collections::HashSet<u32> =
1283                desired.iter().map(|(id, ..)| *id).collect();
1284            // …and the temporal view each one is wanted AT. A track whose id is still
1285            // desired but at the OTHER view is a resubscribe, not a survivor: both
1286            // views share one control-plane track id, so identity here is the pair.
1287            let desired_views: HashMap<u32, hound_media::TemporalView> = desired
1288                .iter()
1289                .map(|&(id, _, _, _, _, view, _)| (id, view))
1290                .collect();
1291            // The tiles that still have a video subscription after this pass — the
1292            // SUCCESSOR set that tells a layer switch apart from a teardown. A switch
1293            // trades one track id for another under the same `(peer, source)` tile, so
1294            // the tile is in here; a tile the allocator hid, a paused call, a withdrawn
1295            // track and a departed peer leave nothing behind. Video only: a screenshare's
1296            // AUDIO track carries the same `(peer, Source::Screen)` pair and must not
1297            // stand in for a video successor.
1298            let successors: HashSet<(PeerId, Source)> = desired
1299                .iter()
1300                .filter(|(_, _, kind, ..)| *kind == MediaKind::Video)
1301                .map(|&(_, peer, _, source, ..)| (peer, source))
1302                .collect();
1303            // A track is torn down when it is no longer wanted at all, OR when it is
1304            // wanted at a different temporal view (a `cam` ⇄ `cam_t0` resubscribe on
1305            // one unchanged track id). The tile is in `successors` either way, so the
1306            // make-before-break texture swap holds and nothing blanks.
1307            let gone: Vec<u32> = active
1308                .iter()
1309                .filter(|(id, sub)| desired_views.get(*id).is_none_or(|&want| want != sub.view))
1310                .map(|(id, _)| *id)
1311                .collect();
1312            for id in gone {
1313                let switching = desired_ids.contains(&id);
1314                if let Some(sub) = active.remove(&id) {
1315                    // Decided BEFORE the stop so the answer is this pass's reconcile,
1316                    // then acted on after it: `stop` is what guarantees no further frame
1317                    // is pushed into the mailbox we are about to release.
1318                    let released = released_key(sub.tile, &successors);
1319                    sub.stop();
1320                    if let Some(key) = released {
1321                        sink.stream_ended(&key);
1322                    }
1323                    // Tell the relay to stop forwarding this track to us (dropping the
1324                    // subscriber closes the moq sub; this also clears server routing).
1325                    //
1326                    // NOT on a temporal switch: the id is re-subscribed a few lines
1327                    // below, and an Unsubscribe→Subscribe pair on one id would tell
1328                    // the server this receiver left a track it never left.
1329                    if !switching {
1330                        let _ = control_outbound.send(ClientMessage::Unsubscribe {
1331                            track: hound_control::TrackId(id),
1332                        });
1333                    }
1334                }
1335            }
1336
1337            // Start pipelines for new tracks.
1338            for (track_id, peer, kind, source, low, view, user_id) in desired {
1339                if active.contains_key(&track_id) {
1340                    continue;
1341                }
1342                // A recently-died pipeline restarts PACED (bounded retry): defer this
1343                // track and wake when its pace elapses.
1344                if let Some(at) = pacer.ready_at(track_id, Instant::now()) {
1345                    next_switch_deadline =
1346                        Some(next_switch_deadline.map_or(at, |d: Instant| d.min(at)));
1347                    continue;
1348                }
1349                // Tell the server we want this track (+ a keyframe to bootstrap video).
1350                let _ = control_outbound.send(ClientMessage::Subscribe {
1351                    track: hound_control::TrackId(track_id),
1352                });
1353                if kind == MediaKind::Video {
1354                    tracing::info!(
1355                        peer = peer.0,
1356                        ?source,
1357                        low,
1358                        track = track_id,
1359                        ?view,
1360                        "simulcast[recv]: subscribing video track ({})",
1361                        match (low, view) {
1362                            (true, _) => "LOW",
1363                            (false, hound_media::TemporalView::BaseOnly) => "source_t0",
1364                            (false, hound_media::TemporalView::Full) => "source",
1365                        }
1366                    );
1367                    let _ = control_outbound.send(ClientMessage::RequestKeyframe {
1368                        track: hound_control::TrackId(track_id),
1369                    });
1370                    // RTC-debug: a PLI we sent for this peer's camera vs screen track.
1371                    stats
1372                        .peer(peer.0)
1373                        .video_keyframe_requested(inbound_video_kind(source));
1374                }
1375                let epoch = next_epoch;
1376                next_epoch += 1;
1377                let sub = start_pipeline(
1378                    peer,
1379                    user_id,
1380                    kind,
1381                    source,
1382                    low,
1383                    view,
1384                    space_id,
1385                    relay.clone(),
1386                    sink.clone(),
1387                    playout_bus.clone(),
1388                    levels.clone(),
1389                    clock,
1390                    stats.clone(),
1391                    control_outbound.clone(),
1392                    track_id,
1393                    volumes.clone(),
1394                    PipelineEndGuard {
1395                        track_id,
1396                        epoch,
1397                        tx: ended_tx.clone(),
1398                    },
1399                );
1400                active.insert(track_id, sub);
1401            }
1402
1403            // Re-reconcile when the roster changes, the pause flag flips, the tile
1404            // demand changes, a debounced layer switch comes due (so a held switch
1405            // actuates even if nothing else moves), OR on the floor tick — the
1406            // per-frame signals (downlink demand, audio liveness) have no event.
1407            let switch_wait = reconcile_wait(next_switch_deadline, Instant::now());
1408            tokio::select! {
1409                r = view_rx.changed() => {
1410                    if r.is_err() { break; }
1411                }
1412                r = video_paused.changed() => {
1413                    if r.is_err() { break; }
1414                }
1415                // No `Err` arm, unlike its sibling wakes: this loop holds a SENDER
1416                // for the same channel (it prunes departed peers' demand), so the
1417                // "all senders dropped" error is unreachable by construction. The
1418                // loop's exit paths are the roster/pause watches closing and the
1419                // supervisor's abort on teardown.
1420                _ = tile_changed.changed() => {}
1421                // A pipeline ended on its own (end-of-track / transport error): REAP
1422                // it so the reconcile re-subscribes — leaving it in `active` would
1423                // freeze that tile permanently (every reconcile would skip it as
1424                // "already subscribed"). Epoch-guarded: a stale signal from a
1425                // pipeline the loop itself stopped (and maybe already replaced)
1426                // must not touch the healthy replacement.
1427                Some((track_id, epoch)) = ended_rx.recv() => {
1428                    if active.get(&track_id).is_some_and(|s| s.epoch == epoch) {
1429                        let sub = active.remove(&track_id).expect("checked present");
1430                        let delivered = sub.delivered.load(Ordering::Relaxed);
1431                        // This tile HAD a subscription, so the evidence the source
1432                        // gate waits for is already in: credit it before the
1433                        // re-subscribe, or a self-healing restart would pay the
1434                        // restart ladder AND a fresh `SOURCE_HOLD_GRACE` of black.
1435                        if let Some((key, low)) = sub.tile {
1436                            lock_policies(&tile_policies)
1437                                .entry(key)
1438                                .or_default()
1439                                .subscription_reaped(low);
1440                        }
1441                        pacer.on_death(track_id, delivered, sub.started.elapsed(), Instant::now());
1442                        tracing::warn!(
1443                            track = track_id,
1444                            lived_ms = sub.started.elapsed().as_millis() as u64,
1445                            delivered,
1446                            "call: pipeline ended on its own — reaping + re-subscribing"
1447                        );
1448                    }
1449                }
1450                _ = tokio::time::sleep(switch_wait) => {}
1451            }
1452        }
1453
1454        // The establish is over: nothing will push to these tiles again on this
1455        // session, so release every mailbox rather than leaving one decoded frame per
1456        // subscribed tile pinned across a reconnect (or for the rest of the process
1457        // after a leave). Safe while the Dart tiles are still attached — a released
1458        // mailbox is a released PENDING frame, and each tile keeps displaying the last
1459        // frame already drawn into its own target buffer.
1460        for (_, sub) in active.drain() {
1461            let released = sub.tile.map(|((peer, source), _)| video_key(peer, source));
1462            sub.stop();
1463            if let Some(key) = released {
1464                sink.stream_ended(&key);
1465            }
1466        }
1467        // The set describes an allocation that no longer exists. Left standing, a
1468        // re-establish's fresh peer ids would inherit the dead session's flags until
1469        // its first reconcile pass — and a teardown that is not followed by one (a
1470        // leave) would leave every cut tile showing an avatar in the next call's
1471        // first frames.
1472        hidden_tiles.publish(&HashSet::new());
1473    })
1474}

1476/// How often the receiver re-sends `RequestKeyframe` while a fresh video
1477/// subscription is still awaiting its first keyframe, and how many times. The
1478/// subscribe-time request can RACE the producer's PLI-forced IDR (the IDR lands in
1479/// a group the moq live-join skips), leaving the receiver black indefinitely on a
1480/// keyframe that never comes. With the screen encoder
1481/// on an on-demand cadence (no frequent periodic IDR to paper over the race), this
1482/// bounded re-request is the fast recovery: producer-side the keyframe throttle
1483/// coalesces duplicates, so nagging is cheap. Once the attempts are exhausted the
1484/// pipeline's [`FIRST_KEYFRAME_DEADLINE`] ends it — a fresh, ladder-paced
1485/// resubscribe replaces a black tile that would otherwise wait on the safety-net IDR.
1486const KEYFRAME_NAG_INTERVAL: Duration = Duration::from_secs(2);
1487const KEYFRAME_NAG_ATTEMPTS: u32 = 5;
1488
1489/// How long a fresh pipeline waits for its MoQ broadcast to be ANNOUNCED before
1490/// giving up. `subscribe_track` awaits the announce with no deadline of its own,
1491/// and a task parked there never drops its [`PipelineEndGuard`] — so the reap +
1492/// restart machinery written for exactly this case never runs and the tile / peer
1493/// stays dead for the rest of the establish (a producer whose `publish_track`
1494/// failed, or a broadcast parked as a same-path backup behind a zombie). Bounded
1495/// in the same spirit as video's `FIRST_KEYFRAME_DEADLINE` (12 s), and slightly
1496/// under it: ending here yields a fresh announce cursor plus a re-sent
1497/// control-plane `Subscribe`, which is the only thing that heals a zombie.
1498const SUBSCRIBE_ANNOUNCE_DEADLINE: Duration = Duration::from_secs(10);

1500/// Start one per-track decode pipeline. Audio → Opus decode → mix bus; video →
1501/// H.264/etc decode → sink keyed `cam:{peer}` / `screen:{peer}`. For video,
1502/// `control_outbound` + `track_id` drive the awaiting-keyframe re-request nag.
1503#[allow(clippy::too_many_arguments)]
1504fn start_pipeline(
1505    peer: PeerId,
1506    user_id: Uuid,
1507    kind: MediaKind,
1508    source: Source,
1509    low: bool,
1510    view: hound_media::TemporalView,
1511    space_id: Uuid,
1512    relay: Arc<MoqRelaySession>,
1513    sink: Arc<dyn VideoSink>,
1514    playout_bus: Arc<PlayoutBus>,
1515    levels: watch::Sender<LevelsView>,
1516    clock: CaptureClock,
1517    stats: Arc<CallStats>,
1518    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
1519    track_id: u32,
1520    volumes: Arc<StreamVolumes>,
1521    end_guard: PipelineEndGuard,
1522) -> Subscription {
1523    let epoch = end_guard.epoch;
1524    let stop = Arc::new(AtomicBool::new(false));
1525    let task_stop = stop.clone();
1526    // First-delivery latch: both media set it on the first frame that DECODES.
1527    // Arrival is deliberately not enough on the video side — a keyframe in a codec
1528    // this platform cannot decode arrives perfectly and never renders, and counting
1529    // it as delivery disarms the first-keyframe deadline and the nag together. Read
1530    // at reap time to pick the restart pace (an undelivered death climbs the ladder).
1531    let delivered = Arc::new(AtomicBool::new(false));
1532    let task_delivered = delivered.clone();
1533    // This pipeline's downlink-driven demand (video only). A FRESH cell every
1534    // call — the whole reset-on-restart contract for `DownlinkDemand` is "a new
1535    // pipeline gets a new cell", so a resubscribe / layer switch / reap can never
1536    // carry a stale capped demand forward.
1537    let downlink = (kind == MediaKind::Video).then(|| Arc::new(DownlinkCell::new()));
1538    let task_downlink = downlink.clone();
1539    // Audio liveness counter (audio only): the pipeline bumps it per frame pulled
1540    // off the wire and the reconcile's [`StallWatch`] samples it. The unbounded
1541    // audio pull has no other out-of-band evidence that the track is alive.
1542    let arrivals = (kind == MediaKind::Audio).then(|| Arc::new(AtomicU64::new(0)));
1543    let task_arrivals = arrivals.clone();
1544    // The per-peer inbound stats handle (cloned once; the pipeline increments on it
1545    // lock-free per frame).
1546    let peer_stats = stats.peer(peer.0);

1548    // receiver subscribes to the `_l0` low track instead of the source. Temporal:
1549    // it may instead take the relay's derived `…_t0` track — same publication,
1550    // base-layer objects only (`services/sfu::derive`) — which nobody declares, so
1551    // it is named here from the committed view rather than looked up in the roster.
1552    let spatial_name = match (kind, source, low) {
1553        (MediaKind::Audio, Source::Screen, _) => SCREEN_AUDIO_TRACK,
1554        (MediaKind::Audio, _, _) => MIC_TRACK,
1555        (MediaKind::Video, Source::Screen, false) => SCREEN_TRACK,
1556        (MediaKind::Video, Source::Screen, true) => "screen_l0",
1557        (MediaKind::Video, _, false) => CAM_TRACK,
1558        (MediaKind::Video, _, true) => "cam_l0",
1559    };

1561    // whole track. A name nobody publishes would cost this tile the full
1562    // subscribe-announce deadline and then a paced restart, forever.
1563    let base_only = view == hound_media::TemporalView::BaseOnly
1564        && hound_media::base_view_offered(video_source(source), hound_media::SpatialLayer::Source)
1565        && !low
1566        && kind == MediaKind::Video;
1567    // Owned: the derived name is built here (nobody declares it) and the spawn
1568    // below is `'static`, so the name moves into the task rather than borrowing.
1569    let track_name = if base_only {
1570        hound_media::base_track_name(spatial_name)
1571    } else {
1572        spatial_name.to_owned()
1573    };

1575
1576    // Audio decode + fill runs on the RT-promoted audio runtime so video CPU can't
1577    // starve it; video stays on the general call runtime.
1578    let runtime = match kind {
1579        MediaKind::Audio => audio_runtime(),
1580        MediaKind::Video => call_runtime(),
1581    };
1582    let task = runtime.spawn(async move {
1583        // Owned by the task so its Drop reports this pipeline's end to the reap
1584        // channel on ANY exit path (clean end, error, panic, abort).
1585        let _end_guard = end_guard;
1586        // Bounded: the announce wait is the ONE await whose non-completion defeats
1587        // the end guard (see [`SUBSCRIBE_ANNOUNCE_DEADLINE`]). Nothing is half-built
1588        // yet, so dropping the future here is clean.
1589        let subscribing = relay.subscribe_track(
1590            &path,
1591            &track_name,
1592            peer.0 as u16,
1593            priority_for(kind),
1594            // Audio drains in arrival order (a reordered Opus frame is still playable);
1595            // video priority-drops stale GOPs. This is what stops a screenshare from
1596            // silently dropping reordered mic/app-audio frames on the shared connection.
1597            kind == MediaKind::Audio,
1598        );
1599        let subscriber = match tokio::time::timeout(SUBSCRIBE_ANNOUNCE_DEADLINE, subscribing).await
1600        {
1601            Ok(Ok(s)) => s,
1602            Ok(Err(e)) => {
1603                tracing::warn!(%e, peer = peer.0, "subscribe failed");
1604                return;
1605            }
1606            Err(_) => {
1607                tracing::warn!(
1608                    peer = peer.0,
1609                    track = track_id,
1610                    %path,
1611                    "subscribe: broadcast never announced within the deadline — ending \
1612                     pipeline for a paced re-subscribe"
1613                );
1614                return;
1615            }
1616        };
1617
1618        match kind {
1619            MediaKind::Audio => {
1620                // This stream's playback-gain cell (the tile volume/mute control),
1621                // looked up by key so a set volume survives resubscribes.
1622                let gain = volumes.cell(peer.0, source);
1623                audio_pipeline(
1624                    subscriber,
1625                    peer,
1626                    source,
1627                    playout_bus,
1628                    levels,
1629                    clock,
1630                    task_stop,
1631                    peer_stats,
1632                    gain,
1633                    task_delivered,
1634                    // `.expect`: this arm is MediaKind::Audio, where `arrivals` is
1635                    // always `Some` (see its construction above).
1636                    task_arrivals.expect("audio pipeline always has an arrival counter"),
1637                )
1638                .await
1639            }
1640            MediaKind::Video => {

1642                // fast re-requests don't race the pipeline's own pull. Ends on the
1643                // first DECODED frame or pipeline stop; after the bounded attempts
1644                // the pipeline's first-keyframe deadline takes over (it ends itself,
1645                // and the reconcile re-subscribes on the restart ladder).
1646                let nag = {
1647                    let got = task_delivered.clone();
1648                    let stop = task_stop.clone();
1649                    let outbound = control_outbound.clone();
1650                    call_runtime().spawn(async move {
1651                        for _ in 0..KEYFRAME_NAG_ATTEMPTS {
1652                            tokio::time::sleep(KEYFRAME_NAG_INTERVAL).await;
1653                            if got.load(Ordering::Relaxed) || stop.load(Ordering::Relaxed) {
1654                                return;
1655                            }
1656                            tracing::info!(
1657                                peer = peer.0,
1658                                track = track_id,
1659                                "video: still awaiting first keyframe — re-requesting"
1660                            );
1661                            let _ = outbound.send(ClientMessage::RequestKeyframe {
1662                                track: hound_control::TrackId(track_id),
1663                            });
1664                        }
1665                    })
1666                };
1667                video_pipeline(
1668                    subscriber,
1669                    peer,
1670                    source,
1671                    sink,
1672                    task_stop,
1673                    peer_stats,
1674                    task_delivered,
1675                    control_outbound,
1676                    track_id,
1677                    // `.expect`: this arm is MediaKind::Video, where `downlink` is
1678                    // always `Some` (see its construction above).
1679                    task_downlink.expect("video pipeline always has a downlink cell"),
1680                    source_top_bps(source),
1681                    // On a `_t0` track half the wire `frame_seq` space is missing
1682                    // BY CONSTRUCTION (the relay filtered the enhancement layer and
1683                    // never rewrites sequence numbers). Told nothing, the delivery
1684                    // estimator would read that as a permanent 50% loss and cap this
1685                    // receiver within one window. See `hound_media::downlink`.
1686                    if base_only {
1687                        hound_media::TEMPORAL_BASE_STRIDE
1688                    } else {
1689                        1
1690                    },
1691                )
1692                .await;
1693                nag.abort();
1694            }
1695        }
1696    });

1701        task,
1702        epoch,
1703        started,
1704        delivered,
1705        downlink,
1706        audio: arrivals.map(|arrivals| AudioLiveness {
1707            arrivals,
1708            watch: StallWatch::new(started),
1709        }),
1710        tile: (kind == MediaKind::Video).then_some(((peer, source), low)),
1711        view,
1712    }
