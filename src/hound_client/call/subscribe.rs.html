1//! The subscribe loop: reconcile remote tracks → decode → sink / mix bus. Diffs
2//! the `SessionView` roster into per-track pipelines ([`start_pipeline`]), picks
3//! source vs low layer per tile demand (debounced through the policy hysteresis),
4//! and self-heals: every pipeline carries a [`PipelineEndGuard`] whose Drop
5//! reports the death so the reconcile reaps + re-subscribes it (restart-paced).
6//! Invariant: audio subscribes with arrival-order drain and the smaller moq
7//! priority value — audio must strictly outrank video on the shared connection.
8
9use super::*;

11/// Lock-free single-slot handle for one video pipeline's downlink-driven demand:
12/// the video pull loop ([`video_pipeline`]) writes its latest
13/// [`hound_media::QualityRequest`] here as frames arrive (via
14/// [`hound_media::DownlinkDemand`]); the reconcile loop below reads it every tick
15/// and folds it with the tile demand via [`hound_media::most_constrained`]. A
16/// plain atomic (not a lock) so the per-frame write on the pull loop never
17/// blocks. `None` (audio, or a pipeline that hasn't decided yet) reads as
18/// [`hound_media::QualityRequest::Source`] — the optimistic default, matching
19/// [`hound_media::DownlinkDemand::new`].
20#[derive(Debug)]
21pub(super) struct DownlinkCell(AtomicU64);
22
23/// Sentinel encoding `QualityRequest::Source` in the packed atomic — a `Capped`
24/// bitrate this large never occurs (an encode rung tops out in the low millions).
25const SOURCE_SENTINEL: u64 = u64::MAX;
26
27impl DownlinkCell {
28    fn new() -> Self {
29        Self(AtomicU64::new(SOURCE_SENTINEL))
30    }
31
32    pub(super) fn store(&self, req: hound_media::QualityRequest) {
33        let v = match req {
34            hound_media::QualityRequest::Source => SOURCE_SENTINEL,
35            hound_media::QualityRequest::Capped(bps) => bps.min(SOURCE_SENTINEL - 1),
36        };
37        self.0.store(v, Ordering::Relaxed);
38    }
39
40    fn load(&self) -> hound_media::QualityRequest {
41        match self.0.load(Ordering::Relaxed) {
42            SOURCE_SENTINEL => hound_media::QualityRequest::Source,
43            bps => hound_media::QualityRequest::Capped(bps),
44        }
45    }
46}

48/// Floor wake-up for the reconcile loop. Every OTHER wake source is event-driven
49/// (roster change, pause flip, tile-demand change, pipeline death, a pending
50/// debounced switch) and in a steady call none of them fire for minutes — so the
51/// signals the pipelines fold per frame ([`DownlinkCell`], the audio arrival
52/// counter) would never be observed and a congestion downgrade / a wedged audio
53/// track could not actuate. [`LayerHysteresis`]'s debounce+dwell bounds any churn
54/// a periodic pass could cause, so polling is safe. 500 ms is the reconcile rate
55/// the dwell tests in [`super::policy`] already model.
56const RECONCILE_TICK: Duration = Duration::from_millis(500);
57
58/// How long an inbound AUDIO subscription may deliver NOTHING before the reconcile
59/// treats it as wedged. The audio pull is deliberately unbounded — the moq group
60/// read is not cancel-safe mid-object, so no timeout may ever be raced against
61/// `next_frame` — which means a relay that stops delivering a track WITHOUT
62/// closing it parks the pipeline task forever: its [`PipelineEndGuard`] never
63/// drops, the reconcile never reaps it, and that peer is silent for the rest of
64/// the establish. This is the out-of-band equivalent of video's
65/// `FIRST_KEYFRAME_DEADLINE`. 8 s is far above any jitter/retransmit tail on a
66/// 50 fps Opus stream (the encoder runs without DTX, so an unmuted sender emits a
67/// frame every 20 ms whether or not anyone is talking) while keeping a wedged
68/// peer's voice loss inside ~10 s. See [`StallWatch`] for why mute is exempt.
69const AUDIO_STALL: Duration = Duration::from_secs(8);
70
71/// Pure liveness policy for one inbound audio subscription (unit-tested). The
72/// pipeline bumps a monotone arrival counter as frames come off the wire; this
73/// samples it on the reconcile tick and reports a subscription that has not moved
74/// for `stall`.
75///
76/// A `muted` observation RE-ARMS the window instead of accumulating stall time: a
77/// muted mic gates before encode and publishes nothing at all, so silence there is
78/// healthy — accusing it would reap + re-subscribe every muted peer in the call
79/// every window, churning the connection audio shares (an "audio is sacred"
80/// violation of exactly the kind this watchdog exists to prevent).
81#[derive(Debug)]
82struct StallWatch {
83    last_count: u64,
84    /// When the counter last moved (or the window was last re-armed).
85    since: Instant,
86}
87
88impl StallWatch {
89    fn new(now: Instant) -> Self {
90        Self {
91            last_count: 0,
92            since: now,
93        }
94    }
95
96    /// Fold one sample of the arrival counter. Returns true once the subscription
97    /// has delivered nothing for `stall` while the roster reports it unmuted.
98    fn observe(&mut self, count: u64, now: Instant, muted: bool, stall: Duration) -> bool {
99        if muted || count != self.last_count {
100            self.last_count = count;
101            self.since = now;
102            return false;
103        }
104        now.duration_since(self.since) >= stall
105    }
106}

108/// The audio-liveness half of a [`Subscription`]: the counter the pipeline bumps
109/// per arriving frame plus the reconcile-side watchdog state that samples it.
110struct AudioLiveness {
111    arrivals: Arc<AtomicU64>,
112    watch: StallWatch,
113}
114
115/// The reconcile select's wait: the soonest pending debounced switch, FLOORED by
116/// [`RECONCILE_TICK`] so the loop always wakes to poll the per-frame signals even
117/// when no event is pending. Pure (takes `now`) so the floor is unit-tested.
118fn reconcile_wait(next_deadline: Option<Instant>, now: Instant) -> Duration {
119    next_deadline
120        .map(|d| d.saturating_duration_since(now).min(RECONCILE_TICK))
121        .unwrap_or(RECONCILE_TICK)
122}

124/// One active remote subscription: the decode task + a stop flag.
125struct Subscription {
126    stop: Arc<AtomicBool>,
127    task: JoinHandle<()>,
128    /// Generation stamp: a reap signal carries the epoch of the pipeline that died,
129    /// so a LATE signal (a stopped pipeline's guard firing after the loop already
130    /// re-subscribed the same track id) can never remove the healthy replacement.
131    epoch: u64,
132    /// When this pipeline started — a death soon after start is paced (see
133    /// [`RestartPacer`]) so a track the relay instantly rejects can't hot-loop.
134    started: Instant,
135    /// Whether the pipeline delivered anything downstream (a decoded audio frame /
136    /// a keyframe past the video gate). A death without delivery climbs the
137    /// restart ladder — the subscription itself is suspect, not just the moment.
138    delivered: Arc<AtomicBool>,
139    /// This pipeline's downlink-driven demand (video only — `None` for audio). A
140    /// FRESH cell per pipeline start, so a resubscribe / layer switch / reap
141    /// naturally resets the demand to `Source` (see [`DownlinkCell`]'s docs).
142    downlink: Option<Arc<DownlinkCell>>,
143    /// Inbound-audio liveness (audio only — `None` for video, which carries its
144    /// own stall/first-keyframe deadlines inside the pull loop). See [`StallWatch`].
145    audio: Option<AudioLiveness>,
146}
147
148impl Subscription {
149    fn stop(self) {
150        self.stop.store(true, Ordering::Relaxed);
151        self.task.abort();
152    }
153}
154
155/// Fires `(track_id, epoch)` on the reap channel when the owning pipeline task ends
156/// — by ANY path: clean end-of-track, a transport error, a panic unwind, or an
157/// abort (tokio drops the future at an await point, running destructors). This is
158/// what lets the subscribe loop REAP a self-terminated pipeline: without it, a
159/// dead track would stay in `active` forever, every reconcile would skip it as
160/// "already subscribed", and its tile would freeze permanently.
161struct PipelineEndGuard {
162    track_id: u32,
163    epoch: u64,
164    tx: tokio::sync::mpsc::UnboundedSender<(u32, u64)>,
165}
166
167impl Drop for PipelineEndGuard {
168    fn drop(&mut self) {
169        let _ = self.tx.send((self.track_id, self.epoch));
170    }
171}

173/// Spawn the subscribe loop: watch the `SessionView`, and for every remote track
174/// start a per-track pipeline (subscribe → decode → sink for video, mix bus for
175/// audio); stop pipelines for tracks that vanish. Sends `Subscribe`/keyframe
176/// intent on the control plane as tracks appear.
177#[allow(clippy::too_many_arguments)]
178pub(super) fn spawn_subscribe_loop(
179    mut view_rx: watch::Receiver<SessionView>,
180    self_id: PeerId,
181    space_id: Uuid,
182    relay: Arc<MoqRelaySession>,
183    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
184    sink: Arc<dyn VideoSink>,
185    playout_bus: Arc<PlayoutBus>,
186    levels: watch::Sender<LevelsView>,
187    clock: CaptureClock,
188    stats: Arc<CallStats>,
189    mut video_paused: watch::Receiver<bool>,
190    mut tile_quality: watch::Receiver<HashMap<(PeerId, Source), hound_media::QualityRequest>>,
191    volumes: Arc<StreamVolumes>,
192) -> JoinHandle<()> {
193    call_runtime().spawn(async move {
194        // Keyed by TrackId.0 so we can diff against the view on each change.
195        let mut active: HashMap<u32, Subscription> = HashMap::new();
196        // The cap we last reported per (peer, source) — `(source track id, cap)` —
197        // so a RequestQuality is sent only when our demand for that peer's video
198        // actually changes, and so a PAUSE can address the withdrawal at the same
199        // track it was reported against.
200        let mut last_sent: HashMap<(PeerId, Source), (u32, Option<u64>)> = HashMap::new();
201        // Per-(peer, source) hysteresis so a transient tile-resize flap (notably a
202        // rejoin relayout) can't thrash the source⇄low subscription on the shared
203        // audio+video connection. Holds a demanded layer switch until it is durable.
204        let mut hysteresis: HashMap<(PeerId, Source), LayerHysteresis> = HashMap::new();
205        // Self-healing: every pipeline task carries a PipelineEndGuard that reports
206        // its (track_id, epoch) here on ANY exit, so a self-terminated pipeline is
207        // reaped from `active` and the reconcile re-subscribes it (paced).
208        let (ended_tx, mut ended_rx) = tokio::sync::mpsc::unbounded_channel::<(u32, u64)>();
209        let mut next_epoch: u64 = 0;
210        let mut pacer = RestartPacer::default();
211        // Remote peers we have inbound stats rows for, so a departed peer's row is
212        // dropped from the debug snapshot instead of freezing there until the next
213        // re-establish.
214        let mut known_peers: std::collections::HashSet<u32> = std::collections::HashSet::new();
215
216        loop {
217            // Soonest pending debounced layer switch — the loop wakes to actuate it
218            // even if no input changes in the meantime.
219            let mut next_switch_deadline: Option<Instant> = None;
220            // When video is paused (app backgrounded / call surface hidden) drop
221            // every inbound video sub — receiving a high-bitrate screenshare nobody
222            // is watching is wasted CPU + bandwidth. Audio always flows.
223            let drop_video = *video_paused.borrow();
224            let demand = tile_quality.borrow().clone();
225            let now = Instant::now();
226
227            // Pausing drops every video subscription — withdraw the standing quality
228            // demand too. The producer keeps a receiver's request until that peer
229            // LEAVES the roster, so a paused-but-present receiver otherwise pins the
230            // shared low layer alive: a full extra downscale + encode of every frame
231            // for a viewer receiving nothing. `last_sent` is cleared but the
232            // hysteresis is NOT, so a resume re-reports the same committed demand at
233            // once and nothing churns on the way back.
234            if drop_video && !last_sent.is_empty() {
235                for (&(peer, source), &(sid, cap)) in last_sent.iter() {
236                    if cap.is_none() {
237                        continue; // nothing standing at the producer
238                    }
239                    tracing::info!(
240                        peer = peer.0,
241                        ?source,
242                        src_track = sid,
243                        "simulcast[recv]: video paused — withdrawing standing RequestQuality"
244                    );
245                    let _ = control_outbound.send(ClientMessage::RequestQuality {
246                        track: hound_control::TrackId(sid),
247                        cap_bps: None,
248                    });
249                }
250                last_sent.clear();
251            }
252
253            // Audio stall watchdog + departed-peer stats reap. Both need the roster,
254            // and both must run BEFORE the `desired` snapshot below borrows `active`.
255            {
256                let (audio_muted, present): (HashMap<u32, bool>, std::collections::HashSet<u32>) = {
257                    let view = view_rx.borrow();
258                    (
259                        view.tracks()
260                            .filter(|t| t.kind == MediaKind::Audio)
261                            .map(|t| (t.id.0, t.muted))
262                            .collect(),
263                        view.peers()
264                            .map(|p| p.id.0)
265                            .filter(|&id| id != self_id.0)
266                            .collect(),
267                    )
268                };
269                // A pipeline parked in the unbounded audio pull never drops its end
270                // guard, so the reap + restart ladder that exists for exactly this
271                // never runs. Kill it here; the guard then drives the normal path.
272                // A track absent from the roster reads as muted (don't accuse it —
273                // the `gone` sweep below removes it this same pass).
274                let wedged: Vec<u32> = active
275                    .iter_mut()
276                    .filter_map(|(id, sub)| {
277                        let live = sub.audio.as_mut()?;
278                        let count = live.arrivals.load(Ordering::Relaxed);
279                        let muted = audio_muted.get(id).copied().unwrap_or(true);
280                        live.watch
281                            .observe(count, now, muted, AUDIO_STALL)
282                            .then_some(*id)
283                    })
284                    .collect();
285                for id in wedged {
286                    if let Some(sub) = active.remove(&id) {
287                        let delivered = sub.delivered.load(Ordering::Relaxed);
288                        tracing::warn!(
289                            track = id,
290                            delivered,
291                            stall_s = AUDIO_STALL.as_secs(),
292                            "audio: subscription delivered nothing for the stall window \
293                             — killing the wedged pipeline for a fresh subscribe"
294                        );
295                        pacer.on_death(id, delivered, sub.started.elapsed(), now);
296                        sub.stop();
297                    }
298                }
299                // Inbound stats rows are created lazily per pipeline and otherwise
300                // only cleared on a full re-establish; drop a departed peer's row so
301                // the snapshot stops reporting its frozen gauges for the rest of the call.
302                let departed: Vec<u32> = known_peers.difference(&present).copied().collect();
303                for id in departed {
304                    stats.remove_peer(id);
305                    // Their streams are gone for good — let the renderer drop the
306                    // mailboxes (each pins the last decoded frame otherwise).
307                    sink.stream_ended(&cam_key(PeerId(id)));
308                    sink.stream_ended(&screen_key(PeerId(id)));
309                }
310                known_peers = present;
311            }
312            // Snapshot the remote tracks we should be subscribed to. Audio is taken
313            // as-is; for each (peer, video source) we pick EITHER the source or the
314            // low track by our demand, and report that demand upstream so the producer
315            // spins up the low layer.
316            let desired: Vec<(u32, PeerId, MediaKind, Source, bool, Uuid)> = {
317                let view = view_rx.borrow();
318                let mut out: Vec<(u32, PeerId, MediaKind, Source, bool, Uuid)> = Vec::new();
319                for t in view.tracks().filter(|t| {
320                    t.peer != self_id
321                        && t.kind == MediaKind::Audio
322                        && matches!(t.source, Source::Microphone | Source::Screen)
323                }) {
324                    // The broadcast path is built from the peer's AUTHENTICATED
325                    // user_id (control-plane roster), never the realtime-derived
326                    // identity map. An old backend that hasn't stamped it yet
327                    // sends the nil default — skip rather than subscribe under a
328                    // wrong (or nobody's) path.
329                    let Some(user_id) =
330                        view.peer(t.peer).map(|p| p.user_id).filter(|u| !u.is_nil())
331                    else {
332                        tracing::warn!(
333                            peer = t.peer.0,
334                            track = t.id.0,
335                            "subscribe: remote peer has no known user_id (old backend?) — skipping"
336                        );
337                        continue;
338                    };
339                    out.push((t.id.0, t.peer, t.kind, t.source, false, user_id));
340                }
341                if !drop_video {
342                    // Group each peer's video into (source track, low track).
343                    let mut groups: HashMap<(PeerId, Source), (Option<u32>, Option<u32>)> =
344                        HashMap::new();
345                    for t in view.tracks().filter(|t| {
346                        t.peer != self_id
347                            && t.kind == MediaKind::Video
348                            && matches!(t.source, Source::Camera | Source::Screen)
349                    }) {
350                        let e = groups.entry((t.peer, t.source)).or_insert((None, None));
351                        if t.low {
352                            e.1 = Some(t.id.0);
353                        } else {
354                            e.0 = Some(t.id.0);
355                        }
356                    }
357                    for (&(peer, source), &(src_id, low_id)) in &groups {
358                        let Some(user_id) =
359                            view.peer(peer).map(|p| p.user_id).filter(|u| !u.is_nil())
360                        else {
361                            tracing::warn!(
362                                peer = peer.0,
363                                ?source,
364                                "subscribe: remote peer has no known user_id (old backend?) — \
365                                 skipping video"
366                            );
367                            continue;
368                        };
369                        let tile_req = demand
370                            .get(&(peer, source))
371                            .copied()
372                            .unwrap_or(hound_media::QualityRequest::Source);
373                        // This receiver's OWN downlink-driven demand for this video,
374                        // read from whichever track (source or low) is CURRENTLY
375                        // subscribed — the pull loop for that pipeline folds each
376                        // arriving frame's `frame_seq` into a `DownlinkDemand` and
377                        // writes the result here (see `DownlinkCell`). Not yet
378                        // subscribed / audio-only / no data yet all read as `Source`
379                        // (optimistic default), so a fresh pipeline never spuriously
380                        // caps.
381                        let bwe_req = [src_id, low_id]
382                            .into_iter()
383                            .flatten()
384                            .find_map(|id| active.get(&id))
385                            .and_then(|s| s.downlink.as_ref())
386                            .map(|c| c.load())
387                            .unwrap_or(hound_media::QualityRequest::Source);
388                        // The EFFECTIVE demand is the more constrained of the tile-size
389                        // demand and this receiver's own bandwidth demand — neither
390                        // signal alone should ever request MORE than the other allows.
391                        let raw_req = hound_media::most_constrained(tile_req, bwe_req);
392                        // Debounce the source⇄low decision so a transient tile-resize
393                        // flap (a rejoin relayout) can't thrash the subscription on the
394                        // shared audio+video connection — only a DURABLE layer change
395                        // actuates. A cap-value change on the same layer passes through.
396                        let req = hysteresis
397                            .entry((peer, source))
398                            .or_insert_with(LayerHysteresis::new)
399                            .observe(raw_req, now, LAYER_SWITCH_DEBOUNCE, LAYER_SWITCH_DWELL);
400                        if let Some(deadline) = hysteresis.get(&(peer, source)).and_then(|h| {
401                            h.pending_deadline(LAYER_SWITCH_DEBOUNCE, LAYER_SWITCH_DWELL)
402                        }) {
403                            next_switch_deadline = Some(
404                                next_switch_deadline.map_or(deadline, |d: Instant| d.min(deadline)),
405                            );
406                        }
407                        // Report our cap upstream (against the SOURCE track) on change,
408                        // so the producer activates / tears down the shared low layer.
409                        if let Some(sid) = src_id {
410                            let cap_bps = match req {
411                                hound_media::QualityRequest::Capped(b) => Some(b),
412                                hound_media::QualityRequest::Source => None,
413                            };
414                            if last_sent.get(&(peer, source)).map(|&(_, c)| c) != Some(cap_bps) {
415                                last_sent.insert((peer, source), (sid, cap_bps));
416                                tracing::info!(
417                                    peer = peer.0,
418                                    ?source,
419                                    ?cap_bps,
420                                    src_track = sid,
421                                    low_present = low_id.is_some(),
422                                    "simulcast[recv]: RequestQuality → producer"
423                                );
424                                let _ = control_outbound.send(ClientMessage::RequestQuality {
425                                    track: hound_control::TrackId(sid),
426                                    cap_bps,
427                                });
428                            }
429                        }
430                        // Subscribe the low track when we want it AND it exists (so a
431                        // demand for low before the producer publishes it falls back to
432                        // source until the low track appears, then switches).
433                        let pick = match req {
434                            hound_media::QualityRequest::Capped(_) => low_id.or(src_id),
435                            hound_media::QualityRequest::Source => src_id.or(low_id),
436                        };
437                        if let Some(id) = pick {
438                            out.push((
439                                id,
440                                peer,
441                                MediaKind::Video,
442                                source,
443                                Some(id) == low_id,
444                                user_id,
445                            ));
446                        }
447                    }
448                }
449                out
450            };
451
452            // Stop pipelines for tracks no longer present (or video we just paused).
453            let desired_ids: std::collections::HashSet<u32> =
454                desired.iter().map(|(id, ..)| *id).collect();
455            let gone: Vec<u32> = active
456                .keys()
457                .copied()
458                .filter(|id| !desired_ids.contains(id))
459                .collect();
460            for id in gone {
461                if let Some(sub) = active.remove(&id) {
462                    sub.stop();
463                    // Tell the relay to stop forwarding this track to us (dropping the
464                    // subscriber closes the moq sub; this also clears server routing).
465                    let _ = control_outbound.send(ClientMessage::Unsubscribe {
466                        track: hound_control::TrackId(id),
467                    });
468                }
469            }
470
471            // Start pipelines for new tracks.
472            for (track_id, peer, kind, source, low, user_id) in desired {
473                if active.contains_key(&track_id) {
474                    continue;
475                }
476                // A recently-died pipeline restarts PACED (bounded retry): defer this
477                // track and wake when its pace elapses.
478                if let Some(at) = pacer.ready_at(track_id, Instant::now()) {
479                    next_switch_deadline =
480                        Some(next_switch_deadline.map_or(at, |d: Instant| d.min(at)));
481                    continue;
482                }
483                // Tell the server we want this track (+ a keyframe to bootstrap video).
484                let _ = control_outbound.send(ClientMessage::Subscribe {
485                    track: hound_control::TrackId(track_id),
486                });
487                if kind == MediaKind::Video {
488                    tracing::info!(
489                        peer = peer.0,
490                        ?source,
491                        low,
492                        track = track_id,
493                        "simulcast[recv]: subscribing video track ({})",
494                        if low { "LOW" } else { "source" }
495                    );
496                    let _ = control_outbound.send(ClientMessage::RequestKeyframe {
497                        track: hound_control::TrackId(track_id),
498                    });
499                    // RTC-debug: a PLI we sent for this peer's camera vs screen track.
500                    stats
501                        .peer(peer.0)
502                        .video_keyframe_requested(inbound_video_kind(source));
503                }
504                let epoch = next_epoch;
505                next_epoch += 1;
506                let sub = start_pipeline(
507                    peer,
508                    user_id,
509                    kind,
510                    source,
511                    low,
512                    space_id,
513                    relay.clone(),
514                    sink.clone(),
515                    playout_bus.clone(),
516                    levels.clone(),
517                    clock,
518                    stats.clone(),
519                    control_outbound.clone(),
520                    track_id,
521                    volumes.clone(),
522                    PipelineEndGuard {
523                        track_id,
524                        epoch,
525                        tx: ended_tx.clone(),
526                    },
527                );
528                active.insert(track_id, sub);
529            }
530
531            // Re-reconcile when the roster changes, the pause flag flips, the tile
532            // demand changes, a debounced layer switch comes due (so a held switch
533            // actuates even if nothing else moves), OR on the floor tick — the
534            // per-frame signals (downlink demand, audio liveness) have no event.
535            let switch_wait = reconcile_wait(next_switch_deadline, Instant::now());
536            tokio::select! {
537                r = view_rx.changed() => {
538                    if r.is_err() { break; }
539                }
540                r = video_paused.changed() => {
541                    if r.is_err() { break; }
542                }
543                r = tile_quality.changed() => {
544                    if r.is_err() { break; }
545                }
546                // A pipeline ended on its own (end-of-track / transport error): REAP
547                // it so the reconcile re-subscribes — leaving it in `active` would
548                // freeze that tile permanently (every reconcile would skip it as
549                // "already subscribed"). Epoch-guarded: a stale signal from a
550                // pipeline the loop itself stopped (and maybe already replaced)
551                // must not touch the healthy replacement.
552                Some((track_id, epoch)) = ended_rx.recv() => {
553                    if active.get(&track_id).is_some_and(|s| s.epoch == epoch) {
554                        let sub = active.remove(&track_id).expect("checked present");
555                        let delivered = sub.delivered.load(Ordering::Relaxed);
556                        pacer.on_death(track_id, delivered, sub.started.elapsed(), Instant::now());
557                        tracing::warn!(
558                            track = track_id,
559                            lived_ms = sub.started.elapsed().as_millis() as u64,
560                            delivered,
561                            "call: pipeline ended on its own — reaping + re-subscribing"
562                        );
563                    }
564                }
565                _ = tokio::time::sleep(switch_wait) => {}
566            }
567        }
568
569        for (_, sub) in active.drain() {
570            sub.stop();
571        }
572    })
573}

575/// How often the receiver re-sends `RequestKeyframe` while a fresh video
576/// subscription is still awaiting its first keyframe, and how many times. The
577/// subscribe-time request can RACE the producer's PLI-forced IDR (the IDR lands in
578/// a group the moq live-join skips), leaving the receiver black indefinitely on a
579/// keyframe that never comes. With the screen encoder
580/// on an on-demand cadence (no frequent periodic IDR to paper over the race), this
581/// bounded re-request is the fast recovery: producer-side the keyframe throttle
582/// coalesces duplicates, so nagging is cheap. Once the attempts are exhausted the
583/// pipeline's [`FIRST_KEYFRAME_DEADLINE`] ends it — a fresh, ladder-paced
584/// resubscribe replaces a black tile that would otherwise wait on the safety-net IDR.
585const KEYFRAME_NAG_INTERVAL: Duration = Duration::from_secs(2);
586const KEYFRAME_NAG_ATTEMPTS: u32 = 5;
587
588/// How long a fresh pipeline waits for its MoQ broadcast to be ANNOUNCED before
589/// giving up. `subscribe_track` awaits the announce with no deadline of its own,
590/// and a task parked there never drops its [`PipelineEndGuard`] — so the reap +
591/// restart machinery written for exactly this case never runs and the tile / peer
592/// stays dead for the rest of the establish (a producer whose `publish_track`
593/// failed, or a broadcast parked as a same-path backup behind a zombie). Bounded
594/// in the same spirit as video's `FIRST_KEYFRAME_DEADLINE` (12 s), and slightly
595/// under it: ending here yields a fresh announce cursor plus a re-sent
596/// control-plane `Subscribe`, which is the only thing that heals a zombie.
597const SUBSCRIBE_ANNOUNCE_DEADLINE: Duration = Duration::from_secs(10);
598
599/// Start one per-track decode pipeline. Audio → Opus decode → mix bus; video →
600/// H.264/etc decode → sink keyed `cam:{peer}` / `screen:{peer}`. For video,
601/// `control_outbound` + `track_id` drive the awaiting-keyframe re-request nag.
602#[allow(clippy::too_many_arguments)]
603fn start_pipeline(
604    peer: PeerId,
605    user_id: Uuid,
606    kind: MediaKind,
607    source: Source,
608    low: bool,
609    space_id: Uuid,
610    relay: Arc<MoqRelaySession>,
611    sink: Arc<dyn VideoSink>,
612    playout_bus: Arc<PlayoutBus>,
613    levels: watch::Sender<LevelsView>,
614    clock: CaptureClock,
615    stats: Arc<CallStats>,
616    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
617    track_id: u32,
618    volumes: Arc<StreamVolumes>,
619    end_guard: PipelineEndGuard,
620) -> Subscription {
621    let epoch = end_guard.epoch;
622    let stop = Arc::new(AtomicBool::new(false));
623    let task_stop = stop.clone();
624    // First-delivery latch: audio sets it on the first decoded frame, video when
625    // the first keyframe passes the gate. Read at reap time to pick the restart
626    // pace (an undelivered death climbs the ladder).
627    let delivered = Arc::new(AtomicBool::new(false));
628    let task_delivered = delivered.clone();
629    // This pipeline's downlink-driven demand (video only). A FRESH cell every
630    // call — the whole reset-on-restart contract for `DownlinkDemand` is "a new
631    // pipeline gets a new cell", so a resubscribe / layer switch / reap can never
632    // carry a stale capped demand forward.
633    let downlink = (kind == MediaKind::Video).then(|| Arc::new(DownlinkCell::new()));
634    let task_downlink = downlink.clone();
635    // Audio liveness counter (audio only): the pipeline bumps it per frame pulled
636    // off the wire and the reconcile's [`StallWatch`] samples it. The unbounded
637    // audio pull has no other out-of-band evidence that the track is alive.
638    let arrivals = (kind == MediaKind::Audio).then(|| Arc::new(AtomicU64::new(0)));
639    let task_arrivals = arrivals.clone();
640    // The per-peer inbound stats handle (cloned once; the pipeline increments on it
641    // lock-free per frame).
642    let peer_stats = stats.peer(peer.0);
643    // The moq track name carries the spatial layer: a constrained receiver subscribes
644    // to the `_l0` low track instead of the source.
645    let track_name = match (kind, source, low) {
646        (MediaKind::Audio, Source::Screen, _) => SCREEN_AUDIO_TRACK,
647        (MediaKind::Audio, _, _) => MIC_TRACK,
648        (MediaKind::Video, Source::Screen, false) => SCREEN_TRACK,
649        (MediaKind::Video, Source::Screen, true) => "screen_l0",
650        (MediaKind::Video, _, false) => CAM_TRACK,
651        (MediaKind::Video, _, true) => "cam_l0",
652    };
653    let path = broadcast_path(space_id, user_id, peer, track_name);
654
655    // Audio decode + fill runs on the RT-promoted audio runtime so video CPU can't
656    // starve it; video stays on the general call runtime.
657    let runtime = match kind {
658        MediaKind::Audio => audio_runtime(),
659        MediaKind::Video => call_runtime(),
660    };
661    let task = runtime.spawn(async move {
662        // Owned by the task so its Drop reports this pipeline's end to the reap
663        // channel on ANY exit path (clean end, error, panic, abort).
664        let _end_guard = end_guard;
665        // Bounded: the announce wait is the ONE await whose non-completion defeats
666        // the end guard (see [`SUBSCRIBE_ANNOUNCE_DEADLINE`]). Nothing is half-built
667        // yet, so dropping the future here is clean.
668        let subscribing = relay.subscribe_track(
669            &path,
670            track_name,
671            peer.0 as u16,
672            priority_for(kind),
673            // Audio drains in arrival order (a reordered Opus frame is still playable);
674            // video priority-drops stale GOPs. This is what stops a screenshare from
675            // silently dropping reordered mic/app-audio frames on the shared connection.
676            kind == MediaKind::Audio,
677        );
678        let subscriber = match tokio::time::timeout(SUBSCRIBE_ANNOUNCE_DEADLINE, subscribing).await
679        {
680            Ok(Ok(s)) => s,
681            Ok(Err(e)) => {
682                tracing::warn!(%e, peer = peer.0, "subscribe failed");
683                return;
684            }
685            Err(_) => {
686                tracing::warn!(
687                    peer = peer.0,
688                    track = track_id,
689                    %path,
690                    "subscribe: broadcast never announced within the deadline — ending \
691                     pipeline for a paced re-subscribe"
692                );
693                return;
694            }
695        };
696
697        match kind {
698            MediaKind::Audio => {
699                // This stream's playback-gain cell (the tile volume/mute control),
700                // looked up by key so a set volume survives resubscribes.
701                let gain = volumes.cell(peer.0, source);
702                audio_pipeline(
703                    subscriber,
704                    peer,
705                    source,
706                    playout_bus,
707                    levels,
708                    clock,
709                    task_stop,
710                    peer_stats,
711                    gain,
712                    task_delivered,
713                    // `.expect`: this arm is MediaKind::Audio, where `arrivals` is
714                    // always `Some` (see its construction above).
715                    task_arrivals.expect("audio pipeline always has an arrival counter"),
716                )
717                .await
718            }
719            MediaKind::Video => {
720                // Awaiting-keyframe re-request nag: a separate small task so the
721                // fast re-requests don't race the pipeline's own pull. Ends on
722                // first keyframe or pipeline stop; after the bounded attempts the
723                // pipeline's first-keyframe deadline takes over (it ends itself,
724                // and the reconcile re-subscribes on the restart ladder).
725                let nag = {
726                    let got = task_delivered.clone();
727                    let stop = task_stop.clone();
728                    let outbound = control_outbound.clone();
729                    call_runtime().spawn(async move {
730                        for _ in 0..KEYFRAME_NAG_ATTEMPTS {
731                            tokio::time::sleep(KEYFRAME_NAG_INTERVAL).await;
732                            if got.load(Ordering::Relaxed) || stop.load(Ordering::Relaxed) {
733                                return;
734                            }
735                            tracing::info!(
736                                peer = peer.0,
737                                track = track_id,
738                                "video: still awaiting first keyframe — re-requesting"
739                            );
740                            let _ = outbound.send(ClientMessage::RequestKeyframe {
741                                track: hound_control::TrackId(track_id),
742                            });
743                        }
744                    })
745                };
746                video_pipeline(
747                    subscriber,
748                    peer,
749                    source,
750                    sink,
751                    task_stop,
752                    peer_stats,
753                    task_delivered,
754                    control_outbound,
755                    track_id,
756                    // `.expect`: this arm is MediaKind::Video, where `downlink` is
757                    // always `Some` (see its construction above).
758                    task_downlink.expect("video pipeline always has a downlink cell"),
759                    source_top_bps(source),
760                )
761                .await;
762                nag.abort();
763            }
764        }
765    });
766
767    let started = Instant::now();
768    Subscription {
769        stop,
770        task,
771        epoch,
772        started,
773        delivered,
774        downlink,
775        audio: arrivals.map(|arrivals| AudioLiveness {
776            arrivals,
777            watch: StallWatch::new(started),
778        }),
779    }
780}
781
782fn priority_for(kind: MediaKind) -> u8 {
783    match kind {
784        MediaKind::Audio => AUDIO_PRIORITY,
785        MediaKind::Video => VIDEO_PRIORITY,
786    }
787}

789/// The top-rung bitrate a video `source` is encoded at — the same per-source
790/// constant the producer's own [`hound_media::SimulcastPlanner`] assumes (see the
791/// `camera_tx`/`screen_tx` publish loops), so the receiver's downlink-demand
792/// thresholds (source/4, source/2) are computed against the SAME notion of
793/// "source" the producer is coalescing low-layer requests against.
794fn source_top_bps(source: Source) -> u64 {
795    match source {
796        Source::Camera => CAM_BITRATE.bps() as u64,
797        Source::Screen => SCREEN_BITRATE.bps() as u64,
798        Source::Microphone => unreachable!("video pipelines never carry Source::Microphone"),
799    }
