1//! The subscribe loop: reconcile remote tracks → decode → sink / mix bus. Diffs
2//! the `SessionView` roster into per-track pipelines ([`start_pipeline`]), picks
3//! nothing / low layer / source per tile demand (debounced through the policy
4//! hysteresis, and gated at join by [`SourceHold`] so an unserved tile waits for
5//! the right rung instead of stampeding the source), and self-heals: every
6//! pipeline carries a [`PipelineEndGuard`] whose Drop reports the death so the
7//! reconcile reaps + re-subscribes it (restart-paced).
8//! Invariant: audio subscribes with arrival-order drain and the smaller moq
9//! priority value — audio must strictly outrank video on the shared connection.

13/// Lock-free single-slot handle for one video pipeline's downlink-driven demand:
14/// the video pull loop ([`video_pipeline`]) writes its latest
15/// [`hound_media::QualityRequest`] here as frames arrive (via
16/// [`hound_media::DownlinkDemand`]); the reconcile loop below reads it every tick.
17/// A plain atomic (not a lock) so the per-frame write on the pull loop never
18/// blocks. `None` (audio, or a pipeline that hasn't decided yet) reads as
19/// [`hound_media::QualityRequest::Source`] — the optimistic default, matching
20/// [`hound_media::DownlinkDemand::new`].
21///
22/// Two consumers, one per half of the receiver's adaptation, and the SAME value
23/// answers both:
24///
25/// * folded per track with that tile's own demand
26///   ([`hound_media::most_constrained`]) — the final clamp, which no budget can lift;
27/// * compared against what the allocator granted that tile
28///   ([`hound_media::downlink_deficient`]) to produce this receiver's DEFICIENT
29///   verdict, which drives the total video budget.
30#[derive(Debug)]
31pub(super) struct DownlinkCell(AtomicU64);
32
33/// Sentinel encoding `QualityRequest::Source` in the packed atomic — a `Capped`
34/// bitrate this large never occurs (an encode rung tops out in the low millions).
35const SOURCE_SENTINEL: u64 = u64::MAX;
36
37impl DownlinkCell {
38    fn new() -> Self {
39        Self(AtomicU64::new(SOURCE_SENTINEL))
40    }
41
42    pub(super) fn store(&self, req: hound_media::QualityRequest) {
43        let v = match req {
44            hound_media::QualityRequest::Source => SOURCE_SENTINEL,
45            // This cell carries the BANDWIDTH half of the demand only; visibility
46            // is the tile half, folded in by `most_constrained` at the call site.
47            // A `Hidden` can never be produced here (see `DownlinkDemand::evaluate`)
48            // and is stored as its cap rather than silently widened to source.
49            hound_media::QualityRequest::Capped(bps) | hound_media::QualityRequest::Hidden(bps) => {
50                bps.min(SOURCE_SENTINEL - 1)
51            }
52        };
53        self.0.store(v, Ordering::Relaxed);
54    }
55
56    fn load(&self) -> hound_media::QualityRequest {
57        match self.0.load(Ordering::Relaxed) {
58            SOURCE_SENTINEL => hound_media::QualityRequest::Source,
59            bps => hound_media::QualityRequest::Capped(bps),
60        }
61    }
62}

64/// Floor wake-up for the reconcile loop. Every OTHER wake source is event-driven
65/// (roster change, pause flip, tile-demand change, pipeline death, a pending
66/// debounced switch) and in a steady call none of them fire for minutes — so the
67/// signals the pipelines fold per frame ([`DownlinkCell`], the audio arrival
68/// counter) would never be observed and a congestion downgrade / a wedged audio
69/// track could not actuate. [`LayerHysteresis`]'s debounce+dwell bounds any churn
70/// a periodic pass could cause, so polling is safe. 500 ms is the reconcile rate
71/// the dwell tests in [`super::policy`] already model.
72const RECONCILE_TICK: Duration = Duration::from_millis(500);
73
74/// How long an inbound AUDIO subscription may deliver NOTHING before the reconcile
75/// treats it as wedged. The audio pull is deliberately unbounded — the moq group
76/// read is not cancel-safe mid-object, so no timeout may ever be raced against
77/// `next_frame` — which means a relay that stops delivering a track WITHOUT
78/// closing it parks the pipeline task forever: its [`PipelineEndGuard`] never
79/// drops, the reconcile never reaps it, and that peer is silent for the rest of
80/// the establish. This is the out-of-band equivalent of video's
81/// `FIRST_KEYFRAME_DEADLINE`. 8 s is far above any jitter/retransmit tail on a
82/// 50 fps Opus stream (the encoder runs without DTX, so an unmuted sender emits a
83/// frame every 20 ms whether or not anyone is talking) while keeping a wedged
84/// peer's voice loss inside ~10 s. See [`StallWatch`] for why mute is exempt.
85const AUDIO_STALL: Duration = Duration::from_secs(8);

87/// Pure liveness policy for one inbound audio subscription (unit-tested). The
88/// pipeline bumps a monotone arrival counter as frames come off the wire; this
89/// samples it on the reconcile tick and reports a subscription that has not moved
90/// for `stall`.
91///
92/// A `muted` observation RE-ARMS the window instead of accumulating stall time: a
93/// muted mic gates before encode and publishes nothing at all, so silence there is
94/// healthy — accusing it would reap + re-subscribe every muted peer in the call
95/// every window, churning the connection audio shares (an "audio is sacred"
96/// violation of exactly the kind this watchdog exists to prevent).
97#[derive(Debug)]
98struct StallWatch {
99    last_count: u64,
100    /// When the counter last moved (or the window was last re-armed).
101    since: Instant,
102}
103
104impl StallWatch {
105    fn new(now: Instant) -> Self {
106        Self {
107            last_count: 0,
108            since: now,
109        }
110    }
111
112    /// Fold one sample of the arrival counter. Returns true once the subscription
113    /// has delivered nothing for `stall` while the roster reports it unmuted.
114    fn observe(&mut self, count: u64, now: Instant, muted: bool, stall: Duration) -> bool {
115        if muted || count != self.last_count {
116            self.last_count = count;
117            self.since = now;
118            return false;
119        }
120        now.duration_since(self.since) >= stall
121    }
122}

131/// The reconcile select's wait: the soonest pending debounced switch, FLOORED by
132/// [`RECONCILE_TICK`] so the loop always wakes to poll the per-frame signals even
133/// when no event is pending. Pure (takes `now`) so the floor is unit-tested.
134fn reconcile_wait(next_deadline: Option<Instant>, now: Instant) -> Duration {
135    next_deadline
136        .map(|d| d.saturating_duration_since(now).min(RECONCILE_TICK))
137        .unwrap_or(RECONCILE_TICK)
138}

140/// Which track of one (peer, video source) group to subscribe to for the committed
141/// demand `req`. Pure, so the fallbacks are unit-tested. `None` means **subscribe to
142/// nothing** — the reconcile's `gone` sweep then tears any existing pipeline down and
143/// sends `Unsubscribe`, and this peer's AUDIO is untouched (a different track).
144///
145/// `req` is `None` while the UI has not reported this tile even once; `source_ok`
146/// carries the bounded escalation gate ([`SourceHold`]) that decides whether an
147/// unserved tile may take the SOURCE track yet.
148///
149/// The roster is the ONLY liveness signal available here, and the producer keeps it
150/// truthful: it withdraws the `_l0` [`TrackDecl`] whenever it deactivates the low
151/// encoder (see [`super::simulcast::LowDecl`]). So:
152///
153/// - a HIDDEN tile subscribes to nothing at all. Its demand still goes to the
154///   producer (`RequestQuality { hidden: true }`) — a fully-hidden receiver set
155///   *sizes* the shared low layer rather than letting it fall back to source, which
156///   is what keeps the thumbnail rung published and makes an unhide a subscribe
157///   rather than a wait on a producer (Octo's "forward every layer so a switch is
158///   instant", inverted to one relay);
159/// - a capped demand takes the low track when its decl is present; when it is not,
160///   it takes the source only if [`SourceHold`] allows. Subscribing to a decl-less
161///   or torn-down layer yields nothing and costs that tile the full `VIDEO_STALL` →
162///   `FIRST_KEYFRAME_DEADLINE` → restart-ladder walk (minutes of black video);
163/// - a low track WITHOUT its source is never picked at all. The producer's source
164///   encode is never dropped (the low layer is purely additive), so a lone `_l0` is
165///   by construction stale — its owner stopped publishing that video and only the
166///   source's `Unpublish` has reached us so far.
167///
168/// When the low decl (re)appears the reconcile switches to it through the normal
169/// [`LayerHysteresis`] path — source→low is a downgrade, so it is debounced.
170fn pick_layer(
171    req: Option<hound_media::QualityRequest>,
172    src_id: Option<u32>,
173    low_id: Option<u32>,
174    source_ok: bool,
175) -> Option<u32> {
176    // A low track is only ever picked alongside its source (see above).
177    let low_id = src_id.and(low_id);
178    match req {
179        Some(hound_media::QualityRequest::Hidden(_)) => None,
180        Some(hound_media::QualityRequest::Capped(_)) => {
181            low_id.or_else(|| src_id.filter(|_| source_ok))
182        }
183        Some(hound_media::QualityRequest::Source) => src_id,
184        // No report yet: this tile has stated no demand, so it gets the source only
185        // once the hold gives up waiting for one.
186        None => src_id.filter(|_| source_ok),
187    }
188}

190/// How long a video track may go unsubscribed while the receiver waits for the
191/// evidence it needs to pick the RIGHT rung, before it gives up and takes the source.
192///
193/// Two things are waited on, both of which arrive in well under a second in a healthy
194/// call: the UI's first size report for a freshly-rostered tile (a texture bind plus
195/// one layout pass), and the producer's `_l0` decl after our `RequestQuality` (a
196/// control round trip, an encoder build and one capture frame). 3 s is ~10× either,
197/// far inside `FIRST_KEYFRAME_DEADLINE`, and — because the tile is blank for the whole
198/// wait either way — invisible when it is spent.
199const SOURCE_HOLD_GRACE: Duration = Duration::from_secs(3);

201/// The bounded gate on escalating a tile to the SOURCE track. One per
202/// (peer, video source); pure, so the whole policy is unit-tested.
203///
204/// ## What it fixes
205///
206/// `pick_layer` used to answer a capped demand with `low.or(source)`, and a tile with
207/// no report yet defaulted to `Source`. At a 32-peer join neither the low decls nor
208/// the tile reports exist yet, so every receiver subscribed to **31 source tracks**
209/// (~46 Mbps) and only walked back down a debounce later — a self-inflicted
210/// congestion collapse at exactly the moment the call is most fragile. Jitsi ships the
211/// same policy as a default (`initial-max-height-px = 0`: forward nothing until the
212/// receiver states its demand), and MoQT draft-19 names the shape `RENDEZVOUS_TIMEOUT`
213/// — "the duration the subscriber is willing to wait for a publisher to become
214/// available".
215///
216/// ## Why it cannot blank a legitimate tile
217///
218/// Three independent escapes, any one of which allows the source immediately:
219///
220/// 1. **`on_source`** — a tile already showing source video is never torn down to
221///    wait for anything. A shrink or a hide is a *downgrade* decision, made by the
222///    hysteresis, not by this gate;
223/// 2. **`low_seen`** — once the `_l0` decl has been in the roster for this
224///    (peer, source), its later absence is a real teardown (the resume-after-pause
225///    path, where the producer withdrew the decl while we were paused), not a join
226///    race. That case keeps its original instant fallback, guard tests and all;
227/// 3. **the grace itself** — a producer that will never publish a low layer (an old
228///    peer, a failed low encoder, a lost request) is indistinguishable from a slow
229///    one, so the wait is bounded and then **waived for good**: the tile takes the
230///    source and never pays the wait again on this establish. Without the latch a
231///    peer that never publishes low would re-pay it on every resubscribe, and the
232///    resume-after-pause budget would be a grace longer for no benefit.
233///
234/// ## Who still reaches the `demand_known: false` branch
235///
236/// In the shipped app, essentially nobody past the first tile that lays out:
237/// [`TileDemand`] seeds every roster peer's silence to `Hidden` once this receiver
238/// has reported anything at all, so a tile the virtualized grid never BUILDS is
239/// treated as not-rendered rather than as unknown. The branch is kept because it is
240/// the honest answer for a receiver that reports no demand at all — houndbot, the
241/// testkit harness, any future surface that renders video without a layout pass —
242/// which must still get video, and gets it one grace after joining.
243///
244/// ## Interaction with the receive-side budget
245///
246/// The waiver outranks the budget, and deliberately. `hound_media::allocate_video` can
247/// hand a tile the thumbnail rung and this gate can still put it on the SOURCE track a
248/// grace later, because the alternative — holding a capped subscription against a
249/// producer that never published `_l0` — is a permanently blank tile, and P2 already
250/// decided a blank tile is worse than an expensive one. The budget is not undermined by
251/// that: it bounds what this receiver ASKS the producers for, the waiver is a
252/// last-resort escape from asking for something nobody publishes, and the escape shows
253/// up honestly in the controller's own accounting — the apply pass prices the track
254/// actually subscribed to, so a waived tile is billed at the source rate it really
255/// costs and the budget backs off accordingly.
256///
257/// How long a tile whose demand is only INFERRED — this receiver reports, but has
258/// never reported *this* tile — stays unsubscribed before it escalates to the
259/// thumbnail rung. See [`TileDemand`]'s two-tier trust model: a reported hide is
260/// trusted forever; an inferred one is not, because a lost report and a tile the
261/// grid never built look identical from here.
262///
263/// 10 s is long enough that the UI's own recovery (a re-assert on connect or roster
264/// change, `demandNeedsReassert`) always wins the race in a healthy app, so this
265/// never fires for a tile that is genuinely off screen — and short enough that a
266/// user staring at a black tile because their report was swallowed gets video back
267/// before they reach for reload.
268const SEEDED_ESCAPE_GRACE: Duration = Duration::from_secs(10);

270/// Per-(peer, video source) subscription policy state that must **survive a
271/// re-establish**: the bounded source gate ([`SourceHold`]) and the seeded tile's
272/// escape clock. Owned by the session (see [`TilePolicies`]), not by the subscribe
273/// loop, because a reconnect must not re-pay a grace this tile already paid — a 1:1
274/// call whose peer publishes no low layer would otherwise go black for
275/// [`SOURCE_HOLD_GRACE`] on every single reconnect.
276#[derive(Debug, Default)]
277pub(super) struct TilePolicy {
278    hold: SourceHold,
279    /// When this tile's demand first became a SEED with nothing subscribed —
280    /// the [`SEEDED_ESCAPE_GRACE`] clock. Cleared the moment the tile reports.
281    seeded_since: Option<Instant>,
282}

284/// The session-owned map of [`TilePolicy`]. The subscribe loop is its only user,
285/// but it outlives each establish (see [`TilePolicy`]). A plain `Mutex`: it is
286/// touched once per reconcile tick (2 Hz), never on a media path.
287pub(super) type TilePolicies = Arc<std::sync::Mutex<HashMap<(PeerId, Source), TilePolicy>>>;

289/// Lock the shared policy map, recovering from a poisoned mutex: the guarded state
290/// is plain policy bookkeeping, and a panic elsewhere must not take the subscribe
291/// loop — and with it every inbound stream — down with it.
292fn lock_policies(
293    policies: &TilePolicies,
294) -> std::sync::MutexGuard<'_, HashMap<(PeerId, Source), TilePolicy>> {
295    policies.lock().unwrap_or_else(|e| e.into_inner())
296}

298impl TilePolicy {
299    /// Fold one reconcile observation of a SEEDED demand (see [`DemandOrigin`]),
300    /// returning the demand to actually act on: the seed itself while the escape
301    /// grace runs, then the thumbnail rung — never the source, which is the
302    /// standing overshoot this whole policy exists to remove.
303    fn escalate_seed(
304        &mut self,
305        seed: hound_media::QualityRequest,
306        now: Instant,
307        grace: Duration,
308    ) -> hound_media::QualityRequest {
309        let since = *self.seeded_since.get_or_insert(now);
310        match seed {
311            hound_media::QualityRequest::Hidden(bps) if now.duration_since(since) >= grace => {
312                hound_media::QualityRequest::Capped(bps)
313            }
314            other => other,
315        }
316    }
317
318    /// This tile reported for itself: the inference — and its escape clock — no
319    /// longer apply.
320    fn reported(&mut self) {
321        self.seeded_since = None;
322    }
323
324    /// A subscription for this tile died on its own and is being reaped. It had a
325    /// pipeline, so the evidence [`SourceHold`] waits for has already been
326    /// collected: the restart must not re-pay the grace on top of the restart
327    /// ladder. `low` marks the death of the `_l0` track specifically, which is
328    /// standing proof the layer exists.
329    fn subscription_reaped(&mut self, low: bool) {
330        self.hold.waived = true;
331        self.hold.since = None;
332        self.hold.low_seen |= low;
333    }
334}

337struct SourceHold {
338    /// Whether the `_l0` decl has EVER been present for this (peer, source).
339    low_seen: bool,
340    /// Whether the grace has already been spent (see escape 3 above).
341    waived: bool,
342    /// When the current wait started; `None` when not waiting.
343    since: Option<Instant>,
344}

346/// One reconcile pass's view of a (peer, video source) group, as [`SourceHold`]
347/// reads it. A struct rather than four bare `bool`s so no call site can transpose
348/// them.
349#[derive(Debug, Clone, Copy)]
350struct HoldInputs {
351    /// The UI has reported this tile's size at least once.
352    demand_known: bool,
353    /// The committed demand wants a below-source rung (`Capped`; `Hidden` does not —
354    /// it wants nothing at all).
355    wants_low: bool,
356    /// The `_l0` decl is in the roster right now.
357    low_present: bool,
358    /// We are already subscribed to this group's SOURCE track.
359    on_source: bool,
360    /// This tile's demand is INFERRED, not reported ([`DemandOrigin::Seeded`]).
361    /// Such a tile may never take the source track: the whole point of seeding is
362    /// that ~19 of 32 peers in a virtualized grid stop paying for a rung nobody
363    /// renders, and letting the gate's own escape hatch hand them the source
364    /// would put every one of them straight back on it. Its escape is
365    /// [`TilePolicy::escalate_seed`] — one thumbnail rung — instead.
366    seeded: bool,
367}
368
369impl SourceHold {
370    /// Fold one reconcile observation; returns whether the SOURCE track may be
371    /// subscribed for this tile right now.
372    fn observe(&mut self, i: HoldInputs, now: Instant, grace: Duration) -> bool {
373        if i.low_present {
374            self.low_seen = true;
375        }
376        if i.seeded {
377            // Not this gate's decision to make, and its timer must not run: a tile
378            // that later REPORTS gets a full, unspent grace.
379            self.since = None;
380            return false;
381        }
382        let waiting = !self.waived
383            && !i.on_source
384            && (!i.demand_known || (i.wants_low && !i.low_present && !self.low_seen));
385        if !waiting {
386            self.since = None;
387            return true;
388        }
389        let since = *self.since.get_or_insert(now);
390        if now.duration_since(since) >= grace {
391            self.waived = true;
392            self.since = None;
393            return true;
394        }
395        false
396    }
397
398    /// When the reconcile must wake to end an ongoing wait, so the fallback actuates
399    /// without depending on the floor tick. `None` when nothing is being waited on.
400    fn pending_deadline(&self, grace: Duration) -> Option<Instant> {
401        self.since.map(|since| since + grace)
402    }
403}

435/// One active remote subscription: the decode task + a stop flag.
436struct Subscription {
437    stop: Arc<AtomicBool>,
438    task: JoinHandle<()>,
439    /// Generation stamp: a reap signal carries the epoch of the pipeline that died,

444    /// [`RestartPacer`]) so a track the relay instantly rejects can't hot-loop.

450    /// This pipeline's downlink-driven demand (video only — `None` for audio). A
451    /// FRESH cell per pipeline start, so a resubscribe / layer switch / reap
452    /// naturally resets the demand to `Source` (see [`DownlinkCell`]'s docs).
453    downlink: Option<Arc<DownlinkCell>>,
454    /// Inbound-audio liveness (audio only — `None` for video, which carries its

456    audio: Option<AudioLiveness>,
457    /// Which tile this subscription serves, and whether it is the low rung —
458    /// video only. The reaper needs it to credit the tile's [`TilePolicy`]: a
459    /// subscription that existed is standing evidence that this tile's rung
460    /// question is already answered, so its restart must not also wait out
461    /// [`SOURCE_HOLD_GRACE`].
462    tile: Option<((PeerId, Source), bool)>,
463}

472/// Fires `(track_id, epoch)` on the reap channel when the owning pipeline task ends
473/// — by ANY path: clean end-of-track, a transport error, a panic unwind, or an
474/// abort (tokio drops the future at an await point, running destructors). This is
475/// what lets the subscribe loop REAP a self-terminated pipeline: without it, a

479    track_id: u32,
480    epoch: u64,
481    tx: tokio::sync::mpsc::UnboundedSender<(u32, u64)>,
482}

484impl Drop for PipelineEndGuard {
485    fn drop(&mut self) {
486        let _ = self.tx.send((self.track_id, self.epoch));
487    }

490/// Spawn the subscribe loop: watch the `SessionView`, and for every remote track

519    volumes: Arc<StreamVolumes>,
520    // The dominant-speaker ranking this establish shares with the control reader
521    // (which feeds it every peer's reported level). Read once per reconcile pass.
522    speakers: SharedSpeakers,
523) -> JoinHandle<()> {
524    call_runtime().spawn(async move {
525        let mut tile_changed = tile_quality.subscribe();
526        // Keyed by TrackId.0 so we can diff against the view on each change.
527        let mut active: HashMap<u32, Subscription> = HashMap::new();
528        // The cap we last reported per (peer, source) — `(source track id, cap)` —
529        // so a RequestQuality is sent only when our demand for that peer's video
530        // actually changes, and so a PAUSE can address the withdrawal at the same
531        // track it was reported against.
532        // The last standing request actually put on the wire per (peer, source), so
533        // an unchanged demand re-sends nothing.
534        let mut last_sent: HashMap<(PeerId, Source), SentQuality> = HashMap::new();
535        // Per-(peer, source) hysteresis so a transient tile-resize flap (notably a
536        // rejoin relayout) can't thrash the source⇄low subscription on the shared
537        // audio+video connection. Holds a demanded layer switch until it is durable.
538        let mut hysteresis: HashMap<(PeerId, Source), LayerHysteresis> = HashMap::new();
539        // Self-healing: every pipeline task carries a PipelineEndGuard that reports
540        // its (track_id, epoch) here on ANY exit, so a self-terminated pipeline is
541        // reaped from `active` and the reconcile re-subscribes it (paced).
542        let (ended_tx, mut ended_rx) = tokio::sync::mpsc::unbounded_channel::<(u32, u64)>();
543        let mut next_epoch: u64 = 0;
544        let mut pacer = RestartPacer::default();
545        // This receiver's TOTAL video budget, walked by AIMD against the DEFICIENT
546        // verdict the per-track downlink cells produce. Per-establish state: a
547        // reconnect starts probing from the neutral 4 Mbps again rather than
548        // inheriting a backoff that belonged to a link that no longer exists.
549        let mut video_budget = hound_media::VideoBudget::new(hound_media::BudgetConfig::default());
550        // The receiver's DECODE budget, the CPU-side twin of the bandwidth one above:
551        // how many tiles may hold a decoder at all. It starts at this platform's
552        // static decode-session ceiling — the only `#[cfg]`-resolved number in the
553        // whole receive path, and it lives behind the engine's capability seam beside
554        // `decode_caps()` rather than here (this crate carries no platform `cfg`) —
555        // and walks down against RT audio health. Per-establish for the same reason
556        // the bandwidth budget is: a reconnect starts from the platform ceiling again
557        // rather than inheriting a cut earned on a machine state that has moved on.
558        let decode_ceiling = Engine::new().decode_session_budget();
559        // The budget controller is sans-IO and takes µs; this is the loop's own
560        // monotone origin for it (`Instant` has no epoch to subtract), and it doubles
561        // as the decode budget's warmup origin — the join storm this establish is
562        // about to cause starts here.
563        let loop_start = Instant::now();
564        let mut decode_budget = DecodeBudget::new(decode_ceiling, loop_start);
565        // What the budget last handed out, so a change is logged once instead of every
566        // pass (the same shape as `last_sent` for the per-tile demand).
567        let mut last_decode_k = decode_ceiling;
568        // The (peer, source) tiles the allocation currently has HIDDEN, republished to
569        // the UI on change so a tile with no subscription renders its owner's avatar
570        // instead of a black rectangle holding an idle texture. See [`HiddenTiles`].
571        // Keyed by the raw peer id, which is the key [`CallView`] carries.
572        let mut hidden_now: HashSet<(u32, Source)> = HashSet::new();
573        // Evidence about the allocation that is currently IN FORCE, produced by the
574        // apply pass below and consumed by the next pass's `observe`. It has to lag by
575        // a tick: the budget decides the allocation, so it cannot also be derived from
576        // it. At a 500 ms tick against a 1 s sustain window that costs nothing.
577        let mut in_force = hound_media::BudgetObservation::default();
578        // Remote peers we have inbound stats rows for, so a departed peer's row is
579        // dropped from the debug snapshot instead of freezing there until the next
580        // re-establish.
581        let mut known_peers: std::collections::HashSet<u32> = std::collections::HashSet::new();
582
583        loop {
584            // Soonest pending debounced layer switch — the loop wakes to actuate it
585            // even if no input changes in the meantime.
586            let mut next_switch_deadline: Option<Instant> = None;
587            // When video is paused (app backgrounded / call surface hidden) drop
588            // every inbound video sub — receiving a high-bitrate screenshare nobody
589            // is watching is wasted CPU + bandwidth. Audio always flows.
590            let drop_video = *video_paused.borrow();
591            let demand = tile_quality.borrow().clone();
592            let now = Instant::now();
593            // Rebuilt from scratch every pass (the allocation is), so a peer that
594            // left, un-paused or was un-hidden cannot leave a stale flag behind.
595            hidden_now.clear();
596
597            // Is this receiver decode-LIMITED? The evidence is the RT audio mix
598            // thread's tick lateness, not any video signal — see
599            // [`DECODE_TICK_LATE_LIMIT_US`] for why video yields to audio here and
600            // never the reverse. The raw tail crosses the seam, not a verdict: the
601            // threshold it is compared against is calibrated per platform inside the
602            // controller (see `DecodeBudget::observe`). While video is PAUSED nothing
603            // is decoding, so the mix thread's schedule is evidence about neither
604            // direction and both of the controller's clocks freeze.
605            let late = if drop_video {
606                TickLate::Unobserved
607            } else {
608                TickLate::Observed(stats.tick_late_p95_us())
609            };
610            let decode_k = decode_budget.observe(late, now);
611            if decode_k != last_decode_k {
612                tracing::info!(
613                    decode_k,
614                    from = last_decode_k,
615                    ceiling = decode_ceiling,
616                    tick_late_p95_us = stats.tick_late_p95_us(),
617                    "simulcast[recv]: decode budget moved"
618                );
619                last_decode_k = decode_k;
620            }
621            // A CPU-motivated cut must not fund an upgrade. Once the controller has
622            // walked BELOW the platform's static ceiling this receiver has evidenced
623            // that it is short of CPU, and the bitrate freed by hiding tiles would
624            // otherwise buy the survivors the source rung — trading 24 thumbnails for
625            // one 720p decode, which is not the saving that was asked for. So every
626            // gathered tile's demand is folded with `below_source_cap` below. Screen
627            // shares included: a 360p share is the lesser evil against a share whose
628            // source rung is what is starving the mix thread, and the screen ladder's
629            // legibility ordering keeps that rung readable. At `k == ceiling` the cut
630            // is the platform's static session cap alone (32 tiles on Apple's 16 says
631            // nothing about this box's CPU) and upgrades stay allowed.
632            let suppress_upgrades = decode_budget.cpu_constrained();
633
634            // Pausing drops every video subscription — withdraw the standing quality
635            // demand too. The producer keeps a receiver's request until that peer
636            // LEAVES the roster, so a paused-but-present receiver otherwise pins the
637            // shared low layer alive: a full extra downscale + encode of every frame
638            // for a viewer receiving nothing. `last_sent` is cleared but the
639            // hysteresis is NOT, so a resume re-reports the same committed demand at
640            // once and nothing churns on the way back.
641            if drop_video && !last_sent.is_empty() {
642                for (&(peer, source), &(sid, (cap, _hidden))) in last_sent.iter() {
643                    if cap.is_none() {
644                        continue; // nothing standing at the producer
645                    }
646                    tracing::info!(
647                        peer = peer.0,
648                        ?source,
649                        src_track = sid,
650                        "simulcast[recv]: video paused — withdrawing standing RequestQuality"
651                    );
652                    let _ = control_outbound.send(ClientMessage::RequestQuality {
653                        track: hound_control::TrackId(sid),
654                        cap_bps: None,
655                        hidden: false,
656                    });
657                }
658                last_sent.clear();
659            }
660
661            // Audio stall watchdog + departed-peer stats reap. Both need the roster,
662            // and both must run BEFORE the `desired` snapshot below borrows `active`.
663            {
664                let (audio_muted, present): (HashMap<u32, bool>, std::collections::HashSet<u32>) = {
665                    let view = view_rx.borrow();
666                    (
667                        view.tracks()
668                            .filter(|t| t.kind == MediaKind::Audio)
669                            .map(|t| (t.id.0, t.muted))
670                            .collect(),
671                        view.peers()
672                            .map(|p| p.id.0)
673                            .filter(|&id| id != self_id.0)
674                            .collect(),
675                    )
676                };
677                // A pipeline parked in the unbounded audio pull never drops its end
678                // guard, so the reap + restart ladder that exists for exactly this
679                // never runs. Kill it here; the guard then drives the normal path.
680                // A track absent from the roster reads as muted (don't accuse it —
681                // the `gone` sweep below removes it this same pass).
682                let wedged: Vec<u32> = active
683                    .iter_mut()
684                    .filter_map(|(id, sub)| {
685                        let live = sub.audio.as_mut()?;
686                        let count = live.arrivals.load(Ordering::Relaxed);
687                        let muted = audio_muted.get(id).copied().unwrap_or(true);
688                        live.watch
689                            .observe(count, now, muted, AUDIO_STALL)
690                            .then_some(*id)
691                    })
692                    .collect();
693                for id in wedged {
694                    if let Some(sub) = active.remove(&id) {
695                        let delivered = sub.delivered.load(Ordering::Relaxed);
696                        tracing::warn!(
697                            track = id,
698                            delivered,
699                            stall_s = AUDIO_STALL.as_secs(),
700                            "audio: subscription delivered nothing for the stall window \
701                             — killing the wedged pipeline for a fresh subscribe"
702                        );
703                        pacer.on_death(id, delivered, sub.started.elapsed(), now);
704                        sub.stop();
705                    }
706                }
707                // Inbound stats rows are created lazily per pipeline and otherwise
708                // only cleared on a full re-establish; drop a departed peer's row so
709                // the snapshot stops reporting its frozen gauges for the rest of the call.
710                let departed: Vec<u32> = known_peers.difference(&present).copied().collect();
711                for id in &departed {
712                    stats.remove_peer(*id);
713                    // Their streams are gone for good — let the renderer drop the
714                    // mailboxes (each pins the last decoded frame otherwise).
715                    sink.stream_ended(&cam_key(PeerId(*id)));
716                    sink.stream_ended(&screen_key(PeerId(*id)));
717                }
718                // …and their tile demand, which the session would otherwise carry for
719                // the rest of the call and hand to whoever inherits that peer id on a
720                // later establish. The UI never reports for self, so pruning to the
721                // remote roster cannot drop a live entry.
722                if !departed.is_empty() {
723                    tile_quality
724                        .send_if_modified(|d| d.retain_peers(|peer| present.contains(&peer.0)));
725                }
726                known_peers = present;
727            }
728            // Advance the dominant-speaker grid and snapshot it for this pass,
729            // pruning peers that have left in the same call (a departed peer must
730            // not keep a rung promotion for whoever inherits its id). Taken AFTER
731            // the roster reap above so `known_peers` is this pass's truth, and
732            // outside the borrow below so the lock is never held while reconciling.
733            let speaking = speakers.decide(|peer| known_peers.contains(&peer));
734            // Snapshot the remote tracks we should be subscribed to. Audio is taken
735            // as-is; for each (peer, video source) we pick EITHER the source or the
736            // low track by our demand, and report that demand upstream so the producer
737            // spins up the low layer.
738            let desired: Vec<(u32, PeerId, MediaKind, Source, bool, Uuid)> = {
739                let view = view_rx.borrow();
740                let mut out: Vec<(u32, PeerId, MediaKind, Source, bool, Uuid)> = Vec::new();
741                for t in view.tracks().filter(|t| {
742                    t.peer != self_id
743                        && t.kind == MediaKind::Audio
744                        && matches!(t.source, Source::Microphone | Source::Screen)
745                }) {
746                    // The broadcast path is built from the peer's AUTHENTICATED
747                    // user_id (control-plane roster), never the realtime-derived
748                    // identity map. An old backend that hasn't stamped it yet
749                    // sends the nil default — skip rather than subscribe under a
750                    // wrong (or nobody's) path.
751                    let Some(user_id) =
752                        view.peer(t.peer).map(|p| p.user_id).filter(|u| !u.is_nil())
753                    else {
754                        tracing::warn!(
755                            peer = t.peer.0,
756                            track = t.id.0,
757                            "subscribe: remote peer has no known user_id (old backend?) — skipping"
758                        );
759                        continue;
760                    };
761                    out.push((t.id.0, t.peer, t.kind, t.source, false, user_id));
762                }
763                if !drop_video {
764                    // Group each peer's video into (source track, low track).
765                    let mut groups: HashMap<(PeerId, Source), (Option<u32>, Option<u32>)> =
766                        HashMap::new();
767                    for t in view.tracks().filter(|t| {
768                        t.peer != self_id
769                            && t.kind == MediaKind::Video
770                            && matches!(t.source, Source::Camera | Source::Screen)
771                    }) {
772                        let e = groups.entry((t.peer, t.source)).or_insert((None, None));
773                        if t.low {
774                            e.1 = Some(t.id.0);
775                        } else {
776                            e.0 = Some(t.id.0);
777                        }
778                    }
779                    // How many remote peers publish video at all — the room-size
780                    // input to the ranking's DOWNGRADE half (a small call can
781                    // afford everyone; see `hound_media::SPEAKER_DEMOTE_MIN_PEERS`).
782                    let video_peers = groups
783                        .keys()
784                        .map(|&(peer, _)| peer)
785                        .collect::<std::collections::HashSet<_>>()
786                        .len();
787
788                    // ── GATHER ──────────────────────────────────────────────────
789                    // Every tile's inputs, collected before ANY of them is committed.
790                    // This pass exists because the rung decision is no longer per
791                    // tile: `hound_media::allocate_video` ranks the whole set against
792                    // one total budget, and deriving each tile's rung inside a single
793                    // loop — which is what this used to be — is exactly how N tiles
794                    // that each fit the link come to overshoot it N-fold.
795                    //
796                    // The gather ORDER is a `HashMap`'s and therefore arbitrary; that
797                    // is safe because the allocator's own priority key is total (it
798                    // ends in the peer id, and a peer's camera and screen are always
799                    // in different classes), so the input order cannot reach a
800                    // decision — pinned by `the_allocation_is_deterministic_and_
801                    // order_independent`.
802                    let mut gathered: Vec<Gathered> = Vec::with_capacity(groups.len());
803                    for (&(peer, source), &(src_id, low_id)) in &groups {
804                        let Some(user_id) =
805                            view.peer(peer).map(|p| p.user_id).filter(|u| !u.is_nil())
806                        else {
807                            tracing::warn!(
808                                peer = peer.0,
809                                ?source,
810                                "subscribe: remote peer has no known user_id (old backend?) — \
811                                 skipping video"
812                            );
813                            continue;
814                        };
815                        // The demand for this tile: its own report, or — once this
816                        // receiver is known to report at all — the roster seed for a
817                        // tile the UI never built, which is `Hidden` (see
818                        // [`TileDemand`]). `None` only for a receiver that has never
819                        // reported anything, and it is then fed to neither the
820                        // allocator (which would be ranking a demand we made up), the
821                        // hysteresis (which would debounce the tile's first real
822                        // report against it) nor the producer: inventing a
823                        // full-quality demand for a tile that has said nothing is half
824                        // of the join transient (see [`SourceHold`]).
825                        let origin = demand.get(peer, source);
826                        let tile_req = origin.map(|(req, origin)| {
827                            let mut policies = lock_policies(&tile_policies);
828                            let policy = policies.entry((peer, source)).or_default();
829                            match origin {
830                                // Trusted forever: a tile that says it is hidden
831                                // mounted and then disposed, and dispose is
832                                // reliable.
833                                DemandOrigin::Reported => {
834                                    policy.reported();
835                                    req
836                                }
837                                // Inferred, so bounded: it escalates to the
838                                // thumbnail rung (never the source) if the
839                                // inference outlives the grace.
840                                DemandOrigin::Seeded => {
841                                    policy.escalate_seed(req, now, SEEDED_ESCAPE_GRACE)
842                                }
843                            }
844                        });
845                        // Ranking modulates the tile's own demand, it does not
846                        // replace it: the dominant speaker's thumbnail buys the
847                        // next rung up, a peer observed silent in a crowded call
848                        // buys one less, and everyone else — including every peer
849                        // we have no speaking evidence about — keeps exactly what
850                        // its tile asked for. It moves a CAP, never a layer axis
851                        // (`speaker_adjusted`'s pinned property), so it can neither
852                        // subscribe, unsubscribe nor layer-switch a tile behind the
853                        // hysteresis's back. Applied BEFORE both the allocator and
854                        // the fold with the downlink demand, so a promotion can never
855                        // lift a receiver above its budget or its link estimate.
856                        let tile_req = tile_req.map(|req| {
857                            hound_media::speaker_adjusted(
858                                video_source(source),
859                                req,
860                                speaking.tier(peer.0),
861                                video_peers,
862                            )
863                        });
864                        let seeded = matches!(origin, Some((_, DemandOrigin::Seeded)));
865                        // This receiver's OWN downlink state for this video, read from
866                        // whichever track (source or low) is CURRENTLY subscribed —
867                        // the pull loop for that pipeline folds each arriving frame's
868                        // `frame_seq` into a `DownlinkDemand` and writes the result
869                        // here (see `DownlinkCell`). Not yet subscribed / audio-only /
870                        // no data yet all read as `Source` with no delivery evidence,
871                        // so a fresh pipeline neither spuriously caps nor drags the
872                        // budget down.
873                        let delivered = [src_id, low_id]
874                            .into_iter()
875                            .flatten()
876                            .find_map(|id| active.get(&id))
877                            .and_then(|s| s.downlink.as_ref())
878                            .map(|c| c.load());
879                        gathered.push(Gathered {
880                            peer,
881                            source,
882                            user_id,
883                            src_id,
884                            low_id,
885                            tile_req,
886                            delivered,
887                            seeded,
888                        });
889                    }
890
891                    // ── ALLOCATE ────────────────────────────────────────────────
892                    // One decision for the whole receiver: rank the tiles, walk the
893                    // total budget by AIMD against whether anything is actually
894                    // arriving, and hand each tile back a rung no richer than what it
895                    // asked for whose SUM fits. This is what closes the hole the
896                    // per-track fold below cannot: a track with no pipeline yet has no
897                    // delivery signal, so it reads as `Source` and the bandwidth half of
898                    // the fold constrains nothing at all at a 32-peer join.
899                    let tiles: Vec<hound_media::TileRequest> = gathered
900                        .iter()
901                        .filter_map(|g| {
902                            let vsource = video_source(g.source);
903                            g.tile_req.map(|req| hound_media::TileRequest {
904                                peer: g.peer.0,
905                                source: vsource,
906                                // The CPU-crisis clamp, applied to the DEMAND rather
907                                // than to the allocator: the allocator may only ever
908                                // lower a demand, so capping the ladder here is
909                                // exactly "no upgrade past the low rung" and it keeps
910                                // `allocate_video` a pure function of its inputs.
911                                demand: if suppress_upgrades {
912                                    hound_media::most_constrained(
913                                        req,
914                                        hound_media::below_source_cap(vsource),
915                                    )
916                                } else {
917                                    req
918                                },
919                                rank: speaking.rank(g.peer.0),
920                                tier: speaking.tier(g.peer.0),
921                                tile_pixels: demand.pixels(g.peer, g.source),
922                            })
923                        })
924                        .collect();
925                    let now_us = now
926                        .duration_since(loop_start)
927                        .as_micros()
928                        .min(i64::MAX as u128) as i64;
929                    // The link evidence (`deficient` / `committed_bps`) describes the
930                    // allocation that is in force; the tile figures describe the roster
931                    // as it is right now. See `hound_media::BudgetObservation`.
932                    //
933                    // Both tile figures are priced over the tiles the decode budget
934                    // will actually SERVE, using the same `decode_k` the allocation
935                    // below is given: a tile the decode cut hides subscribes to
936                    // nothing, so its floor must not seed this controller and its
937                    // demand must not be probed toward. See `hound_media::served_order`.
938                    let budget = video_budget.observe(
939                        hound_media::BudgetObservation {
940                            ceiling_bps: hound_media::demand_ceiling_bps(&tiles, Some(decode_k)),
941                            floor_bps: hound_media::floor_cost_bps(&tiles, Some(decode_k)),
942                            ..in_force
943                        },
944                        now_us,
945                    );
946                    // Two independent caps, one call: `budget` is what this receiver
947                    // can afford to RECEIVE, `decode_k` is what it can afford to
948                    // DECODE. Always a number (never `None`) — see [`DecodeBudget`].
949                    let mut allocation =
950                        hound_media::allocate_video(&tiles, budget, Some(decode_k)).into_iter();
951                    // Re-align to `gathered`: the allocator answers in input order,
952                    // and its input was the gathered tiles that HAVE a demand.
953                    let allocated: Vec<Option<hound_media::QualityRequest>> = gathered
954                        .iter()
955                        .map(|g| {
956                            if g.tile_req.is_some() {
957                                allocation.next()
958                            } else {
959                                None
960                            }
961                        })
962                        .collect();
963
964                    // ── APPLY ───────────────────────────────────────────────────
965                    // What this pass ends up committed to, fed back to the controller
966                    // on the NEXT pass (see `in_force`): the tracks whose delivery is
967                    // measurably short of what they were just allocated, and the priced
968                    // total of the subscriptions actually taken.
969                    let mut deliveries: Vec<hound_media::TrackDelivery> = Vec::new();
970                    let mut committed_bps: u64 = 0;
971                    for (g, alloc) in gathered.iter().zip(allocated) {
972                        let (peer, source, src_id, low_id) = (g.peer, g.source, g.src_id, g.low_id);
973                        let vsource = video_source(source);
974                        // Absent (no pipeline yet) reads as the optimistic `Source`, so
975                        // a fresh pipeline never spuriously caps.
976                        let bwe_req = g.delivered.unwrap_or(hound_media::QualityRequest::Source);
977                        // This track's contribution to the DEFICIENT verdict — only for
978                        // a track that HAS a pipeline, and only ever measured against
979                        // what the allocator just granted it. A peer capped by its own
980                        // bad uplink but delivering the thumbnail it was given says
981                        // nothing about this receiver's downlink.
982                        if let (Some(delivered), Some(allocated)) = (g.delivered, alloc) {
983                            deliveries.push(hound_media::TrackDelivery {
984                                source: vsource,
985                                delivered,
986                                allocated,
987                            });
988                        }
989                        // The EFFECTIVE demand is the more constrained of the
990                        // allocator's answer and this receiver's own per-track
991                        // bandwidth demand. The allocator replaces nothing in
992                        // `DownlinkDemand`: the budget decides what this receiver may
993                        // ASK for across every tile, the per-track decider observes
994                        // what one track actually GETS, and the more constrained wins.
995                        // A budget cannot know a single track is wedged; a per-track
996                        // observation cannot know 31 healthy tracks are 31 too many.
997                        //
998                        // Debounce the off⇄low⇄source decision so a transient
999                        // tile-resize flap (a rejoin relayout, a fast scroll through a
1000                        // virtualized grid) can't thrash the subscription on the
1001                        // shared audio+video connection — only a DURABLE layer change
1002                        // actuates. Directional: the constants below gate DOWNGRADES;
1003                        // an upgrade (the user expanded a tile, or scrolled one back
1004                        // into view) commits after a short flap-absorbing window and
1005                        // ignores the dwell. A cap-value change on the same layer
1006                        // passes through — which is why the allocator emits RUNGS and
1007                        // not raw bitrates: a budget that drifts by a few kbps lands
1008                        // on the same rung and nothing downstream moves at all.
1009                        let req = alloc.map(|allocated| {
1010                            let raw_req = hound_media::most_constrained(allocated, bwe_req);
1011                            hysteresis
1012                                .entry((peer, source))
1013                                .or_insert_with(LayerHysteresis::new)
1014                                .observe(raw_req, now, LAYER_SWITCH_DEBOUNCE, LAYER_SWITCH_DWELL)
1015                        });
1016                        // The COMMITTED request, not the allocator's raw answer, is
1017                        // what the UI is told about: while the hysteresis is still
1018                        // holding a cut back the old subscription is live and frames
1019                        // are arriving, and flipping the tile to its avatar then would
1020                        // blank a picture that is still being painted. So the flag
1021                        // tracks exactly the state "this tile subscribes to nothing".
1022                        if matches!(req, Some(hound_media::QualityRequest::Hidden(_))) {
1023                            hidden_now.insert((peer.0, source));
1024                        }
1025                        if let Some(deadline) = hysteresis.get(&(peer, source)).and_then(|h| {
1026                            h.pending_deadline(LAYER_SWITCH_DEBOUNCE, LAYER_SWITCH_DWELL)
1027                        }) {
1028                            next_switch_deadline = Some(
1029                                next_switch_deadline.map_or(deadline, |d: Instant| d.min(deadline)),
1030                            );
1031                        }
1032                        // Report our cap upstream (against the SOURCE track) on change,
1033                        // so the producer activates / tears down the shared low layer.
1034                        // A tile that has reported nothing yet reports nothing upstream
1035                        // either — an absent request leaves the producer's fold to the
1036                        // receivers that have spoken.
1037                        if let (Some(sid), Some(req)) = (src_id, req) {
1038                            // `hidden` rides alongside the cap rather than replacing
1039                            // it: the producer needs the cap to size a layer when
1040                            // EVERY requester is hidden, and needs the flag to keep
1041                            // this one out of the fold when any of them is not.
1042                            let (cap_bps, hidden) = match req {
1043                                hound_media::QualityRequest::Capped(b) => (Some(b), false),
1044                                hound_media::QualityRequest::Hidden(b) => (Some(b), true),
1045                                hound_media::QualityRequest::Source => (None, false),
1046                            };
1047                            if last_sent.get(&(peer, source)).map(|&(_, c)| c)
1048                                != Some((cap_bps, hidden))
1049                            {
1050                                last_sent.insert((peer, source), (sid, (cap_bps, hidden)));
1051                                tracing::info!(
1052                                    peer = peer.0,
1053                                    ?source,
1054                                    ?cap_bps,
1055                                    hidden,
1056                                    src_track = sid,
1057                                    low_present = low_id.is_some(),
1058                                    budget_kbps = budget / 1000,
1059                                    "simulcast[recv]: RequestQuality → producer"
1060                                );
1061                                let _ = control_outbound.send(ClientMessage::RequestQuality {
1062                                    track: hound_control::TrackId(sid),
1063                                    cap_bps,
1064                                    hidden,
1065                                });
1066                            }
1067                        }
1068                        // The bounded escalation gate: while a tile is waiting for its
1069                        // own first report or for the producer's `_l0` decl, it
1070                        // subscribes to NOTHING rather than pulling the source it
1071                        // cannot use. Never blanks a live tile — see [`SourceHold`].
1072                        let (source_ok, hold_deadline) = {
1073                            let mut policies = lock_policies(&tile_policies);
1074                            let hold = &mut policies.entry((peer, source)).or_default().hold;
1075                            let ok = hold.observe(
1076                                HoldInputs {
1077                                    demand_known: req.is_some(),
1078                                    wants_low: matches!(
1079                                        req,
1080                                        Some(hound_media::QualityRequest::Capped(_))
1081                                    ),
1082                                    low_present: low_id.is_some(),
1083                                    on_source: src_id.is_some_and(|id| active.contains_key(&id)),
1084                                    seeded: g.seeded,
1085                                },
1086                                now,
1087                                SOURCE_HOLD_GRACE,
1088                            );
1089                            (ok, hold.pending_deadline(SOURCE_HOLD_GRACE))
1090                        };
1091                        if let Some(deadline) = hold_deadline {
1092                            next_switch_deadline = Some(
1093                                next_switch_deadline.map_or(deadline, |d: Instant| d.min(deadline)),
1094                            );
1095                        }
1096                        if let Some(id) = pick_layer(req, src_id, low_id, source_ok) {
1097                            // Priced against the track actually taken, not against the
1098                            // request: the `SourceHold` waiver can put a `Capped` tile
1099                            // on the SOURCE when its producer never published `_l0`, and
1100                            // a committed-spend figure that missed that would be the one
1101                            // number the controller must not be able to lie about.
1102                            committed_bps += match req {
1103                                Some(r) if Some(id) == low_id => {
1104                                    hound_media::request_bps(vsource, r)
1105                                }
1106                                _ => hound_media::source_bps(vsource),
1107                            };
1108                            out.push((
1109                                id,
1110                                peer,
1111                                MediaKind::Video,
1112                                source,
1113                                Some(id) == low_id,
1114                                g.user_id,
1115                            ));
1116                        }
1117                    }
1118                    in_force = hound_media::BudgetObservation {
1119                        deficient: hound_media::downlink_deficient(&deliveries),
1120                        committed_bps,
1121                        ..Default::default()
1122                    };
1123                } else {
1124                    // Video is paused: nothing is subscribed, so the evidence about the
1125                    // last allocation is not stale, it is GONE. Carrying a deficiency
1126                    // across a pause would cut the budget on the first pass after the
1127                    // resume for a link nobody was using (`VideoBudget::observe` clears
1128                    // its own clocks over the same gap).
1129                    in_force = hound_media::BudgetObservation::default();
1130                }
1131                out
1132            };
1133            // Tell the UI which tiles have no subscription, so each renders its
1134            // owner's avatar rather than a texture nobody is feeding. Every pass, not
1135            // only on change: `publish` compares before it notifies, and re-patching
1136            // repairs a `CallView` the control reader rebuilt from a snapshot taken
1137            // just before this pass's store. A PAUSED call publishes the empty set —
1138            // nothing is hidden by policy there, the whole surface is away.
1139            hidden_tiles.publish(&hidden_now);
1140
1141            // Stop pipelines for tracks no longer present (or video we just paused).
1142            let desired_ids: std::collections::HashSet<u32> =
1143                desired.iter().map(|(id, ..)| *id).collect();
1144            let gone: Vec<u32> = active
1145                .keys()
1146                .copied()
1147                .filter(|id| !desired_ids.contains(id))
1148                .collect();
1149            for id in gone {
1150                if let Some(sub) = active.remove(&id) {
1151                    sub.stop();
1152                    // Tell the relay to stop forwarding this track to us (dropping the
1153                    // subscriber closes the moq sub; this also clears server routing).
1154                    let _ = control_outbound.send(ClientMessage::Unsubscribe {
1155                        track: hound_control::TrackId(id),
1156                    });
1157                }
1158            }
1159
1160            // Start pipelines for new tracks.
1161            for (track_id, peer, kind, source, low, user_id) in desired {
1162                if active.contains_key(&track_id) {
1163                    continue;
1164                }
1165                // A recently-died pipeline restarts PACED (bounded retry): defer this
1166                // track and wake when its pace elapses.
1167                if let Some(at) = pacer.ready_at(track_id, Instant::now()) {
1168                    next_switch_deadline =
1169                        Some(next_switch_deadline.map_or(at, |d: Instant| d.min(at)));
1170                    continue;
1171                }
1172                // Tell the server we want this track (+ a keyframe to bootstrap video).
1173                let _ = control_outbound.send(ClientMessage::Subscribe {
1174                    track: hound_control::TrackId(track_id),
1175                });
1176                if kind == MediaKind::Video {
1177                    tracing::info!(
1178                        peer = peer.0,
1179                        ?source,
1180                        low,
1181                        track = track_id,
1182                        "simulcast[recv]: subscribing video track ({})",
1183                        if low { "LOW" } else { "source" }
1184                    );
1185                    let _ = control_outbound.send(ClientMessage::RequestKeyframe {
1186                        track: hound_control::TrackId(track_id),
1187                    });
1188                    // RTC-debug: a PLI we sent for this peer's camera vs screen track.
1189                    stats
1190                        .peer(peer.0)
1191                        .video_keyframe_requested(inbound_video_kind(source));
1192                }
1193                let epoch = next_epoch;
1194                next_epoch += 1;
1195                let sub = start_pipeline(
1196                    peer,
1197                    user_id,
1198                    kind,
1199                    source,
1200                    low,
1201                    space_id,
1202                    relay.clone(),
1203                    sink.clone(),
1204                    playout_bus.clone(),
1205                    levels.clone(),
1206                    clock,
1207                    stats.clone(),
1208                    control_outbound.clone(),
1209                    track_id,
1210                    volumes.clone(),
1211                    PipelineEndGuard {
1212                        track_id,
1213                        epoch,
1214                        tx: ended_tx.clone(),
1215                    },
1216                );
1217                active.insert(track_id, sub);
1218            }
1219
1220            // Re-reconcile when the roster changes, the pause flag flips, the tile
1221            // demand changes, a debounced layer switch comes due (so a held switch
1222            // actuates even if nothing else moves), OR on the floor tick — the
1223            // per-frame signals (downlink demand, audio liveness) have no event.
1224            let switch_wait = reconcile_wait(next_switch_deadline, Instant::now());
1225            tokio::select! {
1226                r = view_rx.changed() => {
1227                    if r.is_err() { break; }
1228                }
1229                r = video_paused.changed() => {
1230                    if r.is_err() { break; }
1231                }
1232                // No `Err` arm, unlike its sibling wakes: this loop holds a SENDER
1233                // for the same channel (it prunes departed peers' demand), so the
1234                // "all senders dropped" error is unreachable by construction. The
1235                // loop's exit paths are the roster/pause watches closing and the
1236                // supervisor's abort on teardown.
1237                _ = tile_changed.changed() => {}
1238                // A pipeline ended on its own (end-of-track / transport error): REAP
1239                // it so the reconcile re-subscribes — leaving it in `active` would
1240                // freeze that tile permanently (every reconcile would skip it as
1241                // "already subscribed"). Epoch-guarded: a stale signal from a
1242                // pipeline the loop itself stopped (and maybe already replaced)
1243                // must not touch the healthy replacement.
1244                Some((track_id, epoch)) = ended_rx.recv() => {
1245                    if active.get(&track_id).is_some_and(|s| s.epoch == epoch) {
1246                        let sub = active.remove(&track_id).expect("checked present");
1247                        let delivered = sub.delivered.load(Ordering::Relaxed);
1248                        // This tile HAD a subscription, so the evidence the source
1249                        // gate waits for is already in: credit it before the
1250                        // re-subscribe, or a self-healing restart would pay the
1251                        // restart ladder AND a fresh `SOURCE_HOLD_GRACE` of black.
1252                        if let Some((key, low)) = sub.tile {
1253                            lock_policies(&tile_policies)
1254                                .entry(key)
1255                                .or_default()
1256                                .subscription_reaped(low);
1257                        }
1258                        pacer.on_death(track_id, delivered, sub.started.elapsed(), Instant::now());
1259                        tracing::warn!(
1260                            track = track_id,
1261                            lived_ms = sub.started.elapsed().as_millis() as u64,
1262                            delivered,
1263                            "call: pipeline ended on its own — reaping + re-subscribing"
1264                        );
1265                    }
1266                }
1267                _ = tokio::time::sleep(switch_wait) => {}
1268            }
1269        }
1270
1271        for (_, sub) in active.drain() {
1272            sub.stop();
1273        }
1274        // The set describes an allocation that no longer exists. Left standing, a
1275        // re-establish's fresh peer ids would inherit the dead session's flags until
1276        // its first reconcile pass — and a teardown that is not followed by one (a
1277        // leave) would leave every cut tile showing an avatar in the next call's
1278        // first frames.
1279        hidden_tiles.publish(&HashSet::new());
1280    })
1281}

1283/// How often the receiver re-sends `RequestKeyframe` while a fresh video
1284/// subscription is still awaiting its first keyframe, and how many times. The
1285/// subscribe-time request can RACE the producer's PLI-forced IDR (the IDR lands in
1286/// a group the moq live-join skips), leaving the receiver black indefinitely on a
1287/// keyframe that never comes. With the screen encoder
1288/// on an on-demand cadence (no frequent periodic IDR to paper over the race), this
1289/// bounded re-request is the fast recovery: producer-side the keyframe throttle
1290/// coalesces duplicates, so nagging is cheap. Once the attempts are exhausted the
1291/// pipeline's [`FIRST_KEYFRAME_DEADLINE`] ends it — a fresh, ladder-paced
1292/// resubscribe replaces a black tile that would otherwise wait on the safety-net IDR.
1293const KEYFRAME_NAG_INTERVAL: Duration = Duration::from_secs(2);
1294const KEYFRAME_NAG_ATTEMPTS: u32 = 5;
1295
1296/// How long a fresh pipeline waits for its MoQ broadcast to be ANNOUNCED before
1297/// giving up. `subscribe_track` awaits the announce with no deadline of its own,
1298/// and a task parked there never drops its [`PipelineEndGuard`] — so the reap +
1299/// restart machinery written for exactly this case never runs and the tile / peer
1300/// stays dead for the rest of the establish (a producer whose `publish_track`
1301/// failed, or a broadcast parked as a same-path backup behind a zombie). Bounded
1302/// in the same spirit as video's `FIRST_KEYFRAME_DEADLINE` (12 s), and slightly
1303/// under it: ending here yields a fresh announce cursor plus a re-sent
1304/// control-plane `Subscribe`, which is the only thing that heals a zombie.
1305const SUBSCRIBE_ANNOUNCE_DEADLINE: Duration = Duration::from_secs(10);
1306
1307/// Start one per-track decode pipeline. Audio → Opus decode → mix bus; video →
1308/// H.264/etc decode → sink keyed `cam:{peer}` / `screen:{peer}`. For video,
1309/// `control_outbound` + `track_id` drive the awaiting-keyframe re-request nag.
1310#[allow(clippy::too_many_arguments)]
1311fn start_pipeline(
1312    peer: PeerId,
1313    user_id: Uuid,
1314    kind: MediaKind,
1315    source: Source,
1316    low: bool,
1317    space_id: Uuid,
1318    relay: Arc<MoqRelaySession>,
1319    sink: Arc<dyn VideoSink>,
1320    playout_bus: Arc<PlayoutBus>,
1321    levels: watch::Sender<LevelsView>,
1322    clock: CaptureClock,
1323    stats: Arc<CallStats>,
1324    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
1325    track_id: u32,
1326    volumes: Arc<StreamVolumes>,
1327    end_guard: PipelineEndGuard,
1328) -> Subscription {
1329    let epoch = end_guard.epoch;
1330    let stop = Arc::new(AtomicBool::new(false));
1331    let task_stop = stop.clone();
1332    // First-delivery latch: audio sets it on the first decoded frame, video when
1333    // the first keyframe passes the gate. Read at reap time to pick the restart
1334    // pace (an undelivered death climbs the ladder).
1335    let delivered = Arc::new(AtomicBool::new(false));
1336    let task_delivered = delivered.clone();
1337    // This pipeline's downlink-driven demand (video only). A FRESH cell every
1338    // call — the whole reset-on-restart contract for `DownlinkDemand` is "a new
1339    // pipeline gets a new cell", so a resubscribe / layer switch / reap can never
1340    // carry a stale capped demand forward.
1341    let downlink = (kind == MediaKind::Video).then(|| Arc::new(DownlinkCell::new()));
1342    let task_downlink = downlink.clone();
1343    // Audio liveness counter (audio only): the pipeline bumps it per frame pulled
1344    // off the wire and the reconcile's [`StallWatch`] samples it. The unbounded
1345    // audio pull has no other out-of-band evidence that the track is alive.
1346    let arrivals = (kind == MediaKind::Audio).then(|| Arc::new(AtomicU64::new(0)));
1347    let task_arrivals = arrivals.clone();
1348    // The per-peer inbound stats handle (cloned once; the pipeline increments on it
1349    // lock-free per frame).
1350    let peer_stats = stats.peer(peer.0);
1351    // The moq track name carries the spatial layer: a constrained receiver subscribes
1352    // to the `_l0` low track instead of the source.
1353    let track_name = match (kind, source, low) {
1354        (MediaKind::Audio, Source::Screen, _) => SCREEN_AUDIO_TRACK,
1355        (MediaKind::Audio, _, _) => MIC_TRACK,
1356        (MediaKind::Video, Source::Screen, false) => SCREEN_TRACK,
1357        (MediaKind::Video, Source::Screen, true) => "screen_l0",
1358        (MediaKind::Video, _, false) => CAM_TRACK,
1359        (MediaKind::Video, _, true) => "cam_l0",
1360    };
1361    let path = broadcast_path(space_id, user_id, peer, track_name);
1362
1363    // Audio decode + fill runs on the RT-promoted audio runtime so video CPU can't
1364    // starve it; video stays on the general call runtime.
1365    let runtime = match kind {
1366        MediaKind::Audio => audio_runtime(),
1367        MediaKind::Video => call_runtime(),
1368    };
1369    let task = runtime.spawn(async move {
1370        // Owned by the task so its Drop reports this pipeline's end to the reap
1371        // channel on ANY exit path (clean end, error, panic, abort).
1372        let _end_guard = end_guard;
1373        // Bounded: the announce wait is the ONE await whose non-completion defeats
1374        // the end guard (see [`SUBSCRIBE_ANNOUNCE_DEADLINE`]). Nothing is half-built
1375        // yet, so dropping the future here is clean.
1376        let subscribing = relay.subscribe_track(
1377            &path,
1378            track_name,
1379            peer.0 as u16,
1380            priority_for(kind),
1381            // Audio drains in arrival order (a reordered Opus frame is still playable);
1382            // video priority-drops stale GOPs. This is what stops a screenshare from
1383            // silently dropping reordered mic/app-audio frames on the shared connection.
1384            kind == MediaKind::Audio,
1385        );
1386        let subscriber = match tokio::time::timeout(SUBSCRIBE_ANNOUNCE_DEADLINE, subscribing).await
1387        {
1388            Ok(Ok(s)) => s,
1389            Ok(Err(e)) => {
1390                tracing::warn!(%e, peer = peer.0, "subscribe failed");
1391                return;
1392            }
1393            Err(_) => {
1394                tracing::warn!(
1395                    peer = peer.0,
1396                    track = track_id,
1397                    %path,
1398                    "subscribe: broadcast never announced within the deadline — ending \
1399                     pipeline for a paced re-subscribe"
1400                );
1401                return;
1402            }
1403        };
1404
1405        match kind {
1406            MediaKind::Audio => {
1407                // This stream's playback-gain cell (the tile volume/mute control),
1408                // looked up by key so a set volume survives resubscribes.
1409                let gain = volumes.cell(peer.0, source);
1410                audio_pipeline(
1411                    subscriber,
1412                    peer,
1413                    source,
1414                    playout_bus,
1415                    levels,
1416                    clock,
1417                    task_stop,
1418                    peer_stats,
1419                    gain,
1420                    task_delivered,
1421                    // `.expect`: this arm is MediaKind::Audio, where `arrivals` is
1422                    // always `Some` (see its construction above).
1423                    task_arrivals.expect("audio pipeline always has an arrival counter"),
1424                )
1425                .await
1426            }
1427            MediaKind::Video => {
1428                // Awaiting-keyframe re-request nag: a separate small task so the
1429                // fast re-requests don't race the pipeline's own pull. Ends on
1430                // first keyframe or pipeline stop; after the bounded attempts the
1431                // pipeline's first-keyframe deadline takes over (it ends itself,
1432                // and the reconcile re-subscribes on the restart ladder).
1433                let nag = {
1434                    let got = task_delivered.clone();
1435                    let stop = task_stop.clone();
1436                    let outbound = control_outbound.clone();
1437                    call_runtime().spawn(async move {
1438                        for _ in 0..KEYFRAME_NAG_ATTEMPTS {
1439                            tokio::time::sleep(KEYFRAME_NAG_INTERVAL).await;
1440                            if got.load(Ordering::Relaxed) || stop.load(Ordering::Relaxed) {
1441                                return;
1442                            }
1443                            tracing::info!(
1444                                peer = peer.0,
1445                                track = track_id,
1446                                "video: still awaiting first keyframe — re-requesting"
1447                            );
1448                            let _ = outbound.send(ClientMessage::RequestKeyframe {
1449                                track: hound_control::TrackId(track_id),
1450                            });
1451                        }
1452                    })
1453                };
1454                video_pipeline(
1455                    subscriber,
1456                    peer,
1457                    source,
1458                    sink,
1459                    task_stop,
1460                    peer_stats,
1461                    task_delivered,
1462                    control_outbound,
1463                    track_id,
1464                    // `.expect`: this arm is MediaKind::Video, where `downlink` is
1465                    // always `Some` (see its construction above).
1466                    task_downlink.expect("video pipeline always has a downlink cell"),
1467                    source_top_bps(source),
1468                )
1469                .await;
1470                nag.abort();
1471            }
1472        }
1473    });
1474
1475    let started = Instant::now();
1476    Subscription {
1477        stop,
1478        task,
1479        epoch,
1480        started,
1481        delivered,
1482        downlink,
1483        audio: arrivals.map(|arrivals| AudioLiveness {
1484            arrivals,
1485            watch: StallWatch::new(started),
1486        }),
1487        tile: (kind == MediaKind::Video).then_some(((peer, source), low)),
1488    }
1489}

1495    match kind {
