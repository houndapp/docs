1//! The out-of-call microphone monitor — what the settings screen's mic test runs.
2//!
3//! One blocking capture thread walking the same open → APM → meter path the call's
4//! mic publish walks ([`super::audio_tx::spawn_mic_publish`]), with everything
5//! downstream of the meter removed: no Opus, no MoQ, no OS input-volume controller
6//! (a preview must not leave the user's OS mic slider somewhere else than it found
7//! it). The level is published onto the SAME [`LevelsView`] channel the in-call
8//! meter reads, so both meters are one widget over one stream.
9//!
10//! **The call owns the microphone.** A monitor is refused while a call is live:
11//! two opens of one input device is either an outright `EBUSY` (ALSA's exclusive
12//! hw devices) or two APMs adapting against each other on the same signal. The
13//! refusal lives at the command seam ([`crate::HoundClient::mic_monitor_start`]);
14//! this module assumes it has the device to itself.
15//!
16//! Voice processing is applied through the very [`VoiceProcessing`] cell the call
17//! uses, so "noise suppression off" reaches the meter by the same code path it
18//! reaches a call — which is the whole point of a test that is supposed to sound
19//! like the call will.

38/// How long the loop waits before re-opening after the device faults mid-test.
39/// Slower than the call's ladder on purpose: a settings preview has no far end
40/// waiting on it, and the user is looking straight at the meter — a quiet second
41/// is cheaper than a device thrash.
42const REOPEN_DELAY: Duration = Duration::from_secs(1);
43
44/// A running mic monitor. Dropping it stops the capture thread and blanks the
45/// meter, so the monitor's lifetime *is* the `Option` that holds it.
46pub(crate) struct MicMonitor {
47    stop: Arc<AtomicBool>,
48    /// The live device pick — the same hot-swap seam the call's capture loop has,
49    /// so changing the input device in settings re-opens under the running test
50    /// rather than needing it restarted.
51    device: watch::Sender<DeviceChoice<MicId>>,
52    /// The voice-processing intent + the live APM it applies to. Owned here (not
53    /// shared with a call) because a monitor and a call never coexist.
54    voice: Arc<VoiceProcessing>,
55    levels: watch::Sender<LevelsView>,
56    /// The capture thread, joined on drop. Holding it is what lets "stopped"
57    /// mean the device is RELEASED rather than merely asked to stop — an
58    /// exclusive input (ALSA hw) refuses the next open while the old handle
59    /// lives, and the next open is a device the user is watching a meter for.
60    thread: Option<std::thread::JoinHandle<()>>,
61}
62
63impl MicMonitor {
64    /// Open `mic_id` (`None` = host default) and start metering into `levels`.
65    ///
66    /// The open happens ON the capture thread — the engine's `Microphone` is
67    /// `!Send`, so it must be constructed, driven and dropped there — and its
68    /// result is awaited here. That is what makes a denied or missing device an
69    /// error the settings screen can render, rather than a meter that silently
70    /// never moves.
71    pub(crate) async fn start(
72        mic_id: Option<String>,
73        levels: watch::Sender<LevelsView>,
74    ) -> Result<Self> {
75        let stop = Arc::new(AtomicBool::new(false));
76        let device = watch::channel(choice_of(mic_id)).0;
77        let voice = Arc::new(VoiceProcessing::default());
78        // Built here rather than on the thread so `attach` — which re-asserts the
79        // stored intent onto a fresh APM — runs before the first frame is processed.
80        let apm = Engine::new().audio_processor().map(Arc::new);
81        voice.attach(apm.clone());
82
83        let (opened_tx, opened_rx) = oneshot::channel();
84        let thread_stop = stop.clone();
85        let thread_levels = levels.clone();
86        let device_rx = device.subscribe();
87        let thread = std::thread::Builder::new()
88            .name("mic-monitor".into())
89            .spawn(move || {
90                monitor_loop(thread_stop, apm, device_rx, thread_levels, opened_tx);
91            })
92            .map_err(|e| {
93                HoundError::internal(
94                    InternalCode::MicMonitorThread,
95                    format!("mic monitor thread: {e}"),
96                )
97            })?;
98
99        match opened_rx.await {
100            Ok(Ok(())) => Ok(Self {
101                stop,
102                device,
103                voice,
104                levels,
105                thread: Some(thread),
106            }),
107            Ok(Err(e)) => Err(open_error(&e)),
108            // The thread vanished without reporting — only possible if it panicked
109            // before the first open. Report it as an open failure rather than
110            // leaving the UI with a monitor that is not running.
111            Err(_) => Err(HoundError::media(
112                MediaCode::MicrophoneOpen,
113                "mic monitor thread exited before opening the microphone",
114            )),
115        }
116    }
117
118    /// Re-point the running test at another input device (`None` = host default).
119    pub(crate) fn select_microphone(&self, id: Option<String>) {
120        let _ = self.device.send(choice_of(id));
121    }
122
123    /// Set the noise-suppression mode (0 = off, 1 = moderate, 2 = high).
124    pub(crate) fn set_noise_suppression(&self, mode: u8) {
125        self.voice.set_ns_mode(mode);
126    }
127
128    /// Toggle automatic gain control.
129    pub(crate) fn set_agc_enabled(&self, enabled: bool) {
130        self.voice.set_agc(enabled);
131    }
132
133    /// Set the manual input trim in dB (clamped ±12 by the APM).
134    pub(crate) fn set_input_gain_db(&self, gain_db: f32) {
135        self.voice.set_input_gain_db(gain_db);
136    }
137}

139impl Drop for MicMonitor {
140    fn drop(&mut self) {
141        self.stop.store(true, Ordering::Relaxed);
142        // The loop is parked in `next_frame` and only notices `stop` one frame
143        // later, so blank the meter here: the bar must never outlive the test that
144        // was driving it, the same invariant `silence_mic_meter` holds in a call.
145        self.levels.send_modify(|l| l.mic = 0.0);
146        // Then WAIT for the device to be handed back (see `thread`). Bounded by
147        // one capture frame — the loop's only blocking point — but a blocking
148        // wait all the same, which is why every drop site takes it off the async
149        // worker (`HoundClient::release_mic_monitor`).
150        if let Some(thread) = self.thread.take() {
151            let _ = thread.join();
152        }
153    }
154}

156fn choice_of(id: Option<String>) -> DeviceChoice<MicId> {
157    match id {
158        Some(id) => DeviceChoice::Named(MicId::from(id)),
159        None => DeviceChoice::Default,
160    }
161}

163/// Map an engine open failure onto the typed surface, splitting the one class the
164/// user can act on (a TCC denial) from everything else.
165fn open_error(e: &AudioError) -> HoundError {
166    match OpenFailure::of_audio(e) {
167        OpenFailure::PermissionDenied => HoundError::media(
168            MediaCode::DevicePermissionDenied,
169            "microphone permission denied — enable it in System Settings → \
170             Privacy & Security → Microphone",
171        ),
172        OpenFailure::Other => {
173            HoundError::media(MediaCode::MicrophoneOpen, format!("microphone open: {e}"))
174        }
175    }
176}

178/// The monitor's capture loop: pull PCM, process it through the APM, meter it.
179///
180/// `opened` reports the FIRST open's outcome to [`MicMonitor::start`] and is
181/// consumed by it; a first open that fails ends the loop (there is nothing to
182/// preview and the caller is being told why), while a later one — the device was
183/// unplugged mid-test — retries, because the user can plug it back in while
184/// watching.
185fn monitor_loop(
186    stop: Arc<AtomicBool>,
187    apm: Option<Arc<Apm>>,
188    mut device_rx: watch::Receiver<DeviceChoice<MicId>>,
189    levels: watch::Sender<LevelsView>,
190    opened: oneshot::Sender<std::result::Result<(), AudioError>>,
191) {
192    // The monitor's own timeline: nothing else stamps against it (there is no
193    // playout to stay in sync with), it exists because the capture resampler needs
194    // a clock.
195    let clock = CaptureClock::start();
196    let mut choice = device_rx.borrow_and_update().clone();
197    let mut opened = Some(opened);
198    let mut mic: Option<Microphone> = None;
199
200    while !stop.load(Ordering::Relaxed) {
201        if device_rx.has_changed().unwrap_or(false) {
202            choice = device_rx.borrow_and_update().clone();
203            drop(mic.take());
204        }
205        if mic.is_none() {
206            match Engine::new().open_microphone(&choice, clock) {
207                Ok(m) => {
208                    tracing::debug!(
209                        device_rate = m.device_rate(),
210                        "mic monitor: capture device opened"
211                    );
212                    if let Some(tx) = opened.take() {
213                        let _ = tx.send(Ok(()));
214                    }
215                    mic = Some(m);
216                }
217                Err(e) => {
218                    if let Some(tx) = opened.take() {
219                        let _ = tx.send(Err(e));
220                        return;
221                    }
222                    tracing::warn!(%e, "mic monitor: reopen failed — retrying");
223                    levels.send_modify(|l| l.mic = 0.0);
224                    std::thread::sleep(REOPEN_DELAY);
225                    continue;
226                }
227            }
228        }
229        let Some(cap) = mic.as_mut() else { continue };
230        // Same guard as the call's capture loop: a panicking device backend must
231        // leave a blank meter and a reopen, never a thread that is simply gone and
232        // a bar frozen at whatever it last read.
233        match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| cap.next_frame())) {
234            Ok(Ok(mut frame)) => {
235                if let Some(apm) = apm.as_ref() {
236                    apm.process_capture(&mut frame);
237                }
238                // Metered POST-processing, exactly as the call meters it: the test
239                // has to show what the far end would hear, gain and gate included.
240                levels.send_modify(|l| l.mic = level_from_rms(frame.rms()));
241            }
242            _ => {
243                drop(mic.take());
244                levels.send_modify(|l| l.mic = 0.0);
245                std::thread::sleep(REOPEN_DELAY);
246            }
247        }
248    }
249    levels.send_modify(|l| l.mic = 0.0);
250}
