1//! Sender-side BWE/congestion adaptation. The camera +
2//! screen video the client PUBLISHES adapts its bitrate/resolution to network
3//! conditions and serves keyframes on demand. The brain —
4//! [`hound_media::VideoAdaptation`] — is shared; this is the local IO glue:
5//!
6//! * [`MediaFeedback`] — the in-process form of `ServerMessage::MediaFeedbackReceived`
7//!   fanned out of the control reader to the per-track feedback listeners.
8//! * [`AdaptShared`] — the atomics seam between the async 1 s adaptation TICK (owns
9//!   the `VideoAdaptation` core, reads `conn.rtt()`) and the `!Send` capture/encode
10//!   thread (owns the encoder, applies the decision per frame). One writer per field.
11//! * [`apply_adaptation_tick`] — the pure glue: drain backpressure + a pending PLI +
12//!   the worst receiver-reported loss, advance the core, write the decision back.
13//! * [`drain_keyframe_request`] — the PLI→IDR transition, run BOTH the instant a
14//!   `RequestKeyframe` arrives (the fast path) and on the tick (the safety net), so
15//!   a subscribing receiver never waits a tick for its opening keyframe.
16//! * [`SendCoordinator`] — the shared-uplink referee. Camera and screen publish on
17//!   ONE QUIC connection but adapt independently; when both are live they draw their
18//!   ceilings from a single [`hound_media::SendBudget`] instead of summing past the
19//!   wire. Inert with one publisher.
20//! * [`spawn_keyframe_listener`] / [`spawn_feedback_listener`] — the producer side of
21//!   the PLI relay + the receiver→producer BWE loop, each FILTERED to OUR track of a
22//!   given `Source` (Camera vs Screen) via the `SessionView`.
23//!
24//! With good RTT + no loss the ladder sits at the top rung, so a healthy network
25//! is byte-identical to the pre-adaptation publish path.

29/// Open an encoder at `target`, falling back DOWN the ladder
30/// ([`hound_engine::types::ENCODE_PRIORITY`]: AV1 → H.265 → H.264) to the first codec
31/// that opens. Safe to descend without re-checking decode caps: the negotiated
32/// `target` means every peer decodes it, and real decode hardware is monotonic
33/// (AV1-capable ⇒ H.265-capable ⇒ H.264), so every lower rung is decodable too.
34/// H.264 is the floor, so this normally returns `None` only if even H.264 won't
35/// open — which is no longer unreachable: on a device with no HW H.264, the SW
36/// codec is Cisco's downloaded binary, so before that download lands (or with the
37/// codec switched off in settings) there is nothing to open. The caller treats
38/// `None` as "no video this attempt", which is the honest outcome.
39pub(super) fn open_encoder_laddered(
40    target: VideoCodec,
41    dims: Dimensions,
42    fps: Fps,
43    bitrate: Bitrate,
44    cadence: KeyframeCadence,
45) -> Option<hound_engine::api::VideoEncoder> {
46    use hound_engine::types::ENCODE_PRIORITY;
47    let start = ENCODE_PRIORITY
48        .iter()
49        .position(|&c| c == target)
50        .unwrap_or(0);
51    for &c in &ENCODE_PRIORITY[start..] {
52        if let Ok(e) = Engine::new().video_encoder(c, dims, fps, bitrate, cadence) {
53            return Some(e);
54        }
55    }
56    None
57}

59/// Downscale a capture frame to the simulcast **low** rung's geometry, in whatever
60/// form `low_enc` consumes most cheaply.
61///
62/// A hardware encoder takes a platform surface directly, so the scale runs in the
63/// capture layout straight into a pooled platform buffer and the encoder packs
64/// nothing (Apple: NV12→NV12 into an IOSurface). A software encoder needs planar
65/// I420 regardless, so it keeps the portable scale — as does any platform or frame
66/// with no surface behind it, where `scaled_to_surface` returns `None` and the
67/// fallback runs.
68///
69/// `None` only if *both* paths fail, which the caller treats as "skip this frame".
70/// The source rung is never affected either way.
71pub(super) fn low_rung_frame(
72    frame: &hound_engine::types::RawFrame,
73    low_enc: &hound_engine::api::VideoEncoder,
74    dims: Dimensions,
75) -> Option<hound_engine::types::RawFrame> {
76    use hound_engine::backend::VideoEncode;
77
78    if low_enc.takes_surface_input()
79        && let Some(surface) = frame.scaled_to_surface(dims)
80    {
81        return Some(surface);
82    }
83    frame.scaled_to(dims)
84}

139/// The shared seam between the async **adaptation tick** (owns the sans-IO
140/// [`VideoAdaptation`] core + observes RTT) and the `!Send` **capture/encode thread**
141/// (owns the encoder, applies the decision per frame). Plain atomics — no lock
142/// crosses the thread/async boundary and each field has exactly one writer.
143///
144/// Data flow:
145/// * capture thread → tick: `produced` / `dropped` (per-tick backpressure).
146/// * tick → capture thread: `target_bps` (encoder retarget), `target_fps` (fps gate),
147///   `target_w/target_h` (the RAW per-tick rung geometry — RTC-debug telemetry
148///   only; never read by the capture thread), `encoder_w/encoder_h` (the
149///   DWELLED geometry — see [`hound_media::GeometryDwell`] — the capture thread
150///   rebuilds its encoder to match), `paused` (AudioOnly ⇒ skip encode/publish),
151///   `force_keyframe` (a throttle-approved IDR).
152/// * keyframe listener → publish loop: `pli_pending` + `pli_notify` (drained
153///   immediately, not on the tick); feedback listener → tick:
154///   `remote_loss_bps`/`remote_reported`.
155#[derive(Default)]
156pub(super) struct AdaptShared {
157    /// Encoded access units handed to the publisher since the last tick.
158    produced: AtomicU32,
159    /// Of those, how many were dropped at the bounded channel (publisher behind).
160    dropped: AtomicU32,
161    /// The encoder's current target bitrate (bits/sec); the capture thread retargets
162    /// only when it changes. `0` until the first decision.
163    pub(super) target_bps: AtomicU64,
164    /// The chosen rung's frame rate (fps); the capture thread downsamples to it.
165    /// `0` = no limit (no decision yet / AudioOnly — the publish path is skipped).
166    pub(super) target_fps: AtomicU32,
167    /// The chosen rung's RAW geometry (width/height) — RTC-debug telemetry only
168    /// (the panel's rung label); the capture thread never reads these. `0` = no
169    /// decision yet / AudioOnly.
170    target_w: AtomicU32,
171    target_h: AtomicU32,
172    /// The DWELLED encoder-rebuild geometry (see [`hound_media::GeometryDwell`]):
173    /// unlike `target_w`/`target_h`, this only moves once a candidate rung
174    /// geometry has held stable for [`hound_media::GEOMETRY_REBUILD_DWELL_US`],
175    /// so it never flaps with ordinary AIMD noise. The capture thread compares
176    /// this against the geometry its live encoder was actually built at and
177    /// rebuilds (+ forces a fresh IDR) when they differ. `0` until the first
178    /// tick runs — the capture thread must treat `0` as "no decision yet, keep
179    /// the current encoder" (never rebuild to a degenerate size).
180    pub(super) encoder_w: AtomicU32,
181    pub(super) encoder_h: AtomicU32,
182    /// Whether video is paused (the ladder chose AudioOnly). The capture thread keeps
183    /// capturing (self-preview, A/V-sync clock) but skips encode/publish.
184    pub(super) paused: AtomicBool,
185    /// Whether the SOURCE-rung publish is LIVE — i.e. [`drive_video_publish`]'s live
186    /// gate has opened (a moq subscriber attached, or control-plane demand was seen).
187    /// Latched on, exactly like the publish loop's own `live`: an `AdaptShared` is
188    /// per-establish, so a reconnect starts dark again with a fresh one.
189    ///
190    /// The capture thread reads this to skip the ENCODE while dark. Forwarding was
191    /// always gated (the publish loop discards pre-live frames), but the encode ran
192    /// regardless — a full H.264/AV1 encode per captured frame, every frame, with
193    /// nobody on the other end. Capture keeps running either way (self-preview must
194    /// never freeze); going live sets `force_keyframe` in the same step, so the first
195    /// encoded frame after the transition is an IDR and first-frame latency is one
196    /// capture interval, not a keyframe cadence.
197    live: AtomicBool,
198    /// A PLI arrived for our track (set by the keyframe listener); drained into the
199    /// keyframe throttle — which coalesces a storm — by [`drain_keyframe_request`].
200    pli_pending: AtomicBool,
201    /// Wakes [`drive_video_publish`] the moment `pli_pending` is set, so a
202    /// `RequestKeyframe` becomes a forced IDR immediately instead of at the next 1 Hz
203    /// adaptation tick. It is only a *wake-up*, never a bypass: the woken loop runs the
204    /// same [`drain_keyframe_request`] through the same throttle, and the tick's own
205    /// drain stays as the safety net for a request the throttle held off.
206    ///
207    /// Why this instead of sharing the throttle with the listener task: the
208    /// `VideoAdaptation` core (which owns the throttle) is exclusively owned by the
209    /// publish loop. Waking that owner keeps the single-writer discipline and needs no
210    /// lock across the async/thread boundary.
211    pli_notify: Notify,
212    /// Latched (never cleared) once ANY keyframe request named our track — proof a
213    /// receiver wants this video. The publish loop's live gate accepts it as a
214    /// fallback for the moq subscriber signal, which can be lost when a broadcast
215    /// is replaced on the same path (a share restart): every subscribe sends
216    /// `RequestKeyframe`, so demand always reaches us on the control plane even
217    /// when `used()` never fires.
218    pub(super) demand_seen: AtomicBool,
219    /// Worst receiver-reported loss fraction this interval, in basis points
220    /// (loss × 10_000), max-reduced across reporters. Read+cleared by the tick.
221    remote_loss_bps: AtomicU32,
222    /// Whether any receiver reported this interval — lets the tick pass `None` (no
223    /// remote signal) vs `Some(0.0)` (a receiver explicitly reported a clean leg).
224    remote_reported: AtomicBool,
225    /// A throttle-approved keyframe is due — the capture thread swap-clears this and
226    /// forces the encoder's next frame to an IDR.
227    pub(super) force_keyframe: AtomicBool,
228    /// The codec the encoder should be producing, as a [`codec_rank`] byte. Seeded
229    /// at publish from the roster pick and re-driven by the mid-call re-negotiator
230    /// ([`spawn_codec_renegotiator`]); the encode loop rebuilds its encoder when this
231    /// no longer matches what it built. Defaults to `0` = H.264 (the floor).
232    pub(super) target_codec: AtomicU8,
233}

235impl AdaptShared {
236    /// The SOURCE-rung encode gate, as the capture threads apply it: encode this frame
237    /// only if a receiver is actually attached (`live`) AND the ladder has not dropped
238    /// video (`paused`). Both are "keep capturing, skip encode" states — the device
239    /// stays open and self-preview keeps flowing, so resume costs one capture interval.
240    ///
241    /// The mirror of the mic's `wait_subscribed` (audio_tx) and the low rung's
242    /// `low_active` gate: nobody encodes into the void. It does NOT speak for the LOW
243    /// rung, whose demand signal (`low_active`) is independent — a receiver can want
244    /// only the low layer, and that layer must not wait on the source's gate.
245    ///
246    /// `live` is ACQUIRE: it synchronises with the Release store in [`Self::go_live`],
247    /// so a capture thread that observes the gate open is guaranteed to also observe
248    /// the armed `force_keyframe` written before it. Relaxed here would let aarch64
249    /// reorder the two and lose the opening IDR (see `go_live`).
250    pub(super) fn source_encode_open(&self) -> bool {
251        self.live.load(Ordering::Acquire) && !self.paused.load(Ordering::Relaxed)
252    }
253
254    /// The liveness half of [`Self::source_encode_open`] on its own — has a receiver
255    /// attached? Mirrored into the RTC-debug snapshot each tick so
256    /// [`crate::video_report`] can tell "dark by demand" (encode gated off, healthy)
257    /// from a dead encoder. Kept separate from the pause half because the snapshot
258    /// already carries the pause through the rung label.
259    pub(super) fn source_publish_live(&self) -> bool {
260        self.live.load(Ordering::Acquire)
261    }
262
263    /// Latch the source publish LIVE (a subscriber attached / control-plane demand was
264    /// seen) and arm the opening IDR. The two belong together: the gate opening is
265    /// exactly the moment the capture thread's next encode must be a keyframe, or the
266    /// publish loop (which only forwards from a keyframe) would sit on deltas until
267    /// the periodic IDR. Called only by [`drive_video_publish`].
268    ///
269    /// The ORDER is load-bearing, and it is a release/acquire pair rather than a
270    /// hopeful pair of relaxed stores. `force_keyframe` is armed FIRST; `live` is then
271    /// published with Release, and the capture threads read it with Acquire
272    /// ([`Self::source_encode_open`]). That makes "the IDR is armed" *happen-before*
273    /// "the gate is open", so a capture thread can never see the open gate without the
274    /// arming — the reordering aarch64 is free to do with two relaxed stores, which
275    /// would drop the first encoded frame's IDR and stall the stream until the
276    /// periodic keyframe.
277    fn go_live(&self) {
278        self.force_keyframe.store(true, Ordering::Relaxed);
279        self.live.store(true, Ordering::Release);
280    }
281
282    /// Record one receiver's reception report, max-reducing into this interval's worst
283    /// loss fraction (serve-the-worst-leg). `received`/`lost` are deltas since the
284    /// reporter's previous report. An empty report (`received + lost == 0`) is ignored.
285    fn record_remote_feedback(&self, received: u32, lost: u32) {
286        let total = received as u64 + lost as u64;
287        if total == 0 {
288            return;
289        }
290        // Quantize to basis points (loss × 10_000), rounding to nearest.
291        let bps = ((lost as u64 * 10_000 + total / 2) / total).min(10_000) as u32;
292        self.remote_reported.store(true, Ordering::Relaxed);
293        let mut cur = self.remote_loss_bps.load(Ordering::Relaxed);
294        while bps > cur {
295            match self.remote_loss_bps.compare_exchange_weak(
296                cur,
297                bps,
298                Ordering::Relaxed,
299                Ordering::Relaxed,
300            ) {
301                Ok(_) => break,
302                Err(observed) => cur = observed,
303            }
304        }
305    }
306
307    /// Drain this interval's worst receiver-reported loss fraction (and clear it).
308    /// `None` when no receiver reported — so the tick does not pin the loss-equivalent
309    /// to 0 and mask the local signals on a feedback-free interval.
310    fn take_remote_loss(&self) -> Option<f64> {
311        if self.remote_reported.swap(false, Ordering::Relaxed) {
312            Some(self.remote_loss_bps.swap(0, Ordering::Relaxed) as f64 / 10_000.0)
313        } else {
314            self.remote_loss_bps.store(0, Ordering::Relaxed);
315            None
316        }
317    }
318}

320/// Rate-limits the two "this frame produced nothing" WARN sites in
321/// [`encode_and_publish_adaptive`] (geometry-mismatch skip, encoder error). A lone
322/// failure still logs immediately at WARN; a PERSISTENT failure (encoder wedged,
323/// the geometry rebuild permanently behind) would otherwise repeat that same WARN
324/// once per captured frame — up to encode fps — without saying anything new. Once
325/// a streak is underway, only every [`Self::REMINDER_EVERY`]th failure re-alerts at
326/// WARN; the rest log at DEBUG. Reset by the next successful encode.
327#[derive(Default)]
328pub(super) struct EncodeFailStreak(u32);
329
330impl EncodeFailStreak {
331    /// ~5 s at 30 fps / ~10 s at 15 fps — often enough that a live run still shows
332    /// the failure is ongoing, rare enough that it reads as a heartbeat, not spam.
333    const REMINDER_EVERY: u32 = 150;
334
335    /// Record one failure; returns whether it should log at WARN (true) or DEBUG
336    /// (false).
337    fn bump(&mut self) -> bool {
338        self.0 += 1;
339        self.0 == 1 || self.0.is_multiple_of(Self::REMINDER_EVERY)
340    }
341
342    fn reset(&mut self) {
343        self.0 = 0;
344    }
345}

384// ─── Connection-level coordinated send budget ───────────────────────────────────
385//
386// The MoQ multiplex puts every published track on ONE QUIC connection, but each
387// video publisher runs its own rate controller that sees only its own receivers'
388// loss/RTT — never the connection's uplink saturation. Camera + a heavy screen share
389// therefore sum their ladders past the wire: publish channels drop, bufferbloat
390// builds, and our OWN downlink receive starves. This is the IO shell for
391// `hound_media`'s coordinator core: each publish loop registers a slot, reports its
392// UNCAPPED desire + its publish drops once per tick, and takes back its share of a
393// single AIMD budget. It is inert with one publisher — a lone camera or a lone share
394// is byte-identical to no coordination.
395
396/// Carved off the connection budget before the video split: audio is sacred, so it
397/// takes its reserve first and the video tracks divide only what is left. Opus at
398/// the call's profile plus packet/QUIC overhead fits well inside this.
399const AUDIO_RESERVE_BPS: u64 = 128_000;
400
401/// Where the coordinator's AIMD starts, and its ceiling. Deliberately optimistic —
402/// it only ever clamps DOWN from here toward the wire, so a fast uplink never pays
403/// for the coordinator existing (the desires stay under the budget and the split
404/// hands out the full ceiling); a slow one converges within a few seconds of
405/// sustained publish drops.
406const SEND_BUDGET_MAX_BPS: u64 = 12_000_000;
407
408/// The coordinator's floor: below this there is no useful video left to divide (both
409/// ladders have already collapsed toward AudioOnly on their own).
410const SEND_BUDGET_MIN_BPS: u64 = 600_000;
411
412/// Minimum spacing between coordinator advances. The two publish loops tick at
413/// [`ADAPT_TICK_INTERVAL`] on their OWN `CaptureClock`s, so whichever crosses this
414/// gate first drives the advance and the other reads the fresh allocation. Slightly
415/// under 1 s so ordinary tick jitter can't push the effective cadence out to 2 s
416/// (`SendBudget`'s own decrease cooldown still paces the back-off at ~1 Hz).
417const SEND_BUDGET_INTERVAL_US: i64 = 900_000;
418
419/// Which shared-connection video publisher a budget slot belongs to. Exactly two
420/// exist; the coordinator only bites when BOTH are registered *and* have reported.
421#[derive(Debug, Clone, Copy, PartialEq, Eq)]
422pub(super) enum SendTrack {
423    Camera = 0,
424    Screen = 1,
425}
426
427const SEND_TRACKS: usize = 2;
428
429/// The coordinator's pure state: the AIMD budget plus each slot's registration,
430/// latest desire and current cap. Every method is a `(state, input) -> (state,
431/// output)` transition with `now_us` passed in — the clock read lives one layer up
432/// in [`SendCoordinator`], so the split policy is unit-testable.
433#[derive(Debug)]
434struct CoordState {
435    budget: hound_media::SendBudget,
436    registered: [bool; SEND_TRACKS],
437    /// Each slot's last reported UNCAPPED desire; `None` until it first ticks.
438    desired: [Option<u64>; SEND_TRACKS],
439    caps: [u64; SEND_TRACKS],
440    /// Bumped on every registration of that slot. A [`SendSlot`] only deregisters
441    /// when its own generation is still the live one, so a restarting publisher
442    /// (a share stopping and immediately restarting) can't have its predecessor's
443    /// late drop knock the new registration out.
444    generation: [u64; SEND_TRACKS],
445    /// Whether either track dropped publish AUs since the last advance — the one
446    /// connection-level congestion signal a sender sees without new transport
447    /// surface. Consumed (and cleared) by each advance.
448    over_capacity: bool,
449    last_advance_us: Option<i64>,
450}
451
452impl CoordState {
453    fn fresh() -> Self {
454        Self {
455            budget: hound_media::SendBudget::new(
456                SEND_BUDGET_MAX_BPS,
457                SEND_BUDGET_MIN_BPS,
458                SEND_BUDGET_MAX_BPS,
459            ),
460            registered: [false; SEND_TRACKS],
461            desired: [None; SEND_TRACKS],
462            caps: [u64::MAX; SEND_TRACKS],
463            generation: [0; SEND_TRACKS],
464            over_capacity: false,
465            last_advance_us: None,
466        }
467    }
468
469    /// Both publishers are live AND each has reported a desire at least once. Until
470    /// then the split has nothing honest to divide, so nobody is capped.
471    fn coordinating(&self) -> bool {
472        (0..SEND_TRACKS).all(|i| self.registered[i] && self.desired[i].is_some())
473    }
474
475    /// Any membership change starts the pairing over: a fresh (optimistic) budget and
476    /// no caps, so the surviving publisher is released the moment the other stops.
477    fn membership_changed(&mut self) {
478        self.budget = CoordState::fresh().budget;
479        self.caps = [u64::MAX; SEND_TRACKS];
480        self.over_capacity = false;
481        self.last_advance_us = None;
482    }
483
484    /// Fold one publisher's tick in and return its share. `None` = uncoordinated
485    /// (fewer than two live publishers) — the caller clears its cap and its own
486    /// ladder governs, exactly as before this coordinator existed.
487    fn observe(
488        &mut self,
489        track: SendTrack,
490        desired_bps: u64,
491        dropped: u32,
492        now_us: i64,
493    ) -> Option<u64> {
494        let i = track as usize;
495        self.desired[i] = Some(desired_bps);
496        self.over_capacity |= dropped > 0;
497        if !self.coordinating() {
498            return None;
499        }
500        let due = self
501            .last_advance_us
502            .is_none_or(|prev| now_us.saturating_sub(prev) >= SEND_BUDGET_INTERVAL_US);
503        if due {
504            let total = self.budget.observe(self.over_capacity, now_us);
505            self.over_capacity = false;
506            self.last_advance_us = Some(now_us);
507            let video_budget = total.saturating_sub(AUDIO_RESERVE_BPS);
508            let desired: Vec<u64> = self.desired.iter().map(|d| d.unwrap_or(0)).collect();
509            for (cap, share) in self
510                .caps
511                .iter_mut()
512                .zip(hound_media::allocate_send_budget(video_budget, &desired))
513            {
514                *cap = share;
515            }
516        }
517        Some(self.caps[i])
518    }
519}

521/// The shared-connection send coordinator. One per process (see
522/// [`send_coordinator`]) — a call has exactly one connection, and the RAII
523/// [`SendSlot`]s make it inert between calls.
524pub(super) struct SendCoordinator {
525    /// The coordinator's OWN timeline. Each publish loop stamps its adaptation ticks
526    /// from its own `CaptureClock`, so their `now_us` values are not comparable;
527    /// the AIMD needs one clock, and this is it (the documented single real-clock
528    /// seam — [`CoordState`] itself takes time as an argument).
529    clock: CaptureClock,
530    state: std::sync::Mutex<CoordState>,
531}
532
533impl SendCoordinator {
534    fn new() -> Self {
535        Self {
536            clock: CaptureClock::start(),
537            state: std::sync::Mutex::new(CoordState::fresh()),
538        }
539    }
540
541    /// Claim this publisher's slot for the life of the returned guard.
542    fn register(self: &Arc<Self>, track: SendTrack) -> SendSlot {
543        let mut generation = 0;
544        if let Ok(mut s) = self.state.lock() {
545            let i = track as usize;
546            s.registered[i] = true;
547            s.generation[i] = s.generation[i].wrapping_add(1);
548            generation = s.generation[i];
549            s.membership_changed();
550        }
551        SendSlot {
552            coord: self.clone(),
553            track,
554            generation,
555        }
556    }
557}
558
559/// A publisher's registration in the [`SendCoordinator`], released on drop (so a
560/// publish loop returning — leave, reconnect teardown, a share ending — hands the
561/// wire back to whoever is left within one tick).
562pub(super) struct SendSlot {
563    coord: Arc<SendCoordinator>,
564    track: SendTrack,
565    /// Which registration of `track` this guard is, so a late drop from a replaced
566    /// publisher never deregisters its successor.
567    generation: u64,
568}
569
570impl SendSlot {
571    /// Report this track's uncapped desire + the publish AUs it dropped this
572    /// interval; get back our share of the connection. `None` = uncoordinated.
573    fn observe(&self, desired_bps: u64, dropped: u32) -> Option<u64> {
574        let now_us = self.coord.clock.now();
575        let Ok(mut s) = self.coord.state.lock() else {
576            // A poisoned coordinator degrades to uncoordinated — never to a stale cap.
577            return None;
578        };
579        s.observe(self.track, desired_bps, dropped, now_us)
580    }
581}
582
583impl Drop for SendSlot {
584    fn drop(&mut self) {
585        if let Ok(mut s) = self.coord.state.lock() {
586            let i = self.track as usize;
587            // A publisher that has already been replaced (share restart) must not
588            // deregister the live registration that took its place.
589            if s.generation[i] != self.generation {
590                return;
591            }
592            s.registered[i] = false;
593            s.desired[i] = None;
594            s.membership_changed();
595        }
596    }
597}

603    static COORD: std::sync::OnceLock<Arc<SendCoordinator>> = std::sync::OnceLock::new();

607/// Turn any outstanding keyframe request into a throttle-approved forced IDR, and
608/// arm the encode loop's `force_keyframe` seam when one is due. Pure (time passed
609/// in) — the throttle itself lives in [`hound_media::KeyframeThrottle`].
610///
611/// Called from BOTH the immediate path (a `RequestKeyframe` just arrived — see
612/// [`AdaptShared::pli_notify`]) and the 1 s adaptation tick, so the two can never
613/// disagree about what a request means. Idempotent: with nothing pending and
614/// nothing due it is two relaxed loads.
615///
616/// The immediate path is what makes subscribe→IDR fast. Waiting for the tick cost
617/// up to a full [`ADAPT_TICK_INTERVAL`] of pure latency, which a screenshare —
618/// [`SCREEN_KEYFRAME_CADENCE`] emits IDRs on demand only, so no periodic keyframe
619/// bails it out — paid in full on every layer switch and every loss resync.
620fn drain_keyframe_request(adaptation: &mut VideoAdaptation, shared: &AdaptShared, now_us: i64) {
621    if shared.pli_pending.swap(false, Ordering::Relaxed) {
622        adaptation.on_pli();
623    }
624    if adaptation.poll_keyframe(now_us) {
625        shared.force_keyframe.store(true, Ordering::Relaxed);
626    }
627}

629/// Read the connection RTT (the one bit of IO) and advance the adaptation one tick.
630/// Splitting the RTT read out keeps [`apply_adaptation_tick`] pure + unit-testable.
631pub(super) fn run_adaptation_tick(
632    adaptation: &mut VideoAdaptation,
633    shared: &AdaptShared,
634    geometry: &mut hound_media::GeometryDwell,
635    conn: &iroh::endpoint::Connection,
636    now_us: i64,
637    stats: Option<(&CallStats, bool)>,
638    budget: Option<&SendSlot>,
639) {
640    use iroh::endpoint::PathId;
641    let rtt_us = conn
642        .rtt(PathId::ZERO)
643        .map(|d| d.as_micros().min(i64::MAX as u128) as i64);
644    apply_adaptation_tick(adaptation, shared, geometry, rtt_us, now_us, stats, budget);
645}

647/// The pure glue between [`AdaptShared`] (the atomics the capture thread reads/writes)
648/// and the sans-IO [`VideoAdaptation`] core, with RTT passed in. Drains the
649/// backpressure counters + a pending PLI + the worst receiver-reported loss, advances
650/// the core, and writes the decision back (target bitrate/fps/geometry, paused,
651/// forced keyframe). No IO here — unit-testable by driving the atomics directly
652/// (the optional [`SendSlot`] is the one seam that reads the coordinator's clock,
653/// and it is `None` in those tests).
654///
655/// `geometry` is the encoder-REBUILD geometry dwell (see [`hound_media::GeometryDwell`]),
656/// owned by the caller for the life of the publish task (survives across ticks, seeded
657/// at construction from the source capture geometry). It is driven separately from
658/// `adaptation`'s own rung selection: the rung's bitrate/fps apply to the live encoder
659/// every tick for free, but the rung's *geometry* only commits — and is written to
660/// `shared.encoder_w/encoder_h` — once it has held stable for
661/// [`hound_media::GEOMETRY_REBUILD_DWELL_US`], so an encoder rebuild never flaps.
662fn apply_adaptation_tick(
663    adaptation: &mut VideoAdaptation,
664    shared: &AdaptShared,
665    geometry: &mut hound_media::GeometryDwell,
666    rtt_us: Option<i64>,
667    now_us: i64,
668    // RTC-debug stats sink: `(registry, is_screen)`. `None` in unit tests. RTT is
669    // transport-wide; the decision (target/geometry/rung/loss) is per-track.
670    stats: Option<(&CallStats, bool)>,
671    // This publisher's slot in the shared-connection send budget. `None` in unit
672    // tests that drive one track in isolation.
673    budget: Option<&SendSlot>,
674) {
675    let produced = shared.produced.swap(0, Ordering::Relaxed);
676    let dropped = shared.dropped.swap(0, Ordering::Relaxed);
677
678    // Coordinate with the other video publisher on this connection: report our
679    // UNCAPPED desire + this interval's publish drops, take back our share. `None`
680    // (we're the only publisher) clears the cap, so a single publisher's ladder is
681    // governed by its own BWE exactly as before.
682    if let Some(slot) = budget {
683        adaptation.set_send_cap_bps(slot.observe(adaptation.desired_bps(), dropped));
684    }
685
686    // Drain a pending PLI into the throttle (coalesces a storm into one IDR). The
687    // fast path already does this the instant a request arrives; this is the safety
688    // net for a request the throttle held off, or one whose wake-up was missed.
689    drain_keyframe_request(adaptation, shared, now_us);
690
691    let remote_loss = shared.take_remote_loss();
692
693    let decision = adaptation.on_tick(
694        &AdaptSignals {
695            produced,
696            dropped,
697            rtt_us,
698            remote_loss,
699        },
700        now_us,
701    );
702
703    shared
704        .target_bps
705        .store(decision.target_bps, Ordering::Relaxed);
706    // The rung's fps + RAW geometry (telemetry only) drive the capture thread;
707    // AudioOnly carries no rung (the publish path is skipped while paused anyway).
708    let (target_fps, target_w, target_h) = match decision.quality {
709        Quality::Video(rung) => (rung.fps, rung.width, rung.height),
710        Quality::AudioOnly => (0, 0, 0),
711    };
712    shared.target_fps.store(target_fps, Ordering::Relaxed);
713    shared.target_w.store(target_w, Ordering::Relaxed);
714    shared.target_h.store(target_h, Ordering::Relaxed);
715
716    // The DWELLED encoder-rebuild geometry: feed the rung's height (`None` for
717    // AudioOnly — the dwell holds its last committed geometry, harmless since the
718    // capture thread skips encode entirely while paused) and publish whatever it
719    // currently commits to. Usually a no-op (`Cow`-free atomics store of the same
720    // value); only moves once a candidate has held for the full dwell.
721    let target_height = match decision.quality {
722        Quality::Video(rung) => Some(rung.height),
723        Quality::AudioOnly => None,
724    };
725    let (encoder_w, encoder_h) = geometry.observe(
726        target_height,
727        now_us,
728        hound_media::GEOMETRY_REBUILD_DWELL_US,
729    );
730    shared.encoder_w.store(encoder_w, Ordering::Relaxed);
731    shared.encoder_h.store(encoder_h, Ordering::Relaxed);
732    let was_paused = shared
733        .paused
734        .swap(decision.is_audio_only(), Ordering::Relaxed);
735
736    // Record the RTC-debug stats: transport RTT + this track's outbound decision.
737    if let Some((stats, is_screen)) = stats {
738        if let Some(rtt) = rtt_us {
739            stats.set_rtt_us(rtt);
740        }
741        let loss_bps = (remote_loss.unwrap_or(0.0) * 10_000.0) as u32;
742        stats.set_video_publish_live(is_screen, shared.source_publish_live());
743        let rung_label = match decision.quality {
744            Quality::Video(rung) => format!("{}x{}@{}", rung.width, rung.height, rung.fps),
745            Quality::AudioOnly => "audio_only".to_string(),
746        };
747        if is_screen {
748            stats.set_screen_decision(
749                decision.target_bps,
750                target_w,
751                target_h,
752                &rung_label,
753                loss_bps,
754            );
755        } else {
756            stats.set_video_decision(
757                decision.target_bps,
758                target_w,
759                target_h,
760                &rung_label,
761                loss_bps,
762            );
763        }
764    }
765
766    if decision.quality_changed {
767        match decision.quality {
768            Quality::Video(rung) => tracing::info!(
769                target_bps = decision.target_bps,
770                width = rung.width,
771                height = rung.height,
772                fps = rung.fps,
773                rtt_us,
774                produced,
775                dropped,
776                "call: adapt → video rung"
777            ),
778            Quality::AudioOnly => tracing::info!(
779                target_bps = decision.target_bps,
780                rtt_us,
781                "call: adapt → AudioOnly (video paused)"
782            ),
783        }
784    } else if was_paused && !decision.is_audio_only() {
785        tracing::info!("call: adapt → video resumed");
786    }
787}

789/// Spawn the producer side of the PLI relay: watch the `keyframe_tx` broadcast and,
790/// when the requested `TrackId` is OUR video track of `source` (Camera vs Screen),
791/// mark a PLI pending in [`AdaptShared`]. The adaptation tick drains it through the
792/// throttle, coalescing a storm into a single IDR. "Our track" is resolved from the
793/// `SessionView` (a video track whose `peer == self` AND `source == source`).
794pub(super) fn spawn_keyframe_listener(
795    mut requests: broadcast::Receiver<TrackId>,
796    view_rx: watch::Receiver<SessionView>,
797    self_id: PeerId,
798    source: Source,
799    adapt: Arc<AdaptShared>,
800    // A PLI for OUR LOW track sets this (the capture loop forces a low IDR), instead
801    // of the source's `pli_pending` — so a receiver subscribing to the low layer gets
802    // a fresh keyframe without waiting for the periodic one. `None` for tracks with no
803    // low layer (e.g. camera today).
804    low_keyframe: Option<Arc<AtomicBool>>,
805) -> JoinHandle<()> {
806    call_runtime().spawn(async move {
807        loop {
808            match requests.recv().await {
809                Ok(track) => {
810                    let (is_ours, is_low) = {
811                        let view = view_rx.borrow();
812                        match view.track(track) {
813                            Some(t)
814                                if t.peer == self_id
815                                    && t.kind == MediaKind::Video
816                                    && t.source == source =>
817                            {
818                                (true, t.low)
819                            }
820                            _ => (false, false),
821                        }
822                    };
823                    if is_ours {
824                        adapt.demand_seen.store(true, Ordering::Relaxed);
825                        match (is_low, &low_keyframe) {
826                            // The LOW track has no publish-loop throttle of its own:
827                            // the capture thread swap-clears this flag on its very
828                            // next frame, so it is already immediate.
829                            (true, Some(lk)) => lk.store(true, Ordering::Relaxed),
830                            _ => {
831                                adapt.pli_pending.store(true, Ordering::Relaxed);
832                                // Wake the publish loop NOW rather than leaving the
833                                // request for its next 1 s tick.
834                                adapt.pli_notify.notify_one();
835                            }
836                        }
837                    }
838                }
839                // Lagged: a coalesced keyframe is exactly the debounce we want —
840                // mark pending and continue.
841                Err(broadcast::error::RecvError::Lagged(_)) => {
842                    adapt.pli_pending.store(true, Ordering::Relaxed);
843                    adapt.pli_notify.notify_one();
844                }
845                // Closed = the broadcast sender dropped (session teardown).
846                Err(broadcast::error::RecvError::Closed) => break,
847            }
848        }
849    })
850}

852/// Spawn the producer side of the receiver→producer BWE loop: watch the `feedback_tx`
853/// broadcast and, when a report names OUR video track of `source`, fold its loss into
854/// [`AdaptShared`] (max-reduced across reporters this interval — serve the worst leg).
855/// The adaptation tick drains it into the sender-side rate controller. Reports for
856/// other tracks (or a stale id) are ignored. "Our track" is resolved exactly like the
857/// keyframe listener.
858///
859/// A peer the planner (`layers`) currently has capped — on the low layer, or
860/// transiently still on the source track before the low track's first frame — is
861/// EXCLUDED from the fold even though it names our source track: a low-layer receiver
862/// must never drag the source adaptation down.
863pub(super) fn spawn_feedback_listener(
864    mut feedback: broadcast::Receiver<MediaFeedback>,
865    view_rx: watch::Receiver<SessionView>,
866    self_id: PeerId,
867    source: Source,
868    adapt: Arc<AdaptShared>,
869    layers: Arc<LayerControl>,
870) -> JoinHandle<()> {
871    call_runtime().spawn(async move {
872        loop {
873            match feedback.recv().await {
874                Ok(report) => {
875                    let is_ours = {
876                        let view = view_rx.borrow();
877                        view.track(report.track).is_some_and(|t| {
878                            // SOURCE track only: a low-layer receiver reports on the
879                            // `_l0` track (also Video/`source`), and folding its loss
880                            // here would let a constrained low viewer drag the source's
881                            // BWE down for the focused viewers. The low layer runs at the
882                            // planner's cap, not loss-based adaptation.
883                            t.peer == self_id
884                                && t.kind == MediaKind::Video
885                                && t.source == source
886                                && !t.low
887                        })
888                    };
889                    // A peer the planner currently has capped (on the low layer, or
890                    // transiently still on the source track before the low track's
891                    // first frame) must never drag the SOURCE adaptation down.
892                    if is_ours && !layers.capped_peers().contains(&report.from.0) {
893                        adapt.record_remote_feedback(report.received, report.lost);
894                    }
895                }
896                // Lagged: the next interval's report supersedes the dropped ones
897                // (per-interval deltas) — keep going.
898                Err(broadcast::error::RecvError::Lagged(_)) => {}
899                Err(broadcast::error::RecvError::Closed) => break,
900            }
901        }
902    })
903}

905/// The shared per-frame encode→adapt→publish core for both video capture loops.
906/// Retargets the
907/// encoder to the BWE's latest bitrate (only on a real move), honours a
908/// throttle-approved keyframe (swap-clears so it fires once), downsamples to the
909/// rung's fps, scales the frame to `encoder_dims` when it doesn't already match
910/// (the caller keeps `encoder_dims` in sync with the live encoder — see the
911/// geometry-rebuild note below), encodes, and forwards each access unit via
912/// `forward`. `forward` returns `false` on backpressure (its bounded channel was
913/// full); the caller's drop bumps `dropped`, which the adaptation tick reads as
914/// the congestion signal. A forced keyframe is re-armed when it is dropped, so
915/// the next captured frame re-emits it.
916///
917/// `encoder_dims` is the geometry the passed-in `encoder` was ACTUALLY built at —
918/// not necessarily `frame`'s own (capture) geometry. The caller (the capture
919/// loop) is responsible for rebuilding the encoder to the adaptation's dwelled
920/// rung geometry (`AdaptShared::encoder_w/encoder_h`) and passing the matching
921/// `encoder_dims` here; this function's job is only to make the FRAME match
922/// whatever the encoder actually is, scaling down when the two diverge and
923/// skipping gracefully (never spam-rejecting) if a scale is impossible. This is
924/// the fix for the original resolution-adaptation regression: a rung-downscaled
925/// frame was once fed to an encoder still fixed at a different geometry
926/// ("frame WxH != encoder W'xH'"), rejecting ~every frame (produced≈0,
927/// dropped≈all). Now the two can never diverge for long — the capture loop
928/// rebuilds the encoder to match the dwelled rung, and this scale-to-match is
929/// the defensive fallback for the one-tick window before that rebuild lands.
930#[allow(clippy::too_many_arguments)]
931pub(super) fn encode_and_publish_adaptive(
932    frame: &hound_engine::types::RawFrame,
933    encoder: &mut hound_engine::api::VideoEncoder,
934    adapt: &AdaptShared,
935    applied_bps: &mut u32,
936    fps_gate: &mut hound_media::FrameRateGate,
937    encoder_dims: Dimensions,
938    mut forward: impl FnMut(WireFrame) -> bool,
939    frame_seq: &mut u32,
940    sender: u16,
941    track: TrackKind,
942    stats: &CallStats,
943    fail_streak: &mut EncodeFailStreak,
944) {
945    use hound_engine::backend::VideoEncode;
946
947    // Retarget if the BWE moved the target since we last applied it. The rung target
948    // is an H.264-equivalent quality ceiling; scale it down for a more efficient codec
949    // so the SAME quality ships at fewer bits (the receive-side packet CPU, which
950    // dominates a screenshare, scales with bits — this is where the codec win lands).
951    let target = adapt.target_bps.load(Ordering::Relaxed);
952    let scaled = (target as f64 * codec_bitrate_factor(encoder.codec())) as u64;
953    let target32 = scaled.min(u32::MAX as u64) as u32;
954    if let Some(rate) = Bitrate::try_from_bps(target32)
955        && target32 != *applied_bps
956    {
957        if let Err(e) = encoder.set_bitrate(rate) {
958            tracing::warn!("call: set_bitrate({target32}) failed: {e}");
959        } else {
960            *applied_bps = target32;
961        }
962    }
963
964    // A forced keyframe (new subscriber / PLI) must always emit — gating it would
965    // starve a fresh decoder of its IDR. Otherwise downsample to the rung's fps. A
966    // frame dropped here is a deliberate downsample, NOT backpressure (don't bump
967    // `dropped`).
968    let force_keyframe = adapt.force_keyframe.swap(false, Ordering::Relaxed);
969    let target_fps = adapt.target_fps.load(Ordering::Relaxed);
970    if !force_keyframe && !fps_gate.should_emit(frame.timestamp_us.micros(), target_fps) {
971        return;
972    }
973    if force_keyframe {
974        encoder.request_keyframe();
975    }
976
977    // Match the frame to the encoder's ACTUAL geometry. The healthy/top-rung case
978    // (`frame.dims == encoder_dims`, the common case whenever the caller's rebuild is
979    // caught up) is the zero-copy fast path — byte-identical to no adaptation. A
980    // divergence (the rung dwelled down and the caller's rebuild hasn't landed on
981    // this exact frame yet) downscales in software; if even that fails, skip this
982    // frame gracefully rather than ever handing the encoder a mismatched size — this
983    // is the exact failure mode ("frame WxH != encoder W'xH'") that killed outgoing
984    // video before the geometry rebuild existed.
985    let encode_frame = if frame.dims == encoder_dims {
986        std::borrow::Cow::Borrowed(frame)
987    } else {
988        match frame.scaled_to(encoder_dims) {
989            Some(f) => std::borrow::Cow::Owned(f),
990            None => {
991                if fail_streak.bump() {
992                    tracing::warn!(
993                        frame = %frame.dims,
994                        encoder = %encoder_dims,
995                        "call: frame/encoder geometry mismatch and downscale failed — skipping frame"
996                    );
997                } else {
998                    tracing::debug!(
999                        frame = %frame.dims,
1000                        encoder = %encoder_dims,
1001                        "call: frame/encoder geometry mismatch and downscale failed — skipping frame"
1002                    );
1003                }
1004                if force_keyframe {
1005                    // Re-arm so the next frame still serves the owed IDR.
1006                    adapt.force_keyframe.store(true, Ordering::Relaxed);
1007                }
1008                return;
1009            }
1010        }
1011    };
1012
1013    let wire_codec = encoder.codec().to_wire();
1014    let is_screen = matches!(track, TrackKind::ScrVideo);
1015    // Time the HW-encode call for the RTC-debug encode-latency stat.
1016    let encode_start = std::time::Instant::now();
1017    let encoded = encoder.encode(&encode_frame);
1018    let encode_us = encode_start.elapsed().as_micros().min(i64::MAX as u128) as i64;
1019    match encoded {
1020        Ok(chunks) => {
1021            fail_streak.reset();
1022            for chunk in chunks {
1023                adapt.produced.fetch_add(1, Ordering::Relaxed);
1024                let is_keyframe = chunk.keyframe;
1025                // RTC-debug: one encoded AU (bytes drive send_kbps; keyframe count).
1026                let bytes = chunk.data.len();
1027                if is_screen {
1028                    stats.screen_encoded(bytes, encode_us, is_keyframe);
1029                } else {
1030                    stats.video_encoded(bytes, encode_us, is_keyframe);
1031                }
1032                let wire = WireFrame {
1033                    sender,
1034                    track,
1035                    codec: wire_codec,
1036                    keyframe: is_keyframe,
1037                    frame_seq: *frame_seq,
1038                    timestamp_us: chunk.timestamp_us.micros(),
1039                    data: chunk.data,
1040                };
1041                *frame_seq = frame_seq.wrapping_add(1);
1042                if !forward(wire) {
1043                    adapt.dropped.fetch_add(1, Ordering::Relaxed);
1044                    if is_screen {
1045                        stats.screen_publish_drop();
1046                    } else {
1047                        stats.video_publish_drop();
1048                    }
1049                    // A dropped keyframe must re-arm the forced-IDR flag so the next
1050                    // captured frame re-emits it.
1051                    if is_keyframe {
1052                        adapt.force_keyframe.store(true, Ordering::Relaxed);
1053                    }
1054                }
1055            }
1056        }
1057        Err(e) => {

1067/// Drive one video publish loop: the 1 s adaptation tick + frame forwarding, with
1068/// the **live gate** deciding when forwarding starts. The tick runs from the first
1069/// iteration — before any subscriber — so PLIs are serviced and the BWE keeps
1070/// state even while dark; only *forwarding* waits for proof of a receiver.
1071///
1072/// Live = the moq subscriber signal ([`MoqTrackPublisher::subscribed`]) OR
1073/// control-plane demand (`adapt.demand_seen`, latched by the keyframe listener).
1074/// The fallback matters when a broadcast is replaced on the same path (a share
1075/// restart): moq-net keeps the newcomer as a *backup* until the old broadcast
1076/// closes, so `used()` can arrive late or not at all — while the receiver's
1077/// subscribe always sends `RequestKeyframe`. Going live forces a fresh IDR, so
1078/// the first delivered group always opens decodable; frames before the gate are
1079/// dropped (a dropped keyframe re-arms the forced IDR).
1080///
1081/// Going live also opens the capture thread's ENCODE gate
1082/// ([`AdaptShared::source_encode_open`]) — while dark the source rung is not
1083/// encoded at all, so this branch normally sees no frames to drop. It stays as
1084/// the belt-and-braces stop for anything already in flight across the channel
1085/// when the gate flips (and for the low-rung-only demand window, where the
1086/// capture thread can be live a tick before this loop is).
1087///
1088/// `before_tick` runs ahead of each adaptation tick (the screen path reloads its
1089/// live quality preference there). Returns when the encode channel closes or the
1090/// track dies.
1091///
1092/// The loop also holds this publisher's [`SendSlot`] for its lifetime, so while the
1093/// OTHER video publisher is also live the two ladders draw from one coordinated
1094/// connection budget rather than each claiming the whole wire.
1095#[allow(clippy::too_many_arguments)]
1096pub(super) async fn drive_video_publish(
1097    publisher: &mut MoqTrackPublisher,
1098    enc_rx: &mut tokio::sync::mpsc::Receiver<WireFrame>,
1099    adapt: &AdaptShared,
1100    adaptation: &mut VideoAdaptation,
1101    geometry: &mut hound_media::GeometryDwell,
1102    rtt_conn: &iroh::endpoint::Connection,
1103    stats: &CallStats,
1104    is_screen: bool,
1105    track_label: &'static str,
1106    mut before_tick: impl FnMut(&mut VideoAdaptation),
1107) {
1108    let adapt_clock = CaptureClock::start();
1109    // Our slot in the shared-connection send budget, held for the life of this
1110    // publish loop. Coordination only bites once the OTHER video publisher holds one
1111    // too; released on return (leave, reconnect teardown, a share ending).
1112    let budget = send_coordinator().register(if is_screen {
1113        SendTrack::Screen
1114    } else {
1115        SendTrack::Camera
1116    });
1117    let mut tick = tokio::time::interval(ADAPT_TICK_INTERVAL);
1118    tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
1119
1120    let subscribed = publisher.subscribed();
1121    tokio::pin!(subscribed);
1122    let mut live = false;
1123    // Live but not yet opened on a keyframe (the go-live forced IDR in flight).
1124    let mut primed = false;
1125    let go_live = |live: &mut bool| {
1126        *live = true;
1127        // Opens the capture thread's ENCODE gate AND arms the opening IDR, so the
1128        // first frame it encodes after the transition is a keyframe (the `primed`
1129        // check below only forwards from a keyframe anyway).
1130        adapt.go_live();
1131        tracing::info!(
1132            track = track_label,
1133            "call: video publish live — encode gate open, forcing opening IDR"
1134        );
1135    };
1136
1137    loop {
1138        tokio::select! {
1139            biased;
1140            r = &mut subscribed, if !live => {
1141                if r.is_err() {
1142                    return; // track closed before anyone attached
1143                }
1144                go_live(&mut live);
1145            }
1146            // A `RequestKeyframe` just landed for our track. Serve it here instead of
1147            // at the next tick — a whole second of pure latency between a receiver
1148            // subscribing (or switching simulcast layer) and its first decodable
1149            // frame. Ranked above the tick/frame branches so the IDR is armed before
1150            // the next captured frame is encoded.
1151            _ = adapt.pli_notify.notified() => {
1152                // A keyframe request is also the earliest proof a receiver exists —
1153                // the same fallback the tick uses, applied a tick sooner.
1154                if !live && adapt.demand_seen.load(Ordering::Relaxed) {
1155                    go_live(&mut live);
1156                }
1157                drain_keyframe_request(adaptation, adapt, adapt_clock.now());
1158            }
1159            _ = tick.tick() => {
1160                before_tick(adaptation);
1161                run_adaptation_tick(adaptation, adapt, geometry, rtt_conn, adapt_clock.now(), Some((stats, is_screen)), Some(&budget));
1162                if !live && adapt.demand_seen.load(Ordering::Relaxed) {
1163                    go_live(&mut live);
1164                }
1165            }
1166            frame = enc_rx.recv() => {
1167                let Some(frame) = frame else { return };
1168                if !live {
1169                    if adapt.demand_seen.load(Ordering::Relaxed) {
1170                        go_live(&mut live);
1171                    } else {
1172                        // Dark: no receiver yet. Don't ship (or queue) stale frames;
1173                        // a dropped keyframe re-arms so the first live frame is an IDR.
1174                        if frame.keyframe {
1175                            adapt.force_keyframe.store(true, Ordering::Relaxed);
1176                        }
1177                        continue;
1178                    }
1179                }
1180                // The publish stream must OPEN on a keyframe (a delta has no group
1181                // to ride); skip pre-IDR deltas still in flight from before the
1182                // go-live forced keyframe.
1183                if !primed {
1184                    if !frame.keyframe {
1185                        continue;
1186                    }
1187                    primed = true;
1188                }
1189                // Tell the throttle a keyframe went out so it measures the next
1190                // forced-IDR interval from here.
1191                if frame.keyframe {
1192                    adaptation.on_keyframe_emitted(adapt_clock.now());
1193                }
1194                if let Err(e) = publish_one(publisher, &frame) {
1195                    tracing::warn!(%e, track = track_label, "call: publish video frame failed");
1196                }
1197            }
1198        }
1199    }
1200}

1202/// Watch the roster and re-negotiate the encode codec as peers join/leave, writing
1203/// the decision into `adapt.target_codec` for the encode loop to act on. `encode_caps`
1204/// is probed once by the caller (the probe isn't free) and reused here. Lives for the
1205/// publish — ends when `view_rx` closes (session torn down).
1206pub(super) fn spawn_codec_renegotiator(
1207    mut view_rx: watch::Receiver<SessionView>,
1208    encode_caps: hound_engine::types::VideoCodecCaps,
1209    adapt: Arc<AdaptShared>,
1210    initial: VideoCodec,
1211    track: &'static str,
1212) -> JoinHandle<()> {
1213    call_runtime().spawn(async move {
1214        let mut switch = CodecSwitch::new(initial);
1215        loop {
1216            let want = negotiate_codec(encode_caps, &view_rx.borrow());
1217            if let Some(new) = switch.observe(want, Instant::now(), CODEC_UPGRADE_DEBOUNCE) {
1218                adapt.target_codec.store(codec_rank(new), Ordering::Relaxed);
1219                tracing::info!(track, codec = ?new, "call: re-negotiated encode codec (roster change)");
1220            }
1221            // Wake on the next roster change, or when a debounced upgrade is due.
1222            let wait = switch
1223                .pending_deadline(CODEC_UPGRADE_DEBOUNCE)
1224                .map(|d| d.saturating_duration_since(Instant::now()));
1225            tokio::select! {
1226                r = view_rx.changed() => { if r.is_err() { break; } }
1227                _ = async {
1228                    match wait {
1229                        Some(d) => tokio::time::sleep(d).await,
1230                        None => std::future::pending::<()>().await,
1231                    }
1232                } => {}
1233            }
1234        }
1235    })
1236}
