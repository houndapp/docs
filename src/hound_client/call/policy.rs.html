10/// The LOW simulcast layer's encode geometry for a screen share, derived from the
11/// SOURCE so it **preserves aspect ratio** — a fixed 16:9 rung (e.g. `640×360`) would
12/// stretch any non-16:9 share (a near-square window gets squished horizontally).
13///
14/// The budget is the screen rung's own PIXEL COUNT
15/// (`hound_media::SCREEN_LOW_RUNG`), not a long edge, so the geometry the encoder is
16/// built at always costs what the rung is priced at. A long-edge budget did not: a
17/// near-square window came out 640×640 — 1.78× the pixels of the nominal 640×360 the
18/// cap was priced for — and the encoder then had to spend a 360p budget on 410k
19/// pixels. 16:9 and 9:16 sources are unaffected (they land on exactly the same
20/// geometry the long edge gave them); only the mispriced near-square case moves.
21/// Never upscales, so `scaled_to` (which stretches to the exact target) always
22/// downscales without distortion.
23pub(super) fn low_screen_dims(source: Dimensions) -> Dimensions {
24    let (w, h) = hound_media::fit_encode_area(
25        (source.width(), source.height()),
26        hound_media::SCREEN_LOW_RUNG.pixels(),
27    );
28    Dimensions::even_clamped(w, h)
29}

31/// Map the control-plane [`Source`] onto `hound-media`'s video content kind — the
32/// axis the demand ladder prices against. Camera and screen compress differently
33/// and degrade differently (a small camera tile drops resolution, a small screen
34/// tile drops frame rate), so every rung decision needs it. `Microphone` has no
35/// video ladder; it is treated as camera content, which no caller can reach — the
36/// tile-demand and low-layer paths are both video-only.
37pub(super) fn video_source(source: Source) -> hound_media::VideoSource {
38    match source {
39        Source::Screen => hound_media::VideoSource::Screen,
40        Source::Camera | Source::Microphone => hound_media::VideoSource::Camera,
41    }
42}

44/// How long the screen capture must be IDLE (no real frames delivered) before the
45/// loop starts probing whether the source still exists. Static content is
46/// legitimately idle, so probing starts only after a quiet spell.
47const SOURCE_LIVENESS_IDLE: std::time::Duration = std::time::Duration::from_secs(3);
48/// Minimum spacing between source-liveness probes (each is a blocking ~tens-of-ms
49/// shareable-content fetch on the otherwise-idle capture thread).
50const SOURCE_LIVENESS_INTERVAL: std::time::Duration = std::time::Duration::from_secs(2);
51
52/// Whether an idle capture loop should probe source liveness now. Pure — the
53/// throttle policy in one testable place: probe only after a real-frame drought of
54/// [`SOURCE_LIVENESS_IDLE`], at most every [`SOURCE_LIVENESS_INTERVAL`].
55pub(super) fn source_probe_due(idle_for: Duration, since_last_probe: Duration) -> bool {
56    idle_for >= SOURCE_LIVENESS_IDLE && since_last_probe >= SOURCE_LIVENESS_INTERVAL
57}

83/// Bitrate scale per codec for EQUAL perceptual quality, relative to H.264. A more
84/// efficient codec reaches the same quality at fewer bits, so we bank the efficiency
85/// as lower bandwidth — and the per-packet receive CPU that dominates a screenshare —
86/// instead of a sharper picture at the same bitrate. The BWE rung stays the quality
87/// target (an H.264-equivalent ceiling); this scales what the encoder actually emits.
88/// Conservative vs the codecs' headline gains (HEVC ~40-50%, AV1 ~50%+, more still on
89/// screen content) so quality never regresses.
90pub(super) fn codec_bitrate_factor(c: VideoCodec) -> f64 {
91    match c {
92        VideoCodec::H264 => 1.0,
93        VideoCodec::H265 => 0.65,
94        VideoCodec::Av1 => 0.5,
95    }
96}

98/// Pick the video codec to PUBLISH: the highest rung on the encode ladder
99/// (AV1 → H.265 → H.264, [`hound_engine::types::ENCODE_PRIORITY`]) this device can
100/// **hardware-encode** AND **every current remote peer can decode** (so the
101/// forward-no-transcode relay never hands a peer a codec it can't play), falling back
102/// to H.264. Read from the live roster: a less-capable peer shrinks
103/// `common_remote_decode_caps`, an all-modern call gets HEVC/AV1. `encode_caps` is
104/// probed once by the caller (the probe spins a throwaway VT session) and reused — at
105/// publish time and by the mid-call re-negotiator on every roster change.
106pub(super) fn negotiate_codec(
107    encode_caps: hound_engine::types::VideoCodecCaps,
108    view: &SessionView,
109) -> VideoCodec {
110    let common = view.common_remote_decode_caps();
111    let common_caps = hound_engine::types::VideoCodecCaps {
112        av1: common.av1,
113        h265: common.h265,
114        h264: common.h264,
115    };
116    hound_engine::types::pick_encode_codec(encode_caps, common_caps)
117}

119/// Encode a [`VideoCodec`] for the `AtomicU8` codec slot shared with the encode loop.
120/// The value doubles as a priority rank (higher = richer), so the re-negotiator can
121/// tell a downgrade from an upgrade with a plain comparison.
122pub(super) fn codec_rank(c: VideoCodec) -> u8 {
123    match c {
124        VideoCodec::Av1 => 2,
125        VideoCodec::H265 => 1,
126        VideoCodec::H264 => 0,
127    }
128}

130pub(super) fn codec_from_rank(v: u8) -> VideoCodec {
131    match v {
132        2 => VideoCodec::Av1,
133        1 => VideoCodec::H265,
134        _ => VideoCodec::H264,
135    }
136}

142/// Decides when a live encoder should switch codecs as the roster changes. Pure
143/// (no IO) so the policy is unit-tested. **Downgrades land immediately** — a peer
144/// that can't decode the current codec has no video until we drop to one it can —
145/// while **upgrades are debounced** (a peer leaving lets us climb, but there's no
146/// rush, and rushing invites thrash if they rejoin).
147pub(super) struct CodecSwitch {
148    current: VideoCodec,
149    /// A richer codec seen but not yet committed, and when it was first seen.
150    pending: Option<(VideoCodec, Instant)>,
151}
152
153impl CodecSwitch {
154    pub(super) fn new(initial: VideoCodec) -> Self {
155        Self {
156            current: initial,
157            pending: None,
158        }
159    }
160
161    /// Feed the freshly-negotiated codec for the current roster. Returns `Some(c)`
162    /// when the encoder should switch to `c` now, else `None` (hold).
163    pub(super) fn observe(
164        &mut self,
165        want: VideoCodec,
166        now: Instant,
167        debounce: Duration,
168    ) -> Option<VideoCodec> {
169        if want == self.current {
170            self.pending = None;
171            return None;
172        }
173        if codec_rank(want) < codec_rank(self.current) {
174            // Downgrade: commit immediately (correctness — a peer can't decode us).
175            self.pending = None;
176            self.current = want;
177            return Some(want);
178        }
179        // Upgrade: commit only once `want` has held for `debounce`.
180        match self.pending {
181            Some((p, since)) if p == want && now.duration_since(since) >= debounce => {
182                self.pending = None;
183                self.current = want;
184                Some(want)
185            }
186            Some((p, _)) if p == want => None,
187            _ => {
188                self.pending = Some((want, now));
189                None
190            }
191        }
192    }
193
194    /// When the encode loop should re-poll even without a roster change (to let a
195    /// debounced upgrade fire). `None` when nothing is pending.
196    pub(super) fn pending_deadline(&self, debounce: Duration) -> Option<Instant> {
197        self.pending.map(|(_, since)| since + debounce)
198    }
199}

201/// The resize watch's running state inside the capture loop, factored out PURE so the
202/// debounce classification is unit-testable without a live capture source. Tracks the
203/// baseline content size and a pending (changed-but-not-yet-held) candidate.
204#[derive(Default)]
205pub(super) struct ResizeWatch {
206    baseline: Option<(u32, u32)>,
207    pending: Option<((u32, u32), std::time::Instant)>,
208}
209
210impl ResizeWatch {
211    /// Feed one observed content size at `now`. Returns `true` when a changed size has
212    /// HELD for at least `debounce` (the loop should exit with [`ScreenCaptureExit::Resized`]).
213    /// The first size seen becomes the baseline; a size back at baseline clears any pending.
214    pub(super) fn observe(
215        &mut self,
216        cur: (u32, u32),
217        now: std::time::Instant,
218        debounce: std::time::Duration,
219    ) -> bool {
220        match self.baseline {
221            None => {
222                self.baseline = Some(cur);
223                false
224            }
225            Some(base) if cur != base => match self.pending {
226                Some((size, since)) if size == cur => now.duration_since(since) >= debounce,
227                _ => {
228                    self.pending = Some((cur, now));
229                    false
230                }
231            },
232            // Back at baseline — a transient blip that didn't hold; clear the candidate.
233            Some(_) => {
234                self.pending = None;
235                false
236            }
237        }
238    }
239}

241/// Per-remote-audio-stream playback gain, keyed `(peer, source)` — the tile
242/// right-click volume/mute control. Each audio pipeline holds its stream's gain
243/// CELL (an `AtomicU32` of `f32` bits, loaded lock-free per frame); the map only
244/// locks on the ~UI-rate [`Self::set`] / pipeline-start [`Self::cell`] paths.
245/// Lives on [`ConnectionParams`] so a volume survives reconnects and resubscribes
246/// (the cell is looked up by key, not by pipeline instance).
247#[derive(Default)]
248pub(super) struct StreamVolumes {
249    map: std::sync::Mutex<HashMap<(u32, Source), Arc<std::sync::atomic::AtomicU32>>>,
250}
251
252impl StreamVolumes {
253    /// Get (or create at unity gain) the gain cell for one stream.
254    pub(super) fn cell(&self, peer: u32, source: Source) -> Arc<std::sync::atomic::AtomicU32> {
255        self.map
256            .lock()
257            .unwrap_or_else(|e| e.into_inner())
258            .entry((peer, source))
259            .or_insert_with(|| Arc::new(std::sync::atomic::AtomicU32::new(1.0f32.to_bits())))
260            .clone()
261    }
262
263    /// Set one stream's playback gain. `0.0` = muted, `1.0` = unity; clamped to
264    /// `[0.0, 2.0]` (a modest boost ceiling — beyond that is clipping territory).
265    pub(super) fn set(&self, peer: u32, source: Source, volume: f32) {
266        let v = if volume.is_finite() {
267            volume.clamp(0.0, 2.0)
268        } else {
269            1.0
270        };
271        self.cell(peer, source)
272            .store(v.to_bits(), Ordering::Relaxed);
273    }
274}

276/// Scale decoded PCM by `gain`, saturating at the i16 rails (a >1 boost must clip
277/// gracefully, never wrap). Unity is the caller's fast path — call only when
278/// `gain != 1.0`. Pure; unit-tested.
279pub(super) fn apply_gain(samples: &mut [i16], gain: f32) {
280    for s in samples.iter_mut() {
281        *s = (*s as f32 * gain).clamp(i16::MIN as f32, i16::MAX as f32) as i16;
282    }
283}

285/// One-shot mic route warmup: a non-48 kHz capture device (a Bluetooth headset in
286/// HFP mode) may open onto an SCO route that is not carrying audio yet — the
287/// stream runs, delivering silence, and no error ever fires. Reopening once,
288/// [`hound_engine::api::ROUTE_WARMUP_DELAY`] after open, lands on the settled
289/// route; 48 kHz devices never arm. The playout side has the same policy inside
290/// the mix bus; this is the capture-side half.
291pub(super) struct MicRouteWarmup {
292    pub(super) due: Option<Instant>,
293}
294
295impl MicRouteWarmup {
296    /// Arm (or disarm) for a just-opened device at `device_rate`.
297    pub(super) fn arm(device_rate: u32, now: Instant) -> Self {
298        Self {
299            due: (device_rate != 48_000).then(|| now + hound_engine::api::ROUTE_WARMUP_DELAY),
300        }
301    }
302
303    /// True exactly once, when the warmup deadline passes.
304    pub(super) fn fire(&mut self, now: Instant) -> bool {
305        match self.due {
306            Some(t) if now >= t => {
307                self.due = None;
308                true
309            }
310            _ => false,
311        }
312    }

315/// Consecutive exactly-zero mic frames before warning once that the input is
316/// delivering pure silence — a dead capture route (BT/SCO, revoked permission, a
317/// silent virtual device) produces bit-exact zeros, which a live room's noise
318/// floor never does. ~3 s at 20 ms frames.
319const MIC_SILENCE_WARN_FRAMES: u32 = 150;
320
321/// Detects a dead-silent capture stream (see [`MIC_SILENCE_WARN_FRAMES`]).
322/// [`Self::observe`] returns `true` exactly once per silent streak, at the
323/// threshold; any non-zero sample resets the streak (and re-arms the warning).
324#[derive(Default)]
325pub(super) struct MicSilenceWatch {
326    zero_frames: u32,
327    warned: bool,
328}
329
330impl MicSilenceWatch {
331    pub(super) fn observe(&mut self, samples: &[i16]) -> bool {
332        if samples.iter().all(|&s| s == 0) {
333            self.zero_frames = self.zero_frames.saturating_add(1);
334            if self.zero_frames >= MIC_SILENCE_WARN_FRAMES && !self.warned {
335                self.warned = true;
336                return true;
337            }
338        } else {
339            self.zero_frames = 0;
340            self.warned = false;
341        }
342        false
343    }
344}

348/// Why a capture-device open failed, coarse enough to dedupe logging on.
349///
350/// [`Self::PermissionDenied`] is its own class because it is *sticky*: only the
351/// user flipping a System Settings toggle clears it, so the loop must retry
352/// slowly (a grant can still arrive) and report it as an actionable `error!`.
353/// Everything else is a transient device fault (unplugged, busy, driver hiccup)
354/// that backs off and stays a `warn!`.
355#[derive(Clone, Copy, PartialEq, Eq, Debug)]
356pub(super) enum OpenFailure {
357    PermissionDenied,
358    Other,
359}
360
361impl OpenFailure {
362    pub(super) fn of_capture(e: &hound_engine::error::CaptureError) -> Self {
363        match e {
364            hound_engine::error::CaptureError::PermissionDenied { .. } => Self::PermissionDenied,
365            _ => Self::Other,
366        }
367    }
368
369    pub(super) fn of_audio(e: &hound_engine::error::AudioError) -> Self {
370        match e {
371            hound_engine::error::AudioError::PermissionDenied { .. } => Self::PermissionDenied,
372            _ => Self::Other,
373        }
374    }
375}

377/// A denied device retries at this (slow) cadence: the grant can arrive at any
378/// time — the user flips the toggle in System Settings and capture must resume
379/// without a rejoin — but spinning at capture speed on a permanent denial is pure
380/// waste.
381const OPEN_RETRY_DENIED: Duration = Duration::from_secs(5);
382/// First retry delay for a transient open failure, doubling to [`OPEN_RETRY_MAX`].
383const OPEN_RETRY_MIN: Duration = Duration::from_millis(300);
384/// Ceiling for the transient-failure backoff.
385const OPEN_RETRY_MAX: Duration = Duration::from_secs(5);
386
387/// Retry cadence + log dedupe for a capture-device (re)open loop. Pure: it decides
388/// *whether* this failure is worth a log line and *how long* to wait before the
389/// next attempt; the caller owns the message.
390///
391/// The dedupe is load-bearing, not cosmetic: the Sentry bridge promotes
392/// first-party `tracing::error!` to an event, so a denial logged per retry tick
393/// would be an event storm. A line is emitted only on a TRANSITION into a failure
394/// class — a persisting denial costs exactly one.
395#[derive(Default, Debug)]
396pub(super) struct OpenRetry {
397    last: Option<OpenFailure>,
398    backoff: Option<Duration>,
399}
400
401impl OpenRetry {
402    /// The device opened — re-arm both the log and the backoff, so a LATER failure
403    /// reports again (and a recovered-then-denied device is not silent).
404    pub(super) fn opened(&mut self) {
405        *self = Self::default();
406    }
407
408    /// Record a failed open. Returns `(log_it, wait)`: `log_it` is true only the
409    /// first time this failure class is seen since the last success or class
410    /// change; `wait` is how long to sleep before retrying.
411    pub(super) fn failed(&mut self, failure: OpenFailure) -> (bool, Duration) {
412        let first = self.last != Some(failure);
413        if first {
414            self.backoff = None;
415        }
416        self.last = Some(failure);
417        match failure {
418            OpenFailure::PermissionDenied => (first, OPEN_RETRY_DENIED),
419            OpenFailure::Other => {
420                let wait = self.backoff.unwrap_or(OPEN_RETRY_MIN);
421                self.backoff = Some((wait * 2).min(OPEN_RETRY_MAX));
422                (first, wait)
423            }
424        }
425    }
426}

428/// How long a tile's demand for the LOW layer must HOLD before we downgrade its
429/// video subscription. A switch tears down + reopens a moq subscription on the
430/// connection **shared with audio**; doing it on every transient tile resize (the
431/// Flutter view relays out after a (re)join, firing a burst of sizes that flap
432/// source→low→source within a second) churns the connection's stream state and
433/// disrupts AUDIO delivery (an "audio is sacred" violation). Holding the demand
434/// until it is durable collapses that burst to at most one switch. A downgrade
435/// only ever *saves* bandwidth nobody asked for, so it can afford to be slow.
436/// Upgrades are governed by [`LAYER_UPGRADE_DEBOUNCE`] instead.
437pub(super) const LAYER_SWITCH_DEBOUNCE: Duration = Duration::from_millis(1500);
438
439/// Minimum dwell on a layer after an actuated switch before the next DOWNGRADE
440/// may commit. The debounce alone only filters flaps shorter than itself; a
441/// demand that alternates slower than the debounce (a stage↔filmstrip focus
442/// toggle re-laying tiles out every couple of seconds) would pass it and thrash
443/// the subscription anyway. The dwell bounds actuated churn to one downgrade per
444/// window per stream regardless of the flap period; a held demand still actuates
445/// the moment the dwell expires. Upgrades skip it (see
446/// [`LAYER_UPGRADE_DEBOUNCE`]) — that stays storm-safe because an upgrade can
447/// only ever follow a committed (dwell-gated) downgrade, so the dwell still
448/// bounds the *alternation* rate to one down/up pair per window.
449pub(super) const LAYER_SWITCH_DWELL: Duration = Duration::from_secs(6);
450
451/// How long a tile's demand for the SOURCE layer must hold before we UPGRADE.
452/// Asymmetric with [`LAYER_SWITCH_DEBOUNCE`] on purpose: an upgrade is the direct
453/// answer to a user expanding a tile and they are staring at the result, so this
454/// only has to absorb a relayout flap — a tile briefly large for a frame or two
455/// while the grid reshuffles, i.e. tens of milliseconds. ~200 ms is several frame
456/// periods of margin yet reads as instant.
457pub(super) const LAYER_UPGRADE_DEBOUNCE: Duration = Duration::from_millis(200);

459/// Per-(peer, source) hysteresis for the spatial-layer subscription decision. Pure
460/// (no IO) so the policy is unit-tested in isolation. [`Self::observe`] takes the raw
461/// per-tile demand and returns the **committed** demand the subscribe loop should act
462/// on. The gating is directional:
463///
464/// - a DOWNGRADE (source → low) commits only once the low demand has held for
465///   `debounce` AND the previous switch is at least `dwell` old;
466/// - an UPGRADE (low → source) commits after just [`LAYER_UPGRADE_DEBOUNCE`] and
467///   ignores the dwell, so expanding a tile sharpens it near-instantly;
468/// - a change that does NOT switch tracks (a different cap *value* while already on
469///   the low layer, or no change) passes through immediately.
470///
471/// This is what makes a rejoin's relayout flap — or a slow focus-toggle alternation
472/// — a bounded trickle instead of a resubscribe storm on the shared connection:
473/// every fast upgrade must be preceded by a slow, dwell-gated downgrade.
474#[derive(Debug)]
475pub(super) struct LayerHysteresis {
476    /// The demand we've committed to (what the loop subscribes / reports upstream).
477    committed: hound_media::QualityRequest,
478    /// A demand on the OTHER layer axis, seen but not yet committed, and when first seen.
479    pending: Option<(hound_media::QualityRequest, Instant)>,
480    /// When the last track switch actuated (gates the next one by the dwell).
481    last_switch: Option<Instant>,
482}
483
484/// Whether a quality request selects the low spatial layer (a capped tile) vs the
485/// source track. The track-switch axis the hysteresis debounces.
486fn wants_low_layer(req: hound_media::QualityRequest) -> bool {
487    matches!(
488        req,
489        hound_media::QualityRequest::Capped(_) | hound_media::QualityRequest::Hidden(_)
490    )
491}
492
493impl LayerHysteresis {
494    pub(super) fn new() -> Self {
495        Self {
496            committed: hound_media::QualityRequest::Source,
497            pending: None,
498            last_switch: None,
499        }
500    }
501
502    /// Whether the post-switch dwell has elapsed (a first switch is always free).
503    fn dwell_over(&self, now: Instant, dwell: Duration) -> bool {
504        self.last_switch
505            .is_none_or(|t| now.duration_since(t) >= dwell)
506    }
507
508    /// The (debounce, dwell) that gate committing `want` from the current layer.
509    /// The caller's values are the DOWNGRADE knobs; an upgrade swaps in the short
510    /// [`LAYER_UPGRADE_DEBOUNCE`] and a zero dwell (a zero dwell is always over).
511    fn gates(
512        &self,
513        want: hound_media::QualityRequest,
514        debounce: Duration,
515        dwell: Duration,
516    ) -> (Duration, Duration) {
517        if wants_low_layer(self.committed) && !wants_low_layer(want) {
518            (LAYER_UPGRADE_DEBOUNCE, Duration::ZERO)
519        } else {
520            (debounce, dwell)
521        }
522    }
523
524    /// Feed the freshly-computed per-tile demand; returns the demand to act on now.
525    /// `debounce`/`dwell` gate DOWNGRADES only — see [`Self::gates`].
526    pub(super) fn observe(
527        &mut self,
528        want: hound_media::QualityRequest,
529        now: Instant,
530        debounce: Duration,
531        dwell: Duration,
532    ) -> hound_media::QualityRequest {
533        // A change that stays on the same layer axis (no track switch) is free —
534        // commit it at once. This covers "no change" and a new cap *value* while
535        // already on the low layer (only the reported cap moves, not the subscription).
536        if wants_low_layer(want) == wants_low_layer(self.committed) {
537            self.committed = want;
538            self.pending = None;
539            return self.committed;
540        }
541        // A track switch (source ⇄ low): commit only once `want`'s layer has held for
542        // its debounce and the previous switch is its dwell old — both directional, so
543        // an upgrade needs only ~200 ms and no dwell. A flap back to the committed
544        // layer clears the pending switch via the same-axis branch above.
545        let (debounce, dwell) = self.gates(want, debounce, dwell);
546        match self.pending {
547            Some((p, since))
548                if wants_low_layer(p) == wants_low_layer(want)
549                    && now.duration_since(since) >= debounce
550                    && self.dwell_over(now, dwell) =>
551            {
552                self.committed = want;
553                self.pending = None;
554                self.last_switch = Some(now);
555            }
556            Some((p, since)) if wants_low_layer(p) == wants_low_layer(want) => {
557                // Still waiting; keep the latest target value but hold the timer.
558                self.pending = Some((want, since));
559            }
560            _ => self.pending = Some((want, now)),
561        }
562        self.committed
563    }
564
565    /// When the loop should re-evaluate even without an input change, to let a pending
566    /// switch actuate — the later of the debounce and the dwell expiring, both taken in
567    /// the pending switch's direction (an upgrade's dwell is zero, so it is purely the
568    /// short debounce). `None` when nothing is pending.
569    pub(super) fn pending_deadline(&self, debounce: Duration, dwell: Duration) -> Option<Instant> {
570        self.pending.map(|(want, since)| {
571            let (debounce, dwell) = self.gates(want, debounce, dwell);
572            let due = since + debounce;
573            match self.last_switch {
574                Some(t) => due.max(t + dwell),
575                None => due,
576            }
577        })
578    }
579}

581/// Base restart pace for a reaped pipeline, and the first rung of the escalation
582/// ladder: an UNHEALTHY death (died fast, or never delivered a frame) waits this
583/// long, doubling per consecutive unhealthy death up to [`RESTART_PACE_MAX`].
584const RESTART_PACE: Duration = Duration::from_millis(1000);
585/// Ladder ceiling: a persistently-dead track (producer gone dark, relay routing
586/// wedged) keeps retrying forever — resubscribes are how it eventually heals —
587/// but never faster than this once the ladder tops out.
588const RESTART_PACE_MAX: Duration = Duration::from_secs(20);
589
590/// Pure restart-pacing policy for reaped pipelines (unit-tested). A HEALTHY death
591/// — the pipeline delivered frames and lived ≥ [`RESTART_PACE`] — restarts
592/// immediately and resets the track's ladder (a live stream hit a transient
593/// fault; recover fast). An UNHEALTHY death (instant kill, or a subscription
594/// that never delivered anything) climbs an exponential ladder, so a track the
595/// relay/producer keeps wedging retries forever at a decaying, bounded rate
596/// instead of churning the shared connection. [`Self::ready_at`] is what the
597/// reconcile consults.
598#[derive(Default)]
599pub(super) struct RestartPacer {
600    /// Per-track consecutive unhealthy strikes + the earliest allowed restart.
601    ladders: HashMap<u32, (u32, Instant)>,
602}
603
604impl RestartPacer {
605    /// Record a pipeline death. `delivered` = it produced at least one frame
606    /// downstream; `lived` is how long it ran; `now` is the death time.
607    pub(super) fn on_death(
608        &mut self,
609        track_id: u32,
610        delivered: bool,
611        lived: Duration,
612        now: Instant,
613    ) {
614        if delivered && lived >= RESTART_PACE {
615            self.ladders.remove(&track_id);
616            return;
617        }
618        let strikes = self
619            .ladders
620            .get(&track_id)
621            .map_or(0, |&(s, _)| s.saturating_add(1));
622        let pace = RESTART_PACE
623            .saturating_mul(1u32 << strikes.min(16))
624            .min(RESTART_PACE_MAX);
625        self.ladders.insert(track_id, (strikes, now + pace));
626    }
627
628    /// When `track_id` may restart: `None` = immediately; `Some(t)` = defer to `t`.
629    /// Strikes persist past an elapsed deadline — only a healthy run clears them —
630    /// so a track that dies undelivered on every retry keeps climbing the ladder.
631    pub(super) fn ready_at(&self, track_id: u32, now: Instant) -> Option<Instant> {
632        match self.ladders.get(&track_id) {
633            Some(&(_, t)) if t > now => Some(t),
634            _ => None,
635        }
636    }
637}

639/// Minimum spacing between the recovery keyframe requests a video pipeline sends
640/// outside the subscribe-time nag — on a mid-stream stall or after a
641/// backpressure drop broke decode continuity. The producer coalesces duplicates
642/// through its own throttle; this just keeps a long producer pause from turning
643/// into a control-plane drip.
644const RECOVERY_PLI_MIN_INTERVAL: Duration = Duration::from_secs(2);
645
646/// Rate-limits a pipeline's recovery keyframe requests ([`RECOVERY_PLI_MIN_INTERVAL`]).
647/// Pure; unit-tested.
648#[derive(Default)]
649pub(super) struct PliThrottle {
650    last: Option<Instant>,
651}
652
653impl PliThrottle {
654    /// True when a request may fire now (and records it as fired).
655    pub(super) fn fire(&mut self, now: Instant) -> bool {
656        if self
657            .last
658            .is_some_and(|t| now.duration_since(t) < RECOVERY_PLI_MIN_INTERVAL)
659        {
660            return false;
661        }
662        self.last = Some(now);
663        true
664    }
665}

667/// Max consecutive frames concealed with Opus PLC on a timestamp gap. A gap of up to
668/// this many frames (~100 ms at 20 ms/frame) is plausibly recoverable brief loss worth
669/// concealing; beyond that the gap is the sender being silent (DTX emits no frames) or
670/// a capture discontinuity — fabricating extrapolated audio there is wrong (it mislabels
671/// silence as loss and splices garbage into an active stream). Silence is played across
672/// the larger gap instead.
673pub(super) const MAX_PLC_FRAMES: i64 = 5;
674
675/// Frames genuinely missing in a `gap_us` jump between consecutive in-order frame
676/// timestamps (frame size `frame_us`). Tolerant of sub-frame capture-timing jitter:
677/// SCO/resampled sources never space frames at exactly 20 ms, so a gap under ~1.5
678/// frames is the next frame arriving a little late, NOT a loss. Counting it would
679/// fabricate one spurious PLC frame per real frame (concealment dwarfing real audio).
680/// Pure + unit-tested. The caller separately caps the result at [`MAX_PLC_FRAMES`].
681pub(super) fn frames_missing(gap_us: i64, frame_us: i64) -> i64 {
682    if frame_us <= 0 || gap_us <= 0 {
683        return 0;
684    }
685    let frames = gap_us as f64 / frame_us as f64;
686    if frames < 1.5 {
687        0
688    } else {
689        (frames.round() as i64 - 1).max(0)
690    }
691}
692
693/// How many decoded frames per cadence-diagnostic window (~5 s at 50 fps).
694const CADENCE_WINDOW_FRAMES: u64 = 250;
695
696/// Per-stream receive-cadence diagnostic (a retained debugging tool). Accumulates over a
697/// window and, on completion, reports the signals that explain an active-stream gap the
698/// smoothed jitter number hides — see the call site. Pure bookkeeping; no IO.
699#[derive(Default)]
700pub(super) struct CadenceDiag {
701    real: u64,
702    pub(super) plc: u64,
703    pub(super) big_gaps: u64,
704    max_gap_us: i64,
705    first_ts: Option<i64>,
706    first_now: i64,
707}
708
709impl CadenceDiag {
710    /// Fold a timestamp gap between consecutive in-order frames.
711    pub(super) fn note_gap(&mut self, gap_us: i64) {
712        if gap_us > self.max_gap_us {
713            self.max_gap_us = gap_us;
714        }
715    }
716
717    /// Record one decoded frame at sender timestamp `ts` / local time `now`. Returns
718    /// `Some((real, plc, big_gaps, max_gap_ms, skew_ppm))` once per window, then resets.
719    /// `skew_ppm` > 0 ⇒ the local clock advanced faster than the sender's timestamps
720    /// over the window (receiver consumes faster than sender produces → buffer drains).
721    pub(super) fn observe(&mut self, ts: i64, now: i64) -> Option<(u64, u64, u64, i64, i64)> {
722        if self.first_ts.is_none() {
723            self.first_ts = Some(ts);
724            self.first_now = now;
725        }
726        self.real += 1;
727        if self.real < CADENCE_WINDOW_FRAMES {
728            return None;
729        }
730        let ts_elapsed = ts - self.first_ts.unwrap_or(ts);
731        let now_elapsed = now - self.first_now;
732        let skew_ppm = if ts_elapsed > 0 {
733            ((now_elapsed - ts_elapsed) as f64 / ts_elapsed as f64 * 1_000_000.0) as i64
734        } else {
735            0
736        };
737        let out = (
738            self.real,
739            self.plc,
740            self.big_gaps,
741            self.max_gap_us / 1000,
742            skew_ppm,
743        );
744        *self = CadenceDiag::default();
745        Some(out)
746    }
747}

1703        assert!(!w.observe((1280, 720), t0, DEBOUNCE));
