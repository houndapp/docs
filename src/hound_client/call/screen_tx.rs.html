1//! Outbound screen share: open → `!Send` capture/encode thread → publish, plus
2//! the LOW simulcast rung and the classified exit protocol back to the
3//! supervisor ([`ScreenCaptureExit`]: a clean end vs a held resize vs a
4//! transient fault — a share must NOT end on a resize or a glitch). Static
5//! content is re-emitted on a slow cadence for fresh subscribers, and an idle
6//! source is liveness-probed so a terminated window ends the share instead of
7//! freezing on its last frame — and a capture that never delivers a FIRST frame is
8//! watchdogged into a restart rather than idling behind a black preview forever.

50/// Why the `!Send` screen capture/encode thread exited on its own — routed back to the
51/// supervisor as a [`ScreenCommand`] by the `notify` callback it was spawned with.
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub(super) enum ScreenCaptureExit {
54    /// A genuine end: window closed / OS "stop sharing" / source gone. The supervisor
55    /// stops the pipeline, `Unpublish`es, and drops the desired source (clean stop).
56    Ended,
57    /// The captured content size departed its baseline and HELD past the debounce: the
58    /// shared window was resized. The supervisor restarts the share at the new geometry.
59    Resized,
60    /// A transient `next_frame` error/panic. The supervisor restarts the share in place
61    /// (re-opening cleanly); a restart whose re-open then fails degrades to `Ended`.
62    Retry,
63}
64
65/// The capture thread's exit channel back to the supervisor — a cheap closure that maps
66/// the [`ScreenCaptureExit`] onto a [`ScreenCommand`] and sends it.
67///
68/// The `bool` is **`delivered_any`**: whether THIS incarnation ever captured a real
69/// frame. It is the supervisor's only way to tell a share that broke mid-stream (worth
70/// restarting, indefinitely) from one that never produced anything (restarting it again
71/// runs the same failed experiment) — see `policy::screen_restarts_exhausted`. Only the
72/// capture loop knows it; nothing downstream can infer it, because a source that never
73/// delivers looks exactly like a static share on the wire.
74pub(super) type ScreenExitNotify = Arc<dyn Fn(ScreenCaptureExit, bool) + Send + Sync>;

76/// `Unpublish` our screen video + audio tracks on the control plane (best-effort) so
77/// the roster drops them for everyone. Resolved from the published `CallView` (the
78/// same roster `own_track` reads) against the live session's outbound.
79pub(super) fn unpublish_screen_tracks(live: &LiveSession, params: &ConnectionParams) {
80    let me = live.self_id.0;
81    let ids: Vec<hound_control::TrackId> = {
82        let view = params.channels.view.borrow();
83        view.tracks
84            .iter()
85            .filter(|t| t.peer == me && t.source == Source::Screen)
86            .map(|t| hound_control::TrackId(t.id))
87            .collect()
88    };
89    for id in ids {
90        let _ = live
91            .control_outbound
92            .send(ClientMessage::Unpublish { track: id });
93    }
94}

98/// What the `!Send` screen capture thread hands back once the capture is open: the
99/// app-audio leg, if the engine produced one (a window source on a platform with
100/// app-audio support). The `!Send` `ScreenShare` video handle stays on the capture
101/// thread; only this audio leg crosses to its own thread.
102struct ScreenOpened {
103    audio: Option<hound_engine::api::ScreenAudio>,
104    /// The negotiated capture geometry (post native→target scale) — what the
105    /// capture thread's encoder was ACTUALLY built at. Threaded to the async video
106    /// publish task so its encoder-rebuild geometry dwell (`hound_media::GeometryDwell`)
107    /// seeds from the real starting geometry rather than guessing a 16:9 default (a
108    /// screen source is not always 16:9).
109    dims: Dimensions,
110}

112/// Map an engine screen-capture open error onto a typed [`HoundError`] the UI can
113/// route — a Screen-Recording TCC denial gets a distinct, actionable message, and a
114/// source that opened but produced nothing gets the same code as the supervisor's
115/// dry-restart give-up ([`MediaCode::ScreenCaptureNoFrames`]) rather than the
116/// `ScreenCaptureOpen` catch-all: it is the SAME failure (a live handle over a dead
117/// source), caught at open instead of by the watchdog, and one code per cause is what
118/// keeps the two from landing in different Sentry issues.
119fn map_screen_capture_error(e: hound_engine::error::CaptureError) -> HoundError {
120    use hound_engine::error::CaptureError;
121    use hound_engine::types::DeviceKind;
122    match e {
123        CaptureError::PermissionDenied {
124            kind: DeviceKind::Screen,
125        } => HoundError::media(
126            MediaCode::ScreenPermissionDenied,
127            "screen recording permission denied — enable it in System Settings → \
128             Privacy & Security → Screen Recording",
129        ),
130        e @ CaptureError::NoFrames => HoundError::media(
131            MediaCode::ScreenCaptureNoFrames,
132            format!("open screen capture: {e}"),
133        ),
134        other => HoundError::media(
135            MediaCode::ScreenCaptureOpen,
136            format!("open screen capture: {other}"),
137        ),
138    }
139}

141/// Start the screen-share publish pipeline against the live relay session: spawn the
142/// `!Send` capture+encode thread, await its open (surfacing a permission/unsupported
143/// error), announce the `screen` video track (+ `screen_audio` when the source has an
144/// app-audio leg), and spawn the publish task(s). Returns the handle that stops it.
145///
146/// Mirrors [`spawn_cam_publish`] with screen-tuned differences: the encode geometry
147/// is the source aspect fitted to 1080p-class (not the fixed cam size); a paired
148/// app-audio leg is published on its own track; and a *runtime* capture end (window
149/// closed / OS "stop sharing") just ends the track rather than being an error.
150///
151/// The screen adaptation reads the sharer's [`SharedScreenQuality`] each tick and
152/// reconfigures its ladder live when it changes; the capture thread classifies
153/// its exit (window close = `Ended`, a held resize = `Resized`, a transient fault =
154/// `Retry`) and reports it via `notify` so the supervisor restarts (resize/retry) or
155/// stops (ended) the share — a share never ends on a resize or a transient glitch.
156#[allow(clippy::too_many_arguments)]
157pub(super) async fn spawn_screen_publish(
158    relay: Arc<MoqRelaySession>,
159    space_id: Uuid,
160    self_user_id: Uuid,
161    self_id: PeerId,
162    clock: CaptureClock,
163    source: hound_engine::types::ScreenSource,
164    sink: Arc<dyn VideoSink>,
165    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
166    rtt_conn: iroh::endpoint::Connection,
167    view_rx: watch::Receiver<SessionView>,
168    keyframe_rx: broadcast::Receiver<TrackId>,
169    feedback_rx: broadcast::Receiver<MediaFeedback>,
170    quality_rx: broadcast::Receiver<QualityReport>,
171    screen_quality: SharedScreenQuality,
172    notify: ScreenExitNotify,
173    stats: Arc<CallStats>,
174    mute: Arc<MuteFlags>,
175) -> Result<ScreenShareHandle> {
176    let capture_stop = Arc::new(AtomicBool::new(false));
177    let (vid_tx, vid_rx) = tokio::sync::mpsc::channel::<WireFrame>(8);
178    let (ready_tx, ready_rx) = tokio::sync::oneshot::channel::<Result<ScreenOpened>>();
179
180    // Negotiate the publish codec against the live roster (HW-encode ∩ all-peers
181    // HW-decode): AV1 → H.265 → H.264. The capture/encode thread builds this codec's
182    // encoder; we announce the same choice below. `encode_caps` is probed once (it
183    // spins a throwaway VT session) and reused by the mid-call re-negotiator.
184    // Same bounded wait as the control-plane `Hello` (see `control::connect`): a
185    // share started while the first-run codec download is still in flight would
186    // otherwise negotiate as if this device had no software H.264.
187    crate::openh264::settle().await;
188    let encode_caps = Engine::new().encode_caps();
189    let codec = negotiate_codec(encode_caps, &view_rx.borrow());
190
191    // The shared adaptation seam: the screen video tick (in the publish task) +
192    // keyframe/feedback listeners write it; the capture/encode thread reads it.
193    let adapt = Arc::new(AdaptShared::default());
194    // Seed the codec slot so the encode loop's rebuild check is a no-op until a
195    // roster change actually moves it.
196    adapt
197        .target_codec
198        .store(codec_rank(codec), Ordering::Relaxed);
199
200    // Demand-driven simulcast control for the screen source + the LOW-layer channel
201    // the capture thread feeds. `low_active` is the capture thread's gate; it stays
202    // `None` (low path inert) until a receiver demands the low layer AND actuation is
203    // on. Created here so it can be handed to the capture thread below.
204    let layers = Arc::new(LayerControl::new(SCREEN_BITRATE.bps() as u64));
205    let low_active = layers.low_active.subscribe();
206    let (low_vid_tx, low_vid_rx) = tokio::sync::mpsc::channel::<WireFrame>(8);
207    // A low-track PLI sets this; the capture thread forces a low IDR so a receiver
208    // subscribing to the low layer resyncs promptly (not on the slow safety-net IDR).
209    let low_keyframe = Arc::new(AtomicBool::new(false));
210    // The low-publish task's own view of the gate: it withdraws the `_l0` decl from
211    // the roster when the layer deactivates (subscribed before `layers` is moved).
212    let low_active_decl = layers.low_active.subscribe();
213    // The LOW encoder's live codec (`codec_rank`), written by the capture thread on
214    // every low build and read by the publish task when it announces the `_l0` decl.
215    // Shared rather than captured because the announce lands on the first low frame,
216    // which can be long after a mid-share codec downgrade — see
217    // `spawn_low_video_publish`.
218    let low_codec = Arc::new(AtomicU8::new(codec_rank(codec)));
219
220    // The `!Send` capture+encode thread owns the engine `ScreenShare` + H.264 encoder
221    // for its whole life (the `!Send` contract); it reports open readiness and hands
222    // back the audio leg via `ready_tx`, and classifies + reports its exit via `notify`.
223    {
224        let stop = capture_stop.clone();
225        let adapt = adapt.clone();
226        let notify = notify.clone();
227        let stats = stats.clone();
228        let low_keyframe_capture = low_keyframe.clone();
229        let low_codec_capture = low_codec.clone();
230        std::thread::Builder::new()
231            .name("call-screen-capture-encode".into())
232            .spawn(move || {
233                screen_capture_encode_loop(
234                    stop,
235                    clock,
236                    source,
237                    self_id,
238                    sink,
239                    vid_tx,
240                    ready_tx,
241                    adapt,
242                    notify,
243                    stats,
244                    codec,
245                    low_active,
246                    low_vid_tx,
247                    low_keyframe_capture,
248                    low_codec_capture,
249                )
250            })
251            .ok();
252    }
253
254    // Await the open — a permission/unsupported/bad-source error surfaces to the
255    // caller (so the UI can show the TCC hint) rather than failing silently.
256    let opened = match ready_rx.await {
257        Ok(Ok(o)) => o,
258        Ok(Err(e)) => return Err(e),
259        Err(_) => {
260            return Err(HoundError::media(
261                MediaCode::ScreenCaptureThreadExited,
262                "screen capture thread exited before reporting readiness",
263            ));
264        }
265    };
266
267    // Announce the screen VIDEO track in the negotiated codec.
268    control_outbound
269        .send(ClientMessage::Publish {
270            track: TrackDecl {
271                kind: MediaKind::Video,
272                source: Source::Screen,
273                codec: video_codec_to_ctl(codec),
274                low: false,
275            },
276        })
277        .map_err(|_| {
278            HoundError::media(
279                MediaCode::ControlClosedBeforePublish,
280                "control plane closed before screen publish",
281            )
282        })?;
283
284    let mut tasks: Vec<JoinHandle<()>> = vec![
285        // Producer-side PLI relay + receiver→producer BWE feedback, scoped to the
286        // SCREEN track. Their tasks join the handle so they tear down with the share.
287        spawn_keyframe_listener(
288            keyframe_rx,
289            view_rx.clone(),
290            self_id,
291            Source::Screen,
292            adapt.clone(),
293            Some(low_keyframe.clone()),
294        ),
295        spawn_feedback_listener(
296            feedback_rx,
297            view_rx.clone(),
298            self_id,
299            Source::Screen,
300            adapt.clone(),
301            layers.clone(),
302        ),
303        // Demand-driven simulcast: aggregate receiver quality requests for the screen
304        // track into the layer planner (drives the low layer's `low_active` gate).
305        // Dark until `SIMULCAST_ACTUATE`.
306        spawn_quality_listener(quality_rx, view_rx.clone(), self_id, Source::Screen, layers),
307        // LOW-layer publish: drains the capture thread's low encode and publishes it
308        // on the `screen_l0` track, announcing it on the first frame (so it's
309        // invisible to receivers until actually producing) and withdrawing the decl
310        // again when the layer is torn down (so no receiver picks a dead layer).
311        spawn_low_video_publish(
312            relay.clone(),
313            broadcast_path(space_id, self_user_id, self_id, "screen_l0"),
314            "screen_l0",
315            Source::Screen,
316            low_codec,
317            low_vid_rx,
318            control_outbound.clone(),
319            view_rx.clone(),
320            self_id,
321            low_active_decl,
322        ),
323        // Re-negotiate the codec as the roster changes mid-share (a less-capable peer
324        // joining forces a downgrade so they get video; one leaving lets us climb
325        // back).
326        spawn_codec_renegotiator(view_rx, encode_caps, adapt.clone(), codec, "screen"),
327        spawn_screen_video_publish(
328            relay.clone(),
329            broadcast_path(space_id, self_user_id, self_id, SCREEN_TRACK),
330            vid_rx,
331            adapt,
332            opened.dims,
333            rtt_conn,
334            screen_quality,
335            stats.clone(),
336            codec,
337        ),
338    ];
339
340    // App-audio leg: only when the engine returned one (window + app-audio support).
341    if let Some(audio) = opened.audio {
342        let _ = control_outbound.send(ClientMessage::Publish {
343            track: TrackDecl {
344                kind: MediaKind::Audio,
345                source: Source::Screen,
346                codec: CtlCodec::Opus,
347                low: false,
348            },
349        });
350        let (pcm_tx, pcm_rx) = tokio::sync::mpsc::channel::<hound_engine::types::PcmFrame>(8);
351        let stop = capture_stop.clone();
352        std::thread::Builder::new()
353            .name("call-screen-audio-capture".into())
354            .spawn(move || screen_audio_capture_loop(stop, audio, pcm_tx))
355            .ok();
356        tasks.push(spawn_screen_audio_publish(
357            relay,
358            broadcast_path(space_id, self_user_id, self_id, SCREEN_AUDIO_TRACK),
359            self_id.0 as u16,
360            pcm_rx,
361            stats.clone(),
362            mute,
363        ));
364    }
365
366    Ok(ScreenShareHandle {
367        capture_stop,
368        tasks,
369    })
370}

372/// The `!Send` screen capture+encode thread: open the engine `ScreenShare` (+ maybe
373/// `ScreenAudio`) and the H.264 encoder HERE (so neither crosses a thread boundary),
374/// report readiness (handing back the audio leg), then pull → self-preview tee →
375/// encode → forward. A static source produces no frames, so on an idle poll the last
376/// frame is re-emitted (timestamp advanced) at the static-refresh cadence so a fresh
377/// subscriber isn't stuck on a black tile.
378///
379/// The loop classifies its exit and reports it via `notify`:
380/// * a window CLOSE / OS "stop sharing" / source gone (`next_frame` errs cleanly) →
381///   [`ScreenCaptureExit::Ended`];
382/// * a window RESIZE that holds past [`SCREEN_RESIZE_DEBOUNCE`] (the captured content
383///   size departs its baseline and stays) → [`ScreenCaptureExit::Resized`];
384/// * a transient `next_frame` error/panic → [`ScreenCaptureExit::Retry`] (after a brief
385///   [`SCREEN_RETRY_BACKOFF`]);
386/// * a capture that never delivers its FIRST frame within
387///   [`SCREEN_FIRST_FRAME_DEADLINE`] → [`ScreenCaptureExit::Retry`] as well — the one
388///   failure with no error and no dead source to detect (see the idle branch).
389///
390/// NB `next_frame_timeout` errors split two ways: `SourceEnded` is the engine's typed
391/// clean end (window closed / OS "Stop Sharing"), and every other `CaptureError` is a
392/// fault → `Retry`: the supervisor's restart re-opens cleanly, and a re-open that then
393/// fails (the source is truly gone) degrades to a clean `Ended` there. `notify` is NOT
394/// fired when `stop` was latched (a deliberate stop/leave/reconnect — the supervisor
395/// already owns that transition).
396#[allow(clippy::too_many_arguments)]
397fn screen_capture_encode_loop(
398    stop: Arc<AtomicBool>,
399    clock: CaptureClock,
400    source: hound_engine::types::ScreenSource,
401    self_id: PeerId,
402    sink: Arc<dyn VideoSink>,
403    vid_tx: tokio::sync::mpsc::Sender<WireFrame>,
404    ready_tx: tokio::sync::oneshot::Sender<Result<ScreenOpened>>,
405    adapt: Arc<AdaptShared>,
406    notify: ScreenExitNotify,
407    stats: Arc<CallStats>,
408    codec: VideoCodec,
409    // Demand-driven LOW simulcast layer: `Some(bps)` ⇒ also downscale + encode this
410    // capture at whatever screen rung `bps` buys (`low_rung_for_cap` fitted to the
411    // capture aspect by `policy::low_screen_dims`) and forward on `low_vid_tx` (the
412    // `screen_l0` track). Stays `None` (inert, source path byte-identical) until a
413    // receiver demands it AND `SIMULCAST_ACTUATE` is on.
414    low_active: watch::Receiver<Option<u64>>,
415    low_vid_tx: tokio::sync::mpsc::Sender<WireFrame>,
416    low_keyframe: Arc<AtomicBool>,
417    // The low encoder's live codec (`codec_rank`) for the `screen_l0` publish task's
418    // decl announce — written here on every low build, since the ladder may land a
419    // rung below the negotiated target.
420    low_codec: Arc<AtomicU8>,
421) {
422    use hound_engine::types::{ScreenCaptureConfig, TimestampUs};
423
424    // Capture native, scaled to the 1080p-class cap preserving aspect; request app
425    // audio (the engine ignores it for a display / unsupported platform → `None`).
426    let config = ScreenCaptureConfig {
427        dimensions: Dimensions::even_clamped(SCREEN_MAX_W, SCREEN_MAX_H),
428        fps: SCREEN_FPS,
429        capture_audio: true,
430    };
431    let (mut screen, screen_audio) = match Engine::new().open_screen_capture(&source, config, clock)
432    {
433        Ok(v) => v,
434        Err(e) => {
435            let _ = ready_tx.send(Err(map_screen_capture_error(e)));
436            return;
437        }
438    };
439    // Build the encoder for the delivered geometry (post native→target scale), in
440    // the negotiated codec, descending the ladder if it won't open. The wire-codec
441    // byte is taken from the encoder actually built, so frames stay correctly tagged.
442    let (neg_dims, _fmt) = screen.negotiated();
443    let mut encoder = match open_encoder_laddered(
444        codec,
445        neg_dims,
446        SCREEN_FPS,
447        SCREEN_BITRATE,
448        SCREEN_KEYFRAME_CADENCE,
449    ) {
450        Some(e) => e,
451        None => {
452            let _ = ready_tx.send(Err(HoundError::media(
453                MediaCode::NoVideoEncoder,
454                "no usable video encoder (even H.264 failed to open)",
455            )));
456            return;
457        }
458    };
459    // The codec the encoder is actually producing (may be a lower rung than `codec`
460    // if the ladder fell through) — used for the log/stat label. `last_target` is the
461    // negotiated *target* the live encoder was BUILT for; the rebuild check keys off
462    // it changing, NOT off `current_codec`, so a target the device can't build
463    // (AV1 → built H.265) doesn't re-trigger a rebuild every frame.
464    let mut current_codec = {
465        use hound_engine::backend::VideoEncode;
466        encoder.codec()
467    };
468    let mut last_target = codec;
469    // Throttles the retry of a rebuild that found no usable encoder. `last_target` is
470    // committed only on a successful open, so the pending switch survives a failure —
471    // this is what stops the surviving switch from re-attempting per frame.
472    let mut rebuild_retry = RebuildRetry::default();
473    // The geometry the live `encoder` was actually built at — starts at `neg_dims`
474    // (the negotiated capture geometry) and only moves when the adaptation's
475    // dwelled rung geometry (`adapt.encoder_w/h`) settles on something different.
476    let mut enc_dims = neg_dims;
477    {
478        use hound_engine::backend::VideoEncode;
479        tracing::info!(
480            target: "video_encode",
481            track = "screen",
482            codec = ?encoder.codec(),
483            backend = encoder.backend(),
484            width = neg_dims.width(),
485            height = neg_dims.height(),
486            "video: encode path selected"
487        );
488    }
489    // Hand the audio leg out (for its own thread) and report a successful open.
490    let _ = ready_tx.send(Ok(ScreenOpened {
491        audio: screen_audio,
492        dims: neg_dims,
493    }));
494
495    // RTC-debug: the screen pipeline is publishing at this geometry (sets enabled).
496    stats.set_screen_dims(neg_dims.width(), neg_dims.height());
497
498    let key = screen_key(self_id);
499    let sender = self_id.0 as u16;
500    let mut frame_seq = 0u32;
501    let mut last_frame: Option<hound_engine::types::RawFrame> = None;
502    let mut last_emit = std::time::Instant::now();
503    // The encoder bitrate we last applied (a steady target costs no syscall).
504    let mut applied_bps = SCREEN_BITRATE.bps();
505    // Downsamples capture to the chosen rung's fps; the static-refresh re-emit rides
506    // the forced-keyframe bypass in `encode_and_publish_adaptive` so an idle share
507    // never freezes the gate.
508    let mut fps_gate = hound_media::FrameRateGate::new();
509    let mut fail_streak = EncodeFailStreak::default();
510
511    // LOW simulcast layer state — mirrors the source's, at whatever rung the
512    // coalesced receiver cap buys. Built lazily the first frame `low_active` is
513    // `Some`, torn down when it goes `None`, rebuilt on a codec switch or a SETTLED
514    // geometry change (the camera low path's pattern — `camera_tx`).
515    //
516    // The screen ladder has two below-source rungs (`SCREEN_LOW_RUNG` 360p15,
517    // `SCREEN_MID_RUNG` 720p30), so neither the geometry nor the frame rate is fixed
518    // at open any more. Both come from the rung the cap inverts to, and both differ
519    // from the camera's derivation in one way each:
520    //
521    // * geometry is fitted by AREA (`policy::low_screen_dims`), because a share can
522    //   be any shape and it is the rung's PRICE that must be held — a near-square
523    //   window fitted by height is mispriced by 1.78×;
524    // * the dwell therefore debounces the area-fitted rung (`RungTarget::Pixels`),
525    //   so it can only ever commit to something the producer would actually build.
526    //
527    // The rebuild key is `(geometry, rung)` — `camera_tx`'s, verbatim — and NOT the
528    // geometry alone: the two screen rungs collapse to one geometry whenever the
529    // shared window is already smaller than both pixel budgets, and there they still
530    // differ by 15 ↔ 30 fps and 200 kbps ↔ 1.1 Mbps, neither of which a live encoder
531    // can be retargeted through. See the key itself for the full argument.
532    let mut low_encoder: Option<hound_engine::api::VideoEncoder> = None;
533    let mut low_last_target: Option<VideoCodec> = None;
534    let mut low_applied_bps = 0u32;
535    let mut low_frame_seq = 0u32;
536    let mut low_fps_gate = hound_media::FrameRateGate::new();
537    let mut low_fail_streak = EncodeFailStreak::default();
538    let low_adapt = Arc::new(AdaptShared::default());
539    // Created on activation (seeded at the rung the first cap buys, so the encoder
540    // opens there at once) and dropped with the layer, so a re-activation re-seeds
541    // rather than inheriting a stale committed geometry.
542    let mut low_geometry: Option<hound_media::GeometryDwell> = None;
543    // The (geometry, rung) the live low encoder was actually built at — the rebuild
544    // trigger, and the fps gate's anchor while a change is still dwelling.
545    let mut low_rung: Option<(Dimensions, hound_media::VideoRung)> = None;
546
547    // How this loop exits: classified for the supervisor. Defaults to `Ended` (a clean
548    // capture end) and is upgraded to `Resized`/`Retry` at the break that detected it.
549    let mut exit = ScreenCaptureExit::Ended;
550    // Resize watch: the capture's pixel geometry is fixed at open, so when the per-frame
551    // content size departs its baseline and HOLDS (debounced past live drag-resizes) we
552    // exit with `Resized` and the supervisor restarts at the new geometry.
553    let mut resize = ResizeWatch::default();
554    // Source liveness: damage-driven capture goes SILENT (not errored) when the
555    // shared window's app terminates — indistinguishable from static content by
556    // frame flow alone, so the static-refresh would re-emit the last frame forever.
557    // While idle past SOURCE_LIVENESS_IDLE, probe (throttled) that the source still
558    // exists and END the share when it is gone.
559    let mut last_real_frame = std::time::Instant::now();
560    let mut last_probe = std::time::Instant::now();
561    // First-frame watchdog origin: a source that opens but never delivers frame #1
562    // is invisible to both the error classification and the liveness probe, so the
563    // idle branch times it out against this (`policy::first_frame_overdue`).
564    let loop_started = std::time::Instant::now();
565
566    'capture: while !stop.load(Ordering::Relaxed) {
567        // A held window resize → restart the share at the new geometry.
568        if let Some(cur) = screen.content_size()
569            && resize.observe(cur, std::time::Instant::now(), SCREEN_RESIZE_DEBOUNCE)
570        {
571            tracing::info!("call: shared window resized — restarting share at new geometry");
572            exit = ScreenCaptureExit::Resized;
573            break 'capture;
574        }
575
576        let pulled = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
577            screen.next_frame_timeout(SCREEN_IDLE_POLL)
578        }));
579        match pulled {
580            Ok(Ok(Some(f))) => {
581                // RTC-debug: one real screen frame captured (not a static re-emit).
582                stats.screen_captured();
583                last_frame = Some(f);
584                last_real_frame = std::time::Instant::now();
585            }
586            Ok(Ok(None)) => {
587                // No frame has EVER arrived and the deadline lapsed: the source
588                // opened but never produced (Windows WGC can stay silent forever).
589                // Nothing else in this loop catches it — there is no error to
590                // classify, the liveness probe sees a live handle, and with no
591                // cached frame the re-emit below is a bare `continue`, so the share
592                // would idle behind a black self-preview at 0 fps for the rest of
593                // the call.
594                //
595                // `Retry`, not `Ended`: the supervisor's restart RE-OPENS the
596                // capture, which is the only thing that can fix a source stuck
597                // before its first frame; a re-open that then fails degrades to a
598                // clean `Ended` there (the same argument as the transient-fault arm
599                // below). The backoff bounds the restart rate on a source that keeps
600                // opening and never delivering.
601                //
602                // This cannot false-positive on a genuinely static share: a static
603                // source still delivered frame #1 (which is what the re-emit path
604                // exists to republish), so `last_frame` is `Some` and the predicate
605                // is inert however long the content then stays unchanged.
606                if first_frame_overdue(last_frame.is_some(), loop_started.elapsed()) {
607                    tracing::warn!("call: screen capture never delivered a frame — restarting");
608                    exit = ScreenCaptureExit::Retry;
609                    std::thread::sleep(SCREEN_RETRY_BACKOFF);
610                    break 'capture;
611                }
612                // Idle: first make sure the source still EXISTS — a terminated
613                // window delivers no frames and no error, and re-emitting its last
614                // frame forever would freeze every receiver's tile on stale content.
615                // Gone ⇒ clean end (the supervisor unpublishes; every tile drops).
616                if source_probe_due(last_real_frame.elapsed(), last_probe.elapsed()) {
617                    last_probe = std::time::Instant::now();
618                    if !screen.source_alive() {
619                        tracing::info!("call: shared source no longer exists — ending share");
620                        exit = ScreenCaptureExit::Ended;
621                        break 'capture;
622                    }
623                }
624                // Idle (static source): re-emit the cached frame when a keyframe is
625                // owed (new subscriber / PLI) or the static-refresh cadence lapsed,
626                // with an advanced timestamp so playout stays monotonic. Either
627                // layer's owed IDR counts — a low-layer subscriber on a static share
628                // would otherwise wait out SCREEN_STATIC_REFRESH for the frame that
629                // carries its keyframe.
630                // A source IDR is only "owed" once the source encode gate is open —
631                // while dark nothing consumes `force_keyframe`, so counting it here
632                // would re-emit the cached frame on every idle poll forever.
633                let owed_keyframe = (adapt.source_encode_open()
634                    && adapt.force_keyframe.load(Ordering::Relaxed))
635                    || low_keyframe.load(Ordering::Relaxed);
636                if !owed_keyframe && last_emit.elapsed() < SCREEN_STATIC_REFRESH {
637                    continue;
638                }
639                let Some(cached) = last_frame.as_mut() else {
640                    continue; // nothing captured yet
641                };
642                let advance = cached
643                    .captured_at
644                    .elapsed()
645                    .as_micros()
646                    .min(i64::MAX as u128) as i64;
647                cached.timestamp_us =
648                    TimestampUs::from_micros(cached.timestamp_us.micros() + advance);
649                cached.captured_at = std::time::Instant::now();
650            }
651            // SCK reported the stream STOPPED (window closed / OS "Stop Sharing"):
652            // a clean, typed end — never a retry, which would either fail to reopen
653            // or freeze on the last frame.
654            Ok(Err(hound_engine::error::CaptureError::SourceEnded)) => {
655                tracing::info!("call: capture source ended — ending share");
656                exit = ScreenCaptureExit::Ended;
657                break 'capture;
658            }
659            // Any other `next_frame` error or a pull-callback panic is a TRANSIENT
660            // capture fault (a mid-stream disconnect, a poisoned lock). Don't
661            // permanently end the share — signal `Retry` so the supervisor restarts it
662            // in place; a restart whose re-open then fails (the source is genuinely
663            // gone) degrades to a clean `Ended` there. A brief backoff bounds the
664            // restart rate on a repeated fault.
665            Ok(Err(e)) => {
666                tracing::warn!(%e, "call: screen capture error — restarting share");
667                exit = ScreenCaptureExit::Retry;
668                std::thread::sleep(SCREEN_RETRY_BACKOFF);
669                break 'capture;
670            }
671            Err(_panic) => {
672                tracing::warn!("call: screen capture panicked — restarting share");
673                exit = ScreenCaptureExit::Retry;
674                std::thread::sleep(SCREEN_RETRY_BACKOFF);
675                break 'capture;
676            }
677        }
678        let Some(frame) = last_frame.as_ref() else {
679            continue;
680        };
681        last_emit = std::time::Instant::now();
682
683        // Self-preview tee (keyed `screen:{self}`, matching the subscribe key
684        // scheme). Runs even while paused so the sharer always sees what they're
685        // sharing. Surface-first: SCK frames are NV12 IOSurfaces, so the preview
686        // renders zero-copy — an always-I420 path would convert + copy ~3 MB per
687        // frame (~90 MB/s at 1080p30) just for the sharer's own preview.
688        push_preview(sink.as_ref(), &key, frame);
689
690        // AudioOnly: the ladder dropped screen video under pressure. Keep pulling the
691        // source (resume is instant) but skip encode/publish. Screen audio (if any)
692        // keeps flowing underneath.
693        if adapt.paused.load(Ordering::Relaxed) {
694            continue;
695        }
696
697        // Subscriber-gated SOURCE encode — the camera path's gate, applied to the
698        // share. A screenshare encode is the most expensive thing this client does;
699        // running it while the publish loop's live gate is still dark burns a full
700        // 1080p encode per frame on output nobody receives. Capture, the sharer's own
701        // preview, the liveness probe and the LOW rung below all keep running. Going
702        // live arms `force_keyframe`, so the first encoded frame is an IDR.
703        if adapt.source_encode_open() {
704            // Mid-call codec switch (the re-negotiator moved the target — a peer joined
705            // who can't decode us → downgrade; or one left → upgrade) AND/OR mid-share
706            // resolution-rung change (the adaptation tick's DWELLED geometry —
707            // `adapt.encoder_w/h`, see `hound_media::GeometryDwell` — settled on
708            // something different from what's built). Rebuild the encoder to match
709            // either, at the live bitrate; `0` on the geometry atomics means "no
710            // adaptation decision yet", so fall back to the current `enc_dims` rather
711            // than a bogus 0×0 rebuild. Either change forces a fresh IDR
712            // (force_keyframe) so every receiver re-syncs — a decoder can't switch
713            // codec OR resolution mid-GOP.
714            //
715            // A failed open keeps the current encoder and leaves the switch PENDING
716            // (`last_target`/`enc_dims` move only on success), so it is retried on the
717            // `RebuildRetry` cadence. Committing the target up-front instead makes one
718            // transient failure permanent: a downgrade forced by a peer who cannot
719            // decode the live codec would never be re-attempted, and that peer sits at
720            // 0 fps for the rest of the share with nothing left to renegotiate.
721            let want_codec = codec_from_rank(adapt.target_codec.load(Ordering::Relaxed));
722            let (want_w, want_h) = (
723                adapt.encoder_w.load(Ordering::Relaxed),
724                adapt.encoder_h.load(Ordering::Relaxed),
725            );
726            let want_dims = if want_w > 0 && want_h > 0 {
727                Dimensions::even_clamped(want_w, want_h)
728            } else {
729                enc_dims
730            };
731            let rebuild_due = want_codec != last_target || want_dims != enc_dims;
732            if rebuild_due && rebuild_retry.ready(Instant::now()) {
733                match open_encoder_laddered(
734                    want_codec,
735                    want_dims,
736                    SCREEN_FPS,
737                    Bitrate::from_bps(applied_bps),
738                    SCREEN_KEYFRAME_CADENCE,
739                ) {
740                    Some(e) => {
741                        use hound_engine::backend::VideoEncode;
742                        let got = e.codec();
743                        tracing::info!(
744                            track = "screen",
745                            from_codec = ?current_codec,
746                            to_codec = ?got,
747                            from_dims = %enc_dims,
748                            to_dims = %want_dims,
749                            "call: rebuilding encoder mid-share (codec/geometry)"
750                        );
751                        encoder = e;
752                        current_codec = got;
753                        last_target = want_codec;
754                        enc_dims = want_dims;
755                        rebuild_retry.succeeded();
756                        adapt.force_keyframe.store(true, Ordering::Relaxed);
757                        stats.set_screen_codec(video_codec_to_ctl(got).label());
758                    }
759                    None => {
760                        rebuild_retry.failed(Instant::now());
761                        tracing::warn!(track = "screen", codec = ?want_codec, "call: mid-share encoder rebuild found no usable encoder; keeping current and retrying");
762                    }
763                }
764            }
765
766            encode_and_publish_adaptive(
767                frame,
768                &mut encoder,
769                &adapt,
770                &mut applied_bps,
771                &mut fps_gate,
772                enc_dims,
773                |wire| vid_tx.try_send(wire).is_ok(),
774                &mut frame_seq,
775                sender,
776                TrackKind::ScrVideo,
777                &stats,
778                &mut fail_streak,
779            );
780        }
781
782        // LOW simulcast layer: when a constrained / small-tile receiver demands it,
783        // downscale this same capture frame and encode + forward it on the screen_l0
784        // track. Inert while `low_active` is `None` (dark until SIMULCAST_ACTUATE),
785        // so the source path above is byte-identical to a no-simulcast build.
786        match *low_active.borrow() {
787            Some(low_bps) => {
788                low_adapt.target_bps.store(low_bps, Ordering::Relaxed);
789                // A low-track PLI (new/recovering low subscriber) → force a low IDR.
790                if low_keyframe.swap(false, Ordering::Relaxed) {
791                    low_adapt.force_keyframe.store(true, Ordering::Relaxed);
792                }
793                // What this cap buys, and the geometry it settles on. The dwell is
794                // created on activation seeded at the first rung (so the encoder
795                // opens there at once) and thereafter debounces every change.
796                let want_rung =
797                    hound_media::low_rung_for_cap(hound_media::VideoSource::Screen, low_bps);
798                let fit = |r| low_screen_dims(neg_dims, r);
799                let dwell = low_geometry.get_or_insert_with(|| {
800                    hound_media::GeometryDwell::with_committed(
801                        (neg_dims.width(), neg_dims.height()),
802                        {
803                            let d = fit(want_rung);
804                            (d.width(), d.height())
805                        },
806                    )
807                });
808                let (settled_w, settled_h) = dwell.observe(
809                    // PIXELS, not height: a share's aspect is whatever the user
810                    // picked, so the rung's price is the thing to hold — see
811                    // `policy::low_screen_dims`.
812                    Some(hound_media::RungTarget::Pixels(want_rung.pixels())),
813                    frame.timestamp_us.micros(),
814                    hound_media::GEOMETRY_REBUILD_DWELL_US,
815                );
816                let low_dims = Dimensions::even_clamped(settled_w, settled_h);
817                // The rung the settled geometry belongs to, read off the dwell's own
818                // committed TARGET: while a change is still dwelling this is the rung
819                // the encoder is BUILT at, so its fps gate stays coherent with the
820                // encoder's rate control. It cannot be inferred from the geometry —
821                // a share of a window smaller than both rungs' budgets fits to its
822                // own geometry on either of them.
823                let settled_rung = if dwell.committed_target()
824                    == Some(hound_media::RungTarget::Pixels(want_rung.pixels()))
825                {
826                    want_rung
827                } else {
828                    low_rung.map(|(_, r)| r).unwrap_or(want_rung)
829                };
830                low_adapt
831                    .target_fps
832                    .store(settled_rung.fps, Ordering::Relaxed);
833
834                // (Re)build the low encoder on a codec switch OR a settled
835                // (geometry, RUNG) change — `camera_tx`'s key verbatim. Any of the
836                // three forces a fresh IDR: no decoder can follow a codec or a
837                // resolution change mid-GOP.
838                //
839                // The rung is half the key, not just the geometry, because a rung
840                // change with UNCHANGED geometry is still a change no live encoder
841                // can absorb. A share of a window already smaller than both rungs'
842                // pixel budgets fits to its own geometry on either of them
843                // (`fit_encode_area` never upscales), and the two then differ by
844                // 15 ↔ 30 fps and 200 kbps ↔ 1.1 Mbps: `VideoEncode` has no
845                // frame-rate setter at all (fps is construction-only, and on the
846                // software encoders it fixes the intra period with it), and
847                // `set_bitrate` is a documented no-op on the software paths. Keying
848                // on geometry alone would have left an encoder built at 30 fps /
849                // 1.1 Mbps rate control being fed 15 fps against a latched-but-
850                // unapplied 200 kbps target — a large, sustained overshoot of a link
851                // that had just said it was short. `GeometryDwell` debounces the
852                // rung on the same dwell timestamps, so a cap wobbling across the
853                // boundary still actuates nothing.
854                let low_want = codec_from_rank(adapt.target_codec.load(Ordering::Relaxed));
855                let built = low_rung;
856                if low_encoder.is_none()
857                    || low_last_target != Some(low_want)
858                    || built != Some((low_dims, settled_rung))
859                {
860                    low_last_target = Some(low_want);
861                    low_encoder = open_encoder_laddered(
862                        low_want,
863                        low_dims,
864                        Fps::new(settled_rung.fps),
865                        Bitrate::from_bps(low_bps.min(u32::MAX as u64) as u32),
866                        SCREEN_KEYFRAME_CADENCE,
867                    );
868                    if let Some(enc) = low_encoder.as_ref() {
869                        use hound_engine::backend::VideoEncode;
870                        // What the ladder actually built, which the `_l0` decl must
871                        // name — the roster gate rejects a decl for a codec the
872                        // receivers can't decode, and a rejected decl means no
873                        // `TrackInfo` and so no subscribers at all.
874                        low_codec.store(codec_rank(enc.codec()), Ordering::Relaxed);
875                        if built.is_some_and(|k| k != (low_dims, settled_rung)) {
876                            tracing::info!(
877                                track = "screen_l0",
878                                cap_bps = low_bps,
879                                height = settled_rung.height,
880                                fps = settled_rung.fps,
881                                dims = %low_dims,
882                                "simulcast[send]: low rung re-priced — rebuilding the low encoder"
883                            );
884                        }
885                        low_rung = Some((low_dims, settled_rung));
886                        low_adapt.force_keyframe.store(true, Ordering::Relaxed);
887                        // Same re-baseline the camera low path does on a rebuild:
888                        // `low_applied_bps` holds a CODEC-SCALED target belonging to
889                        // the encoder that just went away, and the new codec's factor
890                        // differs — leaving it latched can make the first
891                        // `set_bitrate` a no-op. The fps gate is re-armed so the
892                        // forced IDR is not swallowed by the old cadence, which
893                        // matters more here than on the camera: the screen rungs move
894                        // 15 ↔ 30 fps.
895                        low_applied_bps = 0;
896                        low_fps_gate = hound_media::FrameRateGate::new();
897                    }
898                }
899                if let Some(low_enc) = low_encoder.as_mut()
900                    && let Some(low_frame) = low_rung_frame(frame, low_enc, low_dims)
901                {
902                    encode_and_publish_adaptive(
903                        &low_frame,
904                        low_enc,
905                        &low_adapt,
906                        &mut low_applied_bps,
907                        &mut low_fps_gate,
908                        low_dims,
909                        |wire| low_vid_tx.try_send(wire).is_ok(),
910                        &mut low_frame_seq,
911                        sender,
912                        TrackKind::ScrVideoLow,
913                        &stats,
914                        &mut low_fail_streak,
915                    );
916                }
917            }
918            // Demand gone — release the low encoder (resumes instantly on re-demand).
919            // The dwell goes with it: the next activation re-seeds at whatever rung
920            // the fresh cap buys rather than inheriting a stale committed geometry.
921            None => {
922                low_encoder = None;
923                low_geometry = None;
924                low_rung = None;
925                // Nothing will consume an owed low IDR while the layer is dark, and a
926                // latched flag would keep the idle loop re-emitting forever.
927                low_keyframe.store(false, Ordering::Relaxed);
928            }
929        }
930    }
931    // RTC-debug: the screen pipeline is no longer publishing (clears the section).
932    stats.set_screen_disabled();
933    // Drop the `!Send` video handle on this owning thread (the `!Send` contract).
934    drop(screen);
935
936    // Report the classified exit to the supervisor — UNLESS this was a deliberate stop
937    // (stop/leave/reconnect already torn down by the supervisor, which owns that
938    // transition; firing would be a redundant no-op the command handler de-dups anyway).
939    // `last_frame` is the incarnation's whole delivery history: it is set by the FIRST
940    // real captured frame and never cleared (the static re-emit path mutates it in
941    // place), so `is_some()` is exactly "this capture ever produced a frame".
942    let delivered_any = last_frame.is_some();
943    if !stop.load(Ordering::Relaxed) {
944        notify(exit, delivered_any);
945    }
946    tracing::info!(
947        ?exit,
948        delivered_any,
949        stop = stop.load(Ordering::Relaxed),
950        "call: screen capture loop exiting"
951    );
952}

954/// The async screen VIDEO publish task: acquire the `screen` track, gate on a live
955/// subscriber, then forward each encoded AU while running the 1 s adaptation tick
956/// (screen ladder). Mirrors the camera publish task. Reads `screen_quality` each tick
957/// and reconfigures its ladder live (`set_screen_quality`) when the sharer changes it —
958/// no restart.
959#[allow(clippy::too_many_arguments)]
960fn spawn_screen_video_publish(
961    relay: Arc<MoqRelaySession>,
962    path: String,
963    mut enc_rx: tokio::sync::mpsc::Receiver<WireFrame>,
964    adapt: Arc<AdaptShared>,
965    neg_dims: Dimensions,
966    rtt_conn: iroh::endpoint::Connection,
967    screen_quality: SharedScreenQuality,
968    stats: Arc<CallStats>,
969    codec: VideoCodec,
970) -> JoinHandle<()> {
971    call_runtime().spawn(async move {
972        let mut publisher = match relay.publish_track(&path, SCREEN_TRACK, VIDEO_PRIORITY) {
973            Ok(p) => p,
974            Err(e) => {
975                tracing::error!(%e, "screen: failed to publish video track");
976                return;
977            }
978        };
979        // RTC-debug: the negotiated screen track codec, set once at publish.
980        stats.set_screen_codec(video_codec_to_ctl(codec).label());
981
982        // Sender-side adaptation core (sans-IO) seeded at the screen launch bitrate,
983        // using the sharer's CURRENT quality preference; the before-tick hook applies
984        // a mid-share change (the `call_set_screen_quality` command) so the ladder is
985        // live-reconfigured without a restart. The preference lives on
986        // `ConnectionParams`, so this re-seeds from it across reconnects too. The
987        // ladder is fitted to the NEGOTIATED capture geometry: its rung table is
988        // built from nominal 16:9 heights, and the encoder never upscales past the
989        // source, so a small or non-16:9 share would otherwise be handed (and spend)
990        // the bits of a full 1080p frame it will never encode.
991        // Not independent of the camera: `drive_video_publish` registers this
992        // publisher with the connection-level `SendCoordinator`, so while both
993        // publish the two ladders divide one budget rather than each claiming the
994        // whole shared uplink.
995        let source_dims = (neg_dims.width(), neg_dims.height());
996        let mut applied_quality = **screen_quality.load();
997        let mut adaptation =
998            VideoAdaptation::screen_with(SCREEN_BITRATE.bps() as u64, applied_quality, source_dims);
999        // The encoder-rebuild geometry dwell, seeded at the negotiated capture
1000        // geometry — exactly what the capture thread's encoder is already built at
1001        // before any tick runs, so a healthy share at the top rung never rebuilds.
1002        let mut geometry = hound_media::GeometryDwell::new(source_dims);
1003        drive_video_publish(
1004            &mut publisher,
1005            &mut enc_rx,
1006            &adapt,
1007            &mut adaptation,
1008            &mut geometry,
1009            &rtt_conn,
1010            &stats,
1011            true,
1012            "screen",
1013            |adaptation| {
1014                let want = **screen_quality.load();
1015                if want != applied_quality {
1016                    adaptation.set_screen_quality(want);
1017                    applied_quality = want;
1018                    tracing::info!(?want, "call: screen quality changed — ladder reconfigured");
1019                }
1020            },
1021        )
1022        .await;
1023    })
