16/// The call's roster as the UI renders it: self id, peers, and the tracks (with
17/// kind/source/mute). Built from the control plane's [`SessionView`] reducer — the
18/// same source of truth the media plane subscribes off of. Empty before `Welcome`.
19#[derive(Clone, Debug, Default, PartialEq, Eq)]
20pub struct CallView {
21    /// This client's own peer id, once the server assigns it (`Welcome`).
22    pub self_id: Option<u32>,
23    /// Everyone in the call, including self (`PeerId.0`).
24    pub peers: Vec<u32>,
25    /// Every known track (local + remote).
26    pub tracks: Vec<CallTrack>,
27    /// peer_id → user_id, for the peers whose identity we know. Purely additive:
28    /// the UI joins these to the display roster (keyed by `user_id`) for the real
29    /// name/avatar; a peer with no entry falls back to "Peer N". Built from the
30    /// control plane's `peers` joined with the realtime-derived [`PeerIdentities`].
31    pub participants: Vec<CallParticipant>,
32}
33
34/// A control-plane peer with a known identity. The UI joins `user_id` against the
35/// display roster (the same one text channels use) for the name + avatar. Carries
36/// no name/avatar itself — identity lives in the roster, not the call plane.
37#[derive(Clone, Debug, PartialEq, Eq)]
38pub struct CallParticipant {
39    pub peer_id: u32,
40    pub user_id: Uuid,
41}

43/// Shared `peer_id (u32) → user_id (Uuid)` map. Seeded at join from the REST
44/// `VoiceJoin.peers` (+ self), then kept current by the realtime voice events
45/// (`VoiceJoined`/`VoiceState`/`VoiceLeft`). An `ArcSwap` so the call's reader task
46/// reads a cheap snapshot while the realtime reducer task swaps in updates — no lock
47/// across the media plane. Cleared on call join/leave so identities never leak.
48pub type PeerIdentities = Arc<ArcSwap<HashMap<u32, Uuid>>>;
49
50/// The sharer's live screen-quality preference (motion↔legibility + a resolution/fps
51/// cap), shared between [`CallSession::set_screen_quality`] (the writer) and the
52/// running screen adaptation tick (the reader, which `set_screen_quality`s its ladder
53/// when it changes). An `ArcSwap` so the command writes lock-free while the tick reads
54/// a cheap snapshot. Lives on [`ConnectionParams`] so it survives a reconnect (the
55/// republished share re-seeds its ladder from it).
56pub(super) type SharedScreenQuality = Arc<ArcSwap<ScreenQuality>>;

58/// Join `peers` (the control-plane roster) against the `peer_id → user_id` map into
59/// a deterministic (sorted by peer_id) list of [`CallParticipant`]. Peers with no
60/// known identity are simply omitted (the UI falls back to "Peer N"). Shared by the
61/// call reader (control-plane changes) and the realtime reducer (identity changes).
62pub fn build_participants(peers: &[u32], map: &HashMap<u32, Uuid>) -> Vec<CallParticipant> {
63    let mut out: Vec<CallParticipant> = peers
64        .iter()
65        .filter_map(|&peer_id| {
66            map.get(&peer_id)
67                .map(|&user_id| CallParticipant { peer_id, user_id })
68        })
69        .collect();
70    out.sort_by_key(|p| p.peer_id);
71    out
72}

74/// One published track, carried with the control-plane enums (not flattened to
75/// strings) so the binding compares them type-driven — `t.kind == MediaKind::Video`
76/// — with exhaustiveness on both sides. The enums are `Copy + Eq` (serde snake_case).
77#[derive(Clone, Debug, PartialEq, Eq)]
78pub struct CallTrack {
79    pub id: u32,
80    pub peer: u32,
81    pub kind: MediaKind,
82    pub source: Source,
83    pub codec: CtlCodec,
84    pub muted: bool,
85    /// Video only: this receiver's own allocator currently gives this track NO
86    /// subscription — the decode budget cut it, or the bandwidth budget's deficit
87    /// phase demoted it. Nothing is arriving and nothing will until the policy
88    /// changes, which the UI has to know: a tile that renders a texture nobody is
89    /// feeding is a black rectangle with a name on it, and on a 32-tile macOS
90    /// gallery that is sixteen black rectangles each holding an idle IOSurface.
91    /// The UI treats it exactly as it treats a camera that is off — the avatar
92    /// fallback, no texture bound — which is also what releases the surface.
93    ///
94    /// Never true for audio: the allocator budgets video, and a hidden tile's owner
95    /// is still heard. Distinct from `muted`, which is what the SENDER is doing.
96    pub hidden: bool,
97}

99/// The `(peer, source)` video tiles this receiver's allocator currently has hidden,
100/// shared between the two tasks that need it. The subscribe loop's reconcile pass is
101/// the sole WRITER (it is the only thing that knows the allocation); the control
102/// reader is a reader, because it rebuilds the whole [`CallView`] from the
103/// [`SessionView`] on every roster frame and would otherwise drop the flags on the
104/// floor. Same shape and the same reason as [`PeerIdentities`], which the reader
105/// joins in at the same point.
106///
107/// The writer also republishes on change — a decode cut is accompanied by no control
108/// frame at all, so nothing else would wake the UI — and re-patches the published
109/// view every pass, which repairs the (harmless, sub-tick) race where the reader
110/// rebuilt from a snapshot taken just before a store.
111#[derive(Clone)]
112pub(super) struct HiddenTiles {
113    set: Arc<ArcSwap<HashSet<(u32, Source)>>>,
114    /// The published `CallView` this set is a field of.
115    view: watch::Sender<CallView>,
116}
117
118impl HiddenTiles {
119    pub(super) fn new(view: watch::Sender<CallView>) -> Self {
120        Self {
121            set: Arc::new(ArcSwap::from_pointee(HashSet::new())),
122            view,
123        }
124    }
125
126    /// The current set, for the control reader's `CallView` rebuild.
127    pub(super) fn load(&self) -> Arc<HashSet<(u32, Source)>> {
128        self.set.load_full()
129    }
130
131    /// Publish this pass's hidden set: store it for the reader, then patch the live
132    /// [`CallView`]. Called every reconcile pass — `send_if_modified` notifies only
133    /// when a flag actually moved, so a steady call costs one comparison per pass and
134    /// wakes nothing.
135    pub(super) fn publish(&self, next: &HashSet<(u32, Source)>) {
136        if **self.set.load() != *next {
137            self.set.store(Arc::new(next.clone()));
138        }
139        self.view
140            .send_if_modified(|cv| mark_hidden(&mut cv.tracks, next));
141    }
142}

144/// Stamp `hidden` onto each video track, returning whether anything changed.
145/// The single definition of the flag, shared by the control reader's rebuild and
146/// the subscribe loop's patch so the two can never disagree.
147fn mark_hidden(tracks: &mut [CallTrack], hidden: &HashSet<(u32, Source)>) -> bool {
148    let mut changed = false;
149    for t in tracks {
150        let want = t.kind == MediaKind::Video && hidden.contains(&(t.peer, t.source));
151        changed |= t.hidden != want;
152        t.hidden = want;
153    }
154    changed
155}

157impl CallView {
158    /// Build from the control-plane view alone — participants empty (identities
159    /// unknown), nothing hidden. Test-only; the live reader always uses
160    /// [`from_view_with`].
161    #[cfg(test)]
162    fn from_view(view: &SessionView) -> Self {
163        Self::from_view_with(view, &HashMap::new(), &HashSet::new())
164    }
165
166    /// Build from the control-plane view, joining each peer against the
167    /// `peer_id → user_id` map to populate [`CallView::participants`] and against the
168    /// receiver's own [`HiddenTiles`] set to populate [`CallTrack::hidden`].
169    pub(super) fn from_view_with(
170        view: &SessionView,
171        identities: &HashMap<u32, Uuid>,
172        hidden: &HashSet<(u32, Source)>,
173    ) -> Self {
174        let peers: Vec<u32> = view.peers().map(|p| p.id.0).collect();
175        let participants = build_participants(&peers, identities);
176        let mut tracks: Vec<CallTrack> = view
177            .tracks()
178            .map(|t| CallTrack {
179                id: t.id.0,
180                peer: t.peer.0,
181                kind: t.kind,
182                source: t.source,
183                codec: t.codec,
184                muted: t.muted,
185                hidden: false,
186            })
187            .collect();
188        mark_hidden(&mut tracks, hidden);
189        Self {
190            self_id: view.self_id().map(|p| p.0),
191            peers,
192            tracks,
193            participants,
194        }
195    }
196}

198/// The connection state of the call's control plane (and, by extension, whether
199/// media can flow). Drives a UI banner. `attempt` is the reconnect attempt count.
200#[derive(Clone, Debug, Default, PartialEq, Eq)]
201pub enum ConnectionState {
202    /// Not in a call (initial / after `leave`).
203    #[default]
204    Idle,
205    /// Dialing the control WS / relay for the first time.
206    Connecting,
207    /// Control WS up and a `Welcome` received — media flowing.
208    Connected,
209    /// Lost the connection; a redial is in progress (`attempt` = which try).
210    Reconnecting { attempt: u32 },
211    /// Gave up (fatal, or reconnect not implemented yet — see module docs).
212    Failed { reason: String },
213}

322    /// THE budget-hidden-tile seam. A tile the allocator leaves without a
323    /// subscription must reach the UI as `hidden` so it renders its owner's avatar
324    /// rather than a black texture — and it must survive the control reader, which
325    /// rebuilds the whole `CallView` from the `SessionView` on every roster frame and
326    /// would otherwise drop the flag on the floor. Both writers go through
327    /// `mark_hidden`, and this pins that they agree.
328    #[test]
329    fn a_budget_hidden_tile_reaches_the_ui_and_survives_a_roster_rebuild() {
330        let mut view = SessionView::new();
331        view.apply(&ServerMessage::Welcome {
332            protocol_version: PROTOCOL_VERSION,
333            self_id: PeerId(1),
334            peers: vec![PeerInfo {
335                id: PeerId(2),
336                decode_caps: DecodeCaps::ALL,
337                user_id: Uuid::nil(),
338            }],
339            tracks: vec![
340                track(10, 2, MediaKind::Video, Source::Camera, Codec::H264),
341                track(11, 2, MediaKind::Audio, Source::Microphone, Codec::Opus),
342                track(12, 2, MediaKind::Video, Source::Screen, Codec::H264),
343            ],
344        });
345
346        let (tx, rx) = watch::channel(CallView::default());
347        let hidden = HiddenTiles::new(tx);
348        assert!(
349            rx.borrow().tracks.iter().all(|t| !t.hidden),
350            "nothing is hidden before the allocator says so"
351        );
352
353        // The subscribe loop's side: publish, and the live view is patched even
354        // though no control frame arrived (a decode cut is accompanied by none).
355        let cut = HashSet::from([(2u32, Source::Camera)]);
356        tx_view(&hidden, &view);
357        hidden.publish(&cut);
358        let cv = rx.borrow().clone();
359        assert!(cv.tracks.iter().find(|t| t.id == 10).unwrap().hidden);
360        assert!(
361            !cv.tracks.iter().find(|t| t.id == 11).unwrap().hidden,
362            "audio is never hidden by the video allocator — the peer is still heard"
363        );
364        assert!(
365            !cv.tracks.iter().find(|t| t.id == 12).unwrap().hidden,
366            "the same peer's share is a separate tile with its own verdict"
367        );
368
369        // The control reader's side: a fresh rebuild re-joins the same set.
370        let rebuilt = CallView::from_view_with(&view, &HashMap::new(), &hidden.load());
371        assert_eq!(rebuilt.tracks, cv.tracks, "the two writers must agree");
372
373        // …and un-hiding clears it, so the tile binds a texture again.
374        hidden.publish(&HashSet::new());
375        assert!(rx.borrow().tracks.iter().all(|t| !t.hidden));
376    }

378    /// Seed the published view the way the control reader does, so the patch above
379    /// has tracks to patch.
380    fn tx_view(hidden: &HiddenTiles, view: &SessionView) {
381        hidden.view.send_replace(CallView::from_view_with(
382            view,
383            &HashMap::new(),
384            &hidden.load(),
385        ));
386    }
