1//! The speaker check: a short sine played through a [`PlayoutBus`] — the same
2//! mixed-playout path (device open, hot-swap fallback, output cushion) a call's
3//! remote audio takes — so "I heard it" vouches for the output a call will use.
4//!
5//! Out of call only: the bus built here has no AEC render reference, so a tone
6//! during a call would reach the far end through the microphone. The refusal
7//! lives at the command seam ([`crate::HoundClient::play_test_tone`]).
8
9use std::time::Duration;
10
11use tokio::sync::watch;
12
13use hound_engine::api::Engine;
14use hound_engine::types::{DeviceChoice, SAMPLE_RATE, SpeakerId};
15use hound_media::CaptureClock;
16
17/// Tone pitch: concert A.
18const FREQ_HZ: f64 = 440.0;
19/// Tone length.
20const DURATION_MS: u32 = 300;
21/// Linear fade at each end; without it the step onset/offset clicks.
22const FADE_MS: u32 = 10;
23/// Peak level. Loud enough to hear over a fan, quiet enough not to startle on
24/// headphones at a high OS volume.
25const LEVEL_DBFS: f64 = -12.0;
26/// Samples per pushed frame: 20 ms, the call's audio frame size.
27const FRAME_SAMPLES: usize = SAMPLE_RATE.hz() as usize / 50;
28/// [`FRAME_SAMPLES`] in µs on the bus clock.
29const FRAME_US: i64 = 20_000;
30/// Jitter-buffer floor for the tone's slot. The whole tone is pushed at once, so
31/// its newest frame lands `floor + 280 ms` ahead of now — under the playout
32/// buffer's 400 ms catch-up ceiling, past which it re-anchors and skips the head.
33const TONE_FLOOR_US: i64 = 40_000;
34/// Head start the bus gets to open the device and prefill before the first frame
35/// is pushed. Frames that come due while the device is still opening would be
36/// released in one burst into the slot's ~100 ms carry and lose the tone's head.
37const DEVICE_OPEN_LEAD: Duration = Duration::from_millis(250);
38/// Held after the last frame is due so the device ring drains before the bus
39/// (and with it the output stream) is dropped.
40const DRAIN_TAIL: Duration = Duration::from_millis(200);
41
42/// The tone as 48 kHz mono PCM: [`FREQ_HZ`] at [`LEVEL_DBFS`] peak for
43/// [`DURATION_MS`], ramped in and out over [`FADE_MS`].
44pub(crate) fn tone_samples() -> Vec<i16> {
45    let rate = SAMPLE_RATE.hz();
46    let n = (rate * DURATION_MS / 1000) as usize;
47    let fade = (rate * FADE_MS / 1000) as usize;
48    let amp = 10f64.powf(LEVEL_DBFS / 20.0) * f64::from(i16::MAX);
49    (0..n)
50        .map(|i| {
51            // Distance to the nearer end, in fade lengths, capped at full gain.
52            let gain = (i.min(n - 1 - i) as f64 / fade as f64).min(1.0);
53            let phase = std::f64::consts::TAU * FREQ_HZ * i as f64 / f64::from(rate);
54            (amp * gain * phase.sin()).round() as i16
55        })
56        .collect()
57}
58
59/// Play the tone on `speaker_id` (`None` = host default) and return once it has
60/// drained. A device that fails to open is the bus's own reopen ladder's
61/// business — the bus reports nothing back, so neither can this.
62pub(crate) async fn play(speaker_id: Option<String>) {
63    let choice = match speaker_id {
64        Some(id) => DeviceChoice::Named(SpeakerId::from(id)),
65        None => DeviceChoice::Default,
66    };
67    // The sender stays alive for the tone's length so the bus never sees the
68    // choice channel close mid-play.
69    let (_choice_tx, choice_rx) = watch::channel(choice);
70    let clock = CaptureClock::start();
71    let bus = Engine::new().start_playout_bus(choice_rx, None, clock, TONE_FLOOR_US);
72    let producer = bus.register("test_tone");
73
74    tokio::time::sleep(DEVICE_OPEN_LEAD).await;
75    let now = clock.now();
76    let mut ts = now;
77    for frame in tone_samples().chunks(FRAME_SAMPLES) {
78        // Concealed = never folded into the jitter estimator: the frames share one
79        // arrival instant, which would otherwise read as a burst of link jitter.
80        producer.push_concealed(now, ts, frame);
81        ts += FRAME_US;
82    }
83    let playing =
84        Duration::from_millis(u64::from(DURATION_MS)) + Duration::from_micros(TONE_FLOOR_US as u64);
85    tokio::time::sleep(playing + DRAIN_TAIL).await;
86    drop(producer);
87    bus.stop();
88}

90#[cfg(test)]
91mod tests {
92    use super::{FADE_MS, FRAME_SAMPLES, tone_samples};
93
94    const RATE: usize = 48_000;
95
96    #[test]
97    fn the_tone_is_300_ms_of_whole_frames() {
98        let tone = tone_samples();
99        assert_eq!(tone.len(), RATE * 300 / 1000);
100        assert_eq!(tone.len() % FRAME_SAMPLES, 0, "no ragged last frame");
101    }
102
103    /// -12 dBFS of i16 full scale is 8231; the sine's sample grid lands within a
104    /// count of the true crest.
105    #[test]
106    fn the_peak_sits_at_minus_12_dbfs() {
107        let peak = tone_samples()
108            .iter()
109            .map(|s| i32::from(*s).abs())
110            .max()
111            .unwrap();
112        assert!((8200..=8231).contains(&peak), "peak {peak}");
113    }
114
115    /// Both ends start from silence and ramp linearly: nothing in the first
116    /// fade can exceed its envelope, and the envelope is full by its end.
117    #[test]
118    fn both_ends_fade_from_silence() {
119        let tone = tone_samples();
120        let fade = RATE * FADE_MS as usize / 1000;
121        let full = 8231.0;
122        assert_eq!(tone[0], 0);
123        assert_eq!(*tone.last().unwrap(), 0);
124        for (i, s) in tone.iter().take(fade).enumerate() {
125            let envelope = full * i as f64 / fade as f64;
126            assert!(f64::from(s.unsigned_abs()) <= envelope + 1.0, "head {i}");
127        }
128        for (i, s) in tone.iter().rev().take(fade).enumerate() {
129            let envelope = full * i as f64 / fade as f64;
130            assert!(f64::from(s.unsigned_abs()) <= envelope + 1.0, "tail {i}");
131        }
132        // Past the fade the full level is reached within one 440 Hz cycle.
133        let cycle = RATE / 440 + 1;
134        let body_peak = tone[fade..fade + cycle]
135            .iter()
136            .map(|s| s.unsigned_abs())
137            .max()
138            .unwrap();
139        assert!(body_peak >= 8200, "body peak {body_peak}");
140    }
