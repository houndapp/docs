1//! Demand-driven simulcast, producer side: [`LayerControl`] folds every
2//! receiver's standing quality demand ([`QualityReport`] via
3//! [`spawn_quality_listener`]) into a [`hound_media::SimulcastPlanner`], driving
4//! the shared LOW-layer supervisor; [`spawn_low_video_publish`] drains the
5//! capture thread's low encode onto its own moq track, announcing it on the
6//! FIRST frame only (dark until actually producing). Invariant: the source
7//! encode is never dropped — the low layer is purely additive, so any low-layer
8//! fault costs a thumbnail, never the full-quality source. That additive-only
9//! design is why the low layer runs unconditionally: a receiver only switches to
10//! the low track for a small/hidden tile it explicitly requested.
11//!
12//! **The roster is truthful about the low layer** ([`LowDecl`]): the `_l0`
13//! `TrackDecl` exists on the control plane exactly while the low encoder is
14//! producing. A decl left behind after a teardown is worse than no decl at all —
15//! the receiver's `pick` chooses it, the subscription yields nothing, and that
16//! tile eats the stall → first-keyframe → restart ladder (minutes of black).
17

20/// The per-source producer seam between the quality listener (records each receiver's
21/// demand into a [`SimulcastPlanner`]) and the low-track publish supervisor (spawns /
22/// tears the low encoder). Mirrors [`AdaptShared`] as a shared producer control block;
23/// the planner is touched ~1 Hz (per report / roster change), never per frame, so the
24/// `Mutex` is free.
25pub(super) struct LayerControl {
26    planner: std::sync::Mutex<hound_media::SimulcastPlanner>,
27    /// Drives the low-publish supervisor: `Some(bps)` = run a shared low layer at this
28    /// bitrate; `None` = tear it down (no receiver wants a below-source layer).
29    pub(super) low_active: watch::Sender<Option<u64>>,
30    /// Lock-free snapshot of the planner's capped-peer set, refreshed on every demand
31    /// change (`set_request`/`retain_peers`) — read on the feedback listener's hot
32    /// path so a per-report fold never contends the `Mutex` the ~1Hz demand listener
33    /// holds. See `Self::capped_peers`.
34    capped: ArcSwap<std::collections::BTreeSet<u32>>,
35}
36
37impl LayerControl {
38    pub(super) fn new(source_bps: u64) -> Self {
39        let planner = hound_media::SimulcastPlanner::new(source_bps);
40        Self {
41            planner: std::sync::Mutex::new(planner),
42            low_active: watch::channel(None).0,
43            capped: ArcSwap::from_pointee(std::collections::BTreeSet::new()),
44        }
45    }
46
47    fn lock(&self) -> std::sync::MutexGuard<'_, hound_media::SimulcastPlanner> {
48        self.planner.lock().unwrap_or_else(|e| e.into_inner())
49    }
50
51    /// Push the planner's effective low-layer target to the supervisor (no-op if same).
52    fn sync_low(&self, plan: hound_media::LayerPlan) {
53        self.low_active.send_if_modified(|cur| {
54            if *cur == plan.low_layer_bps {
55                false
56            } else {
57                tracing::info!(
58                    low_bps = ?plan.low_layer_bps,
59                    send_source = plan.send_source,
60                    "simulcast[send]: low layer active-state changed"
61                );
62                *cur = plan.low_layer_bps;
63                true
64            }
65        });
66    }
67
68    /// Record a receiver's standing request and re-derive the low-layer target.
69    pub(super) fn set_request(&self, peer: u32, req: hound_media::QualityRequest) {
70        let (changed, plan, capped) = {
71            let mut p = self.lock();
72            let changed = p.set_request(peer, req);
73            (changed, p.plan(), p.capped_peers())
74        };
75        self.capped.store(std::sync::Arc::new(capped));
76        if changed {
77            tracing::info!(?plan, "simulcast: receiver demand changed");
78        }
79        self.sync_low(plan);
80    }
81
82    /// Drop departed receivers from the cap map (roster shrank).
83    fn retain_peers(&self, keep: impl Fn(u32) -> bool) {
84        let (plan, capped) = {
85            let mut p = self.lock();
86            p.retain_peers(keep);
87            (p.plan(), p.capped_peers())
88        };
89        self.capped.store(std::sync::Arc::new(capped));
90        self.sync_low(plan);
91    }
92
93    /// Peers currently requesting a capped (below-source) layer — a lock-free snapshot
94    /// read (no `Mutex` contention). Excluded from the SOURCE track's BWE feedback fold
95    /// (see `spawn_feedback_listener` in adapt.rs) so a low-layer receiver — including one
96    /// still transiently subscribed to the source track before its low track's first
97    /// frame — can't drag the source adaptation down.
98    pub(super) fn capped_peers(&self) -> std::sync::Arc<std::collections::BTreeSet<u32>> {
99        self.capped.load_full()
100    }
101}

103/// Spawn the producer side of demand-driven simulcast: watch `quality` and, when a
104/// request names OUR video track of `source`, record it into `layers` (driving the
105/// low-layer supervisor). A slow tick drops requesters who left the call. "Our track"
106/// is resolved exactly like the feedback/keyframe listeners.
107pub(super) fn spawn_quality_listener(
108    mut quality: broadcast::Receiver<QualityReport>,
109    view_rx: watch::Receiver<SessionView>,
110    self_id: PeerId,
111    source: Source,
112    layers: Arc<LayerControl>,
113) -> JoinHandle<()> {
114    call_runtime().spawn(async move {
115        let mut prune = tokio::time::interval(Duration::from_secs(2));
116        prune.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
117        loop {
118            tokio::select! {
119                r = quality.recv() => match r {
120                    Ok(report) => {
121                        let is_ours = {
122                            let view = view_rx.borrow();
123                            view.track(report.track).is_some_and(|t| {
124                                t.peer == self_id
125                                    && t.kind == MediaKind::Video
126                                    && t.source == source
127                            })
128                        };
129                        if is_ours {
130                            let req = match (report.cap_bps, report.hidden) {
131                                (Some(bps), true) => hound_media::QualityRequest::Hidden(bps),
132                                (Some(bps), false) => hound_media::QualityRequest::Capped(bps),
133                                (None, _) => hound_media::QualityRequest::Source,
134                            };
135                            layers.set_request(report.from.0, req);
136                        }
137                    }
138                    Err(broadcast::error::RecvError::Lagged(_)) => {}
139                    Err(broadcast::error::RecvError::Closed) => break,
140                },
141                _ = prune.tick() => {
142                    let present: std::collections::HashSet<u32> =
143                        view_rx.borrow().peers().map(|p| p.id.0).collect();
144                    layers.retain_peers(|peer| present.contains(&peer));
145                }
146            }
147        }
148    })
149}

151/// Whether the `_l0` [`TrackDecl`] is currently in the control-plane roster, and the
152/// one awkward in-between: the layer was torn down before the server echoed the
153/// decl's [`TrackId`] back, so we know we owe an `Unpublish` but cannot address it
154/// yet. Pure (unit-tested); [`spawn_low_video_publish`] is the only caller and does
155/// the IO the returned booleans ask for.
156///
157/// The contract this pins: **decl present ⟺ low encoder producing**. A receiver's
158/// `pick` treats a present `_l0` decl as a live layer, so a stale one costs that tile
159/// the full stall → first-keyframe → restart ladder.
160#[derive(Debug, PartialEq, Eq)]
161enum LowDecl {
162    /// Nothing in the roster (never announced, or the withdrawal completed).
163    Absent,
164    /// A decl is in the roster and the low encoder is backing it.
165    Live,
166    /// The layer went away; the decl must leave the roster, but its server-assigned
167    /// id has not landed in our own roster view yet. Retried on each roster change.
168    Withdrawing,
169}
170
171impl LowDecl {
172    /// A low frame reached the publish task. `active` is the planner's current
173    /// verdict — a frame draining out of the channel *after* a teardown must not
174    /// resurrect the decl. Returns true when the caller must send `Publish`.
175    fn on_frame(&mut self, active: bool) -> bool {
176        if !active || *self != LowDecl::Absent {
177            return false;
178        }
179        *self = LowDecl::Live;
180        true
181    }
182
183    /// The planner's low-layer target moved. Deactivation owes an `Unpublish`;
184    /// re-activation while one is still owed simply CANCELS it (the decl never left
185    /// the roster and is backed again — republishing would leave two `_l0` decls for
186    /// the same source). Returns true when the caller must try the `Unpublish`.
187    fn on_active(&mut self, active: bool) -> bool {
188        match (active, &self) {
189            (false, LowDecl::Live) => {
190                *self = LowDecl::Withdrawing;
191                true
192            }
193            (true, LowDecl::Withdrawing) => {
194                *self = LowDecl::Live;
195                false
196            }
197            _ => false,
198        }
199    }
200
201    /// Our SOURCE decl left the roster, so this layer has nothing to be additive to
202    /// (camera off). Owes the same withdrawal a deactivation does.
203    fn on_source_gone(&mut self) {
204        if *self == LowDecl::Live {
205            *self = LowDecl::Withdrawing;
206        }
207    }
208
209    /// Whether an `Unpublish` is still owed (its track id was not resolvable yet).
210    fn withdraw_pending(&self) -> bool {
211        *self == LowDecl::Withdrawing
212    }
213
214    /// The `Unpublish` went out.
215    fn withdrawn(&mut self) {
216        *self = LowDecl::Absent;
217    }
218}

220/// OUR own video track id for `source` at the given layer, as the server assigned it
221/// (learned from the `TrackPublished` echo of our own `Publish`). `None` until that
222/// echo lands — or, for the source layer, while we are simply not publishing it
223/// (camera off).
224fn own_video_track(
225    view: &SessionView,
226    self_id: PeerId,
227    source: Source,
228    low: bool,
229) -> Option<TrackId> {
230    view.tracks()
231        .find(|t| {
232            t.peer == self_id && t.kind == MediaKind::Video && t.source == source && t.low == low
233        })
234        .map(|t| t.id)
235}
236
237/// Send the `Unpublish` a deactivated low layer owes, if our own decl's id has landed
238/// in the roster yet. A no-op while the `TrackPublished` echo is still in flight — the
239/// publish task retries on the next roster change.
240fn try_withdraw(
241    decl: &mut LowDecl,
242    view_rx: &watch::Receiver<SessionView>,
243    self_id: PeerId,
244    source: Source,
245    track_name: &'static str,
246    control_outbound: &tokio::sync::mpsc::UnboundedSender<ClientMessage>,
247) {
248    let Some(track) = own_video_track(&view_rx.borrow(), self_id, source, true) else {
249        return;
250    };
251    let _ = control_outbound.send(ClientMessage::Unpublish { track });
252    decl.withdrawn();
253    tracing::info!(
254        track = track_name,
255        id = track.0,
256        "simulcast[send]: low layer deactivated — withdrawing its decl from the roster"
257    );
258}

260/// Publish a LOW simulcast layer: drain the capture thread's downscaled encode and
261/// forward it on its own moq track. The control-plane `Publish` announce is sent on
262/// the FIRST frame only, so the low track stays invisible to receivers until it's
263/// actually producing (dark while no one demands it), and it is WITHDRAWN again the
264/// moment the planner deactivates the layer — see [`LowDecl`]. No own BWE tick — the
265/// low layer's bitrate is the planner's cap, applied by the capture thread's
266/// `low_adapt`.
267///
268/// Only the control-plane decl toggles: the moq publisher is opened once and kept, so
269/// nothing here churns the shared moq/QUIC session or touches this peer's other tracks.
270///
271/// NB keyframe-on-(re)subscribe for the low layer is routed by the keyframe listener
272/// (a PLI for the low track must arm the low encoder's IDR, not the source's) — wired
273/// when the layer is taken live; while dark this never runs.
274///
275/// `codec` is the low encoder's CURRENT codec ([`codec_rank`]), written by the capture
276/// thread on every low-encoder build — NOT the codec negotiated at share start. The
277/// announce happens on the first frame, arbitrarily long after that, and the server's
278/// publish gate rejects a decl naming a codec the roster cannot decode: announcing the
279/// start-of-share codec after a mid-share downgrade leaves the low layer with no
280/// `TrackInfo` at all, so no receiver can ever subscribe to it.
281#[allow(clippy::too_many_arguments)]
282pub(super) fn spawn_low_video_publish(
283    relay: Arc<MoqRelaySession>,
284    path: String,
285    track_name: &'static str,
286    source: Source,
287    codec: Arc<AtomicU8>,
288    mut low_rx: tokio::sync::mpsc::Receiver<WireFrame>,
289    control_outbound: tokio::sync::mpsc::UnboundedSender<ClientMessage>,
290    mut view_rx: watch::Receiver<SessionView>,
291    self_id: PeerId,
292    mut low_active: watch::Receiver<Option<u64>>,
293) -> JoinHandle<()> {
294    call_runtime().spawn(async move {
295        let mut publisher = match relay.publish_track(&path, track_name, VIDEO_PRIORITY) {
296            Ok(p) => p,
297            Err(e) => {
298                tracing::error!(%e, track = track_name, "low: failed to publish track");
299                return;
300            }
301        };
302        let mut decl = LowDecl::Absent;
303        loop {
304            tokio::select! {
305                frame = low_rx.recv() => {
306                    let Some(frame) = frame else { break };
307                    if decl.on_frame(low_active.borrow().is_some()) {
308                        let announced =
309                            video_codec_to_ctl(codec_from_rank(codec.load(Ordering::Relaxed)));
310                        let _ = control_outbound.send(ClientMessage::Publish {
311                            track: TrackDecl {
312                                kind: MediaKind::Video,
313                                source,
314                                codec: announced,
315                                low: true,
316                            },
317                        });
318                        tracing::info!(
319                            track = track_name,
320                            codec = ?announced,
321                            "simulcast[send]: low layer now publishing"
322                        );
323                    }
324                    if let Err(e) = publish_one(&mut publisher, &frame) {
325                        tracing::warn!(%e, track = track_name, "low: publish frame failed");
326                    }
327                }
328                r = low_active.changed() => {
329                    if r.is_err() { break; }
330                    let active = low_active.borrow().is_some();
331                    if decl.on_active(active) {
332                        try_withdraw(&mut decl, &view_rx, self_id, source, track_name, &control_outbound);
333                    }
334                }
335                r = view_rx.changed() => {
336                    if r.is_err() { break; }
337                    // Our low decl must never outlive the SOURCE decl it is a layer
338                    // of: an additive layer of a video nobody publishes is exactly the
339                    // stale decl this task exists to prevent (the camera-off path
340                    // `Unpublish`es the source and leaves the low encoder gated).
341                    if own_video_track(&view_rx.borrow(), self_id, source, false).is_none() {
342                        decl.on_source_gone();
343                    }
344                    // A roster change is also what makes an owed withdrawal
345                    // addressable (it may be our own `TrackPublished` echo landing).
346                    if decl.withdraw_pending() {
347                        try_withdraw(&mut decl, &view_rx, self_id, source, track_name, &control_outbound);
348                    }
349                }
350            }
351        }
352    })
353}

359    /// A roster carrying our own camera SOURCE decl (id 1) and, optionally, our low
360    /// decl (id 2) — plus a remote peer's camera, which must never be confused for ours.
361    fn roster(low: bool) -> SessionView {
362        fn cam(id: u32, peer: u32, low: bool) -> hound_control::TrackInfo {
363            hound_control::TrackInfo {
364                id: TrackId(id),
365                peer: PeerId(peer),
366                kind: MediaKind::Video,
367                source: Source::Camera,
368                codec: CtlCodec::H264,
369                muted: false,
370                low,
371            }
372        }
373        let mut tracks = vec![cam(1, 7, false), cam(3, 9, false), cam(4, 9, true)];
374        if low {
375            tracks.push(cam(2, 7, true));
376        }
377        let mut view = SessionView::new();
378        view.apply(&ServerMessage::Welcome {
379            protocol_version: PROTOCOL_VERSION,
380            self_id: PeerId(7),
381            peers: Vec::new(),
382            tracks,
383        });
384        view
385    }

387    #[test]
388    fn own_video_track_picks_our_own_decl_at_the_asked_for_layer() {
389        let view = roster(true);
390        assert_eq!(
391            own_video_track(&view, PeerId(7), Source::Camera, false),
392            Some(TrackId(1))
393        );
394        assert_eq!(
395            own_video_track(&view, PeerId(7), Source::Camera, true),
396            Some(TrackId(2)),
397            "the low layer must resolve to OUR `_l0` decl, not the source's id and \
398             not another peer's"
399        );
400        assert_eq!(
401            own_video_track(&roster(false), PeerId(7), Source::Camera, true),
402            None,
403            "no low decl of ours in the roster (echo not landed / never announced)"
404        );
405        assert_eq!(
406            own_video_track(&view, PeerId(7), Source::Screen, false),
407            None,
408            "a different source is a different track"
409        );
410    }

412    /// THE contract: the `_l0` decl is in the roster exactly while the low encoder is
413    /// producing. A decl left behind after a teardown is worse than none at all — the
414    /// receiver's `pick` reads the roster as liveness, subscribes to a track that
415    /// never produces, and walks VIDEO_STALL → FIRST_KEYFRAME_DEADLINE → the restart
416    /// ladder (minutes of black video) instead of just taking the source.
417    #[test]
418    fn deactivating_the_low_layer_withdraws_its_decl_and_reactivating_republishes() {
419        let mut decl = LowDecl::Absent;
420        // Dark: no receiver wants the layer, so a frame announces nothing.
421        assert!(!decl.on_frame(false));
422        assert_eq!(decl, LowDecl::Absent);
423        // Demand + a first frame → announce, exactly once.
424        assert!(decl.on_frame(true));
425        assert_eq!(decl, LowDecl::Live);
426        assert!(!decl.on_frame(true), "the announce is first-frame only");
427        // Every receiver un-caps (the video-pause withdrawal, or a tile grew) → the
428        // planner tears the encoder down, so the decl must leave the roster.
429        assert!(decl.on_active(false));
430        decl.withdrawn();
431        assert_eq!(decl, LowDecl::Absent);
432        // Demand returns and the encoder produces again → a fresh decl.
433        assert!(
434            decl.on_frame(true),
435            "a re-activated low layer must be re-announced or no receiver can ever \
436             pick it again"
437        );
438        assert_eq!(decl, LowDecl::Live);
439    }
440
441    /// Frames still queued in the low channel when the planner deactivates the layer
442    /// must not resurrect the decl: the encoder is going away regardless, so an
443    /// announce off a straggler would leave exactly the stale decl we just withdrew.
444    #[test]
445    fn straggler_frames_after_a_teardown_do_not_reannounce() {
446        let mut decl = LowDecl::Absent;
447        assert!(decl.on_frame(true));
448        assert!(decl.on_active(false));
449        decl.withdrawn();
450        assert!(!decl.on_frame(false), "the layer is no longer wanted");
451        assert_eq!(decl, LowDecl::Absent);
452    }
453
454    /// A withdrawal is owed as soon as the layer dies, but it can only be SENT once
455    /// the server has echoed the decl's id back. Until then it stays pending — never
456    /// dropped, or the stale decl outlives the encoder forever.
457    #[test]
458    fn a_withdrawal_stays_pending_until_the_track_id_is_known() {
459        let mut decl = LowDecl::Absent;
460        decl.on_frame(true);
461        assert!(decl.on_active(false));
462        assert!(
463            decl.withdraw_pending(),
464            "still owed while the echo is in flight"
465        );
466        decl.withdrawn();
467        assert!(!decl.withdraw_pending());
468    }
469
470    /// Re-activation while a withdrawal is still owed CANCELS it: the decl never left
471    /// the roster and its layer is producing again, so it is truthful as-is.
472    /// Republishing instead would leave the source with two `_l0` decls, the second of
473    /// which nothing would ever withdraw.
474    #[test]
475    fn reactivating_before_the_withdrawal_lands_cancels_it() {
476        let mut decl = LowDecl::Absent;
477        decl.on_frame(true);
478        decl.on_active(false);
479        assert!(!decl.on_active(true), "nothing to withdraw any more");
480        assert_eq!(decl, LowDecl::Live);
481        assert!(!decl.withdraw_pending());
482        assert!(
483            !decl.on_frame(true),
484            "the decl is still in the roster — a second Publish would duplicate it"
485        );
486    }
487
488    /// The low layer is purely additive to its source, so its decl must never outlive
489    /// the source's: turning the camera off `Unpublish`es the source and leaves the
490    /// low encoder gated on nothing, and a lone `_l0` would be picked by the next
491    /// capped receiver as soon as the source is republished.
492    #[test]
493    fn the_low_decl_never_outlives_its_source_decl() {
494        let mut decl = LowDecl::Absent;
495        decl.on_frame(true);
496        decl.on_source_gone();
497        assert!(decl.withdraw_pending());
498        decl.withdrawn();
499        assert_eq!(decl, LowDecl::Absent);
500        // Idempotent: a roster change with no source decl and no low decl is a no-op.
501        decl.on_source_gone();
502        assert_eq!(decl, LowDecl::Absent);
503    }
