1//! Codec choice: negotiation against the roster's common decode caps, the
2//! control-plane/wire mappings, and the mid-call [`CodecSwitch`] policy
3//! (downgrades land immediately, upgrades are debounced).

11/// Map the control-plane codec onto the wire codec byte. The two enums are kept in
12/// sync deliberately (see `hound_control::Codec` docs). Used by the publish path's
13/// codec-negotiation seam (and pinned by a unit test); kept here as the single
14/// source of truth even though the camera codec is currently hardcoded to H.264.
15#[allow(dead_code)]
16fn wire_codec(c: CtlCodec) -> WireCodec {
17    match c {
18        CtlCodec::Av1 => WireCodec::Av1,
19        CtlCodec::H264 => WireCodec::H264,
20        CtlCodec::H265 => WireCodec::H265,
21        CtlCodec::Opus => WireCodec::Opus,
22    }
23}
24
25/// Map the wire codec byte back onto the control-plane [`CtlCodec`], for the stat
26/// labels the receive path derives from a frame it could NOT decode — there is no
27/// engine [`VideoCodec`] to route through in that case, because no decoder was built.
28pub(in crate::call) fn wire_codec_to_ctl(c: WireCodec) -> CtlCodec {
29    match c {
30        WireCodec::Av1 => CtlCodec::Av1,
31        WireCodec::H264 => CtlCodec::H264,
32        WireCodec::H265 => CtlCodec::H265,
33        WireCodec::Opus => CtlCodec::Opus,
34    }
35}
36
37/// Map an engine [`VideoCodec`] onto the control-plane [`CtlCodec`] announced in
38/// `Publish`. Advisory — the per-packet wire byte is the decode source of truth.
39pub(in crate::call) fn video_codec_to_ctl(c: VideoCodec) -> CtlCodec {
40    match c {
41        VideoCodec::Av1 => CtlCodec::Av1,
42        VideoCodec::H265 => CtlCodec::H265,
43        VideoCodec::H264 => CtlCodec::H264,
44    }
45}

47/// Bitrate scale per codec for EQUAL perceptual quality, relative to H.264. A more
48/// efficient codec reaches the same quality at fewer bits, so we bank the efficiency
49/// as lower bandwidth — and the per-packet receive CPU that dominates a screenshare —
50/// instead of a sharper picture at the same bitrate. The BWE rung stays the quality
51/// target (an H.264-equivalent ceiling); this scales what the encoder actually emits.
52/// Conservative vs the codecs' headline gains (HEVC ~40-50%, AV1 ~50%+, more still on
53/// screen content) so quality never regresses.
54pub(in crate::call) fn codec_bitrate_factor(c: VideoCodec) -> f64 {
55    match c {
56        VideoCodec::H264 => 1.0,
57        VideoCodec::H265 => 0.65,
58        VideoCodec::Av1 => 0.5,
59    }
60}

62/// Pick the video codec to PUBLISH: the highest rung on the encode ladder
63/// (AV1 → H.265 → H.264, [`hound_engine::types::ENCODE_PRIORITY`]) this device can
64/// **hardware-encode** AND **every current remote peer can decode** (so the
65/// forward-no-transcode relay never hands a peer a codec it can't play), falling back
66/// to H.264. Read from the live roster: a less-capable peer shrinks
67/// `common_remote_decode_caps`, an all-modern call gets HEVC/AV1. `encode_caps` is
68/// probed once by the caller (the probe spins a throwaway VT session) and reused — at
69/// publish time and by the mid-call re-negotiator on every roster change.
70pub(in crate::call) fn negotiate_codec(
71    encode_caps: hound_engine::types::VideoCodecCaps,
72    view: &SessionView,
73) -> VideoCodec {
74    let common = view.common_remote_decode_caps();
75    let common_caps = hound_engine::types::VideoCodecCaps {
76        av1: common.av1,
77        h265: common.h265,
78        h264: common.h264,
79    };
80    hound_engine::types::pick_encode_codec(encode_caps, common_caps)
81}

83/// Encode a [`VideoCodec`] for the `AtomicU8` codec slot shared with the encode loop.
84/// The value doubles as a priority rank (higher = richer), so the re-negotiator can
85/// tell a downgrade from an upgrade with a plain comparison.
86pub(in crate::call) fn codec_rank(c: VideoCodec) -> u8 {
87    match c {
88        VideoCodec::Av1 => 2,
89        VideoCodec::H265 => 1,
90        VideoCodec::H264 => 0,
91    }
92}
93
94pub(in crate::call) fn codec_from_rank(v: u8) -> VideoCodec {
95    match v {
96        2 => VideoCodec::Av1,
97        1 => VideoCodec::H265,
98        _ => VideoCodec::H264,
99    }
100}

102/// How long a richer codec must stay negotiable before we actually switch UP to it,
103/// so a peer flapping in/out of the call doesn't thrash the encoder.
104pub(in crate::call) const CODEC_UPGRADE_DEBOUNCE: Duration = Duration::from_secs(5);
105
106/// Decides when a live encoder should switch codecs as the roster changes. Pure
107/// (no IO) so the policy is unit-tested. **Downgrades land immediately** — a peer
108/// that can't decode the current codec has no video until we drop to one it can —
109/// while **upgrades are debounced** (a peer leaving lets us climb, but there's no
110/// rush, and rushing invites thrash if they rejoin).
111pub(in crate::call) struct CodecSwitch {
112    current: VideoCodec,
113    /// A richer codec seen but not yet committed, and when it was first seen.
114    pending: Option<(VideoCodec, Instant)>,
115}
116
117impl CodecSwitch {
118    pub(in crate::call) fn new(initial: VideoCodec) -> Self {
119        Self {
120            current: initial,
121            pending: None,
122        }
123    }
124
125    /// Feed the freshly-negotiated codec for the current roster. Returns `Some(c)`
126    /// when the encoder should switch to `c` now, else `None` (hold).
127    pub(in crate::call) fn observe(
128        &mut self,
129        want: VideoCodec,
130        now: Instant,
131        debounce: Duration,
132    ) -> Option<VideoCodec> {
133        if want == self.current {
134            self.pending = None;
135            return None;
136        }
137        if codec_rank(want) < codec_rank(self.current) {
138            // Downgrade: commit immediately (correctness — a peer can't decode us).
139            self.pending = None;
140            self.current = want;
141            return Some(want);
142        }
143        // Upgrade: commit only once `want` has held for `debounce`.
144        match self.pending {
145            Some((p, since)) if p == want && now.duration_since(since) >= debounce => {
146                self.pending = None;
147                self.current = want;
148                Some(want)
149            }
150            Some((p, _)) if p == want => None,
151            _ => {
152                self.pending = Some((want, now));
153                None
154            }
155        }
156    }
157
158    /// When the encode loop should re-poll even without a roster change (to let a
159    /// debounced upgrade fire). `None` when nothing is pending.
160    pub(in crate::call) fn pending_deadline(&self, debounce: Duration) -> Option<Instant> {
161        self.pending.map(|(_, since)| since + debounce)
162    }
163}

243    // ─── Mid-call re-negotiation harness ────────────────────────────────────────
244    //
245    // Drives the FULL chain — roster (`SessionView`) → `common_remote_decode_caps`
246    // → `negotiate_codec` (gated by local HW-encode caps) → `CodecSwitch` — over
247    // simulated peer join/leave sequences, asserting the two reliability invariants
248    // a live mid-stream codec switch must hold: bounded recovery (a peer that can't
249    // decode us forces a decodable codec immediately) and no thrash (a flapping peer
250    // never storms the encoder). The actual VideoToolbox rebuild is hardware and out
251    // of scope here; this pins the decision layer that drives it.
252
253    fn dcaps(av1: bool, h265: bool, h264: bool) -> DecodeCaps {
254        DecodeCaps { av1, h265, h264 }
255    }
256
257    fn ecaps(av1: bool, h265: bool, h264: bool) -> hound_engine::types::VideoCodecCaps {
258        hound_engine::types::VideoCodecCaps { av1, h265, h264 }
259    }
260
261    /// A fresh roster: self (`id` 1, full caps) plus the given remote peers.
262    fn roster(remotes: &[(u32, DecodeCaps)]) -> SessionView {
263        let mut peers = vec![PeerInfo {
264            id: PeerId(1),
265            decode_caps: DecodeCaps::ALL,
266            user_id: uuid::Uuid::nil(),
267        }];
268        peers.extend(remotes.iter().map(|&(id, c)| PeerInfo {
269            id: PeerId(id),
270            decode_caps: c,
271            user_id: uuid::Uuid::nil(),
272        }));
273        let mut v = SessionView::new();
274        v.apply(&ServerMessage::Welcome {
275            protocol_version: PROTOCOL_VERSION,
276            self_id: PeerId(1),
277            peers,
278            tracks: vec![],
279        });
280        v
281    }

283    fn full_encode() -> hound_engine::types::VideoCodecCaps {
284        ecaps(true, true, true)
285    }

336    /// A peer widening its caps mid-call (`UpdateDecodeCaps` → `PeerDecodeCaps`)
337    /// feeds the re-negotiator exactly like a departure: the constraint lifts, and
338    /// the upgrade commits once it has held for the debounce window — never
339    /// instantly, because a codec that just appeared can just as well be turned back
340    /// off, and each switch costs an encoder rebuild + IDR.
341    #[test]
342    fn renegotiate_upgrades_after_a_peer_widens_its_decode_caps() {
343        let enc = full_encode();
344        let mut view = roster(&[(2, DecodeCaps::ALL), (3, dcaps(false, false, true))]);
345        let mut sw = CodecSwitch::new(negotiate_codec(enc, &view)); // H264
346        let t0 = Instant::now();
347
348        // Peer 3's AV1 decoder shows up (its openh264/HW caps changed) and it
349        // re-announces — no join, no leave, same roster.
350        view.apply(&ServerMessage::PeerDecodeCaps {
351            peer: PeerId(3),
352            decode_caps: DecodeCaps::ALL,
353        });
354        let want = negotiate_codec(enc, &view);
355        assert_eq!(want, VideoCodec::Av1, "the common ceiling rose");
356        assert_eq!(sw.observe(want, t0, DB), None, "no instant upgrade");
357        assert_eq!(
358            sw.observe(want, t0 + DB + Duration::from_millis(1), DB),
359            Some(VideoCodec::Av1),
360            "climbs once the widened caps are stable"
361        );
362
363        // The reverse (the user turns the codec off mid-call) is a downgrade, so it
364        // commits at once — a receiver that cannot decode must never wait a window.
365        view.apply(&ServerMessage::PeerDecodeCaps {
366            peer: PeerId(3),
367            decode_caps: dcaps(false, false, true),
368        });
369        assert_eq!(
370            sw.observe(
371                negotiate_codec(enc, &view),
372                t0 + DB + Duration::from_millis(2),
373                DB
374            ),
375            Some(VideoCodec::H264)
376        );
377    }

397    #[test]
398    fn renegotiate_does_not_storm_when_peer_flaps() {
399        let enc = full_encode();
400        let mut view = roster(&[(2, DecodeCaps::ALL)]);
401        let mut sw = CodecSwitch::new(negotiate_codec(enc, &view)); // AV1
402        let t0 = Instant::now();
403
404        // An H.264-only peer joins/leaves every 1 s for 12 s — far faster than the
405        // 5 s upgrade debounce. Downgrades are immediate (correctness); upgrades must
406        // never commit mid-flap, so the encoder switches exactly ONCE (the first
407        // downgrade) rather than thrashing every second.
408        let mut switches = 0;
409        for i in 0..12u64 {
410            if i % 2 == 0 {
411                view.apply(&ServerMessage::PeerJoined {
412                    peer: PeerInfo {
413                        id: PeerId(3),
414                        decode_caps: dcaps(false, false, true),
415                        user_id: uuid::Uuid::nil(),
416                    },
417                });
418            } else {
419                view.apply(&ServerMessage::PeerLeft { peer: PeerId(3) });
420            }
421            let now = t0 + Duration::from_secs(i);
422            if sw.observe(negotiate_codec(enc, &view), now, DB).is_some() {
423                switches += 1;
424            }
425        }
426        assert_eq!(switches, 1, "a flapping peer must not thrash the encoder");
427    }
