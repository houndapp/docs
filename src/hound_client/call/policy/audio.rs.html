1//! Audio policy, both directions: capture-side mic route warmup + dead-silence
2//! detection, per-stream playback gain application, publisher-side silence
3//! suppression ([`SilenceGate`]), and the receiver's gap classification / PLC
4//! math / level reporting / cadence diagnostics.

6use std::time::{Duration, Instant};

8/// Scale decoded PCM by `gain`, saturating at the i16 rails (a >1 boost must clip
9/// gracefully, never wrap). Unity is the caller's fast path — call only when
10/// `gain != 1.0`. Pure; unit-tested.
11pub(in crate::call) fn apply_gain(samples: &mut [i16], gain: f32) {
12    for s in samples.iter_mut() {
13        *s = (*s as f32 * gain).clamp(i16::MIN as f32, i16::MAX as f32) as i16;
14    }
15}

17/// One-shot mic route warmup: a non-48 kHz capture device (a Bluetooth headset in
18/// HFP mode) may open onto an SCO route that is not carrying audio yet — the
19/// stream runs, delivering silence, and no error ever fires. Reopening once,
20/// [`hound_engine::api::ROUTE_WARMUP_DELAY`] after open, lands on the settled
21/// route; 48 kHz devices never arm. The playout side has the same policy inside
22/// the mix bus; this is the capture-side half.
23pub(in crate::call) struct MicRouteWarmup {
24    pub(in crate::call) due: Option<Instant>,
25}
26
27impl MicRouteWarmup {
28    /// Arm (or disarm) for a just-opened device at `device_rate`.
29    pub(in crate::call) fn arm(device_rate: u32, now: Instant) -> Self {
30        Self {
31            due: (device_rate != 48_000).then(|| now + hound_engine::api::ROUTE_WARMUP_DELAY),
32        }
33    }
34
35    /// True exactly once, when the warmup deadline passes.
36    pub(in crate::call) fn fire(&mut self, now: Instant) -> bool {
37        match self.due {
38            Some(t) if now >= t => {
39                self.due = None;
40                true
41            }
42            _ => false,
43        }
44    }
45}

47/// Consecutive exactly-zero mic frames before warning once that the input is
48/// delivering pure silence — a dead capture route (BT/SCO, revoked permission, a
49/// silent virtual device) produces bit-exact zeros, which a live room's noise
50/// floor never does. ~3 s at 20 ms frames.
51const MIC_SILENCE_WARN_FRAMES: u32 = 150;
52
53/// Detects a dead-silent capture stream (see [`MIC_SILENCE_WARN_FRAMES`]).
54/// [`Self::observe`] returns `true` exactly once per silent streak, at the
55/// threshold; any non-zero sample resets the streak (and re-arms the warning).
56#[derive(Default)]
57pub(in crate::call) struct MicSilenceWatch {
58    zero_frames: u32,
59    warned: bool,
60}
61
62impl MicSilenceWatch {
63    pub(in crate::call) fn observe(&mut self, samples: &[i16]) -> bool {
64        if samples.iter().all(|&s| s == 0) {
65            self.zero_frames = self.zero_frames.saturating_add(1);
66            if self.zero_frames >= MIC_SILENCE_WARN_FRAMES && !self.warned {
67                self.warned = true;
68                return true;
69            }
70        } else {
71            self.zero_frames = 0;
72            self.warned = false;
73        }
74        false
75    }
76}

80/// The DTX/coded size threshold. Owned by the codec that produces the sizes
81/// ([`hound_engine::api::DTX_MAX_BYTES`]) and measured against the live encoder there,
82/// so the publisher, the receiver and the codec cannot drift apart.
83pub(in crate::call) use hound_engine::api::DTX_MAX_BYTES;
84
85/// Consecutive DTX-sized frames before publication stops. 10 frames = 200 ms.
86///
87/// libopus keeps CODING for its own hangover (~400 ms measured) after the signal
88/// stops, so by the time frames are DTX-sized the trailing syllable is already fully
89/// encoded and sent — this window is pure insurance on top of that, not the thing that
90/// protects the tail. Sized so a VAD dip *inside* speech (a stop consonant, a breath)
91/// never reaches it: those are one to three frames, and the hangover swallows them
92/// whole.
93pub(in crate::call) const SUPPRESS_AFTER: u32 = 10;
94
95/// How often a suppressed publisher still emits one frame. 500 ms, bounded from both
96/// directions:
97///
98/// * **Upper** — the receiver's `subscribe::AUDIO_STALL` reaps a subscription that
99///   delivers nothing for 8 s. A suppressed sender must stay visibly alive, and it
100///   must do so with margin over a lost keepalive or two, so the interval has to be a
101///   small fraction of that window (16× here).
102/// * **Lower** — every keepalive is an object in the moq count-rolled audio group
103///   (`AUDIO_GROUP_OBJECTS = 4`), so at 2/s a group rolls every ~2 s. That keeps the
104///   ordered substream ticking over — groups still open and close, late-join still has
105///   boundaries to attach at — while costing ~1/25th of the streams a speaking sender
106///   opens. Going much slower would leave a group open for tens of seconds; much
107///   faster would give back the saving.
108pub(in crate::call) const KEEPALIVE_INTERVAL: Duration = Duration::from_millis(500);
109
110/// Consecutive coded frames required to CLEAR suppression. 3 frames = 60 ms.
111///
112/// The way-down hysteresis, and without it suppression barely works on a real
113/// microphone. A room has a noise floor, and the encoder's periodic comfort-noise
114/// refresh on any non-zero floor is not the ~8 B it is on bit-exact zeros — measured on
115/// this encoder it is a **fully coded 36–57 B frame**, i.e. indistinguishable from
116/// speech to a size test. It lands roughly every 20 frames. Treating each one as
117/// "speech" re-arms the [`SUPPRESS_AFTER`] confirmation window every ~400 ms, so a
118/// quiet room ships ~half its frames and suppression never reaches its steady state at
119/// all (`a_room_tone_refresh_never_reopens_the_gate` is the measurement).
120///
121/// The discriminator is RUN LENGTH, not size: measured 46/46 refreshes at −70 and
122/// −60 dBFS were ONE isolated coded frame between runs of ~20 suppressed ones
123/// (`hound_engine::codec::opus::a_room_tone_refresh_is_one_isolated_coded_frame`),
124/// while speech codes continuously for as long as someone is talking. 3 is the smallest
125/// run that clears an isolated refresh with margin, and 60 ms is far shorter than any
126/// utterance — nothing a person says produces two coded frames and then stops.
127///
128/// **This is not a debounce on publication.** Every coded frame is published the instant
129/// it appears, suppressed or not; this only governs when the gate stops *withholding
130/// the silence between them*. The first word is never delayed — see
131/// [`SilenceGate::publish`].
132pub(in crate::call) const RESUME_AFTER: u32 = 3;
133
134/// **Publisher-side silence suppression**: stop putting silence on the wire, keep the
135/// subscription warm. Pure (bytes + `now` in, publish/skip out) so the whole policy is
136/// unit-tested without an encoder or a socket.
137///
138/// ## Why suppress at the SOURCE rather than unsubscribe at the receiver
139///
140/// The alternative — a receiver that unsubscribes all but the top-K loudest peers —
141/// was evaluated and rejected, for the same reason the wire-priority band refuses a
142/// dominant-speaker promotion (see `subscribe::audio_strictly_outranks_video_on_the_wire`):
143/// **it puts a re-subscribe in the path of the first word.** A client-driven resubscribe
144/// cannot beat ~0.9–1.1 s best case (a 300 ms report grid, the 500 ms reconcile tick, a
145/// subscribe RTT, and the 80 ms playout target), so the first word is clipped every
146/// time; a peer who has never won a dominance election has no rank at all and stays
147/// unsubscribed through multi-second holes; and audio resubscribe churn on the shared
148/// QUIC connection is the documented voice-killing incident class. Suppressing at the
149/// source has none of those properties: every subscription stays open, so the first
150/// voiced frame rides a connection that is already there — zero attack latency.
151///
152/// ## What it actually saves, and why it is worth doing at all
153///
154/// Opus DTX already collapses the PAYLOAD of a silent frame to 1–2 bytes, so the cost
155/// of a silent track is ~95% FRAMING, not audio: a 16-byte object header plus moq
156/// varints ≈ 19 B every 20 ms ≈ 8 kbps, and a fresh QUIC uni-stream per 4-object group
157/// = 12.5 streams/s per track. Across 31 silent remotes that is ~250 kbps and ~390
158/// streams/s of pure silence arriving at each receiver, and the SFU's O(N²) stream term
159/// is entirely indifferent to whether those objects contain speech. Suppression
160/// collapses the wire bytes, the stream churn AND the receiver's decode of silence at
161/// once, which no receiver-side policy can do.
162///
163/// ## The rules
164///
165/// * Bytes-driven, never a VAD of our own: the encoder already ran one, and its verdict
166///   is legible in the payload size ([`DTX_MAX_BYTES`]).
167/// * Suppression engages only after [`SUPPRESS_AFTER`] CONSECUTIVE DTX-sized frames.
168/// * While suppressed, one real encoder frame goes out per [`KEEPALIVE_INTERVAL`] — a
169///   genuine DTX packet, never a synthetic one, so the receiver decodes real comfort
170///   noise and the stall watchdog sees real arrivals. A coded frame that lands while
171///   suppressed IS that interval's keepalive (it was published, so the wire and the
172///   watchdog both saw it); no redundant DTX frame follows it.
173/// * **Every coded frame is published, always, at any point in any run.** No coded
174///   frame is ever withheld — that is the invariant the whole design rests on, and the
175///   reason resume costs nothing.
176/// * Clearing the suppressed STATE is the one thing that waits: it needs
177///   [`RESUME_AFTER`] consecutive coded frames, because on a real mic a single
178///   isolated coded frame is the comfort-noise refresh, not somebody talking. This
179///   delays no audio — only the resumption of publishing the *silence* around it.
180/// * Capture timestamps are untouched throughout. The receiver's gap classification is
181///   timestamp-driven ([`classify_audio_gap`]) and the jitter estimator is differential
182///   over `(arrival Δ) − (ts Δ)`, so real timestamps make a suppression gap cost
183///   exactly nothing on both; invented contiguous ones would break both.
184///
185/// ## It can never engage on a headless client, by construction
186///
187/// The synthetic capture source every bot and e2e participant publishes is
188/// deliberately VAD-ACTIVE — `dtx_never_suppresses_the_synthetic_capture_source`
189/// (`hound_engine::codec::opus`) pins ≥98% of its frames as coded — so this gate is
190/// permanently open for it and the suite's audio-liveness floors are unaffected. That
191/// is a load-bearing property, not an accident: do not "fix" the synthetic source into
192/// something quieter. The e2e scenario that exercises suppression injects real silence
193/// through `SyntheticSilence` instead.
194///
195/// Mute is NOT this gate's business: it gates before encode (`audio_tx`), so what
196/// arrives here is only ever unmuted-but-quiet.
197#[derive(Debug, Default)]
198pub(in crate::call) struct SilenceGate {
199    /// Consecutive DTX-sized frames observed (reset by any coded frame). Only counts
200    /// on the way IN — once suppressed, the flag below is the state.
201    dtx_run: u32,
202    /// Consecutive coded frames observed (reset by any DTX-sized frame). Clears
203    /// suppression at [`RESUME_AFTER`]; see that constant for why one is not enough.
204    coded_run: u32,
205    /// Whether silence is being withheld right now. Explicit rather than derived from
206    /// `dtx_run`, because the entry and exit conditions are deliberately asymmetric.
207    suppressed: bool,
208    /// When we last published — the keepalive clock while suppressed.
209    last_published: Option<Instant>,
210}
211
212impl SilenceGate {
213    /// Feed one encoded frame's payload length; returns whether to publish it.
214    pub(in crate::call) fn publish(&mut self, encoded_len: usize, now: Instant) -> bool {
215        if encoded_len > DTX_MAX_BYTES {
216            // A coded frame ALWAYS goes out — mid-suppression included. Publishing it
217            // is also what makes it this interval's keepalive, so `last_published`
218            // advances and no redundant DTX frame chases it 500 ms later.
219            self.dtx_run = 0;
220            self.coded_run = self.coded_run.saturating_add(1);
221            if self.coded_run >= RESUME_AFTER {
222                // A RUN, not one frame: an isolated coded frame in a quiet room is the
223                // comfort-noise refresh (see [`RESUME_AFTER`]).
224                self.suppressed = false;
225            }
226            self.last_published = Some(now);
227            return true;
228        }
229        self.coded_run = 0;
230        if !self.suppressed {
231            self.dtx_run = self.dtx_run.saturating_add(1);
232            if self.dtx_run <= SUPPRESS_AFTER {
233                // Still inside the confirmation window: publish as before.
234                self.last_published = Some(now);
235                return true;
236            }
237            self.suppressed = true;
238        }
239        // Suppressed: one keepalive per interval, and nothing else.
240        match self.last_published {
241            Some(t) if now.duration_since(t) < KEEPALIVE_INTERVAL => false,
242            _ => {
243                self.last_published = Some(now);
244                true
245            }
246        }
247    }
248
249    /// Whether the gate is currently withholding frames (a stats gauge, so a
250    /// "this peer sent nothing" report can name suppression rather than a stall).
251    pub(in crate::call) fn suppressed(&self) -> bool {
252        self.suppressed
253    }
254}

258/// What a jump between consecutive in-order capture timestamps means.
259#[derive(Debug, Clone, Copy, PartialEq, Eq)]
260pub(in crate::call) enum AudioGap {
261    /// The expected next frame (allowing for sub-frame capture jitter).
262    Continuous,
263    /// Brief, plausibly-recoverable loss: conceal this many frames with PLC.
264    Conceal(i64),
265    /// The sender was not transmitting — DTX/suppression, or a capture
266    /// discontinuity. Play silence across it; concealing it would be a lie.
267    Silence,
268}
269
270/// Classify a timestamp gap. Pure, and the ONE place the PLC-vs-silence line is drawn.
271///
272/// A suppressed publisher leaves gaps of a keepalive interval (~500 ms = 25 frames) and,
273/// on the resume edge, of however long it stayed quiet. Both land far past
274/// [`MAX_PLC_FRAMES`] and so classify as [`AudioGap::Silence`]: no PLC is fabricated and
275/// nothing is counted as loss, which is what keeps a quiet peer from reading as a lossy
276/// one in the stats and the audio report.
277///
278/// `prev_was_dtx` — the frame BEFORE the gap was DTX-sized — settles the case the gap
279/// size alone gets wrong: the **resume edge**. A publisher that resumes mid-keepalive
280/// leaves a gap of however long it had been since its last keepalive, which is uniform
281/// in `(0, KEEPALIVE_INTERVAL]` and so lands inside the PLC band ~20% of the time. The
282/// sender was not transmitting across it — that is what a DTX-sized predecessor means —
283/// so there is nothing to conceal: PLC would extrapolate comfort noise into the front of
284/// the first word back, and the frames would be counted as loss, putting a peer who was
285/// merely quiet into the audio report's `network-loss` rule. Silence is the truth.
286pub(in crate::call) fn classify_audio_gap(
287    gap_us: i64,
288    frame_us: i64,
289    prev_was_dtx: bool,
290) -> AudioGap {
291    let missing = frames_missing(gap_us, frame_us);
292    if missing == 0 {
293        AudioGap::Continuous
294    } else if missing <= MAX_PLC_FRAMES && !prev_was_dtx {
295        AudioGap::Conceal(missing)
296    } else {
297        AudioGap::Silence
298    }
299}
300
301/// The speaking level to report for a decoded frame, given the size of the packet it
302/// came from. A DTX-sized packet ([`DTX_MAX_BYTES`]) carries comfort noise, not speech,
303/// so it reports **0** rather than metering the noise: without this a suppressed peer's
304/// UI ring sits at whatever the comfort noise happens to measure (the meter is
305/// logarithmic, so a −40 dBFS floor already reads ~0.3 of full scale) for as long as
306/// they stay quiet — a lit ring on a peer who is saying nothing. More necessary since
307/// the capture chain's loudness contract landed, not less: AGC2's make-up gain rides
308/// the sender's noise floor up with the speech, so the level a suppressed peer would
309/// otherwise report is HIGHER than it used to be. Pure; unit-tested.
310pub(in crate::call) fn reported_audio_level(encoded_len: usize, decoded_rms: f32) -> f32 {
311    if encoded_len <= DTX_MAX_BYTES {
312        0.0
313    } else {
314        hound_engine::api::level_from_rms(decoded_rms)
315    }
316}

318/// Max consecutive frames concealed with Opus PLC on a timestamp gap. A gap of up to
319/// this many frames (~100 ms at 20 ms/frame) is plausibly recoverable brief loss worth
320/// concealing; beyond that the gap is the sender being silent (DTX emits no frames) or
321/// a capture discontinuity — fabricating extrapolated audio there is wrong (it mislabels
322/// silence as loss and splices garbage into an active stream). Silence is played across
323/// the larger gap instead.
324pub(in crate::call) const MAX_PLC_FRAMES: i64 = 5;
325
326/// Frames genuinely missing in a `gap_us` jump between consecutive in-order frame
327/// timestamps (frame size `frame_us`). Tolerant of sub-frame capture-timing jitter:
328/// SCO/resampled sources never space frames at exactly 20 ms, so a gap under ~1.5
329/// frames is the next frame arriving a little late, NOT a loss. Counting it would
330/// fabricate one spurious PLC frame per real frame (concealment dwarfing real audio).
331/// Pure + unit-tested. The caller separately caps the result at [`MAX_PLC_FRAMES`].
332pub(in crate::call) fn frames_missing(gap_us: i64, frame_us: i64) -> i64 {
333    if frame_us <= 0 || gap_us <= 0 {
334        return 0;
335    }
336    let frames = gap_us as f64 / frame_us as f64;
337    if frames < 1.5 {
338        0
339    } else {
340        (frames.round() as i64 - 1).max(0)
341    }
342}

344/// How many decoded frames per cadence-diagnostic window (~5 s at 50 fps).
345const CADENCE_WINDOW_FRAMES: u64 = 250;
346
347/// Per-stream receive-cadence diagnostic (a retained debugging tool). Accumulates over a
348/// window and, on completion, reports the signals that explain an active-stream gap the
349/// smoothed jitter number hides — see the call site. Pure bookkeeping; no IO.
350#[derive(Default)]
351pub(in crate::call) struct CadenceDiag {
352    real: u64,
353    pub(in crate::call) plc: u64,
354    pub(in crate::call) fec: u64,
355    pub(in crate::call) big_gaps: u64,
356    max_gap_us: i64,
357    first_ts: Option<i64>,
358    first_now: i64,
359}
360
361/// One completed [`CadenceDiag`] window. A struct rather than a tuple because the fields
362/// are read individually by a human reading a log line, and `plc` vs `fec` next to each
363/// other is the whole point of the pair.
364pub(in crate::call) struct CadenceWindow {
365    /// Frames genuinely decoded from the wire.
366    pub(in crate::call) real: u64,
367    /// Slots filled by PLC extrapolation.
368    pub(in crate::call) plc: u64,
369    /// Slots redeemed from a carrier packet's in-band FEC — real signal recovered, and
370    /// counted apart from `plc` because the two say different things about the link AND
371    /// because a redeemed slot is the one kind of fill that is easy to mistake for a
372    /// received frame. A nonzero `fec` next to an inflated `jitterbuf_ms` on one peer
373    /// while a co-present peer sits at the floor is the signature of the redemption path
374    /// polluting the jitter estimator; that is exactly how the 2026-08 regression was
375    /// finally attributed, and it was only visible because `plc` happened to count it.
376    pub(in crate::call) fec: u64,
377    /// Gaps classified as sender silence (DTX / suppression / capture discontinuity).
378    pub(in crate::call) big_gaps: u64,
379    pub(in crate::call) max_gap_ms: i64,
380    /// `> 0` ⇒ the local clock advanced faster than the sender's timestamps over the
381    /// window (receiver consumes faster than sender produces → the buffer drains).
382    pub(in crate::call) skew_ppm: i64,
383}
384
385impl CadenceDiag {
386    /// Fold a timestamp gap between consecutive in-order frames.
387    pub(in crate::call) fn note_gap(&mut self, gap_us: i64) {
388        if gap_us > self.max_gap_us {
389            self.max_gap_us = gap_us;
390        }
391    }
392
393    /// Record one decoded frame at sender timestamp `ts` / local time `now`. Returns the
394    /// window summary once per [`CADENCE_WINDOW_FRAMES`], then resets.
395    pub(in crate::call) fn observe(&mut self, ts: i64, now: i64) -> Option<CadenceWindow> {
396        if self.first_ts.is_none() {
397            self.first_ts = Some(ts);
398            self.first_now = now;
399        }
400        self.real += 1;
401        if self.real < CADENCE_WINDOW_FRAMES {
402            return None;
403        }
404        let ts_elapsed = ts - self.first_ts.unwrap_or(ts);
405        let now_elapsed = now - self.first_now;
406        let skew_ppm = if ts_elapsed > 0 {
407            ((now_elapsed - ts_elapsed) as f64 / ts_elapsed as f64 * 1_000_000.0) as i64
408        } else {
409            0
410        };
411        let out = CadenceWindow {
412            real: self.real,
413            plc: self.plc,
414            fec: self.fec,
415            big_gaps: self.big_gaps,
416            max_gap_ms: self.max_gap_us / 1000,
417            skew_ppm,
418        };
419        *self = CadenceDiag::default();
420        Some(out)
421    }
422}

475        // Halving scales; a >1 boost saturates at the rails instead of wrapping.
476        let mut half = [10_000i16, -10_000, 0];

533    /// A coded-speech payload, and a DTX one — the only two inputs the gate has.
534    const VOICED: usize = 62;
535    const DTX: usize = 2;
536    /// The periodic SILK comfort-noise refresh on a BIT-EXACT-ZERO input (a muted
537    /// virtual device, the e2e `SyntheticSilence` seam): 8 B, still under the
538    /// threshold. On a real mic it is not this — see [`TONE_REFRESH`].
539    const REFRESH: usize = 8;
540    /// The periodic comfort-noise refresh on a real ROOM TONE: a fully coded 36–57 B
541    /// frame, measured on this encoder at −70…−55 dBFS
542    /// (`hound_engine::codec::opus::a_room_tone_refresh_is_one_isolated_coded_frame`).
543    /// Indistinguishable from speech by size — only its run length gives it away.
544    const TONE_REFRESH: usize = 47;
545    /// How often that refresh lands, in frames. Measured ~20.6; 21 here.
546    const TONE_REFRESH_EVERY: u32 = 21;
547    const FRAME: Duration = Duration::from_millis(20);

596    fn a_suppressed_sender_publishes_about_two_keepalives_a_second() {
597        // THE saving, asserted at the unit level: 10 s of silence at 50 frames/s must
598        // leave the wire carrying ~2 frames/s, not 50.

600        let mut gate = SilenceGate::default();

613             ({published} of {frames} frames published)"
614        );
615        // And the saving is the point: in the suppressed steady state (past the
616        // confirmation window) >95% of silent frames never reach the wire at all.
617        let steady = frames as usize - SUPPRESS_AFTER as usize;
618        assert!(
619            keepalives * 20 < steady,
620            "expected <5% of the steady-state silent frames on the wire, got \
621             {keepalives}/{steady}"
622        );

641    /// **A real microphone must not defeat suppression** — the failure a size-only gate
642    /// has on every non-synthetic client, and the reason [`RESUME_AFTER`] exists.
643    ///
644    /// The sequence is the measured one: a room with any noise floor produces ~20
645    /// suppressed frames, then ONE fully coded 36–57 B comfort-noise refresh, forever
646    /// (`hound_engine::codec::opus::a_room_tone_refresh_is_one_isolated_coded_frame`).
647    /// Without hysteresis each refresh resets the DTX run, the gate spends the next
648    /// 200 ms back inside its confirmation window, and **26.25 frames/s reach the wire
649    /// — 52% of them, against the 2/s the design claims.** Measured, both numbers, by
650    /// driving the real encoder's output through this gate.
651    #[test]
652    fn a_room_tone_refresh_never_reopens_the_gate() {
653        let t0 = Instant::now();
654        let mut gate = SilenceGate::default();
655        let frames = 3000u32; // 60 s
656        let bytes = |i: u32| {
657            if i.is_multiple_of(TONE_REFRESH_EVERY) {
658                TONE_REFRESH
659            } else {
660                DTX
661            }
662        };
663        let mut published = 0u32;
664        let mut withheld_coded = 0u32;
665        for i in 0..frames {
666            let n = bytes(i);
667            if gate.publish(n, t0 + FRAME * i) {
668                published += 1;
669            } else if n > DTX_MAX_BYTES {
670                withheld_coded += 1;
671            }
672        }
673        assert_eq!(
674            withheld_coded, 0,
675            "THE invariant: no coded frame is ever withheld, refresh or not"
676        );
677        assert!(
678            gate.suppressed(),
679            "an isolated comfort-noise refresh must not clear suppression — that is \
680             what makes a quiet real mic ship half its frames"
681        );
682        // The honest wire rate. Measured against the live encoder at −70…−65 dBFS:
683        // 2.39 frames/s, which is the refresh cadence itself (~1 per 21 frames) — the
684        // refreshes ARE the keepalives, so almost no extra DTX keepalive is due. The
685        // band is generous on both sides; what it must exclude is the 26/s a
686        // hysteresis-free gate produces.
687        let rate = published as f64 / (frames as f64 * 0.02);
688        assert!(
689            (2.0..=4.0).contains(&rate),
690            "a quiet real mic must settle at ~2.4 frames/s (the refresh cadence), got \
691             {rate:.2}/s — a hysteresis-free gate measures 26.25/s here"
692        );
693    }

695    /// Speech onset out of a SUPPRESSED gate publishes from frame one. The hysteresis
696    /// added by [`RESUME_AFTER`] governs when silence starts flowing again, never
697    /// whether audio does: the first three coded frames are published while the gate
698    /// still reads suppressed, and it clears on the third.

716            "and {RESUME_AFTER} sustained coded frames clear the state"
