22/// Own the session for the plane's lifetime: run it, reconnect it, and stop on a
23/// shutdown, a fatal auth outcome, or a version mismatch. `initial` is the
24/// connection [`RealtimeClient::connect`](super::RealtimeClient::connect) already
25/// established, if it managed to.
26pub(super) async fn supervise<D: Dial>(
27    dialer: D,
28    gate: WtGate,
29    events: mpsc::UnboundedSender<RealtimeEvent>,
30    mut commands: mpsc::UnboundedReceiver<Command>,
31    initial: Option<D::Pipe>,
32) {
33    let mut session = ClientSession::new();
34    let mut backoff = Backoff::new();
35    let mut leg = initial;
36    // Outlives every session: a subscribe taken while down, or one whose
37    // connection died before the answer, is confirmed by the next `Welcome`'s
38    // re-issue. Dropped when this task returns, which releases every waiter
39    // rather than making each wait out its own deadline against a dead plane.
40    let mut pending = Pending::new();
41
42    loop {
43        // Establish, unless we are holding a connection already. Commands are
44        // serviced throughout: while down they only move the reducer's desired
45        // set, which is exactly what must survive an outage.
46        let pipe = match leg.take() {
47            Some(pipe) => pipe,
48            None => {
49                let delay = backoff.next_delay(rng_jitter);
50                tracing::info!(
51                    attempt = backoff.attempt(),
52                    delay_ms = delay.as_millis() as u64,
53                    "realtime: reconnect backoff"
54                );
55                // `biased` here (unlike in `run_session`): a shutdown must win
56                // over both the sleep and the dial, or a sign-out would wait out
57                // a 15s backoff rung and then a 13s dial.
58                tokio::select! {
59                    biased;
60                    _ = drain_offline(&mut commands, &mut session, &mut pending) => return,
61                    _ = tokio::time::sleep(delay) => {}
62                }
63                let attempt = tokio::select! {
64                    biased;
65                    _ = drain_offline(&mut commands, &mut session, &mut pending) => return,
66                    result = connect_leg(&dialer, &gate) => result,
67                };
68                match attempt {
69                    Ok(pipe) => pipe,
70                    Err(err) => match classify(&err) {
71                        Disposition::Fatal => {
72                            tracing::error!(%err, "realtime: session refused — giving up");
73                            let _ = events.send(RealtimeEvent::AuthExpired);
74                            return;
75                        }
76                        Disposition::Retryable => {
77                            tracing::warn!(%err, "realtime: connect attempt failed (retrying)");
78                            continue;
79                        }
80                    },
81                }
82            }
83        };
84
85        // Live. `run_session` returns when the pipe dies or a shutdown lands.
86        let started = Instant::now();
87        let stopped = run_session(pipe, &mut session, &events, &mut commands, &mut pending).await;
88        // Read BEFORE `disconnected()` clears it: a session that reached
89        // `Welcome` proves this network path works end to end, which is what
90        // re-arms the WebTransport probe.
91        if session.is_live() {
92            gate.note_welcomed_session();
93        }
94        for event in session.disconnected() {
95            let _ = events.send(event.into());
96        }
97        // Only a session that PROVED durable re-arms the ladder — see
98        // `Backoff::on_session_death`. A server that accepts and immediately
99        // closes must not pin us on the 500ms rung.
100        backoff.on_session_death(started.elapsed());
101        match stopped {
102            Stopped::Shutdown => return,
103            // Reconnecting would re-run a handshake the server refuses
104            // identically forever; the event already told the app.
105            Stopped::VersionMismatch => {
106                tracing::error!("realtime: protocol version mismatch — the plane is stopping");
107                return;
108            }
109            Stopped::Transport => {}
110        }
111    }
112}

114/// Apply commands to the reducer while there is no live transport, forever.
115/// Resolves ONLY when the plane is shutting down (an explicit `Shutdown` or the
116/// last [`RealtimeClient`](super::RealtimeClient) dropped), which is what makes
117/// it usable as the losing arm of a `select!`: the actions it would produce are
118/// all empty while down, so dropping the future mid-poll loses nothing.
119async fn drain_offline(
120    commands: &mut mpsc::UnboundedReceiver<Command>,
121    session: &mut ClientSession,
122    pending: &mut Pending,
123) {
124    while let Some(command) = commands.recv().await {
125        if matches!(command, Command::Shutdown) {
126            return;
127        }
128        // Down: `apply` records desire and yields no frames to drop. A subscribe
129        // taken here parks its waiter, which the next `Welcome`'s re-issue and
130        // its confirmation release.
131        let _ = apply(session, command, pending);
132    }
133}

135/// Why [`run_session`] returned.
136#[derive(Debug, Clone, Copy, PartialEq, Eq)]
137pub(super) enum Stopped {
138    /// The transport ended (peer closed, read error, a send failed, or a
139    /// deadline this shell owns expired).
140    Transport,
141    /// The caller asked the plane to stop.
142    Shutdown,
143    /// The server and this build cannot agree a protocol version. Terminal, and
144    /// deliberately not an auth outcome.
145    VersionMismatch,
146}

148/// Drive one established pipe: handshake, then pump frames and commands until
149/// something ends it.
150///
151/// Mirrors the call plane's reader discipline — a frame that fails to serialize
152/// is skipped, never fatal, and an unparseable inbound frame is skipped by
153/// [`MessagePipeExt::recv_json`] (that is the protocol's additive-compatibility
154/// rule). Generic over the pipe so both legs and the test fake share one loop.
155///
156/// Four deadlines run alongside the two data arms, all absolute (they are NOT
157/// re-armed by an unrelated wakeup, which is exactly the bug a per-iteration
158/// `sleep` would have): the keepalive probe, the inbound-silence limit, the
159/// `Welcome` deadline, and the one retry an `Unavailable` answer earns.
160pub(super) async fn run_session<P: Probe>(
161    mut pipe: P,
162    session: &mut ClientSession,
163    events: &mpsc::UnboundedSender<RealtimeEvent>,
164    commands: &mut mpsc::UnboundedReceiver<Command>,
165    pending: &mut Pending,
166) -> Stopped {
167    // The transport is up: the reducer answers with `Hello`, and nothing else
168    // may be sent until the `Welcome` comes back.
169    if !send_all(&mut pipe, session.connected()).await {
170        return Stopped::Transport;
171    }
172
173    // Read once: the timers must not borrow the pipe, which the frame arm needs
174    // mutably.
175    let idle = pipe.idle_period();
176    let silence = pipe.silence_limit();
177    let mut next_probe = deadline(idle);
178    let mut next_silence_check = deadline(silence);
179    let mut inbound_mark = pipe.inbound_count();
180    let mut welcome_by = deadline(Some(WELCOME_DEADLINE));
181    let mut retry_subscriptions_at: Option<Instant> = None;
182    let mut retried_subscriptions = false;
183
184    let stopped = loop {
185        // NOT `biased`: with a fixed order the inbound arm is always polled
186        // first, so a server flooding events starves the command, keepalive and
187        // deadline arms — a shutdown would never be seen and the liveness timers
188        // would never fire. Random order costs nothing and cannot starve.
189        tokio::select! {
190            frame = pipe.recv_json::<ServerMessage>() => {
191                let Some(msg) = frame else { break Stopped::Transport };
192                let reaction = session.handle(msg);
193                let mismatch = reaction.events.iter().any(|event| {
194                    matches!(event, Event::Error { code: ErrorCode::VersionMismatch, .. })
195                });
196                let unavailable = reaction.events.iter().any(|event| {
197                    matches!(event, Event::Error { code: ErrorCode::Unavailable, .. })
198                });
199                for event in reaction.events {
200                    // A confirmed subscribe releases whoever asked for it. Done
201                    // here rather than in the reducer because the waiters are a
202                    // property of this shell's callers, not of the protocol.
203                    if let Event::Subscribed { space_id } = event {
204                        confirm(pending, space_id);
205                    }
206                    let _ = events.send(event.into());
207                }
208                if mismatch {
209                    break Stopped::VersionMismatch;
210                }
211                // Welcomed: the deadline has done its job.
212                if session.is_live() {
213                    welcome_by = None;
214                    // A server that does not confirm subscribes will never
215                    // answer these, and the `Welcome` is the first moment we
216                    // know that. Waiting them out would stall every open of a
217                    // space against an older backend.
218                    if !session.confirms_subscriptions() {
219                        confirm_all(pending);
220                    }
221                }
222                // The server decided nothing, so the subscription is neither
223                // issued nor refused. Exactly one retry per session — a backend
224                // stuck answering `Unavailable` must not earn a subscribe storm
225                // every 5s — and anything past that is the next `Welcome`'s
226                // reconciliation, which re-issues the whole desired set anyway.
227                if unavailable && !retried_subscriptions {
228                    retried_subscriptions = true;
229                    retry_subscriptions_at = deadline(Some(UNAVAILABLE_RETRY));
230                }
231                if !send_all(&mut pipe, reaction.actions).await {
232                    break Stopped::Transport;
233                }
234            }
235            command = commands.recv() => {
236                // `None` = every `RealtimeClient` handle dropped without a
237                // `disconnect`; treat it as one.
238                let Some(command) = command else { break Stopped::Shutdown };
239                if matches!(command, Command::Shutdown) {
240                    let _ = send_all(&mut pipe, session.bye()).await;
241                    break Stopped::Shutdown;
242                }
243                if !send_all(&mut pipe, apply(session, command, pending)).await {
244                    break Stopped::Transport;
245                }
246            }
247            () = tick_at(next_probe) => {
248                if let Err(error) = pipe.ping().await {
249                    tracing::debug!(%error, "realtime: keepalive failed");
250                    break Stopped::Transport;
251                }
252                next_probe = deadline(idle);
253            }
254            () = tick_at(next_silence_check) => {
255                // Writes into a blackholed TCP path keep succeeding, so silence
256                // is the only evidence the peer is gone.
257                let seen = pipe.inbound_count();
258                if seen == inbound_mark {
259                    tracing::warn!(
260                        limit = ?silence,
261                        "realtime: nothing inbound (not even a pong) — treating the transport as dead"
262                    );
263                    break Stopped::Transport;
264                }
265                inbound_mark = seen;
266                next_silence_check = deadline(silence);
267            }
268            () = tick_at(welcome_by) => {
269                tracing::warn!(
270                    deadline = ?WELCOME_DEADLINE,
271                    "realtime: no Welcome after the handshake — reconnecting"
272                );
273                break Stopped::Transport;
274            }
275            () = tick_at(retry_subscriptions_at) => {
276                retry_subscriptions_at = None;
277                if !send_all(&mut pipe, session.reissue_subscriptions()).await {
278                    break Stopped::Transport;
279                }
280            }
281        }
282    };
283    pipe.close().await;
284    stopped
285}

287/// Apply one command to the reducer, parking or releasing its waiter.
288pub(super) fn apply(
289    session: &mut ClientSession,
290    command: Command,
291    pending: &mut Pending,
292) -> Vec<Action> {
293    match command {
294        Command::Subscribe { space_id, ack } => {
295            let actions = session.subscribe(space_id);
296            if let Some(ack) = ack {
297                // Two ways there is nothing to wait for: this connection is
298                // already in the room (a repeat open of the same space), or its
299                // server does not confirm subscribes at all. Otherwise park the
300                // waiter until the confirmation lands — possibly on a later
301                // connection, since the reducer re-issues the desired set on
302                // every `Welcome`.
303                if session.is_subscribed(space_id)
304                    || (session.is_live() && !session.confirms_subscriptions())
305                {
306                    let _ = ack.send(());
307                } else {
308                    pending.entry(space_id).or_default().push(ack);
309                }
310            }
311            actions
312        }
313        Command::Unsubscribe(space_id) => {
314            // Nobody is waiting to receive a space they just left; dropping the
315            // senders releases them now instead of at their deadline.
316            pending.remove(&space_id);
317            session.unsubscribe(space_id)
318        }
319        Command::Screenshare { space_id, enabled } => session.screenshare(space_id, enabled),
320        // Handled by the caller: it also ends the loop.
321        Command::Shutdown => session.bye(),
322    }
323}

325/// Put every action on the wire. `false` means the pipe is dead. A frame that
326/// fails to ENCODE is skipped (it cannot be fixed by reconnecting, and killing
327/// the session over one malformed frame would be a self-inflicted outage);
328/// anything else is a transport failure.
329async fn send_all<P: MessagePipe>(pipe: &mut P, actions: Vec<Action>) -> bool {
330    for Action::Send(msg) in actions {
331        match pipe.send_json(&msg).await {
332            Ok(()) => {}
333            Err(PipeError::Encode(error)) => {
334                tracing::error!(%error, ?msg, "realtime: skipping unencodable frame");
335            }
336            Err(error) => {
337                tracing::debug!(%error, "realtime: send failed — transport is gone");
338                return false;
339            }
340        }
341    }
342    true
343}
344
345/// An absolute deadline `period` from now, or `None` for a timer that never
346/// fires.
347fn deadline(period: Option<Duration>) -> Option<Instant> {
348    period.map(|period| Instant::now() + period)
349}
350
351/// A `select!` arm that fires at `at` — or never, when there is no deadline.
352async fn tick_at(at: Option<Instant>) {
353    match at {
354        Some(at) => tokio::time::sleep_until(at).await,
355        None => std::future::pending().await,
356    }
357}

495    /// THE fix, at the seam a caller sees: `subscribe_space` does not return on
496    /// having *queued* a subscribe — it returns when the server says the room is
497    /// joined.
498    ///

500    /// that opened a space and immediately expected its traffic (a test's
501    /// `open_text().await` then a send; a person opening a channel someone is
502    /// typing in) was, before this, still several hops from being in the room,
503    /// and anything fanned out in that window went to a room it was not in yet.
504    /// Nothing re-fetches such a message, so it is lost outright — which is what
505    /// made the failure load-correlated and invisible.
506    #[tokio::test]
507    async fn subscribe_space_returns_only_once_the_server_confirms_the_room_join() {
508        let space = id(7);

512

520        assert!(
521            wire.lock()

864
865    /// `Unavailable` means the server could not *decide*, so the subscription is
866    /// neither issued nor refused. The space stays desired and exactly one retry
867    /// goes out; the answer to that retry is what finally delivers events.
868    #[tokio::test(start_paused = true)]
869    async fn an_unavailable_answer_keeps_the_space_desired_and_retries_once() {
870        let space = id(7);
871        // Desired before the session starts, so the `Welcome` is what issues it
872        // and the refusal below is unambiguously the answer to THAT subscribe.
873        let mut session = ClientSession::new();
874        session.subscribe(space);
875        let wire = open_wire([welcome(), error_frame(ErrorCode::Unavailable, "db is down")]);
876        let mut running = start(wire.clone(), session);
877        running.connected().await;
878
879        let subscribes = || {
880            wire.lock()
881                .unwrap()
882                .sent
883                .iter()
884                .filter(|m| **m == ClientMessage::SubscribeSpace { space_id: space })
885                .count()
886        };
887        assert_eq!(subscribes(), 1, "the welcome issued it once");
888
889        // Past the retry delay: exactly one more attempt, unprompted.
890        tokio::time::sleep(UNAVAILABLE_RETRY * 2).await;
891        assert_eq!(subscribes(), 2, "the unanswered subscribe is asked again");
892
893        // That retry is answered: events flow.
894        push(&wire, message_frame(space, "after the blip"));
895        running
896            .wait_for(|event| matches!(event, RealtimeEvent::MessageCreated(_)))
897            .await;
898
899        let (stopped, session, events) = running.stop().await;
900        assert_eq!(stopped, Stopped::Shutdown);
901        assert!(
902            session.desired_spaces().eq([space]),
903            "an unavailable answer must never prune the desired set",
904        );
905        assert!(
906            events.iter().any(|event| matches!(
907                event,
908                RealtimeEvent::Error { code, .. } if *code == ErrorCode::Unavailable
909            )),
910            "the refusal is surfaced, never swallowed: {events:?}",
911        );
912        assert!(
913            events.iter().any(|event| matches!(
914                event,
915                RealtimeEvent::MessageCreated(m) if m.contents == "after the blip"
916            )),
917            "and the eventual success delivers events: {events:?}",
918        );
919    }
