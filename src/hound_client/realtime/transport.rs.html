25/// Where one attempt's legs come from: the config lookup, the credential, and
26/// the two dials.
27///
28/// A trait rather than three free functions so [`connect_leg`]'s *decisions* —
29/// which leg, in which order, and what a failure at each step means — are
30/// testable without a socket, and so the supervisor itself can be driven against
31/// scripted outcomes.
32pub(super) trait Dial: Send + Sync {
33    /// An established connection over either leg.
34    type Pipe: Probe + Send;
35
36    /// `GET /realtime/config`.
37    fn config(&self) -> impl std::future::Future<Output = Result<RealtimeConfig>> + Send;
38
39    /// The current session credential.
40    fn token(&self) -> Result<String>;
41
42    /// Dial the advertised WebTransport endpoint.
43    fn wt(
44        &self,
45        url: &str,
46        token: &str,
47        cert_hash: Option<&str>,
48    ) -> impl std::future::Future<Output = Result<Self::Pipe>> + Send;
49
50    /// Dial the WebSocket leg (derived from the REST base URL).
51    fn ws(&self, token: &str) -> impl std::future::Future<Output = Result<Self::Pipe>> + Send;
52}
53
54/// One attempt: fetch the config, then WebTransport (if advertised, allowed, and
55/// working), else the WebSocket.
56///
57/// The config fetch is what re-mints the credential: it goes through
58/// [`RestClient`], whose 401 hook rotates the session and republishes the access
59/// token, so the token read straight after is current by construction. Reading a
60/// captured token instead is the stale-JWT strand this ordering exists to
61/// prevent.
62///
63/// A config fetch that FAILS does not fail the attempt: the WebSocket leg is
64/// derived from the REST base URL and needs nothing the config carries, so a
65/// backend blip in that one endpoint must not cost the whole plane. Only an
66/// `Auth` error propagates — that one is an answer (it has already survived the
67/// REST layer's refresh-and-retry).
68pub(super) async fn connect_leg<D: Dial>(dialer: &D, gate: &WtGate) -> Result<D::Pipe> {
69    let config = match dialer.config().await {
70        Ok(config) => Some(config),
71        Err(err @ HoundError::Auth { .. }) => return Err(err),
72        Err(err) => {
73            tracing::warn!(%err, "realtime: config fetch failed — going straight to the websocket leg");
74            None
75        }
76    };
77    let token = dialer.token()?;
78
79    let advertised = config
80        .as_ref()
81        .and_then(|config| config.webtransport_url.as_deref())
82        .filter(|url| !url.is_empty());
83    if let Some(url) = advertised {
84        if gate.wt_allowed() {
85            let cert_hash = config
86                .as_ref()
87                .and_then(|config| config.cert_hash.as_deref());
88            match dialer.wt(url, &token, cert_hash).await {
89                Ok(pipe) => {
90                    gate.note_wt_success();
91                    return Ok(pipe);
92                }
93                // EVERY WebTransport failure falls back, refusals included — see
94                // the module docs on where an auth verdict may come from.
95                Err(err) => {
96                    gate.note_wt_failure();
97                    tracing::warn!(%err, "realtime: webtransport unavailable — falling back to websocket");
98                }
99            }
100        } else {
101            tracing::debug!("realtime: skipping webtransport (recent dials all failed)");
102        }
103    }
104    dialer.ws(&token).await
105}

107/// Whether the WebTransport leg is worth dialling at all.
108///
109/// On a network that blocks UDP the WT dial cannot succeed, and paying
110/// [`WT_CONNECT_TIMEOUT`] before every single reconnect is a self-inflicted 3s
111/// outage on top of whatever caused the reconnect. After
112/// [`WT_FAILURE_STREAK`] failures in a row the leg is skipped until either a
113/// session actually reached `Welcome` (proof the path changed) or
114/// [`WT_RETRY_AFTER`] elapses — a laptop moves between networks without
115/// restarting, so this can never be permanent.
116///
117/// Scoped to one plane rather than the process: a fresh plane (a new sign-in)
118/// deserves a fresh probe, and process-global state would be untestable in
119/// parallel.
120pub(super) struct WtGate {
121    state: std::sync::Mutex<GateState>,
122}
123
124#[derive(Default)]
125struct GateState {
126    failures: u32,
127    /// When the streak last tripped the skip, for the expiry.
128    blocked_at: Option<Instant>,
129}
130
131impl WtGate {
132    pub(super) fn new() -> Self {
133        Self {
134            state: std::sync::Mutex::new(GateState::default()),
135        }
136    }
137
138    fn wt_allowed(&self) -> bool {
139        let state = self.state.lock().expect("wt gate mutex");
140        match state.blocked_at {
141            Some(blocked_at) => Instant::now().duration_since(blocked_at) >= WT_RETRY_AFTER,
142            None => state.failures < WT_FAILURE_STREAK,
143        }
144    }
145
146    fn note_wt_failure(&self) {
147        let mut state = self.state.lock().expect("wt gate mutex");
148        state.failures = state.failures.saturating_add(1);
149        if state.failures >= WT_FAILURE_STREAK {
150            // Re-stamped on every failure, so a dial let through by the expiry
151            // that fails again restarts the cooldown rather than retrying hard.
152            state.blocked_at = Some(Instant::now());
153        }
154    }
155
156    fn note_wt_success(&self) {
157        *self.state.lock().expect("wt gate mutex") = GateState::default();
158    }
159
160    /// A session reached `Welcome` — whichever leg carried it, this network path
161    /// works, so re-probe WebTransport on the next attempt.
162    pub(super) fn note_welcomed_session(&self) {
163        *self.state.lock().expect("wt gate mutex") = GateState::default();
164    }
165}

169/// One established connection, per leg. Both arms are [`MessagePipe`]s; the enum
170/// exists only because the trait uses `impl Future` and so is not
171/// dyn-compatible.
172///
173/// Both arms are boxed: a `WebSocketStream`'s buffers alone are ~1.4 KiB, and
174/// this value is moved from the connect helper through the spawn into the
175/// supervisor.
176pub(super) enum Leg {
177    Wt(Box<WtPipe>),
178    Ws(Box<WsPipe>),
179}

181/// The production [`Dial`]: the REST client is the config source, the credential
182/// source, and the origin both legs are derived from.
183pub(super) struct RestDialer(pub(super) Arc<RestClient>);
184
185impl Dial for RestDialer {
186    type Pipe = Leg;
187
188    async fn config(&self) -> Result<RealtimeConfig> {
189        // Bounded by `RestClient`'s own connect + read timeouts (10s/20s), so a
190        // half-dead ingress cannot park an attempt here.
191        self.0.realtime_config().await
192    }
193
194    fn token(&self) -> Result<String> {
195        self.0.token().ok_or_else(|| {
196            HoundError::auth(
197                AuthCode::NoSessionToken,
198                "cannot connect realtime: no session token",
199            )
200        })
201    }
202
203    async fn wt(&self, url: &str, token: &str, cert_hash: Option<&str>) -> Result<Leg> {
204        Ok(Leg::Wt(Box::new(connect_wt(url, token, cert_hash).await?)))
205    }
206
207    async fn ws(&self, token: &str) -> Result<Leg> {
208        Ok(Leg::Ws(Box::new(
209            connect_ws(self.0.base_url(), token, WS_CONNECT_TIMEOUT).await?,
210        )))
211    }
212}

214/// A WebTransport control stream, holding the endpoint and connection it rides.
215///
216/// Both are kept alive deliberately: dropping the `Endpoint` (or the
217/// `Connection`) tears the stream down under us — the same lifetime trap the
218/// call plane's iroh endpoint has.
219pub(super) struct WtPipe {
220    stream: FramedBi,
221    _connection: hound_wt::Connection,
222    _client: WtClient,
223}

225/// The URL the WebTransport leg dials: exactly what the backend advertised, with
226/// no query of our own. The credential rides an `Authorization` header instead —
227/// a `:path` lands in access logs and proxy traces.
228fn wt_dial_url(advertised: &str) -> &str {
229    advertised.trim_end_matches('/')
230}

232/// Dial the advertised WebTransport endpoint and open the control stream.
233///
234/// DNS pins are supplied through a [`PinnedResolver`] closure read at LOOKUP
235/// time, so a harness that installs overrides after this process started still
236/// has them honoured while the URL keeps the real hostname (TLS stays honest).
237async fn connect_wt(url: &str, token: &str, cert_hash: Option<&str>) -> Result<WtPipe> {
238    let trust = match cert_hash {
239        Some(hash) => Trust::CertHash(hound_wt::CertHash::parse(hash).map_err(|e| {
240            HoundError::realtime(
241                RealtimeCode::BadCertHash,
242                format!("bad realtime cert hash: {e}"),
243            )
244        })?),
245        None => Trust::NativeRoots,
246    };
247    let mut options = ClientOptions::new(trust);
248    options.resolver = PinnedResolver::from_fn(|| dns_overrides().to_vec());
249
250    let client = WtClient::new(options).map_err(wt_error)?;
251    let url = wt_dial_url(url);
252    let authorization = format!("Bearer {token}");
253
254    let connection = tokio::time::timeout(
255        WT_CONNECT_TIMEOUT,
256        client.connect_with_headers(url, &[("Authorization", &authorization)]),
257    )
258    .await
259    .map_err(|_| {
260        HoundError::network(
261            NetCode::WebTransportTimeout,
262            "webtransport connect timed out",
263        )
264    })?
265    .map_err(wt_error)?;
266    // The client opens the control stream, mirroring the browser API (and what
267    // the backend's accept loop waits for). Bounded: a session that never gives
268    // us a stream must not park the attempt.
269    let stream = FramedBi::open_within(&connection, WT_STREAM_TIMEOUT)
270        .await
271        .map_err(wt_error)?;
272    Ok(WtPipe {
273        stream,
274        _connection: connection,
275        _client: client,
276    })
277}

279/// Map a WebTransport failure onto the shared error taxonomy — always as a
280/// retryable transport failure, `Rejected` included.
281///
282/// `Rejected` is wtransport's verdict for *any* non-2xx answer to the CONNECT
283/// and for a CONNECT stream stopped before it is answered, so a pod rollout, a
284/// misrouted host and a genuine refusal are indistinguishable here. Calling that
285/// an auth failure signed users out over a deploy; auth verdicts now come only
286/// from the WebSocket handshake's status and from `/realtime/config`.
287fn wt_error(error: WtError) -> HoundError {
288    HoundError::network(NetCode::WebTransport, redact(&error.to_string()))
289}

329/// The universally-reachable leg. Mirrors the backend's `WsPipe`: text frames
330/// are the protocol, and ping/pong/binary are skipped rather than treated as
331/// frames, so a proxy's keepalive can never desync the stream. They are still
332/// COUNTED — a pong is the only evidence a silent peer is alive.
333pub(super) struct WsPipe {
334    socket: tokio_tungstenite::WebSocketStream<
335        tokio_tungstenite::MaybeTlsStream<tokio::net::TcpStream>,
336    >,
337    inbound: u64,
338}
339
340impl MessagePipe for WsPipe {
341    async fn send(&mut self, frame: String) -> std::result::Result<(), PipeError> {
342        use futures::SinkExt;
343        self.socket
344            .send(tokio_tungstenite::tungstenite::Message::Text(frame.into()))
345            .await
346            .map_err(|e| PipeError::Transport(e.to_string()))
347    }
348
349    async fn recv(&mut self) -> Option<std::result::Result<String, PipeError>> {
350        use futures::StreamExt;
351        use tokio_tungstenite::tungstenite::Message;
352        loop {
353            let message = self.socket.next().await?;
354            if message.is_ok() {
355                self.inbound += 1;
356            }
357            return match message {
358                Ok(Message::Text(text)) => Some(Ok(text.to_string())),
359                Ok(Message::Close(_)) => None,
360                // Pongs answer our own Pings; tungstenite answers the server's
361                // Pings itself. Binary is not part of this protocol.
362                Ok(_) => continue,
363                Err(error) => Some(Err(PipeError::Transport(error.to_string()))),
364            };
365        }
366    }
367
368    async fn close(&mut self) {
369        let _ = self.socket.close(None).await;
370    }
371}
372
373impl Liveness for WsPipe {
374    async fn ping(&mut self) -> std::result::Result<(), PipeError> {
375        use futures::SinkExt;
376        self.socket
377            .send(tokio_tungstenite::tungstenite::Message::Ping(
378                Default::default(),
379            ))
380            .await
381            .map_err(|e| PipeError::Transport(e.to_string()))
382    }
383
384    fn inbound_count(&self) -> u64 {
385        self.inbound
386    }
387}

454/// The liveness timings a leg needs, on top of [`hound_wt::Liveness`]'s probe.
455///
456/// Methods rather than associated consts because [`Leg`] is one type covering
457/// two transports with different answers;
458/// [`run_session`](super::supervisor::run_session) reads them once, so the
459/// timers never borrow the pipe.
460pub(super) trait Probe: MessagePipe + Liveness {
461    /// Quiet period after which a keepalive probe is due. `None` for a transport
462    /// that keeps itself alive — WebTransport's QUIC keep-alive is configured on
463    /// the endpoint, so its arm simply never fires.
464    fn idle_period(&self) -> Option<Duration>;
465
466    /// How long total inbound silence (frames AND pongs) may last before the
467    /// peer is declared gone. `None` where the transport reaps its own.
468    fn silence_limit(&self) -> Option<Duration>;
469}

471/// Build the realtime WebSocket URL from the REST base — the same scheme
472/// rewrite the call plane's `control_ws_url` does, against the route the backend
473/// registers (`GET /realtime/ws`). No credential: it goes in a header.
474fn realtime_ws_url(base_url: &str) -> Result<String> {
475    let base = base_url.trim_end_matches('/');
476    let ws_base = if let Some(rest) = base.strip_prefix("https://") {
477        format!("wss://{rest}")
478    } else if let Some(rest) = base.strip_prefix("http://") {
479        format!("ws://{rest}")
480    } else if base.starts_with("ws://") || base.starts_with("wss://") {
481        base.to_owned()
482    } else {
483        return Err(HoundError::realtime(
484            RealtimeCode::BadBaseScheme,
485            format!("api_base has no http(s) scheme: {base_url}"),
486        ));
487    };
488    Ok(format!("{ws_base}/realtime/ws"))
489}

491/// Caps on what a peer can make this client buffer, matched to the frame cap the
492/// WebTransport leg's length-delimited codec enforces and the backend applies to
493/// both legs. `max_message_size` bounds a reassembled fragmented message,
494/// `max_frame_size` a single frame — either alone leaves the other unbounded.
495fn ws_config() -> tokio_tungstenite::tungstenite::protocol::WebSocketConfig {
496    tokio_tungstenite::tungstenite::protocol::WebSocketConfig::default()
497        .max_message_size(Some(hound_wt::MAX_FRAME_BYTES))
498        .max_frame_size(Some(hound_wt::MAX_FRAME_BYTES))
499}

501/// Open the WebSocket leg, honouring the process's DNS pins.
502///
503/// A pinned host is dialled by opening the TCP socket OURSELVES against the
504/// pinned address and only then running the TLS + WebSocket handshake against
505/// the request's original hostname — so SNI and certificate validation still
506/// prove we reached the intended deployment, which an IP-in-the-URL rewrite
507/// would not. An unpinned host takes `connect_async`'s ordinary path.
508///
509/// `timeout` bounds the WHOLE dial (TCP, TLS, upgrade); it is a parameter only
510/// so tests can pin it.
511async fn connect_ws(base_url: &str, token: &str, timeout: Duration) -> Result<WsPipe> {
512    use tokio_tungstenite::tungstenite::client::IntoClientRequest;
513    use tokio_tungstenite::tungstenite::http::header::AUTHORIZATION;
514
515    let url = realtime_ws_url(base_url)?;
516    let mut request = url.into_client_request().map_err(|e| {
517        HoundError::realtime(
518            RealtimeCode::BadWsUrl,
519            format!("bad realtime ws url: {}", redact(&e.to_string())),
520        )
521    })?;
522    let authorization = format!("Bearer {token}").parse().map_err(|_| {
523        HoundError::auth(
524            AuthCode::BadHeaderValue,
525            "session token is not a valid header value",
526        )
527    })?;
528    request.headers_mut().insert(AUTHORIZATION, authorization);
529    let (host, port) = ws_authority(request.uri())?;
530
531    let dial = async {
532        match pinned_addr(dns_overrides(), &host, port) {
533            Some(addr) => {
534                tracing::debug!(%host, %addr, "realtime: dialling a pinned address");
535                let tcp = tokio::net::TcpStream::connect(addr).await.map_err(|e| {
536                    HoundError::network(
537                        NetCode::RealtimeWsTcp,
538                        format!("realtime ws tcp connect: {e}"),
539                    )
540                })?;
541                tokio_tungstenite::client_async_tls_with_config(
542                    request,
543                    tcp,
544                    Some(ws_config()),
545                    None,
546                )
547                .await
548            }
549            None => {
550                tokio_tungstenite::connect_async_with_config(request, Some(ws_config()), false)
551                    .await
552            }
553        }
554        .map_err(ws_error)
555    };
556
557    let (socket, _response) = tokio::time::timeout(timeout, dial).await.map_err(|_| {
558        HoundError::network(
559            NetCode::RealtimeWsTimeout,
560            format!("realtime ws connect timed out after {timeout:?}"),
561        )
562    })??;
563    Ok(WsPipe { socket, inbound: 0 })
564}

566/// Map a WebSocket handshake failure onto the shared error taxonomy, preserving
567/// an HTTP status so [`classify`](crate::call::reconnect::classify) can call
568/// 401/403 fatal.
569fn ws_error(error: tokio_tungstenite::tungstenite::Error) -> HoundError {
570    use tokio_tungstenite::tungstenite::Error;
571    match error {
572        Error::Http(response) => {
573            HoundError::from_status(response.status().as_u16(), "realtime ws handshake refused")
574        }
575        other => HoundError::network(
576            NetCode::RealtimeWsConnect,
577            redact(&format!("realtime ws connect failed: {other}")),
578        ),
579    }
580}

582/// Scrub a `token=…` query value out of anything headed for a log.
583///
584/// This client no longer puts a credential in a URL, but the URLs it dials are
585/// partly server-supplied (`/realtime/config`'s `webtransport_url`), and a
586/// transport error quotes the URL it failed on. Cheap insurance that a
587/// mis-configured deployment cannot leak one through our logs.
588fn redact(text: &str) -> String {
589    const KEY: &str = "token=";
590    let mut out = String::with_capacity(text.len());
591    let mut rest = text;
592    while let Some(at) = rest.find(KEY) {
593        out.push_str(&rest[..at + KEY.len()]);
594        out.push_str("REDACTED");
595        let tail = &rest[at + KEY.len()..];
596        let end = tail
597            .find(['&', ' ', '"', '\'', ',', ')'])
598            .unwrap_or(tail.len());
599        rest = &tail[end..];
600    }
601    out.push_str(rest);
602    out
603}

605/// The host and port a WebSocket request dials (`wss` ⇒ 443, `ws` ⇒ 80).
606fn ws_authority(uri: &tokio_tungstenite::tungstenite::http::Uri) -> Result<(String, u16)> {
607    let host = uri.host().ok_or_else(|| {
608        HoundError::realtime(
609            RealtimeCode::WsUrlNoHost,
610            format!("realtime ws url has no host: {uri}"),
611        )
612    })?;
613    let port = uri.port_u16().or_else(|| match uri.scheme_str() {
614        Some("wss") => Some(443),
615        Some("ws") => Some(80),
616        _ => None,
617    });
618    let port = port.ok_or_else(|| {
619        HoundError::realtime(
620            RealtimeCode::WsUrlNoPort,
621            format!("realtime ws url has no port: {uri}"),
622        )
623    })?;
624    Ok((host.to_owned(), port))
625}

627/// Pick the address to TCP-connect for `host:port` from a `curl --resolve`-style
628/// pin table. Deliberately identical to [`PinnedResolver`]'s rule so both legs
629/// resolve the same way: an exact `(host, port)` pin wins, otherwise a pin for
630/// the host alone is re-pointed at the port actually being dialled.
631fn pinned_addr(pins: &[(String, SocketAddr)], host: &str, port: u16) -> Option<SocketAddr> {
632    pins.iter()
633        .find(|(name, addr)| name == host && addr.port() == port)
634        .or_else(|| pins.iter().find(|(name, _)| name == host))
635        .map(|(_, addr)| SocketAddr::new(addr.ip(), port))
636}

638/// The process's installed DNS pins — empty unless the `dns-override` feature is
639/// on AND a harness installed them.
640fn dns_overrides() -> &'static [(String, SocketAddr)] {
641    #[cfg(feature = "dns-override")]
642    {
643        crate::dns::overrides()
644    }
645    #[cfg(not(feature = "dns-override"))]
646    {
647        &[]
648    }
649}

781    /// On a UDP-blocking network every reconnect would otherwise pay the WT
782    /// timeout before reaching the leg that works. After two failures in a row
783    /// the leg is skipped — and the skip expires, because a laptop changes
784    /// networks without restarting.
785    #[tokio::test(start_paused = true)]
786    async fn webtransport_is_skipped_after_a_failure_streak_and_retried_later() {
787        let blocked = || HoundError::network(NetCode::Transport, "udp is blocked");
788        let dialer = FakeDialer::new([
789            Attempt::WtFails(blocked()),
790            Attempt::WtFails(blocked()),
791            // Third attempt: WT is advertised but must not be dialled.
792            Attempt::WtFails(blocked()),
793            Attempt::WtFails(blocked()),
794        ]);
795        let reached = dialer.reached();
796        let gate = WtGate::new();
797
798        for _ in 0..2 {
799            assert!(connect_leg(&dialer, &gate).await.is_ok());
800        }
801        assert_eq!(reached.lock().unwrap().wt, 2);
802
803        assert!(connect_leg(&dialer, &gate).await.is_ok());
804        assert_eq!(
805            reached.lock().unwrap().wt,
806            2,
807            "the third attempt goes straight to the websocket",
808        );
809
810        tokio::time::advance(WT_RETRY_AFTER + Duration::from_secs(1)).await;
811        assert!(connect_leg(&dialer, &gate).await.is_ok());
812        assert_eq!(
813            reached.lock().unwrap().wt,
814            3,
815            "the skip expires and the leg is probed again",
816        );
817    }

940    /// The pin decision itself (no live TLS): an exact host+port pin wins, a
941    /// host-only pin is re-pointed at the dialled port, and an unpinned host
942    /// takes the ordinary resolver path. Same rule as `hound_wt`'s resolver, so
943    /// both legs reach the same box.
944    #[test]
945    fn pinned_hosts_dial_the_pin_and_others_resolve_normally() {
946        let addr = |s: &str| s.parse::<SocketAddr>().unwrap();
947        let pins = vec![
948            ("api.hound.chat".to_owned(), addr("10.0.0.1:443")),
949            ("rt.hound.chat".to_owned(), addr("10.0.0.2:7891")),
950        ];
951        assert_eq!(
952            pinned_addr(&pins, "api.hound.chat", 443),
953            Some(addr("10.0.0.1:443")),
954        );
955        assert_eq!(
956            pinned_addr(&pins, "api.hound.chat", 8443),
957            Some(addr("10.0.0.1:8443")),
958            "a host-only match keeps the dialled port",
959        );
960        assert_eq!(
961            pinned_addr(&pins, "other.hound.chat", 443),
962            None,
963            "an unpinned host falls through to the system resolver",
964        );
965        assert_eq!(pinned_addr(&[], "api.hound.chat", 443), None);
966    }
