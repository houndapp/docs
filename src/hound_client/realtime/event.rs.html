14/// A decoded server-side realtime event. The reducers fold these into
15/// view-models.
16///
17/// One-to-one with [`hound_realtime::Event`] plus [`RealtimeEvent::AuthExpired`]
18/// (no protocol frame can produce it — it is the shell's verdict that the
19/// session credential is dead) and [`RealtimeEvent::VersionMismatch`] (split out
20/// of `Error` because it, alone among the error codes, ends the plane). Kept as
21/// its own enum rather than a re-export exactly so those variants exist and so
22/// the app layer's `match` arms don't have to name a protocol type.
23#[derive(Clone, Debug, PartialEq)]
24pub enum RealtimeEvent {
25    /// The server joined this connection to a space's room; its events are
26    /// arriving from here on.
27    Subscribed { space_id: Uuid },
28    /// A new message in a subscribed space.
29    MessageCreated(MessageEvent),
30    /// An edit to an existing message.
31    MessageUpdated(MessageEvent),
32    /// A message's link previews are ready — the complete replacement list.
33    ///
34    /// Distinct from [`RealtimeEvent::MessageUpdated`] on purpose: `edited_at`
35    /// is what marks a message "(edited)", and a preview appearing is not an
36    /// edit. Folding this into the update event would stamp that marker on
37    /// every message that happens to contain a link.
38    MessageEmbeds {
39        message_id: Uuid,
40        space_id: Uuid,
41        embeds: Vec<hound_types::LinkEmbed>,
42    },
43    /// A new space in a joined community.
44    SpaceCreated(SpaceEvent),
45    /// A space was deleted.
46    SpaceRemoved { space_id: Uuid },
47    /// A user joined a voice channel (SFU peer id known).
48    VoiceJoined(VoiceJoinEvent),
49    /// A user left a voice channel.
50    VoiceLeft(VoiceLeaveEvent),
51    /// A participant toggled screenshare.
52    VoiceScreenShare(VoiceScreenShareEvent),
53    /// The full voice roster snapshot, pushed with every `Welcome` and on
54    /// server-side resync.
55    VoiceState(Vec<VoiceJoinEvent>),
56    /// The server refused something (an unauthorized subscribe, an unavailable
57    /// one). Surfaced, never swallowed, so contract drift is visible in the
58    /// logs.
59    Error { code: ErrorCode, message: String },
60    /// This build and the server cannot agree a protocol version. The plane will
61    /// NOT reconnect — retrying re-runs the same refused handshake — but the
62    /// SESSION is fine: this is never a reason to sign anyone out.
63    VersionMismatch { message: String },
64    /// The handshake completed: the session is live and its subscriptions are
65    /// re-issued.
66    ///
67    /// `reconnect` distinguishes the first connect of a plane from every later
68    /// one. Only the latter has a gap behind it: subscriptions resync, but the
69    /// message events fanned out while this plane had no connection are gone for
70    /// good, so the app layer repairs them with a history re-fetch (see
71    /// `HoundClient::arm_resync`). Never set on the first connect, where the
72    /// caller's own `open_space` load is the history.
73    Connected { reconnect: bool },
74    /// The transport went away. The supervisor is already reconnecting; desired
75    /// subscriptions survive and are re-issued on the next `Welcome`.
76    Disconnected,
77    /// The session credential is dead — the plane will NOT reconnect. The app
78    /// layer reacts as it does to any dead session (sign out).
79    AuthExpired,
80}
81
82impl From<Event> for RealtimeEvent {
83    fn from(event: Event) -> Self {
84        match event {
85            Event::Connected { reconnect } => Self::Connected { reconnect },
86            Event::Disconnected => Self::Disconnected,
87            Event::Subscribed { space_id } => Self::Subscribed { space_id },
88            Event::MessageCreated(m) => Self::MessageCreated(m),
89            Event::MessageUpdated(m) => Self::MessageUpdated(m),
90            Event::MessageEmbeds {
91                message_id,
92                space_id,
93                embeds,
94            } => Self::MessageEmbeds {
95                message_id,
96                space_id,
97                embeds,
98            },
99            Event::SpaceCreated(s) => Self::SpaceCreated(s),
100            Event::SpaceRemoved { space_id } => Self::SpaceRemoved { space_id },
101            Event::VoiceJoined(v) => Self::VoiceJoined(v),
102            Event::VoiceLeft(v) => Self::VoiceLeft(v),
103            Event::VoiceState(v) => Self::VoiceState(v),
104            Event::Screenshare(s) => Self::VoiceScreenShare(s),
105            // The one error code that ends the plane gets its own variant, so no
106            // caller has to know which codes are terminal.
107            Event::Error {
108                code: ErrorCode::VersionMismatch,
109                message,
110            } => Self::VersionMismatch { message },
111            Event::Error { code, message } => Self::Error { code, message },
112        }
113    }
114}

116/// What the public API asks the supervisor to do. The supervisor owns the
117/// [`ClientSession`](hound_realtime::ClientSession), so every caller-side intent
118/// crosses this channel rather than touching shared state.
119///
120/// Not `PartialEq` (and not `Clone`): a subscribe carries the caller's
121/// one-shot answer channel, which is a capability, not a value.
122#[derive(Debug)]
123pub(super) enum Command {
124    Subscribe {
125        space_id: Uuid,
126        /// Signalled once the server has confirmed the room join. Dropping it
127        /// (an unsubscribe, or a supervisor that stopped) releases the caller
128        /// immediately rather than making it wait out its deadline.
129        ack: Option<oneshot::Sender<()>>,
130    },
131    Unsubscribe(Uuid),
132    Screenshare {
133        space_id: Uuid,
134        enabled: bool,
135    },
136    /// Say goodbye and end the supervisor.
137    Shutdown,
138}
139
140/// Callers waiting for a space's subscribe confirmation. Owned by
141/// [`supervise`](super::supervisor::supervise), not by one session: a subscribe
142/// taken while the plane is down (or one whose connection dies before the
143/// answer) is answered by the NEXT session's confirmation, since the reducer
144/// re-issues the whole desired set on every `Welcome`.
145pub(super) type Pending = HashMap<Uuid, Vec<oneshot::Sender<()>>>;
146
147/// Release everyone waiting on `space_id`.
148pub(super) fn confirm(pending: &mut Pending, space_id: Uuid) {
149    for ack in pending.remove(&space_id).unwrap_or_default() {
150        let _ = ack.send(());
151    }
152}
153
154/// Release everyone waiting on anything — for a server that does not confirm
155/// subscribes at all, where waiting is waiting forever.
156pub(super) fn confirm_all(pending: &mut Pending) {
157    for (_, acks) in pending.drain() {
158        for ack in acks {
159            let _ = ack.send(());
160        }
161    }
162}

175    /// A `message_created` frame decodes into the struct the message reducer
176    /// renders from, attachments and all.
177    #[test]
178    fn a_message_frame_maps_to_the_message_event() {
179        let frame = serde_json::json!({
180            "type": "message_created",
181            "message": {
182                "id": id(10),
183                "space_id": id(2),
184                "user_id": id(3),
185                "contents": "hi",
186                "edited_at": null,
187                "profile": {
188                    "user_id": id(3), "global_profile_id": id(4),
189                    "community_profile_id": null, "name": "Aoife", "picture": "p"
190                },
191                "attachments": [{
192                    "id": id(20), "url": "https://cdn/x.png",
193                    "content_type": "image/png", "size_bytes": 12
194                }]
195            }
196        });
197        let event: RealtimeEvent = decode(frame);
198        let RealtimeEvent::MessageCreated(message) = event else {
199            panic!("expected MessageCreated");
200        };
201        assert_eq!(message.contents, "hi");
202        assert_eq!(message.attachments.len(), 1);
203        assert_eq!(message.attachments[0].size_bytes, 12);
204    }

206    /// The `Welcome`'s roster snapshot decodes into the voice events the roster
207    /// reducer merges — including the SFU peer id the call plane keys on.

209    fn a_welcome_maps_to_connected_plus_the_voice_roster() {
210        let frame = serde_json::json!({
211            "type": "welcome",
212            "protocol_version": PROTOCOL_VERSION,
213            "voice_states": [{

218        let mut session = ClientSession::new();
219        session.connected();
220        let events: Vec<RealtimeEvent> = session

287    /// Decode one server frame the way the shell does and map it to a domain
288    /// event, past the handshake.
289    fn decode(frame: serde_json::Value) -> RealtimeEvent {
290        let mut session = ClientSession::new();
291        session.connected();
292        session.handle(serde_json::from_value(welcome_value()).unwrap());
293        let mut events = session
294            .handle(serde_json::from_value(frame).unwrap())
295            .events;
296        assert_eq!(events.len(), 1, "one frame, one event");
297        events.remove(0).into()
298    }
299
300    fn welcome_value() -> serde_json::Value {
301        serde_json::json!({
302            "type": "welcome",
303            "protocol_version": PROTOCOL_VERSION,
304            "voice_states": []
305        })
306    }
