14/// A decoded server-side realtime event. The reducers fold these into
15/// view-models.
16///
17/// One-to-one with [`hound_realtime::Event`] plus [`RealtimeEvent::AuthExpired`]
18/// (no protocol frame can produce it — it is the shell's verdict that the
19/// session credential is dead) and [`RealtimeEvent::VersionMismatch`] (split out
20/// of `Error` because it, alone among the error codes, ends the plane). Kept as
21/// its own enum rather than a re-export exactly so those variants exist and so
22/// the app layer's `match` arms don't have to name a protocol type.
23#[derive(Clone, Debug, PartialEq)]
24pub enum RealtimeEvent {
25    /// The server joined this connection to a space's room; its events are
26    /// arriving from here on.
27    Subscribed { space_id: Uuid },
28    /// A new message in a subscribed space.
29    MessageCreated(MessageEvent),
30    /// An edit to an existing message.
31    MessageUpdated(MessageEvent),
32    /// A message's link previews are ready — the complete replacement list.
33    ///
34    /// Distinct from [`RealtimeEvent::MessageUpdated`] on purpose: `edited_at`
35    /// is what marks a message "(edited)", and a preview appearing is not an
36    /// edit. Folding this into the update event would stamp that marker on
37    /// every message that happens to contain a link.
38    MessageEmbeds {
39        message_id: Uuid,
40        space_id: Uuid,
41        embeds: Vec<hound_types::LinkEmbed>,
42    },
43    /// A new space in a joined community.
44    SpaceCreated(SpaceEvent),
45    /// A space was renamed or had its privacy toggled — the full summary.
46    SpaceUpdated(SpaceEvent),
47    /// A space was deleted.
48    SpaceRemoved { space_id: Uuid },
49    /// The community's profile changed — the full summary.
50    CommunityUpdated(hound_realtime::CommunityPayload),
51    /// The community was deleted.
52    CommunityRemoved { community_id: Uuid },
53    /// Someone joined the community.
54    MemberJoined {
55        community_id: Uuid,
56        member: hound_realtime::MemberPayload,
57    },
58    /// Someone left or was kicked — possibly this user.
59    MemberRemoved { community_id: Uuid, user_id: Uuid },
60    /// Roles changed; refetch roles and members for the community.
61    RolesChanged { community_id: Uuid },
62    /// Somebody was banned or unbanned; refetch the community's ban list.
63    ///
64    /// A ban also removes a member, so the server sends `member_removed`
65    /// alongside this — the two lists are different pages and each refreshes on
66    /// its own signal.
67    BansChanged { community_id: Uuid },
68    /// A user joined a voice channel (SFU peer id known).
69    VoiceJoined(VoiceJoinEvent),
70    /// A user left a voice channel.
71    VoiceLeft(VoiceLeaveEvent),
72    /// A participant toggled screenshare.
73    VoiceScreenShare(VoiceScreenShareEvent),
74    /// The full voice roster snapshot, pushed with every `Welcome` and on
75    /// server-side resync.
76    VoiceState(Vec<VoiceJoinEvent>),
77    /// The server refused something (an unauthorized subscribe, an unavailable
78    /// one). Surfaced, never swallowed, so contract drift is visible in the
79    /// logs.
80    Error { code: ErrorCode, message: String },
81    /// This build and the server cannot agree a protocol version. The plane will
82    /// NOT reconnect — retrying re-runs the same refused handshake — but the
83    /// SESSION is fine: this is never a reason to sign anyone out.
84    VersionMismatch { message: String },
85    /// The handshake completed: the session is live and its subscriptions are
86    /// re-issued.
87    ///
88    /// `reconnect` distinguishes the first connect of a plane from every later
89    /// one. Only the latter has a gap behind it: subscriptions resync, but the
90    /// message events fanned out while this plane had no connection are gone for
91    /// good, so the app layer repairs them with a history re-fetch (see
92    /// `HoundClient::arm_resync`). Never set on the first connect, where the
93    /// caller's own `open_space` load is the history.
94    Connected { reconnect: bool },
95    /// The transport went away. The supervisor is already reconnecting; desired
96    /// subscriptions survive and are re-issued on the next `Welcome`.
97    Disconnected,
98    /// The session credential is dead — the plane will NOT reconnect. The app
99    /// layer reacts as it does to any dead session (sign out).
100    AuthExpired,
101}
102
103impl From<Event> for RealtimeEvent {
104    fn from(event: Event) -> Self {
105        match event {
106            Event::Connected { reconnect } => Self::Connected { reconnect },
107            Event::Disconnected => Self::Disconnected,
108            Event::Subscribed { space_id } => Self::Subscribed { space_id },
109            Event::MessageCreated(m) => Self::MessageCreated(m),
110            Event::MessageUpdated(m) => Self::MessageUpdated(m),
111            Event::MessageEmbeds {
112                message_id,
113                space_id,
114                embeds,
115            } => Self::MessageEmbeds {
116                message_id,
117                space_id,
118                embeds,
119            },
120            Event::SpaceCreated(s) => Self::SpaceCreated(s),
121            Event::SpaceUpdated(s) => Self::SpaceUpdated(s),
122            Event::SpaceRemoved { space_id } => Self::SpaceRemoved { space_id },
123            Event::CommunityUpdated(c) => Self::CommunityUpdated(c),
124            Event::CommunityRemoved { community_id } => Self::CommunityRemoved { community_id },
125            Event::MemberJoined {
126                community_id,
127                member,
128            } => Self::MemberJoined {
129                community_id,
130                member,
131            },
132            Event::MemberRemoved {
133                community_id,
134                user_id,
135            } => Self::MemberRemoved {
136                community_id,
137                user_id,
138            },
139            Event::RolesChanged { community_id } => Self::RolesChanged { community_id },
140            Event::BansChanged { community_id } => Self::BansChanged { community_id },
141            Event::VoiceJoined(v) => Self::VoiceJoined(v),
142            Event::VoiceLeft(v) => Self::VoiceLeft(v),
143            Event::VoiceState(v) => Self::VoiceState(v),
144            Event::Screenshare(s) => Self::VoiceScreenShare(s),
145            // The one error code that ends the plane gets its own variant, so no
146            // caller has to know which codes are terminal.
147            Event::Error {
148                code: ErrorCode::VersionMismatch,
149                message,
150            } => Self::VersionMismatch { message },
151            Event::Error { code, message } => Self::Error { code, message },
152        }
153    }
154}

156/// What the public API asks the supervisor to do. The supervisor owns the
157/// [`ClientSession`](hound_realtime::ClientSession), so every caller-side intent
158/// crosses this channel rather than touching shared state.
159///
160/// Not `PartialEq` (and not `Clone`): a subscribe carries the caller's
161/// one-shot answer channel, which is a capability, not a value.
162#[derive(Debug)]
163pub(super) enum Command {
164    Subscribe {
165        space_id: Uuid,
166        /// Signalled once the server has confirmed the room join. Dropping it
167        /// (an unsubscribe, or a supervisor that stopped) releases the caller
168        /// immediately rather than making it wait out its deadline.
169        ack: Option<oneshot::Sender<()>>,
170    },
171    Unsubscribe(Uuid),
172    Screenshare {
173        space_id: Uuid,
174        enabled: bool,
175    },
176    /// Say goodbye and end the supervisor.
177    Shutdown,
178}
179
180/// Callers waiting for a space's subscribe confirmation. Owned by
181/// [`supervise`](super::supervisor::supervise), not by one session: a subscribe
182/// taken while the plane is down (or one whose connection dies before the
183/// answer) is answered by the NEXT session's confirmation, since the reducer
184/// re-issues the whole desired set on every `Welcome`.
185pub(super) type Pending = HashMap<Uuid, Vec<oneshot::Sender<()>>>;
186
187/// Release everyone waiting on `space_id`.
188pub(super) fn confirm(pending: &mut Pending, space_id: Uuid) {
189    for ack in pending.remove(&space_id).unwrap_or_default() {
190        let _ = ack.send(());
191    }
192}
193
194/// Release everyone waiting on anything — for a server that does not confirm
195/// subscribes at all, where waiting is waiting forever.
196pub(super) fn confirm_all(pending: &mut Pending) {
197    for (_, acks) in pending.drain() {
198        for ack in acks {
199            let _ = ack.send(());
200        }
201    }
202}

215    /// A `message_created` frame decodes into the struct the message reducer
216    /// renders from, attachments and all.
217    #[test]
218    fn a_message_frame_maps_to_the_message_event() {
219        let frame = serde_json::json!({
220            "type": "message_created",
221            "message": {
222                "id": id(10),
223                "space_id": id(2),
224                "user_id": id(3),
225                "contents": "hi",
226                "edited_at": null,
227                "profile": {
228                    "user_id": id(3), "global_profile_id": id(4),
229                    "community_profile_id": null, "name": "Aoife", "picture": "p"
230                },
231                "attachments": [{
232                    "id": id(20), "url": "https://cdn/x.png",
233                    "content_type": "image/png", "size_bytes": 12
234                }]
235            }
236        });
237        let event: RealtimeEvent = decode(frame);
238        let RealtimeEvent::MessageCreated(message) = event else {
239            panic!("expected MessageCreated");
240        };
241        assert_eq!(message.contents, "hi");
242        assert_eq!(message.attachments.len(), 1);
243        assert_eq!(message.attachments[0].size_bytes, 12);
244    }

246    /// The `Welcome`'s roster snapshot decodes into the voice events the roster

258        let mut session = ClientSession::new();
259        session.connected();
260        let events: Vec<RealtimeEvent> = session

328    /// Decode one server frame the way the shell does and map it to a domain
329    /// event, past the handshake.
330    fn decode(frame: serde_json::Value) -> RealtimeEvent {
331        let mut session = ClientSession::new();
332        session.connected();
333        session.handle(serde_json::from_value(welcome_value()).unwrap());
334        let mut events = session
335            .handle(serde_json::from_value(frame).unwrap())
336            .events;
337        assert_eq!(events.len(), 1, "one frame, one event");
338        events.remove(0).into()
339    }
340
341    fn welcome_value() -> serde_json::Value {
342        serde_json::json!({
343            "type": "welcome",
344            "protocol_version": PROTOCOL_VERSION,
345            "voice_states": []
346        })
347    }
