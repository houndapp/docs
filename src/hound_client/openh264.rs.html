1//! # Cisco's openh264 binary: fetch, verify, install
2//!
3//! The software H.264 codec is not compiled into a shipped build. Cisco's H.264
4//! patent grant covers **their** binary module, and only when it is *downloaded
5//! at the time the product is installed on the user's device* — a codec we
6//! compiled ourselves, or one we bundled into the release artifact, carries no
7//! grant at all. So the app fetches the blob itself, on first run, straight from
8//! Cisco, and hands it to the engine ([`hound_engine::api::Engine::set_openh264_blob`]).
9//!
10//! ## Where it fits in startup
11//!
12//! [`spawn_install`] is fire-and-forget from `init_app`: the download must never
13//! be on the path to the first frame (the Dart bootstrap has a 30 s wedge
14//! watchdog, and a user on a slow link would otherwise stare at a black window
15//! while ~500 KB trickles in). Until it lands there is simply no software H.264 —
16//! `encode_caps`/`decode_caps` report `h264: false`, so peers are told the truth
17//! and never sent a stream this client cannot decode. Hardware H.264
18//! (VideoToolbox, Media Foundation, VAAPI) is unaffected and needs none of this.
19//!
20//! The one place that *does* wait is a capability probe taken moments after
21//! startup — see [`settle`], which bounds the "joined a call before the codec
22//! arrived" race.
23//!
24//! ## Trust
25//!
26//! Three independent checks, none of them skippable:
27//! 1. HTTPS to `ciscobinary.openh264.org` (rustls, webpki roots — the workspace
28//!    TLS stack).
29//! 2. SHA-256 of the *decompressed* library against [`BLOBS`] below, which is
30//!    transcribed from openh264-sys2's own `hashes.txt` — so a corrupted transfer
31//!    or a substituted file never reaches the cache.
32//! 3. The engine re-verifies at install time, because the cache is a file on disk
33//!    that anything could have rewritten between runs.
34//!
35//! ## Cache
36//!
37//! `<app-support>/openh264/<version>/<cisco file name>`. Versioned by directory
38//! so a future openh264 release is a new download rather than an in-place
39//! overwrite, and so [`remove`] can drop the codec entirely (the settings toggle:
40//! openh264's BINARY_LICENSE requires the user to be able to turn it off).
41

52/// Set from the moment a first-run download is scheduled until it finishes, so
53/// [`settle`] knows there is something worth waiting for.
54static DOWNLOADING: AtomicBool = AtomicBool::new(false);
55
56/// Woken when a download finishes, however it finishes.
57static SETTLED: OnceLock<tokio::sync::Notify> = OnceLock::new();
58
59fn settled() -> &'static tokio::sync::Notify {
60    SETTLED.get_or_init(tokio::sync::Notify::new)
61}

63/// How long a capability probe will wait for an in-flight first-run download.
64///
65/// The race this closes: the app starts, the download takes a second or two, and
66/// a call joined inside that window advertises `h264: false` for its WHOLE
67/// lifetime (caps are snapshotted at `Hello`, and there is no mid-session
68/// re-announce). A user who launches the app and immediately joins would get no
69/// video with H.264-only peers until they rejoined. Three seconds is well past a
70/// ~500 KB fetch on any usable link and well inside what "joining a call" already
71/// costs, and it is a ceiling, not a delay — [`settle`] returns instantly once
72/// the download has landed, and on every later call.
73pub const SETTLE_CEILING: Duration = Duration::from_secs(3);
74
75/// Wait for an in-flight first-run download, up to [`SETTLE_CEILING`], so a
76/// capability probe taken moments after startup reports the codec the user is
77/// about to have rather than the one they had a second ago. A no-op once the
78/// download has settled, and when there was never one to wait for.
79///
80/// **Remaining limitation:** if the download is still going after the ceiling —
81/// or fails and succeeds on a later launch — the caps advertised at `Hello` stay
82/// `h264: false` for that session. Re-announcing capabilities mid-call is not
83/// implemented; rejoining picks up the codec.
84pub async fn settle() {
85    if !DOWNLOADING.load(Ordering::Acquire) {
86        return;
87    }
88    // Register for the wake BEFORE re-reading the flag: the download could finish
89    // between the check above and here, and `notify_waiters` only wakes waiters
90    // that already exist.
91    let waker = settled().notified();
92    if !DOWNLOADING.load(Ordering::Acquire) {
93        return;
94    }
95    let _ = tokio::time::timeout(SETTLE_CEILING, waker).await;
96}

98/// Schedule the first-run download, marking it in flight **synchronously** so a
99/// capability probe on the very next line already knows to wait for it (spawning
100/// and then marking would leave exactly the gap [`settle`] exists to close).
101pub fn spawn_install(app_dir: PathBuf) {
102    allow();
103    DOWNLOADING.store(true, Ordering::Release);
104    tokio::spawn(async move {
105        // Non-fatal by construction: a failure leaves the app running on its
106        // hardware codecs, the capability probes keep reporting no software
107        // H.264, and the next launch tries again.
108        if let Err(e) = install(app_dir).await {
109            tracing::warn!("openh264: software H.264 is unavailable: {e}");
110        }
111        DOWNLOADING.store(false, Ordering::Release);
112        settled().notify_waiters();
113    });
114}

207/// The published library for the platform we are running on, if Cisco has one.
208fn host_blob() -> Option<&'static Blob> {
209    BLOBS
210        .iter()
211        .find(|(os, arch, _)| *os == std::env::consts::OS && *arch == std::env::consts::ARCH)
212        .map(|(_, _, blob)| blob)
213}

215/// Where this platform's library is cached under `app_dir`, whether or not it has
216/// been downloaded yet.
217fn cache_path(app_dir: &Path, blob: &Blob) -> PathBuf {
218    app_dir.join("openh264").join(VERSION).join(blob.file)
219}
220
221/// Make Cisco's software H.264 codec available to the engine, downloading it into
222/// `app_dir` if it is not already cached. Returns the installed path.
223///
224/// **Call this off the startup path** — it does network IO. Re-running it is
225/// cheap and safe: a cached, valid library installs without touching the network,
226/// and installing twice is a no-op in the engine.
227///
228/// Failure is not fatal to anything: the app keeps running with whatever hardware
229/// codecs it has, the capability probes keep reporting no software H.264, and the
230/// next launch tries again.
231pub async fn install(app_dir: PathBuf) -> Result<PathBuf> {
232    let Some(blob) = host_blob() else {
233        return Err(HoundError::media(
234            MediaCode::Openh264Unsupported,
235            format!(
236                "Cisco publishes no openh264 build for {}-{}",
237                std::env::consts::OS,
238                std::env::consts::ARCH
239            ),
240        ));
241    };
242    let path = cache_path(&app_dir, blob);
243
244    // Already downloaded? Install straight from the cache — the engine re-verifies
245    // the SHA, so a file that rotted on disk is caught here and re-fetched rather
246    // than trusted.
247    if path.is_file() {
248        match install_at(path.clone()).await {
249            Ok(()) => {
250                tracing::info!(target: "openh264", path = %path.display(), "openh264: installed from cache");
251                return Ok(path);
252            }
253            Err(e) => {
254                tracing::warn!(
255                    target: "openh264",
256                    "openh264: cached library at {} is unusable ({e}); re-downloading",
257                    path.display()
258                );
259                let _ = std::fs::remove_file(&path);
260            }
261        }
262    }
263
264    let url = format!("{BASE_URL}/{}.bz2", blob.file);
265    tracing::info!(target: "openh264", %url, "openh264: downloading Cisco's codec");
266    // Bounded like the REST plane's client: a CDN that accepts the connection and
267    // then goes quiet must not leave this task (and its buffer) alive forever.
268    // ~500 KB over any usable link fits well inside these.
269    let http = reqwest::Client::builder()
270        .connect_timeout(std::time::Duration::from_secs(10))
271        .read_timeout(std::time::Duration::from_secs(30))
272        .build()
273        .map_err(|e| {
274            HoundError::media(
275                MediaCode::Openh264Download,
276                format!("building the openh264 download client: {e}"),
277            )
278        })?;
279    let compressed = http
280        .get(&url)
281        .send()
282        .await
283        .and_then(reqwest::Response::error_for_status)
284        .map_err(|e| {
285            HoundError::media(
286                MediaCode::Openh264Download,
287                format!("downloading {url}: {e}"),
288            )
289        })?
290        .bytes()
291        .await
292        .map_err(|e| {
293            HoundError::media(MediaCode::Openh264Download, format!("reading {url}: {e}"))
294        })?;
295
296    // Decompress + hash + write: all CPU/disk, none of it async. `spawn_blocking`
297    // keeps ~1.2 MB of bzip2 off the runtime's worker threads.
298    let expected = blob.sha256;
299    let target = path.clone();
300    let written = tokio::task::spawn_blocking(move || -> Result<PathBuf> {
301        let library = decompress(&compressed)?;
302        verify(&library, expected)?;
303        // Last responsible moment: the user can flip the setting off while this
304        // download is in flight, and `remove` promises the Cisco binary is then
305        // neither running nor present. Checking here means a toggle-off that
306        // lands before the write leaves nothing behind at all.
307        //
308        // A window remains — a toggle-off between this check and the rename
309        // writes the file — so `remove` deletes the cache rather than assuming
310        // it is absent, and re-running it (or the next launch's disabled path)
311        // clears anything this raced past. The engine gate is authoritative
312        // either way: a blob on disk that the user disabled is never loaded.
313        if !allowed() {
314            return Err(HoundError::media(
315                MediaCode::Openh264Install,
316                "openh264: the Cisco codec was turned off while it was downloading",
317            ));
318        }
319        write_atomically(&target, &library)?;
320        Ok(target)
321    })
322    .await
323    .map_err(|e| {
324        HoundError::media(
325            MediaCode::Openh264Install,
326            format!("openh264 install task failed: {e}"),
327        )
328    })??;
329
330    install_at(written.clone()).await?;
331    tracing::info!(target: "openh264", path = %written.display(), "openh264: downloaded and installed");
332    Ok(written)
333}

335/// Hand a library on disk to the engine (which re-verifies it before loading).
336/// The engine call is synchronous and does file IO + `dlopen`, so it runs on the
337/// blocking pool.
338async fn install_at(path: PathBuf) -> Result<()> {
339    tokio::task::spawn_blocking(move || {
340        hound_engine::api::Engine::new()
341            .set_openh264_blob(path)
342            .map_err(|e| {
343                HoundError::media(
344                    MediaCode::Openh264Install,
345                    format!("installing the openh264 codec: {e}"),
346                )
347            })
348    })
349    .await
350    .map_err(|e| {
351        HoundError::media(
352            MediaCode::Openh264Install,
353            format!("openh264 install task failed: {e}"),
354        )
355    })?
356}

358/// Delete the cached library — the "off" half of the settings toggle. Also tells
359/// the engine to stop using an already-installed one, because a switch the user
360/// can flip has to take effect in this session, not the next.
361///
362/// Deleting a file that is not there is success: the point is that no Cisco codec
363/// remains, not that one was removed.
364pub fn remove(app_dir: &Path) -> Result<()> {
365    hound_engine::api::Engine::new().set_openh264_enabled(false);
366    let Some(blob) = host_blob() else {
367        return Ok(());
368    };
369    let path = cache_path(app_dir, blob);
370    match std::fs::remove_file(&path) {
371        Ok(()) => {
372            tracing::info!(target: "openh264", path = %path.display(), "openh264: removed the cached codec");
373            Ok(())
374        }
375        Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(()),
376        Err(e) => Err(HoundError::media(
377            MediaCode::Openh264Install,
378            format!("removing {}: {e}", path.display()),
379        )),
380    }
381}

383/// Re-permit the Cisco codec (the "on" half of the toggle). Pair with [`install`]
384/// to actually get one back after a [`remove`].
385pub fn allow() {
386    hound_engine::api::Engine::new().set_openh264_enabled(true);
387}

389/// Whether the Cisco codec is currently permitted — the state the settings
390/// screen renders.
391pub fn allowed() -> bool {
392    hound_engine::api::Engine::new().openh264_enabled()
393}

395/// bzip2 → the raw library bytes. Cisco serves one bzip2 stream per release
396/// (a single compressed file, not a tar), so this is a straight decode.
397fn decompress(compressed: &[u8]) -> Result<Vec<u8>> {
398    let mut out = Vec::new();
399    bzip2::read::BzDecoder::new(compressed)
400        .read_to_end(&mut out)
401        .map_err(|e| {
402            HoundError::media(
403                MediaCode::Openh264Download,
404                format!("decompressing the openh264 download: {e}"),
405            )
406        })?;
407    Ok(out)
408}
409
410/// SHA-256 the decompressed library against the published digest. This is the
411/// check that makes it safe to write the file to disk at all.
412fn verify(library: &[u8], expected: &str) -> Result<()> {
413    let digest = sha2::Sha256::digest(library);
414    let actual = digest.iter().fold(String::new(), |mut acc, byte| {
415        use std::fmt::Write;
416        let _ = write!(acc, "{byte:02x}");
417        acc
418    });
419    if actual != expected {
420        return Err(HoundError::media(
421            MediaCode::Openh264Verify,
422            format!("openh264 SHA-256 mismatch: expected {expected}, got {actual}"),
423        ));
424    }
425    Ok(())
426}

428/// Write via a sibling temp file + rename, so a crash or a second instance mid-
429/// write can never leave a half-written library that the next launch would load.
430fn write_atomically(path: &Path, bytes: &[u8]) -> Result<()> {
431    let dir = path.parent().ok_or_else(|| {
432        HoundError::media(
433            MediaCode::Openh264Install,
434            format!("{} has no parent directory", path.display()),
435        )
436    })?;
437    std::fs::create_dir_all(dir).map_err(|e| {
438        HoundError::media(
439            MediaCode::Openh264Install,
440            format!("creating {}: {e}", dir.display()),
441        )
442    })?;
443    // Same directory, so the rename is atomic (a temp dir could be another
444    // filesystem, where rename degrades to copy).
445    let temp = tempfile::NamedTempFile::new_in(dir).map_err(|e| {
446        HoundError::media(
447            MediaCode::Openh264Install,
448            format!("creating a temp file in {}: {e}", dir.display()),
449        )
450    })?;
451    std::fs::write(temp.path(), bytes).map_err(|e| {
452        HoundError::media(
453            MediaCode::Openh264Install,
454            format!("writing {}: {e}", temp.path().display()),
455        )
456    })?;
457    temp.persist(path).map_err(|e| {
458        HoundError::media(
459            MediaCode::Openh264Install,
460            format!("publishing {}: {e}", path.display()),
461        )
462    })?;
463    Ok(())
464}

467mod tests {

470    /// Every entry names a genuine Cisco release: the digests must be lowercase
471    /// hex SHA-256, and the file names must be the ones Cisco serves (which is
472    /// also what openh264-sys2's allowlist checks). A typo here would only show
473    /// up as a failed download on one platform, months later.
474    ///
475    /// The version-in-filename assertion is a PARTIAL guard on a duplication this
476    /// test cannot close: the 2.6.0 filenames are also written out in
477    /// `.mise.toml` (the `http:openh264` tool URLs and the `HOUND_OPENH264_PATH`
478    /// template) and in `.github/scripts/openh264-env.sh`. Bumping the release
479    /// means changing [`VERSION`], this table's names and digests, and those two
480    /// files together — this only proves THIS table is self-consistent.
481    #[test]
482    fn blob_table_is_well_formed() {
483        for (os, arch, blob) in BLOBS {
484            assert_eq!(
485                blob.sha256.len(),
486                64,
487                "{os}-{arch}: SHA-256 must be 64 hex chars"
488            );
489            assert!(
490                blob.sha256
491                    .chars()
492                    .all(|c| c.is_ascii_hexdigit() && !c.is_ascii_uppercase()),
493                "{os}-{arch}: SHA-256 must be lowercase hex"
494            );
495            assert!(
496                blob.file.contains(VERSION),
497                "{os}-{arch}: file name must name version {VERSION}"
498            );
499        }
500    }
501
502    /// No duplicate platform rows — two entries for one target would make which
503    /// library we fetch depend on table order.
504    #[test]
505    fn blob_table_has_one_entry_per_platform() {
506        let mut seen: Vec<(&str, &str)> = Vec::new();
507        for (os, arch, _) in BLOBS {
508            assert!(
509                !seen.contains(&(os, arch)),
510                "duplicate openh264 blob entry for {os}-{arch}"
511            );
512            seen.push((os, arch));
513        }
514    }
515
516    /// The cache is versioned by directory, so a later openh264 release lands
517    /// beside this one instead of overwriting a library that may already be
518    /// mapped into the running process.
519    #[test]
520    fn cache_path_is_versioned() {
521        let blob = &BLOBS[0].2;
522        let path = cache_path(Path::new("/app"), blob);
523        assert_eq!(
524            path,
525            Path::new("/app")
526                .join("openh264")
527                .join(VERSION)
528                .join(blob.file)
529        );
530    }
531
532    /// The digest check rejects a body that is not the published library — the
533    /// one thing standing between a corrupted or substituted download and a
534    /// `dlopen`.
535    #[test]
536    fn verify_rejects_the_wrong_bytes() {
537        let expected = BLOBS[0].2.sha256;
538        assert!(verify(b"not openh264", expected).is_err());
539        // ...and accepts the real digest of the bytes it is given.
540        let bytes = b"not openh264";
541        let actual = sha2::Sha256::digest(bytes)
542            .iter()
543            .fold(String::new(), |mut acc, b| {
544                use std::fmt::Write;
545                let _ = write!(acc, "{b:02x}");
546                acc
547            });
548        assert!(verify(bytes, &actual).is_ok());
549    }
550
551    /// A truncated / non-bzip2 body is a typed error, never a panic — this runs
552    /// on whatever bytes the network hands back.
553    #[test]
554    fn decompress_rejects_garbage() {
555        assert!(decompress(b"BZh9\xde\xad\xbe\xef").is_err());
556        assert!(decompress(b"").is_err());
557    }
