1//! # Agent-readable audio-quality debug report
2//!
3//! A pure function — [`audio_report`] — that turns one [`StatsSnapshot`] into a
4//! compact, deterministic, paste-able text block that **pre-attributes** an audio
5//! glitch to a pipeline stage. The point is that a reader (a human or an agent) sees
6//! a *cause*, not a wall of counters: a `diagnosis:` headline names the most likely
7//! fault, a per-stage table flags each signal `ok` / `HIGH` / `BAD`, and a glitch log
8//! lists the recent discrete events with timing.
9//!
10//! ## The pipeline stages (any can cause a stutter)
11//! ```text
12//! sender ─▶ network ─▶ receiver jitter buffer ─▶ mix tick (RT) ─▶ device ring ─▶ DAC
13//! ```
14//! The receiver can directly observe everything from *network* rightward; the report
15//! is built entirely from signals the [`crate::stats`] snapshot already carries
16//! (inter-arrival jitter, loss/reorder, jitter-buffer depth vs the recommended depth,
17//! drop counts, per-peer underflow, the mix-tick lateness percentiles, output-ring
18//! underruns) plus the discrete glitch log.
19//!
20//! ## Why a text format (not just the JSON)
21//! The raw snapshot is for the live panel; this is for *diagnosis*. It is stable,
22//! self-describing, and small enough to paste into a chat with an agent — and the
23//! [`diagnose`] rules collapse the six-stage signal space to a single most-likely
24//! cause so the reader doesn't re-derive it every time.
25//!
26//! Pure + deterministic ⇒ golden-tested per fault class (shallow buffer / RT
27//! descheduled / loss / healthy), which is what keeps the agent-facing format honest.
28

35// ── Verdict thresholds (named so they are tunable + asserted in tests) ─────────────
36
37/// Smoothed inter-arrival jitter (ms) above which the link is "jittery".
38const JITTER_HIGH_MS: f64 = 30.0;
39/// `recommended_delay > SHALLOW_RATIO × target_delay` ⇒ the buffer can't grow fast
40/// enough for the link's jitter (too shallow → underflows).
41const SHALLOW_RATIO: f64 = 1.3;
42/// Actual occupancy below `DRY_RATIO × target_delay` ⇒ the playout cushion is gone:
43/// the buffer holds far less than the depth it is configured for.
44const DRY_RATIO: f64 = 0.5;
45/// Mix-tick p95 lateness (ms) above which the RT feeder is being descheduled.
46///
47/// `pub(crate)` because the receive-side decode budget acts on the same threshold
48/// (`call::policy::DECODE_TICK_LATE_LIMIT_US`): what this report diagnoses as a
49/// tick slip is exactly what makes video shed decoders to get the mix thread its
50/// core back, and two independently-tuned numbers for one verdict would drift.
51pub(crate) const TICK_LATE_HIGH_MS: f64 = 3.0;
52/// Receive loss fraction (percent) above which the link is lossy.
53const LOSS_HIGH_PCT: f64 = 2.0;
54/// How many recent glitch-log lines to print (newest first).
55const GLITCH_LOG_LINES: usize = 16;
56
57/// A per-row verdict in the stage table.
58#[derive(Clone, Copy, PartialEq, Eq)]
59enum Verdict {
60    Ok,
61    /// Notable but not yet the likely culprit (e.g. elevated jitter).
62    High,
63    /// A real fault contributing to the audible glitch.
64    Bad,
65}
66
67impl Verdict {
68    fn tag(self) -> &'static str {
69        match self {
70            Verdict::Ok => "ok",
71            Verdict::High => "HIGH",
72            Verdict::Bad => "BAD",
73        }
74    }
75}

77/// Receive loss fraction (percent) for one inbound audio stream over the window:
78/// `lost / (received + lost)`.
79fn loss_pct(a: &InboundAudioSnapshot) -> f64 {
80    let denom = (a.frames_received + a.lost) as f64;
81    if denom <= 0.0 {
82        0.0
83    } else {
84        a.lost as f64 / denom * 100.0
85    }
86}

88/// True if this stream's jitter buffer is too shallow for the link (recommendation
89/// meaningfully above the current depth, and the link is actually jittery).
90fn jitterbuf_shallow(a: &InboundAudioSnapshot) -> bool {
91    a.target_delay_us > 0
92        && (a.recommended_delay_us as f64) > SHALLOW_RATIO * (a.target_delay_us as f64)
93        && (a.jitter_us as f64 / 1000.0) >= JITTER_HIGH_MS
94}

96/// Whether a stream is carrying audio this window (worth a table block).
97fn audio_active(a: &InboundAudioSnapshot) -> bool {
98    a.frames_received > 0 || a.underflow > 0 || a.lost > 0
99}

101/// Output-device reopens still in the retained glitch window (~12 s). Each one is a
102/// BLOCKING device open on the real-time mix thread followed by a fresh silence
103/// prefill, so it manufactures exactly the tick-slip + output-underrun signature the
104/// `rt-mix-descheduled` rule reads as CPU contention — which is why it is counted and
105/// diagnosed ahead of it.
106fn device_reopens(snap: &StatsSnapshot) -> usize {
107    snap.glitches
108        .iter()
109        .filter(|g| g.kind == GLITCH_KIND_DEVICE_REOPEN)
110        .count()
111}

113/// A peer's two inbound audio streams (mic + screen-share app audio), labelled, for
114/// the diagnosis scan.
115fn streams(s: &InboundSnapshot) -> [(&'static str, &InboundAudioSnapshot); 2] {
116    [("mic", &s.audio), ("screen_audio", &s.screen_audio)]
117}

119// ── Diagnosis: signals → one most-likely cause ─────────────────────────────────────
120
121/// Approx playout backlog (ms) actually queued in a stream's jitter buffer, assuming
122/// 20 ms Opus frames. This is the ACTUAL occupancy (vs the configured `target_delay`):
123/// a value far above the target means playout is running behind — a burst/skew backlog
124/// the buffer is catching up from.
125fn buffered_ms(a: &InboundAudioSnapshot) -> f64 {
126    a.playout_buffered_frames as f64 * 20.0
127}
128
129/// True if the actual buffered backlog is far above the configured depth — playout is
130/// behind (a burst/skew backlog, the cause of overflow drops).
131fn backlogged(a: &InboundAudioSnapshot) -> bool {
132    let target_ms = a.target_delay_us as f64 / 1000.0;
133    buffered_ms(a) > (2.0 * target_ms).max(300.0)
134}
135
136/// True if the playout cushion has been eroded away: the stream underflows while the
137/// buffer holds far LESS than the depth it is configured for, i.e. frames are being
138/// released at (or after) arrival with nothing in reserve.
139///
140/// This is the shallow case [`jitterbuf_shallow`] structurally cannot see. That rule
141/// compares the recommended depth against the current one, but the recommendation comes
142/// from a *differential* inter-arrival estimator: a one-way-delay STEP (path change,
143/// bufferbloat that only partly drains) shifts every arrival past its playout deadline
144/// without changing the measured jitter at all, so `recommended == target` while the
145/// buffer runs permanently dry. Only the ACTUAL occupancy shows it — the exact
146/// direction `backlogged` covers on the other side.
147fn cushion_lost(a: &InboundAudioSnapshot) -> bool {
148    a.underflow > 0
149        && a.frames_received > 0
150        && a.target_delay_us > 0
151        && buffered_ms(a) < DRY_RATIO * (a.target_delay_us as f64 / 1000.0)
152}

154/// One diagnosis: the pipeline stage at fault, and the headline naming it.
155///
156/// Two forms because they have different jobs. [`headline`](Self::headline) is for a
157/// human/agent reading the report — it embeds the live counters that justify the
158/// verdict, which makes every occurrence a unique string. [`stage`](Self::stage) is
159/// the same verdict as a stable token, so it can be a Sentry tag or a metric label
160/// without exploding cardinality.
161#[derive(Debug, Clone, PartialEq, Eq)]
162pub struct Diagnosis {
163    /// Stable, low-cardinality stage token — one of the values listed on
164    /// [`diagnose_detail`]. Safe to group by.
165    pub stage: &'static str,
166    /// The full sentence, counters and all.
167    pub headline: String,
168}
169
170impl Diagnosis {
171    fn new(stage: &'static str, headline: String) -> Self {
172        Self { stage, headline }
173    }
174}
175
176/// The single most-likely cause. Rules are ordered by specificity/severity; the first
177/// that matches wins. This is the heart of the agent-readable report — it localizes
178/// the fault so the reader acts instead of re-deriving.
179///
180/// Stages: `device-reopen`, `rt-mix-descheduled`, `capture-overflow`,
181/// `jitterbuf-shallow`, `playout-cushion-lost`, `jitterbuf-overflow`, `network-loss`,
182/// `device-underrun`, `output-ring-overrun`, `intermittent-underflow`, `healthy`.
183pub fn diagnose_detail(snap: &StatsSnapshot) -> Diagnosis {
184    let t = &snap.transport;
185    let tick_slip = t.tick_late_p95_ms >= TICK_LATE_HIGH_MS;
186
187    // 0) The output device was re-opened (route change / device loss). This OUTRANKS the
188    //    RT-descheduling rule because it CAUSES its signature: the open blocks the mix
189    //    thread inside the platform audio API (up to ~1 s on CoreAudio) and the new sink
190    //    starts from an empty ring, so a tick slip and a burst of output underruns are
191    //    the expected consequence, not evidence of CPU/encode contention.
192    let reopens = device_reopens(snap);
193    if reopens > 0 {
194        return Diagnosis::new(
195            "device-reopen",
196            format!(
197                "output device re-opened {reopens}× in the last ~12s (route change or device \
198                 loss) — each open blocks the RT mix feeder and restarts the output cushion \
199                 from empty, so the tick slip (p95={:.1}ms) and {} output underruns around it \
200                 are the reopen, not CPU contention",
201                t.tick_late_p95_ms, t.playout_output_underruns
202            ),
203        );
204    }
205
206    // 1) RT feeder descheduled: the device ring drained because the mix thread slipped
207    //    — affects ALL audio at once. The most global, highest-priority cause.
208    if tick_slip && t.playout_output_underruns > 0 {
209        return Diagnosis::new(
210            "rt-mix-descheduled",
211            format!(
212                "RT mix thread descheduled (tick p95={:.1}ms, {} output underruns) — CPU/encode \
213                 contention starving the device feeder; affects all audio",
214                t.tick_late_p95_ms, t.playout_output_underruns
215            ),
216        );
217    }
218
219    // 1b) Mic capture ring overflowing: a SEND-side hole nothing on the receive side
220    //     can see, so it outranks the per-peer rules — no other signal explains it.
221    if snap.outbound_audio.capture_overflow_samples > 0 {
222        return Diagnosis::new(
223            "capture-overflow",
224            format!(
225                "mic capture ring overflow ({} samples dropped) — the capture worker can't \
226                 keep up with the device; the FAR end hears chopped words (every local \
227                 receive-side gauge reads healthy)",
228                snap.outbound_audio.capture_overflow_samples
229            ),
230        );
231    }
232
233    // 2) Jitter buffer too shallow for the link (underflowing + rec≫cur + jittery).
234    for p in &snap.inbound {
235        for (label, a) in streams(p) {
236            if a.underflow > 0 && jitterbuf_shallow(a) {
237                return Diagnosis::new(
238                    "jitterbuf-shallow",
239                    format!(
240                        "jitter buffer too shallow for the link (peer {} {label}): {} underflows, \
241                         jitter {:.0}ms > depth {:.0}ms (recommends {:.0}ms)",
242                        p.peer_id,
243                        a.underflow,
244                        a.jitter_us as f64 / 1000.0,
245                        a.target_delay_us as f64 / 1000.0,
246                        a.recommended_delay_us as f64 / 1000.0,
247                    ),
248                );
249            }
250        }
251    }
252
253    // 2b) Playout cushion eroded: underflowing while running dry. Same stage as (2),
254    //     the other failure mode — the depth is nominally right, the ANCHOR is not.
255    for p in &snap.inbound {
256        for (label, a) in streams(p) {
257            if cushion_lost(a) {
258                return Diagnosis::new(
259                    "playout-cushion-lost",
260                    format!(
261                        "playout cushion lost (peer {} {label}): {} underflows with only ~{:.0}ms \
262                         queued against a {:.0}ms depth — frames land past their playout deadline \
263                         (a sustained one-way-delay increase), so every one releases on arrival",
264                        p.peer_id,
265                        a.underflow,
266                        buffered_ms(a),
267                        a.target_delay_us as f64 / 1000.0,
268                    ),
269                );
270            }
271        }
272    }
273
274    // 3) Buffer overflowing: backed up / sender-faster clock skew.
275    for p in &snap.inbound {
276        for (label, a) in streams(p) {
277            if a.overflow_drops > 0 {
278                return Diagnosis::new(
279                    "jitterbuf-overflow",
280                    format!(
281                        "jitter buffer overflowing (peer {} {label}): {} overflow drops — backed \
282                         up or sender-faster clock skew",
283                        p.peer_id, a.overflow_drops
284                    ),
285                );
286            }
287        }
288    }
289
290    // 4) Network loss.
291    for p in &snap.inbound {
292        for (label, a) in streams(p) {
293            if loss_pct(a) >= LOSS_HIGH_PCT {
294                return Diagnosis::new(
295                    "network-loss",
296                    format!(
297                        "network packet loss (peer {} {label}): {:.1}% lost ({} frames, concealed \
298                         with PLC)",
299                        p.peer_id,
300                        loss_pct(a),
301                        a.lost
302                    ),
303                );
304            }
305        }
306    }
307
308    // 5) Device-side underruns without a tick slip: output device pacing / route.
309    if t.playout_output_underruns > 0 {
310        return Diagnosis::new(
311            "device-underrun",
312            format!(
313                "device-side output underruns ({}) without a tick slip — output device pacing / \
314                 route (e.g. Bluetooth HFP)",
315                t.playout_output_underruns
316            ),
317        );
318    }
319
320    // 5b) The opposite device-side drift: the bus outran the device and the sink threw
321    //     the excess away. Below the underrun rule — a dry ring is the louder fault —
322    //     but above the per-peer catch-all, since it is a whole-output condition.
323    if t.playout_output_overruns > 0 {
324        return Diagnosis::new(
325            "output-ring-overrun",
326            format!(
327                "output ring overrun ({} samples discarded) — the mix bus outran the device; \
328                 output latency pinned at the ring depth (the depth feedback lost control, \
329                 or the sink cannot report its fill)",
330                t.playout_output_overruns
331            ),
332        );
333    }
334
335    // 6) Per-peer mix underflow with no clear upstream cause: intermittent arrival.
336    for p in &snap.inbound {
337        for (label, a) in streams(p) {
338            if a.underflow > 0 {
339                return Diagnosis::new(
340                    "intermittent-underflow",
341                    format!(
342                        "intermittent per-peer underflow (peer {} {label}): {} mix starves — \
343                         bursty arrival or a brief stall",
344                        p.peer_id, a.underflow
345                    ),
346                );
347            }
348        }
349    }
350
351    Diagnosis::new(
352        "healthy",
353        "healthy — no audio glitches in this window".to_string(),
354    )
355}

364/// Render the agent-readable audio-quality report for one snapshot. Pure +
365/// deterministic. See the module docs for the format and rationale.
366pub fn audio_report(snap: &StatsSnapshot) -> String {
367    let t = &snap.transport;
368    let mut out = String::with_capacity(1024);
369
370    let _ = writeln!(
371        out,
372        "AUDIO HEALTH — self peer {} · {} · ~1s window",
373        t.self_peer_id, t.state
374    );
375    let _ = writeln!(out, "diagnosis: {}", diagnose(snap));
376    let _ = writeln!(out);
377
378    // Transport / device-feeder row.
379    let tick_v = if t.tick_late_p95_ms >= TICK_LATE_HIGH_MS {
380        Verdict::Bad
381    } else {
382        Verdict::Ok
383    };
384    let under_v = if t.playout_output_underruns > 0 {
385        Verdict::Bad
386    } else {
387        Verdict::Ok
388    };
389    let _ = writeln!(
390        out,
391        "transport  rtt={:.0}ms  relay={}",
392        t.rtt_ms,
393        if t.sfu_node_id_short.is_empty() {
394            "-"
395        } else {
396            &t.sfu_node_id_short
397        }
398    );
399    let _ = writeln!(
400        out,
401        "mixtick    late p50/p95   {:.1}/{:.1}ms     [{}]   (RT feeder health)",
402        t.tick_late_p50_ms,
403        t.tick_late_p95_ms,
404        tick_v.tag()
405    );
406    let over_v = if t.playout_output_overruns > 0 {
407        Verdict::Bad
408    } else {
409        Verdict::Ok
410    };
411    let capture_v = if snap.outbound_audio.capture_overflow_samples > 0 {
412        Verdict::Bad
413    } else {
414        Verdict::Ok
415    };
416    let _ = writeln!(
417        out,
418        "devicering out-underruns  {:<14} [{}]   mix-starves {}",
419        t.playout_output_underruns,
420        under_v.tag(),
421        t.playout_mix_starves
422    );
423    let _ = writeln!(
424        out,
425        "           out-overruns   {:<14} [{}]   (bus outran the device)",
426        t.playout_output_overruns,
427        over_v.tag()
428    );
429    // Demoted cpal Xrun/RealtimeDenied callbacks absorbed by the settle funnel. A
430    // sustained rate here WITHOUT a reopen row is the storm case: the funnel keeps
431    // re-arming and deliberately never reopens, so this line is the only evidence.
432    let soft_v = if t.playout_soft_faults > 0 {
433        Verdict::High
434    } else {
435        Verdict::Ok
436    };
437    let _ = writeln!(
438        out,
439        "           soft-faults    {:<14} [{}]   (xrun/rt-denied, debounced)",
440        t.playout_soft_faults,
441        soft_v.tag()
442    );
443    let reopens = device_reopens(snap);
444    let reopen_v = if reopens > 0 {
445        Verdict::Bad
446    } else {
447        Verdict::Ok
448    };
449    let _ = writeln!(
450        out,
451        "           reopens        {:<14} [{}]   (route change / device loss, last ~12s)",
452        reopens,
453        reopen_v.tag()
454    );
455    let _ = writeln!(
456        out,
457        "capture    ring overflow  {:<14} [{}]   (send-side: the FAR end hears this)",
458        snap.outbound_audio.capture_overflow_samples,
459        capture_v.tag()
460    );
461    // A FACT, never a verdict: suppression is correct behaviour, so it gets no
462    // `Verdict` and no rule of its own. It is here because it is the one thing that
463    // makes "this peer sent almost nothing" readable — without it a suppressed
464    // publisher and a dead capture path produce the same collapsed send rate.
465    let _ = writeln!(
466        out,
467        "           suppressed     {:<14} [{}]   (silence not published; wire ~2-4/s)",
468        snap.outbound_audio.frames_suppressed,
469        if snap.outbound_audio.suppressed {
470            "now"
471        } else {
472            "-"
473        }
474    );
475    // The loudness ledger. `apm` says whether capture is processed AT ALL (AEC/
476    // NS/AGC) — a fixed build reads on, a broken one off. The raw→processed pair
477    // splits "the mic is quiet" from "the processing made it quiet": raw is the
478    // device level, processed is what the encoder (the far end) gets, and the
479    // difference is the gain chain's contribution. `clipped` is AGC2's limiter
480    // failing — any sustained growth is a gain-chain bug, never a loud user.
481    let clip_v = if snap.outbound_audio.clipped_frames > 0 {
482        Verdict::Bad
483    } else {
484        Verdict::Ok
485    };
486    let _ = writeln!(
487        out,
488        "           apm            {:<14} [{}]   (AEC/NS/AGC on the mic path)",
489        if snap.outbound_audio.apm_active {
490            "on"
491        } else {
492            "OFF"
493        },
494        if snap.outbound_audio.apm_active {
495            Verdict::Ok.tag()
496        } else {
497            Verdict::Bad.tag()
498        }
499    );
500    let _ = writeln!(
501        out,
502        "           level raw/out  {:>5.1} / {:<6.1} [{}]   (dBFS pre/post processing; gap = applied gain)",
503        snap.outbound_audio.raw_dbfs,
504        snap.outbound_audio.processed_dbfs,
505        clip_v.tag()
506    );
507    if snap.outbound_audio.clipped_frames > 0 {
508        let _ = writeln!(
509            out,
510            "           clipped        {:<14} [{}]   (full-scale frames post-APM — gain-chain bug)",
511            snap.outbound_audio.clipped_frames,
512            clip_v.tag()
513        );
514    }
515    // OS input-volume controller (P5). A FACT like `suppressed` above, not a
516    // verdict: no platform provider (Linux today, or an aggregate/USB device
517    // with no settable property) is a normal digital-only degrade, not a bug —
518    // this line answers "did the auto-level move the mic, and where is it now".
519    let _ = writeln!(
520        out,
521        "           os input vol   {:>3.0}% → {:<3.0}%  [{}]   (OS mic slider: applied → AGC's ask)",
522        snap.outbound_audio.os_volume_applied * 100.0,
523        snap.outbound_audio.os_volume_recommended * 100.0,
524        if snap.outbound_audio.os_volume_available {
525            "auto"
526        } else {
527            "n/a"
528        }
529    );
530
531    // Per-peer inbound audio.
532    let mut any_peer = false;
533    for p in &snap.inbound {
534        for (label, a) in [("mic", &p.audio), ("screen_audio", &p.screen_audio)] {
535            if !audio_active(a) {
536                continue;
537            }
538            any_peer = true;
539            let _ = writeln!(out);
540            let _ = writeln!(out, "peer {} · {label}", p.peer_id);
541            write_audio_stage(&mut out, a);
542        }
543    }
544    if !any_peer {
545        let _ = writeln!(out);
546        let _ = writeln!(out, "(no inbound audio streams this window)");
547    }
548
549    // Glitch log.
550    let _ = writeln!(out);
551    if snap.glitches.is_empty() {
552        let _ = writeln!(out, "glitch log: (none in the last ~12s)");
553    } else {
554        let _ = writeln!(out, "glitch log (newest first):");
555        for g in snap.glitches.iter().take(GLITCH_LOG_LINES) {
556            let who = match g.peer {
557                Some(p) => format!("peer{p} {}", g.label),
558                None => g.label.clone(),
559            };
560            let detail = if g.detail_ms > 0.0 {
561                format!(" {:.1}ms", g.detail_ms)
562            } else {
563                String::new()
564            };
565            let _ = writeln!(
566                out,
567                "  t-{:>5.1}s  {:<16} {}{}",
568                g.ago_ms / 1000.0,
569                who,
570                g.kind,
571                detail
572            );
573        }
574        if snap.glitches.len() > GLITCH_LOG_LINES {
575            let _ = writeln!(out, "  … +{} more", snap.glitches.len() - GLITCH_LOG_LINES);
576        }
577    }
578
579    out
580}

582/// Write the per-stream stage rows (network + jitter buffer + level/recv).
583fn write_audio_stage(out: &mut String, a: &InboundAudioSnapshot) {
584    let jitter_ms = a.jitter_us as f64 / 1000.0;
585    let jitter_v = if jitter_ms >= JITTER_HIGH_MS {
586        Verdict::High
587    } else {
588        Verdict::Ok
589    };
590    let loss = loss_pct(a);
591    let loss_v = if loss >= LOSS_HIGH_PCT {
592        Verdict::Bad
593    } else {
594        Verdict::Ok
595    };
596    let shallow = jitterbuf_shallow(a);
597    let depth_v = if shallow { Verdict::Bad } else { Verdict::Ok };
598    let underflow_v = if a.underflow > 0 || a.overflow_drops > 0 {
599        Verdict::Bad
600    } else {
601        Verdict::Ok
602    };
603
604    let _ = writeln!(
605        out,
606        "  network   jitter           {:<14} [{}]",
607        format!("{jitter_ms:.0}ms"),
608        jitter_v.tag()
609    );
610    let _ = writeln!(
611        out,
612        "            loss/reorder     {:<14} [{}]   late-arrivals {}",
613        format!("{:.1}% / {}", loss, a.reordered),
614        loss_v.tag(),
615        a.late_drops
616    );
617    let _ = writeln!(
618        out,
619        "  jitterbuf depth cur/rec    {:<14} [{}]   {}",
620        format!(
621            "{:.0}/{:.0}ms",
622            a.target_delay_us as f64 / 1000.0,
623            a.recommended_delay_us as f64 / 1000.0
624        ),
625        depth_v.tag(),
626        if shallow { "← too shallow" } else { "" }
627    );
628    let _ = writeln!(
629        out,
630        "            underflow/ovfl   {:<14} [{}]",
631        format!("{} / {}", a.underflow, a.overflow_drops),
632        underflow_v.tag()
633    );
634    // ACTUAL occupancy vs the configured depth — the anchor's health, in both
635    // directions. A backlog (≫ target) is the cause of overflow drops; a dry buffer
636    // (≪ target, while underflowing) means the anchor was eroded by a delay step and
637    // the depth reading above is a fiction. Both auto-resync in the playout buffer.
638    let backlog = backlogged(a);
639    let dry = cushion_lost(a);
640    let queued_v = if backlog || dry {
641        Verdict::Bad
642    } else {
643        Verdict::Ok
644    };
645    let _ = writeln!(
646        out,
647        "            queued (actual) {:<14} [{}]   {}",
648        format!("{}fr ~{:.0}ms", a.playout_buffered_frames, buffered_ms(a)),
649        queued_v.tag(),
650        match (backlog, dry) {
651            (true, _) => "← backlog (catching up)",
652            (_, true) => "← dry (cushion lost)",
653            _ => "",
654        }
655    );
656    let _ = writeln!(
657        out,
658        "  level/recv                 {:.2} / {:.1}kbps",
659        a.level, a.recv_kbps
660    );
661}
