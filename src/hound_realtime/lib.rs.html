1//! The realtime plane: wire protocol plus both ends' reducers, sans-IO.
2//!
3//! This is the non-media half of hound's live traffic — messages, space layout,
4//! voice roster, screenshare — carried over whatever transport the client can
5//! get (WebTransport where it works, a WebSocket where it does not). Nothing
6//! here touches a socket, a clock, or a database:
7//!
8//! - [`proto`] is the wire contract, JSON-tagged and versioned.
9//! - [`hub`] is the server reducer: room membership in, addressed messages out.
10//! - [`session`] is the client reducer: it owns the desired-subscription set and
11//!   reconciles it on every reconnect.
12//!
13//! The IO shells (the backend's `api::realtime`, the client's realtime shell)
14//! own connections, authentication, and the database lookups; they feed these
15//! reducers already-resolved facts and route what comes back.
16
17pub mod hub;
18pub mod proto;
19pub mod session;
20

22pub use proto::{
23    Attachment, ClientMessage, ErrorCode, MIN_PROTOCOL_VERSION, MessagePayload, PROTOCOL_VERSION,
24    ServerMessage, SpaceKind, SpacePayload, negotiate_version,
25};
26pub use session::{Action, ClientSession, Event, Reaction};
