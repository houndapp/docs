14/// Keep-alive interval. Well under [`DEFAULT_MAX_IDLE`] so an idle-but-live
15/// session is never mistaken for a dead one, and short enough that a NAT
16/// binding for the UDP flow stays open.
17pub const DEFAULT_KEEP_ALIVE: Duration = Duration::from_secs(5);
18
19/// Idle timeout after which a silent connection is dropped.
20///
21/// **This must always be set.** wtransport's default is *infinite*: a peer that
22/// vanishes without a close (laptop lid, dead NAT, force-quit) would otherwise
23/// leave its connection — and every future awaiting it — parked forever.
24pub const DEFAULT_MAX_IDLE: Duration = Duration::from_secs(30);
25
26/// How long a peer may take to get from "the endpoint saw its first packet" to
27/// admitted or turned away — the whole handshake + authenticate + accept/reject
28/// sequence.
29///
30/// Every step is under the caller's control except the peer's own progress, so
31/// this is the bound that stops an unauthenticated peer from holding a server
32/// task (and, before per-peer admission, the accept loop itself). Matches the
33/// SFU's session-auth deadline.
34pub const DEFAULT_ADMISSION_TIMEOUT: Duration = Duration::from_secs(10);
35
36/// How the server proves its identity to clients.
37#[derive(Debug, Clone)]
38pub enum Identity {
39    /// A certificate chain and private key on disk. Re-read on every
40    /// [`WtServer::reload`], which is what makes rotation a config reload rather
41    /// than a restart.
42    Pem { cert: PathBuf, key: PathBuf },
43    /// A freshly generated self-signed certificate for the given subject alt
44    /// names. Clients must pin its [`CertHash`] — see
45    /// [`Trust::CertHash`](crate::Trust). Development only: the generated
46    /// certificate is valid for 14 days, the maximum a pinning client accepts.
47    SelfSigned { subject_alt_names: Vec<String> },
48}

50/// Everything needed to stand up (or reload) a server endpoint.
51#[derive(Debug, Clone)]
52pub struct ServerOptions {
53    pub bind: SocketAddr,
54    pub identity: Identity,
55    /// `None` disables keep-alive probes entirely.
56    pub keep_alive: Option<Duration>,
57    pub max_idle: Duration,
58}
59
60impl ServerOptions {
61    /// Options with this crate's defaults for the timing knobs.
62    pub fn new(bind: SocketAddr, identity: Identity) -> Self {
63        Self {
64            bind,
65            identity,
66            keep_alive: Some(DEFAULT_KEEP_ALIVE),
67            max_idle: DEFAULT_MAX_IDLE,
68        }
69    }
70
71    /// Build the wtransport config and describe the leaf certificate it serves.
72    async fn build(&self) -> Result<(ServerConfig, LeafCert), WtError> {
73        let identity = match &self.identity {
74            Identity::Pem { cert, key } => wtransport::Identity::load_pemfiles(cert, key)
75                .await
76                .map_err(|e| WtError::Identity(e.to_string()))?,
77            Identity::SelfSigned { subject_alt_names } => {
78                wtransport::Identity::self_signed(subject_alt_names)
79                    .map_err(|e| WtError::Identity(e.to_string()))?
80            }
81        };
82
83        let leaf = identity
84            .certificate_chain()
85            .as_slice()
86            .first()
87            .ok_or_else(|| WtError::Identity("empty certificate chain".to_owned()))?;
88        let leaf = LeafCert {
89            hash: CertHash(leaf.hash()),
90            not_after: not_after_of(leaf.der()),
91        };
92
93        let config = ServerConfig::builder()
94            .with_bind_address(self.bind)
95            .with_identity(identity)
96            .keep_alive_interval(self.keep_alive)
97            .max_idle_timeout(Some(self.max_idle))
98            .map_err(|e| WtError::Identity(e.to_string()))?
99            .build();
100
101        Ok((config, leaf))
102    }
103}

105/// The leaf certificate as clients and operators need to see it.
106#[derive(Debug, Clone)]
107pub struct LeafCert {
108    /// The digest a pinning client matches.
109    pub hash: CertHash,
110    /// `notAfter`. `None` when the DER cannot be parsed, which never blocks
111    /// serving — it only means the expiry cannot be published.
112    pub not_after: Option<SystemTime>,
113}
114
115/// The `notAfter` of a DER-encoded X.509 certificate.
116fn not_after_of(der: &[u8]) -> Option<SystemTime> {
117    use x509_parser::prelude::FromDer;
118
119    let (_, cert) = x509_parser::certificate::X509Certificate::from_der(der).ok()?;
120    let seconds = u64::try_from(cert.validity().not_after.timestamp()).ok()?;
121    Some(SystemTime::UNIX_EPOCH + Duration::from_secs(seconds))
122}

124/// The SHA-256 digest of a server's leaf certificate.
125///
126/// Its string form is dotted lowercase hex (`ab:cd:…`), which is what a pinning
127/// client parses back and what a config endpoint hands out for bootstrap.
128#[derive(Debug, Clone, PartialEq, Eq, Hash)]
129pub struct CertHash(Sha256Digest);
130
131impl CertHash {
132    /// Parse the dotted-hex form produced by [`std::fmt::Display`].
133    pub fn parse(s: &str) -> Result<Self, WtError> {
134        Sha256Digest::from_str_fmt(s, Sha256DigestFmt::DottedHex)
135            .map(Self)
136            .map_err(|e| WtError::Identity(e.to_string()))
137    }
138
139    pub(crate) fn digest(&self) -> Sha256Digest {
140        self.0.clone()
141    }
142}
143
144impl std::fmt::Display for CertHash {
145    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
146        f.write_str(&self.0.fmt(Sha256DigestFmt::DottedHex))
147    }
148}

150/// A bound WebTransport server endpoint.
151pub struct WtServer {
152    endpoint: Endpoint<Server>,
153    /// The current leaf certificate, republished by [`WtServer::reload`] so a
154    /// rotation is visible to whatever advertises or alerts on it.
155    leaf: RwLock<LeafCert>,
156}
157
158impl WtServer {
159    /// Bind the endpoint. The UDP socket is live when this returns, so a
160    /// preflighted port conflict surfaces here rather than mid-run.
161    pub async fn bind(options: &ServerOptions) -> Result<Self, WtError> {
162        let (config, leaf) = options.build().await?;
163        let endpoint = Endpoint::server(config)?;
164        Ok(Self {
165            endpoint,
166            leaf: RwLock::new(leaf),
167        })
168    }
169
170    /// The address actually bound — resolves an ephemeral `:0` port.
171    pub fn local_addr(&self) -> Result<SocketAddr, WtError> {
172        self.endpoint.local_addr().map_err(WtError::Endpoint)
173    }
174
175    /// The current leaf certificate digest, for clients that pin instead of
176    /// validating against a CA.
177    pub fn cert_hash(&self) -> CertHash {
178        self.leaf().hash
179    }
180
181    /// The current leaf certificate — its digest and expiry.
182    pub fn leaf(&self) -> LeafCert {
183        // A poisoned lock means some other task panicked while swapping the
184        // certificate; the value it holds is a fully built `LeafCert` either
185        // way, and refusing to serve every future handshake over it would turn
186        // one panic into an outage.
187        self.leaf
188            .read()
189            .unwrap_or_else(|poisoned| poisoned.into_inner())
190            .clone()
191    }
192
193    /// Await the next peer at the **UDP** boundary: its first packet arrived,
194    /// but neither the QUIC/HTTP/3 handshake nor authentication has happened.
195    ///
196    /// Returning this early is what keeps admission per-peer: the caller spawns
197    /// a task that runs [`Incoming::handshake`] and the authorization decision
198    /// under its own deadline, so one peer that stalls mid-handshake cannot
199    /// delay everyone behind it in the accept queue.
200    pub async fn accept(&self) -> Incoming {
201        Incoming(self.endpoint.accept().await)
202    }
203
204    /// Swap in a freshly built configuration — the certificate-rotation path.
205    ///
206    /// Re-reads the PEM files (so a rotated pair is picked up), replaces the
207    /// endpoint's TLS config, and republishes [`WtServer::leaf`]. Existing
208    /// connections keep running on the old certificate; new handshakes use the
209    /// new one. The socket is never rebound, so no packet is dropped.
210    pub async fn reload(&self, options: &ServerOptions) -> Result<LeafCert, WtError> {
211        let (config, leaf) = options.build().await?;
212        self.endpoint.reload_config(config, false)?;
213        *self
214            .leaf
215            .write()
216            .unwrap_or_else(|poisoned| poisoned.into_inner()) = leaf.clone();
217        Ok(leaf)
218    }
219}

221/// A peer at the UDP boundary, before its handshake.
222///
223/// Nothing about it is trustworthy yet — not even that it speaks HTTP/3. Every
224/// method that waits on the peer is the caller's to bound.
225pub struct Incoming(wtransport::endpoint::IncomingSession);
226
227impl Incoming {
228    /// The peer's UDP address, available before the handshake so a refusal can
229    /// still be attributed.
230    pub fn remote_address(&self) -> SocketAddr {
231        self.0.remote_address()
232    }
233
234    /// Complete the QUIC + HTTP/3 handshake, yielding the peer's CONNECT request
235    /// — the trust boundary where authentication happens.
236    ///
237    /// Unbounded: a peer may keep this pending for as long as its transport
238    /// allows, so callers wrap it in their admission deadline.
239    pub async fn handshake(self) -> Result<SessionRequest, WtError> {
240        self.0
241            .await
242            .map(SessionRequest)
243            .map_err(|e| WtError::Connect(e.to_string()))
244    }
245
246    /// Refuse the peer without completing the handshake — no HTTP status, just a
247    /// QUIC-level refusal. Immediate: nothing is awaited on a peer that has not
248    /// proved it is worth waiting for.
249    pub fn refuse(self) {
250        self.0.refuse();
251    }
252}

254/// Why a session is being turned away. Each maps to the HTTP/3 status the
255/// WebTransport CONNECT response carries, which is what a browser client
256/// surfaces to the page.
257#[derive(Debug, Clone, Copy, PartialEq, Eq)]
258pub enum Reject {
259    /// 403 — authenticated but not allowed (or no credentials at all).
260    Forbidden,
261    /// 404 — no such endpoint path.
262    NotFound,
263    /// 429 — over a quota.
264    TooManyRequests,
265}
266
267/// A peer awaiting admission: the handshake completed, but no session exists
268/// until the caller [`accept`](Self::accept)s.
269///
270/// This is the trust boundary. Authenticate from the request's own fields —
271/// typically a token in [`SessionRequest::query_param`] — and
272/// [`reject`](Self::reject) before any application state is created.
273pub struct SessionRequest(wtransport::endpoint::SessionRequest);
274
275impl SessionRequest {
276    /// The `:path` field of the CONNECT request, query string included.
277    pub fn path(&self) -> &str {
278        self.0.path()
279    }
280
281    /// The `:authority` field (host[:port]) the client dialled.
282    pub fn authority(&self) -> &str {
283        self.0.authority()
284    }
285
286    /// The peer's UDP address.
287    pub fn remote_address(&self) -> SocketAddr {
288        self.0.remote_address()
289    }
290
291    /// All CONNECT request headers.
292    pub fn headers(&self) -> &std::collections::HashMap<String, String> {
293        self.0.headers()
294    }
295
296    /// One CONNECT request header, matched case-insensitively.
297    ///
298    /// HTTP/3 field names are lowercase on the wire, but nothing stops an
299    /// intermediary from inserting a differently-cased one, and a credential
300    /// that silently fails to be found is the worst kind of auth bug.
301    pub fn header(&self, name: &str) -> Option<&str> {
302        self.0
303            .headers()
304            .iter()
305            .find(|(key, _)| key.eq_ignore_ascii_case(name))
306            .map(|(_, value)| value.as_str())
307    }
308
309    /// The bearer credential from the `Authorization` header, if it carries one.
310    ///
311    /// Preferred over [`Self::query_param`]: a token in the `:path` lands in
312    /// access logs and proxy traces. A browser's `WebTransport` constructor
313    /// cannot set headers, so the query parameter remains the portable fallback.
314    pub fn bearer_token(&self) -> Option<&str> {
315        bearer_of(self.header("authorization")?)
316    }
317
318    /// The raw query string, if the path has one.
319    pub fn query(&self) -> Option<&str> {
320        query_of(self.0.path())
321    }
322
323    /// The first value of a percent-decoded query parameter.
324    ///
325    /// The path is the only credential channel a browser's `WebTransport`
326    /// constructor can populate — it cannot set request headers — so a token
327    /// query parameter is the portable way to authenticate a session.
328    pub fn query_param(&self, key: &str) -> Option<String> {
329        query_param_of(self.0.path(), key)
330    }
331
332    /// Admit the peer and return the live session.
333    pub async fn accept(self) -> Result<Connection, WtError> {
334        self.0
335            .accept()
336            .await
337            .map_err(|e| WtError::Connect(e.to_string()))
338    }
339
340    /// Turn the peer away with the matching HTTP/3 status; no session is created.
341    pub async fn reject(self, reason: Reject) {
342        match reason {
343            Reject::Forbidden => self.0.forbidden().await,
344            Reject::NotFound => self.0.not_found().await,
345            Reject::TooManyRequests => self.0.too_many_requests().await,
346        }
347    }
348}

350/// The credential out of an `Authorization: Bearer <token>` value.
351pub fn bearer_of(header: &str) -> Option<&str> {
352    let (scheme, token) = header.trim().split_once(' ')?;
353    scheme
354        .eq_ignore_ascii_case("bearer")
355        .then(|| token.trim())
356        .filter(|token| !token.is_empty())
357}
358
359/// The query string of a `:path`, if any.
360fn query_of(path: &str) -> Option<&str> {
361    path.split_once('?').map(|(_, query)| query)
362}
363
364/// First percent-decoded value of `key` in a `:path`'s query string.
365fn query_param_of(path: &str, key: &str) -> Option<String> {
366    let query = query_of(path)?;
367    url::form_urlencoded::parse(query.as_bytes())
368        .find(|(k, _)| k == key)
369        .map(|(_, value)| value.into_owned())
370}

372#[cfg(test)]
373mod tests {
374    use super::*;

376    /// The token is carried in the CONNECT path's query, so parsing it is part
377    /// of the trust boundary: a missing query, a missing key, and percent
378    /// encoding must all behave predictably before anything is authenticated.
379    #[test]
380    fn query_params_are_extracted_and_percent_decoded() {
381        assert_eq!(
382            query_param_of("/realtime?token=abc.def", "token").as_deref(),
383            Some("abc.def")
384        );
385        assert_eq!(
386            query_param_of("/realtime?x=1&token=a%20b&y=2", "token").as_deref(),
387            Some("a b"),
388            "percent escapes must be decoded"
389        );
390        assert_eq!(query_param_of("/realtime?x=1", "token"), None);
391        assert_eq!(query_param_of("/realtime", "token"), None);
392        assert_eq!(query_of("/realtime"), None);
393        assert_eq!(query_of("/realtime?a=1"), Some("a=1"));
394    }
395
396    /// The header credential channel: the scheme is case-insensitive, the token
397    /// is trimmed, and anything that is not a non-empty bearer value yields
398    /// nothing rather than a credential the validator would have to reject.
399    #[test]
400    fn bearer_tokens_are_parsed_case_insensitively() {
401        assert_eq!(bearer_of("Bearer abc.def"), Some("abc.def"));
402        assert_eq!(bearer_of("bearer abc.def"), Some("abc.def"));
403        assert_eq!(bearer_of("BEARER   abc.def  "), Some("abc.def"));
404        assert_eq!(bearer_of("Basic abc.def"), None, "only bearer counts");
405        assert_eq!(bearer_of("Bearer"), None);
406        assert_eq!(bearer_of("Bearer "), None, "an empty token is no token");
407        assert_eq!(bearer_of("abc.def"), None, "a bare token is not a header");
408    }
409
410    /// A generated self-signed certificate must publish an expiry, which is what
411    /// the endpoint's rotation gauge reports; pinning clients cap validity at two
412    /// weeks, so it is also always in the near future.
413    #[tokio::test]
414    async fn a_self_signed_identity_reports_its_expiry() {
415        let options = ServerOptions::new(
416            "127.0.0.1:0".parse().unwrap(),
417            Identity::SelfSigned {
418                subject_alt_names: vec!["localhost".to_owned()],
419            },
420        );
421        let (_, leaf) = options.build().await.unwrap();
422        let not_after = leaf.not_after.expect("a parseable notAfter");
423        let remaining = not_after
424            .duration_since(SystemTime::now())
425            .expect("not yet expired");
426        assert!(
427            remaining <= Duration::from_secs(15 * 24 * 60 * 60),
428            "a pinnable certificate is short-lived, got {remaining:?}"
429        );
430    }

437    /// The digest's string form is the bootstrap contract: whatever a config
438    /// endpoint hands out must parse back to the same pin on the client.
439    #[test]
440    fn cert_hash_round_trips_through_its_string_form() {

443        assert!(
