1//! The transport-agnostic seam every protocol shell programs against.
2//!
3//! A pipe carries **whole text frames**: one [`MessagePipe::send`] is one frame
4//! the peer receives intact, never split or coalesced. That is the property a
5//! JSON control protocol needs and the only property it needs, so both a
6//! WebTransport bidirectional stream ([`crate::FramedBi`]) and a WebSocket can
7//! provide it — a shell written against this trait serves both legs with one
8//! session loop.
9
10use std::future::Future;
11
12use serde::Serialize;
13use serde::de::DeserializeOwned;
14
15use crate::error::PipeError;
16
17/// Largest frame accepted in either direction (1 MiB). A control frame is
18/// kilobytes at most; the cap bounds what a peer can make the receiver buffer
19/// before the length prefix is even satisfied.
20pub const MAX_FRAME_BYTES: usize = 1024 * 1024;
21
22/// A bidirectional, message-oriented, UTF-8 text pipe.
23///
24/// ## Contract
25///
26/// - `send` delivers one whole frame, or fails terminally for this pipe.
27/// - `recv` yields `None` once the peer closed cleanly or the pipe is finished;
28///   `Some(Err(_))` is a transport failure and is likewise terminal.
29/// - `recv` MUST be cancel-safe: dropping the future (e.g. losing a
30///   `tokio::select!` race) must not consume or truncate a frame. Both
31///   implementations satisfy this by keeping decode state in the pipe rather
32///   than in the future.
33/// - `close` is best-effort and idempotent.
34pub trait MessagePipe: Send {
35    /// Send one frame.
36    fn send(&mut self, frame: String) -> impl Future<Output = Result<(), PipeError>> + Send;
37
38    /// Receive the next frame; `None` once the pipe is finished.
39    fn recv(&mut self) -> impl Future<Output = Option<Result<String, PipeError>>> + Send;
40
41    /// Close the pipe, flushing what is already queued. Best-effort.
42    fn close(&mut self) -> impl Future<Output = ()> + Send;
43}
44
45/// Server-driven liveness for a pipe whose transport carries no keep-alive of
46/// its own.
47///
48/// A WebSocket over TCP has no idle timeout: a peer whose network vanished
49/// (dead NAT, sleeping laptop, force-quit) leaves the socket open indefinitely,
50/// so the server has to probe and reap it. A QUIC-backed pipe already has that
51/// mechanism, which is why [`crate::FramedBi`]'s implementation is a no-op probe
52/// plus an honest frame count — one session loop is written against this trait
53/// and each leg arms the timers it actually needs.
54pub trait Liveness {
55    /// Send a transport-level probe the peer's stack answers without the
56    /// application being involved (a WebSocket `Ping`).
57    fn ping(&mut self) -> impl Future<Output = Result<(), PipeError>> + Send;
58
59    /// Frames received from the peer since the pipe was created, **including**
60    /// ones the protocol skips (ping/pong/binary).
61    ///
62    /// A liveness timer compares this across an interval: unchanged means
63    /// nothing at all arrived, which is the only signal separating a
64    /// silent-but-live peer (it answers pings) from a vanished one.
65    fn inbound_count(&self) -> u64;
66}

68/// JSON framing over any [`MessagePipe`], blanket-implemented so both legs get
69/// identical wire behaviour (and identical skip-on-garbage semantics) for free.
70pub trait MessagePipeExt: MessagePipe {
71    /// Serialize `msg` and send it as one frame.
72    fn send_json<T>(&mut self, msg: &T) -> impl Future<Output = Result<(), PipeError>> + Send
73    where
74        T: Serialize + Sync + ?Sized;
75
76    /// Receive the next frame that deserializes into `T`.
77    ///
78    /// A frame that fails to parse — malformed JSON, or a variant this build
79    /// does not know because the peer is newer — is **skipped, not fatal**: the
80    /// protocol's additive-compatibility rule is that an unknown frame costs
81    /// that frame, never the session. `None` means the pipe is finished.
82    fn recv_json<T>(&mut self) -> impl Future<Output = Option<T>> + Send
83    where
84        T: DeserializeOwned + Send;
85}
86
87impl<P> MessagePipeExt for P
88where
89    P: MessagePipe + ?Sized,
90{
91    fn send_json<T>(&mut self, msg: &T) -> impl Future<Output = Result<(), PipeError>> + Send
92    where
93        T: Serialize + Sync + ?Sized,
94    {
95        // Encode before the async block so the returned future borrows only
96        // `self` — `T` never has to be `'static` or `Send`.
97        let encoded = serde_json::to_string(msg);
98        async move { self.send(encoded?).await }
99    }
100
101    async fn recv_json<T>(&mut self) -> Option<T>
102    where
103        T: DeserializeOwned + Send,
104    {
105        loop {
106            match self.recv().await {
107                Some(Ok(frame)) => match serde_json::from_str::<T>(&frame) {
108                    Ok(msg) => return Some(msg),
109                    Err(error) => {
110                        tracing::warn!(%error, "pipe: skipping unparseable frame");
111                    }
112                },
113                Some(Err(error)) => {
114                    tracing::debug!(%error, "pipe: transport failure, ending");
115                    return None;
116                }
117                None => return None,
118            }
119        }
120    }
121}
