1//! The signed-in user's own global profile: rename and avatar edits.
2//!
3//! Each command answers with the profile as written and has already folded it
4//! into the roster stream, so `selfProfileProvider` (the roster entry for the
5//! signed-in user) shows the edit before the future completes. `set_avatar`
6//! takes the file's bytes, already read by Dart — the same seam as
7//! `set_community_icon`.
8
9use hound_client::error::HoundError;
10use hound_client::model::Profile;
11
12use super::client;
13
14/// Rename the global profile. Refused (`empty_name` / `name_too_long`, with
15/// ready-to-show copy) outside 1..=32 characters after trimming.
16pub async fn set_display_name(name: String) -> Result<Profile, HoundError> {
17    client()?.set_display_name(&name).await
18}
19
20/// Upload `bytes` as the global avatar and point the profile at it. `mime` must
21/// be one of the types this deployment advertises for avatars
22/// ([`supported_upload_mime_types`](super::supported_upload_mime_types)).
23pub async fn set_avatar(bytes: Vec<u8>, mime: String) -> Result<Profile, HoundError> {
24    client()?.set_avatar(bytes, &mime).await
25}
26
27/// Drop the global avatar; every surface falls back to initials.
28pub async fn clear_avatar() -> Result<Profile, HoundError> {
29    client()?.clear_avatar().await
30}
31
32/// Make the caller's Steam photo the global avatar. The server copies it, so
33/// the client never re-uploads Steam's image. Refused `no_steam_identity` (404)
34/// without a linked Steam account, `steam_unavailable` (502) when Steam does
35/// not answer.
36pub async fn import_steam_avatar() -> Result<Profile, HoundError> {
37    client()?.import_steam_avatar().await
38}
