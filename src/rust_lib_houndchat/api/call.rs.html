30/// The actionable messages for a device the OS has denied us. Same shape as the
31/// screen-share denial (`map_screen_capture_error` in `hound-client`), so all three
32/// read alike in the Dart snackbar.
33const CAMERA_DENIED: &str =
34    "camera permission denied — enable it in System Settings → Privacy & Security → Camera";
35const MIC_DENIED: &str =
36    "microphone permission denied — enable it in System Settings → Privacy & Security → Microphone";
37
38/// Refuse to *enable* a device the OS has already denied us.
39///
40/// The capture threads preflight too — and they are what raises the system prompt
41/// — but they run BEHIND this toggle: without this check the command flips a flag,
42/// returns `Ok`, and the capture loop then silently never produces a frame, so the
43/// user just sees a dead tile. Returning a [`HoundError::Media`] instead routes
44/// through the existing Dart `context.guard` → `humanizeError` snackbar with no
45/// Dart change.
46///
47/// `NotDetermined` must PROCEED (only the open path can prompt), as must
48/// `Unknown` — every non-Apple target, where there is no gate to consult and
49/// behaviour is unchanged.
50fn ensure_device_permitted(
51    kind: hound_engine::types::DeviceKind,
52    denied: &str,
53) -> Result<(), HoundError> {
54    use hound_engine::types::DevicePermission;
55    match hound_engine::api::Engine::new().device_permission(kind) {
56        DevicePermission::Denied => {
57            Err(HoundError::media(MediaCode::DevicePermissionDenied, denied))
58        }
59        DevicePermission::Granted | DevicePermission::NotDetermined | DevicePermission::Unknown => {
60            Ok(())
61        }
62    }
63}

65/// Mute/unmute the local mic. Un-muting is an *enable*, so it fails loudly when
66/// the mic is denied (see [`ensure_device_permitted`]); muting always succeeds.
67pub async fn call_set_muted(muted: bool) -> Result<(), HoundError> {
68    if !muted {
69        ensure_device_permitted(hound_engine::types::DeviceKind::Microphone, MIC_DENIED)?;
70    }
71    client()?.call_set_muted(muted).await
72}

91/// The per-source wire tokens [`HoundClient`](hound_client::HoundClient)'s stringly
92/// command surface parses back into `hound_control::Source`. Kept here so the Dart
93/// side deals only in the mirrored [`Source`] enum and no wire literal ever crosses
94/// the bridge.
95fn source_wire(source: Source) -> &'static str {
96    match source {
97        Source::Microphone => "mic",
98        Source::Camera => "camera",
99        Source::Screen => "screen",
100    }
101}

103/// Report a remote video tile's rendered size + visibility, for peer `peer_id`'s
104/// `source` ([`Source::Camera`] | [`Source::Screen`]). The tile takes the cheapest
105/// below-source rung that can fill it (and a large/visible one the full source), so
106/// a receiver only pays to decode what it is actually showing.
107///
108/// `width`/`height` are **PHYSICAL** pixels, not logical ones: the rung ladder
109/// prices a tile by the pixels really decoded and painted, so the caller multiplies
110/// its laid-out extents by the surface's device pixel ratio first (the Flutter side
111/// does this in `DemandBucketer`). Call from each tile's layout/visibility. No-op
112/// when not in a call or for a non-video source.
113pub async fn call_set_tile_quality(
114    peer_id: u32,
115    source: Source,
116    width: u32,
117    height: u32,
118    visible: bool,
119) -> Result<(), HoundError> {
120    client()?
121        .call_set_tile_quality(peer_id, source_wire(source).into(), width, height, visible)
122        .await
123}

210/// Select a call media device. `kind` is `"microphone"` | `"camera"` | `"speaker"`;
211/// `id` is the device id (`None` = host default).
212///
213/// While the mic test is running (and no call is), a `"microphone"` pick re-points
214/// the monitor instead — the same command, whichever pipeline holds the device.
215/// A `"camera"` pick likewise re-points a running camera preview.
216pub async fn call_select_device(kind: String, id: Option<String>) -> Result<(), HoundError> {
217    let c = client()?;
218    match kind.as_str() {
219        "microphone" => c.call_select_microphone(id).await,
220        "camera" => c.call_select_camera(id).await,
221        "speaker" => c.call_select_speaker(id).await,
222        other => Err(HoundError::invalid_state(
223            StateCode::UnknownDeviceKind,
224            format!("unknown device kind: {other}"),
225        )),
226    }
227}
