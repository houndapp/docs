1//! The video/camera surface: the zero-copy texture handshake, plus the
2//! decode→render seam ([`RenderVideoSink`]) that `init_app` installs.
3
4/// The decode→render seam: forwards decoded I420 frames into the per-stream-key
5/// render mailbox ([`hound_render::push_frame`]). Keeps `hound-client`
6/// UI/FFI-free — the texture path lives behind the `hound-render` facade. The
7/// mailbox is cross-platform (macOS GPU render thread / Windows+Linux CPU render
8/// thread both drain it); a frame for a not-yet-attached key just queues-of-one
9/// until attach.
10pub(super) struct RenderVideoSink;
11
12impl hound_client::VideoSink for RenderVideoSink {
13    fn push_frame(&self, stream_key: &str, w: u32, h: u32, i420: Vec<u8>) {
14        // This seam is the only place that sees EVERY frame's geometry, so it is
15        // where a resolution change becomes an event for the texture-sizing stream
16        // ([`watch_frame_sizes`](super::watch_frame_sizes)) instead of waiting up
17        // to a second for the next stats snapshot. Unchanged geometry (the steady
18        // state) costs one hash hit.
19        crate::frame_size::geometry().record(stream_key, w, h);
20        hound_render::push_frame(stream_key, w, h, i420);
21    }
22
23    /// Zero-copy fast path: hand the decoded GPU surface straight to the renderer.
24    /// Every variant is a platform surface pointer + extent and
25    /// `hound_render::push_surface` is platform-uniform, so this stays cfg-free:
26    /// a handle built on one platform only ever reaches that platform's renderer,
27    /// which returns `false` where there is no surface path (or the
28    /// `HOUND_ZEROCOPY_VIDEO=0` opt-out is set) — the caller then falls back to
29    /// the I420 `push_frame` path.
30    fn push_surface(&self, stream_key: &str, surface: hound_client::RenderSurfaceHandle) -> bool {
31        let (ptr, width, height) = match surface {
32            hound_client::RenderSurfaceHandle::AppleCvPixelBuffer { ptr, width, height }
33            | hound_client::RenderSurfaceHandle::WindowsD3d11Nv12 { ptr, width, height } => {
34                (ptr, width, height)
35            }
36        };
37        // Same change-detection as the I420 path — the zero-copy path is the
38        // default for remote video on macOS, so it is the one a simulcast layer
39        // switch actually travels down.
40        crate::frame_size::geometry().record(stream_key, width, height);
41        hound_render::push_surface(stream_key, ptr, width, height)
42    }
43
44    /// The tile's last subscription went away with no replacement (peer left, track
45    /// withdrawn, allocator hid the tile, video paused): prune the render mailbox so
46    /// the last decoded frame — a retained VideoToolbox pool buffer on macOS — is
47    /// released instead of pinned for the process lifetime. A simulcast layer switch
48    /// does NOT come through here; see `hound_client::VideoSink::stream_ended`.
49    fn stream_ended(&self, stream_key: &str) {
50        crate::frame_size::geometry().forget(stream_key);
51        hound_render::drop_stream(stream_key);
52    }
53}

55// ─── Zero-copy video texture handshake ──────────────────────────────────────────
56//
57// Flutter side registers a platform GPU texture (IOSurface-backed Metal texture
58// on Apple) with its TextureRegistry, gets a `texture_id` for the `Texture`
59// widget, and passes the NATIVE texture handle down here. Rust renders decoded
60// frames into that texture (reusing the wgpu I420→RGB pipeline) and signals
61// "frame ready" so Flutter composites it — NO frame bytes cross the FFI.
62
63/// Attach a stream key (e.g. "cam:42", "self", "screen:7") to a native GPU
64/// texture handle the Flutter plugin registered, sized `width`×`height`.
65/// `native_texture` is an opaque pointer (an `IOSurfaceRef` / `id<MTLTexture>` on
66/// Apple) the Rust renderer will draw into. `texture_id` is the Flutter texture id
67/// the same `createTexture` returned: macOS's shared render loop marks it "dirty"
68/// once it draws a new frame, and the plugin's ONE shared `CVDisplayLink` re-pulls
69/// only dirty ids (so `copyPixelBuffer` runs at video-fps, not display refresh). The
70/// plugin allocates the buffer at this SAME `width`×`height` (the decoded frame's
71/// size, so the render target matches the source aspect with no crop/stretch); Dart
72/// recreates + re-attaches at a new size when the decoded resolution changes. Returns
73/// once the render target is bound.
74pub fn attach_video_texture(
75    stream_key: String,
76    native_texture: usize,
77    texture_id: i64,
78    width: u32,
79    height: u32,
80) -> Result<(), String> {
81    // One path per platform behind the uniform `hound-render` facade: macOS wraps
82    // the CVPixelBuffer's IOSurface as a wgpu texture (zero-copy GPU render), reads
83    // the target geometry off the buffer (so `width`/`height` are advisory there),
84    // and uses `texture_id` to drive the dirty-set repaint; Windows/Linux take the
85    // plugin's RGBA pixel-buffer pointer + the size (the raw buffer can't be
86    // introspected) and CPU-convert into it (`texture_id` is unused there today).
87    // `native_texture` is the IOSurface-backed CVPixelBufferRef on macOS, the RGBA
88    // buffer pointer elsewhere.
89    hound_render::attach(stream_key, native_texture, texture_id, width, height)
90}

92/// Detach a stream's texture (tile unmounted / peer left). Stops + joins the render
93/// thread (which then no longer touches the native texture/buffer) so the plugin can
94/// release it.
95pub fn detach_video_texture(stream_key: String) {
96    hound_render::detach(&stream_key);
97}

99/// How long a texture swap will hold the OLD texture on screen waiting for the new
100/// one's first frame. A layer switch delivers an immediate IDR and the renderer
101/// draws on arrival, so this is normally satisfied within one frame interval; the
102/// deadline only covers a stream that stops producing mid-swap, where showing the
103/// (black) new texture is no worse than the frozen old one.
104const TEXTURE_READY_DEADLINE: std::time::Duration = std::time::Duration::from_millis(500);
105
106/// Resolves once a real decoded frame has been drawn into `texture_id` — the signal
107/// Dart's make-before-break texture swap flips on. A freshly allocated
108/// `CVPixelBuffer` is zero-filled, so showing it before this would flash black;
109/// meanwhile the OLD texture keeps compositing its last frame, so the tile never
110/// goes blank either.
111///
112/// Returns `false` if the texture was detached/replaced first, if the deadline
113/// ([`TEXTURE_READY_DEADLINE`]) passed, or on a platform with no observable
114/// first-frame edge — every one of which means "stop waiting and flip".
115pub async fn video_texture_ready(texture_id: i64) -> bool {
116    let (tx, rx) = tokio::sync::oneshot::channel();
117    hound_render::notify_on_draw(
118        texture_id,
119        Box::new(move || {
120            let _ = tx.send(());
121        }),
122    );
123    // A dropped sender (detached/replaced texture) resolves `rx` with `Err`
124    // immediately — that is the "never drawn" path, not a deadline wait.
125    matches!(
126        tokio::time::timeout(TEXTURE_READY_DEADLINE, rx).await,
127        Ok(Ok(()))
128    )
129}
