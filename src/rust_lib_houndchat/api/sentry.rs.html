1//! Crash reporting: Rust panics → Sentry. Policy (event-vs-breadcrumb rules,
2//! event enrichment) lives in [`crate::telemetry`]; this is the bridge half.
3
4use super::CLIENT;

6/// The [`sentry::ClientInitGuard`] returned by [`sentry::init`], parked here for the
7/// process lifetime. Dropping it would flush and shut the client down, so it must
8/// outlive every later panic; a plain local would drop at the end of `init_sentry`.
9static SENTRY_GUARD: std::sync::OnceLock<sentry::ClientInitGuard> = std::sync::OnceLock::new();
10
11/// Install Sentry crash reporting. Idempotent — a second call is a no-op, so the UI
12/// can call this unconditionally on every (re)start without double-initialising.
13///
14/// The `panic` default integration (see the `sentry` feature list in
15/// `crates/frb/Cargo.toml`) is what turns any Rust panic, on any thread, into a
16/// Sentry event; this function does no explicit hooking beyond that. A malformed
17/// `dsn` parses to `None` (Sentry disabled, no panic) rather than erroring — the
18/// caller only invokes this when its own DSN config is non-empty.
19///
20/// `rustls-no-provider` (the event-upload transport's TLS feature, see the `sentry`
21/// Cargo.toml comment) needs a process-default `CryptoProvider` installed before
22/// its `reqwest::Client` is built inside `sentry::init`; install `ring` ourselves so
23/// this works regardless of whether anything else (e.g. `init_app`'s HTTP client)
24/// has run yet.
25///
26/// What each event ends up carrying — and which `tracing` levels become events at
27/// all — is [`crate::telemetry`]'s call, not this function's.
28pub fn init_sentry(
29    dsn: String,
30    release: Option<String>,
31    dist: Option<String>,
32    environment: String,
33) {
34    if SENTRY_GUARD.get().is_some() {
35        return;
36    }
37    let _ = rustls::crypto::ring::default_provider().install_default();
38    // Anchor the process clock now, so `uptime` on every later event is measured from
39    // the earliest moment anything is reportable.
40    let _ = crate::telemetry::uptime();
41    let guard = sentry::init(sentry::ClientOptions {
42        dsn: dsn.parse().ok(),
43        release: release.map(std::borrow::Cow::from),
44        environment: Some(std::borrow::Cow::from(environment)),
45        // Every outgoing event goes through the same stamp — see `telemetry::enrich`
46        // for what it adds and why (`dist` among it: unlike release/environment it
47        // has no `ClientOptions` field, so this is the only place it can be set).
48        before_send: Some(
49            std::sync::Arc::new(move |event: sentry::protocol::Event<'static>| {
50                Some(crate::telemetry::enrich(
51                    event,
52                    dist.as_deref(),
53                    app_snapshot(),
54                ))
55            }) as sentry::BeforeCallback<sentry::protocol::Event<'static>>,
56        ),
57        // The SDK's own transport with the log tail bolted on: an ERROR envelope
58        // carries the last 64 KiB of `houndchat.log`, which is the only thing that
59        // turns the ~100-line breadcrumb ring into the full run-up to the failure.
60        // Wrapping the transport (rather than `before_send`) is what lets the file be
61        // read at SEND time and off the failing thread — see `telemetry::attach_log_tail`.
62        transport: Some(crate::telemetry::attach_log_tail(std::sync::Arc::new(
63            sentry::transports::DefaultTransportFactory,
64        ))),
65        ..Default::default()
66    });
67    // Race-tolerant like `init_app`: if another caller won, let ours flush+drop.
68    let _ = SENTRY_GUARD.set(guard);
69}

71/// The live client's view of itself, for stamping onto an outgoing event. `None`
72/// before [`init_app`](super::init_app) — the reporter comes up first, deliberately,
73/// so a crash during startup still reports.
74///
75/// Runs inside `before_send`, i.e. on the failing thread, so it must not block: both
76/// reads are `watch` borrows whose write side is a 1 Hz publish running no app code,
77/// and the report is a pure format of the borrowed snapshot.
78fn app_snapshot() -> Option<crate::telemetry::AppSnapshot> {
79    let client = CLIENT.get()?;
80    let in_call = client.in_call();
81    Some(crate::telemetry::AppSnapshot {
82        in_call,
83        audio: in_call.then(|| crate::telemetry::AudioSnapshot {
84            stage: client.audio_diagnosis().stage,
85            report: client.audio_debug_report(),
86        }),
87        video: in_call.then(|| crate::telemetry::VideoSnapshot {
88            stage: client.video_diagnosis().stage,
89            report: client.video_debug_report(),
90        }),
91    })
92}

94/// This launch's Sentry join key, for the Dart client to tag its own events with.
95///
96/// The two SDKs mint independent trace ids, so this is the ONLY thing that ties a
97/// Dart exception to the Rust events from the same process — without it, half of an
98/// incident is unreachable from the other half. Stable for the process lifetime.
99pub fn sentry_launch_id() -> String {
100    crate::telemetry::launch_id().to_owned()
101}
102
103/// What this process is — `user`, or the runner label its launcher set (see
104/// `telemetry::APP_ROLE_ENV`). Mirrored onto the Dart client so a nightly smoke run's
105/// failures don't inflate user-facing impact counts on either side.
106pub fn sentry_app_role() -> String {
107    crate::telemetry::app_role().to_owned()
108}
109
110/// Set (or, with `id: None`, clear) the Sentry-scoped user id for this process's
111/// Rust-side client — ID only, no username/email (PII). Mirrors the Dart-side
112/// `Sentry.configureScope` call (see `bootstrap.dart`'s auth-state listener);
113/// the two are separate Sentry clients (Dart events vs. Rust panic events), so
114/// both need setting. A no-op when Sentry was never initialised (`sentry-core`'s
115/// scope calls no-op with no bound client).
116pub fn sentry_set_user(id: Option<String>) {
117    sentry::configure_scope(|scope| {
118        scope.set_user(id.map(|id| sentry::User {
119            id: Some(id),
120            ..Default::default()
121        }));
122    });
