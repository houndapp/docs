1//! Log bridge: Rust `tracing` → Dart (plus the stderr mirror for CI smoke runs
2//! and the persistent log file's bridge commands). The file mechanics themselves
3//! live in [`crate::logfile`]; this is the bridge half.
4
5use std::sync::Arc;
6
7use hound_client::error::{HoundError, InternalCode, StateCode};
8
9use crate::frb_generated::StreamSink;

11/// The live Dart log sink. Swapped on each [`init_log`] (e.g. a hot restart hands us
12/// a fresh sink); the tracing layer — installed exactly once — reads this per event,
13/// so it always forwards to the current sink and never to a dead one.
14static LOG_SINK: std::sync::OnceLock<std::sync::Mutex<Option<Arc<StreamSink<String>>>>> =
15    std::sync::OnceLock::new();
16
17fn log_sink_slot() -> &'static std::sync::Mutex<Option<Arc<StreamSink<String>>>> {
18    LOG_SINK.get_or_init(|| std::sync::Mutex::new(None))
19}
20
21/// Install a `tracing` subscriber that forwards every formatted Rust log line to
22/// Dart over `sink`. WITHOUT this, the core's `tracing::{info,warn,error}!` (the
23/// call/transport/engine diagnostics) have no subscriber and are silently dropped.
24///
25/// **Must stay running**: a flutter_rust_bridge `StreamSink` is closed as soon as its
26/// function RETURNS, so a synchronous installer would tear the stream down and every
27/// later `sink.add` would no-op. So this is `async`
28/// and parks forever — the FRB stream stays open for the process lifetime while the
29/// layer forwards to the sink stored above.
30///
31/// Level filter comes from the `RUST_LOG` env if set, else a debug-on-the-call-path
32/// default. The subscriber is installed once (`try_init`); re-calls just re-point the
33/// sink (handles hot restart).
34pub async fn init_log(sink: StreamSink<String>) {
35    use tracing_subscriber::layer::{Layer, SubscriberExt};
36    use tracing_subscriber::util::SubscriberInitExt;
37    use tracing_subscriber::{EnvFilter, filter::LevelFilter, fmt};
38
39    *log_sink_slot().lock().unwrap() = Some(Arc::new(sink));
40
41    let filter = EnvFilter::try_from_default_env().unwrap_or_else(|_| {
42        // NB: this crate's tracing target is its LIB name (`rust_lib_houndchat`),
43        // not the package name `hound-frb`.
44        EnvFilter::new(
45            "info,hound_client=debug,hound_transport=debug,hound_media=debug,hound_engine=debug,rust_lib_houndchat=debug",
46        )
47    });
48    // Rust-side Sentry, separate from the Dart sink below: TWO Sentry clients
49    // exist (Dart's, initialised by `SentryFlutter.init`; Rust's, by
50    // `init_sentry`), each with its own breadcrumb ring, and each only
51    // attaches to crashes on ITS side (a Dart exception vs. a Rust panic).
52    // Which level becomes an event vs. a breadcrumb is [`telemetry::event_filter`]'s
53    // call; this cap only decides what reaches it at all.
54    //
55    // It MUST admit INFO. The `info!` lines are the app's narrative, and they are
56    // what a Rust event's breadcrumb trail is made of — capping at WARN (as this
57    // did) left every Rust event with an empty trail, saying what broke and never
58    // how it got there. DEBUG/TRACE stay out: too voluminous for a 100-entry ring,
59    // and they still reach the Dart sink above unaffected.
60    //
61    // A safe no-op when `init_sentry` was never called: `sentry-core`'s calls no-op
62    // with no bound client.
63    let sentry_layer = sentry::integrations::tracing::layer()
64        .event_filter(crate::telemetry::event_filter)
65        .with_filter(LevelFilter::INFO);
66    // One formatted line per event (buffered in the writer, flushed on Drop — fmt
67    // makes a fresh writer per event), forwarded to the live Dart sink.
68    // CI smoke runs assert on engine log lines in the captured stdio (the
69    // nightly launch gate greps for `session restore:`). Those must not depend
70    // on the Dart relay above — its delivery in a release-mode Flutter build is
71    // an embedder detail — so a runner-role process also writes every line to
72    // stderr, which provably reaches the harness (native SDK output does).
73    // Never on for real users: stderr is unobserved there, and `app_role` is
74    // fixed at process start (see telemetry::app_role).
75    let stderr_layer = mirrors_logs_to_stderr(crate::telemetry::app_role())
76        .then(|| fmt::layer().with_ansi(false).with_writer(std::io::stderr));
77    // The same lines again, into this instance's file under the OS log directory
78    // (see [`crate::logfile`] for the location, the per-launch rotation and the
79    // five-file retention). The Dart sink above dies with the app, so this is the
80    // only copy a user can attach to a report about last night's call.
81    //
82    // It sits UNDER the shared `filter` like every other layer, so `RUST_LOG` and
83    // the default policy govern the file exactly as they govern Dart and stderr —
84    // a second level policy here is a second thing to be wrong. `ansi(false)`
85    // because a file is not a terminal; timestamps are `fmt`'s default (UTC), which
86    // is also what the rotated file NAMES are stamped in.
87    //
88    // `None` when file logging could not come up: the other layers are untouched,
89    // and `state.problem` is warned about below — after `try_init`, since until
90    // then there is no subscriber to warn to.
91    let state = crate::logfile::init();
92    let file_layer = state
93        .writer()
94        .map(|writer| fmt::layer().with_ansi(false).with_writer(writer));
95    let _ = tracing_subscriber::registry()
96        .with(filter)
97        .with(
98            fmt::layer()
99                .with_ansi(false)
100                .with_writer(|| LogSink { buf: String::new() }),
101        )
102        .with(file_layer)
103        .with(stderr_layer)
104        .with(sentry_layer)
105        .try_init();
106
107    // Say where the file is, in the file and in every other sink: a log that does
108    // not name itself leaves "send me your logs" as a support conversation.
109    match state.path() {
110        Some(path) => tracing::info!("log file: {}", path.display()),
111        None => tracing::warn!("no log file this session"),
112    }
113    if let Some(problem) = &state.problem {
114        tracing::warn!("log file: {problem}");
115    }
116
117    // Keep the FRB stream open for the process lifetime.
118    std::future::pending::<()>().await;
119}

121/// Whether this process mirrors every engine log line to stderr. The CI launch
122/// smoke's pass/fail contract greps captured stdio for `session restore:`, so
123/// the marker's visibility must not hinge on the Dart-side relay.
124fn mirrors_logs_to_stderr(role: &str) -> bool {
125    role == "ci-smoke"
126}

128/// A `std::io::Write` that accumulates a formatted log event and pushes it to the live
129/// Dart `StreamSink` as one line when dropped (fmt drops the writer per event).
130struct LogSink {
131    buf: String,
132}
133
134impl std::io::Write for LogSink {
135    fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
136        self.buf.push_str(&String::from_utf8_lossy(bytes));
137        Ok(bytes.len())
138    }
139    fn flush(&mut self) -> std::io::Result<()> {
140        Ok(())
141    }
142}
143
144impl Drop for LogSink {
145    fn drop(&mut self) {
146        let line = self.buf.trim_end();
147        if line.is_empty() {
148            return;
149        }
150        // Clone the Arc out and release the lock BEFORE `add` so a log emitted from
151        // within `add` can't deadlock on the same non-reentrant mutex.
152        let sink = log_sink_slot()
153            .lock()
154            .ok()
155            .and_then(|g| g.as_ref().cloned());
156        let Some(sink) = sink else { return };
157        if sink.add(line.to_owned()).is_err() {
158            // Dart dropped the log stream (shutdown / hot restart before the new
159            // `init_log` lands). Drop our reference so later lines stop paying a
160            // doomed post — the lock is re-acquired only AFTER `add` returned, never
161            // held across it, so the deadlock rule above still holds. Only clear the
162            // slot if it still holds THIS sink: a concurrent `init_log` may already
163            // have installed a live one.
164            if let Ok(mut slot) = log_sink_slot().lock()
165                && slot.as_ref().is_some_and(|cur| Arc::ptr_eq(cur, &sink))
166            {
167                *slot = None;
168            }
169        }
170    }
171}

173/// Absolute path of the file this instance's logs are written to, or `None` when
174/// file logging could not be brought up (no usable OS log directory — see
175/// [`crate::logfile`]). Settings shows it so the path can be typed into a support
176/// thread or pasted into a file dialog; `None` is why the reveal button is hidden.
177///
178/// Only meaningful after [`init_log`] has run — which the Dart bootstrap does
179/// before anything can call this.
180pub fn log_file_path() -> Option<String> {
181    crate::logfile::path().map(|path| path.display().to_string())
182}
183
184/// Open the current log file with the OS default handler (Console/Notepad/whatever
185/// the desktop associates with `.log`) — the settings screen's "open logs" action.
186///
187/// Detached, like the browser-login launcher: the handler is a foreign GUI app and
188/// awaiting it would block an FRB worker for as long as the user leaves the window
189/// open. So the `Ok` means "handed to the OS", not "the user saw it" — a launcher
190/// that fails *after* forking is indistinguishable from success here, which is why
191/// [`log_file_path`] is shown alongside as the fallback.
192pub fn open_log_file() -> Result<(), HoundError> {
193    let path = crate::logfile::path().ok_or_else(|| {
194        HoundError::invalid_state(
195            StateCode::NotInitialised,
196            "there is no log file this session",
197        )
198    })?;
199    open::that_detached(path).map_err(|e| {
200        HoundError::internal(
201            InternalCode::Unclassified,
202            format!("could not open {}: {e}", path.display()),
203        )
204    })
205}
