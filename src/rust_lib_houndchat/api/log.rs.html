1//! Log bridge: Rust `tracing` → Dart (plus the stderr mirror for CI smoke runs
2//! and the persistent log file's bridge commands), and Dart → `tracing` back the
3//! other way. The file mechanics themselves live in [`crate::logfile`]; this is
4//! the bridge half.
5//!
6//! ## Why the return leg exists
7//!
8//! The rotating log file is written by the Rust `tracing` subscriber, and until
9//! [`log_from_dart`] there was no way for anything that happened in Dart to
10//! reach it. Dart-side failures went to `debugPrint`, which in a bundled app
11//! goes to a console nobody is reading — so the file a user is asked to send to
12//! support, and that Settings offers to open, was structurally incapable of
13//! containing half the client's failures. A silently broken image was the case
14//! that made that concrete.
15

22/// The live Dart log sink. Swapped on each [`init_log`] (e.g. a hot restart hands us
23/// a fresh sink); the tracing layer — installed exactly once — reads this per event,
24/// so it always forwards to the current sink and never to a dead one.
25static LOG_SINK: std::sync::OnceLock<std::sync::Mutex<Option<Arc<StreamSink<String>>>>> =
26    std::sync::OnceLock::new();
27
28fn log_sink_slot() -> &'static std::sync::Mutex<Option<Arc<StreamSink<String>>>> {
29    LOG_SINK.get_or_init(|| std::sync::Mutex::new(None))
30}
31
32/// Install a `tracing` subscriber that forwards every formatted Rust log line to
33/// Dart over `sink`. WITHOUT this, the core's `tracing::{info,warn,error}!` (the
34/// call/transport/engine diagnostics) have no subscriber and are silently dropped.
35///
36/// **Must stay running**: a flutter_rust_bridge `StreamSink` is closed as soon as its
37/// function RETURNS, so a synchronous installer would tear the stream down and every
38/// later `sink.add` would no-op. So this is `async`
39/// and parks forever — the FRB stream stays open for the process lifetime while the
40/// layer forwards to the sink stored above.
41///
42/// Level filter comes from the `RUST_LOG` env if set, else a debug-on-the-call-path
43/// default. The subscriber is installed once (`try_init`); re-calls just re-point the
44/// sink (handles hot restart).
45pub async fn init_log(sink: StreamSink<String>) {
46    use tracing_subscriber::layer::{Layer, SubscriberExt};
47    use tracing_subscriber::util::SubscriberInitExt;
48    use tracing_subscriber::{EnvFilter, filter::LevelFilter, fmt};
49
50    *log_sink_slot().lock().unwrap() = Some(Arc::new(sink));
51
52    let filter = EnvFilter::try_from_default_env().unwrap_or_else(|_| {
53        // NB: this crate's tracing target is its LIB name (`rust_lib_houndchat`),
54        // not the package name `hound-frb`.
55        EnvFilter::new(
56            "info,hound_client=debug,hound_transport=debug,hound_media=debug,hound_engine=debug,rust_lib_houndchat=debug",
57        )
58    });
59    // Rust-side Sentry, separate from the Dart sink below: TWO Sentry clients
60    // exist (Dart's, initialised by `SentryFlutter.init`; Rust's, by
61    // `init_sentry`), each with its own breadcrumb ring, and each only
62    // attaches to crashes on ITS side (a Dart exception vs. a Rust panic).
63    // Which level becomes an event vs. a breadcrumb is [`telemetry::event_filter`]'s
64    // call; this cap only decides what reaches it at all.
65    //
66    // It MUST admit INFO. The `info!` lines are the app's narrative, and they are
67    // what a Rust event's breadcrumb trail is made of — capping at WARN (as this
68    // did) left every Rust event with an empty trail, saying what broke and never
69    // how it got there. DEBUG/TRACE stay out: too voluminous for a 100-entry ring,
70    // and they still reach the Dart sink above unaffected.
71    //
72    // A safe no-op when `init_sentry` was never called: `sentry-core`'s calls no-op
73    // with no bound client.
74    let sentry_layer = sentry::integrations::tracing::layer()
75        .event_filter(crate::telemetry::event_filter)
76        .with_filter(LevelFilter::INFO);
77    // One formatted line per event (buffered in the writer, flushed on Drop — fmt
78    // makes a fresh writer per event), forwarded to the live Dart sink.
79    // CI smoke runs assert on engine log lines in the captured stdio (the
80    // nightly launch gate greps for `session restore:`). Those must not depend
81    // on the Dart relay above — its delivery in a release-mode Flutter build is
82    // an embedder detail — so a runner-role process also writes every line to
83    // stderr, which provably reaches the harness (native SDK output does).
84    // Never on for real users: stderr is unobserved there, and `app_role` is
85    // fixed at process start (see telemetry::app_role).
86    let stderr_layer = mirrors_logs_to_stderr(crate::telemetry::app_role())
87        .then(|| fmt::layer().with_ansi(false).with_writer(std::io::stderr));
88    // The same lines again, into this instance's file under the OS log directory
89    // (see [`crate::logfile`] for the location, the per-launch rotation and the
90    // five-file retention). The Dart sink above dies with the app, so this is the
91    // only copy a user can attach to a report about last night's call.
92    //
93    // It sits UNDER the shared `filter` like every other layer, so `RUST_LOG` and
94    // the default policy govern the file exactly as they govern Dart and stderr —
95    // a second level policy here is a second thing to be wrong. `ansi(false)`
96    // because a file is not a terminal; timestamps are `fmt`'s default (UTC), which
97    // is also what the rotated file NAMES are stamped in.
98    //
99    // `None` when file logging could not come up: the other layers are untouched,
100    // and `state.problem` is warned about below — after `try_init`, since until
101    // then there is no subscriber to warn to.
102    let state = crate::logfile::init();
103    let file_layer = state
104        .writer()
105        .map(|writer| fmt::layer().with_ansi(false).with_writer(writer));
106    let _ = tracing_subscriber::registry()
107        .with(filter)
108        .with(
109            fmt::layer()
110                .with_ansi(false)
111                .with_writer(|| LogSink { buf: String::new() }),
112        )
113        .with(file_layer)
114        .with(stderr_layer)
115        .with(sentry_layer)
116        .try_init();
117
118    // Say where the file is, in the file and in every other sink: a log that does
119    // not name itself leaves "send me your logs" as a support conversation.
120    match state.path() {
121        Some(path) => tracing::info!("log file: {}", path.display()),
122        None => tracing::warn!("no log file this session"),
123    }
124    if let Some(problem) = &state.problem {
125        tracing::warn!("log file: {problem}");
126    }
127
128    // Keep the FRB stream open for the process lifetime.
129    std::future::pending::<()>().await;
130}

132/// Whether this process mirrors every engine log line to stderr. The CI launch
133/// smoke's pass/fail contract greps captured stdio for `session restore:`, so
134/// the marker's visibility must not hinge on the Dart-side relay.
135fn mirrors_logs_to_stderr(role: &str) -> bool {
136    role == "ci-smoke"
137}

139/// A `std::io::Write` that accumulates a formatted log event and pushes it to the live
140/// Dart `StreamSink` as one line when dropped (fmt drops the writer per event).
141struct LogSink {
142    buf: String,
143}
144
145impl std::io::Write for LogSink {
146    fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
147        self.buf.push_str(&String::from_utf8_lossy(bytes));
148        Ok(bytes.len())
149    }
150    fn flush(&mut self) -> std::io::Result<()> {
151        Ok(())
152    }
153}
154
155impl Drop for LogSink {
156    fn drop(&mut self) {
157        let line = self.buf.trim_end();
158        if line.is_empty() {
159            return;
160        }
161        // Clone the Arc out and release the lock BEFORE `add` so a log emitted from
162        // within `add` can't deadlock on the same non-reentrant mutex.
163        let sink = log_sink_slot()
164            .lock()
165            .ok()
166            .and_then(|g| g.as_ref().cloned());
167        let Some(sink) = sink else { return };
168        if sink.add(line.to_owned()).is_err() {
169            // Dart dropped the log stream (shutdown / hot restart before the new
170            // `init_log` lands). Drop our reference so later lines stop paying a
171            // doomed post — the lock is re-acquired only AFTER `add` returned, never
172            // held across it, so the deadlock rule above still holds. Only clear the
173            // slot if it still holds THIS sink: a concurrent `init_log` may already
174            // have installed a live one.
175            if let Ok(mut slot) = log_sink_slot().lock()
176                && slot.as_ref().is_some_and(|cur| Arc::ptr_eq(cur, &sink))
177            {
178                *slot = None;
179            }
180        }
181    }
182}

184/// Severity for [`log_from_dart`]. A plain enum rather than a string so a typo
185/// cannot silently downgrade an error to a debug line nobody reads.
186pub enum DartLogLevel {
187    Error,
188    Warn,
189    Info,
190    Debug,
191}
192
193/// Emit a Dart-side diagnostic through the Rust `tracing` subscriber, so it
194/// lands wherever every other engine log does: the rotating log file, the Dart
195/// sink, and (in a CI smoke run) stderr.
196///
197/// `target` names the Dart origin — a widget, a provider, a subsystem — and is
198/// carried as a field rather than as the tracing target, which has to be a
199/// compile-time constant. Events are emitted under the `dart` target so the two
200/// halves of the log stay separable.
201///
202/// Deliberately synchronous: callers are error handlers and `errorBuilder`s,
203/// which have no business awaiting, and the work is a formatted write to a
204/// subscriber that is already in memory.
205///
206/// A call made before `init_log` has installed the subscriber is dropped by
207/// `tracing` itself, which is why the Dart side buffers the earliest lines and
208/// flushes them once the bridge is up.
209pub fn log_from_dart(level: DartLogLevel, target: String, message: String) {
210    match level {
211        DartLogLevel::Error => tracing::error!(target: "dart", origin = %target, "{message}"),
212        DartLogLevel::Warn => tracing::warn!(target: "dart", origin = %target, "{message}"),
213        DartLogLevel::Info => tracing::info!(target: "dart", origin = %target, "{message}"),
214        DartLogLevel::Debug => tracing::debug!(target: "dart", origin = %target, "{message}"),
215    }
216}

218/// Absolute path of the file this instance's logs are written to, or `None` when
219/// file logging could not be brought up (no usable OS log directory — see
220/// [`crate::logfile`]). Settings shows it so the path can be typed into a support
221/// thread or pasted into a file dialog; `None` is why the reveal button is hidden.
222///
223/// Only meaningful after [`init_log`] has run — which the Dart bootstrap does
224/// before anything can call this.
225pub fn log_file_path() -> Option<String> {
226    crate::logfile::path().map(|path| path.display().to_string())
227}
228
229/// Open the current log file with the OS default handler (Console/Notepad/whatever
230/// the desktop associates with `.log`) — the settings screen's "open logs" action.
231///
232/// Detached, like the browser-login launcher: the handler is a foreign GUI app and
233/// awaiting it would block an FRB worker for as long as the user leaves the window
234/// open. So the `Ok` means "handed to the OS", not "the user saw it" — a launcher
235/// that fails *after* forking is indistinguishable from success here, which is why
236/// [`log_file_path`] is shown alongside as the fallback.
237pub fn open_log_file() -> Result<(), HoundError> {
238    let path = crate::logfile::path().ok_or_else(|| {
239        HoundError::invalid_state(
240            StateCode::NotInitialised,
241            "there is no log file this session",
242        )
243    })?;
244    open::that_detached(path).map_err(|e| {
245        HoundError::internal(
246            InternalCode::Unclassified,
247            format!("could not open {}: {e}", path.display()),
248        )
249    })
250}
