7// Re-exported (not just `use`d) so the generated glue can name them as
8// `crate::api::{HoundError, AuthState, SpaceKind}` — the path FRB emits for a
9// mirrored type. The `#[frb(mirror(...))]` declarations below give codegen their
10// shape; these aliases give the runtime glue a resolvable path.
11//
12// The error-code registry travels with it: every category enum is mirrored too,
13// so the code crosses the bridge as an enum (one `i32`) instead of a string, and
14// the Dart side renders `PREFIX + (index + 1)` from it. The typed constructors
15// remain the only sanctioned way to build a `HoundError` here.

656// The shapes a message's link previews cross the bridge as. Mirrored (not
657// re-declared) for the same reason every other view-model is: a field added to
658// the wire contract reaches Dart for free.
659//
660// Two properties of these types are the UI's contract and are restated here
661// because a renderer has to be able to rely on them:
662//
663//   * every string is ALREADY plain text — entity-decoded, control-character
664//     stripped, length-capped by the server (`api::embed::build`), so Dart
665//     renders them directly with no escaping of its own; and
666//   * every URL in an `EmbedMedia` is relative to the API base and points at
667//     this deployment's media proxy, never at the linked site. The view layer
668//     resolves it (`features/channel/text/embeds/embed_media.dart`).
669//
670// Notably ABSENT: oEmbed's `html` player markup. It is captured server-side for
671// the debugger only and is not part of `LinkEmbed`, so there is no path by which
672// a renderer could be handed remote markup to execute.
673

789#[cfg(test)]
790mod tests {
791    use super::*;
792
793    // ── The mirror↔registry contract, enforced ───────────────────────────────
794    //
795    // Dart renders an error code as `PREFIX + (index + 1)`: the code string never
796    // crosses the bridge, only the variant's ORDINAL does. So a mirror that is
797    // shorter than its registry is not a cosmetic drift — the generated glue hits
798    // `unimplemented!()` and Dart throws a RangeError on an ordinal it has no name
799    // for. That is a crash on the error path, i.e. exactly when something has
800    // already gone wrong. It has happened once (four `MEDIA0xx` codes were added
801    // without mirroring them), which is why these tests exist.
802
803    /// Map every registry code to its mirror variant. The `match` is exhaustive
804    /// and the mirror variants are named, so this **fails to compile** if a code
805    /// is added to `hound_client::error::MediaCode` and not mirrored above (or
806    /// vice-versa) — the drift is caught at build time, not by remembering to run
807    /// a test. The cast then pins the ORDINAL, which is the thing Dart actually
808    /// reads.
809    fn media_mirror_ordinal(code: MediaCode) -> usize {
810        (match code {
811            MediaCode::ControlBaseScheme => _MediaCode::ControlBaseScheme,
812            MediaCode::ControlClosedBeforeWelcome => _MediaCode::ControlClosedBeforeWelcome,
813            MediaCode::BadRelayUrl => _MediaCode::BadRelayUrl,
814            MediaCode::BadSfuNodeId => _MediaCode::BadSfuNodeId,
815            MediaCode::IrohBind => _MediaCode::IrohBind,
816            MediaCode::RelayDial => _MediaCode::RelayDial,
817            MediaCode::ScreenSourceEnumeration => _MediaCode::ScreenSourceEnumeration,
818            MediaCode::ScreenThumbnail => _MediaCode::ScreenThumbnail,
819            MediaCode::MalformedScreenToken => _MediaCode::MalformedScreenToken,
820            MediaCode::UnknownScreenSourceKind => _MediaCode::UnknownScreenSourceKind,
821            MediaCode::ScreenPermissionDenied => _MediaCode::ScreenPermissionDenied,
822            MediaCode::ScreenCaptureOpen => _MediaCode::ScreenCaptureOpen,
823            MediaCode::ScreenCaptureThreadExited => _MediaCode::ScreenCaptureThreadExited,
824            MediaCode::ControlClosedBeforePublish => _MediaCode::ControlClosedBeforePublish,
825            MediaCode::NoVideoEncoder => _MediaCode::NoVideoEncoder,
826            MediaCode::NoSfuConfigured => _MediaCode::NoSfuConfigured,
827            MediaCode::WelcomeTimeout => _MediaCode::WelcomeTimeout,
828            MediaCode::RelayHandshake => _MediaCode::RelayHandshake,
829            MediaCode::DevicePermissionDenied => _MediaCode::DevicePermissionDenied,
830            MediaCode::Openh264Unsupported => _MediaCode::Openh264Unsupported,
831            MediaCode::Openh264Download => _MediaCode::Openh264Download,
832            MediaCode::Openh264Verify => _MediaCode::Openh264Verify,
833            MediaCode::Openh264Install => _MediaCode::Openh264Install,
834            MediaCode::ScreenCaptureNoFrames => _MediaCode::ScreenCaptureNoFrames,
835            MediaCode::MicrophoneOpen => _MediaCode::MicrophoneOpen,
836        }) as usize
837    }
838
839    /// Declaration order is the wire format: the mirror's ordinal must equal the
840    /// registry's index for every code, or Dart renders the wrong `MEDIA0xx`
841    /// string for a real failure. Catches a REORDER, which the exhaustive match
842    /// above cannot.
843    #[test]
844    fn media_code_mirror_preserves_registry_order() {
845        for (index, code) in MediaCode::ALL.iter().enumerate() {
846            assert_eq!(
847                media_mirror_ordinal(*code),
848                index,
849                "mirror ordinal for {code} does not match its registry index — Dart \
850                 would render the wrong code string; fix the order of `_MediaCode`"
851            );
852        }
853    }
854
855    /// Length parity for EVERY category, cheaply: each mirror's last variant must
856    /// sit at `ALL.len() - 1`. Appending to a registry without mirroring it (or
857    /// the reverse) moves that boundary and fails here — which is the whole bug
858    /// class, for the eight categories that have no exhaustive map of their own.
859    ///
860    /// Adding a code therefore means updating the name below too. That is the
861    /// point: it is one line, and it is the line that proves the mirror grew.
862    #[test]
863    fn every_mirror_is_the_same_length_as_its_registry() {
864        for (category, mirror_last, registry_len) in [
865            (
866                "NetCode",
867                _NetCode::RelayConnectionClosed as usize,
868                NetCode::ALL.len(),
869            ),
870            (
871                "AuthCode",
872                _AuthCode::SteamLoginDisabled as usize,
873                AuthCode::ALL.len(),
874            ),
875            (
876                "NotFoundCode",
877                _NotFoundCode::Resource as usize,
878                NotFoundCode::ALL.len(),
879            ),
880            (
881                "DecodeCode",
882                _DecodeCode::ImageTooLarge as usize,
883                DecodeCode::ALL.len(),
884            ),
885            (
886                "RealtimeCode",
887                _RealtimeCode::ControlWsDropped as usize,
888                RealtimeCode::ALL.len(),
889            ),
890            (
891                "MediaCode",
892                _MediaCode::MicrophoneOpen as usize,
893                MediaCode::ALL.len(),
894            ),
895            (
896                "StateCode",
897                _StateCode::CallOwnsMicrophone as usize,
898                StateCode::ALL.len(),
899            ),
900            (
901                "InternalCode",
902                _InternalCode::MicMonitorThread as usize,
903                InternalCode::ALL.len(),
904            ),
905        ] {
906            assert_eq!(
907                mirror_last + 1,
908                registry_len,
909                "{category}: the FRB mirror has {} variants but the registry has \
910                 {registry_len} — a code crossing the bridge with an unmirrored \
911                 ordinal panics in the generated glue and throws a RangeError in Dart",
912                mirror_last + 1
913            );
914        }
915    }
