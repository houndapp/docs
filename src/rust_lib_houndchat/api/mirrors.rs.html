8// `crate::api::{HoundError, AuthState, SpaceKind}` — the path FRB emits for a
9// mirrored type. The `#[frb(mirror(...))]` declarations below give codegen their
10// shape; these aliases give the runtime glue a resolvable path.

13// so the code crosses the bridge as an enum (one `i32`) instead of a string, and

15// remain the only sanctioned way to build a `HoundError` here.

554/// FRB mirror of [`hound_client::model::CommunitySignal`] — "refetch this
555/// community's roles / members".
556///
557/// An edge, not a snapshot: roles and member pages are paginated and rarely
558/// open, so they are fetched on demand and this stream says when a page went
559/// stale. It names the community only — the consumer re-reads whatever page it
560/// is on, which is right whatever the change was.
561/// **`seq` is why a repeated edit is a new value.** Dart's consumers are
562/// latest-wins and compare states for equality, so "roles changed in X" twice
563/// running would otherwise be one event: the second edit would refresh nothing.
564/// Nobody reads the number.
565#[frb(mirror(CommunitySignal))]
566pub enum _CommunitySignal {
567    RolesChanged {
568        seq: u64,
569        community_id: Uuid,
570    },
571    MembersChanged {
572        seq: u64,
573        community_id: Uuid,
574    },
575    /// Somebody was banned or unbanned. Its own signal rather than part of
576    /// `MembersChanged`: the ban list and the member list are different pages,
577    /// and a ban emits both.
578    BansChanged {
579        seq: u64,
580        community_id: Uuid,
581    },
582    /// Signals were LOST (a subscriber fell behind), so every fetched page is
583    /// of unknown freshness. Names no community because nobody can enumerate
584    /// what was dropped: treat it as naming all of them and re-read.
585    Resync {
586        seq: u64,
587    },
588}

1063    use super::*;

1065    // ── The mirror↔registry contract, enforced ───────────────────────────────
1066    //
1067    // Dart renders an error code as `PREFIX + (index + 1)`: the code string never
1068    // crosses the bridge, only the variant's ORDINAL does. So a mirror that is
1069    // shorter than its registry is not a cosmetic drift — the generated glue hits
1070    // `unimplemented!()` and Dart throws a RangeError on an ordinal it has no name
1071    // for. That is a crash on the error path, i.e. exactly when something has
1072    // already gone wrong. It has happened once (four `MEDIA0xx` codes were added
1073    // without mirroring them), which is why these tests exist.
1074
1075    /// Map every registry code to its mirror variant. The `match` is exhaustive
1076    /// and the mirror variants are named, so this **fails to compile** if a code
1077    /// is added to `hound_client::error::MediaCode` and not mirrored above (or
1078    /// vice-versa) — the drift is caught at build time, not by remembering to run
1079    /// a test. The cast then pins the ORDINAL, which is the thing Dart actually
1080    /// reads.
1081    fn media_mirror_ordinal(code: MediaCode) -> usize {
1082        (match code {
1083            MediaCode::ControlBaseScheme => _MediaCode::ControlBaseScheme,
1084            MediaCode::ControlClosedBeforeWelcome => _MediaCode::ControlClosedBeforeWelcome,
1085            MediaCode::BadRelayUrl => _MediaCode::BadRelayUrl,
1086            MediaCode::BadSfuNodeId => _MediaCode::BadSfuNodeId,
1087            MediaCode::IrohBind => _MediaCode::IrohBind,
1088            MediaCode::RelayDial => _MediaCode::RelayDial,
1089            MediaCode::ScreenSourceEnumeration => _MediaCode::ScreenSourceEnumeration,
1090            MediaCode::ScreenThumbnail => _MediaCode::ScreenThumbnail,
1091            MediaCode::MalformedScreenToken => _MediaCode::MalformedScreenToken,
1092            MediaCode::UnknownScreenSourceKind => _MediaCode::UnknownScreenSourceKind,
1093            MediaCode::ScreenPermissionDenied => _MediaCode::ScreenPermissionDenied,
1094            MediaCode::ScreenCaptureOpen => _MediaCode::ScreenCaptureOpen,
1095            MediaCode::ScreenCaptureThreadExited => _MediaCode::ScreenCaptureThreadExited,
1096            MediaCode::ControlClosedBeforePublish => _MediaCode::ControlClosedBeforePublish,
1097            MediaCode::NoVideoEncoder => _MediaCode::NoVideoEncoder,
1098            MediaCode::NoSfuConfigured => _MediaCode::NoSfuConfigured,
1099            MediaCode::WelcomeTimeout => _MediaCode::WelcomeTimeout,
1100            MediaCode::RelayHandshake => _MediaCode::RelayHandshake,
1101            MediaCode::DevicePermissionDenied => _MediaCode::DevicePermissionDenied,
1102            MediaCode::Openh264Unsupported => _MediaCode::Openh264Unsupported,
1103            MediaCode::Openh264Download => _MediaCode::Openh264Download,
1104            MediaCode::Openh264Verify => _MediaCode::Openh264Verify,
1105            MediaCode::Openh264Install => _MediaCode::Openh264Install,
1106            MediaCode::ScreenCaptureNoFrames => _MediaCode::ScreenCaptureNoFrames,
1107            MediaCode::MicrophoneOpen => _MediaCode::MicrophoneOpen,
1108        }) as usize
1109    }
1110
1111    /// Declaration order is the wire format: the mirror's ordinal must equal the
1112    /// registry's index for every code, or Dart renders the wrong `MEDIA0xx`
1113    /// string for a real failure. Catches a REORDER, which the exhaustive match
1114    /// above cannot.
1115    #[test]
1116    fn media_code_mirror_preserves_registry_order() {
1117        for (index, code) in MediaCode::ALL.iter().enumerate() {
1118            assert_eq!(
1119                media_mirror_ordinal(*code),
1120                index,
1121                "mirror ordinal for {code} does not match its registry index — Dart \
1122                 would render the wrong code string; fix the order of `_MediaCode`"
1123            );
1124        }
1125    }
1126
1127    /// Length parity for EVERY category, cheaply: each mirror's last variant must
1128    /// sit at `ALL.len() - 1`. Appending to a registry without mirroring it (or
1129    /// the reverse) moves that boundary and fails here — which is the whole bug
1130    /// class, for the eight categories that have no exhaustive map of their own.
1131    ///
1132    /// Adding a code therefore means updating the name below too. That is the
1133    /// point: it is one line, and it is the line that proves the mirror grew.
1134    #[test]
1135    fn every_mirror_is_the_same_length_as_its_registry() {

1145                AuthCode::ALL.len(),
1146            ),

1151            ),
1152            (
1153                "DecodeCode",
1154                _DecodeCode::ImageTooLarge as usize,
1155                DecodeCode::ALL.len(),
1156            ),
1157            (
1158                "RealtimeCode",
1159                _RealtimeCode::ControlWsDropped as usize,

1162            (
1163                "MediaCode",
