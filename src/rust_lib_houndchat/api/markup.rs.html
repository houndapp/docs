1//! Message-markup commands: parsing preview text, live source highlighting for
2//! the composer, plain-text rendering (the "Copy text" action), autolinking a
3//! reference, and fetching a single message.
4//!
5//! [`NamedId`] + its [`hound_markup::Resolve`] impl are the only logic here,
6//! and both are trivial: they exist to adapt three flat `Vec<NamedId>` lists
7//! (a shape FRB mirrors cleanly) onto `Resolve`'s by-id lookup, because the UI
8//! already holds the roster as lists, not as a map keyed for this purpose.
9
10use flutter_rust_bridge::frb;
11use uuid::Uuid;
12
13use hound_client::error::HoundError;
14use hound_markup::{Document, Reference, StyleRun};
15
16use super::{Message, client};
17
18/// An id the UI already has a display name for (a roster member, a space, a
19/// community), paired for [`markup_plain_text`]'s reference resolution.
20pub struct NamedId {
21    pub id: Uuid,
22    pub name: String,
23}
24
25/// Adapts three flat [`NamedId`] lists onto [`hound_markup::Resolve`].
26struct Names<'a> {
27    users: &'a [NamedId],
28    spaces: &'a [NamedId],
29    communities: &'a [NamedId],
30}
31
32fn find(list: &[NamedId], id: Uuid) -> Option<String> {
33    list.iter().find(|n| n.id == id).map(|n| n.name.clone())
34}
35
36impl hound_markup::Resolve for Names<'_> {
37    fn user(&self, id: Uuid) -> Option<String> {
38        find(self.users, id)
39    }
40    fn space(&self, id: Uuid) -> Option<String> {
41        find(self.spaces, id)
42    }
43    fn community(&self, id: Uuid) -> Option<String> {
44        find(self.communities, id)
45    }
46}

48/// Parse `source` as hound markup. Backs the composer's live preview and
49/// markup-focused tests; the real ingestion path never calls this directly —
50/// it goes through `hound_client::model::Message::body`, parsed once at fetch
51/// time.
52///
53/// Not `async`: parsing is CPU-bound and synchronous by contract (see
54/// `hound_markup::parse`), so this runs on an FRB worker thread rather than
55/// occupying the async executor.
56#[frb(sync)]
57pub fn parse_markup(source: String) -> Document {
58    hound_markup::parse(&source)
59}
60
61/// Style runs over `text` for the composer's live source highlighting: which
62/// UTF-16 spans of the text being typed are bold/quoted/code/…, and which are
63/// the syntax markers to mute. Offsets are UTF-16 code units — what a Dart
64/// `String` (and therefore a `TextEditingValue`) indexes by.
65///
66/// Not `async`, like its siblings, and for a sharper reason: it runs on every
67/// keystroke inside `buildTextSpan`, where an `await` would paint one frame of
68/// unstyled text before catching up.
69#[frb(sync)]
70pub fn highlight_markup(text: String) -> Vec<StyleRun> {
71    hound_markup::highlight(&text)
72}
73
74/// Render `doc` as plain text, resolving references against the roster the UI
75/// already holds (`users`/`spaces`/`communities`). Backs the "Copy text"
76/// action; a reference the caller has no name for falls back to `@unknown
77/// user` etc. (see `hound_markup::plain_text`).
78#[frb(sync)]
79pub fn markup_plain_text(
80    doc: Document,
81    users: Vec<NamedId>,
82    spaces: Vec<NamedId>,
83    communities: Vec<NamedId>,
84) -> String {
85    let names = Names {
86        users: &users,
87        spaces: &spaces,
88        communities: &communities,
89    };
90    hound_markup::plain_text(&doc, &names)
91}

93/// The composer's canonical autolink form for `reference`:
94/// `<hound:<kind>/<uuid>>`, the text inserted when a mention/autocomplete
95/// picks a target.
96#[frb(sync)]
97pub fn reference_autolink(reference: Reference) -> String {
98    reference.autolink()
99}

101/// `GET /message/{id}` — fetch a single message by id, e.g. to resolve a
102/// `<hound:message/…>` reference preview.
103pub async fn fetch_message(message_id: Uuid) -> Result<Message, HoundError> {
104    client()?.fetch_message(message_id).await
105}
