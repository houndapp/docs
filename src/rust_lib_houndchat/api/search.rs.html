1//! Message search: running one, parsing what is being typed, and the
2//! detached-history commands a jump to a hit drives.
3//!
4//! ## Why this module re-declares half of `hound-search`
5//!
6//! Everything on this surface is mirrored by default (see the projection rule
7//! in [`crate::api`]) — but three of the query language's enums are
8//! `#[non_exhaustive]` on purpose ([`hound_search::TokenKind`],
9//! [`hound_search::DiagnosticKind`], [`hound_markup::ContentKind`]: they track
10//! the language, not a wire contract), and a mirror's generated `match` carries
11//! no wildcard arm, so mirroring one stops compiling the moment a variant is
12//! added upstream. They therefore cross as the DTOs below, whose `From` impls
13//! fold an unrecognised variant onto an explicit `Unknown` — the Dart side gets
14//! a value it can render as "something this build doesn't know" instead of a
15//! build that no longer compiles.
16//!
17//! [`Query`](hound_search::Query) and [`Parsed`](hound_search::Parsed) follow
18//! them because they *contain* those enums, and [`SearchToken`] /
19//! [`SearchDiagnostic`] additionally flatten a `Range<u32>` into `start`/`end`
20//! (the same shape [`hound_markup::StyleRun`] already crosses as). Everything
21//! that mirrors cleanly still does: [`Term`], [`DateSpec`], and the whole result
22//! surface ([`SearchPage`], [`SearchHit`], [`SearchScope`], [`SearchSort`]) are
23//! declared in [`crate::api::mirrors`].
24//!
25//! Offsets in [`SearchToken`]/[`SearchDiagnostic`] are UTF-16 code units — what
26//! a Dart `String` and a `TextEditingValue` index by.
27

40/// One coarse content facet of a message — the `has:` vocabulary.
41///
42/// A local copy of [`hound_markup::ContentKind`] (see the module docs).
43#[derive(Clone, Copy, Debug, PartialEq, Eq)]
44pub enum ContentKind {
45    Link,
46    Image,
47    Video,
48    Audio,
49    File,
50    Embed,
51    Code,
52    Spoiler,
53    /// A kind the engine knows and this bridge build does not. Renders as a
54    /// generic facet; never produced by a matching client/engine pair.
55    Unknown,
56}
57
58impl From<hound_markup::ContentKind> for ContentKind {
59    fn from(kind: hound_markup::ContentKind) -> Self {
60        use hound_markup::ContentKind as K;
61        match kind {
62            K::Link => Self::Link,
63            K::Image => Self::Image,
64            K::Video => Self::Video,
65            K::Audio => Self::Audio,
66            K::File => Self::File,
67            K::Embed => Self::Embed,
68            K::Code => Self::Code,
69            K::Spoiler => Self::Spoiler,
70            _ => Self::Unknown,
71        }
72    }
73}
74
75impl ContentKind {
76    /// `None` for [`ContentKind::Unknown`]: a facet this build cannot name is
77    /// dropped from the outgoing query rather than guessed at.
78    fn to_core(self) -> Option<hound_markup::ContentKind> {
79        use hound_markup::ContentKind as K;
80        Some(match self {
81            Self::Link => K::Link,
82            Self::Image => K::Image,
83            Self::Video => K::Video,
84            Self::Audio => K::Audio,
85            Self::File => K::File,
86            Self::Embed => K::Embed,
87            Self::Code => K::Code,
88            Self::Spoiler => K::Spoiler,
89            Self::Unknown => return None,
90        })
91    }
92}

94/// A parsed search: the structured query, plus everything the field needs to
95/// render what it just parsed.
96///
97/// Mirrors [`hound_search::Parsed`]'s three parts. The `query` is runnable no
98/// matter what the diagnostics say — they describe what was *dropped*, not a
99/// failure.
100pub struct ParsedSearchQuery {
101    pub query: SearchQuery,
102    /// What was dropped, and where. Advisory; shown under the field.
103    pub diagnostics: Vec<SearchDiagnostic>,
104    /// Highlight spans over the input, ascending and non-overlapping. They do
105    /// not tile the input — whitespace produces no token.
106    pub tokens: Vec<SearchToken>,
107}
108
109/// The structured search — the only part of a parse that crosses to the
110/// backend. Mirrors [`hound_search::Query`].
111///
112/// Identity filters (`from`/`inSpaces`/`mentions`) are OR within a field;
113/// everything else is AND. A negated term is a NOT.
114///
115/// The two predicates the UI wants are one line of Dart each and are therefore
116/// not bridged: "constrains nothing" is every field empty/null, and "can be
117/// ranked by relevance" (which gates the Relevant sort) is
118/// `terms.any((t) => !t.negated)` — `hound_search::Query::has_positive_text`'s
119/// exact rule.
120pub struct SearchQuery {
121    pub terms: Vec<Term>,
122    /// Message authors. Empty = any author.
123    pub from: Vec<Uuid>,
124    /// Spaces to search, narrowing the request's scope (never widening it).
125    pub in_spaces: Vec<Uuid>,
126    /// Users the message must mention.
127    pub mentions: Vec<Uuid>,
128    /// Content facets the message must have, all of them.
129    pub has: Vec<ContentKind>,
130    pub before: Option<DateSpec>,
131    pub after: Option<DateSpec>,
132    pub during: Option<DateSpec>,
133    /// `is:edited`. `null` = either.
134    pub edited: Option<bool>,
135}
136
137impl From<hound_search::Query> for SearchQuery {
138    fn from(q: hound_search::Query) -> Self {
139        Self {
140            terms: q.terms,
141            from: q.from,
142            in_spaces: q.in_spaces,
143            mentions: q.mentions,
144            has: q.has.into_iter().map(ContentKind::from).collect(),
145            before: q.before,
146            after: q.after,
147            during: q.during,
148            edited: q.edited,
149        }
150    }
151}
152
153impl From<SearchQuery> for hound_search::Query {
154    fn from(q: SearchQuery) -> Self {
155        Self {
156            terms: q.terms,
157            from: q.from,
158            in_spaces: q.in_spaces,
159            mentions: q.mentions,
160            has: q.has.into_iter().filter_map(ContentKind::to_core).collect(),
161            before: q.before,
162            after: q.after,
163            during: q.during,
164            edited: q.edited,
165        }
166    }
167}

169/// One highlightable span of query source. `start`/`end` are half-open UTF-16
170/// code-unit offsets (a flattened [`hound_search::Token::range`]).
171pub struct SearchToken {
172    pub start: u32,
173    pub end: u32,
174    pub kind: SearchTokenKind,
175}
176
177/// What a [`SearchToken`]'s span is. Local copy of
178/// [`hound_search::TokenKind`] (see the module docs).
179#[derive(Clone, Copy, Debug, PartialEq, Eq)]
180pub enum SearchTokenKind {
181    /// A bare word term.
182    Word,
183    /// A quoted phrase term, quotes included.
184    Phrase,
185    /// The leading `-` of a negated term.
186    Negation,
187    /// A recognised filter key **and its `:`**.
188    FilterKey,
189    /// A filter's value, quotes included when it was quoted.
190    FilterValue,
191    /// A filter value that is a `hound:<kind>/<uuid>` reference — the resolved
192    /// pill form. Replaces `FilterValue` for that span.
193    Reference,
194    /// A span kind this bridge build does not know; paint it as plain text.
195    Unknown,
196}
197
198impl From<&hound_search::Token> for SearchToken {
199    fn from(token: &hound_search::Token) -> Self {
200        use hound_search::TokenKind as K;
201        Self {
202            start: token.range.start,
203            end: token.range.end,
204            kind: match token.kind {
205                K::Word => SearchTokenKind::Word,
206                K::Phrase => SearchTokenKind::Phrase,
207                K::Negation => SearchTokenKind::Negation,
208                K::FilterKey => SearchTokenKind::FilterKey,
209                K::FilterValue => SearchTokenKind::FilterValue,
210                K::Reference => SearchTokenKind::Reference,
211                _ => SearchTokenKind::Unknown,
212            },
213        }
214    }
215}

217/// One dropped or overridden part of the query, positioned in the source.
218/// `start`/`end` are UTF-16 code units, spanning the whole chunk.
219pub struct SearchDiagnostic {
220    pub start: u32,
221    pub end: u32,
222    pub kind: SearchDiagnosticKind,
223}
224
225/// Why a chunk was dropped. Local copy of [`hound_search::DiagnosticKind`]
226/// (see the module docs); `key`/`value` are the user's own text.
227pub enum SearchDiagnosticKind {
228    /// `from:` with nothing after the colon.
229    EmptyValue { key: String },
230    /// A value outside a closed vocabulary — `has:` and `is:`.
231    UnknownValue { key: String, value: String },
232    /// An identity filter whose value is a typed name the UI never resolved to
233    /// an id. Names never reach the wire, so the filter is dropped.
234    UnresolvedReference { key: String, value: String },
235    /// A `hound:` reference of the wrong kind — `in:<hound:user/…>`.
236    WrongReferenceKind {
237        key: String,
238        expected: String,
239        found: String,
240    },
241    /// A date filter whose value is none of `YYYY-MM-DD`, `YYYY-MM`, `YYYY`,
242    /// `today`, `yesterday`.
243    InvalidDate { key: String, value: String },
244    /// A negated filter. Only text terms negate in v1.
245    NegationNotSupported { key: String },
246    /// An earlier single-valued filter overridden by a later one. The span is
247    /// the *earlier* chunk's.
248    Superseded { key: String },
249    /// A diagnostic this bridge build cannot name. Render the span as "ignored"
250    /// with no explanation rather than dropping it silently.
251    Unknown,
252}
253
254impl From<&hound_search::Diagnostic> for SearchDiagnostic {
255    fn from(d: &hound_search::Diagnostic) -> Self {
256        use hound_search::DiagnosticKind as K;
257        let kind = match &d.kind {
258            K::EmptyValue { key } => SearchDiagnosticKind::EmptyValue { key: key.clone() },
259            K::UnknownValue { key, value } => SearchDiagnosticKind::UnknownValue {
260                key: key.clone(),
261                value: value.clone(),
262            },
263            K::UnresolvedReference { key, value } => SearchDiagnosticKind::UnresolvedReference {
264                key: key.clone(),
265                value: value.clone(),
266            },
267            K::WrongReferenceKind {
268                key,
269                expected,
270                found,
271            } => SearchDiagnosticKind::WrongReferenceKind {
272                key: key.clone(),
273                expected: expected.clone(),
274                found: found.clone(),
275            },
276            K::InvalidDate { key, value } => SearchDiagnosticKind::InvalidDate {
277                key: key.clone(),
278                value: value.clone(),
279            },
280            K::NegationNotSupported { key } => {
281                SearchDiagnosticKind::NegationNotSupported { key: key.clone() }
282            }
283            K::Superseded { key } => SearchDiagnosticKind::Superseded { key: key.clone() },
284            _ => SearchDiagnosticKind::Unknown,
285        };
286        Self {
287            start: d.range.start,
288            end: d.range.end,
289            kind,
290        }
291    }
292}

294/// One `POST /search` call.
295///
296/// A local DTO rather than a mirror of
297/// [`hound_client::model::SearchRequest`](CoreSearchRequest) for one reason:
298/// that type carries [`hound_search::Query`], which cannot mirror (see the
299/// module docs). The fields are otherwise identical.
300pub struct SearchRequest {
301    pub scope: SearchScope,
302    pub query: SearchQuery,
303    pub sort: SearchSort,
304    /// Opaque keyset cursor from the previous page's `nextCursor`; `null` asks
305    /// for the first page. Never inspected client-side.
306    pub cursor: Option<String>,
307    /// Hits per page, 1..=50. Outside that the backend answers 400.
308    pub limit: u32,
309    /// Minutes east of UTC (`DateTime.now().timeZoneOffset.inMinutes`).
310    /// `today`/`before:`/`after:`/`during:` resolve to the READER's midnight,
311    /// so the offset travels with the request.
312    pub utc_offset_minutes: i32,
313}
314
315impl From<SearchRequest> for CoreSearchRequest {
316    fn from(r: SearchRequest) -> Self {
317        Self {
318            scope: r.scope,
319            query: r.query.into(),
320            sort: r.sort,
321            cursor: r.cursor,
322            limit: r.limit,
323            utc_offset_minutes: r.utc_offset_minutes,
324        }
325    }
326}

330/// Parse one line of search input.
331///
332/// **Runs per keystroke** — the field re-parses on every edit to repaint its
333/// highlighting, so this is `#[frb(sync)]` like `highlight_markup`: an `await`
334/// here would paint one frame of unstyled text before catching up.
335///
336/// Total, and that is the contract: half-typed input is the normal case, not an
337/// error case. Every input yields a runnable
338/// [`query`](ParsedSearchQuery::query); anything unreadable becomes a
339/// [`diagnostic`](ParsedSearchQuery::diagnostics), never a throw.
340#[frb(sync)]
341pub fn parse_search_query(input: String) -> ParsedSearchQuery {
342    let parsed = hound_search::parse(&input);
343    ParsedSearchQuery {
344        query: parsed.query.into(),
345        diagnostics: parsed
346            .diagnostics
347            .iter()
348            .map(SearchDiagnostic::from)
349            .collect(),
350        tokens: parsed.tokens.iter().map(SearchToken::from).collect(),
351    }
352}

354/// The filter keys the language recognises, in the order the helper popover
355/// lists them (`from`, `in`, `mentions`, `has`, `before`, `after`, `during`,
356/// `is`). Anything else before a `:` is ordinary text.
357#[frb(sync)]
358pub fn search_filter_keys() -> Vec<String> {
359    hound_search::FILTER_KEYS
360        .iter()
361        .map(|k| (*k).to_owned())
362        .collect()
363}

365/// The accepted `has:` values, for the value picker.
366#[frb(sync)]
367pub fn search_has_values() -> Vec<String> {
368    hound_search::has_values()
369        .iter()
370        .map(|v| (*v).to_owned())
371        .collect()
372}

374/// `POST /search` — one page of message hits.
375///
376/// Page by handing [`SearchPage::next_cursor`] back as
377/// [`SearchRequest::cursor`]; `null` back means the end of the results. The
378/// authors are joined onto their hits engine-side and merged into the roster on
379/// the way past, so a hit from a community this client never loaded still
380/// renders a name.
381pub async fn search(request: SearchRequest) -> Result<SearchPage, HoundError> {
382    client()?.search(&request.into()).await
383}

385/// Load a window of a space's history centred on `message_id` and mark that
386/// space's message list **detached**: realtime appends stop landing in it until
387/// [`resume_live`] (an edit to a message the window already holds still does).
388///
389/// The jump target when a search hit is not in the loaded window. Observe the
390/// state with [`watch_history_detached`]; the state machine is documented on
391/// `hound_client::app::search`.
392pub async fn open_space_around(space_id: Uuid, message_id: Uuid) -> Result<(), HoundError> {
393    client()?.open_space_around(space_id, message_id).await
394}
395
396/// Leave a detached window: reload page 0 and resume applying realtime events
397/// to the space. The "jump to present" action; harmless on a space that was
398/// never detached.
399pub async fn resume_live(space_id: Uuid) -> Result<(), HoundError> {
400    client()?.resume_live(space_id).await
401}
402
403/// Stream of a space's detached flag: `true` while its list shows an
404/// around-window rather than the live tail. Drives the "viewing older messages"
405/// banner.
406///
407/// Emits the current value immediately (`false` for a space never opened), then
408/// every change — the same forward-loop contract as `watch_messages`.
409pub async fn watch_history_detached(
410    space_id: Uuid,
411    sink: StreamSink<bool>,
412) -> Result<(), HoundError> {
413    let mut rx = client()?.watch_history_detached(space_id).await;
414    if sink.add(*rx.borrow()).is_err() {
415        tracing::debug!("watch_history_detached: dart sink closed, exiting");
416        return Ok(());
417    }
418    while rx.changed().await.is_ok() {
419        if sink.add(*rx.borrow()).is_err() {
420            tracing::debug!("watch_history_detached: dart sink closed, exiting");
421            break;
422        }
423    }
424    Ok(())
425}

428mod tests {
429    use super::*;

431    /// The DTO round trip the bridge depends on: what the parser produced is
432    /// what the backend receives, through two conversions and back.
433    #[test]
434    fn a_parsed_query_round_trips_through_the_bridge_dto() {
435        let source = "grant -spam \"exact phrase\" has:code has:image \
436                      from:<hound:user/018c5b6e-0000-7000-0000-000000000003> \
437                      during:2026-09 is:edited";
438        let core = hound_search::parse(source).query;
439        let dto = SearchQuery::from(core.clone());
440        assert_eq!(hound_search::Query::from(dto), core);
441    }
442
443    /// Tokens and diagnostics flatten their `Range<u32>` into the `start`/`end`
444    /// pair Dart indexes by, and keep the parser's order.
445    #[test]
446    fn tokens_and_diagnostics_flatten_their_ranges() {
447        let parsed = parse_search_query("from:erin -word".to_owned());
448
449        assert!(!parsed.tokens.is_empty());
450        assert!(
451            parsed
452                .tokens
453                .windows(2)
454                .all(|w| w[0].end <= w[1].start && w[0].start <= w[0].end),
455            "tokens stay ascending and non-overlapping across the boundary"
456        );
457        assert!(matches!(parsed.tokens[0].kind, SearchTokenKind::FilterKey));
458
459        // `from:erin` is a name the UI never resolved — dropped, with a
460        // diagnostic pointing at the chunk that produced it.
461        assert_eq!(parsed.diagnostics.len(), 1);
462        let d = &parsed.diagnostics[0];
463        assert!(d.start < d.end);
464        assert!(matches!(
465            d.kind,
466            SearchDiagnosticKind::UnresolvedReference { .. }
467        ));
468        assert!(parsed.query.from.is_empty());
469
470        // The negated term survives, and disables the relevance sort on its own.
471        assert_eq!(parsed.query.terms.len(), 1);
472        assert!(parsed.query.terms[0].negated);
473        assert!(
474            !parsed.query.terms.iter().any(|t| !t.negated),
475            "a lone negated term cannot be ranked by relevance"
476        );
477    }
478
479    /// Every `has:` value the language accepts survives the enum copy — the
480    /// guard against this DTO drifting from `hound_markup::ContentKind`.
481    #[test]
482    fn every_content_kind_round_trips() {
483        for value in hound_search::has_values() {
484            let query = parse_search_query(format!("has:{value}")).query;
485            assert_eq!(query.has.len(), 1, "has:{value} parsed to one facet");
486            assert_ne!(
487                query.has[0],
488                ContentKind::Unknown,
489                "has:{value} is a facet this bridge build cannot name"
490            );
491            let core = hound_search::Query::from(query);
492            assert_eq!(core.has.len(), 1, "has:{value} survives the way back");
493        }
494    }

496    /// The autocomplete lists are the crate's, not a hand-kept copy.
497    #[test]
498    fn autocomplete_lists_come_from_the_language() {

501    }
