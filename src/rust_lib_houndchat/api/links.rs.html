1//! Links: opening one in the system browser, and unfurling one through the
2//! server's embed pipeline for the debugger.
3//!
4//! Both commands are about URLs a *stranger* wrote — a message body is
5//! attacker-controlled text — so both validate the scheme before doing anything
6//! with it. That check is here rather than in the UI because it must hold for
7//! every caller, and because the consequence of missing it is
8//! platform-specific and severe: [`open::that_detached`] resolves to
9//! `ShellExecute` on Windows and `open(1)` on macOS, either of which will
10//! cheerfully act on `file:`, `smb:`, or a registered custom scheme.
11
12use hound_client::error::{HoundError, InternalCode, StateCode};
13
14use super::{EmbedDebugReport, client};
15
16/// Open `url` in the user's browser.
17///
18/// **`http` and `https` only.** Anything else is refused rather than passed to
19/// the platform launcher — see the module docs for why that is not a formality.
20///
21/// Detached, like the browser-login launcher: the browser is a foreign GUI app
22/// and awaiting it would block an FRB worker for as long as the window stays
23/// open. So `Ok` means "handed to the OS", not "the user saw it".
24pub fn open_external_url(url: String) -> Result<(), HoundError> {
25    let parsed = url::Url::parse(&url).map_err(|_| {
26        HoundError::invalid_state(StateCode::InvalidArgument, format!("`{url}` is not a URL"))
27    })?;
28    if !matches!(parsed.scheme(), "http" | "https") {
29        return Err(HoundError::invalid_state(
30            StateCode::InvalidArgument,
31            format!("refusing to open a `{}` link", parsed.scheme()),
32        ));
33    }
34    open::that_detached(parsed.as_str()).map_err(|e| {
35        HoundError::internal(
36            InternalCode::Unclassified,
37            format!("could not open the browser: {e}"),
38        )
39    })
40}

42/// Unfurl `url` through the backend and return the full trace: every HTTP hop
43/// (with the address the guard resolved and allowed), every Open Graph, Twitter
44/// Card and oEmbed key seen, the warnings, and the embed the precedence rules
45/// produced.
46///
47/// The embed debugger in developer settings. It runs **server-side**,
48/// deliberately: the client has no business opening connections to a URL
49/// somebody typed into a text field, and the question the tool answers is what
50/// *the server* makes of a link — which is what a message would actually show.
51///
52/// The server bypasses its unfurl cache for this and rate limits it per user, so
53/// an `HTTP429` here is the limiter working, not a fault.
54pub async fn debug_embed(url: String) -> Result<EmbedDebugReport, HoundError> {
55    client()?.debug_embed(&url).await
