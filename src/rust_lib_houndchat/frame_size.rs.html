1//! Event-driven decoded-geometry reporting for the render path.
2//!
3//! A tile's native texture is allocated at the DECODED frame size, so Flutter has
4//! to learn a resolution change (an inbound simulcast layer switch, a shared-window
5//! resize, a camera hot-swap) to rebuild it. Deriving that from the 1 Hz
6//! [`StatsSnapshot`](hound_client::stats::StatsSnapshot) cadence means up to a
7//! second of the sharp new layer being scaled into the old texture — visibly soft.
8//!
9//! This is the event side of that: the decode→render sink ([`crate::api`]'s
10//! `RenderVideoSink`) sees every frame's dimensions, so it reports them here and
11//! the *change detector* below wakes `watch_frame_sizes` the instant a stream's
12//! geometry moves.
13//!
14//! ## Division of labour with the stats snapshot
15//! The snapshot still decides **membership** (which stream keys exist, and the
16//! self-capture / self-screen geometry it alone knows); this map only decides
17//! **freshness** — it overrides the geometry of keys the snapshot already emits.
18//! That keeps the DTO's contract intact: no phantom key can survive a peer leaving
19//! or a call ending, because a key the snapshot has dropped is never emitted no
20//! matter what this map still remembers.
21//!
22//! ## Cost
23//! One mutex + one borrowed-key hash lookup per decoded frame on the decode task
24//! (a hit in steady state: compare, unlock, no allocation, no wakeup). Only a real
25//! change allocates (the key) and bumps the watch.
26
27use std::collections::HashMap;
28use std::sync::Mutex;
29use std::sync::OnceLock;
30
31use tokio::sync::watch;
32
33/// The latest reported geometry per render stream key, plus a change tick the
34/// stream forwarder awaits.
35///
36/// The tick is a monotonically bumped counter rather than the map itself: the
37/// forwarder re-reads the map (merged with the current stats snapshot) on each
38/// wake, so the channel only needs to say "something moved" — and a `watch` then
39/// coalesces a burst of changes into one wake.
40pub struct FrameGeometry {
41    sizes: Mutex<HashMap<String, (u32, u32)>>,
42    tick: watch::Sender<u64>,
43}
44
45impl FrameGeometry {
46    fn new() -> Self {
47        Self {
48            sizes: Mutex::new(HashMap::new()),
49            tick: watch::channel(0).0,
50        }
51    }
52
53    /// Record `stream_key`'s current decoded geometry. Returns `true` (and bumps
54    /// the change tick) only if it DIFFERS from the last report — the hot path is
55    /// the unchanged case. Degenerate (zero) extents are ignored so a malformed
56    /// frame can't churn a texture rebuild.
57    pub fn record(&self, stream_key: &str, w: u32, h: u32) -> bool {
58        if w == 0 || h == 0 {
59            return false;
60        }
61        {
62            let mut map = self.sizes.lock().unwrap_or_else(|e| e.into_inner());
63            // Borrowed-key lookup first: the steady state is a hit, and `entry()`
64            // would allocate a fresh `String` per frame just to find it.
65            match map.get_mut(stream_key) {
66                Some(cur) if *cur == (w, h) => return false,
67                Some(cur) => *cur = (w, h),
68                None => {
69                    map.insert(stream_key.to_owned(), (w, h));
70                }
71            }
72        }
73        self.tick.send_modify(|v| *v = v.wrapping_add(1));
74        true
75    }
76
77    /// Forget a stream that has ended for good, so the map stays bounded across a
78    /// long session of peers coming and going. Not load-bearing for correctness —
79    /// the snapshot decides membership — so a missed call only wastes an entry.
80    pub fn forget(&self, stream_key: &str) {
81        self.sizes
82            .lock()
83            .unwrap_or_else(|e| e.into_inner())
84            .remove(stream_key);
85    }
86
87    /// Replace the geometry of every entry we have a fresher report for, in place.
88    /// Keys absent from `streams` are NOT added (the snapshot owns membership).
89    pub fn overlay(&self, streams: &mut [(String, u32, u32)]) {
90        let map = self.sizes.lock().unwrap_or_else(|e| e.into_inner());
91        for (key, w, h) in streams.iter_mut() {
92            if let Some(&(rw, rh)) = map.get(key.as_str()) {
93                (*w, *h) = (rw, rh);
94            }
95        }
96    }
97
98    /// A receiver that fires whenever any stream's geometry changes.
99    pub fn subscribe(&self) -> watch::Receiver<u64> {
100        self.tick.subscribe()
101    }
102}
103
104/// The process-wide map. A global for the same reason the render mailboxes are:
105/// the producer is a decode task deep in the call pipeline and the consumer is a
106/// Dart stream forwarder, with no shared owner between them.
107pub fn geometry() -> &'static FrameGeometry {
108    static G: OnceLock<FrameGeometry> = OnceLock::new();
109    G.get_or_init(FrameGeometry::new)
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115
116    /// The whole point of the map: a report is only an EVENT when the geometry
117    /// actually moved, so the per-frame hot path never wakes the Dart stream.
118    #[test]
119    fn only_a_changed_geometry_is_an_event() {
120        let g = FrameGeometry::new();
121        let mut rx = g.subscribe();
122
123        assert!(g.record("cam:7", 640, 360), "first report is a change");
124        assert!(rx.has_changed().unwrap());
125        rx.mark_unchanged();
126
127        assert!(!g.record("cam:7", 640, 360), "same geometry is not");
128        assert!(
129            !rx.has_changed().unwrap(),
130            "no wake for an unchanged report"
131        );
132
133        assert!(g.record("cam:7", 1280, 720), "a layer switch is");
134        assert!(rx.has_changed().unwrap());
135    }
136
137    /// Streams are independent: one tile switching layers must not mask another's
138    /// first report.
139    #[test]
140    fn geometry_is_tracked_per_stream_key() {
141        let g = FrameGeometry::new();
142        assert!(g.record("cam:1", 640, 360));
143        assert!(
144            g.record("cam:2", 640, 360),
145            "a different key is its own change"
146        );
147        assert!(!g.record("cam:1", 640, 360));
148    }
149
150    /// A zero extent (a malformed/torn frame) must never churn a texture rebuild.
151    #[test]
152    fn degenerate_geometry_is_not_recorded() {
153        let g = FrameGeometry::new();
154        assert!(!g.record("cam:1", 0, 360));
155        assert!(!g.record("cam:1", 640, 0));
156        let mut streams = [("cam:1".to_owned(), 640, 360)];
157        g.overlay(&mut streams);
158        assert_eq!(streams[0], ("cam:1".to_owned(), 640, 360));
159    }
160
161    /// The snapshot owns membership: the overlay refreshes keys it already emits
162    /// and never invents one — so a peer that left (or a call that ended) can't be
163    /// resurrected by a stale entry.
164    #[test]
165    fn overlay_refreshes_known_keys_and_adds_none() {
166        let g = FrameGeometry::new();
167        g.record("cam:7", 1280, 720);
168        g.record("screen:9", 1920, 1080);
169
170        let mut streams = [
171            ("cam:7".to_owned(), 640, 360),
172            ("self".to_owned(), 1280, 720),
173        ];
174        g.overlay(&mut streams);
175        assert_eq!(streams[0], ("cam:7".to_owned(), 1280, 720), "refreshed");
176        assert_eq!(
177            streams[1],
178            ("self".to_owned(), 1280, 720),
179            "no report yet — left alone"
180        );
181        assert_eq!(streams.len(), 2, "screen:9 is not conjured into the list");
182    }
183
184    /// `forget` drops the entry, so a re-joining peer starts from the snapshot
185    /// again rather than inheriting its previous call's geometry.
186    #[test]
187    fn forget_drops_the_entry() {
188        let g = FrameGeometry::new();
189        g.record("cam:7", 1280, 720);
190        g.forget("cam:7");
191        let mut streams = [("cam:7".to_owned(), 640, 360)];
192        g.overlay(&mut streams);
193        assert_eq!(streams[0], ("cam:7".to_owned(), 640, 360));
194        assert!(g.record("cam:7", 1280, 720), "re-report is a fresh change");
195    }
196}
