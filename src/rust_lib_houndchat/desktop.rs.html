1//! Handing a file, folder or URL to the desktop: open it with the OS default
2//! handler, or show it selected in the platform file manager.
3//!
4//! **Every call runs on a throwaway thread**, joined before returning. The
5//! Windows shell APIs initialise COM on the thread that calls them (`open`
6//! makes it a single-threaded apartment, STA, to open a folder), and a
7//! thread's apartment is fixed for its lifetime. An FRB worker turned STA
8//! fails WASAPI default-device activation, which works only from a
9//! multithreaded apartment, the next time a device command lands on it —
10//! `ensure_audio_apartment` in hound-engine cannot re-type a thread that is
11//! already an STA. A thread that exits takes its apartment with it. The same
12//! hop keeps zbus's blocking API (the Linux reveal) off the tokio runtime FRB
13//! runs commands on, where it panics.
14
15use std::ffi::OsStr;
16use std::io;
17use std::path::Path;
18use std::thread;
19
20/// Open `target` — a path, or a URL the caller has already vetted — with the
21/// OS default handler.
22///
23/// Detached: `Ok` means the shell accepted the request, not that a window
24/// appeared. On Windows the acceptance is `ShellExecuteExW`'s, so a file type
25/// with no associated handler is an error here; elsewhere a handler that fails
26/// after launch is indistinguishable from success.
27pub(crate) fn open(target: &OsStr) -> io::Result<()> {
28    on_scratch_thread(|| open::that_detached(target))?
29}
30
31/// Show `path` selected in the platform file manager: Explorer on Windows,
32/// Finder on macOS, the freedesktop `FileManager1` service on Linux (or the
33/// containing folder, opened, where no such service answers).
34pub(crate) fn reveal(path: &Path) -> io::Result<()> {
35    on_scratch_thread(|| opener::reveal(path).map_err(io::Error::other))?
36}
37
38/// Run `f` on a fresh OS thread and wait for it. See the module docs for why
39/// the thread must be fresh rather than borrowed from a pool.
40fn on_scratch_thread<T: Send>(f: impl FnOnce() -> T + Send) -> io::Result<T> {
41    thread::scope(|scope| {
42        thread::Builder::new()
43            .name("hound-desktop".into())
44            .spawn_scoped(scope, f)?
45            .join()
46            .map_err(|_| io::Error::other("the desktop launcher panicked"))
47    })
48}

50#[cfg(test)]
51mod tests {
52    use super::*;
53
54    /// The work runs off the calling thread — the property the COM-apartment
55    /// invariant rests on — and its result comes back intact.
56    #[test]
57    fn work_runs_on_a_thread_of_its_own() {
58        let caller = thread::current().id();
59        let worker = on_scratch_thread(|| thread::current().id()).unwrap();
60        assert_ne!(worker, caller);
61    }
62
63    /// A panicking launcher is an error for the caller, not a panic on the
64    /// FRB worker that asked.
65    #[test]
66    fn a_panicking_launcher_is_an_error() {
67        let result = on_scratch_thread::<()>(|| panic!("launcher"));
68        assert!(result.is_err());
69    }
