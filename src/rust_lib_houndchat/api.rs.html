1//! The bridge API surface the Flutter UI binds against.
2//!
3//! One flat surface for codegen, split by domain into children:
4//!  * [`mirrors`] — every `#[frb(mirror(...))]` declaration: the error surface +
5//!    code registries, auth/identity snapshots, media/call enums, call + data
6//!    snapshots (and the tests pinning mirror↔registry parity).
7//!  * [`auth`] — the auth command clusters (passkey, browser-deferred, Steam,
8//!    magic-link, linked identities) + the WebAuthn JSON seam.
9//!  * [`call`] — the `call_*` voice/video commands.
10//!  * [`community`] — community + channel management (profile, icon, channels,
11//!    members, roles, invites, bans, per-channel permission overwrites,
12//!    instance permissions).
13//!  * [`links`] — opening a link externally + the embed debugger's unfurl.
14//!  * [`markup`] — message-markup commands: parse/plain-text/autolink + fetching
15//!    a single message.
16//!  * [`search`] — message search: the query parser (per keystroke), the search
17//!    command, and the detached-history jump/resume pair. Carries its own DTOs
18//!    where the language's `#[non_exhaustive]` enums cannot be mirrored.
19//!  * [`streams`] — the `watch_*` `StreamSink` forwarders + the two reshape DTOs.
20//!  * [`texture`] — the zero-copy texture handshake + the decode→render seam.
21//!  * [`log`] / [`sentry`] — the `tracing`→Dart bridge and Rust-side crash
22//!    reporting.
23//!
24//! This parent keeps the process-wide client handle (`init_app` + the openh264
25//! toggle), the data commands, and re-exports every child item so the generated
26//! glue's `crate::api::X` paths are stable regardless of which child defines `X`.
27//!
28//! **Projection rule (mirror by default; DTO only on a genuine reshape):** the
29//! facade's snapshots are `hound-client`'s own view-models. Rather than re-declare
30//! each one, we `#[frb(mirror(T))]` it — codegen emits the matching Dart class with
31//! zero field duplication, so a new field on the core type flows to Dart for free.
32//! A hand-written struct + `project_*` fn is written *only* where the shape genuinely
33//! differs from the core type:
34//!   - [`LevelsDto`] — `LevelsView.peers` is a `HashMap` (FRB mirrors a `Vec` cleanly).
35//!   - [`CommunityNode`] — the sidebar joins a community with its spaces (a *composed*
36//!     reshape built from the mirrored [`Community`]/[`Space`], still no duplication).
37//!   - [`RegisterStartDto`]/[`LoginStartDto`] — command *results*, not snapshots.
38//!
39//! Ids cross as `uuid::Uuid` (Dart `UuidValue`) via the FRB `uuid` feature.
40

84/// The process-wide [`HoundClient`], set once by [`init_app`]. FRB free functions
85/// fetch it; calling a command before `init_app` is a usage error (surfaced as
86/// [`HoundError::InvalidState`]).
87pub(super) static CLIENT: std::sync::OnceLock<Arc<HoundClient>> = std::sync::OnceLock::new();
88
89pub(super) fn client() -> Result<Arc<HoundClient>, HoundError> {
90    CLIENT.get().cloned().ok_or_else(|| {
91        HoundError::invalid_state(StateCode::NotInitialised, "init_app has not been called")
92    })
93}

95/// The app's private support directory, handed over from Dart (`path_provider`)
96/// by [`init_app`] — the Rust core has no platform API of its own to derive it,
97/// and it is where the downloaded openh264 codec is cached.
98static APP_DIR: std::sync::OnceLock<std::path::PathBuf> = std::sync::OnceLock::new();
99
100pub(super) fn app_dir() -> Result<&'static std::path::Path, HoundError> {
101    APP_DIR
102        .get()
103        .map(std::path::PathBuf::as_path)
104        .ok_or_else(|| {
105            HoundError::invalid_state(StateCode::NotInitialised, "init_app has not been called")
106        })
107}

109/// Build the [`HoundClient`] bound to `base_url` with a keyring-backed session store
110/// and install it as the process-wide handle. Idempotent at the storage layer (the
111/// keyring default store registers once); a second call after a successful first is a
112/// no-op that returns `Ok` (the existing client stays). Must be called before any
113/// other command/stream.
114///
115/// `app_dir` is the app's private support directory (Dart's `path_provider`);
116/// `openh264_enabled` is the user's persisted setting for Cisco's software H.264
117/// codec. Enabled, the codec is fetched **in the background** — deliberately not
118/// awaited, because this call is on the path to the first frame and the download
119/// is a network round-trip against Cisco's CDN. Software H.264 is simply
120/// unavailable until it lands; hardware H.264 is unaffected.
121///
122/// `app_dir` is optional because its ONLY consumer is that codec cache. If the
123/// platform channel that yields it fails, the app must still start: the codec is
124/// then unavailable (hardware paths and everything else are untouched) rather
125/// than the launch being lost to a black screen over an optional feature.
126pub async fn init_app(
127    base_url: String,
128    app_dir: Option<String>,
129    steam_app_id: Option<u32>,
130    openh264_enabled: bool,
131) -> Result<(), HoundError> {
132    match app_dir {
133        Some(dir) => {
134            let dir = std::path::PathBuf::from(dir);
135            let _ = APP_DIR.set(dir.clone());
136            start_openh264(dir, openh264_enabled);
137        }
138        None => tracing::warn!(
139            "openh264: no app-support directory — software H.264 is unavailable this session"
140        ),
141    }
142    if CLIENT.get().is_some() {
143        return Ok(());
144    }
145    // Sub-phases are logged, not merely awaited. Everything from here crosses into
146    // blocking platform code (the OS credential store, then Steamworks IPC), and
147    // Dart's log bridge turns each of these lines into a Sentry breadcrumb — so a
148    // startup that stalls in here names the sub-step instead of leaving the
149    // bootstrap watchdog to say only "init_app".
150    tracing::info!("init_app: opening the credential store");
151    // Off the async worker: the store registers a platform keyring backend, which
152    // on some hosts is a synchronous IPC call. `init_app` is on the path to the
153    // first frame, and a blocked worker there is a black screen.
154    let store = tokio::task::spawn_blocking(KeyringSessionStore::new)
155        .await
156        .map_err(|e| {
157            HoundError::internal(
158                InternalCode::CredentialStore,
159                format!("credential-store init task failed: {e}"),
160            )
161        })??;
162    tracing::info!("init_app: initialising steamworks");
163    // `new_bounded`, never `new`: the Steamworks handshake is the one unbounded
164    // blocking call on this path (see `HoundClient::new_bounded`).
165    let client = HoundClient::new_bounded(&base_url, store, steam_app_id).await?;
166    tracing::info!(steam = client.steam_available(), "init_app: client built");
167    // Install the decode→render video sink: the call orchestration hands decoded
168    // I420 here, keyed `"self"`/`"cam:{peer}"`/`"screen:{peer}"`, and this forwards
169    // to `hound_render::push_frame` — the same per-stream mailbox the self-camera
170    // preview feeds. The Flutter side attaches a texture per tile by stream key.
171    client
172        .set_video_sink(std::sync::Arc::new(RenderVideoSink))
173        .await;
174    // Race-tolerant: if another caller won, drop ours and keep theirs.
175    let _ = CLIENT.set(client);
176    Ok(())
177}

179/// Turn Cisco's software H.264 codec on or off — the settings toggle openh264's
180/// BINARY_LICENSE requires the app to provide (alongside the attribution notice
181/// and the license text, both rendered in the UI).
182///
183/// On: permit the codec and start the download if it is not cached yet. Off:
184/// stop using it immediately *and* delete the cached library, so "off" means the
185/// Cisco binary is neither running nor present. Returns as soon as the intent is
186/// recorded; a fresh download continues in the background.
187pub async fn set_openh264_enabled(enabled: bool) -> Result<(), HoundError> {
188    // `app_dir` is unset only when the platform channel that yields it failed at
189    // startup; there is then no cache to write or delete, and nothing to toggle.
190    let dir = app_dir()?.to_path_buf();
191    if !enabled {
192        return hound_client::openh264::remove(&dir);
193    }
194    start_openh264(dir, true);
195    Ok(())
196}
197
198/// Apply the openh264 setting: permit + fetch in the background, or forbid +
199/// drop the cache. Never blocks the caller — [`init_app`] runs before the first
200/// frame, and the toggle should feel instant.
201fn start_openh264(app_dir: std::path::PathBuf, enabled: bool) {
202    if !enabled {
203        // Honour a persisted "off" from a previous session: no download, and
204        // nothing left on disk from before the user turned it off.
205        if let Err(e) = hound_client::openh264::remove(&app_dir) {
206            tracing::warn!("openh264: could not remove the cached codec: {e}");
207        }
208        return;
209    }
210    // Marks the download in flight synchronously, so a call joined immediately
211    // after startup waits for it rather than advertising no software H.264 for
212    // the whole session (`hound_client::openh264::settle`).
213    hound_client::openh264::spawn_install(app_dir);
214}

247/// The image MIME types this deployment accepts for an avatar / attachment upload,
248/// asked of the server (`OPTIONS` + `Accept-Post`) rather than hard-coded — the set
249/// moves with the BACKEND deploy, not with the client release.
250///
251/// An **empty** list means the server did not advertise (an older backend), not that
252/// it accepts nothing: fall back to the client's own set rather than blocking the
253/// picker. A real failure (offline, non-2xx) throws.
254///
255/// The client core hands these over as typed `mime::Mime`s; they cross the bridge
256/// rendered (`type/subtype`) because Dart has no `Mime` to mirror them into.
257pub async fn supported_upload_mime_types(
258    purpose: UploadPurpose,
259) -> Result<Vec<String>, HoundError> {
260    Ok(client()?
261        .supported_upload_mime_types(purpose)
262        .await?
263        .into_iter()
264        .map(|mime| mime.to_string())
265        .collect())
266}

268/// Decode a JPEG XL image to raw pixels for the UI.
269///
270/// The one remote format Skia cannot decode, so the Dart image provider sniffs the
271/// bytes and routes only these here; everything else stays on the platform codec
272/// path. Returns straight-alpha RGBA8888 (`width * height * 4`) — the same shape as
273/// [`call_screen_thumbnail`], rendered the same way.
274///
275/// Not `async`: a decode is CPU-bound, and a plain function runs on an FRB *worker*
276/// thread (so it never occupies the async executor). An animated JXL yields its
277/// first keyframe; a non-JXL or corrupt payload is a typed `DECODE0xx` error.
278pub fn decode_image(bytes: Vec<u8>) -> Result<DecodedImage, HoundError> {
279    hound_client::image::decode_image(&bytes)
280}

282/// One node of the sidebar tree: a [`Community`] joined with its [`Space`]s. A
283/// *composed* reshape (the core keeps community/space lists separately) built from
284/// the mirrored types — so there is no field duplication, only the join.
285pub struct CommunityNode {
286    pub community: Community,
287    pub spaces: Vec<Space>,
288}
289
290pub(super) fn project_tree(tree: &hound_client::CommunityTree) -> Vec<CommunityNode> {
291    tree.communities()
292        .map(|c| CommunityNode {
293            community: c.clone(),
294            spaces: tree.spaces(c.id).to_vec(),
295        })
296        .collect()
297}
