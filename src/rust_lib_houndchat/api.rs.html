1//! The bridge API surface the Flutter UI binds against.
2//!
3//! One flat surface for codegen, split by domain into children:
4//!  * [`mirrors`] — every `#[frb(mirror(...))]` declaration: the error surface +
5//!    code registries, auth/identity snapshots, media/call enums, call + data
6//!    snapshots (and the tests pinning mirror↔registry parity).
7//!  * [`auth`] — the auth command clusters (passkey, browser-deferred, Steam,
8//!    magic-link, linked identities) + the WebAuthn JSON seam.
9//!  * [`call`] — the `call_*` voice/video commands.
10//!  * [`links`] — opening a link externally + the embed debugger's unfurl.
11//!  * [`markup`] — message-markup commands: parse/plain-text/autolink + fetching
12//!    a single message.
13//!  * [`search`] — message search: the query parser (per keystroke), the search
14//!    command, and the detached-history jump/resume pair. Carries its own DTOs
15//!    where the language's `#[non_exhaustive]` enums cannot be mirrored.
16//!  * [`streams`] — the `watch_*` `StreamSink` forwarders + the two reshape DTOs.
17//!  * [`texture`] — the zero-copy texture handshake + the decode→render seam.
18//!  * [`log`] / [`sentry`] — the `tracing`→Dart bridge and Rust-side crash
19//!    reporting.
20//!
21//! This parent keeps the process-wide client handle (`init_app` + the openh264
22//! toggle), the data commands, and re-exports every child item so the generated
23//! glue's `crate::api::X` paths are stable regardless of which child defines `X`.
24//!
25//! **Projection rule (mirror by default; DTO only on a genuine reshape):** the
26//! facade's snapshots are `hound-client`'s own view-models. Rather than re-declare
27//! each one, we `#[frb(mirror(T))]` it — codegen emits the matching Dart class with
28//! zero field duplication, so a new field on the core type flows to Dart for free.
29//! A hand-written struct + `project_*` fn is written *only* where the shape genuinely
30//! differs from the core type:
31//!   - [`LevelsDto`] — `LevelsView.peers` is a `HashMap` (FRB mirrors a `Vec` cleanly).
32//!   - [`CommunityNode`] — the sidebar joins a community with its spaces (a *composed*
33//!     reshape built from the mirrored [`Community`]/[`Space`], still no duplication).
34//!   - [`RegisterStartDto`]/[`LoginStartDto`] — command *results*, not snapshots.
35//!
36//! Ids cross as `uuid::Uuid` (Dart `UuidValue`) via the FRB `uuid` feature.
37

79/// The process-wide [`HoundClient`], set once by [`init_app`]. FRB free functions
80/// fetch it; calling a command before `init_app` is a usage error (surfaced as
81/// [`HoundError::InvalidState`]).
82pub(super) static CLIENT: std::sync::OnceLock<Arc<HoundClient>> = std::sync::OnceLock::new();
83
84pub(super) fn client() -> Result<Arc<HoundClient>, HoundError> {
85    CLIENT.get().cloned().ok_or_else(|| {
86        HoundError::invalid_state(StateCode::NotInitialised, "init_app has not been called")
87    })
88}

90/// The app's private support directory, handed over from Dart (`path_provider`)
91/// by [`init_app`] — the Rust core has no platform API of its own to derive it,
92/// and it is where the downloaded openh264 codec is cached.
93static APP_DIR: std::sync::OnceLock<std::path::PathBuf> = std::sync::OnceLock::new();
94
95pub(super) fn app_dir() -> Result<&'static std::path::Path, HoundError> {
96    APP_DIR
97        .get()
98        .map(std::path::PathBuf::as_path)
99        .ok_or_else(|| {
100            HoundError::invalid_state(StateCode::NotInitialised, "init_app has not been called")
101        })
102}

104/// Build the [`HoundClient`] bound to `base_url` with a keyring-backed session store
105/// and install it as the process-wide handle. Idempotent at the storage layer (the
106/// keyring default store registers once); a second call after a successful first is a
107/// no-op that returns `Ok` (the existing client stays). Must be called before any
108/// other command/stream.
109///
110/// `app_dir` is the app's private support directory (Dart's `path_provider`);
111/// `openh264_enabled` is the user's persisted setting for Cisco's software H.264
112/// codec. Enabled, the codec is fetched **in the background** — deliberately not
113/// awaited, because this call is on the path to the first frame and the download
114/// is a network round-trip against Cisco's CDN. Software H.264 is simply
115/// unavailable until it lands; hardware H.264 is unaffected.
116///
117/// `app_dir` is optional because its ONLY consumer is that codec cache. If the
118/// platform channel that yields it fails, the app must still start: the codec is
119/// then unavailable (hardware paths and everything else are untouched) rather
120/// than the launch being lost to a black screen over an optional feature.
121pub async fn init_app(
122    base_url: String,
123    app_dir: Option<String>,
124    steam_app_id: Option<u32>,
125    openh264_enabled: bool,
126) -> Result<(), HoundError> {
127    match app_dir {
128        Some(dir) => {
129            let dir = std::path::PathBuf::from(dir);
130            let _ = APP_DIR.set(dir.clone());
131            start_openh264(dir, openh264_enabled);
132        }
133        None => tracing::warn!(
134            "openh264: no app-support directory — software H.264 is unavailable this session"
135        ),
136    }
137    if CLIENT.get().is_some() {
138        return Ok(());
139    }
140    // Sub-phases are logged, not merely awaited. Everything from here crosses into
141    // blocking platform code (the OS credential store, then Steamworks IPC), and
142    // Dart's log bridge turns each of these lines into a Sentry breadcrumb — so a
143    // startup that stalls in here names the sub-step instead of leaving the
144    // bootstrap watchdog to say only "init_app".
145    tracing::info!("init_app: opening the credential store");
146    // Off the async worker: the store registers a platform keyring backend, which
147    // on some hosts is a synchronous IPC call. `init_app` is on the path to the
148    // first frame, and a blocked worker there is a black screen.
149    let store = tokio::task::spawn_blocking(KeyringSessionStore::new)
150        .await
151        .map_err(|e| {
152            HoundError::internal(
153                InternalCode::CredentialStore,
154                format!("credential-store init task failed: {e}"),
155            )
156        })??;
157    tracing::info!("init_app: initialising steamworks");
158    // `new_bounded`, never `new`: the Steamworks handshake is the one unbounded
159    // blocking call on this path (see `HoundClient::new_bounded`).
160    let client = HoundClient::new_bounded(&base_url, store, steam_app_id).await?;
161    tracing::info!(steam = client.steam_available(), "init_app: client built");
162    // Install the decode→render video sink: the call orchestration hands decoded
163    // I420 here, keyed `"self"`/`"cam:{peer}"`/`"screen:{peer}"`, and this forwards
164    // to `hound_render::push_frame` — the same per-stream mailbox the self-camera
165    // preview feeds. The Flutter side attaches a texture per tile by stream key.
166    client
167        .set_video_sink(std::sync::Arc::new(RenderVideoSink))
168        .await;
169    // Race-tolerant: if another caller won, drop ours and keep theirs.
170    let _ = CLIENT.set(client);
171    Ok(())
172}

174/// Turn Cisco's software H.264 codec on or off — the settings toggle openh264's
175/// BINARY_LICENSE requires the app to provide (alongside the attribution notice
176/// and the license text, both rendered in the UI).
177///
178/// On: permit the codec and start the download if it is not cached yet. Off:
179/// stop using it immediately *and* delete the cached library, so "off" means the
180/// Cisco binary is neither running nor present. Returns as soon as the intent is
181/// recorded; a fresh download continues in the background.
182pub async fn set_openh264_enabled(enabled: bool) -> Result<(), HoundError> {
183    // `app_dir` is unset only when the platform channel that yields it failed at
184    // startup; there is then no cache to write or delete, and nothing to toggle.
185    let dir = app_dir()?.to_path_buf();
186    if !enabled {
187        return hound_client::openh264::remove(&dir);
188    }
189    start_openh264(dir, true);
190    Ok(())
191}
192
193/// Apply the openh264 setting: permit + fetch in the background, or forbid +
194/// drop the cache. Never blocks the caller — [`init_app`] runs before the first
195/// frame, and the toggle should feel instant.
196fn start_openh264(app_dir: std::path::PathBuf, enabled: bool) {
197    if !enabled {
198        // Honour a persisted "off" from a previous session: no download, and
199        // nothing left on disk from before the user turned it off.
200        if let Err(e) = hound_client::openh264::remove(&app_dir) {
201            tracing::warn!("openh264: could not remove the cached codec: {e}");
202        }
203        return;
204    }
205    // Marks the download in flight synchronously, so a call joined immediately
206    // after startup waits for it rather than advertising no software H.264 for
207    // the whole session (`hound_client::openh264::settle`).
208    hound_client::openh264::spawn_install(app_dir);
209}

242/// The image MIME types this deployment accepts for an avatar / attachment upload,
243/// asked of the server (`OPTIONS` + `Accept-Post`) rather than hard-coded — the set
244/// moves with the BACKEND deploy, not with the client release.
245///
246/// An **empty** list means the server did not advertise (an older backend), not that
247/// it accepts nothing: fall back to the client's own set rather than blocking the
248/// picker. A real failure (offline, non-2xx) throws.
249///
250/// The client core hands these over as typed `mime::Mime`s; they cross the bridge
251/// rendered (`type/subtype`) because Dart has no `Mime` to mirror them into.
252pub async fn supported_upload_mime_types(
253    purpose: UploadPurpose,
254) -> Result<Vec<String>, HoundError> {
255    Ok(client()?
256        .supported_upload_mime_types(purpose)
257        .await?
258        .into_iter()
259        .map(|mime| mime.to_string())
260        .collect())
261}

263/// Decode a JPEG XL image to raw pixels for the UI.
264///
265/// The one remote format Skia cannot decode, so the Dart image provider sniffs the
266/// bytes and routes only these here; everything else stays on the platform codec
267/// path. Returns straight-alpha RGBA8888 (`width * height * 4`) — the same shape as
268/// [`call_screen_thumbnail`], rendered the same way.
269///
270/// Not `async`: a decode is CPU-bound, and a plain function runs on an FRB *worker*
271/// thread (so it never occupies the async executor). An animated JXL yields its
272/// first keyframe; a non-JXL or corrupt payload is a typed `DECODE0xx` error.
273pub fn decode_image(bytes: Vec<u8>) -> Result<DecodedImage, HoundError> {
274    hound_client::image::decode_image(&bytes)
275}

277/// One node of the sidebar tree: a [`Community`] joined with its [`Space`]s. A
278/// *composed* reshape (the core keeps community/space lists separately) built from
279/// the mirrored types — so there is no field duplication, only the join.
280pub struct CommunityNode {
281    pub community: Community,
282    pub spaces: Vec<Space>,
283}
284
285pub(super) fn project_tree(tree: &hound_client::CommunityTree) -> Vec<CommunityNode> {
286    tree.communities()
287        .map(|c| CommunityNode {
288            community: c.clone(),
289            spaces: tree.spaces(c.id).to_vec(),
290        })
291        .collect()
292}
