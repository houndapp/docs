1//! Sentry policy for the Rust client: what becomes an event, what stays a
2//! breadcrumb, and what every event carries.
3//!
4//! TWO Sentry clients live in this process — Dart's (`SentryFlutter.init`) and
5//! Rust's ([`crate::api::init_sentry`]) — each with its own scope and breadcrumb
6//! ring, each attaching only to failures on its own side. This module is the Rust
7//! half's policy. Its job is to make an event *self-contained*: a bare message with
8//! no story, no join key and no app state is not triageable, and every rule below
9//! was earned by an event that wasn't.
10//!
11//! ## What becomes an event
12//!
13//! - **Only first-party ERROR.** A dependency's error-level hygiene warning —
14//!   `iroh::socket`'s "Endpoint dropped without calling `Endpoint::close`" — is not
15//!   an app failure. Promoted, it files an issue with no stack, no user and nothing
16//!   to act on; demoted, it still shows up in the breadcrumb trail of whatever
17//!   first-party failure follows it. See [`is_first_party`].
18//! - **INFO and above become breadcrumbs.** The core's `info!` lines *are* the
19//!   narrative ("session restore: …", "call joined"); without them an event says
20//!   what broke and never how it got there.
21//! - **One event per callsite per [`EVENT_COOLDOWN`].** A first-party `error!` on a
22//!   hot path (a per-frame encode failure) would otherwise file thousands of
23//!   identical issues and burn the project quota. Repeats demote to breadcrumbs, so
24//!   the burst stays visible locally.
25//!
26//! ## What every event carries
27//!
28//! [`enrich`] stamps the join key ([`LAUNCH_ID_TAG`] — the *only* thing tying a Rust
29//! event to the Dart event from the same launch, since the two clients share no
30//! trace), what this process is ([`APP_ROLE_ENV`]), how long it had been up, whether
31//! a call was live, and the audio pipeline's verdict when one was.
32
33use std::collections::HashMap;
34use std::sync::{Mutex, OnceLock, PoisonError};
35use std::time::{Duration, Instant};

37use sentry::integrations::tracing::EventFilter;
38use sentry::protocol::Event;

41/// Tag joining a Rust event to the Dart event from the same process launch. The two
42/// SDKs mint independent trace ids, so without this there is no query that returns
43/// both halves of one incident; the Dart side reads it back over the bridge (see
44/// `api::sentry_launch_id`) and tags its own client with the same value.
45pub const LAUNCH_ID_TAG: &str = "launch_id";
46
47/// Env var naming what this process *is*: `user` (the default) or a runner label set
48/// by whatever launched it — today `ci-smoke`, from the nightly launch-smoke job in
49/// `.github/workflows/nightly.yaml`. A real install never sets it. Without it a
50/// runner's failures land in the same issue stream as users' and inflate every
51/// impact count, which is how you end up triaging a robot.
52pub const APP_ROLE_ENV: &str = "HOUND_APP_ROLE";
53
54/// Minimum spacing between Sentry *events* from one `tracing` callsite.
55const EVENT_COOLDOWN: Duration = Duration::from_secs(30);
56
57/// This launch's id, minted on first use. Stable for the process lifetime.
58pub fn launch_id() -> &'static str {
59    static LAUNCH_ID: OnceLock<String> = OnceLock::new();
60    LAUNCH_ID.get_or_init(|| uuid::Uuid::new_v4().to_string())
61}
62
63/// See [`APP_ROLE_ENV`]. Read once — the environment does not change under us.
64pub fn app_role() -> &'static str {
65    static APP_ROLE: OnceLock<String> = OnceLock::new();
66    APP_ROLE
67        .get_or_init(|| std::env::var(APP_ROLE_ENV).unwrap_or_else(|_| "user".to_owned()))
68        .as_str()
69}

71/// How long this process has been reporting. Anchored on the first call, which
72/// [`crate::api::init_sentry`] makes at startup — milliseconds into the process, and
73/// the earliest point at which anything is reportable anyway.
74pub fn uptime() -> Duration {
75    static START: OnceLock<Instant> = OnceLock::new();
76    START.get_or_init(Instant::now).elapsed()
77}
78
79/// Coarse [`uptime`] band, for filtering. The same message means different things at
80/// different ages — "endpoint dropped" seconds after launch or at teardown is
81/// lifecycle noise, the same line an hour in is the wedge — and the exact seconds
82/// (carried as `uptime_seconds`) are too high-cardinality to group by.
83fn uptime_bucket(uptime: Duration) -> &'static str {
84    match uptime.as_secs() {
85        0..=9 => "startup",
86        10..=59 => "under-1m",
87        60..=599 => "under-10m",
88        _ => "over-10m",
89    }
90}

92/// Whether a `tracing` target is ours. Targets are module paths, so this is a prefix
93/// test; NB this crate's target is its LIB name (`rust_lib_houndchat`), not the
94/// package name `hound-frb`.
95///
96/// This is also the extension point: a third-party target whose ERROR genuinely
97/// warrants an issue of its own gets an arm here. None has earned one — every
98/// dependency error seen so far reads as an app failure through first-party code
99/// first, and arrives as a breadcrumb under it.
100fn is_first_party(target: &str) -> bool {
101    target.starts_with("hound_") || target.starts_with("rust_lib_houndchat")
102}
103
104/// Whether this callsite may file an event again yet, recording the attempt if so.
105///
106/// Keyed by `tracing`'s callsite [`Identifier`](tracing::callsite::Identifier), so
107/// repeats of one `error!` collapse regardless of the values interpolated into the
108/// message. The map is bounded by the number of distinct callsites compiled into the
109/// binary, so it needs no eviction. Lock poisoning is ignored deliberately: this runs
110/// on the failing thread, and a reporter that panics while reporting reports nothing.
111fn cooled_down(md: &Metadata<'_>) -> bool {
112    static LAST_EVENT: OnceLock<Mutex<HashMap<tracing::callsite::Identifier, Instant>>> =
113        OnceLock::new();
114    let mut seen = LAST_EVENT
115        .get_or_init(Default::default)
116        .lock()
117        .unwrap_or_else(PoisonError::into_inner);
118    admit(&mut seen, md.callsite(), Instant::now())
119}
120
121/// The rate limiter itself, over any key — split out from [`cooled_down`] so it is
122/// testable without synthesising `tracing` callsites.
123fn admit<K: std::hash::Hash + Eq>(seen: &mut HashMap<K, Instant>, key: K, now: Instant) -> bool {
124    match seen.get(&key) {
125        Some(last) if now.duration_since(*last) < EVENT_COOLDOWN => false,
126        _ => {
127            seen.insert(key, now);
128            true
129        }
130    }
131}

133/// The event-vs-breadcrumb decision for one `tracing` event. See the module docs for
134/// the three rules; DEBUG/TRACE are dropped entirely (they still reach the Dart log
135/// sink, which has its own filter).
136pub fn event_filter(md: &Metadata<'_>) -> EventFilter {
137    match *md.level() {
138        Level::ERROR if is_first_party(md.target()) => {
139            if cooled_down(md) {
140                EventFilter::Event
141            } else {
142                EventFilter::Breadcrumb
143            }
144        }
145        Level::ERROR | Level::WARN | Level::INFO => EventFilter::Breadcrumb,
146        Level::DEBUG | Level::TRACE => EventFilter::Ignore,
147    }
148}

150/// What the live client can say about itself at event time. `None` before the client
151/// exists — the reporter comes up first, deliberately, so a startup crash still
152/// reports.
153pub struct AppSnapshot {
154    pub in_call: bool,
155    /// The audio verdict + full debug report. `Some` only during a call: between
156    /// calls the report is a fixed "healthy" block that would bloat every event for
157    /// nothing.
158    pub audio: Option<AudioSnapshot>,
159    /// The video verdict + full debug report, on the same in-call-only terms as
160    /// [`AppSnapshot::audio`].
161    pub video: Option<VideoSnapshot>,
162}
163
164/// The audio pipeline's self-assessment, in both forms (see
165/// `hound_client::audio_report::Diagnosis`): a low-cardinality stage token fit for a
166/// tag, and the paste-able report behind it.
167pub struct AudioSnapshot {
168    pub stage: &'static str,
169    pub report: String,
170}
171
172/// The video pipeline's self-assessment, in the same two forms as
173/// [`AudioSnapshot`] (see `hound_client::video_report::Diagnosis`).
174pub struct VideoSnapshot {
175    pub stage: &'static str,
176    pub report: String,
177}

179/// Stamp an outgoing event with everything triage needs that the message itself
180/// can't carry. Runs as the client's `before_send`, i.e. after the scope and the
181/// default integrations have had their turn.
182pub fn enrich(
183    mut event: Event<'static>,
184    dist: Option<&str>,
185    app: Option<AppSnapshot>,
186) -> Event<'static> {
187    // `sentry-contexts` fills `server_name` with the machine's hostname whenever the
188    // client didn't set one, and exposes no opt-out — so strip it here, downstream of
189    // that. The Dart side reports an opaque user id and nothing else; the Rust side
190    // must not be the half that leaks a device name.
191    event.server_name = None;
192    // Unlike release/environment, `dist` has no `ClientOptions` field — it lives on
193    // the wire event, so it can only be set here.
194    if let Some(dist) = dist {
195        event.dist = Some(dist.to_owned().into());
196    }
197    let uptime = uptime();
198    event
199        .tags
200        .insert(LAUNCH_ID_TAG.to_owned(), launch_id().to_owned());
201    event
202        .tags
203        .insert("app_role".to_owned(), app_role().to_owned());
204    event
205        .tags
206        .insert("uptime".to_owned(), uptime_bucket(uptime).to_owned());
207    event
208        .extra
209        .insert("uptime_seconds".to_owned(), uptime.as_secs().into());
210    if let Some(app) = app {
211        event
212            .tags
213            .insert("in_call".to_owned(), app.in_call.to_string());
214        if let Some(audio) = app.audio {
215            event
216                .tags
217                .insert("audio_stage".to_owned(), audio.stage.to_owned());
218            event
219                .extra
220                .insert("audio_report".to_owned(), audio.report.into());
221        }
222        if let Some(video) = app.video {
223            event
224                .tags
225                .insert("video_stage".to_owned(), video.stage.to_owned());
226            event
227                .extra
228                .insert("video_report".to_owned(), video.report.into());
229        }
230    }
231    event
232}

238    #[test]
239    fn only_first_party_targets_are_ours() {
240        assert!(is_first_party("hound_client::session"));
241        assert!(is_first_party("hound_media::downlink"));
242        assert!(is_first_party("rust_lib_houndchat::api"));
243        // The HOUNDCHAT-58 case: a dependency's error-level hygiene warning.
244        assert!(!is_first_party("iroh::socket"));
245        assert!(!is_first_party("moq_lite::session"));
246        // Prefix-matching must not catch an unrelated crate that merely starts the
247        // same way.
248        assert!(!is_first_party("houndini::thing"));
249    }

251    #[test]
252    fn uptime_buckets_stay_low_cardinality() {
253        assert_eq!(uptime_bucket(Duration::from_secs(0)), "startup");
254        assert_eq!(uptime_bucket(Duration::from_secs(9)), "startup");
255        assert_eq!(uptime_bucket(Duration::from_secs(10)), "under-1m");
256        assert_eq!(uptime_bucket(Duration::from_secs(59)), "under-1m");
257        assert_eq!(uptime_bucket(Duration::from_secs(60)), "under-10m");
258        assert_eq!(uptime_bucket(Duration::from_secs(599)), "under-10m");
259        assert_eq!(uptime_bucket(Duration::from_secs(600)), "over-10m");
260        assert_eq!(uptime_bucket(Duration::from_secs(86_400)), "over-10m");
261    }

263    /// One callsite may file once per cooldown and its repeats demote, while a
264    /// different callsite keeps its own budget — a hot-path `error!` must not mute
265    /// the rest of the app.
266    #[test]
267    fn repeats_from_one_callsite_are_rate_limited_independently() {
268        let mut seen = HashMap::new();
269        let t0 = Instant::now();
270
271        assert!(admit(&mut seen, "encode-failed", t0));
272        assert!(!admit(&mut seen, "encode-failed", t0 + EVENT_COOLDOWN / 2));
273        assert!(
274            admit(&mut seen, "decode-failed", t0),
275            "a different callsite has its own budget"
276        );
277        assert!(
278            admit(&mut seen, "encode-failed", t0 + EVENT_COOLDOWN),
279            "the budget refreshes once the cooldown elapses"
280        );
281    }
