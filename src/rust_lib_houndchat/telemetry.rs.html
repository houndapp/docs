1//! Sentry policy for the Rust client: what becomes an event, what stays a
2//! breadcrumb, and what every event carries.
3//!
4//! TWO Sentry clients live in this process — Dart's (`SentryFlutter.init`) and
5//! Rust's ([`crate::api::init_sentry`]) — each with its own scope and breadcrumb
6//! ring, each attaching only to failures on its own side. This module is the Rust
7//! half's policy. Its job is to make an event *self-contained*: a bare message with
8//! no story, no join key and no app state is not triageable, and every rule below
9//! was earned by an event that wasn't.
10//!
11//! ## What becomes an event
12//!
13//! - **Only first-party ERROR** — where first-party includes the media stack's
14//!   stage-labelled targets (`wgc`, `audio_endpoint`, …), which are ours despite not
15//!   looking like module paths. A dependency's error-level hygiene warning —
16//!   `iroh::socket`'s "Endpoint dropped without calling `Endpoint::close`" — is not
17//!   an app failure. Promoted, it files an issue with no stack, no user and nothing
18//!   to act on; demoted, it still shows up in the breadcrumb trail of whatever
19//!   first-party failure follows it. See [`is_first_party`].
20//! - **INFO and above become breadcrumbs.** The core's `info!` lines *are* the
21//!   narrative ("session restore: …", "call joined"); without them an event says
22//!   what broke and never how it got there.
23//! - **One event per callsite per [`EVENT_COOLDOWN`].** A first-party `error!` on a
24//!   hot path (a per-frame encode failure) would otherwise file thousands of
25//!   identical issues and burn the project quota. Repeats demote to breadcrumbs, so
26//!   the burst stays visible locally.
27//!
28//! ## What every event carries
29//!
30//! [`enrich`] stamps the join key ([`LAUNCH_ID_TAG`] — the *only* thing tying a Rust
31//! event to the Dart event from the same launch, since the two clients share no
32//! trace), what this process is ([`APP_ROLE_ENV`]), how long it had been up, whether
33//! a call was live, the audio pipeline's verdict when one was, and the emitting
34//! callsite's typed error code ([`ERROR_CODE_TAG`], lifted off the event's `code`
35//! field) so Rust events group by cause exactly as the Dart half's do.

45/// Tag joining a Rust event to the Dart event from the same process launch. The two
46/// SDKs mint independent trace ids, so without this there is no query that returns
47/// both halves of one incident; the Dart side reads it back over the bridge (see
48/// `api::sentry_launch_id`) and tags its own client with the same value.
49pub const LAUNCH_ID_TAG: &str = "launch_id";
50
51/// Env var naming what this process *is*: `user` (the default) or a runner label set
52/// by whatever launched it — today `ci-smoke`, from the nightly launch-smoke job in
53/// `.github/workflows/nightly.yaml`. A real install never sets it. Without it a
54/// runner's failures land in the same issue stream as users' and inflate every
55/// impact count, which is how you end up triaging a robot.
56pub const APP_ROLE_ENV: &str = "HOUND_APP_ROLE";
57
58/// Minimum spacing between Sentry *events* from one `tracing` callsite.
59const EVENT_COOLDOWN: Duration = Duration::from_secs(30);

61/// This launch's id, minted on first use. Stable for the process lifetime.
62pub fn launch_id() -> &'static str {
63    static LAUNCH_ID: OnceLock<String> = OnceLock::new();
64    LAUNCH_ID.get_or_init(|| uuid::Uuid::new_v4().to_string())
65}
66
67/// See [`APP_ROLE_ENV`]. Read once — the environment does not change under us.
68pub fn app_role() -> &'static str {
69    static APP_ROLE: OnceLock<String> = OnceLock::new();
70    APP_ROLE
71        .get_or_init(|| std::env::var(APP_ROLE_ENV).unwrap_or_else(|_| "user".to_owned()))
72        .as_str()
73}

75/// How long this process has been reporting. Anchored on the first call, which
76/// [`crate::api::init_sentry`] makes at startup — milliseconds into the process, and
77/// the earliest point at which anything is reportable anyway.
78pub fn uptime() -> Duration {
79    static START: OnceLock<Instant> = OnceLock::new();
80    START.get_or_init(Instant::now).elapsed()
81}
82
83/// Coarse [`uptime`] band, for filtering. The same message means different things at
84/// different ages — "endpoint dropped" seconds after launch or at teardown is
85/// lifecycle noise, the same line an hour in is the wedge — and the exact seconds
86/// (carried as `uptime_seconds`) are too high-cardinality to group by.
87fn uptime_bucket(uptime: Duration) -> &'static str {
88    match uptime.as_secs() {
89        0..=9 => "startup",
90        10..=59 => "under-1m",
91        60..=599 => "under-10m",
92        _ => "over-10m",
93    }
94}

96/// First-party `tracing` targets that are NOT module paths.
97///
98/// The media/render stack labels its lines by PIPELINE STAGE rather than by module
99/// (`target: "wgc"`, `target: "audio_endpoint"`, …) so a log reads as a pipeline
100/// instead of a crate tree. That is deliberate and useful — and it silently excluded
101/// every one of those callsites from [`is_first_party`], i.e. from event promotion, at
102/// any level: the capture/encode/decode/playout layer, which is where the failures
103/// users actually feel live, could only ever produce breadcrumbs. This list is that
104/// layer, enumerated (it is a closed set — `grep -r 'target: "' crates/`).
105///
106/// Keep it in sync when a new stage label appears; an unlisted label is not a bug so
107/// much as a stage that quietly stops reporting, which is the failure this fixes.
108const STAGE_TARGETS: &[&str] = &[
109    "audio_devices",
110    "audio_endpoint",
111    "avf_capture",
112    "engine",
113    "mf_capture",
114    "openh264",
115    "portal_screen",
116    "pw_audio",
117    "sck_capture",
118    "v4l2_capture",
119    "video_decode",
120    "video_encode",
121    "video_render_cpu",
122    "video_render_win",
123    "wasapi_loopback",
124    "wgc",
125];
126
127/// Whether a `tracing` target is ours: a first-party module path (targets default to
128/// the module path, so this is a prefix test; NB this crate's target is its LIB name
129/// `rust_lib_houndchat`, not the package name `hound-frb`) OR one of the media stack's
130/// explicit stage labels ([`STAGE_TARGETS`]).
131///
132/// This is also the extension point: a third-party target whose ERROR genuinely
133/// warrants an issue of its own gets an arm here. None has earned one — every
134/// dependency error seen so far reads as an app failure through first-party code
135/// first, and arrives as a breadcrumb under it.
136fn is_first_party(target: &str) -> bool {
137    target.starts_with("hound_")
138        || target.starts_with("rust_lib_houndchat")
139        || STAGE_TARGETS.contains(&target)
140}

142/// Whether this callsite may file an event again yet, recording the attempt if so.
143///
144/// Keyed by `tracing`'s callsite [`Identifier`](tracing::callsite::Identifier), so
145/// repeats of one `error!` collapse regardless of the values interpolated into the
146/// message. The map is bounded by the number of distinct callsites compiled into the
147/// binary, so it needs no eviction. Lock poisoning is ignored deliberately: this runs
148/// on the failing thread, and a reporter that panics while reporting reports nothing.
149fn cooled_down(md: &Metadata<'_>) -> bool {
150    static LAST_EVENT: OnceLock<Mutex<HashMap<tracing::callsite::Identifier, Instant>>> =
151        OnceLock::new();
152    let mut seen = LAST_EVENT
153        .get_or_init(Default::default)
154        .lock()
155        .unwrap_or_else(PoisonError::into_inner);
156    admit(&mut seen, md.callsite(), Instant::now())
157}
158
159/// The rate limiter itself, over any key — split out from [`cooled_down`] so it is
160/// testable without synthesising `tracing` callsites.
161fn admit<K: std::hash::Hash + Eq>(seen: &mut HashMap<K, Instant>, key: K, now: Instant) -> bool {
162    match seen.get(&key) {
163        Some(last) if now.duration_since(*last) < EVENT_COOLDOWN => false,
164        _ => {
165            seen.insert(key, now);
166            true
167        }
168    }
169}

171/// The event-vs-breadcrumb decision for one `tracing` event. See the module docs for
172/// the three rules; DEBUG/TRACE are dropped entirely (they still reach the Dart log
173/// sink, which has its own filter).
174pub fn event_filter(md: &Metadata<'_>) -> EventFilter {
175    match *md.level() {
176        Level::ERROR if is_first_party(md.target()) => {
177            if cooled_down(md) {
178                EventFilter::Event
179            } else {
180                EventFilter::Breadcrumb
181            }
182        }
183        Level::ERROR | Level::WARN | Level::INFO => EventFilter::Breadcrumb,
184        Level::DEBUG | Level::TRACE => EventFilter::Ignore,
185    }
186}

188/// The tag [`enrich`] lifts a `tracing` event's `code` field onto — the Rust half of
189/// what the Dart `beforeSend` fingerprints on, so both clients' events for one failure
190/// group and filter by the same stable [`hound_client::error`] code (`MEDIA024`, …)
191/// rather than by build-shifted frames or interpolated message text. The key is
192/// deliberately identical to the one `ui/lib/main.dart`'s `beforeSend` sets — one
193/// query has to return both halves of a failure.
194pub const ERROR_CODE_TAG: &str = "hound_error_code";
195
196/// Where `sentry-tracing` files a `tracing` event's structured fields.
197///
198/// This is the empirical half of the lift, and it is worth stating because it is not
199/// obvious from the API: `sentry-tracing`'s `event_from_event` puts every field that is
200/// not the message into `contexts["Rust Tracing Fields"]` as `Context::Other`, and
201/// promotes to `tags` ONLY fields whose name is prefixed `tags.` — which a plain
202/// `error!(code = %e.code(), …)` is not. So `before_send` ([`enrich`]) has to find the
203/// code in the context map — never in `extra`, never in `tags`. Lifting there rather
204/// than renaming the field at every callsite keeps the emission sites plain `tracing`
205/// and makes the tag automatic for callsites not yet written.
206const TRACING_FIELDS_CONTEXT: &str = "Rust Tracing Fields";
207
208/// Longest `code` value accepted as a tag. A tag is a GROUPING key, so an
209/// unexpectedly free-form `code` field (some future callsite passing a sentence) must
210/// not become a high-cardinality dimension; real codes are ≤ 12 characters.
211const MAX_ERROR_CODE_LEN: usize = 32;
212
213/// Pull the `code` field out of a `sentry-tracing` event, if it carries one that can
214/// serve as a tag. Accepts the field however `tracing` recorded it (`%`/`?` land as a
215/// string, a bare integer as a number); rejects empty, over-long and structured values.
216fn error_code(event: &Event<'static>) -> Option<String> {
217    let sentry::protocol::Context::Other(fields) = event.contexts.get(TRACING_FIELDS_CONTEXT)?
218    else {
219        return None;
220    };
221    let code = match fields.get("code")? {
222        serde_json::Value::String(s) => s.clone(),
223        serde_json::Value::Number(n) => n.to_string(),
224        // Bool/Null/Array/Object are not codes — a tag built from one would be noise.
225        _ => return None,
226    };
227    (!code.is_empty() && code.len() <= MAX_ERROR_CODE_LEN).then_some(code)
228}

230/// What the live client can say about itself at event time. `None` before the client
231/// exists — the reporter comes up first, deliberately, so a startup crash still
232/// reports.
233pub struct AppSnapshot {
234    pub in_call: bool,
235    /// The audio verdict + full debug report. `Some` only during a call: between
236    /// calls the report is a fixed "healthy" block that would bloat every event for
237    /// nothing.
238    pub audio: Option<AudioSnapshot>,
239    /// The video verdict + full debug report, on the same in-call-only terms as
240    /// [`AppSnapshot::audio`].
241    pub video: Option<VideoSnapshot>,
242}
243
244/// The audio pipeline's self-assessment, in both forms (see
245/// `hound_client::audio_report::Diagnosis`): a low-cardinality stage token fit for a
246/// tag, and the paste-able report behind it.
247pub struct AudioSnapshot {
248    pub stage: &'static str,
249    pub report: String,
250}
251
252/// The video pipeline's self-assessment, in the same two forms as
253/// [`AudioSnapshot`] (see `hound_client::video_report::Diagnosis`).
254pub struct VideoSnapshot {
255    pub stage: &'static str,
256    pub report: String,
257}

259/// Stamp an outgoing event with everything triage needs that the message itself
260/// can't carry. Runs as the client's `before_send`, i.e. after the scope and the
261/// default integrations have had their turn.
262pub fn enrich(
263    mut event: Event<'static>,
264    dist: Option<&str>,
265    app: Option<AppSnapshot>,
266) -> Event<'static> {
267    // `sentry-contexts` fills `server_name` with the machine's hostname whenever the
268    // client didn't set one, and exposes no opt-out — so strip it here, downstream of
269    // that. The Dart side reports an opaque user id and nothing else; the Rust side
270    // must not be the half that leaks a device name.
271    event.server_name = None;
272    // Unlike release/environment, `dist` has no `ClientOptions` field — it lives on
273    // the wire event, so it can only be set here.
274    if let Some(dist) = dist {
275        event.dist = Some(dist.to_owned().into());
276    }
277    let uptime = uptime();
278    event
279        .tags
280        .insert(LAUNCH_ID_TAG.to_owned(), launch_id().to_owned());
281    event
282        .tags
283        .insert("app_role".to_owned(), app_role().to_owned());
284    event
285        .tags
286        .insert("uptime".to_owned(), uptime_bucket(uptime).to_owned());
287    event
288        .extra
289        .insert("uptime_seconds".to_owned(), uptime.as_secs().into());
290    // The typed error code the emitting `error!` carried, promoted from a field
291    // nobody can query into a tag Sentry groups and filters on. See [`error_code`].
292    if let Some(code) = error_code(&event) {
293        event.tags.insert(ERROR_CODE_TAG.to_owned(), code);
294    }
295    if let Some(app) = app {
296        event
297            .tags
298            .insert("in_call".to_owned(), app.in_call.to_string());
299        if let Some(audio) = app.audio {
300            event
301                .tags
302                .insert("audio_stage".to_owned(), audio.stage.to_owned());
303            event
304                .extra
305                .insert("audio_report".to_owned(), audio.report.into());
306        }
307        if let Some(video) = app.video {
308            event
309                .tags
310                .insert("video_stage".to_owned(), video.stage.to_owned());
311            event
312                .extra
313                .insert("video_report".to_owned(), video.report.into());
314        }
315    }
316    event
317}

457    /// Build the event shape `sentry-tracing` produces for
458    /// `error!(code = %err.code(), …)`: non-message fields land in the
459    /// "Rust Tracing Fields" context, NOT in `tags`/`extra`.
460    fn event_with_fields(fields: &[(&str, serde_json::Value)]) -> Event<'static> {
461        let map = fields
462            .iter()
463            .map(|(k, v)| ((*k).to_owned(), v.clone()))
464            .collect();
465        let mut event = Event::default();
466        event.contexts.insert(
467            TRACING_FIELDS_CONTEXT.to_owned(),
468            sentry::protocol::Context::Other(map),
469        );
470        event
471    }
