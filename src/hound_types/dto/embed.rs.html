1//! Link-embed wire types — the unfurled preview of a URL found in a message.
2//!
3//! One shape serves three consumers, which is why it lives here rather than in
4//! the backend: the REST message endpoints return it, the realtime plane pushes
5//! it (`MessageEmbedsUpdated`), and the client mirrors it into Dart untouched.
6//!
7//! ## What the server promises the renderer
8//!
9//! Everything here is **already sanitised**. The backend is the only party that
10//! parses third-party HTML, so a client may render every string as plain text
11//! with no further escaping: text fields are entity-decoded, control characters
12//! are stripped, and lengths are capped ([`MAX_TITLE_CHARS`] and friends).
13//!
14//! Two deliberate omissions keep that promise cheap to hold:
15//!
16//! * **No HTML reaches a client.** oEmbed's `html` field and `twitter:player`
17//!   are iframe players. They are captured for the embed debugger
18//!   ([`EmbedDebugReport::oembed_html`]) and are never part of a [`LinkEmbed`],
19//!   so there is no path by which a renderer could be tempted to execute one.
20//! * **No third-party URL is fetched by a client.** Every media URL a
21//!   [`LinkEmbed`] carries points at *this deployment's* media proxy
22//!   ([`EmbedMedia::url`]); the origin URL travels alongside it in
23//!   [`EmbedMedia::source_url`] for attribution and for the debugger, not to be
24//!   loaded.
25
26use serde::{Deserialize, Serialize};

28/// Longest embed title a client will be handed, in `char`s.
29pub const MAX_TITLE_CHARS: usize = 256;
30/// Longest embed description a client will be handed, in `char`s.
31pub const MAX_DESCRIPTION_CHARS: usize = 1024;
32/// Longest short label (site name, author, provider), in `char`s.
33pub const MAX_LABEL_CHARS: usize = 128;
34/// Most images one embed carries — an `og:image` gallery is truncated to this.
35pub const MAX_EMBED_IMAGES: usize = 4;
36/// Most embeds one message carries, however many links it contains.
37pub const MAX_MESSAGE_EMBEDS: usize = 4;

39/// What a link turned out to be. Drives layout, not trust: every variant is
40/// rendered from the same sanitised fields.
41///
42/// Derived from oEmbed's `type` where one was found, else from the Open Graph
43/// `og:type` / Twitter `twitter:card` pair, else [`EmbedKind::Link`].
44#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize)]
45#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
46#[serde(rename_all = "snake_case")]
47pub enum EmbedKind {
48    /// Nothing more specific — a plain page with a title and maybe a summary.
49    /// The default, and what an unrecognised page unfurls to.
50    #[default]
51    Link,
52    /// `og:type=article`: has (or may have) a byline and a publication time.
53    Article,
54    /// oEmbed `photo`, `og:type` in the image family, or a URL that *is* an
55    /// image. Renders image-first, with the text demoted.
56    Photo,
57    /// oEmbed `video` / `og:video` / `twitter:card=player`. Renders as a poster
58    /// with a play affordance; see [`EmbedVideo::playable`] for whether the
59    /// client may play it in place.
60    Video,
61    /// `og:type=music.*` or an audio file.
62    Audio,
63    /// oEmbed `rich` — a player or widget whose real form is HTML this protocol
64    /// deliberately does not carry. Renders as a card.
65    Rich,
66    /// `og:type=profile` — a person or account page.
67    Profile,
68}

70/// One image, audio track, or video the embed refers to.
71///
72/// [`url`](Self::url) is the **only** URL a renderer should load. It addresses
73/// this deployment's media proxy, which re-fetches the origin under the same
74/// SSRF guard the unfurl itself ran under, caps the payload, and answers from a
75/// cache — so a client's IP never reaches the linked site and a hot embed
76/// cannot be turned into an amplifier against its origin.
77///
78/// It is **relative** on the wire (`/embed/media/{embed}/{key}`). The client
79/// resolves it against its configured API base at decode time, which keeps the
80/// backend from needing to know its own public URL.
81#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
82#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
83pub struct EmbedMedia {
84    /// Proxy URL, relative to the API base. The one to fetch.
85    #[cfg_attr(feature = "utoipa", schema(example = "/embed/media/018c.../image-0"))]
86    pub url: String,
87    /// Where the proxy got it. Attribution and debugging only — never fetched
88    /// by a client.
89    pub source_url: String,
90    /// The origin's `Content-Type`, once the proxy has seen the bytes. `None`
91    /// before the first proxy fetch: the unfurl learns a media URL from markup,
92    /// not from a request.
93    #[serde(default, skip_serializing_if = "Option::is_none")]
94    pub content_type: Option<String>,
95    /// Declared pixel width (`og:image:width`, oEmbed `thumbnail_width`, …).
96    /// Advisory: it lets a client reserve the right box before the bytes land,
97    /// and it is the *publisher's* claim, not a measurement.
98    #[serde(default, skip_serializing_if = "Option::is_none")]
99    pub width: Option<u32>,
100    /// Declared pixel height. Same caveat as [`width`](Self::width).
101    #[serde(default, skip_serializing_if = "Option::is_none")]
102    pub height: Option<u32>,
103    /// `og:image:alt` / `twitter:image:alt`, for screen readers.
104    #[serde(default, skip_serializing_if = "Option::is_none")]
105    pub alt: Option<String>,
106}

108/// The embed's video, when it has one.
109///
110/// Split from [`EmbedMedia`] for one field: [`playable`](Self::playable). Open
111/// Graph uses `og:video` for *both* a direct media file and an iframe player,
112/// distinguished only by `og:video:type`, and confusing the two is how a
113/// renderer ends up trying to decode a page of HTML as a video stream.
114#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
115#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
116pub struct EmbedVideo {
117    /// The video itself, proxied like every othermedia reference.
118    pub media: EmbedMedia,
119    /// Whether [`media`](Self::media) is a **direct video file** the client may
120    /// hand to a player.
121    ///
122    /// True only for a `video/*` content type. An iframe player
123    /// (`og:video:type=text/html`, `twitter:player`) is false: it is a web page,
124    /// and this client renders no remote HTML. A false video still carries its
125    /// poster and opens externally.
126    pub playable: bool,
127    /// Duration in seconds where the source declared one (oEmbed `duration`,
128    /// `og:video:duration`, `music:duration`).
129    #[serde(default, skip_serializing_if = "Option::is_none")]
130    pub duration_secs: Option<u32>,
131}

153/// The unfurled preview of one URL, ready to render.
154///
155/// Additive by construction: every field after the first three is optional or
156/// defaulted, so a client built against an older field set still decodes a
157/// newer server's embed (see the compatibility rules on the realtime protocol).
158#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
159#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
160pub struct LinkEmbed {
161    /// The URL **as it appeared in the message**. This is what a client
162    /// highlights in the body text, and what pairs an embed with its link.
163    pub request_url: String,
164    /// Where the fetch actually ended up: redirects followed, then `og:url` /
165    /// oEmbed `url` if the page declared a canonical form. What a click opens.
166    pub url: String,
167    #[serde(default)]
168    pub kind: EmbedKind,
169    #[serde(default, skip_serializing_if = "Option::is_none")]
170    pub title: Option<String>,
171    /// Summary text. Newlines survive (a publisher's paragraph breaks are
172    /// meaningful); every other control character does not.
173    #[serde(default, skip_serializing_if = "Option::is_none")]
174    pub description: Option<String>,
175    /// `og:site_name`, else the registrable domain.
176    #[serde(default, skip_serializing_if = "Option::is_none")]
177    pub site_name: Option<String>,
178    #[serde(default, skip_serializing_if = "Option::is_none")]
179    pub author: Option<EmbedAuthor>,
180    #[serde(default, skip_serializing_if = "Option::is_none")]
181    pub provider: Option<EmbedProvider>,
182    /// The page's `theme-color`, as `#rrggbb`. Tints the card's accent rail.
183    /// Validated to exactly that form — a client can parse it with confidence.
184    #[serde(default, skip_serializing_if = "Option::is_none")]
185    pub accent_color: Option<String>,
186    /// `article:published_time`, as unix milliseconds.
187    #[serde(default, skip_serializing_if = "Option::is_none")]
188    pub published_at: Option<i64>,
189    /// `og:locale`, e.g. `en_GB`.
190    #[serde(default, skip_serializing_if = "Option::is_none")]
191    pub locale: Option<String>,
192    /// Every `og:image` the page declared, in document order, capped at
193    /// [`MAX_EMBED_IMAGES`]. The first is the primary/poster; the rest render as
194    /// a gallery. Empty is normal.
195    #[serde(default)]
196    pub images: Vec<EmbedMedia>,
197    #[serde(default, skip_serializing_if = "Option::is_none")]
198    pub video: Option<EmbedVideo>,
199    #[serde(default, skip_serializing_if = "Option::is_none")]
200    pub audio: Option<EmbedMedia>,
201    /// The site's icon, proxied like any other media.
202    #[serde(default, skip_serializing_if = "Option::is_none")]
203    pub favicon: Option<EmbedMedia>,
204}
205
206impl LinkEmbed {
207    /// A bare embed for `request_url`, with nothing discovered yet. The starting
208    /// point every builder fills in.
209    pub fn new(request_url: impl Into<String>) -> Self {
210        let request_url = request_url.into();
211        Self {
212            url: request_url.clone(),
213            request_url,
214            kind: EmbedKind::Link,
215            title: None,
216            description: None,
217            site_name: None,
218            author: None,
219            provider: None,
220            accent_color: None,
221            published_at: None,
222            locale: None,
223            images: Vec::new(),
224            video: None,
225            audio: None,
226            favicon: None,
227        }
228    }
229
230    /// Whether there is anything worth drawing a card for.
231    ///
232    /// A page with no title, no summary and no media unfurls to a card that says
233    /// only what the link text already said — so callers drop it rather than
234    /// stack an empty box under every URL.
235    pub fn is_renderable(&self) -> bool {
236        self.title.is_some()
237            || self.description.is_some()
238            || !self.images.is_empty()
239            || self.video.is_some()
240            || self.audio.is_some()
241    }
242}

246/// Which metadata vocabulary a value came from. The debugger colour-codes by
247/// this, so "why did the title come out like that" is answerable at a glance.
248#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
249#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
250#[serde(rename_all = "snake_case")]
251pub enum EmbedSource {
252    /// An `og:*` meta property.
253    OpenGraph,
254    /// A `twitter:*` meta name.
255    TwitterCard,
256    /// The discovered oEmbed endpoint's JSON.
257    OEmbed,
258    /// Plain HTML: `<title>`, `<meta name=description>`, `<link rel=icon>`.
259    Html,
260    /// Derived by the server (the domain as a site name, a proxy URL, the
261    /// content type the proxy observed).
262    Derived,
263}

265/// One HTTP hop the unfurl made. Redirects appear as their own entries, which is
266/// what makes "this link goes somewhere else" visible in the debugger.
267#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
268#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
269pub struct EmbedFetchStep {
270    pub url: String,
271    /// The IP the guard resolved and pinned the connection to. Present even on a
272    /// refusal — a blocked hop's address is exactly what you want to see.
273    #[serde(default, skip_serializing_if = "Option::is_none")]
274    pub resolved_ip: Option<String>,
275    #[serde(default, skip_serializing_if = "Option::is_none")]
276    pub status: Option<u16>,
277    #[serde(default, skip_serializing_if = "Option::is_none")]
278    pub content_type: Option<String>,
279    /// Bytes read from this hop. `u32` rather than `u64` because every fetch is
280    /// capped in single-digit megabytes, and the narrower type crosses the
281    /// FRB bridge as a plain Dart `int`.
282    #[serde(default, skip_serializing_if = "Option::is_none")]
283    pub bytes: Option<u32>,
284    pub elapsed_ms: u32,
285    /// Why this hop stopped the unfurl, if it did.
286    #[serde(default, skip_serializing_if = "Option::is_none")]
287    pub error: Option<String>,
288}

299/// Everything the unfurl of one URL saw and decided — the embed debugger's
300/// payload.
301///
302/// Deliberately not what a message carries: this is the working-out, and some of
303/// it ([`oembed_html`](Self::oembed_html)) is markup that must never reach a
304/// renderer. The debugger shows it as **escaped source text**.
305#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
306#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
307pub struct EmbedDebugReport {
308    /// The URL as typed into the debugger.
309    pub request_url: String,
310    /// What a message would render, or `None` if this link unfurls to nothing.
311    #[serde(default, skip_serializing_if = "Option::is_none")]
312    pub embed: Option<LinkEmbed>,
313    /// Every HTTP hop, in order.
314    #[serde(default)]
315    pub steps: Vec<EmbedFetchStep>,
316    /// Every metadata key seen, whether or not it won.
317    #[serde(default)]
318    pub raw: Vec<EmbedRawField>,
319    /// The oEmbed endpoint discovered via `<link rel="alternate">`, if any.
320    #[serde(default, skip_serializing_if = "Option::is_none")]
321    pub oembed_endpoint: Option<String>,
322    /// oEmbed's `html` player markup, verbatim. **Never rendered** — shown as
323    /// source so an operator can see what a site is offering.
324    #[serde(default, skip_serializing_if = "Option::is_none")]
325    pub oembed_html: Option<String>,
326    /// Non-fatal observations: a dropped `javascript:` URL, an image over the
327    /// gallery cap, a malformed timestamp.
328    #[serde(default)]
329    pub warnings: Vec<String>,
330    /// Why there is no embed. `None` with no embed means "fetched fine, nothing
331    /// worth showing".
332    #[serde(default, skip_serializing_if = "Option::is_none")]
333    pub error: Option<String>,
334    /// Whether this answer came from the shared unfurl cache. The debugger always
335    /// bypasses it on the way in, so this reports what a *message* would have
336    /// hit.
337    pub cached: bool,
338    pub elapsed_ms: u32,
339}

349#[cfg(test)]
350mod tests {
351    use super::*;
352
353    /// The wire contract's additive rule, checked rather than asserted in a
354    /// comment: a payload written by a build that predates a field must still
355    /// decode, because embed rows outlive deployments and both halves of the
356    /// system are rolled independently.
357    #[test]
358    fn a_minimal_payload_decodes() {
359        let embed: LinkEmbed = serde_json::from_str(
360            r#"{"request_url":"https://example.com","url":"https://example.com"}"#,
361        )
362        .expect("the two required fields are enough");
363        assert_eq!(embed.kind, EmbedKind::Link);
364        assert!(embed.images.is_empty());
365        assert!(embed.title.is_none());
366    }
367
368    /// Absent fields stay absent on the wire. This is not cosmetic: a message
369    /// page carries up to four embeds per message, and serialising a dozen
370    /// `null`s each would be most of the payload.
371    #[test]
372    fn empty_fields_are_omitted() {
373        let json = serde_json::to_string(&LinkEmbed::new("https://example.com")).unwrap();
374        assert!(!json.contains("null"), "got: {json}");
375        assert!(json.contains("request_url"));
376    }
377
378    /// A card with nothing to say is not drawn — an empty box under every URL is
379    /// worse than no box.
380    #[test]
381    fn renderability_needs_something_to_render() {
382        let mut embed = LinkEmbed::new("https://example.com");
383        assert!(!embed.is_renderable());
384
385        embed.favicon = Some(EmbedMedia {
386            url: "/embed/media/x/favicon".to_owned(),
387            source_url: "https://example.com/favicon.ico".to_owned(),
388            content_type: None,
389            width: None,
390            height: None,
391            alt: None,
392        });
393        assert!(
394            !embed.is_renderable(),
395            "a favicon alone says nothing the link text did not"
396        );
397
398        embed.title = Some("Something".to_owned());
399        assert!(embed.is_renderable());
400    }
401
402    #[test]
403    fn the_kind_tag_is_snake_case_on_the_wire() {
404        assert_eq!(
405            serde_json::to_string(&EmbedKind::Article).unwrap(),
406            "\"article\""
407        );
408        assert_eq!(
409            serde_json::to_string(&EmbedSource::OpenGraph).unwrap(),
410            "\"open_graph\""
411        );
412    }
