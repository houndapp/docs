1//! Backend<->client wire DTOs. Field sets and serde attributes are byte-identical
2//! to what the backend used to declare locally (and the client used to
3//! hand-mirror) — this module is the single declaration both sides now share.
4//!
5//! Not every backend wire type lives here: types that embed a DB-entity enum
6//! (`sea_orm_active_enums::Permission`/`InstancePermission`) or that the backend
7//! deliberately keeps non-generic for OpenAPI schema-naming reasons
8//! (`FinishLoginResponse`, `ProfileResponse` — see
9//! `services/backend/src/api/auth.rs`) stay backend-local; see the crate
10//! extraction report for the full list of deliberate exclusions.

82/// Minimal information about a voice channel participant.
83///
84/// Clients use this to render participant avatars and names before any media
85/// arrives, and `sfu_peer_id` to map media/control peer ids to a user identity
86/// for peers who confirmed before the caller joined (their `voice:join`
87/// broadcast pre-dates the caller's socket subscription).
88#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
89#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
90pub struct VoicePeerInfo {
91    /// Stable UUID for this user. Matches the `user_id` in socket.io events.
92    pub user_id: Uuid,
93    /// Display name (from community profile, falls back to global profile).
94    pub name: Option<String>,
95    /// Avatar URL.
96    pub picture: String,
97    /// The peer's SFU/control-plane peer id (reported via `voice/confirm`).
98    /// `None` until they confirm.
99    pub sfu_peer_id: Option<u16>,
100}

102/// Response for joining a voice channel.
103///
104/// The client uses this to:
105/// 1. Present `relay_token` to the SFU on connect (proves they are allowed in).
106/// 2. Configure their iroh endpoint to use `relay_url` for NAT traversal.
107/// 3. Render the participant list with `peers` before any media arrives.
108#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
109#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
110pub struct VoiceJoinResponse {
111    /// The joining user's merged community profile (name, avatar, etc.).
112    pub profile: MergedProfile,
113    /// iroh relay URL. Clients must not hardcode this value — use whatever the
114    /// server returns here so that changing relay infrastructure only requires
115    /// a backend deploy, not a client release.
116    pub relay_url: String,
117    /// Short-lived HMAC-SHA256 token (15 min TTL) that proves this user may
118    /// connect to the SFU for the given voice channel.
119    ///
120    /// Format: `{user_id}:{space_id}:{expiry_unix_secs}:{hex_mac}`
121    pub relay_token: String,
122    /// Other participants already in the channel at the time of joining.
123    /// Populated from the in-memory voice-participant registry.
124    pub peers: Vec<VoicePeerInfo>,
125    /// iroh node ID of the SFU (hex-encoded public key).
126    ///
127    /// `None` when the backend is not configured with `SFU_NODE_ID` (e.g. in
128    /// development without a running SFU). Clients skip the direct QUIC
129    /// connection when this is absent.
130    #[serde(skip_serializing_if = "Option::is_none")]
131    pub sfu_node_id: Option<String>,
132}

134/// Response for re-minting a relay token mid-call (`voice/relay-token`).
135///
136/// Carries ONLY the fresh token — the client already has the relay URL and SFU
137/// node id from its `voice/join`, and the roster is maintained over the live
138/// control plane, so a refresh re-mints nothing else.
139#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
140#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
141pub struct RelayTokenResponse {
142    /// Short-lived HMAC-SHA256 token (15 min TTL), same format as
143    /// `VoiceJoinResponse::relay_token`: `{user_id}:{space_id}:{expiry}:{hex_mac}`.
144    pub relay_token: String,
145}

147/// Payload broadcast as `voice:join` when a user confirms their SFU connection.
148///
149/// Stored in-memory in the backend's `voice_participants` map and emitted as
150/// the socket event. `sfu_peer_id` is `None` in the brief window between
151/// `POST /voice/join` and `POST /voice/confirm`; it is always `Some` by the
152/// time the event is broadcast.
153#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
154#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
155pub struct VoiceJoinEvent {
156    pub user_id: Uuid,
157    pub space_id: Uuid,
158    pub community_id: Uuid,
159    pub name: Option<String>,
160    pub picture: String,
161    /// SFU-assigned peer_id. Set at `POST /voice/confirm`; always present in
162    /// the broadcast payload so recipients can immediately map SFU frames.
163    #[serde(skip_serializing_if = "Option::is_none")]
164    pub sfu_peer_id: Option<u16>,
165}

188/// Response for `GET /realtime/config`: how to reach the realtime plane.
189///
190/// The client reads this once at startup. `webtransport_url` present means "try
191/// WebTransport, fall back to the WebSocket on failure"; absent means the
192/// instance runs no WebTransport endpoint and the WebSocket is the only leg.
193/// `cert_hash` is the SHA-256 digest (dotted lowercase hex) the client must pin
194/// — present only for a self-signed development endpoint; a production endpoint
195/// carries a CA-issued certificate and omits it.
196#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
197#[cfg_attr(feature = "utoipa", derive(utoipa::ToSchema))]
198pub struct RealtimeConfig {
199    #[cfg_attr(feature = "utoipa", schema(example = "https://rt.hound.chat/realtime"))]
200    #[serde(default, skip_serializing_if = "Option::is_none")]
201    pub webtransport_url: Option<String>,
202    #[serde(default, skip_serializing_if = "Option::is_none")]
203    pub cert_hash: Option<String>,
204}

206/// One SFU node's registration status, as reported by `GET /internal/sfu/nodes`.
207#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
208pub struct SfuNodeStatus {
209    pub node_id: String,
210    pub draining: bool,
211    /// Seen within the registry's liveness TTL — eligible for selection.
212    pub live: bool,
213    /// Current active relay-session count, from the node's last registration
214    /// heartbeat. `#[serde(default)]` so a response from an older backend (or
215    /// a node that hasn't upgraded to report it yet) still deserializes —
216    /// `None` in both cases.
217    #[serde(default)]
218    pub sessions: Option<usize>,
219}
