1//! # hound-iroh
2//!
3//! The iroh lifetime rules shared by both ends of the wire. Every iroh `Endpoint`
4//! in the tree — the client's per-call endpoint (`crates/app/client`) and the SFU's
5//! service endpoint (`services/sfu`) — is owned through [`GracefulEndpoint`], so no
6//! exit path on either side can leave a socket aborted.
7
8use std::ops::Deref;
9use std::time::Duration;
10
11use iroh::Endpoint;
12use tokio::runtime::Handle;
13
14/// How long a close may run before we stop waiting on it. `Endpoint::close`
15/// retransmits close frames until the peer acknowledges them or QUIC's probe
16/// timeout (~3 s on a bad path) fires, so this is a backstop against a wedged
17/// socket pinning the endpoint (its UDP socket + actors) — not a normal bound.
18/// Elapsing means the close never finished, which is exactly the ungraceful abort
19/// this type exists to avoid, so it is worth a warning of its own.
20const CLOSE_TIMEOUT: Duration = Duration::from_secs(10);
21
22/// An iroh [`Endpoint`] that is always closed **gracefully**, on every path.
23///
24/// Two things go wrong when an endpoint is merely dropped:
25///
26/// * iroh aborts the socket and logs at `error!` ("Endpoint dropped without calling
27///   `Endpoint::close`") — which our tracing→Sentry bridges report as a crash-level
28///   event on both the client and the SFU.
29/// * The peer never sees a close frame, so it waits out an idle timeout before
30///   concluding the connection is gone.
31///
32/// The trap this guard closes is that iroh marks the socket closed only at the
33/// **very end** of `Endpoint::close` (after the close frames are acknowledged, or
34/// their probe timeout fires). A close that is awaited under a short timeout is
35/// therefore *worse* than useless: the elapsed timeout drops the close future
36/// mid-flight and the endpoint still counts as never-closed. So the guard closes
37/// either fully awaited ([`GracefulEndpoint::close`], for a caller that is about to
38/// exit) or detached onto a runtime ([`Drop`], for every other path — a `?` bail
39/// mid-setup, an aborted task, a teardown that must not block).
40///
41/// [`Deref`] gives the underlying endpoint's full API.
42pub struct GracefulEndpoint {
43    endpoint: Endpoint,
44    /// Where [`Drop`] detaches the close to. Captured up front because a drop can
45    /// run anywhere — including on a task the runtime just aborted, where
46    /// `Handle::current` is not available. Must outlive the drop (in practice a
47    /// process-lifetime runtime), or the detached close never runs.
48    runtime: Handle,
49}
50
51impl GracefulEndpoint {
52    /// Take ownership of `endpoint`, detaching its eventual close onto the CURRENT
53    /// runtime.
54    ///
55    /// # Panics
56    ///
57    /// If called outside a tokio runtime. An `Endpoint` can only be built from
58    /// async context, so a caller that has one is already inside one.
59    pub fn new(endpoint: Endpoint) -> Self {
60        Self::on_runtime(endpoint, Handle::current())
61    }
62
63    /// Take ownership of `endpoint`, detaching its eventual close onto `runtime`.
64    /// For callers that pin their work to a specific runtime rather than whichever
65    /// one happens to be current (the client's dedicated call runtime).
66    pub fn on_runtime(endpoint: Endpoint, runtime: Handle) -> Self {
67        Self { endpoint, runtime }
68    }
69
70    /// Close now, waiting for the flush. For a caller that is about to exit the
71    /// process (or otherwise stop polling), where a detached close would never get
72    /// to run. Idempotent, and leaves nothing for [`Drop`] to do.
73    pub async fn close(&self) {
74        close_bounded(&self.endpoint).await;
75    }
76}
77
78impl Deref for GracefulEndpoint {
79    type Target = Endpoint;
80
81    fn deref(&self) -> &Self::Target {
82        &self.endpoint
83    }
84}
85
86impl Drop for GracefulEndpoint {
87    fn drop(&mut self) {
88        if self.endpoint.is_closed() {
89            return; // already closed explicitly
90        }
91        // The clone is what keeps iroh's inner state — and its close check — alive
92        // until the detached close finishes; the endpoint is an `Arc` handle, so
93        // this drop is not the inner one.
94        let endpoint = self.endpoint.clone();
95        self.runtime
96            .spawn(async move { close_bounded(&endpoint).await });
97    }
98}
99
100async fn close_bounded(endpoint: &Endpoint) {
101    if tokio::time::timeout(CLOSE_TIMEOUT, endpoint.close())
102        .await
103        .is_err()
104    {
105        tracing::warn!(
106            timeout_s = CLOSE_TIMEOUT.as_secs(),
107            "iroh endpoint close timed out — the socket will be aborted ungracefully"
108        );
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115
116    async fn bound_endpoint() -> Endpoint {
117        Endpoint::builder(iroh::endpoint::presets::N0)
118            .relay_mode(iroh::RelayMode::Disabled)
119            .bind()
120            .await
121            .expect("bind iroh endpoint")
122    }
123
124    /// Wait out the detached close. Polls rather than sleeping a fixed span so the
125    /// test is not a race on a loaded machine.
126    async fn await_closed(endpoint: &Endpoint) {
127        for _ in 0..100 {
128            if endpoint.is_closed() {
129                return;
130            }
131            tokio::time::sleep(Duration::from_millis(50)).await;
132        }
133        panic!("endpoint still open 5s after the guard dropped");
134    }
135
136    /// Dropping the guard must leave the endpoint CLOSED, not aborted — iroh flips
137    /// `is_closed` only at the very end of `Endpoint::close`, and a drop that misses
138    /// it logs an `error!` and strands the peer on an idle timeout.
139    #[tokio::test]
140    async fn dropping_the_guard_closes_the_endpoint() {
141        let endpoint = bound_endpoint().await;
142        // A clone outlives the guard so the close can be observed: iroh's inner
143        // state lives until the LAST handle goes.
144        let probe = endpoint.clone();
145        drop(GracefulEndpoint::new(endpoint));
146        await_closed(&probe).await;
147    }
148
149    /// The awaited close is complete when it returns (nothing is left detached),
150    /// and the subsequent drop is a no-op.
151    #[tokio::test]
152    async fn explicit_close_completes_before_returning() {
153        let guard = GracefulEndpoint::new(bound_endpoint().await);
154        guard.close().await;
155        assert!(
156            guard.is_closed(),
157            "close returned with the socket still open"
158        );
159        drop(guard);
160    }
161}
