1//! Prometheus metrics: the process-wide recorder + render handle, and the
2//! HTTP request-instrumentation middleware. `/internal/metrics`'s handler
3//! lives in [`super::internal`] (co-located with the other internal-auth-gated
4//! routes); this module owns everything that feeds it.
5
6use std::time::Instant;
7
8use axum::extract::{MatchedPath, Request};
9use axum::middleware::Next;
10use axum::response::Response;
11use metrics_exporter_prometheus::{Matcher, PrometheusHandle};
12
13/// Explicit bucket boundaries (seconds) for `http_request_duration_seconds`.
14/// Without an override the exporter renders `histogram!` as a SUMMARY (rolling
15/// quantiles), which can't feed `histogram_quantile` or bucket-ratio SLIs —
16/// the buckets below make it a real Prometheus histogram (`_bucket` series).
17/// Spaced for web-request latency: dense sub-100ms, out to a 10s tail.
18const HTTP_DURATION_BUCKETS: &[f64] = &[
19    0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1.0, 2.5, 5.0, 10.0,
20];
21
22/// Installs the global recorder and returns its render handle. `OnceLock`
23/// makes repeated calls within one process return the same handle instead of
24/// erroring on a second `install_recorder()` — irrelevant in production
25/// (`Api::serve` calls this once) but keeps the test harness, which
26/// constructs a fresh [`super::Api`] per test function, safe regardless of
27/// how the test runner schedules those calls.
28pub(crate) fn handle() -> PrometheusHandle {
29    static HANDLE: std::sync::OnceLock<PrometheusHandle> = std::sync::OnceLock::new();
30    HANDLE
31        .get_or_init(|| {
32            let handle = metrics_exporter_prometheus::PrometheusBuilder::new()
33                .set_buckets_for_metric(
34                    Matcher::Full("http_request_duration_seconds".into()),
35                    HTTP_DURATION_BUCKETS,
36                )
37                .expect("non-empty http duration buckets")
38                .install_recorder()
39                .expect("installing prometheus recorder");
40            init_series();
41            handle
42        })
43        .clone()
44}
45
46/// Zero-initialise every low-cardinality series the process owns, so a fresh
47/// pod's first scrape already exposes them instead of each appearing lazily on
48/// its first event — dashboards and alerts can't `rate()`/threshold a series
49/// that doesn't exist yet. Runs once, inside [`handle`]'s recorder install (so
50/// the global recorder is already live). Every known label combination of a
51/// labelled counter is touched; gauges get their real startup value. Only
52/// bounded label sets appear here — never a per-request/route/user series.
53fn init_series() {
54    for method in ["magic", "passkey", "steam"] {
55        for outcome in ["success", "failure"] {
56            metrics::counter!("backend_auth_logins_total", "method" => method, "outcome" => outcome)
57                .increment(0);
58        }
59    }
60    for outcome in ["sent", "error"] {
61        metrics::counter!("backend_emails_sent_total", "outcome" => outcome).increment(0);
62    }
63    metrics::counter!("backend_socketio_connects_total").increment(0);
64    metrics::counter!("backend_socketio_disconnects_total").increment(0);
65    metrics::counter!("backend_messages_created_total").increment(0);
66
67    // Gauges — real startup value (a fresh pod has no live sockets and no SFU
68    // has registered yet; the registry recomputes `backend_sfu_nodes` on the
69    // first heartbeat).
70    metrics::gauge!("backend_socketio_connections").set(0.0);
71    metrics::gauge!("backend_sfu_nodes", "state" => "live").set(0.0);
72    metrics::gauge!("backend_sfu_nodes", "state" => "draining").set(0.0);
73}

75/// Records `http_requests_total{method,route,status}` and
76/// `http_request_duration_seconds{method,route,status}` for every request.
77///
78/// Must be mounted with [`axum::Router::route_layer`], not `.layer()`:
79/// `route_layer` only wraps matched-route dispatch, so [`MatchedPath`] is
80/// already populated when this runs; `.layer()` wraps the router itself
81/// (including unmatched requests), where it's always `None`. Labeling by the
82/// matched route template (e.g. `/space/{id}/message`) rather than the raw
83/// URI keeps the series count bounded — the raw path carries UUIDs that
84/// would mint a new series per request.
85pub(crate) async fn track_http_requests(request: Request, next: Next) -> Response {
86    let method = request.method().to_string();
87    let route = request
88        .extensions()
89        .get::<MatchedPath>()
90        .map(|p| p.as_str().to_owned())
91        .unwrap_or_else(|| "unmatched".to_owned());
92
93    let start = Instant::now();
94    let response = next.run(request).await;
95    let elapsed = start.elapsed();
96    let status = response.status().as_u16().to_string();
97
98    metrics::counter!(
99        "http_requests_total",
100        "method" => method.clone(), "route" => route.clone(), "status" => status.clone(),
101    )
102    .increment(1);
103    metrics::histogram!(
104        "http_request_duration_seconds",
105        "method" => method, "route" => route, "status" => status,
106    )
107    .record(elapsed.as_secs_f64());
108
109    response
110}
