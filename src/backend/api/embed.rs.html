1//! Link embeds: unfurling URLs found in messages into renderable previews, the
2//! media proxy that serves their images, and the debugger that explains both.
3//!
4//! ## The shape of the feature
5//!
6//! ```text
7//!   POST /space/{id}/message
8//!        │  stores + answers immediately
9//!        └─▶ spawn_unfurl ──▶ guard ──▶ fetch ──▶ parse ──▶ build ──▶ cache
10//!                                                              │
11//!                                        realtime: MessageEmbedsUpdated
12//!
13//!   GET /embed/media/{embed}/{slot}   the only URL a client loads
14//!   POST /embed/debug                 the same pipeline, traced, cache bypassed
15//! ```
16//!
17//! ## Module map
18//!
19//! * [`guard`] — the SSRF/size/redirect boundary. Read this one first; it is
20//!   what makes the rest of the feature safe to have.
21//! * [`parse`] — HTML → raw ordered metadata.
22//! * [`oembed`] — the oEmbed 1.0 document.
23//! * [`build`] — precedence and sanitisation → a [`LinkEmbed`].
24//! * [`store`] — the per-URL cache and the message→embed join.
25//! * [`media`] — the proxy route and its byte cache.
26//! * [`unfurl`] — the pipeline that drives all of the above.
27//!
28//! ## Trust boundary, stated once
29//!
30//! Everything upstream of [`build`] is attacker-controlled: the URL comes from a
31//! message, the page comes from whoever that URL resolves to, and the oEmbed
32//! endpoint is named by that page. Nothing downstream of [`build`] is: a
33//! [`LinkEmbed`] is entity-decoded, control-character-stripped, length-capped
34//! plain text plus `http(s)` URLs that point at this deployment. Clients are
35//! told they may render it as text, and this module is where that becomes true.
36

59/// Debug unfurls allowed per user per window.
60const DEBUG_RATE_LIMIT: u32 = 20;

69/// The media proxy, mounted outside the OpenAPI router.
70///
71/// Its response is raw bytes of a runtime-decided content type, which is not a
72/// body OpenAPI can usefully describe — and it is the one embed route that must
73/// answer an unauthenticated `<img>`-style request, so it does not belong beside
74/// the token-bearing surface either.
75pub(crate) fn media_router() -> axum::Router<Api> {
76    axum::Router::new().route(
77        "/embed/media/{embed_id}/{key}",
78        axum::routing::get(media::get_embed_media),
79    )
80}

95/// Unfurl one URL and report everything the pipeline saw.
96///
97/// Backs the embed debugger in the client's developer settings. Available to any
98/// signed-in user, deliberately: posting a message already makes this service
99/// fetch a URL of your choosing, so the debugger grants no capability that
100/// sending a message does not — it only *shows* you what happened.
101///
102/// What it does add is an **uncached** fetch, so it is rate limited per user.
103pub(crate) async fn post_embed_debug(
104    LoggedIn(claims): LoggedIn,
105    State(api): State<Api>,
106    Json(req): Json<EmbedDebugRequest>,
107) -> ApiResult<Json<EmbedDebugReport>> {
108    if !api.allow_debug_unfurl(claims.sub) {
109        return Err(ApiError::TooManyRequests());
110    }
111    metrics::counter!("backend_embed_debug_total").increment(1);
112    Ok(Json(api.debug_unfurl(&req.url).await))
113}

115/// A fixed-window request counter, keyed by user.
116///
117/// Fixed window rather than a token bucket because the thing being limited is a
118/// human clicking a button in a debug panel: the burst behaviour at a window
119/// edge that makes fixed windows a poor choice for API quotas is irrelevant at
120/// twenty-per-minute, and this costs one integer per active user.
121#[derive(Debug, Default)]
122pub(crate) struct DebugRateLimiter {
123    windows: RwLock<HashMap<Uuid, (i64, u32)>>,
124}
125
126impl Api {
127    fn allow_debug_unfurl(&self, user_id: Uuid) -> bool {
128        let now = chrono::Utc::now().timestamp_millis();
129        let mut windows = self
130            .embed_debug_limiter
131            .windows
132            .write()
133            .unwrap_or_else(|e| e.into_inner());
134        // Bounded by the number of users who have used the debugger this
135        // minute; sweeping here keeps a long-running process from accumulating
136        // an entry per user who ever did.
137        windows.retain(|_, (started, _)| now - *started < DEBUG_RATE_WINDOW_MS);
138        let entry = windows.entry(user_id).or_insert((now, 0));
139        if now - entry.0 >= DEBUG_RATE_WINDOW_MS {
140            *entry = (now, 0);
141        }
142        if entry.1 >= DEBUG_RATE_LIMIT {
143            return false;
144        }
145        entry.1 += 1;
146        true
147    }
148}
