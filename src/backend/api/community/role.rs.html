21/// Inclusive upper bound on a role name, in characters after trimming. Matches
22/// the community- and space-name bounds: all three are one sidebar row.
23pub(crate) const MAX_ROLE_NAME_CHARS: usize = 64;
24
25/// Trims and bounds a role name.
26fn validate_role_name(name: &str) -> ApiResult<String> {
27    crate::api::validate::bounded(
28        name,
29        MAX_ROLE_NAME_CHARS,
30        ApiError::EmptyRoleName,
31        ApiError::RoleNameTooLong {
32            max: MAX_ROLE_NAME_CHARS,
33        },
34    )
35}

156    /// Checks that `user_id` is not the last organiser in `community_id`.
157    /// Must be called inside a transaction to avoid a TOCTOU race.
158    pub(super) async fn assert_not_last_organiser(
159        &self,
160        conn: &impl sea_orm::ConnectionTrait,
161        community_id: Uuid,
162        user_id: Uuid,
163    ) -> ApiResult<()> {
164        let is_org = Role::find()
165            .filter(role::Column::CommunityId.eq(community_id))
166            .filter(role::Column::IsOrganisers.eq(true))
167            .inner_join(role_member::Entity)
168            .filter(role_member::Column::UserId.eq(user_id))
169            .count(conn)
170            .await?
171            > 0;
172        if is_org && self.count_organiser_members(conn, community_id).await? <= 1 {
173            return Err(ApiError::LastOrganiser);
174        }
175        Ok(())

192    pub(super) async fn find_role_permissions(&self, role_id: Uuid) -> DbResult<Vec<Permission>> {
193        role_permission::Entity::find()
194            .filter(role_permission::Column::RoleId.eq(role_id))

200    /// Every role of a community with its grants, in two queries.
201    ///
202    /// The listing is the one place a per-role `find_role_permissions` would be
203    /// an N+1: the whole community's `role_permission` rows come back in one
204    /// `IN` and are grouped in memory. `@organisers` still reports every
205    /// variant — it holds them by construction rather than by rows.
206    pub(super) async fn roles_to_responses(
207        &self,
208        community_id: Uuid,
209    ) -> DbResult<Vec<RoleResponse>> {
210        let roles = self.find_roles_by_community(community_id).await?;
211        let grants = role_permission::Entity::find()
212            .filter(role_permission::Column::RoleId.is_in(roles.iter().map(|r| r.id)))
213            .all(&self.db)
214            .await?;
215        Ok(roles
216            .into_iter()
217            .map(|role| RoleResponse {
218                permissions: if role.is_organisers {
219                    Permission::iter().collect()
220                } else {
221                    grants
222                        .iter()
223                        .filter(|grant| grant.role_id == role.id)
224                        .map(|grant| grant.permission)
225                        .collect()
226                },
227                id: role.id,
228                community_id: role.community_id,
229                name: role.name,
230                is_organisers: role.is_organisers,
231                is_everyone: role.is_everyone,
232            })
233            .collect())
234    }
235
236    pub(super) async fn role_to_response(&self, role: role::Model) -> DbResult<RoleResponse> {
237        let permissions = if role.is_organisers {
238            Permission::iter().collect()
239        } else {
240            self.find_role_permissions(role.id).await?
241        };
242        Ok(RoleResponse {
243            id: role.id,
244            community_id: role.community_id,
245            name: role.name,
246            is_organisers: role.is_organisers,
247            is_everyone: role.is_everyone,
248            permissions,
249        })
250    }

290/// Resolves `role_id` inside `community_id` and applies the role-hierarchy
291/// guard: a caller may only act on a role whose grants they already hold.
292///
293/// The community scoping is part of the check, not a nicety — `role_id` is a
294/// free path parameter, and the `{id}` guard authorised the caller for *this*
295/// community only.
296///
297/// See [`Api::can_manage_role`] for the rule; the refusal is
298/// [`ApiError::RoleAboveYou`] rather than a bare `Forbidden` so a client can
299/// distinguish "you can't manage roles" from "not that one".
300async fn manageable_role(
301    api: &Api,
302    user_id: Uuid,
303    community_id: Uuid,
304    role_id: Uuid,
305) -> ApiResult<role::Model> {
306    let role = api
307        .find_role_by_id(role_id)
308        .await?
309        .filter(|role| role.community_id == community_id)
310        .ok_or(ApiError::RoleNotFound(role_id))?;
311    if !api.can_manage_role(user_id, community_id, &role).await? {
312        return Err(ApiError::RoleAboveYou);
313    }
314    Ok(role)
315}

463/// Replaces all permissions for a role (full replace, not a patch).
464///
465/// Two containment checks, not one: the role's **current** grants must already
466/// be held by the caller ([`Api::can_manage_role`]), and so must every grant in
467/// the **new** set. Without the second, an editor of a lesser role could grant
468/// it anything and then join it.
469pub(crate) async fn put_role_permissions(
470    CommunityPermission {
471        claims,
472        community_id,
473        ..
474    }: CommunityPermission<perm::ManageRoles>,
475    Path(ids): Path<(Uuid, Uuid)>,
476    State(api): State<Api>,
477    Json(req): Json<PutRolePermissionsRequest>,
478) -> ApiResult<Json<RoleResponse>> {
479    let (_, role_id) = ids;
480    let role = manageable_role(&api, claims.sub, community_id, role_id).await?;
481    if role.is_organisers {
482        return Err(ApiError::Forbidden);
483    }
484    let held = api
485        .community_permissions_for(claims.sub, community_id)
486        .await?;
487    if let Some(missing) = req.permissions.iter().find(|p| !held.contains(p)) {
488        tracing::debug!(?missing, %role_id, "refused granting an unheld permission");
489        return Err(ApiError::PermissionNotHeld);
490    }
491
492    let txn = api.db.begin().await?;
493    role_permission::Entity::delete_many()
494        .filter(role_permission::Column::RoleId.eq(role_id))
495        .exec(&txn)
496        .await?;
497    for perm in &req.permissions {
498        role_permission::ActiveModel {
499            role_id: Set(role_id),
500            permission: Set(*perm),
501        }
502        .insert(&txn)
503        .await?;
504    }
505    txn.commit().await?;
506    api.emit_roles_changed(community_id).await;
507    Ok(Json(api.role_to_response(role).await?))
508}

529/// Adds a user to a role.
530///
531/// `@organisers` is organiser-only (the role-hierarchy guard): handing out the
532/// role that holds every permission cannot be reachable from `ManageRoles`.
533///
534/// The target must already be a member: a role row for someone outside the
535/// community grants them nothing they can act through, and would come back to
536/// life as a silent privilege the day they joined.
537pub(crate) async fn post_role_member(
538    CommunityPermission {
539        claims,
540        community_id,
541        ..
542    }: CommunityPermission<perm::ManageRoles>,
543    Path(ids): Path<(Uuid, Uuid)>,
544    State(api): State<Api>,
545    Json(req): Json<PostRoleMemberRequest>,
546) -> ApiResult<()> {
547    let (_, role_id) = ids;
548    let role = manageable_role(&api, claims.sub, community_id, role_id).await?;
549    if role.is_everyone {
550        return Err(ApiError::Forbidden);
551    }
552    // The check and the insert share a transaction: on the pool, a departure
553    // committing between them leaves exactly the row this refuses.
554    let txn = api.db.begin().await?;
555    let members = api
556        .community_members_among(&txn, community_id, [req.user_id])
557        .await?;
558    if !members.contains(&req.user_id) {
559        txn.rollback().await?;
560        return Err(ApiError::NotAMember(req.user_id));
561    }
562    api.insert_role_member(&txn, role_id, req.user_id).await?;
563    txn.commit().await?;
564    api.emit_roles_changed(community_id).await;
565    Ok(())
566}

590        ..

592    Path(ids): Path<(Uuid, Uuid, Uuid)>,

604            return Err(ApiError::LastOrganiser);
605        }
606    }
607    role_member::Entity::delete_many()
608        .filter(role_member::Column::RoleId.eq(role_id))
609        .filter(role_member::Column::UserId.eq(user_id))
610        .exec(&txn)
611        .await?;
612    txn.commit().await?;
613    api.emit_roles_changed(community_id).await;
614    Ok(())
615}
