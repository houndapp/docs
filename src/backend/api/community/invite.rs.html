1//! Community invites: minting shareable codes, and redeeming one.
2//!
3//! An invite is a row that carries its own whole policy — an optional expiry, an
4//! optional seat count, and a revocation stamp. Its **state** is therefore
5//! derived on every read ([`InviteState::of`]) rather than stored, so nothing
6//! has to run for an expired invite to stop working.
7//!
8//! The seat count is the one thing a derived read cannot enforce: two people
9//! holding the last seat both see "active". Redemption closes that with a
10//! conditional `UPDATE … WHERE uses < max_uses` inside the joining transaction,
11//! so the database, not the handler, decides who got the seat — see
12//! [`Api::claim_invite_seat`].
13//!
14//! Minting and listing are community-scoped (`ManageUsers`); redemption is not,
15//! because a caller holding a code does not yet know which community it opens.
16//! Those two routes live under `/invite` (see [`router`]) and need only a
17//! session.
18

40/// The code alphabet: Crockford base32 — the digits plus the letters, less
41/// `I`, `L`, `O` and `U`. The first three are dropped because they are the
42/// glyphs a human mis-copies as `1`/`0`; `U` because dropping it is what keeps
43/// a random code from spelling something. Lookups uppercase the input, so a
44/// code read off a screen and typed in lowercase still resolves.
45const CODE_ALPHABET: &[u8] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
46
47/// 8 characters of a 32-symbol alphabet: 40 bits, ~1.1e12 codes. Short enough
48/// to read aloud, wide enough that guessing is not an attack.
49const CODE_LEN: usize = 8;
50
51/// How many fresh codes a mint will try before giving up. A collision needs two
52/// draws from 1.1e12 to match, so more than one attempt is already
53/// astronomically unlikely — the retry exists so a collision is a retry rather
54/// than a 500, not because it is expected.
55const MINT_ATTEMPTS: usize = 8;
56
57/// Inclusive bound on `expires_in_secs`. An invite that outlives the reason it
58/// was made is the one that leaks a community; 30 days is the longest a link
59/// stays useful without someone re-deciding.
60pub(crate) const MAX_EXPIRY_SECS: u64 = 30 * 24 * 60 * 60;
61
62/// Inclusive bounds on `max_uses`. Zero would mint something already dead.
63pub(crate) const MIN_MAX_USES: u32 = 1;
64pub(crate) const MAX_MAX_USES: u32 = 1000;
65

75/// Why an invite does or does not work, derived from the row and the clock.
76///
77/// Checked in order of finality: a revoked invite reads `revoked` even if it
78/// had also expired, because revoking is the deliberate act and the one a
79/// moderator wants reflected back.
80#[derive(
81    Clone, Copy, Debug, PartialEq, Eq, serde::Serialize, serde::Deserialize, utoipa::ToSchema,
82)]
83#[serde(rename_all = "snake_case")]
84pub enum InviteState {
85    Active,
86    Expired,
87    Exhausted,
88    Revoked,
89}
90
91impl InviteState {
92    fn of(invite: &community_invite::Model, now_ms: i64) -> Self {
93        if invite.revoked_at.is_some() {
94            return Self::Revoked;
95        }
96        if invite.expires_at.is_some_and(|at| at <= now_ms) {
97            return Self::Expired;
98        }
99        if invite.max_uses.is_some_and(|max| invite.uses >= max) {
100            return Self::Exhausted;
101        }
102        Self::Active
103    }
104
105    /// The wire spelling, for the message half of an `invite_not_active` 400.
106    fn as_str(self) -> &'static str {
107        match self {
108            Self::Active => "active",
109            Self::Expired => "expired",
110            Self::Exhausted => "exhausted",
111            Self::Revoked => "revoked",
112        }
113    }
114}

198/// Mints a shareable invite code for the community.
199pub(crate) async fn post_invite(
200    CommunityPermission {
201        claims,
202        community_id,
203        ..
204    }: CommunityPermission<perm::ManageUsers>,
205    State(api): State<Api>,
206    Json(req): Json<PostInviteRequest>,
207) -> ApiResult<Json<InviteModel>> {
208    if let Some(secs) = req.expires_in_secs
209        && secs > MAX_EXPIRY_SECS
210    {
211        return Err(ApiError::InviteExpiryTooLong {
212            max_days: MAX_EXPIRY_SECS / (24 * 60 * 60),
213        });
214    }
215    if let Some(max_uses) = req.max_uses
216        && !(MIN_MAX_USES..=MAX_MAX_USES).contains(&max_uses)
217    {
218        return Err(ApiError::InvalidMaxUses {
219            min: MIN_MAX_USES,
220            max: MAX_MAX_USES,
221        });
222    }
223
224    let now_ms = chrono::Utc::now().timestamp_millis();
225    let expires_at = req
226        .expires_in_secs
227        .map(|secs| now_ms + (secs as i64) * 1_000);
228    let invite = api
229        .insert_invite(
230            community_id,
231            claims.sub,
232            now_ms,
233            expires_at,
234            req.max_uses.map(|n| n as i32),
235        )
236        .await?;
237
238    Ok(Json(api.render_invite(invite, now_ms)))
239}

256/// Lists every invite ever minted for the community, newest first.
257///
258/// Dead invites are listed too, carrying the `state` that killed them: a
259/// moderator looking at a link someone says "doesn't work" needs to see the
260/// revoked and expired ones, not an empty list. Which is also why it is paged —
261/// nothing is ever removed from this list.
262pub(crate) async fn get_invites(
263    CommunityPermission { community_id, .. }: CommunityPermission<perm::ManageUsers>,
264    State(api): State<Api>,
265    Query(pagination): Query<PaginationParameters>,
266) -> ApiResult<Json<Page<InviteModel>>> {
267    let now_ms = chrono::Utc::now().timestamp_millis();
268    let page = api
269        .find_invites_by_community(community_id, pagination)
270        .await?;
271    Ok(Json(page.map(|invite| api.render_invite(invite, now_ms))))
272}

360/// Redeems an invite code, joining the caller to the community it opens.
361///
362/// Idempotent for an existing member: they get the community back and the
363/// invite keeps its seat. Emits `MemberJoined` on an actual join.
364pub(crate) async fn post_invite_join(
365    LoggedIn(claims): LoggedIn,
366    Path(code): Path<String>,
367    State(api): State<Api>,
368) -> ApiResult<Json<CommunityModel>> {
369    let invite = api.find_invite_by_code(&code).await?;
370    let community_id = invite.community_id;
371
372    // Checked before the seat is claimed: re-redeeming must not burn one.
373    let already_member = api.is_community_member(community_id, claims.sub).await?;
374    if !already_member {
375        // One transaction: the ban check, the seat and the membership land
376        // together, so a failed join never leaves a consumed use behind and a
377        // ban landing mid-redemption is not stepped over.
378        let txn = api.db.begin().await?;
379        if api.is_banned(&txn, community_id, claims.sub).await? {
380            txn.rollback().await?;
381            return Err(ApiError::Banned);
382        }
383        let claimed = api
384            .claim_invite_seat(&txn, invite.id, chrono::Utc::now().timestamp_millis())
385            .await?;
386        if !claimed {
387            // Released before the re-read below: the write lock this
388            // transaction may hold would otherwise be in the way of it.
389            txn.rollback().await?;
390            // The conditional UPDATE matched nothing, so re-read to say which
391            // of the three ways it is dead.
392            let state = InviteState::of(
393                &api.find_invite_by_code(&code).await?,
394                chrono::Utc::now().timestamp_millis(),
395            );
396            return Err(ApiError::InviteNotActive(state.as_str()));
397        }
398        api.admit_member(&txn, community_id, claims.sub, &[])
399            .await?;
400        txn.commit().await?;
401
402        // The redeemer's own live sessions joined their rooms at admission, so
403        // the community they just joined is not among them.
404        api.grant_realtime_membership(claims.sub, community_id);
405        if let Some(member) = api.member_payload(community_id, claims.sub).await? {
406            api.emit_member_joined(community_id, member).await;
407        }
408    }
409
410    let community = api
411        .find_community(community_id)
412        .await?
413        .ok_or(ApiError::CommunityNotFound(community_id))?;
414    Ok(Json(
415        api.render_community(claims.sub, community, SpaceScope::Visible)
416            .await?,
417    ))
418}

422impl Api {
423    fn render_invite(&self, invite: community_invite::Model, now_ms: i64) -> InviteModel {
424        InviteModel {
425            url: format!("{}/invite/{}", self.public_url, invite.code),
426            state: InviteState::of(&invite, now_ms),
427            revoked: invite.revoked_at.is_some(),
428            id: invite.id,
429            community_id: invite.community_id,
430            code: invite.code,
431            created_by: invite.created_by,
432            created_at: invite.created_at,
433            expires_at: invite.expires_at,
434            // Both are non-negative by construction (`max_uses` is bounds-checked
435            // on mint, `uses` only ever increments from 0); the cast is the
436            // i32-column to wire-u32 crossing, not a conversion of meaning.
437            max_uses: invite.max_uses.map(|n| n as u32),
438            uses: invite.uses as u32,
439        }
440    }
441
442    /// Mints an invite, redrawing the code on the unique-index collision.
443    async fn insert_invite(
444        &self,
445        community_id: Uuid,
446        created_by: Uuid,
447        created_at: i64,
448        expires_at: Option<i64>,
449        max_uses: Option<i32>,
450    ) -> ApiResult<community_invite::Model> {
451        let mut last = None;
452        for _ in 0..MINT_ATTEMPTS {
453            let result = community_invite::ActiveModel {
454                id: Set(Uuid::now_v7()),
455                community_id: Set(community_id),
456                code: Set(mint_code()),
457                created_by: Set(Some(created_by)),
458                created_at: Set(created_at),
459                expires_at: Set(expires_at),
460                max_uses: Set(max_uses),
461                uses: Set(0),
462                revoked_at: Set(None),
463            }
464            .insert(&self.db)
465            .await;
466
467            match result {
468                Ok(invite) => return Ok(invite),
469                Err(err) if is_unique_violation(&err) => last = Some(err),
470                Err(err) => return Err(err.into()),
471            }
472        }
473        // Unreachable short of a broken RNG, and a 500 is the honest answer if
474        // it ever is: the caller cannot fix it by asking again.
475        Err(last
476            .map(ApiError::from)
477            .unwrap_or_else(|| ApiError::Database(DbErr::Custom("invite code mint".into()))))
478    }
479
480    /// Resolves a code to its invite. Case-insensitive: codes are stored
481    /// uppercase, so uppercasing the input is the whole normalisation.
482    async fn find_invite_by_code(&self, code: &str) -> ApiResult<community_invite::Model> {
483        let code = code.trim().to_uppercase();
484        community_invite::Entity::find()
485            .filter(community_invite::Column::Code.eq(code.clone()))
486            .one(&self.db)
487            .await?
488            .ok_or(ApiError::InviteNotFound(code))
489    }
490
491    async fn find_invites_by_community(
492        &self,
493        community_id: Uuid,
494        params: PaginationParameters,
495    ) -> DbResult<Page<community_invite::Model>> {
496        community_invite::Entity::find()
497            .filter(community_invite::Column::CommunityId.eq(community_id))
498            // UUIDv7 ids are time-ordered, so the primary key *is* the
499            // newest-first ordering — no second index needed.
500            .order_by_desc(community_invite::Column::Id)
501            .paginate(&self.db, params.per_page())
502            .to_page(params.page())
503            .await
504    }
505
506    async fn revoke_invite(&self, community_id: Uuid, invite_id: Uuid) -> ApiResult<()> {
507        let updated = community_invite::Entity::update_many()
508            .col_expr(
509                community_invite::Column::RevokedAt,
510                Expr::value(chrono::Utc::now().timestamp_millis()),
511            )
512            .filter(community_invite::Column::Id.eq(invite_id))
513            // Scoped to the community the guard authorised: an invite id from
514            // another community must not be revocable through this route.
515            .filter(community_invite::Column::CommunityId.eq(community_id))
516            .filter(community_invite::Column::RevokedAt.is_null())
517            .exec(&self.db)
518            .await?;
519        if updated.rows_affected == 0 {
520            // Either no such invite here, or it was already revoked. Both are
521            // "there is nothing to revoke"; an already-revoked invite is still
522            // findable, so distinguish them.
523            let exists = community_invite::Entity::find_by_id(invite_id)
524                .filter(community_invite::Column::CommunityId.eq(community_id))
525                .count(&self.db)
526                .await?
527                > 0;
528            if !exists {
529                return Err(ApiError::InviteNotFound(invite_id.to_string()));
530            }
531        }
532        Ok(())
533    }
534
535    /// Takes one seat on an invite, atomically. `false` means there was none to
536    /// take — expired, revoked, or the last one went to someone else.
537    ///
538    /// The whole policy lives in the `WHERE`, so the decision is the database's
539    /// single row-level write rather than a read the handler then acts on: two
540    /// concurrent redemptions of a one-seat invite serialise here, and exactly
541    /// one sees `rows_affected == 1`.
542    async fn claim_invite_seat(
543        &self,
544        conn: &impl sea_orm::ConnectionTrait,
545        invite_id: Uuid,
546        now_ms: i64,
547    ) -> DbResult<bool> {
548        let updated = community_invite::Entity::update_many()
549            .col_expr(
550                community_invite::Column::Uses,
551                Expr::col(community_invite::Column::Uses).add(1),
552            )
553            .filter(community_invite::Column::Id.eq(invite_id))
554            .filter(community_invite::Column::RevokedAt.is_null())
555            .filter(
556                Condition::any()
557                    .add(community_invite::Column::ExpiresAt.is_null())
558                    .add(community_invite::Column::ExpiresAt.gt(now_ms)),
559            )
560            .filter(
561                Condition::any()
562                    .add(community_invite::Column::MaxUses.is_null())
563                    .add(
564                        Expr::col(community_invite::Column::Uses)
565                            .lt(Expr::col(community_invite::Column::MaxUses)),
566                    ),
567            )
568            .exec(conn)
569            .await?;
570        Ok(updated.rows_affected > 0)
571    }
572}

574/// A fresh code. Uniform over [`CODE_ALPHABET`] — its length is a power of two,
575/// so index selection introduces no bias.
576fn mint_code() -> String {
577    use rand::Rng;
578    let mut rng = rand::rng();
579    (0..CODE_LEN)
580        .map(|_| CODE_ALPHABET[rng.random_range(0..CODE_ALPHABET.len())] as char)
581        .collect()
582}
583
584/// Whether a write failed on a unique index, on either backend.
585///
586/// Matched on the driver's SQLSTATE/extended result code rather than on the
587/// message text: `23505` is Postgres' `unique_violation`, `1555` and `2067` are
588/// SQLite's `SQLITE_CONSTRAINT_PRIMARYKEY` and `SQLITE_CONSTRAINT_UNIQUE`.
589fn is_unique_violation(err: &DbErr) -> bool {
590    let (DbErr::Query(sea_orm::RuntimeErr::SqlxError(err))
591    | DbErr::Exec(sea_orm::RuntimeErr::SqlxError(err))) = err
592    else {
593        return false;
594    };
595    err.as_database_error()
596        .and_then(|err| err.code())

639    /// Rewrites an invite's policy columns directly — the only way to reach a
640    /// past expiry or a used-up seat without waiting or looping.
641    async fn force(
642        api: &Api,
643        invite_id: Uuid,
644        expires_at: Option<Option<i64>>,
645        uses: Option<i32>,
646        revoked_at: Option<Option<i64>>,
647    ) {
648        let mut active = community_invite::ActiveModel {
649            id: Set(invite_id),
650            ..Default::default()
651        };
652        if let Some(expires_at) = expires_at {
653            active.expires_at = Set(expires_at);
654        }
655        if let Some(uses) = uses {
656            active.uses = Set(uses);
657        }
658        if let Some(revoked_at) = revoked_at {
659            active.revoked_at = Set(revoked_at);
660        }
661        active.update(&api.db).await.unwrap();
662    }
