26/// The rows [`Api::provision_default_layout`] creates, for callers that need to build a

52/// Resets a community to the same default layout `POST /community` produces, while
53/// preserving its ID and name. Every space, message, role, member, and community-scoped
54/// profile is discarded; the caller becomes the sole member and sole `@organisers` member.
55///
56/// The two membership controls that live outside the community's own rows are
57/// deliberately treated differently. **Invites are revoked** — every code
58/// minted before the reset opened a community that no longer exists, and a live
59/// link into a freshly emptied community is exactly the hole invites exist to
60/// close. **Bans are kept**: a ban is a standing refusal of a person, not a
61/// property of the layout, and a reset is not an amnesty. Lift one with
62/// `DELETE /community/{id}/ban/{user_id}`.
63pub(crate) async fn reset_community(
64    InstancePerm { claims, .. }: InstancePerm<perm::instance::ManageCommunities>,
65    Path(id): Path<Uuid>,
66    State(api): State<Api>,
67) -> ApiResult<Json<CommunityModel>> {
68    let txn = api.db.begin().await?;
69
70    let name = CommunitySettings::find()
71        .filter(community_settings::Column::CommunityId.eq(id))
72        .one(&txn)
73        .await?
74        .ok_or(ApiError::CommunityNotFound(id))?
75        .name;
76
77    let old_space_ids: Vec<Uuid> = CommunitySpace::find()
78        .filter(community_space::Column::CommunityId.eq(id))
79        .select_only()
80        .column(community_space::Column::Id)
81        .into_tuple()
82        .all(&txn)
83        .await?;
84
85    // Deleting spaces cascades their messages, message history, attachment links, and
86    // per-space overwrites; community_settings also cascades away via its default_space
87    // FK, which is why `name` was captured above before this point.
88    community_space::Entity::delete_many()
89        .filter(community_space::Column::CommunityId.eq(id))
90        .exec(&txn)
91        .await?;
92    // Idempotent: covers the case where the cascade above didn't already remove it.
93    community_settings::Entity::delete_many()
94        .filter(community_settings::Column::CommunityId.eq(id))
95        .exec(&txn)
96        .await?;
97    db_role::Entity::delete_many()
98        .filter(db_role::Column::CommunityId.eq(id))
99        .exec(&txn)
100        .await?;
101    community_member::Entity::delete_many()
102        .filter(community_member::Column::CommunityId.eq(id))
103        .exec(&txn)
104        .await?;
105    profile::Entity::delete_many()
106        .filter(profile::Column::CommunityId.eq(id))
107        .exec(&txn)
108        .await?;
109    // Revoked rather than deleted, like every other revocation: the row stays
110    // listable so a moderator can see what happened to a link that stopped
111    // working.
112    community_invite::Entity::update_many()
113        .col_expr(
114            community_invite::Column::RevokedAt,
115            sea_orm::sea_query::Expr::value(chrono::Utc::now().timestamp_millis()),
116        )
117        .filter(community_invite::Column::CommunityId.eq(id))
118        .filter(community_invite::Column::RevokedAt.is_null())
119        .exec(&txn)
120        .await?;
121
122    let layout = api
123        .provision_default_layout(&txn, id, name, Some(claims.sub))
124        .await?;
125    txn.commit().await?;
126
127    for old_space_id in old_space_ids {
128        api.emit_space_removed(id, old_space_id).await;
129    }
130    api.emit_space_created(&layout.general).await;
131    api.emit_space_created(&layout.voice).await;
132
133    Ok(Json(CommunityModel {
134        name: layout.settings.name,
135        // A reset discards the profile along with everything else: the settings
136        // row is dropped and re-provisioned, so description and icon start empty.
137        description: layout.settings.description,
138        icon: None,
139        default_space: layout.settings.default_space,
140        id,
141        spaces: Page::single_page(vec![
142            CommunitySpaceSummary::from(layout.general),
143            CommunitySpaceSummary::from(layout.voice),
144        ]),
145        // The resetting caller becomes the community's sole organiser.
146        permissions: Permission::iter().collect(),
147    }))
148}

150impl Api {
151    async fn insert_community_settings(
152        &self,
153        conn: &impl sea_orm::ConnectionTrait,
154        community_id: Uuid,
155        space_id: Uuid,
156        name: String,
157    ) -> DbResult<community_settings::Model> {
158        community_settings::ActiveModel {
159            id: Set(uuid::Uuid::now_v7()),
160            name: Set(name),
161            community_id: Set(community_id),
162            default_space: Set(space_id),
163            // A fresh community has no profile yet; `PATCH /community/{id}` fills it.
164            ..Default::default()
165        }
166        .insert(conn)
167        .await
168    }
169
170    /// Seeds the default layout for a community: `@organisers`/`@everyone` roles,
171    /// `@everyone`'s default channel permissions, `organiser` as a member and the
172    /// sole `@organisers` member, the `general`/`voice` spaces, and the
173    /// `community_settings` row pointing `default_space` at `general`.
174    ///
175    /// Shared by community creation and community reset so the two produce
176    /// identical layouts by construction.
177    ///
178    /// `organiser` is `None` only for the instance's boot-provisioned default
179    /// community ([`Api::provision_default_community`]), which is created before
180    /// any user exists to own it — the layout is otherwise identical, and the
181    /// first account to register adopts the empty `@organisers` role (see
182    /// [`Api::adopt_default_community_organiser`]).
183    pub(crate) async fn provision_default_layout(
184        &self,
185        conn: &impl sea_orm::ConnectionTrait,
186        community_id: Uuid,
187        name: String,
188        organiser: Option<Uuid>,
189    ) -> DbResult<DefaultLayout> {
190        // Seed special roles before join_community so @everyone exists when the
191        // organiser joins.
192        let organisers_role = self
193            .insert_role(conn, community_id, "@organisers", true, false)
194            .await?;
195        let everyone_role = self
196            .insert_role(conn, community_id, "@everyone", false, true)
197            .await?;
198        for perm in DEFAULT_EVERYONE_PERMISSIONS {
199            role_permission::ActiveModel {
200                role_id: Set(everyone_role.id),
201                permission: Set(perm),
202            }
203            .insert(conn)
204            .await?;
205        }
206        // Membership, profile, `@everyone` and `@organisers` in one admission —
207        // the same one every other join path takes.
208        if let Some(organiser) = organiser {
209            self.admit_member(conn, community_id, organiser, &[organisers_role.id])
210                .await?;
211        }
212        let general = self
213            .insert_community_space(
214                conn,
215                community_id,
216                "general".into(),
217                CommunitySpaceKind::Text,
218                false,
219            )
220            .await?;
221        let voice = self
222            .insert_community_space(
223                conn,
224                community_id,
225                "voice".into(),
226                CommunitySpaceKind::Voice,
227                false,
228            )
229            .await?;
230        let settings = self
231            .insert_community_settings(conn, community_id, general.id, name)
232            .await?;
233        Ok(DefaultLayout {
234            settings,
235            general,
236            voice,
237        })
238    }
239}

253    #[tokio::test]
254    async fn reset_restores_default_layout() {
255        use crate::api::space::{PostCommunitySpaceMessageRequest, PostCommunitySpaceRequest};
256
257        let (api, server) = crate::api::test().await;
258        let (owner, token, community) = setup(&api, &server).await;
259
260        let old_space_ids: std::collections::HashSet<Uuid> =
261            community.spaces.contents.iter().map(|s| s.id).collect();
262
263        // Dirty the community: a second member, messages, an extra space, an extra role.
264        let (_member, _member_token) = plain_member(&api, community.id).await;
265
266        server
267            .post(&format!("/space/{}/message", community.default_space))
268            .authorization_bearer(&token)
269            .json(&PostCommunitySpaceMessageRequest {
270                contents: "hello".into(),
271                attachments: vec![],
272            })
273            .await;
274
275        let extra_space: crate::api::space::CompleteCommunitySpace = server
276            .post("/space")
277            .authorization_bearer(&token)
278            .json(&PostCommunitySpaceRequest {
279                id: community.id,
280                name: "extra".into(),
281                kind: CommunitySpaceKind::Text,
282                is_private: None,
283            })
284            .await
285            .json();
286
287        server
288            .post(&format!("/community/{}/role", community.id))
289            .authorization_bearer(&token)
290            .json(&role::PostRoleRequest {
291                name: "extra-role".into(),
292            })
293            .await;
294
295        let response = server
296            .post(&format!("/community/{}/reset", community.id))
297            .authorization_bearer(&token)
298            .await;
299        assert_eq!(response.status_code(), 200);
300
301        let reset: CommunityModel = response.json();
302        assert_eq!(reset.id, community.id);
303        assert_eq!(reset.name, "test-community");
304
305        assert_eq!(reset.spaces.contents.len(), 2);
306        let general = reset
307            .spaces
308            .contents
309            .iter()
310            .find(|s| s.name == "general")
311            .expect("general space present");
312        let voice = reset
313            .spaces
314            .contents
315            .iter()
316            .find(|s| s.name == "voice")
317            .expect("voice space present");
318        assert!(matches!(general.kind, CommunitySpaceKind::Text));
319        assert!(matches!(voice.kind, CommunitySpaceKind::Voice));
320        assert!(!old_space_ids.contains(&general.id));
321        assert!(!old_space_ids.contains(&voice.id));
322        assert_ne!(general.id, extra_space.id);
323        assert_eq!(reset.default_space, general.id);
324
325        let roles = api.find_roles_by_community(community.id).await.unwrap();
326        assert_eq!(roles.len(), 2);
327        assert!(roles.iter().any(|r| r.is_organisers));
328        let everyone = roles.iter().find(|r| r.is_everyone).unwrap();
329
330        let everyone_perms: Vec<Permission> = RolePermission::find()
331            .filter(role_permission::Column::RoleId.eq(everyone.id))
332            .all(&api.db)
333            .await
334            .unwrap()
335            .into_iter()
336            .map(|p| p.permission)
337            .collect();
338        assert_eq!(everyone_perms.len(), 3);
339        for perm in [
340            Permission::ViewChannels,
341            Permission::SendMessages,
342            Permission::ReadHistory,
343        ] {
344            assert!(everyone_perms.contains(&perm));
345        }
346
347        let members = CommunityMember::find()
348            .filter(community_member::Column::CommunityId.eq(community.id))
349            .all(&api.db)
350            .await
351            .unwrap();
352        assert_eq!(members.len(), 1);
353        assert_eq!(members[0].user_id, owner.id);
354        assert!(api.is_organiser(community.id, owner.id).await.unwrap());
355
356        let messages = community_space_message::Entity::find()
357            .all(&api.db)
358            .await
359            .unwrap();
360        assert!(messages.is_empty());
361    }

377    /// A reset empties the community, so every code minted into the old one is
378    /// revoked: a live link into a freshly emptied community is the hole
379    /// invites exist to close. A ban is not about the layout, and survives —
380    /// a reset is not an amnesty.
381    #[tokio::test]
382    async fn reset_revokes_every_invite_and_keeps_every_ban() {
383        let (api, server) = crate::api::test().await;
384        let (_owner, token, community) = setup(&api, &server).await;
385        let (banned, _banned_token) = plain_member(&api, community.id).await;
386
387        let invite: InviteModel = server
388            .post(&format!("/community/{}/invite", community.id))
389            .authorization_bearer(&token)
390            .json(&serde_json::json!({}))
391            .await
392            .json();
393        assert_eq!(invite.state, InviteState::Active);
394        server
395            .post(&format!("/community/{}/ban/{}", community.id, banned.id))
396            .authorization_bearer(&token)
397            .json(&serde_json::json!({}))
398            .await;
399
400        let response = server
401            .post(&format!("/community/{}/reset", community.id))
402            .authorization_bearer(&token)
403            .await;
404        assert_eq!(response.status_code(), 200);
405
406        let invites: Page<InviteModel> = server
407            .get(&format!("/community/{}/invites", community.id))
408            .authorization_bearer(&token)
409            .await
410            .json();
411        assert_eq!(invites.contents.len(), 1, "the row stays listable");
412        assert_eq!(invites.contents[0].state, InviteState::Revoked);
413
414        assert!(
415            api.is_banned(&api.db, community.id, banned.id)
416                .await
417                .unwrap(),
418            "a ban is a standing refusal of a person, not a property of the layout"
419        );
420    }
