24/// The rows [`Api::provision_default_layout`] creates, for callers that need to build a
25/// response or emit events from them.
26pub(crate) struct DefaultLayout {
27    pub(super) settings: community_settings::Model,
28    pub(super) general: community_space::Model,
29    pub(super) voice: community_space::Model,
30}

34    path = "/{id}/reset",
35    tag = TAG,
36    params(
37        ("id" = Uuid, Path, description = "The community ID to reset"),
38    ),
39    responses(
40        (status = 200, description = "community reset to a fresh default layout", body = CommunityModel),
41        (status = 403, description = "missing ManageCommunities instance permission"),
42        (status = 404, description = "community not found"),

50/// Resets a community to the same default layout `POST /community` produces, while
51/// preserving its ID and name. Every space, message, role, member, and community-scoped
52/// profile is discarded; the caller becomes the sole member and sole `@organisers` member.
53pub(crate) async fn reset_community(
54    InstancePerm { claims, .. }: InstancePerm<perm::instance::ManageCommunities>,
55    Path(id): Path<Uuid>,
56    State(api): State<Api>,
57) -> ApiResult<Json<CommunityModel>> {
58    let txn = api.db.begin().await?;
59
60    let name = CommunitySettings::find()
61        .filter(community_settings::Column::CommunityId.eq(id))
62        .one(&txn)
63        .await?
64        .ok_or(ApiError::CommunityNotFound(id))?
65        .name;
66
67    let old_space_ids: Vec<Uuid> = CommunitySpace::find()
68        .filter(community_space::Column::CommunityId.eq(id))
69        .select_only()
70        .column(community_space::Column::Id)
71        .into_tuple()
72        .all(&txn)
73        .await?;
74
75    // Deleting spaces cascades their messages, message history, attachment links, and
76    // per-space overwrites; community_settings also cascades away via its default_space
77    // FK, which is why `name` was captured above before this point.
78    community_space::Entity::delete_many()
79        .filter(community_space::Column::CommunityId.eq(id))
80        .exec(&txn)
81        .await?;
82    // Idempotent: covers the case where the cascade above didn't already remove it.
83    community_settings::Entity::delete_many()
84        .filter(community_settings::Column::CommunityId.eq(id))
85        .exec(&txn)
86        .await?;
87    db_role::Entity::delete_many()
88        .filter(db_role::Column::CommunityId.eq(id))
89        .exec(&txn)
90        .await?;
91    community_member::Entity::delete_many()
92        .filter(community_member::Column::CommunityId.eq(id))
93        .exec(&txn)
94        .await?;
95    profile::Entity::delete_many()
96        .filter(profile::Column::CommunityId.eq(id))
97        .exec(&txn)
98        .await?;
99
100    let layout = api
101        .provision_default_layout(&txn, id, name, Some(claims.sub))
102        .await?;
103    txn.commit().await?;
104
105    for old_space_id in old_space_ids {
106        api.emit_space_removed(id, old_space_id).await;
107    }
108    api.emit_space_created(&layout.general).await;
109    api.emit_space_created(&layout.voice).await;
110
111    Ok(Json(CommunityModel {
112        name: layout.settings.name,
113        default_space: layout.settings.default_space,
114        id,
115        spaces: Page::single_page(vec![
116            CommunitySpaceSummary::from(layout.general),
117            CommunitySpaceSummary::from(layout.voice),
118        ]),
119    }))
120}

122impl Api {
123    async fn insert_community_settings(
124        &self,
125        conn: &impl sea_orm::ConnectionTrait,
126        community_id: Uuid,
127        space_id: Uuid,
128        name: String,
129    ) -> DbResult<community_settings::Model> {
130        community_settings::ActiveModel {
131            id: Set(uuid::Uuid::now_v7()),
132            name: Set(name),
133            community_id: Set(community_id),
134            default_space: Set(space_id),
135        }
136        .insert(conn)
137        .await
138    }
139
140    /// Seeds the default layout for a community: `@organisers`/`@everyone` roles,
141    /// `@everyone`'s default channel permissions, `organiser` as a member and the
142    /// sole `@organisers` member, the `general`/`voice` spaces, and the
143    /// `community_settings` row pointing `default_space` at `general`.
144    ///
145    /// Shared by community creation and community reset so the two produce
146    /// identical layouts by construction.
147    ///
148    /// `organiser` is `None` only for the instance's boot-provisioned default
149    /// community ([`Api::provision_default_community`]), which is created before
150    /// any user exists to own it — the layout is otherwise identical, and the
151    /// first account to register adopts the empty `@organisers` role (see
152    /// [`Api::adopt_default_community_organiser`]).
153    pub(crate) async fn provision_default_layout(
154        &self,
155        conn: &impl sea_orm::ConnectionTrait,
156        community_id: Uuid,
157        name: String,
158        organiser: Option<Uuid>,
159    ) -> DbResult<DefaultLayout> {
160        // Seed special roles before join_community so @everyone exists when the
161        // organiser joins.
162        let organisers_role = self
163            .insert_role(conn, community_id, "@organisers", true, false)
164            .await?;
165        let everyone_role = self
166            .insert_role(conn, community_id, "@everyone", false, true)
167            .await?;
168        for perm in DEFAULT_EVERYONE_PERMISSIONS {
169            role_permission::ActiveModel {
170                role_id: Set(everyone_role.id),
171                permission: Set(perm),
172            }
173            .insert(conn)
174            .await?;
175        }
176        // join_community adds the organiser as a member, creates their profile, and
177        // adds them to @everyone.
178        if let Some(organiser) = organiser {
179            self.join_community(conn, organiser, community_id).await?;
180            self.insert_role_member(conn, organisers_role.id, organiser)
181                .await?;
182        }
183        let general = self
184            .insert_community_space(
185                conn,
186                community_id,
187                "general".into(),
188                CommunitySpaceKind::Text,
189                false,
190            )
191            .await?;
192        let voice = self
193            .insert_community_space(
194                conn,
195                community_id,
196                "voice".into(),
197                CommunitySpaceKind::Voice,
198                false,
199            )
200            .await?;
201        let settings = self
202            .insert_community_settings(conn, community_id, general.id, name)
203            .await?;
204        Ok(DefaultLayout {
205            settings,
206            general,
207            voice,
208        })
209    }
210}

214    use sea_orm::{ColumnTrait, EntityTrait, QueryFilter, TransactionTrait};

223    #[tokio::test]
224    async fn reset_restores_default_layout() {
225        use crate::api::space::{PostCommunitySpaceMessageRequest, PostCommunitySpaceRequest};
226
227        let (api, server) = crate::api::test().await;
228        let (owner, token, community) = setup(&api, &server).await;
229
230        let old_space_ids: std::collections::HashSet<Uuid> =
231            community.spaces.contents.iter().map(|s| s.id).collect();
232
233        // Dirty the community: a second member, messages, an extra space, an extra role.
234        let member = api.insert_test_user().await;
235        let txn = api.db.begin().await.unwrap();
236        api.join_community(&txn, member.id, community.id)
237            .await
238            .unwrap();
239        txn.commit().await.unwrap();
240
241        server
242            .post(&format!("/space/{}/message", community.default_space))
243            .authorization_bearer(&token)
244            .json(&PostCommunitySpaceMessageRequest {
245                contents: "hello".into(),
246                attachments: vec![],
247            })
248            .await;
249
250        let extra_space: crate::api::space::CompleteCommunitySpace = server
251            .post("/space")
252            .authorization_bearer(&token)
253            .json(&PostCommunitySpaceRequest {
254                id: community.id,
255                name: "extra".into(),
256                kind: CommunitySpaceKind::Text,
257                is_private: None,
258            })
259            .await
260            .json();
261
262        server
263            .post(&format!("/community/{}/role", community.id))
264            .authorization_bearer(&token)
265            .json(&role::PostRoleRequest {
266                name: "extra-role".into(),
267            })
268            .await;
269
270        let response = server
271            .post(&format!("/community/{}/reset", community.id))
272            .authorization_bearer(&token)
273            .await;
274        assert_eq!(response.status_code(), 200);
275
276        let reset: CommunityModel = response.json();
277        assert_eq!(reset.id, community.id);
278        assert_eq!(reset.name, "test-community");
279
280        assert_eq!(reset.spaces.contents.len(), 2);
281        let general = reset
282            .spaces
283            .contents
284            .iter()
285            .find(|s| s.name == "general")
286            .expect("general space present");
287        let voice = reset
288            .spaces
289            .contents
290            .iter()
291            .find(|s| s.name == "voice")
292            .expect("voice space present");
293        assert!(matches!(general.kind, CommunitySpaceKind::Text));
294        assert!(matches!(voice.kind, CommunitySpaceKind::Voice));
295        assert!(!old_space_ids.contains(&general.id));
296        assert!(!old_space_ids.contains(&voice.id));
297        assert_ne!(general.id, extra_space.id);
298        assert_eq!(reset.default_space, general.id);
299
300        let roles = api.find_roles_by_community(community.id).await.unwrap();
301        assert_eq!(roles.len(), 2);
302        assert!(roles.iter().any(|r| r.is_organisers));
303        let everyone = roles.iter().find(|r| r.is_everyone).unwrap();
304
305        let everyone_perms: Vec<Permission> = RolePermission::find()
306            .filter(role_permission::Column::RoleId.eq(everyone.id))
307            .all(&api.db)
308            .await
309            .unwrap()
310            .into_iter()
311            .map(|p| p.permission)
312            .collect();
313        assert_eq!(everyone_perms.len(), 3);
314        for perm in [
315            Permission::ViewChannels,
316            Permission::SendMessages,
317            Permission::ReadHistory,
318        ] {
319            assert!(everyone_perms.contains(&perm));
320        }
321
322        let members = CommunityMember::find()
323            .filter(community_member::Column::CommunityId.eq(community.id))
324            .all(&api.db)
325            .await
326            .unwrap();
327        assert_eq!(members.len(), 1);
328        assert_eq!(members[0].user_id, owner.id);
329        assert!(api.is_organiser(community.id, owner.id).await.unwrap());
330
331        let messages = community_space_message::Entity::find()
332            .all(&api.db)
333            .await
334            .unwrap();
335        assert!(messages.is_empty());
336    }
