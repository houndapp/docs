1//! The display identity a community shows for a user: merged profile name and
2//! resolved avatar URL, keyed by user id.
3//!
4//! Two surfaces render people — the member roster and the ban list — and both
5//! face the same problem: the set of users they were asked about is not
6//! guaranteed to have a profile row each. A ban outlives the account it names
7//! (it is what stops a deleted-and-recreated user walking back in), and a
8//! roster row whose profile is missing is still a member the caller has to be
9//! shown. Dropping such a row silently shortens a page below its own `total`,
10//! so **every** requested id comes back, a placeholder standing in for what
11//! could not be resolved.
12
13use std::collections::HashMap;
14
15use uuid::Uuid;
16
17use crate::api::{Api, data::prelude::*};
18
19/// One user's community-scoped display identity.
20#[derive(Clone, Debug)]
21pub(super) struct Identity {
22    pub(super) user_id: Uuid,
23    /// Community profile name, falling back to the global one; `None` when the
24    /// user has set neither — or when there is no profile left to read.
25    pub(super) name: Option<String>,
26    /// Resolved public avatar URL: a real upload, the generated placeholder
27    /// every profile gets, or empty where no profile resolved at all.
28    pub(super) picture: String,
29}
30
31impl Api {
32    /// The display identity of every user in `user_ids`, keyed by id.
33    ///
34    /// One query regardless of how many are asked for. The map always contains
35    /// every requested id: an id with no profile gets a placeholder rather than
36    /// being dropped, so a caller can render exactly the rows it asked about.
37    pub(super) async fn community_identities(
38        &self,
39        community_id: Uuid,
40        user_ids: Vec<Uuid>,
41    ) -> DbResult<HashMap<Uuid, Identity>> {
42        if user_ids.is_empty() {
43            return Ok(HashMap::new());
44        }
45        let profiles = self
46            .find_merged_profiles(user_ids.iter().copied(), community_id)
47            .await?;
48
49        Ok(user_ids
50            .into_iter()
51            .map(|user_id| {
52                let profile = profiles.iter().find(|p| p.user_id == user_id);
53                (
54                    user_id,
55                    Identity {
56                        user_id,
57                        name: profile.and_then(|p| p.name.clone()),
58                        picture: profile.map(|p| p.picture.clone()).unwrap_or_default(),
59                    },
60                )
61            })
62            .collect())
63    }
64}
