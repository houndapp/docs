1//! Compiling a [`hound_search::Query`] to one statement.
2//!
3//! The only raw SQL in the backend, and it is raw because neither text engine
4//! has a `sea_query` expression: PGroonga's `&@` and FTS5's `MATCH` are
5//! operators against index-backed pseudo-columns. Everything else is written
6//! by hand alongside them rather than half in the query builder.
7//!
8//! ## The two dialects
9//!
10//! Postgres matches per term — `search_text &@ $term`, one predicate each,
11//! `NOT (…)` for a negation — and takes its relevance from
12//! `pgroonga_score(tableoid, ctid)`.
13//!
14//! SQLite has one `MATCH` string for the whole query, because FTS5 admits a
15//! single match expression per table reference: positives joined by `AND`,
16//! each negation appended as `NOT "…"`. With no positive term the polarity
17//! flips and the whole thing becomes a `rowid NOT IN` of the negations OR-ed
18//! together. Relevance is `-bm25(…)` (FTS5's bm25 is negative, best-first),
19//! which needs the FTS table joined rather than used as a subquery.
20//!
21//! ## Terms are phrases
22//!
23//! A term's words must appear adjacent, quoted or not. That is already what
24//! `&@` does with a multi-token argument under a position-recording tokenizer,
25//! and what a quoted FTS5 string does, so the two agree without either side
26//! being special-cased. `Term::phrase` therefore does not change the SQL — it
27//! is a fact about how the user wrote it, which the UI's highlighting uses.
28//!
29//! ## Nothing is post-filtered
30//!
31//! Every constraint, permissions included, is a predicate here. A row this
32//! statement returns is a row the caller may read, so hydration never has to
33//! drop one — and a bug in hydration cannot leak a message.
34

48/// Everything the statement is built from. The spaces are already
49/// permission-resolved; this module never consults the ACL.
50pub(super) struct Plan<'a> {
51    pub spaces: &'a [Uuid],
52    pub query: &'a Query,
53    pub sort: SearchSort,
54    pub cursor: Option<&'a Cursor>,
55    pub limit: u32,
56    /// `(lo, hi)` from [`Query::id_bounds`]: `id >= lo`, `id < hi`.
57    pub bounds: (Option<Uuid>, Option<Uuid>),
58}
59
60/// The statement, or `None` when the query is structurally unsatisfiable and
61/// the database should not be asked at all.
62pub(super) fn statement(backend: DbBackend, plan: &Plan<'_>) -> Option<Statement> {
63    let text = TextPlan::of(plan.query);
64    if matches!(text, TextPlan::Never) {
65        return None;
66    }
67    let mut binder = Binder::new(backend);
68    let sql = match backend {
69        DbBackend::Postgres => postgres_sql(&mut binder, plan, &text),
70        _ => sqlite_sql(&mut binder, plan, &text),
71    };
72    Some(Statement::from_sql_and_values(backend, sql, binder.values))
73}

77/// The text terms, reduced to what each engine can be handed.
78#[derive(Debug, PartialEq)]
79enum TextPlan<'a> {
80    /// No text terms at all: every message in scope is a candidate.
81    Any,
82    Terms {
83        positive: Vec<&'a str>,
84        negative: Vec<&'a str>,
85    },
86    /// A positive term with nothing a tokenizer would keep (`"+++"`). No
87    /// message can contain it, so the search is over before it starts.
88    Never,
89}
90
91impl<'a> TextPlan<'a> {
92    fn of(query: &'a Query) -> Self {
93        let mut positive = Vec::new();
94        let mut negative = Vec::new();
95        for term in &query.terms {
96            // Both tokenizers keep alphanumeric runs and discard everything
97            // between them, so a term with no alphanumeric character indexes
98            // as nothing on either engine. Checked here rather than per
99            // dialect so the two agree about it.
100            if !term.text.chars().any(char::is_alphanumeric) {
101                if !term.negated {
102                    return TextPlan::Never;
103                }
104                // "must not contain something unindexable" excludes nothing.
105                continue;
106            }
107            if term.negated {
108                negative.push(term.text.as_str());
109            } else {
110                positive.push(term.text.as_str());
111            }
112        }
113        if positive.is_empty() && negative.is_empty() {
114            return TextPlan::Any;
115        }
116        TextPlan::Terms { positive, negative }
117    }
118}

120/// One term as an FTS5 phrase literal.
121///
122/// Re-tokenized here rather than quoted verbatim: `unicode61` splits on every
123/// non-alphanumeric character, so joining the alphanumeric runs with spaces
124/// inside one quoted phrase is exactly what FTS5 would have parsed — and it
125/// makes the literal unquotable-in, since no `"` can survive the split.
126fn fts_phrase(text: &str) -> Option<String> {
127    let tokens: Vec<&str> = text
128        .split(|c: char| !c.is_alphanumeric())
129        .filter(|token| !token.is_empty())
130        .collect();
131    (!tokens.is_empty()).then(|| format!("\"{}\"", tokens.join(" ")))
132}
133
134/// The FTS5 match expression, and whether membership means include or exclude.
135///
136/// `NOT` binds tighter than `AND` in FTS5, so `"a" AND "b" NOT "c"` parses as
137/// `"a" AND ("b" NOT "c")` — which is still "has a, has b, lacks c".
138fn fts_match(text: &TextPlan<'_>) -> Option<(String, bool)> {
139    let TextPlan::Terms { positive, negative } = text else {
140        return None;
141    };
142    let negatives: Vec<String> = negative.iter().copied().filter_map(fts_phrase).collect();
143    if positive.is_empty() {
144        // Negation-only: match anything the user asked to exclude, then
145        // exclude those rows.
146        return (!negatives.is_empty()).then(|| (negatives.join(" OR "), false));
147    }
148    let mut expression = positive
149        .iter()
150        .copied()
151        .filter_map(fts_phrase)
152        .collect::<Vec<_>>()
153        .join(" AND ");
154    for negation in negatives {
155        expression.push_str(" NOT ");
156        expression.push_str(&negation);
157    }
158    Some((expression, true))
159}

161// ── binding ────────────────────────────────────────────────────────────────

163/// Accumulates bind values and hands back the placeholder each one is
164/// addressed by — `$1`-style on Postgres, positional `?` elsewhere.
165/// `Statement::from_sql_and_values` passes the SQL to the driver verbatim, so
166/// the dialect's own syntax has to be written here.
167struct Binder {
168    backend: DbBackend,
169    values: Vec<Value>,
170}
171
172impl Binder {
173    fn new(backend: DbBackend) -> Self {
174        Binder {
175            backend,
176            values: Vec::new(),
177        }
178    }
179
180    fn bind(&mut self, value: impl Into<Value>) -> String {
181        self.values.push(value.into());
182        match self.backend {
183            DbBackend::Postgres => format!("${}", self.values.len()),
184            _ => "?".to_string(),
185        }
186    }
187
188    /// A comma-separated placeholder list for an `IN (…)`.
189    ///
190    /// `IN` rather than Postgres's `= ANY($1)`: an array bind needs
191    /// sea-query's `postgres-array` support, and the lists here are bounded by
192    /// how many channels a person can read.
193    fn bind_all(&mut self, values: impl IntoIterator<Item = impl Into<Value>>) -> String {
194        values
195            .into_iter()
196            .map(|value| self.bind(value))
197            .collect::<Vec<_>>()
198            .join(", ")
199    }
200}

202// ── the shared predicates ──────────────────────────────────────────────────
203
204/// Everything both dialects express identically, as `AND`-able clauses over
205/// the message alias `m`.
206fn common_predicates(binder: &mut Binder, plan: &Plan<'_>) -> Vec<String> {
207    let query = plan.query;
208    let mut clauses = vec![format!(
209        "m.space_id IN ({})",
210        binder.bind_all(plan.spaces.iter().copied())
211    )];
212
213    let (lo, hi) = plan.bounds;
214    if let Some(lo) = lo {
215        clauses.push(format!("m.id >= {}", binder.bind(lo)));
216    }
217    if let Some(hi) = hi {
218        clauses.push(format!("m.id < {}", binder.bind(hi)));
219    }
220
221    if !query.from.is_empty() {
222        clauses.push(format!(
223            "m.user_id IN ({})",
224            binder.bind_all(query.from.iter().copied())
225        ));
226    }
227
228    if !query.mentions.is_empty() {
229        clauses.push(format!(
230            "EXISTS (SELECT 1 FROM community_space_message_reference r \
231             WHERE r.message_id = m.id AND r.kind = 'user' AND r.target_id IN ({}))",
232            binder.bind_all(query.mentions.iter().copied())
233        ));
234    }
235
236    // One EXISTS per facet: `has:` repeats are AND, and a single IN would read
237    // as OR.
238    for (index, kind) in query.has.iter().enumerate() {
239        clauses.push(format!(
240            "EXISTS (SELECT 1 FROM community_space_message_kind k{index} \
241             WHERE k{index}.message_id = m.id AND k{index}.kind = {})",
242            binder.bind(kind.as_str())
243        ));
244    }
245
246    match query.edited {
247        Some(true) => clauses.push("m.edited_at IS NOT NULL".to_string()),
248        Some(false) => clauses.push("m.edited_at IS NULL".to_string()),
249        None => {}
250    }
251
252    clauses
253}

255/// The `newest` keyset, pushed into the inner query so the primary key can
256/// serve it.
257fn newest_keyset(binder: &mut Binder, plan: &Plan<'_>) -> Option<String> {
258    let cursor = plan.cursor?;
259    matches!(plan.sort, SearchSort::Newest).then(|| format!("m.id < {}", binder.bind(cursor.id)))
260}

262/// Wraps the scored inner query so the `relevant` keyset can compare against
263/// the computed score, and orders by it.
264fn relevance_wrapper(binder: &mut Binder, plan: &Plan<'_>, inner: String) -> String {
265    let mut sql = format!("SELECT hits.id AS id, hits.score AS score FROM ({inner}) AS hits");
266    if let Some(cursor) = plan.cursor {
267        // A cursor minted under `newest` carries no score; treat it as the
268        // id-only tiebreak it is rather than refusing the page.
269        let score = cursor.score.unwrap_or(f64::MAX);
270        let score_a = binder.bind(score);
271        let score_b = binder.bind(score);
272        let id = binder.bind(cursor.id);
273        sql.push_str(&format!(
274            " WHERE hits.score < {score_a} OR (hits.score = {score_b} AND hits.id < {id})"
275        ));
276    }
277    sql.push_str(&format!(
278        " ORDER BY hits.score DESC, hits.id DESC LIMIT {}",
279        plan.limit
280    ));
281    sql
282}

284/// `LIMIT` is interpolated, not bound: it is a `u32` the handler has already
285/// checked against [`super::LIMIT_RANGE`], and Postgres will not take an
286/// unsigned bind for it.
287fn limit_clause(limit: u32) -> String {
288    format!(" LIMIT {limit}")
289}

291// ── Postgres ───────────────────────────────────────────────────────────────
292
293fn postgres_sql(binder: &mut Binder, plan: &Plan<'_>, text: &TextPlan<'_>) -> String {
294    let mut clauses = common_predicates(binder, plan);
295    if let TextPlan::Terms { positive, negative } = text {
296        for term in positive {
297            clauses.push(format!("m.search_text &@ {}", binder.bind(*term)));
298        }
299        for term in negative {
300            clauses.push(format!("NOT (m.search_text &@ {})", binder.bind(*term)));
301        }
302    }
303
304    match plan.sort {
305        SearchSort::Newest => {
306            if let Some(keyset) = newest_keyset(binder, plan) {
307                clauses.push(keyset);
308            }
309            format!(
310                "SELECT m.id AS id, pgroonga_score(m.tableoid, m.ctid) AS score \
311                 FROM community_space_message m WHERE {} ORDER BY m.id DESC{}",
312                clauses.join(" AND "),
313                limit_clause(plan.limit)
314            )
315        }
316        SearchSort::Relevant => {
317            let inner = format!(
318                "SELECT m.id AS id, pgroonga_score(m.tableoid, m.ctid) AS score \
319                 FROM community_space_message m WHERE {}",
320                clauses.join(" AND ")
321            );
322            relevance_wrapper(binder, plan, inner)
323        }
324    }
325}

327// ── SQLite ─────────────────────────────────────────────────────────────────
328
329const FTS: &str = "community_space_message_fts";
330
331fn sqlite_sql(binder: &mut Binder, plan: &Plan<'_>, text: &TextPlan<'_>) -> String {
332    match plan.sort {
333        SearchSort::Newest => {
334            let mut clauses = common_predicates(binder, plan);
335            if let Some((expression, include)) = fts_match(text) {
336                let negate = if include { "" } else { "NOT " };
337                clauses.push(format!(
338                    "m.rowid {negate}IN (SELECT rowid FROM {FTS} WHERE {FTS} MATCH {})",
339                    binder.bind(expression)
340                ));
341            }
342            if let Some(keyset) = newest_keyset(binder, plan) {
343                clauses.push(keyset);
344            }
345            format!(
346                "SELECT m.id AS id, 0.0 AS score FROM community_space_message m \
347                 WHERE {} ORDER BY m.id DESC{}",
348                clauses.join(" AND "),
349                limit_clause(plan.limit)
350            )
351        }
352        // Only reachable with a positive term (the handler forces `newest`
353        // otherwise), so there is always an FTS row to take a score from.
354        SearchSort::Relevant => {
355            let (expression, _) = fts_match(text).expect("relevance requires a positive term");
356            let match_placeholder = binder.bind(expression);
357            let clauses = common_predicates(binder, plan);
358            let inner = format!(
359                "SELECT m.id AS id, -bm25({FTS}) AS score FROM {FTS} \
360                 JOIN community_space_message m ON m.rowid = {FTS}.rowid \
361                 WHERE {FTS} MATCH {match_placeholder} AND {}",
362                clauses.join(" AND ")
363            );
364            relevance_wrapper(binder, plan, inner)
365        }
366    }
367}

374    fn term(text: &str, negated: bool) -> Term {
375        Term {
376            text: text.to_string(),
377            phrase: false,

389    #[test]
390    fn a_term_of_punctuation_is_unsatisfiable() {
391        assert_eq!(
392            TextPlan::of(&query(vec![term("+++", false)])),
393            TextPlan::Never
394        );
395        let plan = Plan {
396            spaces: &[Uuid::nil()],
397            query: &query(vec![term("+++", false)]),
398            sort: SearchSort::Newest,
399            cursor: None,
400            limit: 25,
401            bounds: (None, None),
402        };
403        assert!(statement(DbBackend::Sqlite, &plan).is_none());
404    }

406    /// The same term must not be able to exclude everything on one engine and
407    /// nothing on the other.
408    #[test]
409    fn an_unindexable_negation_is_dropped() {
410        assert_eq!(
411            TextPlan::of(&query(vec![term("!!", true)])),
412            TextPlan::Any,
413            "a negation nothing can contain excludes nothing"
414        );
415    }
416
417    #[test]
418    fn phrases_are_requoted_from_their_tokens() {
419        assert_eq!(fts_phrase("two words").as_deref(), Some(r#""two words""#));
420        // A quote cannot survive tokenization, so the literal is always well
421        // formed.
422        assert_eq!(fts_phrase(r#"a"b"#).as_deref(), Some(r#""a b""#));
423        assert_eq!(fts_phrase("---"), None);
424    }
425
426    #[test]
427    fn negations_ride_the_one_match_string() {
428        let terms = query(vec![
429            term("alpha", false),
430            term("beta", false),
431            term("spam", true),
432            term("ham", true),
433        ]);
434        let text = TextPlan::of(&terms);
435        assert_eq!(
436            fts_match(&text),
437            Some((
438                r#""alpha" AND "beta" NOT "spam" NOT "ham""#.to_string(),
439                true
440            ))
441        );
442    }
443
444    #[test]
445    fn negation_only_flips_to_an_exclusion() {
446        let terms = query(vec![term("spam", true), term("ham", true)]);
447        let text = TextPlan::of(&terms);
448        assert_eq!(
449            fts_match(&text),
