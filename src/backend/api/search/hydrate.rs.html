1//! Turning a page of `(id, score)` pairs into renderable results.
2//!
3//! Same rule as the history endpoint it mirrors: a fixed number of queries
4//! per page, never one per row. A page is at most
5//! [`super::LIMIT_RANGE`]'s ceiling of rows, and the hydration is four reads
6//! (messages, attachments, embeds, profiles) plus one per community the page
7//! touches.
8//!
9//! The hit order is the database's, and it is preserved: the ranking is the
10//! whole point of the response, and a `HashMap` iteration would throw it away.
11
12use std::collections::{HashMap, HashSet};
13
14use hound_markup::Reference;
15use hound_types::MergedProfile;
16use sea_orm::{ColumnTrait, DerivePartialModel, QueryFilter, prelude::*};
17use uuid::Uuid;
18
19use super::{SearchHit, sql::HitRow};
20use crate::api::{Api, ApiResult, data::prelude::*, space::CommunitySpaceMessageModel};
21
22/// The message columns a hit renders from. `space_id` is here and not on the
23/// history endpoint's projection because a result row has to say which channel
24/// it is in — and jumping to it needs the id, not the name.
25#[derive(DerivePartialModel, Debug)]
26#[sea_orm(entity = "community_space_message::Entity", from_query_result)]
27struct HitMessage {
28    id: Uuid,
29    space_id: Uuid,
30    user_id: Uuid,
31    contents: String,
32    edited_at: Option<i64>,
33}
34
35impl Api {
36    /// `community_of_space` comes from the ACL resolution that produced the
37    /// search's space list, so no space lookup is needed here — and a hit
38    /// whose space is somehow not in it is impossible, because the statement
39    /// filtered on exactly those ids.
40    pub(super) async fn hydrate_hits(
41        &self,
42        rows: Vec<HitRow>,
43        community_of_space: &HashMap<Uuid, Uuid>,
44    ) -> ApiResult<(Vec<SearchHit>, Vec<MergedProfile>)> {
45        if rows.is_empty() {
46            return Ok((Vec::new(), Vec::new()));
47        }
48        let ids: Vec<Uuid> = rows.iter().map(|row| row.id).collect();
49
50        let mut messages: HashMap<Uuid, HitMessage> = community_space_message::Entity::find()
51            .filter(community_space_message::Column::Id.is_in(ids.clone()))
52            .into_partial_model::<HitMessage>()
53            .all(&self.db)
54            .await?
55            .into_iter()
56            .map(|message| (message.id, message))
57            .collect();
58
59        let mut attachments = self.attachments_for_messages(ids.clone()).await?;
60        let mut embeds = self.embeds_for_messages(ids).await?;
61
62        // Authors and mentioned users, grouped by the community whose
63        // profiles apply: a person can have a different display name in each
64        // community a search spans.
65        let mut wanted: HashMap<Uuid, HashSet<Uuid>> = HashMap::new();
66        let mut hits = Vec::with_capacity(rows.len());
67        for row in rows {
68            let Some(message) = messages.remove(&row.id) else {
69                // Deleted between the search and the hydration. Dropping it is
70                // the only honest answer; the page is simply one row shorter.
71                continue;
72            };
73            let Some(&community_id) = community_of_space.get(&message.space_id) else {
74                continue;
75            };
76            let users = wanted.entry(community_id).or_default();
77            users.insert(message.user_id);
78            for reference in hound_markup::references(&hound_markup::parse(&message.contents)) {
79                if let Reference::User(id) = reference {
80                    users.insert(id);
81                }
82            }
83            hits.push(SearchHit {
84                message: CommunitySpaceMessageModel {
85                    id: message.id,
86                    space_id: message.space_id,
87                    user_id: message.user_id,
88                    contents: message.contents,
89                    edited_at: message.edited_at,
90                    attachments: attachments.remove(&message.id).unwrap_or_default(),
91                    embeds: embeds.remove(&message.id).unwrap_or_default(),
92                },
93                community_id,
94                score: row.score,
95            });
96        }
97
98        // First community wins a duplicate: the same user's profile in two
99        // communities is two rows with one `user_id`, and the response is a
100        // flat list keyed by that.
101        let mut seen = HashSet::new();
102        let mut users = Vec::new();
103        for (community_id, user_ids) in wanted {
104            for profile in self.find_merged_profiles(user_ids, community_id).await? {
105                if seen.insert(profile.user_id) {
106                    users.push(profile);
107                }
108            }
109        }
110        Ok((hits, users))
111    }
112}
