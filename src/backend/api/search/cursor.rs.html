1//! The opaque page cursor.
2//!
3//! A keyset, not an offset: it names the last row of the page just served, and
4//! the next page is everything strictly "after" it in the active sort order.
5//! That is what makes paging stable while messages are still being posted —
6//! an offset would re-serve or skip a row every time the table grows under it.
7//!
8//! The encoding (base64url of a JSON object) is **private to the backend**.
9//! Clients round-trip the string and never build or read one, so the shape can
10//! change without a client release: `v` is what an older cursor is recognised
11//! and rejected by.
12
13use base64::Engine as _;
14use uuid::Uuid;
15
16use crate::api::{ApiError, ApiResult};
17
18/// The only shape this build mints or accepts.
19pub(super) const VERSION: u8 = 1;
20
21const ENGINE: base64::engine::general_purpose::GeneralPurpose =
22    base64::engine::general_purpose::URL_SAFE_NO_PAD;
23
24#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
25pub(super) struct Cursor {
26    /// The last row's message id. Sole key under `newest`, tiebreak under
27    /// `relevant`.
28    pub id: Uuid,
29    /// The last row's score, present only under `relevant` sort.
30    #[serde(default)]
31    pub score: Option<f64>,
32    pub v: u8,
33}
34
35pub(super) fn encode(cursor: &Cursor) -> String {
36    // Infallible: the struct is plain data with no map keys and no non-finite
37    // floats (a score is a database double, and `serde_json` only fails a
38    // struct on those).
39    ENGINE.encode(serde_json::to_vec(cursor).unwrap_or_default())
40}
41
42/// Anything that is not a cursor this build minted is a
43/// [`ApiError::BadRequest`] — a caller cannot fix it by retrying, and it is
44/// never a symptom of missing data.
45pub(super) fn decode(raw: &str) -> ApiResult<Cursor> {
46    let bytes = ENGINE.decode(raw).map_err(|_| ApiError::BadRequest)?;
47    let cursor: Cursor = serde_json::from_slice(&bytes).map_err(|_| ApiError::BadRequest)?;
48    if cursor.v != VERSION {
49        return Err(ApiError::BadRequest);
50    }
51    Ok(cursor)
52}

54#[cfg(test)]
55mod tests {
56    use super::*;
57
58    #[test]
59    fn round_trips() {
60        let cursor = Cursor {
61            id: Uuid::from_u128(7),
62            score: Some(1.5),
63            v: VERSION,
64        };
65        assert_eq!(decode(&encode(&cursor)).unwrap(), cursor);
66    }
67
68    #[test]
69    fn url_safe_and_unpadded() {
70        // A cursor rides in a JSON body today, but it is the kind of value
71        // that ends up in a query string; `+`/`/`/`=` would need escaping.
72        let encoded = encode(&Cursor {
73            id: Uuid::from_u128(u128::MAX),
74            score: None,
75            v: VERSION,
76        });
77        assert!(
78            encoded
79                .bytes()
80                .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_'),
81            "{encoded}"
82        );
83    }
84
85    #[test]
86    fn rejects_junk_and_foreign_versions() {
87        assert!(decode("not base64!").is_err());
88        assert!(decode(&ENGINE.encode(b"{}")).is_err());
89        assert!(
90            decode(&encode(&Cursor {
91                id: Uuid::nil(),
92                score: None,
93                v: VERSION + 1,
94            }))
95            .is_err()
96        );
97    }
