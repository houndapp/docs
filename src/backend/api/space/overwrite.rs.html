61/// Reads a channel's permission overwrites — the round-trip half of the PUT.
62///
63/// `ManageChannels`, like the write: an ACL names the roles and users a channel
64/// treats specially, which is moderator-visible information rather than
65/// member-visible.
66///
67/// `user_overwrites` is filtered to current members. The write refuses a
68/// stranger and departure deletes the leaver's rows, so a row that survives
69/// both is one written before either guard existed — it must not reach an
70/// editor that would then send it straight back.
71pub(crate) async fn get_space_overwrites(
72    SpacePermission { space, .. }: SpacePermission<perm::ManageChannels>,
73    State(api): State<Api>,
74) -> ApiResult<Json<SpaceOverwritesResponse>> {
75    let (role_overwrites, user_overwrites) = api.find_space_overwrites(space.id).await?;
76    let members = api
77        .community_members_among(
78            &api.db,
79            space.community_id,
80            user_overwrites.iter().map(|ow| ow.user_id),
81        )
82        .await?;
83    Ok(Json(SpaceOverwritesResponse {
84        role_overwrites: role_overwrites
85            .into_iter()
86            .map(|ow| RoleOverwrite {
87                role_id: ow.role_id,
88                permission: ow.permission,
89                allow: ow.allow,
90            })
91            .collect(),
92        user_overwrites: user_overwrites
93            .into_iter()
94            .filter(|ow| members.contains(&ow.user_id))
95            .map(|ow| UserOverwrite {
96                user_id: ow.user_id,
97                permission: ow.permission,
98                allow: ow.allow,
99            })
100            .collect(),
101    }))
102}

118/// Replaces all permission overwrites for a channel (full replace, not a patch).
119/// Only channel-applicable permissions (`channels:view`, `messages:send`,
120/// `messages:history`, `messages:manage`) are accepted; community-only permissions
121/// are rejected with 400.
122///
123/// The submitted ACL is a *set*: `(role, permission)` and `(user, permission)`
124/// are each a primary key, so naming one twice has no defined winner and is
125/// refused (`duplicate_overwrite`) rather than resolved arbitrarily — which is
126/// also what keeps the insert below from failing halfway as a 500.
127///
128/// Every `role_overwrites` entry must name a role of *this* community, and
129/// every `user_overwrites` entry a current member of it: an ACL entry for a
130/// stranger is a rule nothing can match, and it would outlive whatever made the
131/// client offer it.
132///
133/// `is_private` moves with the ACL: the flag *means* "`@everyone` is denied
134/// `channels:view` here", so this write recomputes it from the submitted
135/// overwrites rather than leaving a padlock that disagrees with the rules
136/// behind it. Emits `SpaceUpdated` with the resulting summary.
137pub(crate) async fn put_space_overwrites(
138    SpacePermission { space, .. }: SpacePermission<perm::ManageChannels>,
139    State(api): State<Api>,
140    Json(req): Json<PutSpaceOverwritesRequest>,
141) -> ApiResult<()> {
142    validate_overwrites(&req)?;
143    let updated = api.replace_space_overwrites(&space, &req).await?;
144    api.emit_space_updated(&updated).await;
145    Ok(())
146}

148/// Everything about a submitted ACL that can be decided without the database:
149/// each permission is channel-applicable, and no subject names one twice.
150fn validate_overwrites(req: &PutSpaceOverwritesRequest) -> ApiResult<()> {
151    // Linear scans, not a set: an ACL is a handful of entries, and the
152    // permission enum is `Eq` but not `Hash`.
153    let mut seen: Vec<(Uuid, sea_orm_active_enums::Permission)> = Vec::new();
154    for ow in &req.role_overwrites {
155        if !ow.permission.is_channel_applicable() {
156            return Err(ApiError::BadRequest);
157        }
158        if !seen.iter().all(|key| *key != (ow.role_id, ow.permission)) {
159            return Err(ApiError::DuplicateOverwrite);
160        }
161        seen.push((ow.role_id, ow.permission));
162    }
163    seen.clear();
164    for ow in &req.user_overwrites {
165        if !ow.permission.is_channel_applicable() {
166            return Err(ApiError::BadRequest);
167        }
168        if !seen.iter().all(|key| *key != (ow.user_id, ow.permission)) {
169            return Err(ApiError::DuplicateOverwrite);
170        }
171        seen.push((ow.user_id, ow.permission));
172    }
173    Ok(())
174}

189                .await?,
190            space_user_overwrite::Entity::find()

192                .all(&self.db)
193                .await?,
194        ))

197    /// Atomic full replace of a space's role and user overwrites, returning the
198    /// space row with `is_private` brought back in step with them.
199    async fn replace_space_overwrites(
200        &self,
201        space: &community_space::Model,
202        req: &PutSpaceOverwritesRequest,
203    ) -> ApiResult<community_space::Model> {
204        let space_id = space.id;
205        let txn = self.db.begin().await?;
206
207        // Every subject is settled before the first delete, so a refused PUT
208        // leaves the previous ACL exactly as it was. One `IN` query per table —
209        // the request is a full replace, so a per-entry lookup would scale with
210        // the channel's member list.
211        let community_roles: std::collections::HashMap<Uuid, role::Model> = Role::find()
212            .filter(role::Column::CommunityId.eq(space.community_id))
213            .all(&txn)
214            .await?
215            .into_iter()
216            .map(|role| (role.id, role))
217            .collect();
218        if let Some(ow) = req
219            .role_overwrites
220            .iter()
221            .find(|ow| !community_roles.contains_key(&ow.role_id))
222        {
223            // Scoped like `manageable_role`: a role id is a free request field,
224            // and the guard authorised the caller for *this* community only.
225            txn.rollback().await?;
226            return Err(ApiError::RoleNotFound(ow.role_id));
227        }
228        let members = self
229            .community_members_among(
230                &txn,
231                space.community_id,
232                req.user_overwrites.iter().map(|ow| ow.user_id),
233            )
234            .await?;
235        if let Some(ow) = req
236            .user_overwrites
237            .iter()
238            .find(|ow| !members.contains(&ow.user_id))
239        {
240            txn.rollback().await?;
241            return Err(ApiError::NotAMember(ow.user_id));
242        }
243
244        // The flag is derived from the submitted ACL, not from the old value: a
245        // PUT that drops the @everyone view deny un-privates the channel, and
246        // one that adds it privates it.
247        let is_private = privacy::is_private_acl(
248            community_roles
249                .values()
250                .find(|role| role.is_everyone)
251                .map(|role| role.id),
252            &req.role_overwrites,
253        );
254
255        // Full replace: delete all existing overwrites for this space.
256        space_role_overwrite::Entity::delete_many()
257            .filter(space_role_overwrite::Column::SpaceId.eq(space_id))
258            .exec(&txn)
259            .await?;
260        space_user_overwrite::Entity::delete_many()
261            .filter(space_user_overwrite::Column::SpaceId.eq(space_id))
262            .exec(&txn)
263            .await?;
264        for ow in &req.role_overwrites {
265            space_role_overwrite::ActiveModel {
266                space_id: Set(space_id),
267                role_id: Set(ow.role_id),
268                permission: Set(ow.permission),
269                allow: Set(ow.allow),
270            }
271            .insert(&txn)
272            .await?;
273        }
274        for ow in &req.user_overwrites {
275            space_user_overwrite::ActiveModel {
276                space_id: Set(space_id),
277                user_id: Set(ow.user_id),
278                permission: Set(ow.permission),
279                allow: Set(ow.allow),
280            }
281            .insert(&txn)
282            .await?;
283        }
284        let updated = community_space::ActiveModel {
285            id: Set(space_id),
286            is_private: Set(is_private),
287            ..Default::default()
288        }
289        .update(&txn)
290        .await?;
291        txn.commit().await?;
292        Ok(updated)
293    }

295    /// Drops every per-user overwrite this user holds in any of the community's
296    /// channels. The overwrite half of a membership teardown — leave, kick and
297    /// ban all call it inside the transaction that removes the membership row,
298    /// so a departure never leaves a rule addressed to someone who is gone.
299    ///
300    /// No `SpaceUpdated` follows: a departed user's overwrites bind nobody
301    /// else, so no other member's view of the channel changes.
302    pub(crate) async fn remove_member_overwrites(
303        &self,
304        conn: &impl sea_orm::ConnectionTrait,
305        community_id: Uuid,
306        user_id: Uuid,
307    ) -> DbResult<()> {
308        let space_ids: Vec<Uuid> = community_space::Entity::find()
309            .select_only()
310            .column(community_space::Column::Id)
311            .filter(community_space::Column::CommunityId.eq(community_id))
312            .into_tuple()
313            .all(conn)
314            .await?;
315        if space_ids.is_empty() {
316            return Ok(());
317        }
318        space_user_overwrite::Entity::delete_many()
319            .filter(space_user_overwrite::Column::SpaceId.is_in(space_ids))
320            .filter(space_user_overwrite::Column::UserId.eq(user_id))
321            .exec(conn)
322            .await?;
323        Ok(())
324    }

668    /// The other direction of the same invariant: `PATCH /space/{id}` writes
669    /// the `@everyone` deny that privatising *means*, and a subsequent read of
670    /// the overwrites shows it — so the two surfaces describe one ACL, not two.

672    async fn patching_is_private_shows_up_in_the_overwrites() {

675        let everyone = api

677            .await

679            .unwrap();

682                .filter(space_role_overwrite::Column::SpaceId.eq(community.default_space))

691                .into_iter()
692                .filter(|ow| !ow.allow)

811    /// Every way out of a community takes the leaver's channel rules with it,
812    /// across the whole community — and touches nobody else's. A ban is a kick
813    /// that sticks, and a kick is a leave somebody else asked for; all three
814    /// run the same teardown, so they are pinned by the same test.
815    #[tokio::test]
816    async fn every_departure_removes_that_users_overwrites_in_every_space() {
817        for departure in [Departure::Leave, Departure::Kick, Departure::Ban] {
818            let (api, server) = crate::api::test().await;
819            let (_owner, token, community) = setup(&api, &server).await;
820            let (gone, bystander) = two_spaces_two_members(&api, &server, &token, &community).await;
821
822            let response = match departure {
823                Departure::Leave => {
824                    server
825                        .delete("/community/leave")
826                        .add_query_param("id", community.id)
827                        .authorization_bearer(api.test_access_token(gone))
828                        .await
829                }
830                Departure::Kick => {
831                    server
832                        .delete(&format!("/community/{}/member/{}", community.id, gone))
833                        .authorization_bearer(&token)
834                        .await
835                }
836                Departure::Ban => {
837                    server
838                        .post(&format!("/community/{}/ban/{}", community.id, gone))
839                        .authorization_bearer(&token)
840                        .json(&serde_json::json!({}))
841                        .await
842                }
843            };
844            assert_eq!(response.status_code(), 200, "{departure:?}");
845
846            assert_eq!(
847                overwrite_rows(&api, community.id, gone).await,
848                0,
849                "{departure:?} clears every rule addressed to the departed"
850            );
851            assert_eq!(
852                overwrite_rows(&api, community.id, bystander).await,
853                2,
854                "{departure:?} leaves every other member's rules standing"
855            );
856        }
