1//! The single writer for a message row and everything derived from it.
2//!
3//! A stored message is one source string plus four derived projections: its
4//! `hound:` reference rows, its [`ContentKind`] rows, its flattened
5//! `search_text` column, and its attachment links. [`Api::store_message`] is
6//! the only place any of them is written, and it takes a whole
7//! [`ParsedMessage`] rather than the pieces — so a body and the columns that
8//! describe it are physically incapable of disagreeing.
9//!
10//! ## The invariant
11//!
12//! Every derived column equals what a fresh [`ParsedMessage`] would produce
13//! from the row's current `contents` and its current attachments and embeds.
14//! Nothing else may write them, and [`Api::rederive_message`] restores the
15//! invariant for one message from those three inputs alone — which is what
16//! makes the whole projection rebuildable (`POST /internal/messages/reindex`)
17//! and what lets an unfurl landing minutes later add its `image`/`embed`
18//! facets without touching the body.
19//!
20//! ## Why the writes are replace-all
21//!
22//! Delete-then-insert, not a diff. The sets are tiny (a handful of rows), and
23//! a full replace is what makes an edit that *removes* a reference, a facet,
24//! or an attachment unable to leave its row behind.
25

37/// One message's complete stored form, ready to be written.
38///
39/// `edited_at` is the caller's to decide: a create leaves it `None`, an edit
40/// stamps it, and a re-derivation carries the existing value through
41/// untouched — re-indexing must never look like an edit.
42pub(crate) struct MessageWrite {
43    pub id: Uuid,
44    pub space_id: Uuid,
45    pub user_id: Uuid,
46    pub parsed: ParsedMessage,
47    /// Upload ids, already authorised by the caller (`reference_upload`).
48    pub attachment_ids: Vec<Uuid>,
49    pub edited_at: Option<i64>,
50}
51
52impl Api {
53    /// Insert-or-replace the message row and every projection derived from
54    /// it, inside `txn`.
55    ///
56    /// One transaction because the projections are only useful if they are
57    /// never observable apart from the body they describe: a reader must not
58    /// be able to see an edited `contents` with the previous edit's reference
59    /// rows, or a facet list that names a picture the message no longer has.
60    pub(crate) async fn store_message(
61        &self,
62        txn: &DatabaseTransaction,
63        write: MessageWrite,
64    ) -> Result<(), DbErr> {
65        let MessageWrite {
66            id,
67            space_id,
68            user_id,
69            parsed,
70            attachment_ids,
71            edited_at,
72        } = write;
73
74        // Upsert rather than insert-or-update-by-branch: a create and a
75        // re-derivation take the identical path, so a rebuild can never
76        // diverge from a first write.
77        community_space_message::Entity::insert(community_space_message::ActiveModel {
78            id: Set(id),
79            space_id: Set(space_id),
80            user_id: Set(user_id),
81            contents: Set(parsed.source.clone()),
82            search_text: Set(parsed.search_text.clone()),
83            edited_at: Set(edited_at),
84        })
85        .on_conflict(
86            OnConflict::column(community_space_message::Column::Id)
87                .update_columns([
88                    community_space_message::Column::Contents,
89                    community_space_message::Column::SearchText,
90                    community_space_message::Column::EditedAt,
91                ])
92                .to_owned(),
93        )
94        .exec(txn)
95        .await?;
96
97        self.replace_message_references(txn, id, &parsed.references)
98            .await?;
99        self.replace_message_kinds(txn, id, &parsed.kinds).await?;
100        self.replace_message_attachments(txn, id, &attachment_ids)
101            .await?;
102        Ok(())
103    }

105    /// Rebuild one message's derived columns from its current body,
106    /// attachments and embeds.
107    ///
108    /// The repair operation for the invariant this module holds, and the
109    /// normal path by which an unfurl's facets arrive: the embeds are stored
110    /// and this runs in the same transaction, so the `image`/`video`/`embed`
111    /// facets appear at the same instant the cards do.
112    ///
113    /// A body that no longer passes today's limits is a `DbErr::Custom` — the
114    /// reindex counts it and moves on rather than rewriting or dropping what
115    /// a user wrote.
116    pub(crate) async fn rederive_message(
117        &self,
118        txn: &DatabaseTransaction,
119        message_id: Uuid,
120    ) -> Result<(), DbErr> {
121        let row = community_space_message::Entity::find_by_id(message_id)
122            .one(txn)
123            .await?
124            .ok_or_else(|| DbErr::Custom(format!("message {message_id} is gone")))?;
125        let attachment_ids = self.attachment_ids(txn, message_id).await?;
126        let (attachment_content_types, embed_kinds) = self
127            .message_context_inputs(txn, message_id, &attachment_ids)
128            .await?;
129        let parsed = ParsedMessage::parse_allowing_empty(
130            &row.contents,
131            &MessageContext {
132                attachment_content_types: &attachment_content_types,
133                embed_kinds: &embed_kinds,
134            },
135            &Limits::default(),
136        )
137        .map_err(|violation| {
138            DbErr::Custom(format!(
139                "message {message_id} no longer parses: {violation}"
140            ))
141        })?;
142
143        self.store_message(
144            txn,
145            MessageWrite {
146                id: row.id,
147                space_id: row.space_id,
148                user_id: row.user_id,
149                parsed,
150                attachment_ids,
151                edited_at: row.edited_at,
152            },
153        )
154        .await
155    }

157    /// The two facts a message body cannot tell about itself: what its
158    /// attachments are, and what its links unfurled to.
159    ///
160    /// `attachment_ids` is passed in rather than read, because a create knows
161    /// them before any link row exists — the uploads are ones the caller just
162    /// referenced.
163    pub(crate) async fn message_context_inputs(
164        &self,
165        conn: &impl ConnectionTrait,
166        message_id: Uuid,
167        attachment_ids: &[Uuid],
168    ) -> Result<(Vec<String>, Vec<EmbedKind>), DbErr> {
169        let attachment_content_types = if attachment_ids.is_empty() {
170            Vec::new()
171        } else {
172            upload::Entity::find()
173                .filter(upload::Column::Id.is_in(attachment_ids.to_vec()))
174                .all(conn)
175                .await?
176                .into_iter()
177                .map(|upload| upload.content_type)
178                .collect()
179        };
180
181        let embed_kinds = community_space_message_embed::Entity::find()
182            .filter(community_space_message_embed::Column::MessageId.eq(message_id))
183            .order_by_asc(community_space_message_embed::Column::Ordinal)
184            .find_also_related(link_embed::Entity)
185            .all(conn)
186            .await?
187            .into_iter()
188            // A join row whose unfurl is still pending, or unfurled to
189            // nothing, contributes no facet — exactly as it contributes no
190            // card.
191            .filter_map(|(_, embed)| embed?.payload)
192            .filter_map(|payload| serde_json::from_value::<LinkEmbed>(payload).ok())
193            .map(|embed| embed.kind)
194            .collect();
195
196        Ok((attachment_content_types, embed_kinds))
197    }

199    /// The upload ids currently attached to a message, in insertion order.
200    pub(crate) async fn attachment_ids(
201        &self,
202        conn: &impl ConnectionTrait,
203        message_id: Uuid,
204    ) -> Result<Vec<Uuid>, DbErr> {
205        Ok(community_space_message_attachment::Entity::find()
206            .filter(community_space_message_attachment::Column::MessageId.eq(message_id))
207            .all(conn)
208            .await?
209            .into_iter()
210            .map(|link| link.upload_id)
211            .collect())
212    }

214    /// Replace every reference row for `message_id` with `references`.
215    ///
216    /// `community_space_message_reference` is a derived index of
217    /// [`hound_markup::references`]; this is the one place it is written.
218    pub(crate) async fn replace_message_references(
219        &self,
220        conn: &impl ConnectionTrait,
221        message_id: Uuid,
222        references: &[Reference],
223    ) -> Result<(), DbErr> {
224        community_space_message_reference::Entity::delete_many()
225            .filter(community_space_message_reference::Column::MessageId.eq(message_id))
226            .exec(conn)
227            .await?;
228        for reference in references {
229            community_space_message_reference::Entity::insert(
230                community_space_message_reference::ActiveModel {
231                    message_id: Set(message_id),
232                    kind: Set(reference.kind().as_str().to_string()),
233                    target_id: Set(reference.id()),
234                },
235            )
236            .exec(conn)
237            .await?;
238        }
239        Ok(())
240    }

242    /// Replace every facet row for `message_id` with `kinds`.
243    async fn replace_message_kinds(
244        &self,
245        conn: &impl ConnectionTrait,
246        message_id: Uuid,
247        kinds: &[ContentKind],
248    ) -> Result<(), DbErr> {
249        community_space_message_kind::Entity::delete_many()
250            .filter(community_space_message_kind::Column::MessageId.eq(message_id))
251            .exec(conn)
252            .await?;
253        for kind in kinds {
254            community_space_message_kind::Entity::insert(
255                community_space_message_kind::ActiveModel {
256                    message_id: Set(message_id),
257                    kind: Set(kind.as_str().to_string()),
258                },
259            )
260            .exec(conn)
261            .await?;
262        }
263        Ok(())
264    }

266    /// Replace the message's attachment links.
267    ///
268    /// The uploads themselves are authorised by the caller before this runs
269    /// (`reference_upload` is a per-attachment ownership check, not something
270    /// to fold into the message's own write); this only writes the join.
271    async fn replace_message_attachments(
272        &self,
273        conn: &impl ConnectionTrait,
274        message_id: Uuid,
275        attachment_ids: &[Uuid],
276    ) -> Result<(), DbErr> {
277        community_space_message_attachment::Entity::delete_many()
278            .filter(community_space_message_attachment::Column::MessageId.eq(message_id))
279            .exec(conn)
280            .await?;
281        for upload_id in attachment_ids {
282            community_space_message_attachment::Entity::insert(
283                community_space_message_attachment::ActiveModel {
284                    message_id: Set(message_id),
285                    upload_id: Set(*upload_id),
286                },
287            )
288            .exec(conn)
289            .await?;
290        }
291        Ok(())
292    }

294    /// Every message id in the instance, in id order, from `after` onwards.
295    ///
296    /// Keyset paging rather than offset: a reindex walks the whole table one
297    /// batch at a time and must not re-scan what it has already passed.
298    pub(crate) async fn message_ids_after(
299        &self,
300        after: Option<Uuid>,
301        limit: u64,
302    ) -> Result<Vec<Uuid>, DbErr> {
303        let mut query = community_space_message::Entity::find()
304            .select_only()
305            .column(community_space_message::Column::Id)
306            .order_by_asc(community_space_message::Column::Id)
307            .limit(limit);
308        if let Some(after) = after {
309            query = query.filter(community_space_message::Column::Id.gt(after));
310        }
311        query.into_tuple::<Uuid>().all(&self.db).await
312    }

317    /// Every reference currently stored for a message, in no particular
318    /// order. Test-only: production code re-derives what it needs from
319    /// `contents` rather than reading this index back.
320    pub(crate) async fn message_references(&self, message_id: Uuid) -> Vec<Reference> {
321        use hound_markup::ReferenceKind;
322
323        community_space_message_reference::Entity::find()
324            .filter(community_space_message_reference::Column::MessageId.eq(message_id))
325            .all(&self.db)
326            .await
327            .unwrap()
328            .into_iter()
329            .filter_map(|row| {
330                let reference = match ReferenceKind::parse_str(&row.kind)? {
331                    ReferenceKind::User => Reference::User(row.target_id),
332                    ReferenceKind::Space => Reference::Space(row.target_id),
333                    ReferenceKind::Message => Reference::Message(row.target_id),
334                    ReferenceKind::Community => Reference::Community(row.target_id),
335                };
336                Some(reference)
337            })
338            .collect()
339    }

341    /// Every facet currently stored for a message, sorted — the order
342    /// [`ParsedMessage::kinds`] is in, so the two compare directly.
343    pub(crate) async fn message_kinds(&self, message_id: Uuid) -> Vec<ContentKind> {
344        let mut kinds: Vec<ContentKind> = community_space_message_kind::Entity::find()
345            .filter(community_space_message_kind::Column::MessageId.eq(message_id))
346            .all(&self.db)
347            .await
348            .unwrap()
349            .into_iter()
350            .filter_map(|row| ContentKind::parse_str(&row.kind))
351            .collect();
352        kinds.sort();
353        kinds
354    }
