1//! `POST /profile/global/avatar/steam`: copies the caller's Steam avatar into
2//! this deployment's storage and makes it their global photo.
3//!
4//! Steam's CDN is never hotlinked: the bytes are fetched once, pass the same
5//! type/size/magic checks a client upload does, and become an ordinary
6//! referenced `upload` row under `avatar/{user}/`.
7

34/// Sets the authenticated user's global photo to their Steam avatar.
35pub(crate) async fn import_steam_avatar(
36    LoggedIn(claims): LoggedIn,
37    State(api): State<Api>,
38) -> ApiResult<Json<ProfileResponse>> {
39    let steam = api.auth.steam()?;
40    let steam_id: u64 = api
41        .user_identities(claims.sub)
42        .await?
43        .into_iter()
44        .find(|identity| identity.provider == IdentityProvider::Steam)
45        .and_then(|identity| identity.external_id.parse().ok())
46        .ok_or(ApiError::NoSteamIdentity)?;
47
48    let persona = steam
49        .persona(steam_id)
50        .await
51        .map_err(|error| ApiError::SteamUnavailable(error.to_string()))?;
52    let url = persona
53        .avatar_url
54        .ok_or_else(|| ApiError::SteamUnavailable("persona has no avatar".into()))?;
55    let (mime, data) = api.fetch_steam_avatar(&url).await?;
56
57    let upload = api
58        .store_upload(
59            claims.sub,
60            &format!("avatar/{}", claims.sub),
61            AVATAR_MIME_TYPES,
62            &mime,
63            data,
64        )
65        .await
66        // Steam sent something that is not an acceptable avatar: an upstream
67        // fault, not a request the caller can fix.
68        .map_err(|error| match error {
69            ApiError::UnsupportedMediaType(_) | ApiError::PayloadTooLarge => {
70                ApiError::SteamUnavailable(error.to_string())
71            }
72            error => error,
73        })?;
74    let upload = api.reference_upload(upload.id, claims.sub).await?;
75    let profile = api
76        .update_global_profile(claims.sub, None, Some(Some(upload.key)))
77        .await?;
78    Ok(Json(api.profile_response(profile)))
79}

96impl Api {
97    /// GETs the avatar a Steam persona names, bounded by `upload_max_bytes`
98    /// while streaming (`Content-Length` is a claim, not a limit). Returns the
99    /// declared media type without parameters, and the bytes; every failure is
100    /// `SteamUnavailable`.
101    async fn fetch_steam_avatar(&self, url: &str) -> ApiResult<(String, Bytes)> {
102        let unavailable = |error: &dyn std::fmt::Display| {
103            ApiError::SteamUnavailable(format!("avatar fetch: {error}"))
104        };
105        let mut response = self
106            .outbound_http
107            .get(url)
108            .send()
109            .await
110            .and_then(reqwest::Response::error_for_status)
111            .map_err(|error| unavailable(&error))?;
112        let mime = response
113            .headers()
114            .get(header::CONTENT_TYPE)
115            .and_then(|value| value.to_str().ok())
116            .and_then(|value| value.split(';').next())
117            .map(|value| value.trim().to_ascii_lowercase())
118            .unwrap_or_default();
119
120        let mut data = Vec::new();
121        while let Some(chunk) = response
122            .chunk()
123            .await
124            .map_err(|error| unavailable(&error))?
125        {
126            if (data.len() + chunk.len()) as u64 > self.upload_max_bytes {
127                return Err(unavailable(&"over the upload size limit"));
128            }
129            data.extend_from_slice(&chunk);
130        }
131        Ok((mime, Bytes::from(data)))
132    }

146    fn jpeg_bytes() -> Vec<u8> {
147        let mut v = vec![0xFF, 0xD8, 0xFF, 0xE0, 0x00, 0x10];
148        v.extend_from_slice(b"JFIF\x00");
149        v.extend_from_slice(&[0x01, 0x01, 0x00, 0x00, 0x01, 0x00, 0x01, 0x00, 0x00]);
150        v.extend_from_slice(&[0xFF, 0xD9]);
151        v
152    }

154    /// A stand-in for Steam's avatar CDN on a loopback port; returns its origin.
155    async fn serve_avatars() -> String {
156        let app = axum::Router::new()
157            .route(
158                "/avatar.jpg",
159                get(|| async { ([(header::CONTENT_TYPE, "image/jpeg")], jpeg_bytes()) }),
160            )
161            .route(
162                "/not-an-image.jpg",
163                get(|| async { ([(header::CONTENT_TYPE, "image/jpeg")], "<html></html>") }),
164            )
165            .route(
166                "/broken.jpg",
167                get(|| async { StatusCode::INTERNAL_SERVER_ERROR }),
168            );
169        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
170        let addr = listener.local_addr().unwrap();
171        tokio::spawn(async move { axum::serve(listener, app).await.unwrap() });
172        format!("http://{addr}")
173    }

188    #[tokio::test]
189    async fn imports_the_steam_avatar_as_a_referenced_upload() {
190        let (api, server) = crate::api::test().await;
191        let user = api.insert_test_user().await;
192        let token = api.test_access_token(user.id);
193        link_steam(&server, &token).await;
194        set_persona(
195            &api,
196            FakePersona::Avatar(format!("{}/avatar.jpg", serve_avatars().await)),
197        );
198
199        let response = server
200            .post("/profile/global/avatar/steam")
201            .authorization_bearer(&token)
202            .await;
203        assert_eq!(response.status_code(), 200);
204        let imported: ProfileResponse = response.json();
205        let picture = imported.picture.clone().expect("the import sets a picture");
206        let prefix = api.storage.public_url(&format!("avatar/{}/", user.id));
207        assert!(picture.starts_with(&prefix), "{picture}");
208        assert!(picture.ends_with(".jpg"), "{picture}");
209
210        // An unreferenced upload, to prove the sweep below actually runs.
211        let orphan = api
212            .store_upload(
213                user.id,
214                &format!("avatar/{}", user.id),
215                AVATAR_MIME_TYPES,
216                "image/jpeg",
217                jpeg_bytes().into(),
218            )
219            .await
220            .unwrap();
221        api.sweep_orphaned_uploads(std::time::Duration::ZERO).await;
222
223        let profile = api.find_global_profile_by_user_id(user.id).await.unwrap();
224        let key = profile
225            .avatar_key
226            .expect("the profile references the upload");
227        let surviving = upload::Entity::find()
228            .filter(upload::Column::Key.eq(key))
229            .one(&api.db)
230            .await
231            .unwrap();
232        assert!(
233            surviving.is_some(),
234            "a referenced upload survives the sweep"
235        );
236        assert!(
237            upload::Entity::find_by_id(orphan.id)
238                .one(&api.db)
239                .await
240                .unwrap()
241                .is_none(),
242            "the orphan was swept"
243        );
244
245        let fetched: ProfileResponse = server
246            .get("/profile/global")
247            .authorization_bearer(&token)
248            .await
249            .json();
250        assert_eq!(fetched.picture, Some(picture));
251    }
