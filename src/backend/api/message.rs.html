1//! Standalone message lookup, independent of the `/space/{id}/message/*`
2//! routes: those need the owning space's id up front, but a `<hound:message/…>`
3//! reference embedded in another message carries only the message's own id.
4//! This is the resolution path a client uses to render that reference as a
5//! chip (channel, author, first line) — permission-checked against the
6//! message's space exactly as every other message read is.
7
8use axum::{
9    Json,
10    extract::{Path, State},
11};
12use uuid::Uuid;
13
14use crate::api::{Api, ApiResult, middleware::LoggedIn, space::CompleteCommunitySpaceMessage};
15
16pub const TAG: &str = "message";
17
18/// Returns a router containing all `/message` related routes.
19pub(crate) fn router() -> utoipa_axum::router::OpenApiRouter<Api> {
20    utoipa_axum::router::OpenApiRouter::with_openapi(Default::default())
21        .routes(utoipa_axum::routes!(get_message))
22}
23
24#[utoipa::path(
25    get,
26    path = "/{message_id}",
27    tag = TAG,
28    params(
29        ("message_id" = Uuid, Path, description = "The ID of the message to retrieve"),
30    ),
31    responses(
32        (status = 200, description = "the message", body = CompleteCommunitySpaceMessage),
33        (status = 403, description = "no access to the message's space"),
34        (status = 404, description = "message not found"),
35    ),
36    security(
37      ("token" = [])
38    ),
39)]
40/// Retrieves a single message by ID, for resolving a `<hound:message/…>`
41/// reference. 404 for an unknown id, 403 for a caller without view access to
42/// the message's space (same check `SpaceAccess` applies to every other
43/// message route).
44pub(crate) async fn get_message(
45    LoggedIn(claims): LoggedIn,
46    Path(message_id): Path<Uuid>,
47    State(api): State<Api>,
48) -> ApiResult<Json<CompleteCommunitySpaceMessage>> {
49    api.find_complete_message(message_id, claims.sub)
50        .await
51        .map(Json)
52}
