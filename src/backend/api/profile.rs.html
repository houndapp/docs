103/// Request body for editing the global profile.
104///
105/// Every field is optional and an omitted one is left alone.
106/// `avatar_upload_id` is doubly optional so `null` is expressible and means
107/// *clear the photo* — omitting the key and sending `null` are different
108/// requests.
109#[derive(Debug, Default, serde::Serialize, serde::Deserialize, utoipa::ToSchema)]
110pub struct PatchProfileRequest {
111    /// Display name; trimmed, 1–32 characters.
112    #[schema(example = "Aoife")]
113    #[serde(default, skip_serializing_if = "Option::is_none")]
114    pub name: Option<String>,
115    /// An upload the caller owns (`POST /upload/avatar`), or `null` to clear
116    /// the photo.
117    #[schema(value_type = Option<Uuid>)]
118    #[serde(
119        default,
120        deserialize_with = "present_option",
121        skip_serializing_if = "Option::is_none"
122    )]
123    pub avatar_upload_id: Option<Option<Uuid>>,
124}
125
126#[utoipa::path(
127    patch,
128    path = "/global",
129    tag = TAG,
130    request_body = PatchProfileRequest,
131    responses(
132        (status = 204, description = "Profile updated"),
133        (status = 400, description = "Name empty after trimming, or over 32 characters"),
134        (status = 401, description = "Missing or invalid token"),
135        (status = 403, description = "Upload belongs to a different user"),
136        (status = 404, description = "Upload not found"),
137    ),
138    security(("token" = [])),
139)]
140/// Edits the authenticated user's global profile: display name and photo.
141///
142/// A partial update — an omitted field is untouched, and a `null`
143/// `avatar_upload_id` clears the photo.
144pub(crate) async fn patch_global_profile(
145    LoggedIn(claims): LoggedIn,
146    State(api): State<Api>,
147    Json(body): Json<PatchProfileRequest>,
148) -> ApiResult<StatusCode> {
149    let name = body
150        .name
151        .map(|name| validate_profile_name(&name))
152        .transpose()?;
153    // Resolved before the write so an unknown or someone else's upload fails
154    // the request rather than half-applying the name.
155    let avatar_key = match body.avatar_upload_id {
156        Some(Some(upload_id)) => Some(Some(api.reference_upload(upload_id, claims.sub).await?.key)),
157        Some(None) => Some(None),
158        None => None,
159    };
160    api.update_global_profile(claims.sub, name, avatar_key)
161        .await?;
162    Ok(StatusCode::NO_CONTENT)
163}

165/// Trims and bounds a display name.
166pub(crate) fn validate_profile_name(name: &str) -> ApiResult<String> {
167    validate::bounded(
168        name,
169        MAX_PROFILE_NAME_CHARS,
170        ApiError::EmptyName,
171        ApiError::NameTooLong {
172            max: MAX_PROFILE_NAME_CHARS,
173        },
174    )
175}

227    /// Writes the given fields of the user's global profile; `None` leaves a
228    /// field alone, `Some(None)` clears the avatar. Returns the updated row.
229    pub(crate) async fn update_global_profile(
230        &self,
231        user_id: Uuid,
232        name: Option<String>,
233        avatar_key: Option<Option<String>>,
234    ) -> DbResult<profile::Model> {
235        let profile = self.find_global_profile_by_user_id(user_id).await?;
236        if name.is_none() && avatar_key.is_none() {
237            return Ok(profile);
238        }
239        let mut active = profile::ActiveModel {
240            id: Set(profile.id),
241            ..Default::default()
242        };
243        if let Some(name) = name {
244            active.name = Set(Some(name));
245        }
246        if let Some(avatar_key) = avatar_key {
247            active.avatar_key = Set(avatar_key);
248        }
249        active.update(&self.db).await
250    }

268    /// Creates the user's community-scoped profile row if they have none.
269    ///
270    /// Idempotent against the `(user_id, community_id)` unique index: a member
271    /// who leaves keeps their community profile (it is their identity there,
272    /// not their membership), so a rejoin must not mint a second one — two rows
273    /// would make the merged read pick one arbitrarily.
274    pub(crate) async fn insert_community_profile(
275        &self,
276        conn: &impl sea_orm::ConnectionTrait,
277        user_id: Uuid,
278        community_id: Uuid,
279    ) -> DbResult<()> {
280        profile::Entity::insert(profile::ActiveModel {
281            id: Set(uuid::Uuid::now_v7()),
282            user_id: Set(user_id),
283            community_id: Set(Some(community_id)),
284            ..Default::default()
285        })
286        .on_conflict(
287            sea_orm::sea_query::OnConflict::columns([
288                profile::Column::UserId,
289                profile::Column::CommunityId,
290            ])
291            .do_nothing()
292            .to_owned(),
293        )
294        .exec_without_returning(conn)
295        .await?;
296        Ok(())
297    }

329    /// [`Api::resolve_merged_profiles`] in the shared wire shape.
330    pub(crate) async fn find_merged_profiles(
331        &self,
332        user_ids: impl IntoIterator<Item = Uuid>,
333        community_id: Uuid,
334    ) -> DbResult<Vec<MergedProfile>> {
335        Ok(self
336            .resolve_merged_profiles(user_ids, community_id)
337            .await?
338            .into_iter()
339            .map(MergedProfile::from)
340            .collect())
341    }

343    /// Fetches global and community profiles for the given users in one query, then
344    /// merges them: community profile fields take precedence over global ones.
345    pub(crate) async fn resolve_merged_profiles(
346        &self,
347        user_ids: impl IntoIterator<Item = Uuid>,
348        community_id: Uuid,
349    ) -> DbResult<Vec<ResolvedProfile>> {
350        let profiles = Profile::find()
351            .filter(
352                Condition::all()
353                    .add(profile::Column::UserId.is_in(user_ids))
354                    .add(
355                        Condition::any()
356                            .add(profile::Column::CommunityId.is_null())
357                            .add(profile::Column::CommunityId.eq(community_id)),
358                    ),
359            )
360            .all(&self.db)
361            .await?;
362
363        // Group into (global, community) pairs keyed by user_id.
364        let mut by_user: std::collections::HashMap<
365            Uuid,
366            (Option<profile::Model>, Option<profile::Model>),
367        > = std::collections::HashMap::new();
368        for p in profiles {
369            let entry = by_user.entry(p.user_id).or_default();
370            if p.community_id.is_none() {
371                entry.0 = Some(p);
372            } else {
373                entry.1 = Some(p);
374            }
375        }
376
377        Ok(by_user
378            .into_iter()
379            .filter_map(|(user_id, (global, community))| {
380                // Global profile must exist per invariant; skip if somehow absent.
381                let global = global?;
382                let name = community
383                    .as_ref()
384                    .and_then(|c| c.name.clone())
385                    .or_else(|| global.name.clone());
386                // Prefer community avatar_key, then global.
387                let avatar_key = community
388                    .as_ref()
389                    .and_then(|c| c.avatar_key.as_deref())
390                    .or(global.avatar_key.as_deref());
391                let picture = self.profile_picture(avatar_key);
392                Some(ResolvedProfile {
393                    user_id,
394                    global_profile_id: global.id,
395                    community_profile_id: community.map(|c| c.id),
396                    name,
397                    picture,
398                })
399            })
400            .collect())
401    }
