1//! The WebSocket leg: universally reachable, and the fallback whenever the
2//! WebTransport endpoint is unconfigured or unreachable.
3//!
4//! Two things this leg must do that the WebTransport one gets from QUIC: cap what
5//! a peer can make the server buffer (frame and message size), and prove the peer
6//! is still there (a server-driven ping, answered by every WebSocket stack). Both
7//! are wired here; the reaping policy itself lives in [`hub::run_session`].
8
9use axum::extract::ws::{Message, WebSocket, WebSocketUpgrade};
10use axum::extract::{Query, State};
11use axum::response::{IntoResponse, Response};
12use axum_extra::TypedHeader;
13use axum_extra::headers::Authorization;
14use axum_extra::headers::authorization::Bearer;
15use futures::SinkExt;
16use hound_wt::{Liveness, MAX_FRAME_BYTES, MessagePipe, PipeError};
17
18use crate::api::Api;
19use crate::api::realtime::{Transport, hub, unauthorized};
20
21/// The `/realtime/ws` upgrade route. A plain axum route rather than an OpenAPI
22/// one — a protocol upgrade has no response body to document.
23pub(crate) fn router() -> axum::Router<Api> {
24    axum::Router::new().route("/realtime/ws", axum::routing::get(realtime_ws_handler))
25}

27#[derive(serde::Deserialize)]
28pub(crate) struct TokenQuery {
29    /// Fallback credential channel for clients that cannot set headers (a
30    /// browser's `WebSocket` constructor). Prefer `Authorization: Bearer`.
31    token: Option<String>,
32}
33
34/// Authenticate, then upgrade. Auth runs before the upgrade so an unauthorized
35/// caller gets a plain 401 and never reaches the session machinery.
36async fn realtime_ws_handler(
37    State(api): State<Api>,
38    bearer: Option<TypedHeader<Authorization<Bearer>>>,
39    Query(query): Query<TokenQuery>,
40    ws: WebSocketUpgrade,
41) -> Response {
42    // Header first: a token in the query string is logged by every proxy between
43    // here and the client.
44    let token = bearer
45        .as_ref()
46        .map(|TypedHeader(header)| header.token())
47        .or(query.token.as_deref());
48    let Some(user_id) = api.authenticate_realtime(token.unwrap_or_default()).await else {
49        tracing::warn!("/realtime/ws — rejected: invalid access token");
50        return unauthorized().into_response();
51    };
52
53    // The same cap the WebTransport leg's length-delimited codec enforces, so a
54    // peer cannot make one leg buffer what the other refuses. `max_message_size`
55    // bounds a reassembled fragmented message; `max_frame_size` bounds a single
56    // frame — both are needed, since either alone leaves the other unbounded.
57    ws.max_message_size(MAX_FRAME_BYTES)
58        .max_frame_size(MAX_FRAME_BYTES)
59        .on_upgrade(move |socket| {
60            hub::run_session(WsPipe::new(socket), api, user_id, Transport::WebSocket)
61        })
62}

64/// A [`MessagePipe`] over an axum WebSocket.
65///
66/// Text frames are the protocol; ping/pong and binary are skipped rather than
67/// treated as frames, so a keepalive from a proxy can never desync the stream.
68/// They are still *counted*: a peer answering pings is alive, and that count is
69/// the only thing distinguishing it from one that vanished.
70pub(crate) struct WsPipe {
71    socket: WebSocket,
72    inbound: u64,
73}
74
75impl WsPipe {
76    pub(crate) fn new(socket: WebSocket) -> Self {
77        Self { socket, inbound: 0 }
78    }
79}
80
81impl MessagePipe for WsPipe {
82    async fn send(&mut self, frame: String) -> Result<(), PipeError> {
83        self.socket
84            .send(Message::Text(frame.into()))
85            .await
86            .map_err(|e| PipeError::Transport(e.to_string()))
87    }
88
89    async fn recv(&mut self) -> Option<Result<String, PipeError>> {
90        loop {
91            let message = self.socket.recv().await?;
92            if message.is_ok() {
93                self.inbound += 1;
94            }
95            return match message {
96                Ok(Message::Text(text)) => Some(Ok(text.to_string())),
97                Ok(Message::Close(_)) => None,
98                // Ping/Pong are answered by axum; binary is not part of this
99                // protocol.
100                Ok(_) => continue,
101                Err(error) => Some(Err(PipeError::Transport(error.to_string()))),
102            };
103        }
104    }
105
106    async fn close(&mut self) {
107        let _ = SinkExt::close(&mut self.socket).await;
108    }
109}
110
111impl Liveness for WsPipe {
112    async fn ping(&mut self) -> Result<(), PipeError> {
113        self.socket
114            .send(Message::Ping(Default::default()))
115            .await
116            .map_err(|e| PipeError::Transport(e.to_string()))
117    }
118
119    fn inbound_count(&self) -> u64 {
120        self.inbound
121    }
122}
