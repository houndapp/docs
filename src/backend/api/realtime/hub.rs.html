1//! The IO shell around [`hound_realtime::Hub`]: connection registry, fanout, and
2//! the per-connection session loop both transports share.
3//!
4//! ## Locking
5//!
6//! Two std `Mutex`es: the hub (routing state) and the connection mailboxes.
7//! Neither is ever held across an `.await` — every send onto a mailbox is
8//! `try_send`, which is synchronous. Fanout takes them in sequence (resolve
9//! targets under the hub lock, push under the conns lock); [`Realtime::admit`] is
10//! the one path that nests them, hub → conns, and nothing anywhere takes them in
11//! the other order. Both are acquired with poison recovery: a session task that
12//! panics must not take the realtime plane down with it, since every HTTP emit
13//! routes through the same locks.
14//!
15//! ## One task per connection
16//!
17//! Each session is a single task owning its pipe, selecting between "a frame
18//! arrived", "this connection has an outgoing message queued", and — on a leg
19//! with no transport keep-alive — the liveness timers. Outgoing sends are awaited
20//! inline, so a slow client backs up only its own mailbox, never the fanout and
21//! never another connection.
22//!
23//! ## Bounds
24//!
25//! Every resource a connection can consume is bounded before it is trusted: the
26//! per-user connection cap ([`MAX_CONNS_PER_USER`]) is reserved before the
27//! handshake, `Hello` must arrive inside [`SessionLimits::hello_deadline`], the
28//! outbound mailbox is [`MAILBOX_CAPACITY`] deep and overflowing it is a
29//! disconnect, and a WebSocket that answers nothing is reaped. Teardown runs from
30//! a drop guard, so a panicking session cannot leak any of it.
31

52/// How long after a connection drops before the user is evicted from voice.
53///
54/// A grace period, not a timeout: a client that reconnects (or rejoins voice)
55/// within it bumps the eviction generation and cancels the pending eviction, so
56/// a brief network blip does not drop the user out of a call.
57const VOICE_EVICTION_GRACE: Duration = Duration::from_secs(30);
58
59/// Depth of one connection's outbound mailbox.
60///
61/// Deep enough to absorb a burst (a busy community's fanout, a resync) without
62/// touching a client that is merely mid-`await`; shallow enough that a client
63/// which has stopped reading is detected in bounded memory rather than growing a
64/// queue until the pod is OOM-killed. Overflow is a disconnect, not a drop: see
65/// [`Realtime::disconnect_overflowed`].
66pub(crate) const MAILBOX_CAPACITY: usize = 256;
67
68/// Most simultaneous connections one authenticated user may hold.
69///
70/// Generous against real use (desktop + phone + a reconnect racing its
71/// predecessor's teardown) and small enough that a stolen token cannot mint
72/// sessions until the process runs out of tasks.
73pub(crate) const MAX_CONNS_PER_USER: usize = 8;
74
75/// The timing bounds every session is held to. A field rather than constants so
76/// tests can compress them; production always uses [`SessionLimits::default`].
77#[derive(Debug, Clone, Copy)]
78pub(crate) struct SessionLimits {
79    /// A connection that has not sent `Hello` within this is closed. Bounds what
80    /// an authenticated-but-idle peer can park.
81    pub(crate) hello_deadline: Duration,
82    /// Server ping cadence on a leg with no transport keep-alive.
83    pub(crate) ping_interval: Duration,
84    /// No inbound frame *of any kind* for this long means the peer is gone. Must
85    /// be a comfortable multiple of `ping_interval`, so a peer that answers
86    /// every probe is never reaped for a single lost round trip.
87    pub(crate) idle_timeout: Duration,
88}
89
90impl Default for SessionLimits {
91    fn default() -> Self {
92        Self {
93            hello_deadline: Duration::from_secs(10),
94            ping_interval: Duration::from_secs(25),
95            idle_timeout: Duration::from_secs(60),
96        }
97    }
98}

100/// Live realtime state: the routing reducer, the per-connection mailboxes, and
101/// the endpoint advertisement `/realtime/config` serves.
102#[derive(Debug)]
103pub(crate) struct Realtime {
104    hub: Mutex<Hub>,
105    conns: Mutex<HashMap<ConnId, mpsc::Sender<ServerMessage>>>,
106    /// Connections per user, for [`MAX_CONNS_PER_USER`]. Counted here rather than
107    /// from the hub because a connection occupies a slot from the moment it is
108    /// authenticated — including while it is still parked before `Hello`, which
109    /// the hub knows nothing about.
110    user_conns: Mutex<HashMap<Uuid, usize>>,
111    next_conn: AtomicU64,
112    config: RealtimeConfig,
113    limits: SessionLimits,
114}

116impl Realtime {
117    /// `config` describes the WebTransport endpoint, if one is running. With no
118    /// endpoint (the default) it is empty and clients use the WebSocket leg.
119    pub(crate) fn new(config: RealtimeConfig) -> Self {
120        Self::with_limits(config, SessionLimits::default())
121    }

123    pub(crate) fn with_limits(config: RealtimeConfig, limits: SessionLimits) -> Self {
124        Self {
125            hub: Mutex::new(Hub::new()),
126            conns: Mutex::new(HashMap::new()),
127            user_conns: Mutex::new(HashMap::new()),
128            next_conn: AtomicU64::new(0),
129            config,
130            limits,
131        }
132    }

134    pub(crate) fn config(&self) -> RealtimeConfig {
135        self.config.clone()
136    }

138    /// The routing state.
139    ///
140    /// Poison-recovering: the guarded value is a map whose invariants never span
141    /// more than one method call, so the worst a panicking session leaves behind
142    /// is its own entry — which its drop guard removes anyway. Propagating the
143    /// poison instead would take every future emit down with it.
144    fn hub(&self) -> MutexGuard<'_, Hub> {
145        self.hub
146            .lock()
147            .unwrap_or_else(|poisoned| poisoned.into_inner())
148    }
149
150    /// The connection mailboxes. Poison-recovering for the same reason as
151    /// [`Self::hub`].
152    fn conns(&self) -> MutexGuard<'_, HashMap<ConnId, mpsc::Sender<ServerMessage>>> {
153        self.conns
154            .lock()
155            .unwrap_or_else(|poisoned| poisoned.into_inner())
156    }
157
158    /// Live connection count. Test-only: production reads this through the
159    /// `backend_realtime_connections` gauge, which the session loop maintains.
160    #[cfg(test)]
161    pub(crate) fn connection_count(&self) -> usize {
162        self.hub().len()
163    }
164
165    /// How many connections a space's messages would currently fan out to. The
166    /// observable a test synchronizes on: a subscribe is answered only when it
167    /// is denied, so room membership is the only evidence it took effect.
168    #[cfg(test)]
169    pub(crate) fn space_subscriber_count(&self, space_id: Uuid) -> usize {
170        self.hub()
171            .recipients(&Target::Room(Room::Space(space_id)))
172            .len()
173    }
174
175    fn allocate_conn(&self) -> ConnId {
176        ConnId(self.next_conn.fetch_add(1, Ordering::Relaxed))
177    }
178
179    /// Connections `user` currently holds. A peek, not a reservation: the
180    /// WebTransport leg uses it to refuse a peer before admitting it, and
181    /// [`Realtime::reserve`] is what actually claims a slot.
182    pub(crate) fn user_conn_count(&self, user: Uuid) -> usize {
183        *self
184            .user_conns
185            .lock()
186            .unwrap_or_else(|poisoned| poisoned.into_inner())
187            .get(&user)
188            .unwrap_or(&0)
189    }
190
191    /// Claim one of `user`'s [`MAX_CONNS_PER_USER`] slots. `None` when they are
192    /// all taken; the returned guard releases the slot on drop, so a panicking
193    /// session cannot strand one.
194    fn reserve(&self, api: &Api, user: Uuid) -> Option<UserSlot> {
195        let mut counts = self
196            .user_conns
197            .lock()
198            .unwrap_or_else(|poisoned| poisoned.into_inner());
199        let count = counts.entry(user).or_insert(0);
200        if *count >= MAX_CONNS_PER_USER {
201            return None;
202        }
203        *count += 1;
204        Some(UserSlot {
205            api: api.clone(),
206            user,
207        })
208    }
209
210    fn release(&self, user: Uuid) {
211        let mut counts = self
212            .user_conns
213            .lock()
214            .unwrap_or_else(|poisoned| poisoned.into_inner());
215        if let Some(count) = counts.get_mut(&user) {
216            *count = count.saturating_sub(1);
217            if *count == 0 {
218                counts.remove(&user);
219            }
220        }
221    }
222
223    /// Join a connection's rooms and queue its `Welcome`, atomically.
224    ///
225    /// This is the one path that holds the hub lock while taking the conns lock,
226    /// and that nesting is what makes the `Welcome` provably first. A concurrent
227    /// [`Realtime::route`] has to resolve its recipients under the hub lock, so it
228    /// either ran *before* this section — in which case the connection was not yet
229    /// in the room, and the event it carried is instead in the voice snapshot
230    /// sampled here — or it runs *after*, and queues behind the `Welcome` already
231    /// in the mailbox. There is no third case, so no event is lost and none can
232    /// precede the `Welcome`.
233    ///
234    /// The caller must register the mailbox before calling this: a fanout that
235    /// resolves the connection as a recipient and finds no mailbox drops the
236    /// event silently.
237    ///
238    /// `snapshot` is invoked by the hub *after* the rooms are joined. It may only
239    /// read state that nothing reads-or-writes while holding the hub lock; the
240    /// voice-participant registry qualifies, because its writers release it before
241    /// they emit.
242    fn admit(
243        &self,
244        conn: ConnId,
245        connect: Connect,
246        snapshot: impl FnOnce() -> Vec<VoiceJoinEvent>,
247    ) -> Result<(), Box<ServerMessage>> {
248        let mut hub = self.hub();
249        let outgoings = hub.connect(conn, connect, snapshot)?;
250        let conns = self.conns();
251        for out in &outgoings {
252            for recipient in hub.recipients(&out.target) {
253                if let Some(tx) = conns.get(&recipient) {
254                    // A mailbox this fresh cannot be full, and a `Welcome` that
255                    // cannot be queued means the session is already gone.
256                    let _ = tx.try_send(out.message.clone());
257                }
258            }
259        }
260        Ok(())
261    }
262
263    /// Resolve each outgoing's target and push it onto every recipient's mailbox.
264    pub(crate) fn route(&self, outgoings: &[Outgoing]) {
265        if outgoings.is_empty() {
266            return;
267        }
268        let plan: Vec<(Vec<ConnId>, &ServerMessage)> = {
269            let hub = self.hub();
270            outgoings
271                .iter()
272                .map(|out| (hub.recipients(&out.target), &out.message))
273                .collect()
274        };
275
276        let mut overflowed: Vec<ConnId> = Vec::new();
277        {
278            let conns = self.conns();
279            for (recipients, message) in plan {
280                for conn in recipients {
281                    let Some(tx) = conns.get(&conn) else { continue };
282                    // Sync, so the lock is never held across an await. A closed
283                    // mailbox is a session already tearing down — nothing to do.
284                    if let Err(mpsc::error::TrySendError::Full(_)) = tx.try_send(message.clone())
285                        && !overflowed.contains(&conn)
286                    {
287                        overflowed.push(conn);
288                    }
289                }
290            }
291        }
292        self.disconnect_overflowed(&overflowed);
293    }
294
295    /// Disconnect connections whose mailbox overflowed.
296    ///
297    /// A full mailbox means the client has stopped draining
298    /// [`MAILBOX_CAPACITY`] control frames. Dropping individual events instead
299    /// would leave it silently inconsistent — it cannot know which it missed —
300    /// so the connection is ended and its recovery is a reconnect, whose
301    /// `Welcome` is a full resync by design. Dropping the mailbox *is* the
302    /// signal: the session's `recv` ends once it has drained what is queued.
303    fn disconnect_overflowed(&self, overflowed: &[ConnId]) {
304        if overflowed.is_empty() {
305            return;
306        }
307        let mut conns = self.conns();
308        for &conn in overflowed {
309            if conns.remove(&conn).is_some() {
310                metrics::counter!("backend_realtime_mailbox_overflows_total").increment(1);
311                tracing::warn!(
312                    conn = conn.0,
313                    "realtime: mailbox full — disconnecting; the client resyncs on reconnect"
314                );
315            }
316        }
317    }
318
319    /// Drop a user's rooms for a community they no longer belong to.
320    pub(crate) fn revoke(&self, user_id: Uuid, community_id: Uuid) {
321        let outgoings = self.hub().revoke(user_id, community_id);
322        self.route(&outgoings);
323    }
324}

364/// Drive one authenticated connection for its full lifetime, whichever leg it
365/// arrived on.
366///
367/// `user_id` is the identity from the validated token — never client-supplied —
368/// and is what every membership check and every stamped event uses.
369pub(crate) async fn run_session<P>(mut pipe: P, api: Api, user_id: Uuid, transport: Transport)
370where
371    P: MessagePipe + Liveness,
372{
373    let limits = api.realtime.limits;
374
375    // The per-user cap is claimed before anything else is waited on, so
376    // connections parked in the handshake count against it too.
377    let Some(slot) = api.realtime.reserve(&api, user_id) else {
378        tracing::warn!(
379            %user_id,
380            transport = transport.label(),
381            cap = MAX_CONNS_PER_USER,
382            "realtime: refused — per-user connection cap reached"
383        );
384        metrics::counter!("backend_realtime_refused_total", "reason" => "conn_cap").increment(1);
385        pipe.close().await;
386        return;
387    };
388    let conn = api.realtime.allocate_conn();
389
390    // From here on every exit — return, `break`, or panic — runs teardown.
391    let mut guard = SessionGuard {
392        api: api.clone(),
393        conn,
394        user_id,
395        transport,
396        admitted: false,
397        _slot: slot,
398    };
399
400    // The first frame must be Hello, and must arrive promptly: a connection that
401    // sends nothing is indistinguishable from one probing for free resources.
402    let hello =
403        tokio::time::timeout(limits.hello_deadline, pipe.recv_json::<ClientMessage>()).await;
404    let Ok(Some(ClientMessage::Hello { protocol_version })) = hello else {
405        let reason = if hello.is_err() {
406            "no_hello"
407        } else {
408            "bad_hello"
409        };
410        tracing::warn!(
411            %user_id,
412            transport = transport.label(),
413            reason,
414            "realtime: closing before admission"
415        );
416        metrics::counter!("backend_realtime_refused_total", "reason" => reason).increment(1);
417        pipe.close().await;
418        return;
419    };
420
421    let communities = match api.find_community_ids_by_user(user_id).await {
422        Ok(communities) => communities,
423        Err(error) => {
424            tracing::error!(%error, %user_id, "realtime: failed to load memberships");
425            pipe.close().await;
426            return;
427        }
428    };
429
430    // The mailbox is registered BEFORE the hub join: a fanout that resolves this
431    // connection as a recipient and finds no mailbox drops the event silently, so
432    // the window between "in the room" and "reachable" must not exist.
433    let (tx, mut rx) = mpsc::channel::<ServerMessage>(MAILBOX_CAPACITY);
434    api.realtime.conns().insert(conn, tx);
435    guard.admitted = true;
436
437    let admitted = api.realtime.admit(
438        conn,
439        Connect {
440            protocol_version,
441            user: user_id,
442            communities: communities.clone(),
443        },
444        || api.voice_roster_for_communities(&communities),
445    );
446    if let Err(goodbye) = admitted {
447        // Version mismatch: the Err IS the frame to send before closing.
448        if let Err(error) = pipe.send_json(&goodbye).await {
449            tracing::debug!(%error, %user_id, "realtime: goodbye frame not delivered");
450        }
451        pipe.close().await;
452        return;
453    }
454
455    metrics::counter!("backend_realtime_connects_total", "transport" => transport.label())
456        .increment(1);
457    metrics::gauge!("backend_realtime_connections", "transport" => transport.label())
458        .increment(1.0);
459    tracing::debug!(conn = conn.0, %user_id, transport = transport.label(), "realtime: connected");
460
461    // A leg whose transport reaps a vanished peer for us needs no probing; one
462    // that does not (a WebSocket over TCP) gets a server-driven ping and an
463    // inbound-silence deadline.
464    let probing = transport.needs_server_liveness();
465    let mut ping = tokio::time::interval_at(
466        tokio::time::Instant::now() + limits.ping_interval,
467        limits.ping_interval,
468    );
469    ping.set_missed_tick_behavior(MissedTickBehavior::Delay);
470    let mut seen_inbound = pipe.inbound_count();
471    let mut idle_at = tokio::time::Instant::now() + limits.idle_timeout;
472
473    loop {
474        // Only ONE branch borrows the pipe, and the borrow ends with the select,
475        // so the outgoing send below can take it again. `recv_json` is
476        // cancel-safe (decode state lives in the pipe), so losing this race
477        // never drops a frame. The timers are absolute deadlines, so re-creating
478        // their futures each turn is likewise safe.
479        let step = tokio::select! {
480            queued = rx.recv() => match queued {
481                Some(message) => Step::Send(Box::new(message)),
482                None => break,
483            },
484            inbound = pipe.recv_json::<ClientMessage>() => match inbound {
485                Some(message) => Step::Received(message),
486                None => break,
487            },
488            _ = ping.tick(), if probing => Step::Ping,
489            _ = tokio::time::sleep_until(idle_at), if probing => Step::IdleCheck,
490        };
491
492        match step {
493            Step::Send(message) => {
494                if let Err(error) = pipe.send_json(&message).await {
495                    tracing::debug!(
496                        %error, conn = conn.0, %user_id, transport = transport.label(),
497                        "realtime: send failed — ending session"
498                    );
499                    break;
500                }
501            }
502            Step::Ping => {
503                if let Err(error) = pipe.ping().await {
504                    tracing::debug!(
505                        %error, conn = conn.0, %user_id, transport = transport.label(),
506                        "realtime: liveness probe failed — ending session"
507                    );
508                    break;
509                }
510            }
511            Step::IdleCheck => {
512                // Any inbound frame counts, protocol or not: a peer answering
513                // pings is alive even when it has nothing to say.
514                let inbound = pipe.inbound_count();
515                if inbound == seen_inbound {
516                    tracing::info!(
517                        conn = conn.0, %user_id, transport = transport.label(),
518                        timeout = ?limits.idle_timeout,
519                        "realtime: no inbound frame within the idle timeout — reaping"
520                    );
521                    metrics::counter!(
522                        "backend_realtime_idle_reaps_total", "transport" => transport.label()
523                    )
524                    .increment(1);
525                    break;
526                }
527                seen_inbound = inbound;
528                idle_at = tokio::time::Instant::now() + limits.idle_timeout;
529            }
530            Step::Received(ClientMessage::Bye) => break,
531            Step::Received(message) => {
532                seen_inbound = pipe.inbound_count();
533                idle_at = tokio::time::Instant::now() + limits.idle_timeout;
534
535                let Some(input) = resolve(&api, user_id, message).await else {
536                    continue;
537                };
538                let outgoings = api.realtime.hub().apply(conn, input);
539                api.realtime.route(&outgoings);
540            }
541        }
542    }
543
544    // Registry teardown, the connection gauge, and the voice-eviction timer are
545    // the guard's job — it runs on a panic too. Closing the pipe is not, because
546    // it awaits.
547    drop(guard);
548    pipe.close().await;
549}

551/// Runs a session's teardown exactly once, on drop.
552///
553/// A session task that panics mid-loop must not leave its connection in the hub,
554/// its mailbox in the registry, or the connection gauge over-counting for the
555/// life of the pod — so every one of those lives here rather than after the loop.
556struct SessionGuard {
557    api: Api,
558    conn: ConnId,
559    user_id: Uuid,
560    transport: Transport,
561    /// `false` until the mailbox is registered: a session refused before
562    /// admission has nothing to tear down but its user slot.
563    admitted: bool,
564    /// Dropped with the guard, releasing the per-user connection slot.
565    _slot: UserSlot,
566}
567
568impl Drop for SessionGuard {
569    fn drop(&mut self) {
570        if !self.admitted {
571            return;
572        }
573        // Drop the mailbox first so nothing is routed to a connection that is
574        // going away, then clear its rooms.
575        self.api.realtime.conns().remove(&self.conn);
576        let outgoings = self.api.realtime.hub().disconnect(self.conn);
577        self.api.realtime.route(&outgoings);
578
579        metrics::counter!(
580            "backend_realtime_disconnects_total", "transport" => self.transport.label()
581        )
582        .increment(1);
583        metrics::gauge!("backend_realtime_connections", "transport" => self.transport.label())
584            .decrement(1.0);
585        tracing::debug!(
586            conn = self.conn.0, user_id = %self.user_id, transport = self.transport.label(),
587            "realtime: disconnected"
588        );
589
590        schedule_voice_eviction(self.api.clone(), self.user_id);
591    }
592}

594/// What one turn of the session loop does. The outgoing message is boxed to
595/// keep the enum pointer-sized — negligible against the JSON encode that
596/// immediately follows it.
597enum Step {
598    Send(Box<ServerMessage>),
599    Received(ClientMessage),
600    /// Probe the peer's transport.
601    Ping,
602    /// The inbound-silence deadline elapsed; decide whether anything arrived.
603    IdleCheck,
604}

606/// Turn a client frame into a hub input, resolving anything the reducer cannot
607/// know. `None` means the frame was handled here (or is not actionable) and the
608/// hub sees nothing.
609async fn resolve(api: &Api, user_id: Uuid, message: ClientMessage) -> Option<Input> {
610    match message {
611        // The handshake is consumed at admission; a duplicate is a no-op.
612        ClientMessage::Hello { .. } | ClientMessage::Bye => None,
613
614        ClientMessage::SubscribeSpace { space_id } => {
615            let (community_id, auth) = authorize_space(api, user_id, space_id).await;
616            Some(Input::Subscribe {
617                space_id,
618                community_id,
619                auth,
620            })
621        }
622
623        ClientMessage::UnsubscribeSpace { space_id } => Some(Input::Unsubscribe { space_id }),
624
625        ClientMessage::Screenshare { space_id, enabled } => {
626            // Authorized exactly like a subscribe: a screenshare is a broadcast
627            // into a community's room, so a non-member must not be able to put
628            // one there.
629            let (community_id, auth) = authorize_space(api, user_id, space_id).await;
630            Some(Input::Screenshare {
631                event: VoiceScreenShareEvent {
632                    user_id,
633                    space_id,
634                    // The event is stamped server-side: the client supplies only
635                    // the space and the flag, never whose share it is or where it
636                    // fans out. With no space there is no community to stamp, and
637                    // `auth` is never `Allowed` in that case, so the nil id is
638                    // never routed anywhere.
639                    community_id: community_id.unwrap_or_default(),
640                    enabled,
641                },
642                auth,
643            })
644        }
645    }
646}

648/// Resolve a space's owning community and whether `user_id` may see it.
649///
650/// Three-way on purpose. A database failure is [`Authorization::Unavailable`],
651/// never `Denied`: telling a member they are not one because a query failed is
652/// both wrong and unrecoverable from the client's side, where a retryable answer
653/// is exactly what the situation calls for.
654async fn authorize_space(
655    api: &Api,
656    user_id: Uuid,
657    space_id: Uuid,
658) -> (Option<Uuid>, Authorization) {
659    let space = match api.find_space_by_id(space_id).await {
660        Ok(Some(space)) => space,
661        Ok(None) => {
662            tracing::warn!(%space_id, %user_id, "realtime: request for an unknown space");
663            return (None, Authorization::Denied);
664        }
665        Err(error) => {
666            tracing::error!(%error, %space_id, %user_id, "realtime: space lookup failed");
667            return (None, Authorization::Unavailable);
668        }
669    };
670    let community_id = space.community_id;
671    match api.is_community_member(community_id, user_id).await {
672        Ok(true) => (Some(community_id), Authorization::Allowed),
673        Ok(false) => (Some(community_id), Authorization::Denied),
674        Err(error) => {
675            tracing::error!(%error, %space_id, %user_id, "realtime: membership check failed");
676            (Some(community_id), Authorization::Unavailable)
677        }
678    }
679}

681/// After a connection drops, evict the user from voice once the grace period
682/// passes — unless a newer session (or a voice rejoin) has bumped the generation
683/// in the meantime, which cancels this eviction.
684fn schedule_voice_eviction(api: Api, user_id: Uuid) {
685    // Reached from a drop guard, which may run while the runtime is shutting
686    // down; spawning there would panic inside an unwind.
687    if tokio::runtime::Handle::try_current().is_err() {
688        return;
689    }
690    let generation = api.bump_eviction_generation(user_id);
691    tokio::spawn(async move {
692        tokio::time::sleep(VOICE_EVICTION_GRACE).await;
693        if api.current_eviction_generation(user_id) == generation {
694            api.evict_voice_participant_by_user(user_id).await;
695        }
696    });
697}

699/// Zero-initialise the per-transport connection series. Called from
700/// [`crate::api::metrics::init_series`] so a fresh pod exposes both legs'
701/// counters and gauges before either has seen a connection.
702pub(crate) fn init_metrics() {
703    for transport in TRANSPORTS {
704        metrics::counter!("backend_realtime_connects_total", "transport" => transport.label())
705            .increment(0);
706        metrics::counter!("backend_realtime_disconnects_total", "transport" => transport.label())
707            .increment(0);
708        metrics::counter!("backend_realtime_idle_reaps_total", "transport" => transport.label())
709            .increment(0);
710        metrics::gauge!("backend_realtime_connections", "transport" => transport.label()).set(0.0);
711    }
712    for reason in ["conn_cap", "no_hello", "bad_hello"] {
713        metrics::counter!("backend_realtime_refused_total", "reason" => reason).increment(0);
714    }
715    metrics::counter!("backend_realtime_mailbox_overflows_total").increment(0);
716    crate::api::realtime::wt::init_metrics();
717}
