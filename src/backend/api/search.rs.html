1//! Message search: `POST /search`.
2//!
3//! One endpoint, two engines. The client sends a parsed
4//! [`hound_search::Query`] — never the raw text — and this module compiles it
5//! to whichever dialect the connection speaks: a PGroonga full-text match on
6//! Postgres, an FTS5 match on SQLite (see [`sql`]). Everything else about the
7//! query — authors, mentions, content facets, date bounds, the keyset — is
8//! identical SQL on both.
9//!
10//! ## The two halves
11//!
12//! 1. **Which spaces**, resolved from the request's scope and the caller's
13//!    permissions ([`Api::searchable_spaces`]). The resulting id list is a
14//!    `space_id IN (…)` predicate, so a space the caller cannot read is
15//!    structurally absent from the search rather than filtered out of its
16//!    results. `in:` only ever narrows that list.
17//! 2. **Which messages**, a single statement returning `(id, score)` pairs,
18//!    then a batch hydration into the same shape the message history endpoint
19//!    returns ([`hydrate`]).
20//!
21//! ## Ordering and paging
22//!
23//! Newest is `id DESC` — message ids are UUIDv7, so that is send order and it
24//! is the primary key. Relevant is `score DESC, id DESC`. Both page by keyset
25//! ([`cursor`]) rather than offset: a search re-run against a table that has
26//! grown must not repeat or skip a row at a page boundary. There is no total
27//! count; the client pages until `next_cursor` is null.
28//!
29//! Relevance needs something to score against, so a query with no positive
30//! text term is sorted newest-first whatever the request asked for — on
31//! SQLite there is no FTS row to take a `bm25` from at all.

55/// The `limit` a request may ask for. The ceiling is a page a client can
56/// render and a hydration this endpoint can do in a fixed handful of queries,
57/// not a database limit.
58const LIMIT_RANGE: std::ops::RangeInclusive<u32> = 1..=50;
59
60/// What a request that omits `limit` gets.
61const DEFAULT_LIMIT: u32 = 25;

68/// Which communities a search covers before `in:` narrows it.
69///
70/// Serialises as `{"community": "<uuid>"}` or the bare string `"everywhere"`
71/// — serde's external tagging, which is exactly the shape the UI's scope pill
72/// reads as (a named community, or the pill deleted).
73#[derive(
74    Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize, utoipa::ToSchema,
75)]
76#[serde(rename_all = "snake_case")]
77pub enum SearchScope {
78    /// One community. 403 if the caller is not a member of it.
79    Community(Uuid),
80    /// Every community the caller is a member of.
81    Everywhere,
82}
83
84/// Result ordering.
85#[derive(
86    Debug,
87    Clone,
88    Copy,
89    Default,
90    PartialEq,
91    Eq,
92    serde::Serialize,
93    serde::Deserialize,
94    utoipa::ToSchema,
95)]
96#[serde(rename_all = "snake_case")]
97pub enum SearchSort {
98    /// Most recently sent first.
99    #[default]
100    Newest,
101    /// Best text match first, most recent breaking ties. Degrades to
102    /// [`SearchSort::Newest`] when the query has no positive text term —
103    /// there is nothing to score.
104    Relevant,
105}
106
107/// A search request.
108#[derive(Debug, serde::Serialize, serde::Deserialize, utoipa::ToSchema)]
109pub struct SearchRequest {
110    pub scope: SearchScope,
111    /// The parsed query. Absent (or `{}`) is a legitimate "newest messages in
112    /// scope" browse, not an error.
113    #[serde(default)]
114    pub query: hound_search::Query,
115    #[serde(default)]
116    pub sort: SearchSort,
117    /// Opaque; the `next_cursor` of the previous page. Backend-private —
118    /// clients pass it back untouched and never construct one.
119    #[serde(default)]
120    pub cursor: Option<String>,
121    /// 1..=50.
122    #[serde(default = "default_limit")]
123    pub limit: u32,
124    /// Minutes east of UTC, the caller's own offset. Decides where the day
125    /// boundaries of `before:`/`after:`/`during:`/`today` fall — see
126    /// [`hound_search::DateContext`].
127    #[serde(default)]
128    pub utc_offset_minutes: i32,
129}
130
131fn default_limit() -> u32 {
132    DEFAULT_LIMIT
133}

135/// One matching message.
136#[derive(Debug, serde::Serialize, serde::Deserialize, utoipa::ToSchema)]
137pub struct SearchHit {
138    pub message: CommunitySpaceMessageModel,
139    /// The community owning the hit's space — a search may span several, and
140    /// a result row has to name the community it will jump into.
141    #[schema(example = "018c5b6e-0000-7000-0000-000000000001")]
142    pub community_id: Uuid,
143    /// Relevance, higher is better. `0` under [`SearchSort::Newest`] and for
144    /// any query with no positive text term.
145    pub score: f64,
146}

148/// A page of search results.
149#[derive(Debug, serde::Serialize, serde::Deserialize, utoipa::ToSchema)]
150pub struct SearchResults {
151    pub hits: Vec<SearchHit>,
152    /// Merged profiles for every hit's author, plus everyone the hits
153    /// `<hound:user/…>` reference — hoisted exactly as `GET
154    /// /space/{id}/message` hoists them, so a result row renders without a
155    /// follow-up profile fetch.
156    pub users: Vec<MergedProfile>,
157    /// Pass back as `cursor` for the next page. `null` when this page is the
158    /// last one.
159    pub next_cursor: Option<String>,
160}
161
162impl SearchResults {
163    fn empty() -> Self {
164        SearchResults {
165            hits: Vec::new(),
166            users: Vec::new(),
167            next_cursor: None,
168        }
169    }
170}

186/// Search messages the caller can read.
187pub(crate) async fn post_search(
188    LoggedIn(claims): LoggedIn,
189    State(api): State<Api>,
190    Json(request): Json<SearchRequest>,
191) -> ApiResult<Json<SearchResults>> {
192    api.search_messages(claims.sub, request).await.map(Json)
193}
194
195impl Api {
196    async fn search_messages(
197        &self,
198        user_id: Uuid,
199        request: SearchRequest,
200    ) -> ApiResult<SearchResults> {
201        if !LIMIT_RANGE.contains(&request.limit) {
202            return Err(ApiError::BadRequest);
203        }
204        let cursor = request.cursor.as_deref().map(cursor::decode).transpose()?;
205
206        // Asking to rank by relevance without a term to rank against is not
207        // an error — it is what the UI sends the instant a user deletes the
208        // last word of a query it had already put in Relevant mode.
209        let sort = if request.query.has_positive_text() {
210            request.sort
211        } else {
212            SearchSort::Newest
213        };
214
215        let (spaces, community_of_space) = self
216            .searchable_spaces(request.scope, user_id, &request.query.in_spaces)
217            .await?;
218        if spaces.is_empty() {
219            return Ok(SearchResults::empty());
220        }
221
222        let context = hound_search::DateContext {
223            now_ms: chrono::Utc::now().timestamp_millis(),
224            utc_offset_minutes: request.utc_offset_minutes,
225        };
226        let plan = sql::Plan {
227            spaces: &spaces,
228            query: &request.query,
229            sort,
230            cursor: cursor.as_ref(),
231            limit: request.limit,
232            bounds: request.query.id_bounds(&context),
233        };
234        let Some(statement) = sql::statement(self.db.get_database_backend(), &plan) else {
235            // A positive term nothing can tokenize, e.g. `"+++"`.
236            return Ok(SearchResults::empty());
237        };
238
239        let rows = self.run_search(statement).await?;
240        // A short page is the last page. A full one might be, but claiming so
241        // would need a lookahead row; an empty follow-up page is cheaper than
242        // being wrong.
243        let next_cursor = (rows.len() as u32 == request.limit)
244            .then(|| rows.last())
245            .flatten()
246            .map(|row| {
247                cursor::encode(&cursor::Cursor {
248                    id: row.id,
249                    score: matches!(sort, SearchSort::Relevant).then_some(row.score),
250                    v: cursor::VERSION,
251                })
252            });
253
254        let (hits, users) = self.hydrate_hits(rows, &community_of_space).await?;
255        Ok(SearchResults {
256            hits,
257            users,
258            next_cursor,
259        })
260    }
261
262    /// Runs the compiled statement.
263    ///
264    /// Postgres runs it inside a transaction with `enable_seqscan` off:
265    /// `pgroonga_score` reads a per-row score the *index scan* records, so a
266    /// planner that chose a sequential scan would return every score as zero
267    /// and silently flatten the relevance ordering.
268    async fn run_search(&self, statement: sea_orm::Statement) -> ApiResult<Vec<sql::HitRow>> {
269        use sea_orm::{FromQueryResult, TransactionTrait};
270
271        if self.db.get_database_backend() != sea_orm::DbBackend::Postgres {
272            return Ok(sql::HitRow::find_by_statement(statement)
273                .all(&self.db)
274                .await?);
275        }
276        let txn = self.db.begin().await?;
277        txn.execute_unprepared("SET LOCAL enable_seqscan = off")
278            .await?;
279        let rows = sql::HitRow::find_by_statement(statement).all(&txn).await?;
280        txn.commit().await?;
281        Ok(rows)
282    }

284    /// The spaces a search may read, and the community each belongs to.
285    ///
286    /// The whole ACL, resolved once: scope → communities → every space in
287    /// them → [`Api::filter_accessible_spaces`] → intersect with `in:`. An
288    /// `in:` naming a space the caller cannot read is silently dropped rather
289    /// than refused — the id came from a client-side channel list, and a
290    /// permission change between rendering that list and searching is not the
291    /// user's mistake.
292    async fn searchable_spaces(
293        &self,
294        scope: SearchScope,
295        user_id: Uuid,
296        in_spaces: &[Uuid],
297    ) -> ApiResult<(Vec<Uuid>, HashMap<Uuid, Uuid>)> {
298        let communities = match scope {
299            SearchScope::Community(community_id) => {
300                if !self.is_community_member(community_id, user_id).await? {
301                    return Err(ApiError::NotInCommunity(user_id, community_id));
302                }
303                vec![community_id]
304            }
305            SearchScope::Everywhere => self.find_community_ids_by_user(user_id).await?,
306        };
307
308        let mut spaces = Vec::new();
309        let mut community_of_space = HashMap::new();
310        for community_id in communities {
311            let all = self.find_all_spaces_by_community(community_id).await?;
312            for space in self
313                .filter_accessible_spaces(community_id, user_id, all)
314                .await?
315            {
316                if !in_spaces.is_empty() && !in_spaces.contains(&space.id) {
317                    continue;
318                }
319                community_of_space.insert(space.id, community_id);
320                spaces.push(space.id);
321            }
322        }
323        Ok((spaces, community_of_space))
324    }
