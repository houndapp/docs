1//! HTML → raw metadata. Parsing only: nothing here decides what an embed says,
2//! which is [`super::build`]'s job.
3//!
4//! ## Why document order, not a map
5//!
6//! Open Graph's *structured properties* (`og:image:width`, `og:image:alt`, …)
7//! attach to the most recently declared parent property, so a page with three
8//! images is three `og:image` tags each followed by its own dimensions. Folding
9//! those into a `HashMap` — the obvious first implementation — silently keeps
10//! one image and staples the last width onto it. So the tags come out of here as
11//! an ordered list and stay that way until the builder walks it.
12//!
13//! ## Charset
14//!
15//! Decoded through `encoding_rs` with the label from the `Content-Type` header,
16//! falling back to a `<meta charset>` sniff over the head of the document and
17//! then to UTF-8. Real pages are still served as `windows-1252`, and reading one
18//! as UTF-8 turns every curly quote in a title into a replacement character.
19
20use scraper::{Html, Selector};

22/// One `<link rel="icon">` candidate.
23#[derive(Debug, Clone)]
24pub(crate) struct IconLink {
25    pub href: String,
26    /// The `sizes` attribute's largest edge, if it parsed. Used only to prefer a
27    /// bigger icon; `None` sorts last.
28    pub size: Option<u32>,
29}

31/// Everything scraped from one HTML document, before any precedence is applied.
32#[derive(Debug, Default)]
33pub(crate) struct PageMetadata {
34    /// `og:*` properties, keys lowercased and `og:` stripped, in document order.
35    pub og: Vec<(String, String)>,
36    /// `twitter:*` names, keys lowercased and `twitter:` stripped, in document
37    /// order.
38    pub twitter: Vec<(String, String)>,
39    pub html_title: Option<String>,
40    pub html_description: Option<String>,
41    /// `<meta name="author">` — the last-resort byline.
42    pub html_author: Option<String>,
43    /// `<meta name="theme-color">`, unvalidated — the builder decides whether it
44    /// is a colour it can hand to a renderer.
45    pub theme_color: Option<String>,
46    pub icons: Vec<IconLink>,
47    /// `<link rel="canonical">`.
48    pub canonical: Option<String>,
49    /// oEmbed endpoints from `<link rel="alternate">`, JSON ones first — the XML
50    /// flavour is legal but nothing here parses it, so it is kept only so the
51    /// debugger can say the site offered one.
52    pub oembed_json: Option<String>,
53    pub oembed_xml: Option<String>,
54    /// `<html lang>`.
55    pub lang: Option<String>,
56}

58/// Decode a response body to text, honouring the declared charset.
59pub(crate) fn decode_html(bytes: &[u8], content_type: Option<&str>) -> String {
60    let label = content_type
61        .and_then(charset_from_content_type)
62        .or_else(|| sniff_meta_charset(bytes));
63    let encoding = label
64        .and_then(|l| encoding_rs::Encoding::for_label(l.as_bytes()))
65        .unwrap_or(encoding_rs::UTF_8);
66    let (text, _, _) = encoding.decode(bytes);
67    text.into_owned()
68}

70fn charset_from_content_type(value: &str) -> Option<String> {
71    value.split(';').skip(1).find_map(|param| {
72        let (k, v) = param.split_once('=')?;
73        (k.trim().eq_ignore_ascii_case("charset"))
74            .then(|| v.trim().trim_matches('"').to_ascii_lowercase())
75    })
76}

78/// Look for `<meta charset=…>` / `<meta http-equiv=content-type …>` in the head
79/// of the document.
80///
81/// Bounded to the first 4 KiB deliberately: per the HTML spec a declaration
82/// after that point does not apply, and scanning a whole megabyte for it would
83/// be a second pass over the body for nothing.
84fn sniff_meta_charset(bytes: &[u8]) -> Option<String> {
85    let head = &bytes[..bytes.len().min(4096)];
86    let text = String::from_utf8_lossy(head).to_ascii_lowercase();
87    let at = text.find("charset")?;
88    let rest = &text[at + "charset".len()..];
89    let rest = rest.trim_start().strip_prefix('=')?.trim_start();
90    let value: String = rest
91        .trim_start_matches(['"', '\''])
92        .chars()
93        .take_while(|c| c.is_ascii_alphanumeric() || *c == '-' || *c == '_')
94        .collect();
95    (!value.is_empty()).then_some(value)
96}

98/// Scrape one document.
99///
100/// Attribute values arrive entity-decoded from the parser, so `&amp;` in an
101/// `og:url` is already `&` — nothing downstream needs to unescape, and nothing
102/// should try to.
103pub(crate) fn scrape(html: &str) -> PageMetadata {
104    let document = Html::parse_document(html);
105    let mut meta = PageMetadata::default();
106
107    // `Selector::parse` on a literal cannot fail; `expect` documents that rather
108    // than threading a Result nobody can act on.
109    let meta_sel = Selector::parse("meta").expect("static selector");
110    let title_sel = Selector::parse("title").expect("static selector");
111    let link_sel = Selector::parse("link").expect("static selector");
112    let html_sel = Selector::parse("html").expect("static selector");
113
114    for element in document.select(&meta_sel) {
115        let value = element.attr("content").unwrap_or_default().trim();
116        if value.is_empty() {
117            continue;
118        }
119        // `property` is Open Graph's spelling, `name` is Twitter's — but plenty
120        // of sites use the other one, so both attributes are read for both
121        // vocabularies rather than trusting anyone's spec compliance.
122        let key = element
123            .attr("property")
124            .or_else(|| element.attr("name"))
125            .unwrap_or_default()
126            .trim()
127            .to_ascii_lowercase();
128        if let Some(rest) = key.strip_prefix("og:") {
129            meta.og.push((rest.to_owned(), value.to_owned()));
130        } else if let Some(rest) = key.strip_prefix("twitter:") {
131            meta.twitter.push((rest.to_owned(), value.to_owned()));
132        } else if key == "description" && meta.html_description.is_none() {
133            meta.html_description = Some(value.to_owned());
134        } else if key == "theme-color" && meta.theme_color.is_none() {
135            meta.theme_color = Some(value.to_owned());
136        } else if key == "author" && meta.html_author.is_none() {
137            meta.html_author = Some(value.to_owned());
138        } else if key.starts_with("article:") || key.starts_with("music:") {
139            // Open Graph's vertical extensions are declared WITHOUT the `og:`
140            // prefix (`article:published_time`, `music:duration`). They belong to
141            // the same ordered list so the builder reads them in document order
142            // alongside the properties they qualify.
143            meta.og.push((key.clone(), value.to_owned()));
144        }
145    }
146
147    meta.html_title = document
148        .select(&title_sel)
149        .next()
150        .map(|t| t.text().collect::<String>())
151        .map(|t| t.trim().to_owned())
152        .filter(|t| !t.is_empty());
153
154    for element in document.select(&link_sel) {
155        let rels = element.attr("rel").unwrap_or_default().to_ascii_lowercase();
156        let Some(href) = element
157            .attr("href")
158            .map(str::trim)
159            .filter(|h| !h.is_empty())
160        else {
161            continue;
162        };
163        // `rel` is a space-separated token list ("shortcut icon"), so it is
164        // matched token-wise rather than by equality.
165        let tokens: Vec<&str> = rels.split_whitespace().collect();
166        if tokens.contains(&"canonical") && meta.canonical.is_none() {
167            meta.canonical = Some(href.to_owned());
168        }
169        if tokens.contains(&"icon") || tokens.contains(&"apple-touch-icon") {
170            meta.icons.push(IconLink {
171                href: href.to_owned(),
172                size: element.attr("sizes").and_then(parse_icon_size),
173            });
174        }
175        if tokens.contains(&"alternate") {
176            match element
177                .attr("type")
178                .unwrap_or_default()
179                .to_ascii_lowercase()
180                .as_str()
181            {
182                "application/json+oembed" | "text/json+oembed" => {
183                    meta.oembed_json.get_or_insert_with(|| href.to_owned());
184                }
185                "text/xml+oembed" | "application/xml+oembed" => {
186                    meta.oembed_xml.get_or_insert_with(|| href.to_owned());
187                }
188                _ => {}
189            }
190        }
191    }
192
193    meta.lang = document
194        .select(&html_sel)
195        .next()
196        .and_then(|e| e.attr("lang"))
197        .map(str::trim)
198        .filter(|l| !l.is_empty())
199        .map(str::to_owned);
200
201    // Biggest declared icon first; undeclared sizes last.
202    meta.icons
203        .sort_by_key(|i| std::cmp::Reverse(i.size.unwrap_or(0)));
204    meta
205}

207/// `sizes="32x32"` / `"any"` / `"16x16 32x32"` → the largest edge found.
208fn parse_icon_size(value: &str) -> Option<u32> {
209    value
210        .split_whitespace()
211        .filter_map(|token| {
212            let lowered = token.to_ascii_lowercase();
213            let (w, h) = lowered.split_once('x')?;
214            Some(w.parse::<u32>().ok()?.max(h.parse::<u32>().ok()?))
215        })
216        .max()
217}

219/// First value for `key` in an ordered property list.
220pub(crate) fn first<'a>(pairs: &'a [(String, String)], key: &str) -> Option<&'a str> {
221    pairs
222        .iter()
223        .find(|(k, _)| k == key)
224        .map(|(_, v)| v.as_str())
225}
