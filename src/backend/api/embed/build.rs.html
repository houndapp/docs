1//! Precedence and sanitisation: raw metadata in, a renderable
2//! [`LinkEmbed`] out.
3//!
4//! ## The precedence rules, and why
5//!
6//! Three vocabularies describe the same page and disagree constantly. The order
7//! below is applied per *field*, not per vocabulary, because no single source is
8//! best at everything:
9//!
10//! | field | order |
11//! |---|---|
12//! | title | oEmbed → `og:title` → `twitter:title` → `<title>` |
13//! | description | `og:description` → `twitter:description` → oEmbed → `<meta description>` |
14//! | site name | `og:site_name` → oEmbed `provider_name` → the registrable domain |
15//! | author | oEmbed `author_name` → `article:author` → `<meta author>` |
16//! | images | `og:image` (all of them) → `twitter:image` → oEmbed thumbnail |
17//! | video | `og:video` → `twitter:player:stream` |
18//!
19//! oEmbed wins the title because it is the *service* describing its own
20//! resource, where `og:title` is whatever the page author typed — but it loses
21//! the description, because endpoints routinely return a boilerplate one ("Watch
22//! on …") while `og:description` carries the actual summary.
23//!
24//! ## Sanitisation is not optional here
25//!
26//! Every string leaving this module has been through [`clean_line`] or
27//! [`clean_block`]: control characters removed (they let a title forge line
28//! structure in a renderer that lays out its own text), whitespace collapsed,
29//! and lengths capped. Clients are told they may render these as plain text, and
30//! this is the function that makes that true.
31

36};

42/// What the unfurl learned about one URL, before precedence.
43pub(crate) struct Scraped {
44    /// The URL as it appeared in the message.
45    pub request_url: String,
46    /// The URL that actually answered, after redirects.
47    pub final_url: Url,
48    pub meta: PageMetadata,
49    pub oembed: Option<OEmbed>,
50}

52/// Build the embed for a fetched HTML page.
53///
54/// `warnings` collects everything dropped along the way — a `javascript:` image
55/// URL, an unparseable timestamp, images past the gallery cap. They are the
56/// debugger's payload and are ignored by the message path.
57pub(crate) fn from_html(scraped: Scraped, warnings: &mut Vec<String>) -> LinkEmbed {
58    let Scraped {
59        request_url,
60        final_url,
61        meta,
62        oembed,
63    } = scraped;
64    let base = &final_url;
65    let oe = oembed.as_ref();
66    let mut embed = LinkEmbed::new(request_url);
67
68    // ── the click target ────────────────────────────────────────────────────
69    //
70    // `og:url` is the publisher's canonical form and is normally what a reader
71    // should land on. It is still a URL a stranger controls, so it is accepted
72    // only if it is an absolute http(s) one — a page declaring
73    // `og:url: javascript:…` gets its canonical claim ignored, not honoured.
74    // Written as steps rather than a chain: each `absolute` call needs `warnings`
75    // mutably, and two of those inside one expression is a borrow the compiler
76    // has no reason to accept.
77    let mut canonical = first(&meta.og, "url").and_then(|u| canonical_url(u, warnings, "og:url"));
78    if canonical.is_none()
79        && let Some(declared) = meta.canonical.as_deref()
80    {
81        canonical = canonical_url(declared, warnings, "rel=canonical");
82    }
83    embed.url = canonical
84        .map(String::from)
85        .unwrap_or_else(|| final_url.to_string());
86
87    embed.title = pick_line(&[
88        oe.and_then(|o| o.title.as_deref()),
89        first(&meta.og, "title"),
90        first(&meta.twitter, "title"),
91        meta.html_title.as_deref(),
92    ]);
93
94    embed.description = pick(
95        &[
96            first(&meta.og, "description"),
97            first(&meta.twitter, "description"),
98            oe.and_then(|o| o.description.as_deref()),
99            meta.html_description.as_deref(),
100        ],
101        |s| clean_block(s, MAX_DESCRIPTION_CHARS),
102    );
103
104    embed.site_name = pick_label(&[
105        first(&meta.og, "site_name"),
106        oe.and_then(|o| o.provider_name.as_deref()),
107        first(&meta.twitter, "site"),
108    ])
109    // A domain is a poor site name but a good one is rare, and "example.com"
110    // beats an unlabelled card.
111    .or_else(|| {
112        base.host_str()
113            .map(|h| h.trim_start_matches("www.").to_owned())
114    });
115
116    if let Some(name) = pick_label(&[
117        oe.and_then(|o| o.author_name.as_deref()),
118        first(&meta.og, "article:author"),
119        meta.html_author.as_deref(),
120        first(&meta.twitter, "creator"),
121    ]) {
122        let declared_url = oe
123            .and_then(|o| o.author_url.as_deref())
124            .or_else(|| first(&meta.og, "article:author"));
125        embed.author = Some(EmbedAuthor {
126            url: declared_url
127                .and_then(|u| absolute(base, u, warnings, "author url"))
128                .map(String::from),
129            // `article:author` is often a profile URL rather than a name; when
130            // it is, the label would read as a raw URL, so the last path
131            // segment stands in for it.
132            name: name_or_handle(&name),
133        });
134    }
135
136    if let Some(name) = oe
137        .and_then(|o| o.provider_name.as_deref())
138        .and_then(|n| clean_line(n, MAX_LABEL_CHARS))
139    {
140        let declared_url = oe.and_then(|o| o.provider_url.as_deref());
141        embed.provider = Some(EmbedProvider {
142            name,
143            url: declared_url
144                .and_then(|u| absolute(base, u, warnings, "provider url"))
145                .map(String::from),
146        });
147    }
148
149    embed.accent_color = meta
150        .theme_color
151        .as_deref()
152        .and_then(|c| normalise_hex_colour(c, warnings));
153
154    embed.published_at =
155        first(&meta.og, "article:published_time").and_then(|t| parse_timestamp(t, warnings));
156
157    embed.locale = pick_label(&[first(&meta.og, "locale"), meta.lang.as_deref()]);
158
159    // ── media ───────────────────────────────────────────────────────────────
160
161    let mut images: Vec<EmbedMedia> = Vec::new();
162    for candidate in collect_structured(&meta.og, "image") {
163        // `og:image:secure_url` is the https form of the same asset; prefer it,
164        // because the proxy would otherwise fetch a cleartext copy of something
165        // available over TLS.
166        let href = candidate.secure_url.as_deref().unwrap_or(&candidate.url);
167        let Some(url) = absolute(base, href, warnings, "og:image") else {
168            continue;
169        };
170        images.push(EmbedMedia {
171            url: String::new(), // stamped with the proxy path once the row has an id
172            source_url: url.to_string(),
173            content_type: candidate.mime.clone(),
174            width: candidate.width,
175            height: candidate.height,
176            alt: candidate
177                .alt
178                .as_deref()
179                .and_then(|a| clean_line(a, MAX_LABEL_CHARS)),
180        });
181    }
182    if images.is_empty() {
183        let declared_twitter =
184            first(&meta.twitter, "image").or_else(|| first(&meta.twitter, "image:src"));
185        let twitter_image =
186            declared_twitter.and_then(|u| absolute(base, u, warnings, "twitter:image"));
187        // For `type: photo` oEmbed's `url` IS the full-size image and the
188        // thumbnail is a reduction of it, so the real thing wins.
189        let declared_oembed = oe
190            .filter(|o| {
191                o.r#type
192                    .as_deref()
193                    .is_some_and(|t| t.eq_ignore_ascii_case("photo"))
194            })
195            .and_then(|o| o.url.as_deref())
196            .or_else(|| oe.and_then(|o| o.thumbnail_url.as_deref()));
197        let thumbnail = declared_oembed.and_then(|u| absolute(base, u, warnings, "oembed image"));
198        if let Some(url) = twitter_image {
199            images.push(EmbedMedia {
200                url: String::new(),
201                source_url: url.to_string(),
202                content_type: None,
203                width: None,
204                height: None,
205                alt: first(&meta.twitter, "image:alt").and_then(|a| clean_line(a, MAX_LABEL_CHARS)),
206            });
207        } else if let Some(url) = thumbnail {
208            images.push(EmbedMedia {
209                url: String::new(),
210                source_url: url.to_string(),
211                content_type: None,
212                width: oe.and_then(|o| o.thumbnail_width).and_then(|d| d.as_u32()),
213                height: oe.and_then(|o| o.thumbnail_height).and_then(|d| d.as_u32()),
214                alt: None,
215            });
216        }
217    }
218    if images.len() > MAX_EMBED_IMAGES {
219        warnings.push(format!(
220            "{} images declared; keeping the first {MAX_EMBED_IMAGES}",
221            images.len()
222        ));
223        images.truncate(MAX_EMBED_IMAGES);
224    }
225    embed.images = images;
226
227    embed.video = build_video(&meta, base, warnings);
228    embed.audio = collect_structured(&meta.og, "audio")
229        .into_iter()
230        .next()
231        .and_then(|candidate| {
232            let href = candidate
233                .secure_url
234                .clone()
235                .unwrap_or(candidate.url.clone());
236            let url = absolute(base, &href, warnings, "og:audio")?;
237            Some(EmbedMedia {
238                url: String::new(),
239                source_url: url.to_string(),
240                content_type: candidate.mime,
241                width: None,
242                height: None,
243                alt: None,
244            })
245        });
246
247    embed.favicon = meta.icons.first().and_then(|icon| {
248        let url = absolute(base, &icon.href, warnings, "rel=icon")?;
249        Some(EmbedMedia {
250            url: String::new(),
251            source_url: url.to_string(),
252            content_type: None,
253            width: icon.size,
254            height: icon.size,
255            alt: None,
256        })
257    });
258    // Nothing declared an icon: `/favicon.ico` is the convention every browser
259    // falls back to, and the proxy will simply 404 if it isn't there.
260    if embed.favicon.is_none()
261        && let Ok(guess) = base.join("/favicon.ico")
262    {
263        embed.favicon = Some(EmbedMedia {
264            url: String::new(),
265            source_url: guess.to_string(),
266            content_type: None,
267            width: None,
268            height: None,
269            alt: None,
270        });
271    }
272
273    embed.kind = classify(&meta, oe, &embed);
274    embed
275}

277/// Build the embed from an oEmbed document ALONE, with no page behind it.
278///
279/// The fallback for a known provider whose page could not be fetched or carried
280/// nothing — a YouTube watch URL for an unavailable video returns a shell whose
281/// only metadata is the site boilerplate, while `youtube.com/oembed` answers for
282/// the same URL perfectly well.
283///
284/// Thinner than [`from_html`] by nature: oEmbed has no description field in the
285/// spec and no `theme-color`, so the card is title + author + thumbnail. That is
286/// still far more than the shell page yields.
287pub(crate) fn from_oembed(
288    request_url: String,
289    target: &Url,
290    oe: &OEmbed,
291    warnings: &mut Vec<String>,
292) -> LinkEmbed {
293    let mut embed = LinkEmbed::new(request_url);
294    embed.url = target.to_string();
295    embed.title = oe
296        .title
297        .as_deref()
298        .and_then(|t| clean_line(t, MAX_TITLE_CHARS));
299    embed.description = oe
300        .description
301        .as_deref()
302        .and_then(|d| clean_block(d, MAX_DESCRIPTION_CHARS));
303    embed.site_name = oe
304        .provider_name
305        .as_deref()
306        .and_then(|n| clean_line(n, MAX_LABEL_CHARS))
307        .or_else(|| {
308            target
309                .host_str()
310                .map(|h| h.trim_start_matches("www.").to_owned())
311        });
312
313    if let Some(name) = oe
314        .author_name
315        .as_deref()
316        .and_then(|n| clean_line(n, MAX_LABEL_CHARS))
317    {
318        let declared_url = oe.author_url.as_deref();
319        embed.author = Some(EmbedAuthor {
320            url: declared_url
321                .and_then(|u| absolute(target, u, warnings, "author url"))
322                .map(String::from),
323            name: name_or_handle(&name),
324        });
325    }
326    if let Some(name) = oe
327        .provider_name
328        .as_deref()
329        .and_then(|n| clean_line(n, MAX_LABEL_CHARS))
330    {
331        let declared_url = oe.provider_url.as_deref();
332        embed.provider = Some(EmbedProvider {
333            name,
334            url: declared_url
335                .and_then(|u| absolute(target, u, warnings, "provider url"))
336                .map(String::from),
337        });
338    }
339
340    // `photo` puts the full image in `url`; every other type offers a thumbnail.
341    let is_photo = oe
342        .r#type
343        .as_deref()
344        .is_some_and(|t| t.eq_ignore_ascii_case("photo"));
345    let declared_image = if is_photo {
346        oe.url.as_deref().or(oe.thumbnail_url.as_deref())
347    } else {
348        oe.thumbnail_url.as_deref()
349    };
350    if let Some(image) = declared_image.and_then(|u| absolute(target, u, warnings, "oembed image"))
351    {
352        let (width, height) = if is_photo {
353            (oe.width, oe.height)
354        } else {
355            (oe.thumbnail_width, oe.thumbnail_height)
356        };
357        embed.images.push(EmbedMedia {
358            url: String::new(),
359            source_url: image.to_string(),
360            content_type: None,
361            width: width.and_then(|d| d.as_u32()),
362            height: height.and_then(|d| d.as_u32()),
363            alt: None,
364        });
365    }
366
367    embed.kind = match oe.r#type.as_deref().map(str::to_ascii_lowercase).as_deref() {
368        Some("photo") => EmbedKind::Photo,
369        Some("video") => EmbedKind::Video,
370        Some("rich") => EmbedKind::Rich,
371        _ => EmbedKind::Link,
372    };
373    // A video with no playable file still renders as a poster with a play badge
374    // that opens externally — which is the whole point of the card.
375    if embed.kind == EmbedKind::Video && !embed.images.is_empty() {
376        embed.video = Some(EmbedVideo {
377            media: EmbedMedia {
378                url: String::new(),
379                source_url: target.to_string(),
380                content_type: Some("text/html".to_owned()),
381                width: oe.width.and_then(|d| d.as_u32()),
382                height: oe.height.and_then(|d| d.as_u32()),
383                alt: None,
384            },
385            playable: false,
386            duration_secs: oe.duration.and_then(|d| d.as_u32()),
387        });
388    }
389    embed
390}

392/// Build the embed for a URL that *is* a media file — the response was an image,
393/// a video or an audio track, so there is no HTML and no metadata, only the
394/// resource itself.
395pub(crate) fn from_media(request_url: String, final_url: &Url, content_type: &str) -> LinkEmbed {
396    let mut embed = LinkEmbed::new(request_url);
397    embed.url = final_url.to_string();
398    embed.site_name = final_url
399        .host_str()
400        .map(|h| h.trim_start_matches("www.").to_owned());
401    // The filename is the only title available, and it is better than none: a
402    // card reading "diagram-final-v3.png" says what the link is.
403    embed.title = final_url
404        .path_segments()
405        .and_then(|mut s| s.next_back())
406        .and_then(|name| clean_line(name, MAX_TITLE_CHARS));
407
408    let mime = content_type
409        .split(';')
410        .next()
411        .unwrap_or_default()
412        .trim()
413        .to_ascii_lowercase();
414    let media = EmbedMedia {
415        url: String::new(),
416        source_url: final_url.to_string(),
417        content_type: Some(mime.clone()),
418        width: None,
419        height: None,
420        alt: None,
421    };
422    if mime.starts_with("image/") {
423        embed.kind = EmbedKind::Photo;
424        embed.images = vec![media];
425    } else if mime.starts_with("video/") {
426        embed.kind = EmbedKind::Video;
427        embed.video = Some(EmbedVideo {
428            media,
429            playable: true,
430            duration_secs: None,
431        });
432    } else if mime.starts_with("audio/") {
433        embed.kind = EmbedKind::Audio;
434        embed.audio = Some(media);
435    }
436    embed
437}

439/// `og:video` first, then Twitter's player *stream* — never the player iframe.
440fn build_video(meta: &PageMetadata, base: &Url, warnings: &mut Vec<String>) -> Option<EmbedVideo> {
441    if let Some(candidate) = collect_structured(&meta.og, "video").into_iter().next() {
442        let href = candidate
443            .secure_url
444            .clone()
445            .unwrap_or(candidate.url.clone());
446        let url = absolute(base, &href, warnings, "og:video")?;
447        // The whole reason `playable` exists: `og:video` with
448        // `og:video:type=text/html` is a page, not a stream. Absent a declared
449        // type, the file extension is the only evidence there is.
450        let mime = candidate
451            .mime
452            .clone()
453            .or_else(|| guess_video_mime(url.path()));
454        let playable = mime.as_deref().is_some_and(|m| m.starts_with("video/"));
455        if !playable {
456            warnings.push(
457                "og:video is an HTML player, not a video file — it will open externally".to_owned(),
458            );
459        }
460        return Some(EmbedVideo {
461            media: EmbedMedia {
462                url: String::new(),
463                source_url: url.to_string(),
464                content_type: mime,
465                width: candidate.width,
466                height: candidate.height,
467                alt: None,
468            },
469            playable,
470            duration_secs: candidate.duration,
471        });
472    }
473
474    // `twitter:player:stream` IS the direct file (the iframe lives in
475    // `twitter:player`, which is never read).
476    let stream = first(&meta.twitter, "player:stream")?;
477    let url = absolute(base, stream, warnings, "twitter:player:stream")?;
478    let mime = first(&meta.twitter, "player:stream:content_type")
479        .map(str::to_owned)
480        .or_else(|| guess_video_mime(url.path()));
481    Some(EmbedVideo {
482        media: EmbedMedia {
483            url: String::new(),
484            source_url: url.to_string(),
485            content_type: mime.clone(),
486            width: first(&meta.twitter, "player:width").and_then(|v| v.parse().ok()),
487            height: first(&meta.twitter, "player:height").and_then(|v| v.parse().ok()),
488            alt: None,
489        },
490        playable: mime.as_deref().is_some_and(|m| m.starts_with("video/")),
491        duration_secs: None,
492    })
493}

495/// Decide the layout. oEmbed's `type` is authoritative where present; otherwise
496/// `og:type` and `twitter:card`; otherwise the shape of what was found.
497fn classify(meta: &PageMetadata, oe: Option<&OEmbed>, embed: &LinkEmbed) -> EmbedKind {
498    if let Some(kind) = oe
499        .and_then(|o| o.r#type.as_deref())
500        .map(str::to_ascii_lowercase)
501    {
502        match kind.as_str() {
503            "photo" => return EmbedKind::Photo,
504            "video" => return EmbedKind::Video,
505            "rich" => return EmbedKind::Rich,
506            "link" => {}
507            _ => {}
508        }
509    }
510    if let Some(og_type) = first(&meta.og, "type").map(str::to_ascii_lowercase) {
511        if og_type == "article" || og_type.starts_with("article:") {
512            return EmbedKind::Article;
513        }
514        if og_type == "profile" {
515            return EmbedKind::Profile;
516        }
517        if og_type.starts_with("video") {
518            return EmbedKind::Video;
519        }
520        if og_type.starts_with("music") {
521            return EmbedKind::Audio;
522        }
523    }
524    if first(&meta.twitter, "card").is_some_and(|c| c.eq_ignore_ascii_case("player")) {
525        return EmbedKind::Video;
526    }
527    if embed.video.is_some() {
528        return EmbedKind::Video;
529    }
530    if embed.audio.is_some() {
531        return EmbedKind::Audio;
532    }
533    // An image with nothing to read beside it is a photo, however the page
534    // labelled itself.
535    if !embed.images.is_empty() && embed.title.is_none() && embed.description.is_none() {
536        return EmbedKind::Photo;
537    }
538    EmbedKind::Link
539}

541// ─── Open Graph structured properties ────────────────────────────────────────
542
543/// One `og:<root>` and the sub-properties that qualified it.
544#[derive(Debug, Default, Clone)]
545struct Structured {
546    url: String,
547    secure_url: Option<String>,
548    mime: Option<String>,
549    width: Option<u32>,
550    height: Option<u32>,
551    alt: Option<String>,
552    duration: Option<u32>,
553}
554
555/// Walk the ordered property list, grouping each `og:<root>` with the
556/// `og:<root>:*` tags that follow it.
557///
558/// This is the whole reason [`PageMetadata::og`] is a `Vec`: the association
559/// between an image and its dimensions is *positional*, and any map loses it.
560fn collect_structured(og: &[(String, String)], root: &str) -> Vec<Structured> {
561    let sub_prefix = format!("{root}:");
562    let mut out: Vec<Structured> = Vec::new();
563    for (key, value) in og {
564        if key == root {
565            out.push(Structured {
566                url: value.clone(),
567                ..Default::default()
568            });
569            continue;
570        }
571        let Some(sub) = key.strip_prefix(&sub_prefix) else {
572            continue;
573        };
574        match sub {
575            // `og:<root>:url` and `og:<root>:secure_url` are the two ways to
576            // name the asset itself, and either may arrive with no bare
577            // `og:<root>` before it — plenty of pages declare only
578            // `og:video:secure_url`. Each opens a new structure when the
579            // current one has already been given that same slot; that is what
580            // separates "the https form of this image" from "the next image".
581            "url" | "secure_url" => {
582                let restates_current = out.last().is_some_and(|current| {
583                    if sub == "url" {
584                        // `og:image` then `og:image:url` with the same value is
585                        // the spec's alias spelled twice, not two images.
586                        &current.url == value
587                    } else {
588                        false
589                    }
590                });
591                if restates_current {
592                    continue;
593                }
594                let needs_new = match out.last() {
595                    None => true,
596                    Some(current) if sub == "url" => !current.url.is_empty(),
597                    Some(current) => current.secure_url.is_some(),
598                };
599                if needs_new {
600                    out.push(Structured::default());
601                }
602                // `expect`: either the vec was non-empty or a structure was just
603                // pushed.
604                let current = out.last_mut().expect("a structure to fill");
605                if sub == "url" {
606                    current.url = value.clone();
607                } else {
608                    current.secure_url = Some(value.clone());
609                }
610            }
611            _ => {
612                // A qualifier with no structure to qualify (`og:image:width`
613                // before any `og:image`) describes nothing and is dropped.
614                let Some(current) = out.last_mut() else {
615                    continue;
616                };
617                match sub {
618                    "type" => current.mime = Some(value.to_ascii_lowercase()),
619                    "width" => current.width = value.parse().ok(),
620                    "height" => current.height = value.parse().ok(),
621                    "alt" => current.alt = Some(value.clone()),
622                    "duration" => current.duration = value.parse().ok(),
623                    _ => {}
624                }
625            }
626        }
627    }
628    out.retain(|s| !s.url.is_empty() || s.secure_url.is_some());
629    out
630}

634/// A page's claim about its own address, accepted only if it is absolute.
635///
636/// Deliberately NOT resolved against the page like every other URL here, because
637/// this one becomes the card's click target. A relative `rel=canonical` joined
638/// onto the fetched URL produces something that is syntactically fine and
639/// semantically nonsense — a truncated YouTube page yielded
640/// `<link rel="canonical" href="undefined">`, which joined to
641/// `https://www.youtube.com/undefined` and opened a blank tab.
642///
643/// Requiring an absolute URL costs nothing real: a page that cannot state its
644/// own address properly falls back to the URL that actually answered, which is
645/// correct by construction. The two JS-templating leaks are named explicitly
646/// because they ARE syntactically valid relative paths and would otherwise be
647/// indistinguishable from a real one.
648fn canonical_url(raw: &str, warnings: &mut Vec<String>, what: &str) -> Option<Url> {
649    let raw = raw.trim();
650    if raw.is_empty() || raw.eq_ignore_ascii_case("undefined") || raw.eq_ignore_ascii_case("null") {
651        warnings.push(format!(
652            "{what}: `{}` is a templating leak, not an address — using the URL that answered",
653            truncate_for_log(raw)
654        ));
655        return None;
656    }
657    let Ok(url) = Url::parse(raw) else {
658        warnings.push(format!(
659            "{what}: `{}` is not an absolute URL — using the URL that answered",
660            truncate_for_log(raw)
661        ));
662        return None;
663    };
664    if !matches!(url.scheme(), "http" | "https") {
665        warnings.push(format!(
666            "{what}: dropped a `{}` URL — only http(s) is used",
667            truncate_for_log(url.scheme())
668        ));
669        return None;
670    }
671    Some(url)
672}
673
674/// Resolve `href` against `base` and accept it only if it is `http(s)`.
675///
676/// The gate that keeps `javascript:`, `data:` and `file:` out of every URL field
677/// a client might make tappable. A rejection is a warning, not an error: one bad
678/// image URL should not cost the page its title.
679fn absolute(base: &Url, href: &str, warnings: &mut Vec<String>, what: &str) -> Option<Url> {
680    let href = href.trim();
681    if href.is_empty() {
682        return None;
683    }
684    let Ok(url) = base.join(href) else {
685        warnings.push(format!("{what}: `{}` is not a URL", truncate_for_log(href)));
686        return None;
687    };
688    if !matches!(url.scheme(), "http" | "https") {
689        warnings.push(format!(
690            "{what}: dropped `{}` — only http(s) URLs are used",
691            truncate_for_log(url.scheme())
692        ));
693        return None;
694    }
695    Some(url)
696}

699fn pick(candidates: &[Option<&str>], clean: impl Fn(&str) -> Option<String>) -> Option<String> {
700    candidates.iter().flatten().copied().find_map(clean)
701}

711/// A single-line string: every run of whitespace (newlines included) becomes one
712/// space, control characters are dropped, and the result is capped.
713pub(crate) fn clean_line(raw: &str, max_chars: usize) -> Option<String> {
714    let collapsed: String = {
715        let mut out = String::with_capacity(raw.len());
716        let mut pending_space = false;
717        for ch in raw.chars().filter(|c| !is_forbidden(*c)) {
718            if ch.is_whitespace() {
719                pending_space = !out.is_empty();
720                continue;
721            }
722            if pending_space {
723                out.push(' ');
724                pending_space = false;
725            }
726            out.push(ch);
727        }
728        out
729    };
730    finish(collapsed, max_chars)
731}
732
733/// A multi-line string: paragraph breaks survive (a publisher's line structure
734/// is content), but nothing else does. Runs of three or more newlines collapse to
735/// two, so a page cannot pad a card to arbitrary height.
736pub(crate) fn clean_block(raw: &str, max_chars: usize) -> Option<String> {
737    let mut out = String::with_capacity(raw.len());
738    let mut newlines = 0usize;
739    let mut pending_space = false;
740    for ch in raw.chars().filter(|c| !is_forbidden(*c)) {
741        if ch == '\n' || ch == '\r' {
742            // \r\n counts once: a bare \r after a \n adds nothing.
743            if ch == '\r' {
744                continue;
745            }
746            newlines += 1;
747            pending_space = false;
748            continue;
749        }
750        if ch.is_whitespace() {
751            pending_space = !out.is_empty();
752            continue;
753        }
754        if newlines > 0 {
755            out.push_str(if newlines == 1 { "\n" } else { "\n\n" });
756            newlines = 0;
757        } else if pending_space {
758            out.push(' ');
759        }
760        pending_space = false;
761        out.push(ch);
762    }
763    finish(out, max_chars)
764}

766/// Characters no embed field may contain.
767///
768/// C0/C1 controls (bar the newlines `clean_block` handles itself), the
769/// bidirectional-override set, and the zero-width joiners/spaces. The bidi
770/// characters are the pointed ones: `U+202E` in a title reverses the text that
771/// follows it, which is how "…evil.exe" is made to read "…exe.live".
772fn is_forbidden(ch: char) -> bool {
773    matches!(ch,
774        '\u{0}'..='\u{8}'
775        | '\u{b}'..='\u{c}'
776        | '\u{e}'..='\u{1f}'
777        | '\u{7f}'..='\u{9f}'
778        | '\u{200b}'..='\u{200f}'
779        | '\u{202a}'..='\u{202e}'
780        | '\u{2060}'..='\u{2064}'
781        | '\u{2066}'..='\u{2069}'
782        | '\u{feff}')
783}

785/// Trim, drop if empty, and cap at `max_chars` on a char boundary with an
786/// ellipsis. The cap counts `char`s rather than bytes for the same reason the
787/// message limit does: a byte cap would cut a line of emoji to a quarter.
788fn finish(value: String, max_chars: usize) -> Option<String> {
789    let trimmed = value.trim();
790    if trimmed.is_empty() {
791        return None;
792    }
793    if trimmed.chars().count() <= max_chars {
794        return Some(trimmed.to_owned());
795    }
796    let mut out: String = trimmed.chars().take(max_chars.saturating_sub(1)).collect();
797    out.push('…');
798    Some(out)
799}

801/// `#abc` / `#aabbcc` → `#aabbcc`. Anything else is dropped with a warning: a
802/// client is promised a value it can parse without a CSS engine.
803fn normalise_hex_colour(raw: &str, warnings: &mut Vec<String>) -> Option<String> {
804    let hex = raw.trim().strip_prefix('#')?;
805    if !hex.chars().all(|c| c.is_ascii_hexdigit()) {
806        warnings.push(format!(
807            "theme-color `{}` is not a hex colour",
808            truncate_for_log(raw)
809        ));
810        return None;
811    }
812    match hex.len() {
813        3 => {
814            let mut out = String::with_capacity(7);
815            out.push('#');
816            for ch in hex.chars() {
817                out.push(ch);
818                out.push(ch);
819            }
820            Some(out.to_ascii_lowercase())
821        }
822        6 => Some(format!("#{}", hex.to_ascii_lowercase())),
823        // 8 digits is `#rrggbbaa`; the alpha is dropped rather than passed to a
824        // client that would read it as part of the RGB.
825        8 => Some(format!("#{}", hex[..6].to_ascii_lowercase())),
826        _ => {
827            warnings.push(format!(
828                "theme-color `{}` is not 3, 6 or 8 hex digits",
829                truncate_for_log(raw)
830            ));
831            None
832        }
833    }
834}

836/// `article:published_time` → unix ms. RFC 3339 first (what the spec says),
837/// then a bare date, which is what a good number of sites actually emit.
838fn parse_timestamp(raw: &str, warnings: &mut Vec<String>) -> Option<i64> {
839    let raw = raw.trim();
840    if let Ok(dt) = chrono::DateTime::parse_from_rfc3339(raw) {
841        return Some(dt.timestamp_millis());
842    }
843    if let Ok(date) = chrono::NaiveDate::parse_from_str(raw, "%Y-%m-%d") {
844        return date
845            .and_hms_opt(0, 0, 0)
846            .and_then(|dt| chrono::Utc.from_local_datetime(&dt).single())
847            .map(|dt| dt.timestamp_millis());
848    }
849    warnings.push(format!(
850        "published time `{}` is not a recognised timestamp",
851        truncate_for_log(raw)
852    ));
853    None
854}

856/// A byline that is really a URL becomes its last path segment, so a card shows
857/// "jane" rather than "https://example.com/authors/jane".
858fn name_or_handle(name: &str) -> String {
859    if !(name.starts_with("http://") || name.starts_with("https://")) {
860        return name.to_owned();
861    }
862    Url::parse(name)
863        .ok()
864        .and_then(|u| {
865            u.path_segments()
866                .and_then(|mut s| s.rfind(|p: &&str| !p.is_empty()))
867                .map(str::to_owned)
868        })
869        .unwrap_or_else(|| name.to_owned())
870}

891/// Every metadata key seen, flattened for the debugger. Ordered by vocabulary so
892/// the report reads as three blocks rather than an interleaving.
893pub(crate) fn raw_fields(meta: &PageMetadata, oembed: Option<&OEmbed>) -> Vec<EmbedRawField> {
894    let mut out = Vec::new();
895    for (key, value) in &meta.og {
896        out.push(EmbedRawField {
897            source: EmbedSource::OpenGraph,
898            // The vertical extensions were stored with their own prefix
899            // intact; everything else lost the `og:` on the way in.
900            key: if key.starts_with("article:") || key.starts_with("music:") {
901                key.clone()
902            } else {
903                format!("og:{key}")
904            },
905            value: value.clone(),
906        });
907    }
908    for (key, value) in &meta.twitter {
909        out.push(EmbedRawField {
910            source: EmbedSource::TwitterCard,
911            key: format!("twitter:{key}"),
912            value: value.clone(),
913        });
914    }
915    let mut html = Vec::new();
916    if let Some(v) = &meta.html_title {
917        html.push(("<title>", v.clone()));
918    }
919    if let Some(v) = &meta.html_description {
920        html.push(("meta[name=description]", v.clone()));
921    }
922    if let Some(v) = &meta.html_author {
923        html.push(("meta[name=author]", v.clone()));
924    }
925    if let Some(v) = &meta.theme_color {
926        html.push(("meta[name=theme-color]", v.clone()));
927    }
928    if let Some(v) = &meta.canonical {
929        html.push(("link[rel=canonical]", v.clone()));
930    }
931    if let Some(v) = &meta.lang {
932        html.push(("html[lang]", v.clone()));
933    }
934    for icon in &meta.icons {
935        html.push(("link[rel=icon]", icon.href.clone()));
936    }
937    if let Some(v) = &meta.oembed_xml {
938        html.push(("link[type=text/xml+oembed]", v.clone()));
939    }
940    for (key, value) in html {
941        out.push(EmbedRawField {
942            source: EmbedSource::Html,
943            key: key.to_owned(),
944            value,
945        });
946    }
947    if let Some(oe) = oembed {
948        let mut push = |key: &str, value: Option<String>| {
949            if let Some(value) = value {
950                out.push(EmbedRawField {
951                    source: EmbedSource::OEmbed,
952                    key: key.to_owned(),
953                    value,
954                });
955            }
956        };
957        push("type", oe.r#type.clone());
958        push("version", oe.version.clone());
959        push("title", oe.title.clone());
960        push("description", oe.description.clone());
961        push("author_name", oe.author_name.clone());
962        push("author_url", oe.author_url.clone());
963        push("provider_name", oe.provider_name.clone());
964        push("provider_url", oe.provider_url.clone());
965        push("thumbnail_url", oe.thumbnail_url.clone());
966        push("url", oe.url.clone());
967        push(
968            "width",
969            oe.width.and_then(|d| d.as_u32()).map(|v| v.to_string()),
970        );
971        push(
972            "height",
973            oe.height.and_then(|d| d.as_u32()).map(|v| v.to_string()),
974        );
975        push(
976            "duration",
977            oe.duration.and_then(|d| d.as_u32()).map(|v| v.to_string()),
978        );
979    }
980    out
981}
