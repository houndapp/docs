1//! Known oEmbed providers, looked up by host.
2//!
3//! ## Why a registry, when discovery already exists
4//!
5//! [`super::parse`] finds `<link rel="alternate" type="application/json+oembed">`
6//! in the page, which works right up until it doesn't:
7//!
8//! * the page a bot is served may not carry the link. A YouTube watch page for
9//!   an unavailable or region-blocked video returns a shell whose only metadata
10//!   is the site's boilerplate — which unfurled to a card reading "- YouTube"
11//!   over "Enjoy the videos and music you love…", and is what prompted this;
12//! * the page fetch may fail outright, while the provider's own endpoint answers
13//!   perfectly well;
14//! * some providers publish an endpoint and simply never advertise it.
15//!
16//! For a host we recognise, the endpoint is a fact we already know, so there is
17//! no reason to ask the page for permission to use it.
18//!
19//! ## What it buys, concretely
20//!
21//! For YouTube, oEmbed is the ONLY source of `provider_name` ("YouTube", not the
22//! host-derived "youtube.com") and of the channel — `author_name` and
23//! `author_url`. Neither appears in the page's Open Graph tags at all. It is not
24//! the source of the thumbnail: oEmbed offers `hqdefault` at 480×360 where
25//! `og:image` offers `maxresdefault` at 1280×720, which is why the precedence in
26//! [`super::build`] keeps Open Graph ahead of oEmbed for images.
27//!
28//! ## Adding one
29//!
30//! Verify the endpoint answers before adding it — every entry here was checked
31//! against a live URL. Match on the REGISTRABLE host and let the suffix rule
32//! cover subdomains, so `m.youtube.com` and `music.youtube.com` arrive without
33//! separate entries.
34
35use url::Url;
36
37/// A provider whose oEmbed endpoint is known without asking the page.
38#[derive(Debug, Clone, Copy)]
39pub(crate) struct Provider {
40    /// Human name, for logs and the debugger. The card's provider label comes
41    /// from the endpoint's own `provider_name`, not from here.
42    pub(crate) name: &'static str,
43    /// Hosts this provider owns. Matched exactly OR as a dot-suffix, so one
44    /// entry covers every subdomain.
45    hosts: &'static [&'static str],
46    /// The endpoint, with any fixed query parameters already on it. The target
47    /// URL is appended as `url=…` at lookup time.
48    endpoint: &'static str,
49}
50
51impl Provider {
52    /// The endpoint to GET for `target`.
53    ///
54    /// Built with `query_pairs_mut` rather than string formatting so the target
55    /// is percent-encoded properly — these URLs routinely carry `?`, `&` and
56    /// `#`, and splicing one in raw truncates it at the first separator.
57    pub(crate) fn oembed_endpoint(&self, target: &Url) -> Option<Url> {
58        let mut endpoint = Url::parse(self.endpoint).ok()?;
59        endpoint
60            .query_pairs_mut()
61            .append_pair("url", target.as_str());
62        Some(endpoint)
63    }
64}

115/// The provider owning `url`, if this build knows one.
116pub(crate) fn lookup(url: &Url) -> Option<&'static Provider> {
117    let host = url.host_str()?.trim_end_matches('.').to_ascii_lowercase();
118    PROVIDERS
119        .iter()
120        .find(|provider| provider.hosts.iter().any(|known| owns(known, &host)))
121}
122
123/// Whether `host` is `known` or a subdomain of it.
124///
125/// The dot in the suffix check is load-bearing: without it `notyoutube.com`
126/// matches `youtube.com`, and a lookalike domain would be handed a request
127/// carrying its target URL.
128fn owns(known: &str, host: &str) -> bool {
129    host == known || host.ends_with(&format!(".{known}"))
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    fn url(text: &str) -> Url {
137        Url::parse(text).expect("test URL parses")
138    }
139
140    #[test]
141    fn every_form_of_a_provider_url_resolves_to_it() {
142        for text in [
143            "https://www.youtube.com/watch?v=abc",
144            "https://youtu.be/abc",
145            "https://m.youtube.com/watch?v=abc",
146            "https://music.youtube.com/watch?v=abc",
147            "https://www.youtube-nocookie.com/embed/abc",
148            "https://YouTube.com/watch?v=abc",
149        ] {
150            assert_eq!(
151                lookup(&url(text)).map(|p| p.name),
152                Some("YouTube"),
153                "{text}"
154            );
155        }
156    }
157
158    /// The lookalike case the dot in the suffix rule exists for. Getting this
159    /// wrong hands an attacker-registered domain a request naming the URL a user
160    /// posted.
161    #[test]
162    fn a_lookalike_domain_is_not_the_provider() {
163        for text in [
164            "https://notyoutube.com/watch?v=abc",
165            "https://youtube.com.evil.example/watch?v=abc",
166            "https://myvimeo.com/1",
167            "https://example.com/",
168        ] {
169            assert!(lookup(&url(text)).is_none(), "{text} must not match");
170        }
171    }
172
173    /// A target URL's own query string must survive into the endpoint intact —
174    /// splicing it in raw truncates it at the first `&`, which for YouTube means
175    /// losing the video id entirely.
176    #[test]
177    fn the_target_url_is_percent_encoded_into_the_endpoint() {
178        let target = url("https://www.youtube.com/watch?v=abc&t=42s");
179        let provider = lookup(&target).expect("youtube is known");
180        let endpoint = provider.oembed_endpoint(&target).expect("endpoint builds");
181
182        let pairs: Vec<(String, String)> = endpoint
183            .query_pairs()
184            .map(|(k, v)| (k.into_owned(), v.into_owned()))
185            .collect();
186        assert!(
187            pairs.contains(&("url".to_owned(), target.to_string())),
188            "the target must round-trip out of the endpoint: {pairs:?}"
189        );
190        // The endpoint's own fixed parameters survive alongside it.
191        assert!(pairs.contains(&("format".to_owned(), "json".to_owned())));
192    }
193
194    #[test]
195    fn every_registered_endpoint_is_a_usable_https_url() {
196        let target = url("https://example.com/x");
197        for provider in PROVIDERS {
198            let endpoint =
199                Url::parse(provider.endpoint).unwrap_or_else(|e| panic!("{}: {e}", provider.name));
200            assert_eq!(endpoint.scheme(), "https", "{}", provider.name);
201            assert!(
202                provider.oembed_endpoint(&target).is_some(),
203                "{}",
204                provider.name
205            );
206            assert!(!provider.hosts.is_empty(), "{}", provider.name);
207        }
208    }
