1//! Persistence for link embeds: the per-URL unfurl cache and the message→embed
2//! join.
3//!
4//! ## The cache is keyed by URL, not by message
5//!
6//! A link posted in fifty spaces is one row and one outbound fetch. That is the
7//! whole point of the split: without it, unfurl traffic would scale with how
8//! often people quote each other, and a popular link would be a small denial of
9//! service aimed at whoever published it.
10//!
11//! ## Failures are cached too
12//!
13//! On a shorter clock ([`TTL_ERROR`]) than a success, but cached all the same. A
14//! link to a host that times out must not cost eight seconds every time somebody
15//! mentions it, and a 404 is a stable fact for at least half an hour.
16//!
17//! ## Refresh keeps the row id
18//!
19//! A stale row is updated in place rather than replaced, because the id is in
20//! every proxy URL the embed hands out. Minting a new one would break the media
21//! links of every message already pointing at it.
22

31/// Successful unfurl: a day. Pages change, but not so fast that a preview being
32/// a few hours behind is a defect.
33const TTL_OK: i64 = 24 * 60 * 60 * 1000;
34/// Fetched fine, nothing worth showing. Re-checked sooner than a success — a
35/// site adding Open Graph tags is a common enough event.
36const TTL_EMPTY: i64 = 6 * 60 * 60 * 1000;
37/// Refused, unreachable, or an error status.
38const TTL_ERROR: i64 = 30 * 60 * 1000;
39
40/// How a cached unfurl turned out. Stored as a string so a new outcome is a
41/// value, not a schema migration.
42#[derive(Debug, Clone, Copy, PartialEq, Eq)]
43pub(crate) enum EmbedStatus {
44    Ok,
45    Empty,
46    Error,
47}
48
49impl EmbedStatus {
50    fn as_str(self) -> &'static str {
51        match self {
52            Self::Ok => "ok",
53            Self::Empty => "empty",
54            Self::Error => "error",
55        }
56    }
57
58    fn ttl_ms(self) -> i64 {
59        match self {
60            Self::Ok => TTL_OK,
61            Self::Empty => TTL_EMPTY,
62            Self::Error => TTL_ERROR,
63        }
64    }
65}

67/// A cache lookup's answer.
68pub(crate) struct CachedEmbed {
69    pub id: Uuid,
70    /// `None` for a cached `empty` or `error` — the fact that this URL unfurls
71    /// to nothing is itself the cached value.
72    pub embed: Option<LinkEmbed>,
73}
74
75/// The lookup key: SHA-256 of the URL, hex.
76///
77/// Hashed rather than indexed directly because a URL is unbounded (query strings
78/// run to kilobytes) while an index key should not be. Deliberately *not*
79/// normalised beyond what the URL parser already did: two URLs differing by a
80/// tracking parameter genuinely are two URLs, and folding them together would
81/// serve one page's preview under another's link.
82pub(crate) fn url_hash(url: &str) -> String {
83    use sha2::{Digest, Sha256};
84    hex::encode(Sha256::digest(url.as_bytes()))
85}
86
87impl Api {
88    /// A fresh cache entry for `url`, or `None` to unfurl.
89    pub(crate) async fn find_fresh_embed(&self, url: &str) -> DbResult<Option<CachedEmbed>> {
90        let now = chrono::Utc::now().timestamp_millis();
91        let Some(row) = link_embed::Entity::find()
92            .filter(link_embed::Column::UrlHash.eq(url_hash(url)))
93            .one(&self.db)
94            .await?
95        else {
96            return Ok(None);
97        };
98        if row.expires_at <= now {
99            return Ok(None);
100        }
101        Ok(Some(CachedEmbed {
102            id: row.id,
103            embed: decode_payload(row.payload, row.id),
104        }))
105    }
106
107    /// The embed id for `url`, minting the row if this URL has never been seen.
108    ///
109    /// Split from the write below because the id is needed *before* the payload
110    /// can be finished: the proxy URLs inside an embed name the row that carries
111    /// it, so an unfurl has to know its own id to stamp them.
112    pub(crate) async fn reserve_embed_id(&self, url: &str) -> DbResult<Uuid> {
113        let hash = url_hash(url);
114        if let Some(existing) = link_embed::Entity::find()
115            .filter(link_embed::Column::UrlHash.eq(hash.clone()))
116            .one(&self.db)
117            .await?
118        {
119            return Ok(existing.id);
120        }
121        let now = chrono::Utc::now().timestamp_millis();
122        let id = Uuid::now_v7();
123        // A concurrent unfurl of the same URL can win this insert; the unique
124        // index on the hash is what makes that a *conflict* rather than two rows
125        // with two sets of proxy URLs. The loser re-reads and adopts the winner's
126        // id, which is the outcome we want either way.
127        match (link_embed::ActiveModel {
128            id: Set(id),
129            url_hash: Set(hash.clone()),
130            url: Set(url.to_owned()),
131            status: Set(EmbedStatus::Error.as_str().to_owned()),
132            payload: Set(None),
133            error: Set(Some("unfurl in progress".to_owned())),
134            fetched_at: Set(now),
135            // Expired on arrival: if the unfurl dies before writing its result,
136            // this placeholder must not shadow the URL for the error TTL.
137            expires_at: Set(now),
138        })
139        .insert(&self.db)
140        .await
141        {
142            Ok(model) => Ok(model.id),
143            Err(_) => link_embed::Entity::find()
144                .filter(link_embed::Column::UrlHash.eq(hash))
145                .one(&self.db)
146                .await?
147                .map(|m| m.id)
148                .ok_or_else(|| DbErr::Custom("embed row vanished after an insert conflict".into())),
149        }
150    }
151
152    /// Record an unfurl's outcome against an id from [`Api::reserve_embed_id`].
153    pub(crate) async fn write_embed(
154        &self,
155        id: Uuid,
156        url: &str,
157        status: EmbedStatus,
158        embed: Option<&LinkEmbed>,
159        error: Option<String>,
160    ) -> DbResult<()> {
161        let now = chrono::Utc::now().timestamp_millis();
162        let payload = embed
163            .map(serde_json::to_value)
164            .transpose()
165            .map_err(|e| DbErr::Custom(format!("embed payload is not serialisable: {e}")))?;
166        link_embed::ActiveModel {
167            id: Set(id),
168            url_hash: Set(url_hash(url)),
169            url: Set(url.to_owned()),
170            status: Set(status.as_str().to_owned()),
171            payload: Set(payload),
172            error: Set(error),
173            fetched_at: Set(now),
174            expires_at: Set(now + status.ttl_ms()),
175        }
176        .update(&self.db)
177        .await?;
178        // A refreshed row keeps its id, so its proxy URLs are unchanged — which
179        // means the media cache is holding the PREVIOUS fetch's bytes under
180        // exactly the keys the new payload is about to hand out.
181        self.embed_media_cache.invalidate(id);
182        Ok(())
183    }
184
185    /// The stored embed for `embed_id`, for the media proxy's URL lookup.
186    pub(crate) async fn load_embed_payload(&self, embed_id: Uuid) -> Option<LinkEmbed> {
187        let row = link_embed::Entity::find_by_id(embed_id)
188            .one(&self.db)
189            .await
190            .ok()??;
191        decode_payload(row.payload, row.id)
192    }

194    /// Replace a message's embed list wholesale.
195    ///
196    /// Wholesale because an edit can remove a link, and a merge could never
197    /// express that. Ordinals are re-assigned from zero so they always describe
198    /// the current body.
199    pub(crate) async fn set_message_embeds(
200        &self,
201        message_id: Uuid,
202        embed_ids: &[Uuid],
203    ) -> DbResult<()> {
204        community_space_message_embed::Entity::delete_many()
205            .filter(community_space_message_embed::Column::MessageId.eq(message_id))
206            .exec(&self.db)
207            .await?;
208        for (ordinal, embed_id) in embed_ids.iter().enumerate() {
209            community_space_message_embed::ActiveModel {
210                message_id: Set(message_id),
211                ordinal: Set(ordinal as i32),
212                embed_id: Set(*embed_id),
213            }
214            .insert(&self.db)
215            .await?;
216        }
217        Ok(())
218    }

220    /// Embeds for a whole page of messages, keyed by message id.
221    ///
222    /// One query for the page, for the same reason attachments are: a history
223    /// load is on the path of every space open and every post-reconnect resync,
224    /// and per-row lookups would make it N round trips.
225    pub(crate) async fn embeds_for_messages(
226        &self,
227        message_ids: Vec<Uuid>,
228    ) -> DbResult<HashMap<Uuid, Vec<LinkEmbed>>> {
229        if message_ids.is_empty() {
230            return Ok(HashMap::new());
231        }
232        let mut by_message: HashMap<Uuid, Vec<LinkEmbed>> = HashMap::new();
233        for (link, embed) in community_space_message_embed::Entity::find()
234            .filter(community_space_message_embed::Column::MessageId.is_in(message_ids))
235            .order_by_asc(community_space_message_embed::Column::Ordinal)
236            .find_also_related(link_embed::Entity)
237            .all(&self.db)
238            .await?
239        {
240            // A join row whose embed is still pending (or unfurled to nothing)
241            // contributes no card. It stays in the table so a later refresh has
242            // somewhere to land.
243            let Some(embed) = embed.and_then(|row| decode_payload(row.payload, row.id)) else {
244                continue;
245            };
246            by_message.entry(link.message_id).or_default().push(embed);
247        }
248        Ok(by_message)
249    }
250
251    /// Embeds for one message, in ordinal order.
252    pub(crate) async fn embeds_for_message(&self, message_id: Uuid) -> Vec<LinkEmbed> {
253        self.embeds_for_messages(vec![message_id])
254            .await
255            .unwrap_or_default()
256            .remove(&message_id)
257            .unwrap_or_default()
258    }
259}

261/// Decode a stored payload, treating a shape this build no longer understands as
262/// "no embed" rather than an error.
263///
264/// These rows outlive deployments: a payload written by a newer build, or by one
265/// whose field set has since changed incompatibly, must degrade to a missing card
266/// and a log line — never to a failed history load for the whole space.
267fn decode_payload(payload: Option<serde_json::Value>, id: Uuid) -> Option<LinkEmbed> {
268    let payload = payload?;
269    match serde_json::from_value::<LinkEmbed>(payload) {
270        Ok(embed) => Some(embed),
271        Err(e) => {
272            tracing::warn!(embed_id = %id, error = %e, "discarding an undecodable embed payload");
273            None
274        }
275    }
