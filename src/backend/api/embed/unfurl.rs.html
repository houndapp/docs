1//! The unfurl pipeline: message text in, stored embeds and a realtime push out.
2//!
3//! ## Why this runs after the message is stored
4//!
5//! Unfurling is a network round-trip against a stranger's server. Doing it on
6//! the POST would put an eight-second worst case on every send containing a
7//! link, and would let a slow host pin a request worker. So the handler stores
8//! the message, answers, and spawns [`Api::spawn_unfurl`]; the previews arrive
9//! moments later as [`ServerMessage::MessageEmbedsUpdated`].
10//!
11//! The cost of that choice is a visible two-step in the UI (message, then card),
12//! which is the same behaviour every other chat client has, and is why the
13//! realtime frame is a *replacement list* rather than an append — a client that
14//! misses it and re-reads history still converges.
15//!
16//! ## Which links get unfurled
17//!
18//! The first [`MAX_MESSAGE_EMBEDS`] distinct `http(s)` URLs in the body, in the
19//! order they appear. Distinct by exact URL: the same link twice is one card.
20//!
21//! ## Failure is not an error
22//!
23//! Nothing here can fail the message. Every outcome — refused by the guard,
24//! timed out, a 404, a page with no metadata — is recorded in the cache so it is
25//! not retried on every mention, and produces no card.
26

39/// What this service will accept as a previewable page.
40const PAGE_ACCEPT: &str = "text/html,application/xhtml+xml;q=0.9,*/*;q=0.5";

42/// Extract the URLs worth unfurling from a message body, in document order,
43/// de-duplicated.
44///
45/// `linkify` rather than a regular expression: matching URLs in prose is a
46/// famously bad fit for one, and the crate already handles the cases that bite —
47/// a trailing full stop, a URL inside brackets, an email address that is not a
48/// link.
49pub(crate) fn extract_urls(contents: &str) -> Vec<Url> {
50    let mut finder = linkify::LinkFinder::new();
51    finder.kinds(&[linkify::LinkKind::Url]);
52    let mut seen: Vec<Url> = Vec::new();
53    for found in finder.links(contents) {
54        let Ok(url) = Url::parse(found.as_str()) else {
55            continue;
56        };
57        if !matches!(url.scheme(), "http" | "https") {
58            continue;
59        }
60        if seen.iter().any(|existing| existing == &url) {
61            continue;
62        }
63        seen.push(url);
64        if seen.len() == MAX_MESSAGE_EMBEDS {
65            break;
66        }
67    }
68    seen
69}

71impl Api {
72    /// Unfurl a message's links in the background and push the result.
73    ///
74    /// Detached on purpose: the caller has already answered the client. The task
75    /// holds only a cloned [`Api`] handle (an `Arc`), so it cannot outlive the
76    /// process or hold a request open.
77    ///
78    /// A message with no links still runs, and still emits — an *edit* that
79    /// removed the last link has to be able to take its card away.
80    pub(crate) fn spawn_unfurl(&self, message_id: Uuid, space_id: Uuid, contents: String) {
81        let api = self.clone();
82        tokio::spawn(async move {
83            let urls = extract_urls(&contents);
84            let had_embeds = !api.embeds_for_message(message_id).await.is_empty();
85            if urls.is_empty() && !had_embeds {
86                return;
87            }
88
89            let mut ids = Vec::with_capacity(urls.len());
90            let mut embeds = Vec::with_capacity(urls.len());
91            for url in urls {
92                match api.unfurl_cached(&url).await {
93                    Some((id, Some(embed))) => {
94                        ids.push(id);
95                        embeds.push(embed);
96                    }
97                    // Unfurled to nothing: the join row is still recorded, so a
98                    // later refresh of that URL lands on this message without
99                    // re-parsing the body.
100                    Some((id, None)) => ids.push(id),
101                    None => {}
102                }
103            }
104
105            if let Err(e) = api.set_message_embeds(message_id, &ids).await {
106                tracing::warn!(%message_id, error = %e, "could not store message embeds");
107                return;
108            }
109            metrics::counter!("backend_message_embeds_total").increment(embeds.len() as u64);
110            api.emit_message_embeds(message_id, space_id, embeds).await;
111        });
112    }

114    /// The cache-aware unfurl of one URL: `(embed id, embed)`.
115    ///
116    /// `None` only when the database itself would not cooperate — every
117    /// *fetch* outcome is a successful cache write.
118    async fn unfurl_cached(&self, url: &Url) -> Option<(Uuid, Option<LinkEmbed>)> {
119        let key = url.as_str();
120        match self.find_fresh_embed(key).await {
121            Ok(Some(cached)) => {
122                metrics::counter!("backend_embed_cache_hits_total").increment(1);
123                return Some((cached.id, cached.embed));
124            }
125            Ok(None) => {}
126            Err(e) => {
127                tracing::warn!(%url, error = %e, "embed cache lookup failed");
128                return None;
129            }
130        }
131
132        let id = match self.reserve_embed_id(key).await {
133            Ok(id) => id,
134            Err(e) => {
135                tracing::warn!(%url, error = %e, "could not reserve an embed row");
136                return None;
137            }
138        };
139
140        metrics::counter!("backend_embed_cache_misses_total").increment(1);
141        let outcome = unfurl(url).await;
142        let (status, embed, error) = match outcome {
143            Outcome::Embed(mut embed) => {
144                assign_proxy_urls(&mut embed, id);
145                (EmbedStatus::Ok, Some(*embed), None)
146            }
147            Outcome::Empty => (EmbedStatus::Empty, None, None),
148            Outcome::Failed(message) => {
149                metrics::counter!("backend_embed_unfurl_failures_total").increment(1);
150                (EmbedStatus::Error, None, Some(message))
151            }
152        };
153        if let Err(e) = self
154            .write_embed(id, key, status, embed.as_ref(), error)
155            .await
156        {
157            tracing::warn!(%url, error = %e, "could not store an unfurled embed");
158        }
159        Some((id, embed))
160    }

162    /// Unfurl one URL with the cache **bypassed**, collecting the full trace.
163    /// Backs the embed debugger.
164    pub(crate) async fn debug_unfurl(&self, raw: &str) -> EmbedDebugReport {
165        let started = std::time::Instant::now();
166        let mut report = EmbedDebugReport {
167            request_url: raw.to_owned(),
168            embed: None,
169            steps: Vec::new(),
170            raw: Vec::new(),
171            oembed_endpoint: None,
172            oembed_html: None,
173            warnings: Vec::new(),
174            error: None,
175            cached: false,
176            elapsed_ms: 0,
177        };
178
179        let Ok(url) = Url::parse(raw.trim()) else {
180            report.error = Some("that is not a URL".to_owned());
181            report.elapsed_ms = started.elapsed().as_millis() as u32;
182            return report;
183        };
184        if !matches!(url.scheme(), "http" | "https") {
185            report.error = Some(format!(
186                "`{}` links are never previewed — only http and https",
187                url.scheme()
188            ));
189            report.elapsed_ms = started.elapsed().as_millis() as u32;
190            return report;
191        }
192
193        // Reported, not honoured: the debugger always re-fetches, so an operator
194        // sees what the site says *now* rather than what it said yesterday. What
195        // the cache holds is still worth showing — "a message would have hit the
196        // cache" explains a stale-looking card.
197        report.cached = matches!(self.find_fresh_embed(url.as_str()).await, Ok(Some(_)));
198
199        let trace = unfurl_traced(&url).await;
200        report.steps = trace.steps;
201        report.raw = trace.raw;
202        report.oembed_endpoint = trace.oembed_endpoint;
203        report.oembed_html = trace.oembed_html;
204        report.warnings = trace.warnings;
205        match trace.outcome {
206            Outcome::Embed(mut embed) => {
207                // The preview is only honest if its images actually load, and a
208                // proxy URL names the ROW that carries it — so the row has to
209                // exist. Reserving the id (idempotent: it adopts an existing row
210                // for this URL) and writing what was just fetched is what makes
211                // the debugger's card byte-identical to the one a channel
212                // renders, which is the tool's entire promise.
213                //
214                // Stamping only when a fresh row happened to exist is what this
215                // replaced, and it was wrong in the case the tool exists for: a
216                // URL nobody had posted yet has no row, so every media URL came
217                // back as the empty string it was built with and every image in
218                // the preview silently failed to load.
219                //
220                // Writing also refreshes the cache with what the site says NOW,
221                // so debugging a stale-looking card is what fixes it. Only a
222                // success is written: a failure the operator is actively
223                // investigating should not be pinned into the cache underneath
224                // them.
225                match self.reserve_embed_id(url.as_str()).await {
226                    Ok(id) => {
227                        assign_proxy_urls(&mut embed, id);
228                        if let Err(e) = self
229                            .write_embed(id, url.as_str(), EmbedStatus::Ok, Some(&embed), None)
230                            .await
231                        {
232                            tracing::warn!(%url, error = %e, "debug unfurl: could not store the embed");
233                            report.warnings.push(
234                                "the preview could not be stored, so its images may not load"
235                                    .to_owned(),
236                            );
237                        }
238                    }
239                    Err(e) => {
240                        tracing::warn!(%url, error = %e, "debug unfurl: could not reserve a row");
241                        report.warnings.push(
242                            "no row could be reserved for this preview, so its images will not load"
243                                .to_owned(),
244                        );
245                    }
246                }
247                report.embed = Some(*embed);
248            }
249            Outcome::Empty => {
250                report.error = Some(
251                    "the page was fetched but carries no title, description or image".to_owned(),
252                );
253            }
254            Outcome::Failed(message) => report.error = Some(message),
255        }
256        report.elapsed_ms = started.elapsed().as_millis() as u32;
257        report
258    }

260    /// Push a message's embed list to its space.
261    pub(crate) async fn emit_message_embeds(
262        &self,
263        message_id: Uuid,
264        space_id: Uuid,
265        embeds: Vec<LinkEmbed>,
266    ) {
267        self.realtime.route(&[hound_realtime::Outgoing::to(
268            hound_realtime::Target::Room(hound_realtime::Room::Space(space_id)),
269            hound_realtime::ServerMessage::MessageEmbedsUpdated {
270                message_id,
271                space_id,
272                embeds,
273            },
274        )]);
275    }

278/// What one URL unfurled to.
279enum Outcome {
280    /// Boxed: a `LinkEmbed` is ~700 bytes and the other two variants are a
281    /// `String` and nothing, so an unboxed enum would pay the largest variant's
282    /// size on every unfurl that failed or found nothing — the common case.
283    Embed(Box<LinkEmbed>),
284    /// Fetched successfully; there was nothing worth a card.
285    Empty,
286    /// Refused, unreachable, or an error status. The string is shown in the
287    /// debugger and stored on the cache row.
288    Failed(String),
289}
290
291/// Everything a traced unfurl saw. The message path throws all but `outcome`
292/// away; the debugger keeps the lot.
293struct Trace {
294    outcome: Outcome,
295    steps: Vec<hound_types::EmbedFetchStep>,
296    raw: Vec<hound_types::EmbedRawField>,
297    oembed_endpoint: Option<String>,
298    oembed_html: Option<String>,
299    warnings: Vec<String>,
300}
301
302async fn unfurl(url: &Url) -> Outcome {
303    unfurl_traced(url).await.outcome
304}

306/// Fetch, parse, discover oEmbed, and build — the whole pipeline for one URL.
307async fn unfurl_traced(url: &Url) -> Trace {
308    let mut trace = Trace {
309        outcome: Outcome::Empty,
310        steps: Vec::new(),
311        raw: Vec::new(),
312        oembed_endpoint: None,
313        oembed_html: None,
314        warnings: Vec::new(),
315    };
316    let fetcher = SafeFetcher::new();
317
318    let fetched = match fetcher
319        .get(url, FetchLimits::page(), PAGE_ACCEPT, &mut trace.steps)
320        .await
321    {
322        Ok(fetched) => fetched,
323        Err(e) => {
324            trace.outcome = Outcome::Failed(describe(&e));
325            return trace;
326        }
327    };
328
329    let mime = fetched
330        .content_type
331        .as_deref()
332        .and_then(|ct| ct.split(';').next())
333        .unwrap_or_default()
334        .trim()
335        .to_ascii_lowercase();
336
337    // A URL that IS an image/video/audio file has no markup to scrape; the
338    // resource is its own preview.
339    if mime.starts_with("image/") || mime.starts_with("video/") || mime.starts_with("audio/") {
340        trace.outcome = Outcome::Embed(Box::new(build::from_media(
341            url.to_string(),
342            &fetched.final_url,
343            &mime,
344        )));
345        return trace;
346    }
347    if !(mime.is_empty() || mime.contains("html") || mime.contains("xml")) {
348        // A PDF, a zip, a JSON API response: fetched fine, nothing to scrape.
349        trace.outcome = Outcome::Failed(describe(&FetchError::UnsupportedContentType(mime)));
350        return trace;
351    }
352    if fetched.truncated {
353        trace
354            .warnings
355            .push("the page was larger than the fetch limit and was read only in part".to_owned());
356    }
357
358    let html = parse::decode_html(&fetched.body, fetched.content_type.as_deref());
359    let meta = parse::scrape(&html);
360    trace.oembed_endpoint = meta.oembed_json.clone();
361
362    let oembed = match &meta.oembed_json {
363        Some(endpoint) => fetch_oembed(&fetcher, &fetched.final_url, endpoint, &mut trace).await,
364        None => None,
365    };
366    if let Some(oe) = &oembed {
367        trace.oembed_html = oe.html.clone();
368    }
369
370    trace.raw = build::raw_fields(&meta, oembed.as_ref());
371    let embed = build::from_html(
372        Scraped {
373            request_url: url.to_string(),
374            final_url: fetched.final_url.clone(),
375            meta,
376            oembed,
377        },
378        &mut trace.warnings,
379    );
380    trace.outcome = if embed.is_renderable() {
381        Outcome::Embed(Box::new(embed))
382    } else {
383        Outcome::Empty
384    };
385    trace
386}

388/// Fetch and decode the discovered oEmbed endpoint.
389///
390/// Guarded exactly like the page fetch — the endpoint URL comes out of the same
391/// untrusted markup, so it gets no more trust than the page did. A failure here
392/// is a warning, never the end of the unfurl: Open Graph alone usually carries
393/// enough for a card.
394async fn fetch_oembed(
395    fetcher: &SafeFetcher,
396    base: &Url,
397    endpoint: &str,
398    trace: &mut Trace,
399) -> Option<OEmbed> {
400    let Ok(url) = base.join(endpoint) else {
401        trace
402            .warnings
403            .push("the declared oEmbed endpoint is not a URL".to_owned());
404        return None;
405    };
406    let fetched: Fetched = match fetcher
407        .get(
408            &url,
409            FetchLimits::oembed(),
410            "application/json,text/json;q=0.9",
411            &mut trace.steps,
412        )
413        .await
414    {
415        Ok(fetched) => fetched,
416        Err(e) => {
417            trace
418                .warnings
419                .push(format!("oEmbed endpoint: {}", describe(&e)));
420            return None;
421        }
422    };
423    match serde_json::from_slice::<OEmbed>(&fetched.body) {
424        Ok(oe) if oe.is_useful() => Some(oe),
425        Ok(_) => {
426            trace
427                .warnings
428                .push("the oEmbed endpoint answered with nothing usable".to_owned());
429            None
430        }
431        Err(e) => {
432            trace
433                .warnings
434                .push(format!("the oEmbed response is not valid JSON: {e}"));
435            None
436        }
437    }
438}

440/// A fetch failure as a person should read it.
441///
442/// The guard's own `Display` is already written for humans; this exists so the
443/// one case that is genuinely ambiguous — a blocked address — says why rather
444/// than leaking the impression that the service is broken.
445fn describe(error: &FetchError) -> String {
446    match error {
447        FetchError::PrivateAddress { .. } => {
448            "that address is not on the public internet, so it is not previewed".to_owned()
449        }
450        other => other.to_string(),
451    }
