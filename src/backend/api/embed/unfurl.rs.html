1//! The unfurl pipeline: message text in, stored embeds and a realtime push out.
2//!
3//! ## Why this runs after the message is stored
4//!
5//! Unfurling is a network round-trip against a stranger's server. Doing it on
6//! the POST would put an eight-second worst case on every send containing a
7//! link, and would let a slow host pin a request worker. So the handler stores
8//! the message, answers, and spawns [`Api::spawn_unfurl`]; the previews arrive
9//! moments later as [`ServerMessage::MessageEmbedsUpdated`].
10//!
11//! The cost of that choice is a visible two-step in the UI (message, then card),
12//! which is the same behaviour every other chat client has, and is why the
13//! realtime frame is a *replacement list* rather than an append — a client that
14//! misses it and re-reads history still converges.
15//!
16//! ## Which links get unfurled
17//!
18//! The first [`MAX_MESSAGE_EMBEDS`] distinct `http(s)` URLs in the body, in the
19//! order they appear. Distinct by exact URL: the same link twice is one card.
20//!
21//! ## Failure is not an error
22//!
23//! Nothing here can fail the message. Every outcome — refused by the guard,
24//! timed out, a 404, a page with no metadata — is recorded in the cache so it is
25//! not retried on every mention, and produces no card.
26

40/// What this service will accept as a previewable page.
41const PAGE_ACCEPT: &str = "text/html,application/xhtml+xml;q=0.9,*/*;q=0.5";

43/// Extract the URLs worth unfurling from a message body, in document order,
44/// de-duplicated.
45///
46/// `linkify` rather than a regular expression: matching URLs in prose is a
47/// famously bad fit for one, and the crate already handles the cases that bite —
48/// a trailing full stop, a URL inside brackets, an email address that is not a
49/// link.
50pub(crate) fn extract_urls(contents: &str) -> Vec<Url> {
51    let mut finder = linkify::LinkFinder::new();
52    finder.kinds(&[linkify::LinkKind::Url]);
53    let mut seen: Vec<Url> = Vec::new();
54    for found in finder.links(contents) {
55        let Ok(url) = Url::parse(found.as_str()) else {
56            continue;
57        };
58        if !matches!(url.scheme(), "http" | "https") {
59            continue;
60        }
61        if seen.iter().any(|existing| existing == &url) {
62            continue;
63        }
64        seen.push(url);
65        if seen.len() == MAX_MESSAGE_EMBEDS {
66            break;
67        }
68    }
69    seen
70}

72impl Api {
73    /// Unfurl a message's links in the background and push the result.
74    ///
75    /// Detached on purpose: the caller has already answered the client. The task
76    /// holds only a cloned [`Api`] handle (an `Arc`), so it cannot outlive the
77    /// process or hold a request open.
78    ///
79    /// A message with no links still runs, and still emits — an *edit* that
80    /// removed the last link has to be able to take its card away.
81    pub(crate) fn spawn_unfurl(&self, message_id: Uuid, space_id: Uuid, contents: String) {
82        let api = self.clone();
83        tokio::spawn(async move {
84            let urls = extract_urls(&contents);
85            let had_embeds = !api.embeds_for_message(message_id).await.is_empty();
86            if urls.is_empty() && !had_embeds {
87                return;
88            }
89
90            let mut ids = Vec::with_capacity(urls.len());
91            let mut embeds = Vec::with_capacity(urls.len());
92            for url in urls {
93                match api.unfurl_cached(&url).await {
94                    Some((id, Some(embed))) => {
95                        ids.push(id);
96                        embeds.push(embed);
97                    }
98                    // Unfurled to nothing: the join row is still recorded, so a
99                    // later refresh of that URL lands on this message without
100                    // re-parsing the body.
101                    Some((id, None)) => ids.push(id),
102                    None => {}
103                }
104            }
105
106            if let Err(e) = api.set_message_embeds(message_id, &ids).await {
107                tracing::warn!(%message_id, error = %e, "could not store message embeds");
108                return;
109            }
110            metrics::counter!("backend_message_embeds_total").increment(embeds.len() as u64);
111            api.emit_message_embeds(message_id, space_id, embeds).await;
112        });
113    }

115    /// The cache-aware unfurl of one URL: `(embed id, embed)`.
116    ///
117    /// `None` only when the database itself would not cooperate — every
118    /// *fetch* outcome is a successful cache write.
119    async fn unfurl_cached(&self, url: &Url) -> Option<(Uuid, Option<LinkEmbed>)> {
120        let key = url.as_str();
121        match self.find_fresh_embed(key).await {
122            Ok(Some(cached)) => {
123                metrics::counter!("backend_embed_cache_hits_total").increment(1);
124                return Some((cached.id, cached.embed));
125            }
126            Ok(None) => {}
127            Err(e) => {
128                tracing::warn!(%url, error = %e, "embed cache lookup failed");
129                return None;
130            }
131        }
132
133        let id = match self.reserve_embed_id(key).await {
134            Ok(id) => id,
135            Err(e) => {
136                tracing::warn!(%url, error = %e, "could not reserve an embed row");
137                return None;
138            }
139        };
140
141        metrics::counter!("backend_embed_cache_misses_total").increment(1);
142        let outcome = unfurl(url).await;
143        let (status, embed, error) = match outcome {
144            Outcome::Embed(mut embed) => {
145                assign_proxy_urls(&mut embed, id);
146                (EmbedStatus::Ok, Some(*embed), None)
147            }
148            Outcome::Empty => (EmbedStatus::Empty, None, None),
149            Outcome::Failed(message) => {
150                metrics::counter!("backend_embed_unfurl_failures_total").increment(1);
151                (EmbedStatus::Error, None, Some(message))
152            }
153        };
154        if let Err(e) = self
155            .write_embed(id, key, status, embed.as_ref(), error)
156            .await
157        {
158            tracing::warn!(%url, error = %e, "could not store an unfurled embed");
159        }
160        Some((id, embed))
161    }

163    /// Unfurl one URL with the cache **bypassed**, collecting the full trace.
164    /// Backs the embed debugger.
165    pub(crate) async fn debug_unfurl(&self, raw: &str) -> EmbedDebugReport {
166        let started = std::time::Instant::now();
167        let mut report = EmbedDebugReport {
168            request_url: raw.to_owned(),
169            embed: None,
170            steps: Vec::new(),
171            raw: Vec::new(),
172            oembed_endpoint: None,
173            oembed_html: None,
174            warnings: Vec::new(),
175            error: None,
176            cached: false,
177            elapsed_ms: 0,
178        };
179
180        let Ok(url) = Url::parse(raw.trim()) else {
181            report.error = Some("that is not a URL".to_owned());
182            report.elapsed_ms = started.elapsed().as_millis() as u32;
183            return report;
184        };
185        if !matches!(url.scheme(), "http" | "https") {
186            report.error = Some(format!(
187                "`{}` links are never previewed — only http and https",
188                url.scheme()
189            ));
190            report.elapsed_ms = started.elapsed().as_millis() as u32;
191            return report;
192        }
193
194        // Reported, not honoured: the debugger always re-fetches, so an operator
195        // sees what the site says *now* rather than what it said yesterday. What
196        // the cache holds is still worth showing — "a message would have hit the
197        // cache" explains a stale-looking card.
198        report.cached = matches!(self.find_fresh_embed(url.as_str()).await, Ok(Some(_)));
199
200        let trace = unfurl_traced(&url).await;
201        report.steps = trace.steps;
202        report.raw = trace.raw;
203        report.oembed_endpoint = trace.oembed_endpoint;
204        report.oembed_html = trace.oembed_html;
205        report.warnings = trace.warnings;
206        match trace.outcome {
207            Outcome::Embed(mut embed) => {
208                // The preview is only honest if its images actually load, and a
209                // proxy URL names the ROW that carries it — so the row has to
210                // exist. Reserving the id (idempotent: it adopts an existing row
211                // for this URL) and writing what was just fetched is what makes
212                // the debugger's card byte-identical to the one a channel
213                // renders, which is the tool's entire promise.
214                //
215                // Stamping only when a fresh row happened to exist is what this
216                // replaced, and it was wrong in the case the tool exists for: a
217                // URL nobody had posted yet has no row, so every media URL came
218                // back as the empty string it was built with and every image in
219                // the preview silently failed to load.
220                //
221                // Writing also refreshes the cache with what the site says NOW,
222                // so debugging a stale-looking card is what fixes it. Only a
223                // success is written: a failure the operator is actively
224                // investigating should not be pinned into the cache underneath
225                // them.
226                match self.reserve_embed_id(url.as_str()).await {
227                    Ok(id) => {
228                        assign_proxy_urls(&mut embed, id);
229                        if let Err(e) = self
230                            .write_embed(id, url.as_str(), EmbedStatus::Ok, Some(&embed), None)
231                            .await
232                        {
233                            tracing::warn!(%url, error = %e, "debug unfurl: could not store the embed");
234                            report.warnings.push(
235                                "the preview could not be stored, so its images may not load"
236                                    .to_owned(),
237                            );
238                        }
239                    }
240                    Err(e) => {
241                        tracing::warn!(%url, error = %e, "debug unfurl: could not reserve a row");
242                        report.warnings.push(
243                            "no row could be reserved for this preview, so its images will not load"
244                                .to_owned(),
245                        );
246                    }
247                }
248                report.embed = Some(*embed);
249            }
250            Outcome::Empty => {
251                report.error = Some(
252                    "the page was fetched but carries no title, description or image".to_owned(),
253                );
254            }
255            Outcome::Failed(message) => report.error = Some(message),
256        }
257        report.elapsed_ms = started.elapsed().as_millis() as u32;
258        report
259    }

261    /// Push a message's embed list to its space.
262    pub(crate) async fn emit_message_embeds(
263        &self,
264        message_id: Uuid,
265        space_id: Uuid,
266        embeds: Vec<LinkEmbed>,
267    ) {
268        self.realtime.route(&[hound_realtime::Outgoing::to(
269            hound_realtime::Target::Room(hound_realtime::Room::Space(space_id)),
270            hound_realtime::ServerMessage::MessageEmbedsUpdated {
271                message_id,
272                space_id,
273                embeds,
274            },
275        )]);
276    }

279/// What one URL unfurled to.
280enum Outcome {
281    /// Boxed: a `LinkEmbed` is ~700 bytes and the other two variants are a
282    /// `String` and nothing, so an unboxed enum would pay the largest variant's
283    /// size on every unfurl that failed or found nothing — the common case.
284    Embed(Box<LinkEmbed>),
285    /// Fetched successfully; there was nothing worth a card.
286    Empty,
287    /// Refused, unreachable, or an error status. The string is shown in the
288    /// debugger and stored on the cache row.
289    Failed(String),
290}
291
292/// Everything a traced unfurl saw. The message path throws all but `outcome`
293/// away; the debugger keeps the lot.
294struct Trace {
295    outcome: Outcome,
296    steps: Vec<hound_types::EmbedFetchStep>,
297    raw: Vec<hound_types::EmbedRawField>,
298    oembed_endpoint: Option<String>,
299    oembed_html: Option<String>,
300    warnings: Vec<String>,
301}
302
303async fn unfurl(url: &Url) -> Outcome {
304    unfurl_traced(url).await.outcome
305}

307/// Fetch, parse, discover oEmbed, and build — the whole pipeline for one URL.
308async fn unfurl_traced(url: &Url) -> Trace {
309    let mut trace = Trace {
310        outcome: Outcome::Empty,
311        steps: Vec::new(),
312        raw: Vec::new(),
313        oembed_endpoint: None,
314        oembed_html: None,
315        warnings: Vec::new(),
316    };
317    let fetcher = SafeFetcher::new();
318
319    let fetched = match fetcher
320        .get(url, FetchLimits::page(), PAGE_ACCEPT, &mut trace.steps)
321        .await
322    {
323        Ok(fetched) => fetched,
324        Err(e) => {
325            // The page is not the only thing that can describe this URL. For a
326            // host we recognise, the provider's own endpoint answers for it —
327            // and routinely answers well when the page did not load at all.
328            if let Some(embed) = provider_only(&fetcher, url, &mut trace).await {
329                trace.outcome = if embed.is_renderable() {
330                    Outcome::Embed(Box::new(embed))
331                } else {
332                    Outcome::Empty
333                };
334                return trace;
335            }
336            trace.outcome = Outcome::Failed(describe(&e));
337            return trace;
338        }
339    };
340
341    let mime = fetched
342        .content_type
343        .as_deref()
344        .and_then(|ct| ct.split(';').next())
345        .unwrap_or_default()
346        .trim()
347        .to_ascii_lowercase();
348
349    // A URL that IS an image/video/audio file has no markup to scrape; the
350    // resource is its own preview.
351    if mime.starts_with("image/") || mime.starts_with("video/") || mime.starts_with("audio/") {
352        trace.outcome = Outcome::Embed(Box::new(build::from_media(
353            url.to_string(),
354            &fetched.final_url,
355            &mime,
356        )));
357        return trace;
358    }
359    if !(mime.is_empty() || mime.contains("html") || mime.contains("xml")) {
360        // A PDF, a zip, a JSON API response: fetched fine, nothing to scrape.
361        trace.outcome = Outcome::Failed(describe(&FetchError::UnsupportedContentType(mime)));
362        return trace;
363    }
364    if fetched.truncated {
365        trace
366            .warnings
367            .push("the page was larger than the fetch limit and was read only in part".to_owned());
368    }
369
370    let html = parse::decode_html(&fetched.body, fetched.content_type.as_deref());
371    let meta = parse::scrape(&html);
372    // The registry wins over discovery. A page that advertises nothing is the
373    // common failure — a shell served to a bot, or a provider that simply never
374    // adds the link — and for a host we know, the endpoint is not something we
375    // need the page's permission to use.
376    let endpoint = providers::lookup(url)
377        .and_then(|provider| provider.oembed_endpoint(url))
378        .map(|endpoint| endpoint.to_string())
379        .or_else(|| meta.oembed_json.clone());
380    trace.oembed_endpoint = endpoint.clone();
381
382    let oembed = match &endpoint {
383        Some(endpoint) => fetch_oembed(&fetcher, &fetched.final_url, endpoint, &mut trace).await,
384        None => None,
385    };
386    if let Some(oe) = &oembed {
387        trace.oembed_html = oe.html.clone();
388    }
389
390    trace.raw = build::raw_fields(&meta, oembed.as_ref());
391    let embed = build::from_html(
392        Scraped {
393            request_url: url.to_string(),
394            final_url: fetched.final_url.clone(),
395            meta,
396            oembed,
397        },
398        &mut trace.warnings,
399    );
400    trace.outcome = if embed.is_renderable() {
401        Outcome::Embed(Box::new(embed))
402    } else {
403        Outcome::Empty
404    };
405    trace
406}

408/// Build a card from a known provider's oEmbed endpoint alone.
409///
410/// Used when the page fetch failed. Returns `None` when the host is unknown or
411/// the endpoint could not answer either, so the caller reports the page's own
412/// failure — the more useful error in that case.
413async fn provider_only(fetcher: &SafeFetcher, url: &Url, trace: &mut Trace) -> Option<LinkEmbed> {
414    let provider = providers::lookup(url)?;
415    let endpoint = provider.oembed_endpoint(url)?;
416    trace.oembed_endpoint = Some(endpoint.to_string());
417    let oembed = fetch_oembed(fetcher, url, endpoint.as_str(), trace).await?;
418    trace.oembed_html = oembed.html.clone();
419    trace.raw = build::raw_fields(&parse::PageMetadata::default(), Some(&oembed));
420    trace.warnings.push(format!(
421        "the page itself could not be read; this preview comes from {}'s oEmbed endpoint",
422        provider.name
423    ));
424    Some(build::from_oembed(
425        url.to_string(),
426        url,
427        &oembed,
428        &mut trace.warnings,
429    ))
430}

432/// Fetch and decode the discovered oEmbed endpoint.
433///
434/// Guarded exactly like the page fetch — the endpoint URL comes out of the same
435/// untrusted markup, so it gets no more trust than the page did. A failure here
436/// is a warning, never the end of the unfurl: Open Graph alone usually carries
437/// enough for a card.
438async fn fetch_oembed(
439    fetcher: &SafeFetcher,
440    base: &Url,
441    endpoint: &str,
442    trace: &mut Trace,
443) -> Option<OEmbed> {
444    let Ok(url) = base.join(endpoint) else {
445        trace
446            .warnings
447            .push("the declared oEmbed endpoint is not a URL".to_owned());
448        return None;
449    };
450    let fetched: Fetched = match fetcher
451        .get(
452            &url,
453            FetchLimits::oembed(),
454            "application/json,text/json;q=0.9",
455            &mut trace.steps,
456        )
457        .await
458    {
459        Ok(fetched) => fetched,
460        Err(e) => {
461            trace
462                .warnings
463                .push(format!("oEmbed endpoint: {}", describe(&e)));
464            return None;
465        }
466    };
467    match serde_json::from_slice::<OEmbed>(&fetched.body) {
468        Ok(oe) if oe.is_useful() => Some(oe),
469        Ok(_) => {
470            trace
471                .warnings
472                .push("the oEmbed endpoint answered with nothing usable".to_owned());
473            None
474        }
475        Err(e) => {
476            trace
477                .warnings
478                .push(format!("the oEmbed response is not valid JSON: {e}"));
479            None
480        }
481    }
482}

484/// A fetch failure as a person should read it.
485///
486/// The guard's own `Display` is already written for humans; this exists so the
487/// one case that is genuinely ambiguous — a blocked address — says why rather
488/// than leaking the impression that the service is broken.
489fn describe(error: &FetchError) -> String {
490    match error {
491        FetchError::PrivateAddress { .. } => {
492            "that address is not on the public internet, so it is not previewed".to_owned()
493        }
494        other => other.to_string(),
495    }
