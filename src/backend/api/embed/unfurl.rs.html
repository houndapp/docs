1//! The unfurl pipeline: message text in, stored embeds and a realtime push out.
2//!
3//! ## Why this runs after the message is stored
4//!
5//! Unfurling is a network round-trip against a stranger's server. Doing it on
6//! the POST would put an eight-second worst case on every send containing a
7//! link, and would let a slow host pin a request worker. So the handler stores
8//! the message, answers, and spawns [`Api::spawn_unfurl`]; the previews arrive
9//! moments later as [`ServerMessage::MessageEmbedsUpdated`].
10//!
11//! The cost of that choice is a visible two-step in the UI (message, then card),
12//! which is the same behaviour every other chat client has, and is why the
13//! realtime frame is a *replacement list* rather than an append — a client that
14//! misses it and re-reads history still converges.
15//!
16//! ## Which links get unfurled
17//!
18//! The first [`MAX_MESSAGE_EMBEDS`] distinct `http(s)` URLs in the body, in the
19//! order they appear. Distinct by exact URL: the same link twice is one card.
20//!
21//! ## Failure is not an error
22//!
23//! Nothing here can fail the message. Every outcome — refused by the guard,
24//! timed out, a 404, a page with no metadata — is recorded in the cache so it is
25//! not retried on every mention, and produces no card.
26

39/// What this service will accept as a previewable page.
40const PAGE_ACCEPT: &str = "text/html,application/xhtml+xml;q=0.9,*/*;q=0.5";

42/// Extract the URLs worth unfurling from a message body, in document order,
43/// de-duplicated.
44///
45/// `linkify` rather than a regular expression: matching URLs in prose is a
46/// famously bad fit for one, and the crate already handles the cases that bite —
47/// a trailing full stop, a URL inside brackets, an email address that is not a
48/// link.
49pub(crate) fn extract_urls(contents: &str) -> Vec<Url> {
50    let mut finder = linkify::LinkFinder::new();
51    finder.kinds(&[linkify::LinkKind::Url]);
52    let mut seen: Vec<Url> = Vec::new();
53    for found in finder.links(contents) {
54        let Ok(url) = Url::parse(found.as_str()) else {
55            continue;
56        };
57        if !matches!(url.scheme(), "http" | "https") {
58            continue;
59        }
60        if seen.iter().any(|existing| existing == &url) {
61            continue;
62        }
63        seen.push(url);
64        if seen.len() == MAX_MESSAGE_EMBEDS {
65            break;
66        }
67    }
68    seen
69}

71impl Api {
72    /// Unfurl a message's links in the background and push the result.
73    ///
74    /// Detached on purpose: the caller has already answered the client. The task
75    /// holds only a cloned [`Api`] handle (an `Arc`), so it cannot outlive the
76    /// process or hold a request open.
77    ///
78    /// A message with no links still runs, and still emits — an *edit* that
79    /// removed the last link has to be able to take its card away.
80    pub(crate) fn spawn_unfurl(&self, message_id: Uuid, space_id: Uuid, contents: String) {
81        let api = self.clone();
82        tokio::spawn(async move {
83            let urls = extract_urls(&contents);
84            let had_embeds = !api.embeds_for_message(message_id).await.is_empty();
85            if urls.is_empty() && !had_embeds {
86                return;
87            }
88
89            let mut ids = Vec::with_capacity(urls.len());
90            let mut embeds = Vec::with_capacity(urls.len());
91            for url in urls {
92                match api.unfurl_cached(&url).await {
93                    Some((id, Some(embed))) => {
94                        ids.push(id);
95                        embeds.push(embed);
96                    }
97                    // Unfurled to nothing: the join row is still recorded, so a
98                    // later refresh of that URL lands on this message without
99                    // re-parsing the body.
100                    Some((id, None)) => ids.push(id),
101                    None => {}
102                }
103            }
104
105            if let Err(e) = api.set_message_embeds(message_id, &ids).await {
106                tracing::warn!(%message_id, error = %e, "could not store message embeds");
107                return;
108            }
109            metrics::counter!("backend_message_embeds_total").increment(embeds.len() as u64);
110            api.emit_message_embeds(message_id, space_id, embeds).await;
111        });
112    }

114    /// The cache-aware unfurl of one URL: `(embed id, embed)`.
115    ///
116    /// `None` only when the database itself would not cooperate — every
117    /// *fetch* outcome is a successful cache write.
118    async fn unfurl_cached(&self, url: &Url) -> Option<(Uuid, Option<LinkEmbed>)> {
119        let key = url.as_str();
120        match self.find_fresh_embed(key).await {
121            Ok(Some(cached)) => {
122                metrics::counter!("backend_embed_cache_hits_total").increment(1);
123                return Some((cached.id, cached.embed));
124            }
125            Ok(None) => {}
126            Err(e) => {
127                tracing::warn!(%url, error = %e, "embed cache lookup failed");
128                return None;
129            }
130        }
131
132        let id = match self.reserve_embed_id(key).await {
133            Ok(id) => id,
134            Err(e) => {
135                tracing::warn!(%url, error = %e, "could not reserve an embed row");
136                return None;
137            }
138        };
139
140        metrics::counter!("backend_embed_cache_misses_total").increment(1);
141        let outcome = unfurl(url).await;
142        let (status, embed, error) = match outcome {
143            Outcome::Embed(mut embed) => {
144                assign_proxy_urls(&mut embed, id);
145                (EmbedStatus::Ok, Some(*embed), None)
146            }
147            Outcome::Empty => (EmbedStatus::Empty, None, None),
148            Outcome::Failed(message) => {
149                metrics::counter!("backend_embed_unfurl_failures_total").increment(1);
150                (EmbedStatus::Error, None, Some(message))
151            }
152        };
153        if let Err(e) = self
154            .write_embed(id, key, status, embed.as_ref(), error)
155            .await
156        {
157            tracing::warn!(%url, error = %e, "could not store an unfurled embed");
158        }
159        Some((id, embed))
160    }

162    /// Unfurl one URL with the cache **bypassed**, collecting the full trace.
163    /// Backs the embed debugger.
164    pub(crate) async fn debug_unfurl(&self, raw: &str) -> EmbedDebugReport {
165        let started = std::time::Instant::now();
166        let mut report = EmbedDebugReport {
167            request_url: raw.to_owned(),
168            embed: None,
169            steps: Vec::new(),
170            raw: Vec::new(),
171            oembed_endpoint: None,
172            oembed_html: None,
173            warnings: Vec::new(),
174            error: None,
175            cached: false,
176            elapsed_ms: 0,
177        };
178
179        let Ok(url) = Url::parse(raw.trim()) else {
180            report.error = Some("that is not a URL".to_owned());
181            report.elapsed_ms = started.elapsed().as_millis() as u32;
182            return report;
183        };
184        if !matches!(url.scheme(), "http" | "https") {
185            report.error = Some(format!(
186                "`{}` links are never previewed — only http and https",
187                url.scheme()
188            ));
189            report.elapsed_ms = started.elapsed().as_millis() as u32;
190            return report;
191        }
192
193        // Reported, not honoured: the debugger always re-fetches, so an operator
194        // sees what the site says *now* rather than what it said yesterday. What
195        // the cache holds is still worth showing — "a message would have hit the
196        // cache" explains a stale-looking card.
197        report.cached = matches!(self.find_fresh_embed(url.as_str()).await, Ok(Some(_)));
198
199        let trace = unfurl_traced(&url).await;
200        report.steps = trace.steps;
201        report.raw = trace.raw;
202        report.oembed_endpoint = trace.oembed_endpoint;
203        report.oembed_html = trace.oembed_html;
204        report.warnings = trace.warnings;
205        match trace.outcome {
206            Outcome::Embed(mut embed) => {
207                // Stamped with the id a real unfurl would have used, so the
208                // debugger's media URLs resolve against the stored row where one
209                // exists — and are at least shaped correctly where it does not.
210                if let Ok(Some(cached)) = self.find_fresh_embed(url.as_str()).await {
211                    assign_proxy_urls(&mut embed, cached.id);
212                }
213                report.embed = Some(*embed);
214            }
215            Outcome::Empty => {
216                report.error = Some(
217                    "the page was fetched but carries no title, description or image".to_owned(),
218                );
219            }
220            Outcome::Failed(message) => report.error = Some(message),
221        }
222        report.elapsed_ms = started.elapsed().as_millis() as u32;
223        report
224    }

226    /// Push a message's embed list to its space.
227    pub(crate) async fn emit_message_embeds(
228        &self,
229        message_id: Uuid,
230        space_id: Uuid,
231        embeds: Vec<LinkEmbed>,
232    ) {
233        self.realtime.route(&[hound_realtime::Outgoing::to(
234            hound_realtime::Target::Room(hound_realtime::Room::Space(space_id)),
235            hound_realtime::ServerMessage::MessageEmbedsUpdated {
236                message_id,
237                space_id,
238                embeds,
239            },
240        )]);
241    }

244/// What one URL unfurled to.
245enum Outcome {
246    /// Boxed: a `LinkEmbed` is ~700 bytes and the other two variants are a
247    /// `String` and nothing, so an unboxed enum would pay the largest variant's
248    /// size on every unfurl that failed or found nothing — the common case.
249    Embed(Box<LinkEmbed>),
250    /// Fetched successfully; there was nothing worth a card.
251    Empty,
252    /// Refused, unreachable, or an error status. The string is shown in the
253    /// debugger and stored on the cache row.
254    Failed(String),
255}
256
257/// Everything a traced unfurl saw. The message path throws all but `outcome`
258/// away; the debugger keeps the lot.
259struct Trace {
260    outcome: Outcome,
261    steps: Vec<hound_types::EmbedFetchStep>,
262    raw: Vec<hound_types::EmbedRawField>,
263    oembed_endpoint: Option<String>,
264    oembed_html: Option<String>,
265    warnings: Vec<String>,
266}
267
268async fn unfurl(url: &Url) -> Outcome {
269    unfurl_traced(url).await.outcome
270}

272/// Fetch, parse, discover oEmbed, and build — the whole pipeline for one URL.
273async fn unfurl_traced(url: &Url) -> Trace {
274    let mut trace = Trace {
275        outcome: Outcome::Empty,
276        steps: Vec::new(),
277        raw: Vec::new(),
278        oembed_endpoint: None,
279        oembed_html: None,
280        warnings: Vec::new(),
281    };
282    let fetcher = SafeFetcher::new();
283
284    let fetched = match fetcher
285        .get(url, FetchLimits::page(), PAGE_ACCEPT, &mut trace.steps)
286        .await
287    {
288        Ok(fetched) => fetched,
289        Err(e) => {
290            trace.outcome = Outcome::Failed(describe(&e));
291            return trace;
292        }
293    };
294
295    let mime = fetched
296        .content_type
297        .as_deref()
298        .and_then(|ct| ct.split(';').next())
299        .unwrap_or_default()
300        .trim()
301        .to_ascii_lowercase();
302
303    // A URL that IS an image/video/audio file has no markup to scrape; the
304    // resource is its own preview.
305    if mime.starts_with("image/") || mime.starts_with("video/") || mime.starts_with("audio/") {
306        trace.outcome = Outcome::Embed(Box::new(build::from_media(
307            url.to_string(),
308            &fetched.final_url,
309            &mime,
310        )));
311        return trace;
312    }
313    if !(mime.is_empty() || mime.contains("html") || mime.contains("xml")) {
314        // A PDF, a zip, a JSON API response: fetched fine, nothing to scrape.
315        trace.outcome = Outcome::Failed(describe(&FetchError::UnsupportedContentType(mime)));
316        return trace;
317    }
318    if fetched.truncated {
319        trace
320            .warnings
321            .push("the page was larger than the fetch limit and was read only in part".to_owned());
322    }
323
324    let html = parse::decode_html(&fetched.body, fetched.content_type.as_deref());
325    let meta = parse::scrape(&html);
326    trace.oembed_endpoint = meta.oembed_json.clone();
327
328    let oembed = match &meta.oembed_json {
329        Some(endpoint) => fetch_oembed(&fetcher, &fetched.final_url, endpoint, &mut trace).await,
330        None => None,
331    };
332    if let Some(oe) = &oembed {
333        trace.oembed_html = oe.html.clone();
334    }
335
336    trace.raw = build::raw_fields(&meta, oembed.as_ref());
337    let embed = build::from_html(
338        Scraped {
339            request_url: url.to_string(),
340            final_url: fetched.final_url.clone(),
341            meta,
342            oembed,
343        },
344        &mut trace.warnings,
345    );
346    trace.outcome = if embed.is_renderable() {
347        Outcome::Embed(Box::new(embed))
348    } else {
349        Outcome::Empty
350    };
351    trace
352}

354/// Fetch and decode the discovered oEmbed endpoint.
355///
356/// Guarded exactly like the page fetch — the endpoint URL comes out of the same
357/// untrusted markup, so it gets no more trust than the page did. A failure here
358/// is a warning, never the end of the unfurl: Open Graph alone usually carries
359/// enough for a card.
360async fn fetch_oembed(
361    fetcher: &SafeFetcher,
362    base: &Url,
363    endpoint: &str,
364    trace: &mut Trace,
365) -> Option<OEmbed> {
366    let Ok(url) = base.join(endpoint) else {
367        trace
368            .warnings
369            .push("the declared oEmbed endpoint is not a URL".to_owned());
370        return None;
371    };
372    let fetched: Fetched = match fetcher
373        .get(
374            &url,
375            FetchLimits::oembed(),
376            "application/json,text/json;q=0.9",
377            &mut trace.steps,
378        )
379        .await
380    {
381        Ok(fetched) => fetched,
382        Err(e) => {
383            trace
384                .warnings
385                .push(format!("oEmbed endpoint: {}", describe(&e)));
386            return None;
387        }
388    };
389    match serde_json::from_slice::<OEmbed>(&fetched.body) {
390        Ok(oe) if oe.is_useful() => Some(oe),
391        Ok(_) => {
392            trace
393                .warnings
394                .push("the oEmbed endpoint answered with nothing usable".to_owned());
395            None
396        }
397        Err(e) => {
398            trace
399                .warnings
400                .push(format!("the oEmbed response is not valid JSON: {e}"));
401            None
402        }
403    }
404}

406/// A fetch failure as a person should read it.
407///
408/// The guard's own `Display` is already written for humans; this exists so the
409/// one case that is genuinely ambiguous — a blocked address — says why rather
410/// than leaking the impression that the service is broken.
411fn describe(error: &FetchError) -> String {
412    match error {
413        FetchError::PrivateAddress { .. } => {
414            "that address is not on the public internet, so it is not previewed".to_owned()
415        }
416        other => other.to_string(),
417    }
