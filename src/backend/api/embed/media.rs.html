1//! The embed media proxy: the only path by which an embed's images, poster
2//! frames and video reach a client.
3//!
4//! ## Why a proxy at all
5//!
6//! A client that loaded `og:image` directly would hand its IP address — and,
7//! by the timing of the request, its presence — to whatever host the poster
8//! chose. A one-pixel `og:image` is a read receipt for a whole channel. The
9//! proxy makes every embed asset come from this deployment instead.
10//!
11//! ## Why it cannot be turned into a weapon
12//!
13//! Three properties, together:
14//!
15//! * **A closed URL set.** The route addresses an embed row and a slot key
16//!   (`image-0`, `video`, `favicon`); the remote URL is looked up, never
17//!   supplied. There is no request shape that makes this fetch an arbitrary URL,
18//!   which is what separates it from an open proxy.
19//! * **A cache in front of the origin.** A hot embed is served from memory, so a
20//!   channel full of people opening the same message is one upstream fetch, not
21//!   hundreds. Without this, an embed would be a bandwidth amplifier pointed at
22//!   whatever it linked.
23//! * **The same guard as the unfurl.** The fetch runs through
24//!   [`SafeFetcher`](super::guard::SafeFetcher), so a stored URL that now
25//!   resolves to a private address is refused at fetch time, not merely at
26//!   unfurl time.
27//!
28//! ## What it serves
29//!
30//! Bytes whose type was confirmed by sniffing, under `nosniff` and a
31//! `default-src 'none'` CSP, with SVG refused outright — it is a script-bearing
32//! document wearing an image's content type, and nothing in this client can
33//! render one anyway.
34

49/// How long a client may reuse a proxied asset. Long: the URL is content-
50/// addressed by embed id, and an embed's media never changes under it — a new
51/// unfurl mints a new row.
52const CLIENT_CACHE_SECS: u32 = 60 * 60 * 24 * 7;
53
54/// Server-side byte budget across all cached assets.
55const CACHE_MAX_BYTES: usize = 64 * 1024 * 1024;
56
57/// Most distinct assets held, independent of size — a cap on bookkeeping, not on
58/// memory (that is [`CACHE_MAX_BYTES`]).
59const CACHE_MAX_ENTRIES: usize = 2048;
60
61/// The content types the proxy will serve. `image/svg+xml` is deliberately
62/// absent: an SVG is an XML document with script and external-reference
63/// capabilities, and no client here can render one regardless.
64const ALLOWED_TYPES: &[&str] = &[
65    "image/png",
66    "image/jpeg",
67    "image/gif",
68    "image/webp",
69    "image/avif",
70    "image/bmp",
71    "image/x-icon",
72    "image/vnd.microsoft.icon",
73    "image/apng",
74    "image/jxl",
75    "video/mp4",
76    "video/webm",
77    "video/ogg",
78    "video/quicktime",
79    "audio/mpeg",
80    "audio/mp4",
81    "audio/ogg",
82    "audio/wav",
83    "audio/webm",
84    "audio/flac",
85];

87/// A proxied asset held in memory.
88#[derive(Clone)]
89struct CachedMedia {
90    content_type: String,
91    bytes: Bytes,
92}

94/// LRU over proxied assets, bounded by both entry count and total bytes.
95pub(crate) struct MediaCache {
96    inner: Mutex<Inner>,
97}
98
99struct Inner {
100    lru: lru::LruCache<String, CachedMedia>,
101    bytes: usize,
102}
103
104impl std::fmt::Debug for MediaCache {
105    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
106        f.debug_struct("MediaCache").finish_non_exhaustive()
107    }
108}

110impl Default for MediaCache {
111    fn default() -> Self {
112        Self::new()
113    }
114}
115
116impl MediaCache {
117    pub(crate) fn new() -> Self {
118        Self {
119            inner: Mutex::new(Inner {
120                // `expect`: the constant is non-zero by inspection.
121                lru: lru::LruCache::new(
122                    NonZeroUsize::new(CACHE_MAX_ENTRIES).expect("non-zero cache capacity"),
123                ),
124                bytes: 0,
125            }),
126        }
127    }
128
129    fn get(&self, key: &str) -> Option<CachedMedia> {
130        // A poisoned lock here means a panic while holding it, which cannot
131        // corrupt anything a cache miss doesn't already handle — so the lock is
132        // recovered rather than propagated into a 500 on an image request.
133        let mut inner = self.inner.lock().unwrap_or_else(|e| e.into_inner());
134        inner.lru.get(key).cloned()
135    }
136
137    /// Drop everything held for one embed.
138    ///
139    /// Called when a stale row is refreshed in place: the id — and therefore
140    /// every proxy key — survives the refresh, so without this the cache would
141    /// keep serving the previous fetch's bytes for a payload that has moved on.
142    pub(crate) fn invalidate(&self, embed_id: Uuid) {
143        let prefix = format!("{embed_id}/");
144        let mut inner = self.inner.lock().unwrap_or_else(|e| e.into_inner());
145        let stale: Vec<String> = inner
146            .lru
147            .iter()
148            .filter(|(k, _)| k.starts_with(&prefix))
149            .map(|(k, _)| k.clone())
150            .collect();
151        for key in stale {
152            if let Some(entry) = inner.lru.pop(&key) {
153                inner.bytes = inner.bytes.saturating_sub(entry.bytes.len());
154            }
155        }
156    }
157
158    fn put(&self, key: String, value: CachedMedia) {
159        let len = value.bytes.len();
160        if len > CACHE_MAX_BYTES {
161            return;
162        }
163        let mut inner = self.inner.lock().unwrap_or_else(|e| e.into_inner());
164        if let Some(old) = inner.lru.put(key, value) {
165            inner.bytes = inner.bytes.saturating_sub(old.bytes.len());
166        }
167        inner.bytes += len;
168        while inner.bytes > CACHE_MAX_BYTES {
169            match inner.lru.pop_lru() {
170                Some((_, evicted)) => inner.bytes = inner.bytes.saturating_sub(evicted.bytes.len()),
171                None => break,
172            }
173        }
174    }
175}

179/// Stamp every media reference with its proxy path.
180///
181/// Called once, when the embed row gets its id: until then the media entries
182/// carry only their `source_url`, because the path they will be served at
183/// contains that id.
184pub(crate) fn assign_proxy_urls(embed: &mut LinkEmbed, embed_id: Uuid) {
185    for (index, image) in embed.images.iter_mut().enumerate() {
186        image.url = proxy_url(embed_id, &format!("image-{index}"));
187    }
188    if let Some(video) = embed.video.as_mut() {
189        video.media.url = proxy_url(embed_id, "video");
190    }
191    if let Some(audio) = embed.audio.as_mut() {
192        audio.url = proxy_url(embed_id, "audio");
193    }
194    if let Some(favicon) = embed.favicon.as_mut() {
195        favicon.url = proxy_url(embed_id, "favicon");
196    }
197}

199/// Relative on purpose — see [`EmbedMedia::url`](hound_types::EmbedMedia::url).
200fn proxy_url(embed_id: Uuid, key: &str) -> String {
201    format!("/embed/media/{embed_id}/{key}")
202}

204/// Resolve a slot key back to the media entry it names.
205///
206/// The inverse of [`assign_proxy_urls`], and the reason the key set is a closed
207/// vocabulary rather than an index into a list: an unknown key is a 404, not an
208/// out-of-bounds read.
209pub(crate) fn slot<'a>(embed: &'a LinkEmbed, key: &str) -> Option<&'a EmbedMedia> {
210    match key {
211        "video" => embed.video.as_ref().map(|v| &v.media),
212        "audio" => embed.audio.as_ref(),
213        "favicon" => embed.favicon.as_ref(),
214        other => {
215            let index: usize = other.strip_prefix("image-")?.parse().ok()?;
216            embed.images.get(index)
217        }
218    }
219}

223/// `GET /embed/media/{embed_id}/{key}`
224///
225/// Unauthenticated, matching the upload path: an embed asset is a copy of a
226/// public web page's image, and the id is only discoverable from a message you
227/// could already read. Authenticating it would also mean teaching the image
228/// loader to carry a bearer token, for no secret worth protecting.
229pub(crate) async fn get_embed_media(
230    State(api): State<Api>,
231    Path((embed_id, key)): Path<(Uuid, String)>,
232    headers: HeaderMap,
233) -> Response {
234    let cache_key = format!("{embed_id}/{key}");
235    if let Some(hit) = api.embed_media_cache.get(&cache_key) {
236        return serve(hit, &headers);
237    }
238
239    let Some(embed) = api.load_embed_payload(embed_id).await else {
240        return StatusCode::NOT_FOUND.into_response();
241    };
242    let Some(media) = slot(&embed, &key) else {
243        return StatusCode::NOT_FOUND.into_response();
244    };
245    let Ok(url) = url::Url::parse(&media.source_url) else {
246        return StatusCode::NOT_FOUND.into_response();
247    };
248
249    let mut steps = Vec::new();
250    let fetched = match SafeFetcher::new()
251        .get(
252            &url,
253            FetchLimits::media(),
254            "image/*,video/*,audio/*;q=0.9,*/*;q=0.5",
255            &mut steps,
256        )
257        .await
258    {
259        Ok(fetched) => fetched,
260        Err(e) => {
261            tracing::debug!(%embed_id, key, error = %e, "embed media fetch failed");
262            metrics::counter!("backend_embed_media_fetch_failures_total").increment(1);
263            // 502, not 404: the asset exists as far as this service knows and the
264            // *origin* is what failed. A client retries a 502 and gives up on a
265            // 404, which is the behaviour each case deserves.
266            return StatusCode::BAD_GATEWAY.into_response();
267        }
268    };
269
270    let Some(content_type) = resolve_type(&fetched.body, fetched.content_type.as_deref()) else {
271        return StatusCode::UNSUPPORTED_MEDIA_TYPE.into_response();
272    };
273
274    let entry = CachedMedia {
275        content_type: content_type.to_owned(),
276        bytes: fetched.body,
277    };
278    api.embed_media_cache.put(cache_key, entry.clone());
279    metrics::counter!("backend_embed_media_fetched_total").increment(1);
280    serve(entry, &headers)
281}

283/// Decide what content type to serve, refusing anything not in the allowlist.
284///
285/// The sniffed type wins over the declared one wherever the bytes are
286/// recognisable — the origin's header is a claim, and serving image bytes as
287/// whatever a third party said they were is exactly the confusion `nosniff` is
288/// meant to end.
289fn resolve_type(bytes: &[u8], declared: Option<&str>) -> Option<&'static str> {
290    if let Some(sniffed) = sniff(bytes) {
291        return Some(sniffed);
292    }
293    let declared = declared?.split(';').next()?.trim().to_ascii_lowercase();
294    ALLOWED_TYPES.iter().copied().find(|t| *t == declared)
295}
296
297/// Magic-byte sniff for the image formats worth recognising. Container formats
298/// (MP4, WebM) are left to the declared type: their signatures are far enough
299/// into the file to be unreliable on a range-served body, and they are already
300/// confined by the allowlist.
301fn sniff(bytes: &[u8]) -> Option<&'static str> {
302    const PNG: &[u8] = &[0x89, b'P', b'N', b'G', 0x0d, 0x0a, 0x1a, 0x0a];
303    if bytes.starts_with(PNG) {
304        // An APNG is a PNG with an extra chunk; both decode as PNG.
305        return Some("image/png");
306    }
307    if bytes.starts_with(&[0xff, 0xd8, 0xff]) {
308        return Some("image/jpeg");
309    }
310    if bytes.starts_with(b"GIF87a") || bytes.starts_with(b"GIF89a") {
311        return Some("image/gif");
312    }
313    if bytes.len() >= 12 && bytes.starts_with(b"RIFF") && &bytes[8..12] == b"WEBP" {
314        return Some("image/webp");
315    }
316    if bytes.len() >= 12 && &bytes[4..8] == b"ftyp" && matches!(&bytes[8..12], b"avif" | b"avis") {
317        return Some("image/avif");
318    }
319    if bytes.starts_with(b"BM") {
320        return Some("image/bmp");
321    }
322    if bytes.starts_with(&[0x00, 0x00, 0x01, 0x00]) {
323        return Some("image/x-icon");
324    }
325    if bytes.starts_with(&[0xff, 0x0a])
326        || bytes.starts_with(&[0x00, 0x00, 0x00, 0x0c, 0x4a, 0x58, 0x4c, 0x20])
327    {
328        return Some("image/jxl");
329    }
330    None
331}

333/// Send the bytes, honouring a single-range `Range` request.
334///
335/// Range support is here for one reason: a video element seeking. Everything
336/// else asks for the whole body, and the bytes are already in memory, so the
337/// implementation is a slice rather than a streaming reader.
338fn serve(entry: CachedMedia, headers: &HeaderMap) -> Response {
339    let total = entry.bytes.len() as u64;
340    let mut response_headers = HeaderMap::new();
341    response_headers.insert(
342        header::CONTENT_TYPE,
343        HeaderValue::from_str(&entry.content_type)
344            .unwrap_or(HeaderValue::from_static("application/octet-stream")),
345    );
346    response_headers.insert(
347        header::CACHE_CONTROL,
348        HeaderValue::from_str(&format!("public, max-age={CLIENT_CACHE_SECS}, immutable"))
349            .unwrap_or(HeaderValue::from_static("public")),
350    );
351    response_headers.insert(
352        header::X_CONTENT_TYPE_OPTIONS,
353        HeaderValue::from_static("nosniff"),
354    );
355    // Belt and braces for any web client that ever renders these: the asset is
356    // inert even if a future content type slips past the allowlist.
357    response_headers.insert(
358        header::CONTENT_SECURITY_POLICY,
359        HeaderValue::from_static("default-src 'none'; sandbox"),
360    );
361    response_headers.insert(header::ACCEPT_RANGES, HeaderValue::from_static("bytes"));
362
363    match parse_range(headers, total) {
364        Some(Ok((start, end))) => {
365            let slice = entry.bytes.slice(start as usize..=(end as usize));
366            response_headers.insert(
367                header::CONTENT_RANGE,
368                HeaderValue::from_str(&format!("bytes {start}-{end}/{total}"))
369                    .unwrap_or(HeaderValue::from_static("bytes */0")),
370            );
371            (
372                StatusCode::PARTIAL_CONTENT,
373                response_headers,
374                Body::from(slice),
375            )
376                .into_response()
377        }
378        Some(Err(())) => {
379            response_headers.insert(
380                header::CONTENT_RANGE,
381                HeaderValue::from_str(&format!("bytes */{total}"))
382                    .unwrap_or(HeaderValue::from_static("bytes */0")),
383            );
384            (StatusCode::RANGE_NOT_SATISFIABLE, response_headers).into_response()
385        }
386        None => (StatusCode::OK, response_headers, Body::from(entry.bytes)).into_response(),
387    }
388}

390/// `bytes=a-b` / `bytes=a-` / `bytes=-n` → an inclusive byte range.
391///
392/// `None` means no (or an unhandled) range header — serve the whole body, which
393/// is always a legal answer. `Some(Err(()))` means a range that cannot be
394/// satisfied. Multi-range requests fall into the `None` arm deliberately: a
395/// multipart response buys nothing for a seeking video player.
396fn parse_range(headers: &HeaderMap, total: u64) -> Option<Result<(u64, u64), ()>> {
397    let raw = headers.get(header::RANGE)?.to_str().ok()?;
398    let spec = raw.trim().strip_prefix("bytes=")?.trim();
399    if spec.contains(',') {
400        return None;
401    }
402    if total == 0 {
403        return Some(Err(()));
404    }
405    let (start_txt, end_txt) = spec.split_once('-')?;
406    let (start, end) = match (start_txt.trim(), end_txt.trim()) {
407        ("", "") => return Some(Err(())),
408        // `bytes=-500`: the LAST 500 bytes.
409        ("", suffix) => {
410            let len: u64 = suffix.parse().ok()?;
411            if len == 0 {
412                return Some(Err(()));
413            }
414            (total.saturating_sub(len), total - 1)
415        }
416        (start, "") => (start.parse().ok()?, total - 1),
417        (start, end) => (start.parse().ok()?, end.parse::<u64>().ok()?.min(total - 1)),
418    };
419    if start > end || start >= total {
420        return Some(Err(()));
421    }
422    Some(Ok((start, end)))
