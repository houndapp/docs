3use sea_orm::{
4    DatabaseTransaction, DbErr, SqliteTransactionMode, TransactionOptions, TransactionTrait,
5};
6
7use crate::api::Api;
8
9impl Api {
10    /// Begin a transaction that may write.
11    ///
12    /// Invariant: every transaction that writes MUST start here, never with a
13    /// bare `begin()`. On SQLite this issues `BEGIN IMMEDIATE`: the write lock
14    /// is taken up front, so no two writers can each hold a read lock on a
15    /// table and then both try to upgrade it. Under the test harness's
16    /// shared-cache `sqlite::memory:` pool, locks are per-table and that
17    /// upgrade is not retried — sqlx's `sqlite3_unlock_notify` wait detects
18    /// the cycle and fails with SQLITE_LOCKED ("database is deadlocked"). With
19    /// IMMEDIATE, a second writer instead waits in `BEGIN` holding nothing.
20    ///
21    /// Postgres ignores the mode; this is a plain `BEGIN` there.
22    ///
23    /// Deferred `begin()` is reserved for read-only transactions; the only one
24    /// is `Api::run_search`'s Postgres-only `SET LOCAL` wrapper.
25    pub(crate) async fn write_txn(&self) -> Result<DatabaseTransaction, DbErr> {
26        self.db
27            .begin_with_options(TransactionOptions {
28                sqlite_transaction_mode: Some(SqliteTransactionMode::Immediate),
29                ..Default::default()
30            })
31            .await
32    }
33}

49    /// Read a message row, hold the read across a yield, then write the row.
50    async fn read_then_write(api: Api, message_id: Uuid) -> Result<(), sea_orm::DbErr> {
51        let txn = api.write_txn().await?;
52        let row = community_space_message::Entity::find_by_id(message_id)
53            .one(&txn)
54            .await?
55            .expect("message exists");
56        // Lets the other task take its own read before either writes.
57        tokio::time::sleep(Duration::from_millis(50)).await;
58        community_space_message::ActiveModel {
59            search_text: Set(format!("{} rewritten", row.search_text)),
60            ..row.into()
61        }
62        .update(&txn)
63        .await?;
64        txn.commit().await
65    }
66
67    /// Two concurrent read→write transactions on `community_space_message`,
68    /// different rows. Under a deferred `BEGIN` each takes a shared-cache table
69    /// read lock, then each upgrade waits on the other's read lock and one
70    /// fails with SQLITE_LOCKED ("database is deadlocked"). `write_txn`'s
71    /// IMMEDIATE begin serialises them instead.
72    #[tokio::test]
73    async fn concurrent_read_then_write_transactions_do_not_deadlock() {
74        let (api, server) = crate::api::test().await;
75        let (_user, token, community) = setup(&api, &server).await;
76
77        let mut ids = Vec::new();
78        for contents in ["first", "second"] {
79            let msg: CompleteCommunitySpaceMessage = server
80                .post(&format!("/space/{}/message", community.default_space))
81                .authorization_bearer(&token)
82                .json(&PostCommunitySpaceMessageRequest {
83                    contents: contents.into(),
84                    attachments: vec![],
85                })
86                .await
87                .json();
88            ids.push(msg.id);
89        }
90
91        let a = tokio::spawn(read_then_write(api.clone(), ids[0]));
92        let b = tokio::spawn(read_then_write(api.clone(), ids[1]));
93        let (a, b) = tokio::time::timeout(Duration::from_secs(10), async {
94            (a.await.unwrap(), b.await.unwrap())
95        })
96        .await
97        .expect("neither writer wedged");
98        a.expect("first writer");
99        b.expect("second writer");
100    }
