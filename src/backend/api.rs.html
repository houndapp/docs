33/// What [`serve`] hands back.
34///
35/// The realtime WebTransport leg is **bound but not accepting**: binding early is
36/// what makes a port conflict a startup failure and puts the self-signed
37/// certificate hash into the advertisement, while accepting is deliberately
38/// deferred. The HTTP leg is fenced by its server future not being polled yet; a
39/// WebTransport peer has no such fence, so [`realtime::wt::spawn`] runs only once
40/// migrations and provisioning are done.
41pub(crate) struct Serving {
42    pub(crate) api: Api,
43    pub(crate) wt_leg: Option<realtime::wt::WtLeg>,
44    pub(crate) server: axum::serve::Serve<tokio::net::TcpListener, axum::Router, axum::Router>,
45}
46
47pub(crate) async fn serve(cli: &super::Cli) -> eyre::Result<Serving> {
48    // Bound before the `Api` exists: a self-signed endpoint mints its
49    // certificate here, and its hash has to be in the advertisement the `Api`
50    // is constructed with.
51    let wt_leg = realtime::wt::configure(cli).await?;
52
53    let api = Api {
54        inner: Arc::new(Inner {
55            auth: auth::Auth::from_cli(cli)?,
56            db: sea_orm::Database::connect(&cli.database_url.to_string()).await?,
57            realtime: realtime::hub::Realtime::new(
58                wt_leg
59                    .as_ref()
60                    .map(realtime::wt::WtLeg::config)
61                    .unwrap_or_default(),
62            ),
63            storage: storage::Storage::from_cli(cli)?,
64            public_url: cli.public_url.trim_end_matches('/').to_owned(),
65            embed_media_cache: embed::media::MediaCache::new(),
66            embed_debug_limiter: embed::DebugRateLimiter::default(),
67            upload_max_bytes: cli.upload_max_bytes,
68            relay_url: cli.relay_url.to_string(),
69            relay_hmac_secret: cli.relay_hmac_secret.clone(),
70            // Empty string (the chart's default when the feature is off) and
71            // "unset" are the same thing — see `Cli::metrics_import_url`.
72            metrics_import_url: cli
73                .metrics_import_url
74                .as_deref()
75                .filter(|s| !s.is_empty())
76                .map(str::parse)
77                .transpose()?,
78            metrics_import_client: reqwest::Client::new(),
79            sfu_nodes: RwLock::new(sfu_registry::seed(cli.sfu_node_id.clone())),
80            advertised_sfu: RwLock::new(None),
81            voice_participants: RwLock::new(HashMap::new()),
82            voice_eviction_generations: RwLock::new(HashMap::new()),
83            control_calls: RwLock::new(HashMap::new()),
84            metrics_handle: metrics::handle(),
85            #[cfg(test)]
86            _test_storage_dir: tempfile::tempdir()?,
87        }),
88    };
89
90    let (axum_router, _) = router()
91        .nest("/internal", internal::router().into())
92        .with_state(api.clone())
93        .split_for_parts();
94
95    // Protocol-upgrade routes are plain axum (no OpenAPI response body to
96    // document), merged after the OpenAPI split.
97    let mut axum_router = axum_router
98        .merge(control::router().with_state(api.clone()))
99        .merge(realtime::ws::router().with_state(api.clone()))
100        .merge(embed::media_router().with_state(api.clone()));
101
102    axum_router = mount_uploads(axum_router, &api.storage);
103
104    // `route_layer` (not `layer`): needs `MatchedPath` populated, which only
105    // happens once a route inside this router has matched — see
106    // `metrics::track_http_requests`. Applied after every route above is in
107    // place so it covers all of them, including the /uploads fallback.
108    let mut axum_router =
109        axum_router.route_layer(axum::middleware::from_fn(metrics::track_http_requests));
110
111    if cli.environment.is_dev() {
112        axum_router = axum_router.layer(tower_http::cors::CorsLayer::very_permissive());
113    } else {
114        let origins: Vec<http::HeaderValue> = cli
115            .cors_origins
116            .iter()
117            .map(|o| o.parse().expect("invalid CORS origin"))
118            .collect();
119        axum_router = axum_router
120            .layer(
121                tower_http::cors::CorsLayer::new()
122                    .allow_origin(origins)
123                    .allow_methods([
124                        http::Method::GET,
125                        http::Method::POST,
126                        http::Method::PUT,
127                        http::Method::DELETE,
128                        http::Method::OPTIONS,
129                    ])
130                    .allow_headers([http::header::AUTHORIZATION, http::header::CONTENT_TYPE])
131                    .allow_credentials(true),
132            )
133            .layer(tower_http::decompression::RequestDecompressionLayer::new())
134            .layer(tower_http::compression::CompressionLayer::new());
135    }
136
137    axum_router = axum_router
138        .layer(
139            tower_http::sensitive_headers::SetSensitiveRequestHeadersLayer::new(std::iter::once(
140                http::header::AUTHORIZATION,
141            )),
142        )
143        .layer(tower_http::trace::TraceLayer::new_for_http());
144
145    tracing::info!(host=%cli.host, port=%cli.port, "serving http api");
146    let server = axum::serve(
147        tokio::net::TcpListener::bind(std::net::SocketAddr::from((cli.host, cli.port)))
148            .await
149            .unwrap(),
150        axum_router,
151    );
152
153    Ok(Serving {
154        api,
155        wt_leg,
156        server,
157    })
158}

160/// Serves stored objects at `/uploads`, but only for the local-disk backend:
161/// with S3 the object carries its own content-type and is fetched from the
162/// bucket's public base instead. Content-type here is `mime_guess`'s, by
163/// extension — the extensions `upload::mime_to_ext` mints are all mapped.
164fn mount_uploads(router: axum::Router, storage: &storage::Storage) -> axum::Router {
165    match storage.local_dir() {
166        Some(dir) => router.nest_service("/uploads", tower_http::services::ServeDir::new(dir)),
167        None => router.route(
168            "/uploads/{*path}",
169            axum::routing::any(|| async { axum::http::StatusCode::NOT_IMPLEMENTED }),
170        ),
171    }
172}

174pub(crate) fn router() -> utoipa_axum::router::OpenApiRouter<Api> {
175    utoipa_axum::router::OpenApiRouter::with_openapi(openapi())
176        .nest("/.well-known", well_known::router())
177        .nest("/auth", auth::router())
178        .nest("/community", community::router())
179        .nest("/embed", embed::router())
180        .nest("/instance", instance::router())
181        // Invite *redemption* is not scoped to a community — a caller holding a
182        // code does not yet know which one it opens — so it hangs off the root
183        // rather than under /community/{id}. Minting and listing stay there.
184        .nest("/invite", community::invite::router())
185        .nest("/message", message::router())
186        .nest("/profile", profile::router())
187        .nest("/realtime", realtime::router())
188        .nest("/search", search::router())
189        .nest("/space", space::router())
190        .nest("/upload", upload::router())
191        .nest("/user", user::router())
192}

241pub(crate) struct Inner {
242    auth: auth::Auth,
243    pub(crate) db: sea_orm::DatabaseConnection,
244    /// The realtime plane: connection registry, room routing, and the
245    /// WebTransport advertisement `/realtime/config` serves. In-memory and
246    /// per-process — see `src/api/realtime.rs` for the lifecycle contract.
247    pub(crate) realtime: realtime::hub::Realtime,
248    pub(crate) storage: storage::Storage,
249    /// Origin of the app this instance's shareable links point at, without a
250    /// trailing slash — see `Cli::public_url`. Today only invite URLs are built
251    /// from it; a link a human is handed is a deployment fact, so it is
252    /// configured once here rather than assembled by each client.
253    pub(crate) public_url: String,
254    /// In-memory LRU of proxied embed assets. The thing that keeps a hot embed
255    /// from turning into an amplifier aimed at the site it links — see
256    /// `api::embed::media`. Per-process and lossy by design: a miss is one
257    /// guarded refetch, never an error.
258    pub(crate) embed_media_cache: embed::media::MediaCache,
259    /// Per-user fixed-window counter for the embed debugger's uncached unfurls.
260    pub(crate) embed_debug_limiter: embed::DebugRateLimiter,
261    pub(crate) upload_max_bytes: u64,
262    /// iroh relay server URL returned to clients in the voice-join response.
263    ///
264    /// Clients use this to configure their iroh endpoint. Storing it here
265    /// (rather than hardcoding in clients) means changing the relay in
266    /// production requires only a backend redeploy, not a client release.
267    pub(crate) relay_url: String,
268    /// Shared HMAC secret used to sign relay tokens.
269    ///
270    /// Must match `RELAY_SECRET` on the SFU process. Stored separately from
271    /// `relay_url` so the URL can be logged freely while the secret is treated
272    /// as sensitive.
273    pub(crate) relay_hmac_secret: String,
274    /// CI metrics import target — see `api::internal::post_metrics_ci`.
275    /// `None` (default, unset `METRICS_IMPORT_URL`) disables the route
276    /// (503); when set, incoming Prometheus text is forwarded verbatim here
277    /// via `metrics_import_client`.
278    pub(crate) metrics_import_url: Option<url::Url>,
279    /// Shared HTTP client for forwarding to `metrics_import_url`. Built once
280    /// regardless of whether the feature is enabled (cheap, and avoids an
281    /// `Option<Client>` split) — mirrors `SteamWebApi`'s single persistent
282    /// client.
283    pub(crate) metrics_import_client: reqwest::Client,
284    /// Live SFU nodes, keyed by iroh node ID (hex-encoded public key) →
285    /// last-known registration state. Nodes register (and re-register every
286    /// ~30s) via `POST /internal/sfu/register`; see `api::sfu_registry` for
287    /// the liveness/selection/migrate-broadcast logic. Empty until the first
288    /// registration (or the `SFU_NODE_ID` env seed) lands.
289    pub(crate) sfu_nodes: RwLock<HashMap<String, sfu_registry::SfuNode>>,
290    /// The node ID last returned to a fresh `voice/join` / advertised to
291    /// `/control` sessions, tracked only to detect a change in
292    /// `api::sfu_registry::select_best`'s pick and trigger an `SfuMigrate`
293    /// broadcast — reads that need the *current* best node call
294    /// `Api::best_sfu_node()` instead, which recomputes liveness fresh.
295    pub(crate) advertised_sfu: RwLock<Option<String>>,
296    /// Active voice participants: space_id → (user_id → event)
297    pub(crate) voice_participants:
298        RwLock<HashMap<Uuid, HashMap<Uuid, hound_types::VoiceJoinEvent>>>,
299    /// Per-user eviction generation counter. Incremented on socket disconnect and on
300    /// voice join. The pending eviction task checks the generation matches before evicting,

302    pub(crate) voice_eviction_generations: RwLock<HashMap<Uuid, u64>>,
303    /// Authoritative call control-plane state, keyed by `space_id` (a "call" is a

306    ///
307    /// NOTE: this is in-memory and per-process — it does not survive a backend

312    /// Render handle for the process-wide Prometheus recorder — see

331impl utoipa::Modify for SecurityAddon {
332    fn modify(&self, openapi: &mut utoipa::openapi::OpenApi) {
333        let components = openapi.components.as_mut().unwrap();
334        components.add_security_scheme(
335            "refreshToken",
336            utoipa::openapi::security::SecurityScheme::Http(
337                utoipa::openapi::security::HttpBuilder::new()
338                    .scheme(utoipa::openapi::security::HttpAuthScheme::Bearer)
339                    .bearer_format("JWT")
340                    .description(Some("JWT refresh token used to obtain a new access token"))
341                    .build(),
342            ),
343        );
344        components.add_security_scheme(
345            "token",
346            utoipa::openapi::security::SecurityScheme::Http(
347                utoipa::openapi::security::HttpBuilder::new()
348                    .scheme(utoipa::openapi::security::HttpAuthScheme::Bearer)
349                    .bearer_format("JWT")
350                    .description(Some("JWT access token for authenticating API requests"))
351                    .build(),
352            ),
353        )
354    }
355}

529/// The body of a "this instance has no credentials for that feature" 501, in the
530/// same `{ error, code }` shape [`auth::jwt::AuthError`] uses.
531///
532/// A bodiless 501 is indistinguishable from one an ingress or proxy invented, so
533/// the client could only report it as `HTTP501` with an empty message — which is
534/// exactly how a deployment that simply had Steam switched off arrived in crash
535/// triage. `code` is the stable half the client branches on.
536fn feature_disabled(message: &str, code: &'static str) -> axum::Json<serde_json::Value> {
537    axum::Json(serde_json::json!({ "error": message, "code": code }))
538}
539
540/// A refusal that names the rule it broke, in the same `{ error, code }` shape
541/// [`auth::jwt::AuthError`] uses.
542///
543/// A status alone collapses distinct refusals into one unactionable answer: a
544/// bodiless 400 tells a user their edit failed but not which field to fix, and
545/// a bodiless 403 makes "the target outranks you" indistinguishable from "you
546/// lack the permission this route needs". `code` is the stable half a client
547/// branches on; `message` is the sentence, and never the client's to parse.
548fn coded(
549    status: axum::http::StatusCode,
550    message: &str,
551    code: &'static str,
552) -> axum::response::Response {
553    use axum::response::IntoResponse;
554    (
555        status,
556        axum::Json(serde_json::json!({ "error": message, "code": code })),
557    )
558        .into_response()
559}
560
561/// [`coded`] at 400 — a request the caller can fix by changing it.
562fn invalid(message: &str, code: &'static str) -> axum::response::Response {
563    coded(axum::http::StatusCode::BAD_REQUEST, message, code)
564}
565
566/// [`coded`] at 403 — a refusal that is not about the request's shape.
567fn forbidden(message: &str, code: &'static str) -> axum::response::Response {
568    coded(axum::http::StatusCode::FORBIDDEN, message, code)
569}

571impl axum::response::IntoResponse for ApiError {
572    fn into_response(self) -> axum::response::Response {
573        // Log level tracks the response class, not "an error occurred": routine
574        // 4xx (not-found, forbidden, bad request) are expected client outcomes and
575        // would otherwise spam ERROR on every miss; only a genuine 5xx (our own
576        // bug or a downstream dependency failure) is worth an ERROR-level page.
577        let message = self.to_string();
578        let response = match self {
579            Self::CommunitySpaceNotFound(_) => {
580                (axum::http::StatusCode::NOT_FOUND, ()).into_response()
581            }
582            Self::CommunityNotFound(_) => (axum::http::StatusCode::NOT_FOUND, ()).into_response(),
583            Self::NotInCommunity(_, _) => (axum::http::StatusCode::FORBIDDEN, ()).into_response(),
584            Self::InvalidMessageContents() | Self::TooManyMessageReferences { .. } => {
585                axum::http::StatusCode::UNPROCESSABLE_ENTITY.into_response()
586            }
587            Self::MessageNotFound(_) | Self::UploadNotFound(_) => {
588                (axum::http::StatusCode::NOT_FOUND, ()).into_response()
589            }
590            Self::Forbidden => (axum::http::StatusCode::FORBIDDEN, ()).into_response(),
591            Self::RoleNotFound(_) => (axum::http::StatusCode::NOT_FOUND, ()).into_response(),
592            Self::LastOrganiser => invalid(&message, "last_organiser"),
593            Self::NotAnOrganiser(_) => (axum::http::StatusCode::NOT_FOUND, ()).into_response(),
594            Self::RoleAboveYou => forbidden(&message, "role_above_you"),
595            Self::PermissionNotHeld => forbidden(&message, "permission_not_held"),
596            Self::CannotBanSelf => forbidden(&message, "cannot_ban_self"),
597            Self::CannotKickSelf => forbidden(&message, "cannot_kick_self"),
598            Self::BanReasonTooLong { .. } => invalid(&message, "ban_reason_too_long"),
599            Self::EmptyRoleName => invalid(&message, "empty_role_name"),
600            Self::RoleNameTooLong { .. } => invalid(&message, "role_name_too_long"),
601            Self::DuplicateOverwrite => invalid(&message, "duplicate_overwrite"),
602            Self::Banned => forbidden(&message, "banned"),
603            Self::NotAMember(_) => invalid(&message, "not_a_member"),
604            Self::InviteNotFound(_) => (axum::http::StatusCode::NOT_FOUND, ()).into_response(),
605            Self::InviteNotActive(_) => invalid(&message, "invite_not_active"),
606            Self::InviteExpiryTooLong { .. } => invalid(&message, "invite_expiry_too_long"),
607            Self::InvalidMaxUses { .. } => invalid(&message, "invalid_max_uses"),
608            Self::BadRequest => (axum::http::StatusCode::BAD_REQUEST, ()).into_response(),
609            Self::UserNotFound(_) => (axum::http::StatusCode::NOT_FOUND, ()).into_response(),
610            Self::RegistrationStateNotFound(_) | Self::LoginStateNotFound(_) => {
611                (axum::http::StatusCode::BAD_REQUEST, ()).into_response()
612            }
613            Self::EmptyCommunityName => invalid(&message, "empty_community_name"),
614            Self::CommunityNameTooLong { .. } => invalid(&message, "community_name_too_long"),
615            Self::CommunityDescriptionTooLong { .. } => {
616                invalid(&message, "community_description_too_long")
617            }
618            Self::InvalidDefaultSpace => invalid(&message, "invalid_default_space"),
619            Self::EmptySpaceName => invalid(&message, "empty_space_name"),
620            Self::SpaceNameTooLong { .. } => invalid(&message, "space_name_too_long"),
621            Self::DefaultSpace => invalid(&message, "default_space"),
622            Self::InstanceAlreadyProvisioned => {
623                (axum::http::StatusCode::CONFLICT, ()).into_response()
624            }
625            Self::UnsupportedMediaType(allowed) => (
626                axum::http::StatusCode::UNSUPPORTED_MEDIA_TYPE,
627                [(upload::ACCEPT_POST, upload::accept_post(allowed))],
628            )
629                .into_response(),
630            Self::PayloadTooLarge => {
631                (axum::http::StatusCode::PAYLOAD_TOO_LARGE, ()).into_response()
632            }
633            Self::MultipartError(_) => {
634                (axum::http::StatusCode::UNPROCESSABLE_ENTITY, ()).into_response()
635            }
636            Self::Jwt(error) => error.into_response(),
637            Self::SteamDisabled => (
638                axum::http::StatusCode::NOT_IMPLEMENTED,
639                feature_disabled(&message, "steamdisabled"),
640            )
641                .into_response(),
642            Self::Steam(auth::steam::SteamError::Rejected | auth::steam::SteamError::Banned) => {
643                (axum::http::StatusCode::UNAUTHORIZED, ()).into_response()
644            }
645            Self::Steam(auth::steam::SteamError::Upstream(_)) => {
646                (axum::http::StatusCode::BAD_GATEWAY, ()).into_response()
647            }
648            Self::IdentityAlreadyLinked => (axum::http::StatusCode::CONFLICT, ()).into_response(),
649            Self::LastLoginMethod => (axum::http::StatusCode::BAD_REQUEST, ()).into_response(),
650            Self::IdentityNotFound => (axum::http::StatusCode::NOT_FOUND, ()).into_response(),
651            Self::EmailDisabled => (
652                axum::http::StatusCode::NOT_IMPLEMENTED,
653                feature_disabled(&message, "emaildisabled"),
654            )
655                .into_response(),
656            Self::TooManyMagicRequests | Self::TooManyRequests() => {
657                (axum::http::StatusCode::TOO_MANY_REQUESTS, ()).into_response()
658            }
659            Self::InvalidMagicCode => (axum::http::StatusCode::UNAUTHORIZED, ()).into_response(),
660            Self::PasskeyCredentialCodec(..)
661            | Self::Database(_)
662            | Self::WebAuthn(_)
663            | Self::Storage(_) => {
664                (axum::http::StatusCode::INTERNAL_SERVER_ERROR, ()).into_response()
665            }
666        };
667
668        if response.status().is_server_error() {
669            tracing::error!(error=%message, "api error");
670        } else {
671            tracing::debug!(error=%message, "api error");
672        }
673
674        response
675    }
676}

696}

745#[cfg(test)]
746pub(crate) async fn test_with_options(options: TestOptions) -> (Api, axum::Router) {
747    let TestOptions {
748        metrics_import_url,
749        session_limits,
750        wt_leg,
751        database_url,
752    } = options;
753    static TRACING: std::sync::Once = std::sync::Once::new();
754
755    TRACING.call_once(|| {
756        crate::LogFormats::Pretty.init_tracing_subscriber(false, None);
757    });
758
759    // Generous, because this bounds how long a test waits for a pooled
760    // connection, not how long a query may take. nextest runs the suite with
761    // one process per test across every core, so a 100 ms budget is a function
762    // of machine load rather than of the code under test: it surfaces as
763    // `PoolTimedOut` in a different handful of tests on every run.
764    const TIMEOUT: std::time::Duration = std::time::Duration::from_secs(30);
765
766    let test_storage_dir = tempfile::tempdir().unwrap();
767
768    let api = Api {
769        inner: Arc::new(Inner {
770            auth: auth::test(),
771            storage: storage::Storage::local_test(test_storage_dir.path().to_owned()),
772            public_url: "https://app.test".into(),
773            embed_media_cache: embed::media::MediaCache::new(),
774            embed_debug_limiter: embed::DebugRateLimiter::default(),
775            upload_max_bytes: 8 * 1024 * 1024,
776            relay_url: "http://localhost:3340".into(),
777            relay_hmac_secret: "test-relay-secret".into(),
778            metrics_import_url,
779            metrics_import_client: reqwest::Client::new(),
780            realtime: realtime::hub::Realtime::with_limits(
781                wt_leg
782                    .as_ref()
783                    .map(realtime::wt::WtLeg::config)
784                    .unwrap_or_default(),
785                session_limits.unwrap_or_default(),
786            ),
787            sfu_nodes: RwLock::new(HashMap::new()),
788            advertised_sfu: RwLock::new(None),
789            voice_participants: RwLock::new(HashMap::new()),
790            voice_eviction_generations: RwLock::new(HashMap::new()),
791            control_calls: RwLock::new(HashMap::new()),
792            metrics_handle: metrics::handle(),
793            db: sea_orm::Database::connect(
794                sea_orm::ConnectOptions::new(
795                    database_url.unwrap_or_else(|| "sqlite::memory:".to_string()),
796                )
797                .min_connections(10)
798                .max_connections(10)
799                .connect_timeout(TIMEOUT)
800                .acquire_timeout(TIMEOUT)
801                .max_lifetime(TIMEOUT * 100)
802                .clone(),
803            )
804            .await
805            .unwrap(),
806            _test_storage_dir: test_storage_dir,
807        }),
808    };
809
810    <migration::Migrator as migration::MigratorTrait>::fresh(&api.db)
811        .await
812        .unwrap();
813    if sea_orm::ConnectionTrait::get_database_backend(&api.db) == sea_orm::DbBackend::Sqlite {
814        restore_sqlite_foreign_keys(&api.db).await;
815    }
816
817    // Mirrors `Cli::drive`: the WebTransport leg starts accepting only once the
818    // schema exists.
819    if let Some(leg) = wt_leg {
820        realtime::wt::spawn(leg, api.clone());
821    }
822
823    let (axum_router, _) = router()
824        .nest("/internal", internal::router().into())
825        .with_state(api.clone())
826        .split_for_parts();
827    let axum_router = axum_router
828        .merge(control::router().with_state(api.clone()))
829        .merge(realtime::ws::router().with_state(api.clone()))
830        .merge(embed::media_router().with_state(api.clone()));
831    let axum_router = mount_uploads(axum_router, &api.storage);
832    let axum_router =
833        axum_router.route_layer(axum::middleware::from_fn(metrics::track_http_requests));
834
835    (api, axum_router)
836}

838/// Re-enable `PRAGMA foreign_keys` on EVERY connection in the test pool.
839///
840/// `Migrator::fresh` drops the schema with foreign keys off and turns them back
841/// on afterwards, but it issues both pragmas through the *pool* — so the `OFF`
842/// and the `ON` can land on different connections, leaving one of the pool's
843/// connections with enforcement permanently disabled. `PRAGMA foreign_keys` is
844/// per-connection, so ~1 transaction in `max_connections` then silently skips
845/// every `ON DELETE CASCADE` (this is what made `reset_restores_default_layout`
846/// flake: the reset's space delete left its messages orphaned).
847///
848/// Acquiring the whole pool at once is what makes this total — issuing the
849/// pragma N times through the pool would just keep hitting the same connection.
850#[cfg(test)]
851async fn restore_sqlite_foreign_keys(db: &sea_orm::DatabaseConnection) {
852    use sea_orm::sqlx::Executor;
853
854    let pool = db.get_sqlite_connection_pool();
855    let mut held = Vec::new();
856    for _ in 0..pool.options().get_max_connections() {
857        let mut conn = pool.acquire().await.expect("acquiring a pooled connection");
858        (&mut *conn)
859            .execute("PRAGMA foreign_keys = ON")
860            .await
861            .expect("re-enabling foreign keys");
862        held.push(conn);
863    }
864}
