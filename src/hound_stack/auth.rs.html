1//! The host `hound-auth` page process — the browser-deferred auth pages
2//! (passkey sign-in/sign-up, Steam, email link) a dev client opens in the
3//! system browser, and the page every magic link points at
4//! (`MAGIC_BASE_URL`, `crate::backend`).
5//!
6//! A HOST process rather than the `hound-auth` compose container (which
7//! `compose.dev.yaml` still defines for the prod-like topology): the
8//! container meant a release cargo build inside docker on every `--fresh`,
9//! for a binary the host already builds under [`crate::cargo::STACK_SELECTION`].
10//! The HTML is `include_str!`'d, so the binary needs no working directory.
11//!
12//! Stateless: its only upstream is a fixed URL (`HOUND_API_URL`), so a
13//! backend restart does NOT restart it — only a rebuilt binary does.
14//!
15//! DETACHED like the backend and SFU beside it — see `crate::process`.
16
17use std::path::Path;
18use std::process::{Child, Command};
19use std::time::Duration;
20
21use eyre::{Context, Result};
22
23use crate::backend::env_or;
24use crate::process;
25use crate::wait::{Deadline, wait_for_http_or_exit};
26
27/// The page's port — the same one the container published, and what the
28/// backend's `RP_ORIGIN`/`MAGIC_BASE_URL` default to.
29pub const PORT: u16 = 3100;
30
31/// Bind address. NOT loopback, unlike the SFU: the opt-in `passkeys-front`
32/// Caddy container reaches this page from inside docker via
33/// `host.docker.internal` (see `tools/houndbot/compose.e2e-override.yaml`),
34/// which a 127.0.0.1 bind would refuse. Safe to widen: it is a plain HTTP
35/// page server — no media, and nothing it serves makes a LAN peer dial it.
36const LISTEN: &str = "0.0.0.0:3100";
37
38/// How long the page gets to answer after spawn. It binds one TCP listener
39/// and does no I/O first, so this is generous.
40const READY_BUDGET: Duration = Duration::from_secs(15);
41
42/// The page's root, as the host sees it.
43pub fn url() -> String {
44    format!("http://localhost:{PORT}/")
45}
46
47/// Loopback probe URL (IPv4 literal: `localhost` may resolve `::1` first,
48/// and the listener is IPv4-only).
49fn probe_url() -> String {
50    format!("http://127.0.0.1:{PORT}/")
51}
52
53/// Spawn the prebuilt `hound-auth` `binary` DETACHED. Output → `log_path`.
54///
55/// Both URLs are consumed by the PAGE in the user's browser, so they are
56/// host-reachable URLs. The passkeys mode (`//ui:compose:passkeys`) re-homes
57/// them onto the dotted dev domain via `PASSKEYS_API_URL` /
58/// `PASSKEYS_PUBLIC_URL` — the same variables `compose.dev.yaml`'s
59/// container reads.
60pub fn spawn(binary: &Path, root: &Path, log_path: &Path) -> Result<Child> {
61    let mut cmd = Command::new(binary);
62    cmd.current_dir(root)
63        .env(
64            "HOUND_API_URL",
65            env_or("PASSKEYS_API_URL", "http://localhost:8080"),
66        )
67        .env(
68            "PUBLIC_URL",
69            env_or("PASSKEYS_PUBLIC_URL", "http://localhost:3100"),
70        )
71        .env("LISTEN", LISTEN)
72        .env("RUST_LOG", "hound_auth=info,tower_http=info");
73
74    process::spawn_detached(&mut cmd, log_path)
75        .with_context(|| format!("spawning hound-auth ({})", binary.display()))
76}
77
78/// Wait for the freshly spawned page to answer (any HTTP response counts),
79/// bailing early if the process exits first.
80pub async fn wait_ready(child: &mut Child) -> Result<()> {
81    wait_for_http_or_exit(
82        &reqwest::Client::new(),
83        &probe_url(),
84        child,
85        &Deadline::after(READY_BUDGET, Duration::from_millis(200)),
86    )
87    .await
88    .context("hound-auth never became reachable")
89}
90
91/// Whether something answers on the page's port — the liveness half of
92/// the reuse probe in `crate::lifecycle`.
93pub async fn is_healthy(client: &reqwest::Client) -> bool {
94    client
95        .get(probe_url())
96        .timeout(Duration::from_secs(5))
97        .send()
98        .await
99        .is_ok()
100}
