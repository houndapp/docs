1//! Quiet cargo builds, separated from runtime logs. Every binary the harness
2//! runs (backend, SFU, auth page, houndbot, snapdiff, the nextest test
3//! binaries) is built HERE first — full compiler output captured into the
4//! run's `build.log`, surfaced only on failure — and then executed directly,
5//! so each runtime artifact (`backend.log`, bot logs, test output) contains
6//! runtime lines only, never pages of "Compiling…" chatter.
7

16/// A nested `cargo` invocation from `root`, with the environment `cargo run`
17/// injected into THIS process scrubbed off it.
18///
19/// `hound-stack` is itself launched by `cargo run`, which sets
20/// `CARGO_MANIFEST_DIR`, `CARGO_PKG_*` and friends for the child. A nested
21/// cargo inherits them and evaluates every build script's
22/// `rerun-if-env-changed` against them — and `ring`'s build.rs declares
23/// `CARGO_MANIFEST_DIR`, so under the harness it always sees a value the
24/// shell never sets. Measured: 28 crates (ring, rustls, quinn, reqwest, iroh…)
25/// recompiled on every harness run, then again from the next shell build,
26/// and each relink of `backend` tripped the staleness check into restarting
27/// a backend that had not changed. Everything under `CARGO_*` that is cargo
28/// CONFIG (`CARGO_HOME`, `CARGO_TARGET_DIR`, `CARGO_BUILD_JOBS`,
29/// `CARGO_INCREMENTAL`, `CARGO_TERM_COLOR`, `CARGO_LOG`, …) is kept.
30pub fn cargo(root: &Path) -> Command {
31    let mut cmd = Command::new("cargo");
32    cmd.current_dir(root);
33    for (name, _) in std::env::vars_os() {
34        let Some(name) = name.to_str() else { continue };
35        let injected = name == "CARGO"
36            || name == "OUT_DIR"
37            || name.starts_with("CARGO_MANIFEST_")
38            || name.starts_with("CARGO_PKG_")
39            || name == "CARGO_CRATE_NAME"
40            || name == "CARGO_BIN_NAME"
41            || name == "CARGO_PRIMARY_PACKAGE"
42            || name == "CARGO_TARGET_TMPDIR";
43        if injected {
44            cmd.env_remove(name);
45        }
46    }
47    cmd
48}

50/// The ONE package selection every binary this harness builds is compiled
51/// under — `hound-testkit` (the scenario suite nextest builds separately),
52/// `hound-sfu`, `backend`, `hound-auth` (the host auth page) and `hound-bot`
53/// (the visual flow's peer).
54///
55/// Fixed, and shared by every call site, because a `-p` set is a feature
56/// resolve. A second, different set means cargo unifies features differently,
57/// compiles a second copy of every dependency whose resolved features moved,
58/// AND re-uplifts `target/debug/backend` with a fresh mtime — which
59/// `crate::lifecycle`'s staleness check reads as "the backend changed",
60/// restarting a perfectly good one. So `e2e`, `visual` and `up` all pass
61/// THIS, and vary only the `--bin` filter.
62///
63/// `snapdiff` is deliberately outside it: it shares no engine/media
64/// dependencies with these, so folding it in would only widen the resolve
65/// (see `visual`'s [`build_bin`] call).
66pub const STACK_SELECTION: &[&str] = &[
67    "hound-testkit",
68    "hound-sfu",
69    "backend",
70    "hound-auth",
71    "hound-bot",
72];
73
74/// Where `cargo build` puts binaries: `CARGO_TARGET_DIR` if set, else the
75/// workspace-root `target/`.
76pub fn target_debug_dir(root: &Path) -> PathBuf {
77    std::env::var_os("CARGO_TARGET_DIR")
78        .map(PathBuf::from)
79        .unwrap_or_else(|| root.join("target"))
80        .join("debug")
81}

83/// Run `cmd` with all output captured into a `build_log` section titled
84/// `title`. Silent on success; on failure prints the log tail inline and
85/// bails with a pointer at the full file.
86pub fn run_captured(mut cmd: Command, build_log: &Path, title: &str) -> Result<()> {
87    let mut f = logs::append_section(build_log, title)?;
88    let out = cmd.output().with_context(|| title.to_owned())?;
89    f.write_all(&out.stdout)?;
90    f.write_all(&out.stderr)?;
91    if !out.status.success() {
92        logs::print_tail(title, build_log, 40);
93        bail!(
94            "{title} exited {} — full output in {}",
95            out.status,
96            build_log.display()
97        );
98    }
99    Ok(())
100}

102/// `cargo build -q --message-format=short -p package --bin bin`, output to
103/// `build_log`. Returns the built binary's path for the caller to exec.
104///
105/// The single-package escape hatch, for a binary that shares no dependency
106/// graph worth unifying with [`STACK_SELECTION`] (`snapdiff`). Anything the
107/// stack itself runs goes through [`build_bins`] instead.
108pub fn build_bin(root: &Path, package: &str, bin: &str, build_log: &Path) -> Result<PathBuf> {
109    let mut cmd = cargo(root);
110    cmd.args([
111        "build",
112        "-q",
113        "--message-format=short",
114        "-p",
115        package,
116        "--bin",
117        bin,
118    ]);
119    run_captured(
120        cmd,
121        build_log,
122        &format!("cargo build -p {package} --bin {bin}"),
123    )?;
124    Ok(target_debug_dir(root).join(bin))
125}

127/// ONE `cargo build` for several binaries: `-p` for every package in
128/// `packages`, `--bin` for every name in `bins`. Returns the built paths in
129/// `bins` order.
130///
131/// Why one invocation and not a [`build_bin`] per binary: each distinct `-p`
132/// set is a distinct feature resolve, and a crate already compiled under one
133/// resolve is recompiled under the next. Every caller therefore passes
134/// [`STACK_SELECTION`] and varies only `bins` — a package contributing no
135/// `--bin` (e.g. `hound-testkit`, whose test binaries nextest builds itself)
136/// still belongs in `packages`, purely so this build resolves the features
137/// that one did. Measured on a cold scratch target dir, after `cargo nextest
138/// run --no-run -p hound-testkit --profile e2e`: 478 units recompiled with
139/// `-p hound-testkit` in the selection, 482 without (the extra four — objc2,
140/// block2, dispatch2, brotli-decompressor — are ones nextest had already
141/// built under features this selection otherwise fails to reproduce). Adding
142/// `-p hound-bot` on top measured 478 as well: it is already inside the same
143/// dependency graph, so carrying it costs nothing and buys the visual flow's
144/// `houndbot` the same resolve.
145///
146/// What must NOT join this call is `--tests`: it would pull in `backend`'s
147/// and `hound-sfu`'s own (heavy) test binaries on every backend edit.
148pub fn build_bins(
149    root: &Path,
150    packages: &[&str],
151    bins: &[&str],
152    build_log: &Path,
153) -> Result<Vec<PathBuf>> {
154    let mut cmd = cargo(root);
155    cmd.args(["build", "-q", "--message-format=short"]);
156    for package in packages {
157        cmd.args(["-p", package]);
158    }
159    for bin in bins {
160        cmd.args(["--bin", bin]);
161    }
162    let title = format!(
163        "cargo build {} {}",
164        packages
165            .iter()
166            .map(|p| format!("-p {p}"))
167            .collect::<Vec<_>>()
168            .join(" "),
169        bins.iter()
170            .map(|b| format!("--bin {b}"))
171            .collect::<Vec<_>>()
172            .join(" ")
173    );
174    run_captured(cmd, build_log, &title)?;
175    let debug = target_debug_dir(root);
176    Ok(bins.iter().map(|bin| debug.join(bin)).collect())
177}

179/// The test binaries nextest would run for `package`/`profile`, from
180/// `cargo nextest list --list-type binaries-only --message-format json`
181/// (`rust-binaries.<id>.binary-path`). Assumes they are already built — the
182/// harness's `--no-run` pass does that.
183///
184/// Only called when there is something to do with the paths (see
185/// `crate::codesign::enabled`): it is an extra cargo invocation.
186pub fn nextest_binaries(root: &Path, package: &str, profile: &str) -> Result<Vec<PathBuf>> {
187    let out = cargo(root)
188        .args([
189            "nextest",
190            "list",
191            "--list-type",
192            "binaries-only",
193            "--message-format",
194            "json",
195            "-p",
196            package,
197            "--profile",
198            profile,
199        ])
200        .output()
201        .context("cargo nextest list")?;
202    if !out.status.success() {
203        bail!(
204            "cargo nextest list exited {}: {}",
205            out.status,
206            String::from_utf8_lossy(&out.stderr).trim()
207        );
208    }
209    let listing: serde_json::Value =
210        serde_json::from_slice(&out.stdout).context("parsing cargo nextest list JSON")?;
211    let binaries = listing["rust-binaries"]
212        .as_object()
213        .ok_or_else(|| eyre::eyre!("cargo nextest list JSON has no rust-binaries object"))?;
214    Ok(binaries
215        .values()
216        .filter_map(|b| b["binary-path"].as_str())
217        .map(PathBuf::from)
218        .collect())
219}
