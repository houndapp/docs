1//! `hound-stack e2e`: ensure the stack is up, run the `hound-testkit`
2//! scenario suite (`cargo nextest run -p hound-testkit --profile e2e`)
3//! against it, collect artifacts. Mirrors the retired
4//! `tools/houndbot/e2e.sh`. Extra CLI args forward to `cargo nextest run`
5//! (e.g. an `-E` filter for a fast local iteration loop against one scenario
6//! file).
7//!
8//! The stack is REUSED when one is already live and LEFT RUNNING afterwards
9//! (see `crate::lifecycle` for the contract). `--fresh` wipes and rebuilds it
10//! first — the landing gate's shape, and the answer to any scenario that
11//! needs a pristine database; `--teardown` takes it down after the run, which
12//! is what CI wants.
13//!
14//! The test binaries are compiled in a separate quiet pass
15//! (`cargo nextest run --no-run`, output → `build.log`) before the real run,
16//! so `nextest.log` holds test results, not compiler chatter.
17//!
18//! # Remote-target mode (`--target <url>` / `E2E_TARGET_URL`)
19//!
20//! Points the SAME suite at a DEPLOYED hound (staging, or a new cluster
21//! before DNS cuts over) instead of the local stack: no compose/backend
22//! bring-up, no teardown, no compose artifacts, and no local `hound-sfu`
23//! build. Never the default — the plain `mise run e2e:call` path is
24//! byte-for-byte what it always was.
25//!
26//! Two things the operator owns in this mode:
27//!
28//! - **Mail.** Scenarios authenticate by magic link, read out of Mailpit —
29//!   there is no other auth path here, and none is added. So the target
30//!   backend must deliver its mail somewhere this process can read as
31//!   `MAILPIT_URL`; for a cluster that means port-forwarding its Mailpit
32//!   (`kubectl port-forward -n hound svc/mailpit 8025:8025`, then
33//!   `MAILPIT_URL=http://localhost:8025`). A deployment wired to a real SMTP
34//!   provider can't run these scenarios at all.
35//! - **DNS.** `E2E_RESOLVE=host:port:ip[,…]` (curl `--resolve` shape) pins
36//!   hostnames for the run, so a pre-cutover cluster is reachable at its real
37//!   hostnames — TLS SNI/certs still validate. It reaches every `reqwest`
38//!   client (auth + REST + Mailpit), NOT the realtime/media planes; see
39//!   `hound_testkit::remote` for why, and for the `/etc/hosts` alternative
40//!   that does cover them.

52/// Durable-artifact override — the nextest run log + junit report,
53/// build/backend/sfu/compose logs — instead of the default throwaway temp
54/// dir (still left in place for local inspection either way).
55pub const OUT_DIR_ENV: &str = "E2E_OUT_DIR";
56
57/// Remote-target URL, the env equivalent of `--target` (which wins).
58pub const TARGET_URL_ENV: &str = "E2E_TARGET_URL";
59
60const INFRA_FAILURE: i32 = 2;
61
62/// The nextest profile every phase of this flow selects — the `--no-run`
63/// build, the binary listing that feeds codesigning, and the run itself must
64/// agree, or they are three different builds.
65const NEXTEST_PROFILE: &str = "e2e";
66
67pub struct E2eOptions {
68    /// Tear any existing stack down (volumes included) and rebuild before
69    /// running.
70    pub fresh: bool,
71    /// Tear the stack down after the run instead of leaving it up.
72    pub teardown: bool,
73    /// Run against a DEPLOYED hound at this base URL instead of a local
74    /// stack — no bring-up, no teardown (see the module doc).
75    pub target: Option<String>,
76    /// Forwarded verbatim to `cargo nextest run`.
77    pub nextest_args: Vec<String>,
78}
79
80pub async fn run(root: PathBuf, opts: E2eOptions) -> Result<i32> {
81    let E2eOptions {
82        fresh,
83        teardown,
84        target,
85        nextest_args: extra_nextest_args,
86    } = opts;
87    let workdir = artifacts::resolve(OUT_DIR_ENV)?;
88    info!(workdir = %workdir.display(), "artifact dir");
89    let build_log = workdir.join("build.log");
90
91    // Remote mode is opt-in and explicit: `--target` first, then the env
92    // equivalent (an empty value is "unset", so an unset CI variable doesn't
93    // silently flip the mode).
94    let target = target
95        .or_else(|| std::env::var(TARGET_URL_ENV).ok())
96        .filter(|t| !t.trim().is_empty());
97    let remote = target.is_some();
98
99    let backend_url = target.clone().unwrap_or_else(|| {
100        std::env::var("BACKEND_URL").unwrap_or_else(|_| "http://localhost:8080".into())
101    });
102    let mailpit_url =
103        std::env::var("MAILPIT_URL").unwrap_or_else(|_| "http://localhost:8025".into());
104
105    // The pins themselves reach the suite by plain env inheritance; what
106    // happens here is validation. The suite warns and falls back to real DNS
107    // on a malformed spec, which in remote mode means every scenario quietly
108    // talks to whatever the hostname resolves to today — the one outcome this
109    // mode exists to prevent, so the harness refuses to start instead.
110    if let Ok(spec) = std::env::var(hound_testkit::remote::RESOLVE_ENV) {
111        let pins = hound_testkit::remote::parse_resolve(&spec)
112            .with_context(|| format!("parsing {}", hound_testkit::remote::RESOLVE_ENV))?;
113        info!(?pins, "DNS pins for this run (REST plane only)");
114    }
115    if remote {
116        info!(%backend_url, %mailpit_url, "remote target — skipping stack bring-up");
117    }
118
119    // Build the suite before the stack comes up: a compile failure then
120    // costs nothing to tear down, and the containers don't sit idle through
121    // a long build.
122    info!("building the scenario suite (output → build.log)");
123    let mut build_cmd = cargo::cargo(&root);
124    build_cmd.args([
125        "nextest",
126        "run",
127        "--no-run",
128        "-p",
129        "hound-testkit",
130        "--profile",
131        NEXTEST_PROFILE,
132    ]);
133    cargo::run_captured(
134        build_cmd,
135        &build_log,
136        "cargo nextest run --no-run (suite build)",
137    )?;
138    sign_test_binaries(&root, &build_log)?;
139
140    // All three host binaries in ONE cargo invocation (see
141    // `lifecycle::build_stack_bins`). `hound-sfu` is needed twice over: the
142    // stack runs one on the host, and the seamless-rollout scenario spawns a
143    // SECOND process from the same binary (fresh ports, a short drain
144    // deadline — `crates/test/testkit/src/sfu_process.rs`), which is why its
145    // path is exported to the run below. Skipped entirely in remote mode:
146    // nothing here owns the target's backend, SFU fleet or auth page.
147    let bins = if remote {
148        None
149    } else {
150        Some(lifecycle::build_stack_bins(&root, &build_log)?)
151    };
152
153    let mut stack = match &bins {
154        None => None, // remote
155        Some(bins) => {
156            let paths = StackPaths::discover(root.clone());
157            match LiveStack::ensure_up(paths, &workdir, bins, &backend_url, &mailpit_url, fresh)
158                .await
159            {
160                Ok(stack) => Some(stack),
161                Err(e) => {
162                    eprintln!("[e2e] stack bring-up failed: {e:#}");
163                    return Ok(INFRA_FAILURE);
164                }
165            }
166        }
167    };
168
169    info!("running the hound-testkit scenario suite (output → nextest.log)");
170    let mut cmd = cargo::cargo(&root);
171    cmd.args([
172        "nextest",
173        "run",
174        "-p",
175        "hound-testkit",
176        "--profile",
177        NEXTEST_PROFILE,
178        // One scenario's failure must not cancel the rest — a full
179        // per-test result set is what makes the CI artifact diagnosable.
180        "--no-fail-fast",
181    ])
182    .args(&extra_nextest_args)
183    .env("BACKEND_URL", &backend_url)
184    .env("MAILPIT_URL", &mailpit_url);
185    // The rollout scenario's second SFU: same binary, same relay/secret as
186    // the stack's own host SFU, so both register with this backend.
187    if let Some(bins) = &bins {
188        cmd.env("HOUND_SFU_BIN", &bins.sfu)
189            .env("HOUND_SFU_RELAY_SECRET", crate::backend::RELAY_SECRET_DEV)
190            .env("HOUND_SFU_RELAY_URL", crate::backend::RELAY_URL_DEV);
191    }
192    if remote {
193        // The suite's own local-only gate (`hound_testkit::remote`) — a
194        // runtime skip with a printed reason, so remote mode stays one
195        // command and the skipped scenario still appears in the report.
196        cmd.env(hound_testkit::remote::REMOTE_ENV, "1");
197    }
198    let nextest_log = workdir.join("nextest.log");
199    let status = run_tee(cmd, &nextest_log).context("running cargo nextest")?;
200
201    let junit_src = root.join("target/nextest/e2e/junit.xml");
202    if junit_src.exists() {
203        let _ = std::fs::copy(&junit_src, workdir.join("junit.xml"));
204    }
205
206    // Propagate nextest's own exit code (0 pass, its own nonzero on
207    // failures) — the same contract the old shell harness had via
208    // `set -o pipefail`.
209    let code = status.code().unwrap_or(1);
210    // Both host processes' logs, because a scenario failure is as often the
211    // media path as the API. Each may belong to an EARLIER invocation's
212    // workdir (a reused stack), so the paths come from the running
213    // processes' state file, not from this run's artifact dir.
214    if code != 0 && !remote {
215        let recorded = state::load(&root);
216        let backend_log = stack
217            .as_ref()
218            .and_then(|s| s.backend_log_path().map(Path::to_owned))
219            .or(recorded.backend.map(|r| r.log))
220            .unwrap_or_else(|| workdir.join("backend.log"));
221        logs::print_tail("backend log", &backend_log, 100);
222        let sfu_log = stack
223            .as_ref()
224            .and_then(|s| s.sfu_log_path().map(Path::to_owned))
225            .or(recorded.sfu.map(|r| r.log))
226            .unwrap_or_else(|| workdir.join("sfu.log"));
227        logs::print_tail("sfu log", &sfu_log, 100);
228    }
229
230    match stack.as_mut() {
231        Some(stack) if teardown => stack.teardown(),
232        Some(_) => info!("stack left up — reuse it, or tear it down with `hound-stack down`"),
233        None => {}
234    }
235
236    info!(code, workdir = %workdir.display(), "e2e suite finished (workdir left in place)");
237    Ok(code)
238}

240/// Give this run's test binaries the machine's stable code signature, if one
241/// is configured (see `crate::codesign` for why the suite in particular needs
242/// it). The listing costs an extra cargo invocation, so it only happens when
243/// there is something to sign with.
244fn sign_test_binaries(root: &Path, build_log: &Path) -> Result<()> {
245    if !codesign::enabled() {
246        return Ok(());
247    }
248    for bin in cargo::nextest_binaries(root, "hound-testkit", NEXTEST_PROFILE)? {
249        codesign::sign(&bin, build_log)?;
250    }
251    Ok(())
