1//! Optional stable code signature for the binaries this harness builds
2//! (macOS only, opt-in).
3//!
4//! # Why
5//!
6//! Every Mach-O cargo produces is ad-hoc signed by the linker: no team, no
7//! stable identifier, and a designated requirement that is a bare `cdhash`
8//! of the binary's own bytes. macOS keys its permission grants on that
9//! requirement, so a rebuild — any rebuild, one changed line — produces a
10//! different cdhash, a different "application" as far as TCC is concerned,
11//! and every grant the previous binary held is void.
12//!
13//! The grant that bites this harness is Local Network privacy: the scenario
14//! suite's test client runs an iroh endpoint that binds `0.0.0.0`, and macOS
15//! gates such a socket's non-loopback traffic behind a per-application grant.
16//! Ad-hoc signed, each run is a new application — either a fresh prompt, or
17//! (unattended) silent denial and a suite that fails on "no route" in ways
18//! that look like an SFU bug.
19//!
20//! The stack's SFU now runs on the host and advertises 127.0.0.1 only (see
21//! `crate::sfu`), so the only peer address a scenario is handed is loopback
22//! and there is no LAN traffic left to gate. Signing stays worth doing as
23//! belt and braces — the wildcard bind itself is unchanged, a firewall-on
24//! machine still prompts per application, and a stable identity keeps every
25//! other TCC grant (screen recording, microphone) alive across rebuilds.
26//!
27//! Signing with a real Apple identity replaces that requirement with one
28//! keyed on identifier + team, both of which we hold fixed here
29//! (`chat.hound.stack.<binary>`). The grant is then made once and survives
30//! every rebuild.
31//!
32//! # Opting in
33//!
34//! Set `APPLE_SIGNING_IDENTITY` in the gitignored per-developer
35//! `.mise.local.toml` `[env]` (the same variable the Tauri-era note in that
36//! file already documents); list candidates with
37//! `security find-identity -v -p codesigning`. Unset — the default, and CI —
38//! this module is a no-op and the harness behaves exactly as it did before
39//! it existed.
40

44use eyre::{Result, eyre};

47/// The signing identity, as `codesign --sign` takes it. Named to match what
48/// `.mise.local.toml` already exports for the macOS app bundle.
49pub const IDENTITY_ENV: &str = "APPLE_SIGNING_IDENTITY";
50
51/// Whether signing is configured for this run: a non-empty identity AND a
52/// macOS host. Callers check this before doing work that only exists to feed
53/// [`sign`] (e.g. asking nextest to list its test binaries).
54pub fn enabled() -> bool {
55    cfg!(target_os = "macos") && identity().is_some()
56}
57
58fn identity() -> Option<String> {
59    std::env::var(IDENTITY_ENV)
60        .ok()
61        .filter(|v| !v.trim().is_empty())
62}
63
64/// Sign `binary` in place with the configured identity, output appended to
65/// `build_log`. No-op when signing isn't [`enabled`].
66///
67/// **Invariant: this must not touch a binary that is already correctly
68/// signed.** `codesign --force` rewrites the Mach-O unconditionally, moving
69/// its mtime (and often its length) — and `crate::lifecycle` reads exactly
70/// that (mtime, len) afterwards to decide whether a reused stack's backend is
71/// stale. Signing on every invocation would therefore make every reuse look
72/// like a code change and restart the backend each time, which is the whole
73/// thing reuse exists to avoid. So the requirement is CHECKED first
74/// ([`satisfies_requirement`]) and signing happens only when it does not hold.
75///
76/// A failure here is hard: silently leaving a binary ad-hoc signed puts the
77/// run back on the every-rebuild-is-a-new-app path this module exists to
78/// close, and the symptom (a denied Local Network bind) surfaces much later
79/// as an unrelated-looking media failure.
80pub fn sign(binary: &Path, build_log: &Path) -> Result<()> {
81    let Some(identity) = identity().filter(|_| cfg!(target_os = "macos")) else {
82        return Ok(());
83    };
84    let name = binary
85        .file_name()
86        .ok_or_else(|| eyre!("cannot sign {}: no file name", binary.display()))?
87        .to_string_lossy()
88        .into_owned();
89    let identifier = identifier(&name);
90
91    if satisfies_requirement(binary, &identifier) {
92        debug!(binary = %binary.display(), "already signed by us — leaving it (and its mtime) alone");
93        return Ok(());
94    }
95
96    info!(binary = %binary.display(), %identifier, "codesigning with the configured identity");
97    let mut cmd = Command::new("codesign");
98    cmd.arg("--force")
99        .arg("--sign")
100        .arg(&identity)
101        .arg("--identifier")
102        .arg(&identifier)
103        .arg(binary);
104    crate::cargo::run_captured(cmd, build_log, &format!("codesign {name}"))
105}

107/// The stable identifier for a harness binary: half of the designated
108/// requirement (the team anchor is the other half). It must NOT be derived
109/// from the binary's content — that is precisely what the linker's ad-hoc
110/// signature does — and must not collide with the shipped app's
111/// `chat.hound.app`.
112fn identifier(file_name: &str) -> String {
113    format!("chat.hound.stack.{file_name}")
114}
115
116/// The designated requirement this module signs for: our identifier, signed
117/// under an Apple-issued certificate chain.
118///
119/// `anchor apple generic` (not a pinned team OID) is what keeps the string
120/// free of per-developer detail while still excluding the linker's ad-hoc
121/// signature, which has no anchor at all.
122fn requirement(identifier: &str) -> String {
123    format!(r#"identifier "{identifier}" and anchor apple generic"#)
124}
125
126/// Does `binary` ALREADY carry a signature satisfying [`requirement`]?
127///
128/// `codesign --verify --strict -R=<req>` exits 0 only then. A freshly
129/// relinked binary carries the linker's ad-hoc identifier
130/// (`<name>-<hash>`) and no anchor, so it fails and gets signed; one this
131/// harness signed earlier passes and is left untouched — mtime intact, which
132/// is what the staleness stamp depends on. Any error running `codesign` at
133/// all counts as "not signed": the worst case is one redundant signing pass.
134fn satisfies_requirement(binary: &Path, identifier: &str) -> bool {
135    Command::new("codesign")
136        .arg("--verify")
137        .arg("--strict")
138        .arg(format!("-R={}", requirement(identifier)))
139        .arg(binary)
140        .stdout(Stdio::null())
141        .stderr(Stdio::null())
142        .status()
143        .is_ok_and(|s| s.success())
144}

146#[cfg(test)]
147mod tests {
148    use super::*;
149
150    /// The identifier must be a pure function of the file NAME — never of
151    /// its contents, or it would move on every rebuild exactly like the
152    /// ad-hoc one it replaces.
153    #[test]
154    fn identifier_is_namespaced_and_content_independent() {
155        assert_eq!(identifier("backend"), "chat.hound.stack.backend");
156        assert_eq!(identifier("hound-sfu"), "chat.hound.stack.hound-sfu");
157        // Not the shipped app's identifier, whose grants are separate.
158        assert_ne!(identifier("app"), "chat.hound.app");
159    }
160
161    /// The requirement string is what both halves of the round trip agree
162    /// on: `--identifier` on signing, `-R=` on verification. Quoting matters
163    /// — `identifier` takes a quoted string in the requirement language.
164    #[test]
165    fn requirement_quotes_the_identifier_and_anchors_to_apple() {
166        assert_eq!(
167            requirement("chat.hound.stack.backend"),
168            r#"identifier "chat.hound.stack.backend" and anchor apple generic"#
169        );
170    }
