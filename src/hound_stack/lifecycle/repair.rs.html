1//! What to do with whatever answers on the stack's ports: the pure half of
2//! [`super::is_live`]'s successor. Probing is I/O ([`probe`]); turning the
3//! four probe results into a [`Plan`] is not ([`plan`]), so that is where
4//! the decision lives and what the tests cover.
5//!
6//! The backend is the only probe that decides between reuse and bring-up.
7//! Everything else a live backend is missing is REPAIRED in place: a
8//! bring-up would run the port preflight, which the answering backend fails
9//! by definition — refusing with "run `hound-stack down`" for a state the
10//! harness can fix itself.
11
12use std::time::Duration;
13
14use crate::{auth, sfu};
15
16/// The four probes, taken once, in order, against one backend URL.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub struct Liveness {
19    /// `GET /instance` answers (any status).
20    pub backend: bool,
21    /// The host SFU's `GET /health` answers 2xx.
22    pub sfu_health: bool,
23    /// The backend's registry holds a live, non-draining node.
24    pub sfu_registered: bool,
25    /// The auth page's `GET /` answers.
26    pub auth: bool,
27}
28
29/// What the SFU needs on a reuse.
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum SfuRepair {
32    /// Healthy and registered — only the staleness check applies.
33    Keep,
34    /// Nothing on the health port: spawn one (after stopping any recorded
35    /// pid, which is then a wedged or half-dead process).
36    Spawn,
37    /// Healthy but absent from the registry — the backend was restarted by
38    /// hand, or the registry TTL lapsed. The SFU registers at startup, so a
39    /// restart is the immediate fix (its heartbeat is up to 30 s away).
40    Restart,
41}
42
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub enum Plan {
45    /// No backend: full bring-up, preflight first.
46    BringUp,
47    /// A backend answers: reuse it, repairing the rest.
48    Reuse { sfu: SfuRepair, spawn_auth: bool },
49}
50
51pub fn plan(l: Liveness) -> Plan {
52    if !l.backend {
53        return Plan::BringUp;
54    }
55    let sfu = match (l.sfu_health, l.sfu_registered) {
56        (false, _) => SfuRepair::Spawn,
57        (true, false) => SfuRepair::Restart,
58        (true, true) => SfuRepair::Keep,
59    };
60    Plan::Reuse {
61        sfu,
62        spawn_auth: !l.auth,
63    }
64}
65
66/// Take the four probes. The later three are skipped (false) when the
67/// backend is down — the plan ignores them then.
68pub async fn probe(backend_url: &str) -> Liveness {
69    let client = reqwest::Client::new();
70    if !backend_answers(&client, backend_url).await {
71        return Liveness {
72            backend: false,
73            sfu_health: false,
74            sfu_registered: false,
75            auth: false,
76        };
77    }
78    Liveness {
79        backend: true,
80        sfu_health: sfu::is_healthy(&client).await,
81        sfu_registered: sfu::has_live_node(&client, backend_url).await,
82        auth: auth::is_healthy(&client).await,
83    }
84}
85
86/// Any HTTP response proves the server is up — an auth-gated 401 counts (the
87/// same rule as `wait::wait_http_reachable`).
88pub async fn backend_answers(client: &reqwest::Client, backend_url: &str) -> bool {
89    client
90        .get(format!("{backend_url}/instance"))
91        .timeout(Duration::from_secs(5))
92        .send()
93        .await
94        .is_ok()
95}

97#[cfg(test)]
98mod tests {
99    use super::{Liveness, Plan, SfuRepair, plan};
100
101    fn live(backend: bool, sfu_health: bool, sfu_registered: bool, auth: bool) -> Liveness {
102        Liveness {
103            backend,
104            sfu_health,
105            sfu_registered,
106            auth,
107        }
108    }
109
110    #[test]
111    fn all_four_up_is_plain_reuse() {
112        assert_eq!(
113            plan(live(true, true, true, true)),
114            Plan::Reuse {
115                sfu: SfuRepair::Keep,
116                spawn_auth: false
117            }
118        );
119    }
120
121    /// No backend ⇒ bring-up, whatever else answers (a leftover SFU or auth
122    /// page is the preflight's problem to report).
123    #[test]
124    fn backend_down_is_bring_up_regardless() {
125        for bits in 0..8u8 {
126            let l = live(false, bits & 1 != 0, bits & 2 != 0, bits & 4 != 0);
127            assert_eq!(plan(l), Plan::BringUp, "{l:?}");
128        }
129    }
130
131    /// Silent health port ⇒ spawn, even if the registry still lists a node
132    /// (a dead SFU's entry outlives it until the TTL).
133    #[test]
134    fn silent_sfu_is_spawned() {
135        for registered in [false, true] {
136            assert_eq!(
137                plan(live(true, false, registered, true)),
138                Plan::Reuse {
139                    sfu: SfuRepair::Spawn,
140                    spawn_auth: false
141                }
142            );
143        }
144    }
145
146    #[test]
147    fn healthy_unregistered_sfu_is_restarted() {
148        assert_eq!(
149            plan(live(true, true, false, true)),
150            Plan::Reuse {
151                sfu: SfuRepair::Restart,
152                spawn_auth: false
153            }
154        );
155    }
156
157    #[test]
158    fn missing_auth_is_spawned_independently_of_the_sfu() {
159        assert_eq!(
160            plan(live(true, true, true, false)),
161            Plan::Reuse {
162                sfu: SfuRepair::Keep,
163                spawn_auth: true
164            }
165        );
166        assert_eq!(
167            plan(live(true, true, false, false)),
168            Plan::Reuse {
169                sfu: SfuRepair::Restart,
170                spawn_auth: true
171            }
172        );
173    }
