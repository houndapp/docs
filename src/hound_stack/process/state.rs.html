1//! The detached processes' state file: `target/hound-stack/stack.json`.
2//!
3//! The host backend, SFU and auth page all outlive the `hound-stack`
4//! invocation that started them (reuse-by-default — see `crate::lifecycle`),
5//! so the next invocation has to learn about them from disk rather than from
6//! `Child`ren it owns. This is that record, one entry per process: who to
7//! signal (`pid`), where its output went (`log`), and which binary it is
8//! running (`binary` + [`BinaryStamp`]) so a later run can tell "the same
9//! process is still fine" from "the binary on disk has moved on".
10//!
11//! One file for all of them, because they are one stack: a teardown stops
12//! whatever it finds, and a split record would strand a process.
13//!
14//! Deliberately under `<root>/target/`, not `CARGO_TARGET_DIR`: the record
15//! describes processes running on THIS host against THIS checkout, and
16//! `target/` is the one directory that is already both per-checkout and
17//! gitignored. It is not a build artifact and nothing consumes it but this
18//! crate.
19
20use std::path::{Path, PathBuf};
21use std::time::UNIX_EPOCH;
22
23use eyre::{Context, Result};
24use serde::{Deserialize, Serialize};

26/// Identity of the built binary a running process was spawned from. Content
27/// hashing would be exact but costs a full read of a ~200 MB debug binary on
28/// every invocation; (mtime, len) moves whenever cargo relinks, which is the
29/// only way the binary can change here.
30#[derive(Clone, Copy, Debug, PartialEq, Eq, Serialize, Deserialize)]
31pub struct BinaryStamp {
32    /// Nanoseconds since the Unix epoch (APFS/ext4 resolution — a relink
33    /// inside one second still compares unequal).
34    pub mtime_nanos: u64,
35    pub len: u64,
36}
37
38impl BinaryStamp {
39    pub fn of(binary: &Path) -> Result<Self> {
40        let meta =
41            std::fs::metadata(binary).with_context(|| format!("stat {}", binary.display()))?;
42        let mtime_nanos = meta
43            .modified()
44            .ok()
45            .and_then(|t| t.duration_since(UNIX_EPOCH).ok())
46            .map(|d| d.as_nanos())
47            .and_then(|n| u64::try_from(n).ok())
48            .unwrap_or_default();
49        Ok(Self {
50            mtime_nanos,
51            len: meta.len(),
52        })
53    }
54}

56/// What one `hound-stack` invocation needs to know about a process some
57/// EARLIER invocation left running.
58#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
59pub struct ProcessRecord {
60    pub pid: u32,
61    /// Where the running process's stdout+stderr go — the tail a failing run
62    /// prints, which may belong to a workdir from a previous run.
63    pub log: PathBuf,
64    pub binary: PathBuf,
65    #[serde(flatten)]
66    pub stamp: BinaryStamp,
67    /// The artifact dir of the invocation that spawned it.
68    pub workdir: PathBuf,
69}
70
71impl ProcessRecord {
72    /// Whether the process recorded here is running the binary now on disk.
73    /// A different path counts as stale too — the target dir moved under it.
74    pub fn matches(&self, binary: &Path, stamp: &BinaryStamp) -> bool {
75        self.binary == binary && &self.stamp == stamp
76    }
77}

79/// The stack's detached processes, as last recorded. A missing entry means
80/// "hound-stack is not running one" — either nothing started it, or whatever
81/// answers on its port belongs to somebody else and must be left alone.
82#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
83pub struct StackState {
84    #[serde(default, skip_serializing_if = "Option::is_none")]
85    pub backend: Option<ProcessRecord>,
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub sfu: Option<ProcessRecord>,
88    #[serde(default, skip_serializing_if = "Option::is_none")]
89    pub auth: Option<ProcessRecord>,
90}
91
92impl StackState {
93    pub fn is_empty(&self) -> bool {
94        self.backend.is_none() && self.sfu.is_none() && self.auth.is_none()
95    }
96}

98/// `<root>/target/hound-stack/stack.json`.
99pub fn path(root: &Path) -> PathBuf {
100    root.join("target/hound-stack/stack.json")
101}
102
103/// The recorded state, or an empty one when there is no file (nothing was
104/// started, or a `down` removed it) or it is unreadable/unparsable — a
105/// corrupt record is treated as absent so it can never wedge a bring-up.
106pub fn load(root: &Path) -> StackState {
107    std::fs::read(path(root))
108        .ok()
109        .and_then(|raw| serde_json::from_slice(&raw).ok())
110        .unwrap_or_default()
111}
112
113/// Read-modify-write: the processes are recorded independently but share one
114/// file, so every mutation goes through here. A state with no process left
115/// REMOVES the file, keeping "no file" the single meaning of
116/// "there is nothing of ours to stop" (what `down` short-circuits on).
117pub fn update(root: &Path, f: impl FnOnce(&mut StackState)) -> Result<()> {
118    let mut state = load(root);
119    f(&mut state);
120    if state.is_empty() {
121        remove(root);
122        return Ok(());
123    }
124    store(root, &state)
125}
126
127pub fn store(root: &Path, state: &StackState) -> Result<()> {
128    let file = path(root);
129    if let Some(dir) = file.parent() {
130        std::fs::create_dir_all(dir).with_context(|| format!("creating {}", dir.display()))?;
131    }
132    let json = serde_json::to_vec_pretty(state).context("serialising stack state")?;
133    std::fs::write(&file, json).with_context(|| format!("writing {}", file.display()))
134}
135
136/// Forget every process. Missing is success — `down` is idempotent.
137pub fn remove(root: &Path) {
138    let _ = std::fs::remove_file(path(root));
139}

141#[cfg(test)]
142mod tests {
143    use super::*;
144
145    fn record(pid: u32, bin: &str) -> ProcessRecord {
146        ProcessRecord {
147            pid,
148            log: PathBuf::from("/tmp/hound-stack-abc/backend.log"),
149            binary: PathBuf::from(bin),
150            stamp: BinaryStamp {
151                mtime_nanos: 1_700_000_000_123_456_789,
152                len: 219_000_000,
153            },
154            workdir: PathBuf::from("/tmp/hound-stack-abc"),
155        }
156    }
157
158    fn sample() -> StackState {
159        StackState {
160            backend: Some(record(4321, "/repo/target/debug/backend")),
161            sfu: Some(record(4322, "/repo/target/debug/hound-sfu")),
162            auth: Some(record(4323, "/repo/target/debug/hound-auth")),
163        }
164    }
165
166    #[test]
167    fn state_round_trips_through_json() {
168        let json = serde_json::to_string(&sample()).unwrap();
169        assert_eq!(serde_json::from_str::<StackState>(&json).unwrap(), sample());
170    }
171
172    /// `#[serde(flatten)]` keeps each stamp's fields at its entry's top level
173    /// — the file stays flat objects a human can read in a failure report.
174    #[test]
175    fn stamp_fields_are_flattened() {
176        let v: serde_json::Value = serde_json::to_value(sample()).unwrap();
177        let backend = &v["backend"];
178        assert!(backend.get("mtime_nanos").is_some() && backend.get("len").is_some());
179        assert!(backend.get("stamp").is_none());
180    }
181
182    #[test]
183    fn staleness_compares_both_path_and_stamp() {
184        let state = record(1, "/repo/target/debug/backend");
185        let same = state.stamp;
186        assert!(state.matches(Path::new("/repo/target/debug/backend"), &same));
187        // Relinked in place: same path and length, later mtime.
188        let relinked = BinaryStamp {
189            mtime_nanos: same.mtime_nanos + 1,
190            ..same
191        };
192        assert!(!state.matches(Path::new("/repo/target/debug/backend"), &relinked));
193        // Same bytes, different target dir.
194        assert!(!state.matches(Path::new("/elsewhere/debug/backend"), &same));
195    }
196
197    #[test]
198    fn a_corrupt_record_reads_as_absent() {
199        let dir = tempfile::tempdir().unwrap();
200        let file = path(dir.path());
201        std::fs::create_dir_all(file.parent().unwrap()).unwrap();
202        std::fs::write(&file, b"{ not json").unwrap();
203        assert!(load(dir.path()).is_empty());
204    }
205
206    /// The entries are written and cleared independently, and clearing the
207    /// last one takes the file with it.
208    #[test]
209    fn entries_update_independently_and_the_last_one_out_removes_the_file() {
210        let dir = tempfile::tempdir().unwrap();
211        let root = dir.path();
212        update(root, |s| s.backend = Some(record(1, "/b"))).unwrap();
213        update(root, |s| s.sfu = Some(record(2, "/s"))).unwrap();
214        update(root, |s| s.auth = Some(record(3, "/a"))).unwrap();
215        let loaded = load(root);
216        assert_eq!(loaded.backend.unwrap().pid, 1);
217        assert_eq!(loaded.sfu.unwrap().pid, 2);
218        assert_eq!(loaded.auth.unwrap().pid, 3);
219
220        update(root, |s| s.sfu = None).unwrap();
221        assert!(load(root).sfu.is_none());
222        assert!(path(root).exists());
223
224        update(root, |s| s.auth = None).unwrap();
225        assert!(path(root).exists());
226
227        update(root, |s| s.backend = None).unwrap();
228        assert!(!path(root).exists());
229        assert!(load(root).is_empty());
230    }
