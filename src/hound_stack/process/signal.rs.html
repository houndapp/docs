1//! Signalling a detached stack process by pid.
2//!
3//! The processes this stops belong to no `Child` we hold — an earlier
4//! `hound-stack` invocation spawned them and exited (see
5//! [`crate::process::state`]), so `Child::kill`/`wait` are unavailable and
6//! the raw signals are the whole interface. `SIGTERM`, then a bounded poll
7//! with signal 0 (the "does this pid exist" probe), then `SIGKILL`: the
8//! backend closes its listeners on TERM and the SFU drains-then-exits, and
9//! nothing here may leave :8080 or :7890 held once teardown claims it
10//! finished.
11//!
12//! Pid reuse is the one hazard, and it is bounded by hygiene rather than
13//! checked: the state entry is removed the moment we stop or replace the
14//! process it names, so a stale pid only survives a `kill -9` of this
15//! harness itself. Unix-only — Windows dev is not a supported workflow here.
16
17use std::time::{Duration, Instant};
18
19use eyre::{Result, bail};
20
21/// How long a process gets to exit on `SIGTERM` before it is killed. Covers
22/// the SFU's drain as well as the backend's listener close — the dev SFU runs
23/// with a 1 s drain deadline (see `crate::sfu`), so neither is slow here.
24pub const STOP_GRACE: Duration = Duration::from_secs(10);
25
26const POLL: Duration = Duration::from_millis(100);
27
28/// Whether `pid` still exists (signal 0 — permission errors count as alive,
29/// since the process is demonstrably there).
30pub fn alive(pid: u32) -> bool {
31    #[cfg(unix)]
32    {
33        // SAFETY: `kill` with signal 0 performs the permission/existence
34        // check and delivers nothing.
35        unsafe { libc::kill(pid as libc::pid_t, 0) == 0 }
36    }
37    #[cfg(not(unix))]
38    {
39        let _ = pid;
40        false
41    }
42}
43
44/// `SIGTERM`, wait up to [`STOP_GRACE`] for the pid to go, then `SIGKILL`.
45/// Returns once the pid is gone, or errors if it outlived even the kill.
46pub fn stop(what: &str, pid: u32) -> Result<()> {
47    #[cfg(unix)]
48    {
49        if !alive(pid) {
50            return Ok(());
51        }
52        // SAFETY: plain signal delivery to a pid we recorded.
53        unsafe { libc::kill(pid as libc::pid_t, libc::SIGTERM) };
54        if wait_gone(pid, STOP_GRACE) {
55            return Ok(());
56        }
57        tracing::warn!(pid, "{what} ignored SIGTERM for {STOP_GRACE:?} — SIGKILL");
58        // SAFETY: as above.
59        unsafe { libc::kill(pid as libc::pid_t, libc::SIGKILL) };
60        if wait_gone(pid, Duration::from_secs(5)) {
61            return Ok(());
62        }
63        bail!("{what} pid {pid} survived SIGKILL — stop it by hand before rerunning");
64    }
65    #[cfg(not(unix))]
66    {
67        let _ = (what, pid);
68        bail!(
69            "stopping a detached stack process is unix-only — Windows-native development is not \
70             a supported workflow here (see tools/stack/src/process/signal.rs)"
71        )
72    }
73}
74
75#[cfg(unix)]
76fn wait_gone(pid: u32, budget: Duration) -> bool {
77    let deadline = Instant::now() + budget;
78    loop {
79        if !alive(pid) {
80            return true;
81        }
82        if Instant::now() >= deadline {
83            return false;
84        }
85        std::thread::sleep(POLL);
86    }
87}
