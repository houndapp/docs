1//! Port preflight: fail fast if this run's fixed host ports are already
2//! taken by a leftover/concurrent stack, instead of racing it through
3//! `compose up`/backend-startup timeouts.
4//!
5//! Unlike the shell scripts' `lsof`-based check (silently skipped when
6//! `lsof` isn't installed, and scoped to `LISTEN` sockets only), this binds
7//! each port directly with `SO_REUSEADDR` set — the same "can we actually use
8//! this port" question, without an external tool dependency, and without
9//! false positives from a recently-closed socket still in `TIME_WAIT`.
10
11use eyre::{Result, bail};
12use socket2::{Domain, Socket, Type};
13
14#[derive(Clone, Copy)]
15pub enum Proto {
16    Tcp,
17    Udp,
18}
19
20/// One port this stack binds on the host.
21#[derive(Clone, Copy)]
22pub struct Port {
23    pub number: u16,
24    pub proto: Proto,
25}
26
27impl Port {
28    pub const fn tcp(number: u16) -> Self {
29        Self {
30            number,
31            proto: Proto::Tcp,
32        }
33    }
34    pub const fn udp(number: u16) -> Self {
35        Self {
36            number,
37            proto: Proto::Udp,
38        }
39    }
40}

42/// Postgres, mailpit (smtp+web), iroh-relay (compose), the auth page (host
43/// process, `crate::auth`), backend, the SFU's health server (TCP) plus the
44/// SFU's iroh endpoint and the backend's realtime WebTransport endpoint
45/// (UDP) — every fixed host port this stack's compose services and host
46/// processes bind (see `services/backend/compose.dev.yaml`, `crate::backend`,
47/// `crate::sfu` and `crate::auth`).
48/// The `passkeys`-profile TLS front (:443) is opt-in and deliberately not
49/// preflighted.
50pub const STACK_PORTS: &[Port] = &[
51    Port::tcp(5432),
52    Port::tcp(1025),
53    Port::tcp(8025),
54    Port::tcp(3340),
55    Port::tcp(crate::auth::PORT),
56    Port::tcp(8080),
57    Port::tcp(2019),
58    Port::udp(7890),
59    Port::udp(7891),
60];

62fn busy(port: &Port) -> bool {
63    let (domain, ty) = match port.proto {
64        Proto::Tcp => (Domain::IPV4, Type::STREAM),
65        Proto::Udp => (Domain::IPV4, Type::DGRAM),
66    };
67    let Ok(socket) = Socket::new(domain, ty, None) else {
68        // Can't even create a probe socket — don't block the run on that;
69        // the real bind (compose/backend) will surface the actual failure.
70        return false;
71    };
72    let _ = socket.set_reuse_address(true);
73    let addr = std::net::SocketAddr::from(([0, 0, 0, 0], port.number)).into();
74    socket.bind(&addr).is_err()
75}

77/// Whether anything holds any of `ports` — the cheap, docker-free "is there
78/// something to tear down at all" question `hound-stack down` asks before it
79/// goes near compose.
80pub fn any_busy(ports: &[Port]) -> bool {
81    ports.iter().any(busy)
82}
83
84/// Wait (bounded) for every one of `ports` to be free. For the moment right
85/// after a teardown: docker's host-side port forwarder (OrbStack's in
86/// particular) releases a published port a beat AFTER `compose down` has
87/// reported the container removed, so a preflight run straight away sees the
88/// relay's :3340 still held and refuses a stack that is, in fact, gone.
89/// Returns `false` if something still holds a port at the deadline — the
90/// caller then runs [`preflight`] for the named error.
91pub fn wait_free(ports: &[Port], budget: std::time::Duration) -> bool {
92    let deadline = std::time::Instant::now() + budget;
93    loop {
94        if !any_busy(ports) {
95            return true;
96        }
97        if std::time::Instant::now() >= deadline {
98            return false;
99        }
100        std::thread::sleep(std::time::Duration::from_millis(250));
101    }
102}

104/// Bail with a clear cause (not a downstream compose/backend timeout) if any
105/// `ports` are already bound.
106///
107/// Only run when this invocation is about to bind those ports ITSELF — i.e.
108/// after the liveness probe said there is no stack to reuse. Run
109/// unconditionally it would reject the healthy stack it is supposed to
110/// rejoin.
111pub fn preflight(ports: &[Port]) -> Result<()> {
112    for port in ports {
113        if busy(port) {
114            let proto = match port.proto {
115                Proto::Tcp => "tcp",
116                Proto::Udp => "udp",
117            };
118            bail!(
119                "port {} ({proto}) is held, but there is no stack here to reuse (the backend \
120                 doesn't answer, the host SFU's health port or the auth page doesn't, or no SFU \
121                 is registered) \
122                 — a half-up, wedged or foreign stack. Clear it with `hound-stack down` (`mise \
123                 run //ui:compose:down`), or rerun with `--fresh` to wipe and rebuild; if \
124                 something else on this host legitimately owns that port, stop that instead.",
125                port.number
126            );
127        }
128    }
129    Ok(())
130}
