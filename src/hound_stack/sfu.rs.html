1//! The host SFU process, and the "is voice routable" contract.
2//!
3//! The dev SFU runs on the HOST, not as the `hound-sfu` compose container
4//! (which `compose.dev.yaml` still defines for the prod-like topology). Two
5//! reasons, both about the address clients end up dialling:
6//!
7//! - A container SFU advertises the Docker bridge address, so every media
8//!   packet leaves loopback. On macOS that is Local Network traffic: the
9//!   client needs a per-application privacy grant, and without one the
10//!   packets are dropped silently and the failure reads like an SFU bug.
11//! - `--fresh` had to rebuild the SFU's docker image on every wipe.
12//!
13//! [`ADVERTISE_IPV4`] is what makes the host process loopback-only: iroh
14//! advertises exactly the address it is bound to, so binding 127.0.0.1 is the
15//! only way to publish one reachable loopback address and nothing else (see
16//! `Sfu::bind` in `services/sfu/src/sfu.rs`, and `ADVERTISE_IPV4`'s clap doc
17//! in `services/sfu/src/main.rs`).
18//!
19//! DETACHED like the backend beside it — see `crate::process`.
20

31/// The UDP port the host SFU's iroh endpoint binds — the same fixed port the
32/// container published, so nothing downstream had to move.
33pub const LISTEN_PORT: u16 = 7890;
34
35/// The SFU's `GET /health` port. Answering here is what distinguishes OUR
36/// host SFU from any other node that may have registered with the backend
37/// (the reuse probe in `crate::lifecycle` checks both).
38pub const HEALTH_PORT: u16 = 2019;
39
40/// Bind address AND advertised address (the SFU makes them the same thing —
41/// see the module doc). Loopback only: no media packet this stack carries
42/// ever leaves the host.
43const ADVERTISE_IPV4: &str = "127.0.0.1";
44
45/// Dev drain deadline. The SFU drains on SIGTERM before exiting; with no
46/// sessions that is immediate, and this caps what a teardown waits for when
47/// one is still up.
48const DRAIN_DEADLINE_SECS: &str = "1";
49
50/// How long the SFU gets to bind its iroh endpoint and answer `/health` (the
51/// health server only starts once the node id exists — see
52/// `services/sfu/src/main.rs`).
53const HEALTHY_BUDGET: Duration = Duration::from_secs(30);
54
55/// How long the backend's registry gets to show the node after it is healthy.
56/// One registration POST is issued as soon as the SFU starts, so this covers
57/// a retry, not the 30 s steady-state interval.
58const REGISTERED_BUDGET: Duration = Duration::from_secs(15);
59
60/// `GET /health` on the host SFU.
61pub fn health_url() -> String {
62    format!("http://127.0.0.1:{HEALTH_PORT}/health")
63}
64
65/// Spawn the prebuilt `hound-sfu` `binary` DETACHED, bound and advertising
66/// loopback only, registering with `backend_url`. Output → `log_path`.
67pub fn spawn(binary: &Path, root: &Path, log_path: &Path, backend_url: &str) -> Result<Child> {
68    let mut cmd = Command::new(binary);
69    cmd.current_dir(root)
70        .env("RELAY_SECRET", RELAY_SECRET_DEV)
71        .env("RELAY_URL", RELAY_URL_DEV)
72        .env("BACKEND_URL", backend_url)
73        .env("LISTEN_PORT", LISTEN_PORT.to_string())
74        .env("HEALTH_PORT", HEALTH_PORT.to_string())
75        .env("ADVERTISE_IPV4", ADVERTISE_IPV4)
76        .env("DRAIN_DEADLINE_SECS", DRAIN_DEADLINE_SECS)
77        .env("RUST_LOG", "info");
78
79    process::spawn_detached(&mut cmd, log_path)
80        .with_context(|| format!("spawning hound-sfu ({})", binary.display()))
81}

83/// Wait for the freshly spawned SFU to bind its endpoint (`/health` answers)
84/// and then to appear in the backend's registry. Bails early if the process
85/// exits first — a far more useful cause than either timeout.
86pub async fn wait_ready(child: &mut Child, backend_url: &str) -> Result<()> {
87    let client = reqwest::Client::new();
88    wait_for_http_or_exit(
89        &client,
90        &health_url(),
91        child,
92        &Deadline::after(HEALTHY_BUDGET, Duration::from_millis(200)),
93    )
94    .await
95    .context("SFU never became healthy")?;
96
97    let deadline = Deadline::after(REGISTERED_BUDGET, Duration::from_secs(1));
98    wait_for_condition_or_exit("SFU registration", child, &deadline, || {
99        let client = client.clone();
100        async move { has_live_node(&client, backend_url).await }
101    })
102    .await
103    .context("SFU never registered with the backend — voice/join would fail")
104}
105
106/// Whether the host SFU's `/health` answers — "the process this harness runs
107/// is up", as opposed to "something is registered with the backend".
108pub async fn is_healthy(client: &reqwest::Client) -> bool {
109    client
110        .get(health_url())
111        .timeout(Duration::from_secs(5))
112        .send()
113        .await
114        .is_ok_and(|r| r.status().is_success())
115}

117/// Whether the backend's `GET /internal/sfu/nodes` reports at least one live,
118/// non-draining node — the typed "is voice routable" contract.
119///
120/// Shared by the two callers that ask the same question for different
121/// reasons: the post-spawn registration wait above, and the liveness probe
122/// that decides whether an already-running stack needs its SFU repaired
123/// (`crate::lifecycle`). Anything short of a parsed success response
124/// is `false` — an unreachable backend is not a registered SFU.
125pub async fn has_live_node(client: &reqwest::Client, backend_url: &str) -> bool {
126    let url = format!("{backend_url}/internal/sfu/nodes");
127    let Ok(resp) = client.get(&url).bearer_auth(RELAY_SECRET_DEV).send().await else {
128        return false;
129    };
130    if !resp.status().is_success() {
131        return false;
132    }
133    match resp.json::<hound_types::SfuNodesResponse>().await {
134        Ok(nodes) => nodes.nodes.iter().any(|n| n.live && !n.draining),
135        Err(_) => false,
136    }
137}
