1//! Two membership-hygiene constraints the tables were missing.
2//!
3//! **`profile(user_id, community_id)` is unique.** A community profile is a
4//! user's identity *in* that community, not their membership of it, so leaving
5//! keeps the row — and a rejoin used to mint a second one, after which the
6//! merged read picked whichever the planner returned first. NULL
7//! `community_id` is the global profile, and both backends treat NULLs in a
8//! unique index as distinct, so the one global row per user is unaffected by
9//! this index (nothing here constrains it).
10//!
11//! **`community_ban.banned_by` and `community_invite.created_by` release
12//! instead of cascading.** Both pointed at `user` with `ON DELETE CASCADE`, so
13//! deleting a moderator deleted the bans they had issued — the standing
14//! refusal vanished and everyone they had banned could walk back in. The
15//! actor is provenance, not ownership: the columns become nullable and the
16//! foreign keys `ON DELETE SET NULL`, so the ban outlives the account that
17//! wrote it and the row simply stops naming anyone.
18//!
19//! Postgres can alter both in place. SQLite cannot alter a column's
20//! nullability or a foreign key at all, so each table is rebuilt through the
21//! same `sea-query` builder that created it — which is what keeps the
22//! rebuilt column types identical to the originals by construction rather
23//! than by a hand-written `CREATE TABLE` that has to be kept in step.
24

30const PROFILE_UNIQUE: &str = "idx-profile-user-community";
31const INVITE_COMMUNITY_INDEX: &str = "idx-community-invite-community-id";

33/// Keeps the oldest row of every duplicated `(user_id, community_id)` pair, so
34/// the unique index below can be created on a database that already has some.
35/// UUIDs order as their storage type on both backends, and these are v7, so
36/// "smallest id" is "written first".
37const DEDUPE_PROFILES: &str = "\
38DELETE FROM profile
39WHERE community_id IS NOT NULL
40  AND EXISTS (
41    SELECT 1 FROM profile AS earlier
42    WHERE earlier.user_id = profile.user_id
43      AND earlier.community_id = profile.community_id
44      AND earlier.id < profile.id
45  )";
46
47/// An actor column carried across, with any already-orphaned value released to
48/// `NULL` rather than refused.
49///
50/// SQLite enforces foreign keys only while its pragma is on, so a dev database
51/// can hold a `banned_by` pointing at an account that is gone. Both the
52/// rebuild's copy and Postgres' `ADD CONSTRAINT` validate the column, and
53/// either would fail half-way on such a row — for want of doing exactly what
54/// the new foreign key does to it from here on.
55fn released(column: &str) -> String {
56    format!(r#"CASE WHEN {column} IN (SELECT id FROM "user") THEN {column} END"#)
57}

101/// `ALTER TABLE` per column and per constraint. Quoted `"user"` throughout:
102/// it is a reserved word in Postgres.
103async fn release_actors_postgres(manager: &SchemaManager<'_>) -> Result<(), DbErr> {
104    let db = manager.get_connection();
105    for (table, column, constraint) in [
106        ("community_ban", "banned_by", "fk-community-ban-banned-by"),
107        (
108            "community_invite",
109            "created_by",
110            "fk-community-invite-created-by",
111        ),
112    ] {
113        db.execute_unprepared(&format!(
114            r#"ALTER TABLE {table} ALTER COLUMN {column} DROP NOT NULL"#
115        ))
116        .await?;
117        db.execute_unprepared(&format!(
118            r#"ALTER TABLE {table} DROP CONSTRAINT IF EXISTS "{constraint}""#
119        ))
120        .await?;
121        // Nullable now, so an orphan can be released before the constraint
122        // below validates the column.
123        db.execute_unprepared(&format!(
124            "UPDATE {table} SET {column} = {}",
125            released(column)
126        ))
127        .await?;
128        db.execute_unprepared(&format!(
129            r#"ALTER TABLE {table} ADD CONSTRAINT "{constraint}"
130               FOREIGN KEY ({column}) REFERENCES "user" (id)
131               ON DELETE SET NULL ON UPDATE CASCADE"#
132        ))
133        .await?;
134    }
135    Ok(())
136}

138/// SQLite's only way to change a column: build the table again beside the old
139/// one, copy, and drop. Index names are database-wide, so the old table's
140/// named index goes first — the rename would otherwise carry it into a
141/// collision with the new table's.
142async fn release_actors_by_rebuild(manager: &SchemaManager<'_>) -> Result<(), DbErr> {
143    let db = manager.get_connection();
144
145    db.execute_unprepared("ALTER TABLE community_ban RENAME TO community_ban_old")
146        .await?;
147    manager.create_table(community_ban_table()).await?;
148    db.execute_unprepared(&format!(
149        "INSERT INTO community_ban (community_id, user_id, banned_by, reason, created_at)
150         SELECT community_id, user_id, {}, reason, created_at FROM community_ban_old",
151        released("banned_by")
152    ))
153    .await?;
154    db.execute_unprepared("DROP TABLE community_ban_old")
155        .await?;
156
157    db.execute_unprepared(&format!(
158        r#"DROP INDEX IF EXISTS "{INVITE_COMMUNITY_INDEX}""#
159    ))
160    .await?;
161    db.execute_unprepared("ALTER TABLE community_invite RENAME TO community_invite_old")
162        .await?;
163    manager.create_table(community_invite_table()).await?;
164    db.execute_unprepared(&format!(
165        "INSERT INTO community_invite
166             (id, community_id, code, created_by, created_at, expires_at, max_uses, uses, revoked_at)
167         SELECT id, community_id, code, {}, created_at, expires_at, max_uses, uses, revoked_at
168         FROM community_invite_old",
169        released("created_by")
170    ))
171    .await?;
172    db.execute_unprepared("DROP TABLE community_invite_old")
173        .await?;
174    manager
175        .create_index(
176            Index::create()
177                .if_not_exists()
178                .name(INVITE_COMMUNITY_INDEX)
179                .table(CommunityInvite::Table)
180                .col(CommunityInvite::CommunityId)
181                .to_owned(),
182        )
183        .await?;
184    Ok(())
185}
