1//! The text index over `community_space_message.search_text` — the one
2//! structure `POST /search` reads, in each backend's own idiom.
3//!
4//! Postgres gets a PGroonga index on the column itself. Nothing else changes:
5//! the index is maintained by the write that sets `search_text`, in that
6//! write's own transaction, so a search is read-your-writes for free.
7//!
8//! SQLite (tests, and local dev — the backend's default database is
9//! `sqlite::memory:`) has no such thing, so it gets an **external-content**
10//! FTS5 table shadowing the column plus the three triggers that keep it in
11//! step. External content means the index stores only the terms; the rows
12//! stay in `community_space_message` and are reached by rowid. Same
13//! transaction, same guarantee.
14//!
15//! ## The rowid caveat
16//!
17//! `community_space_message` has a UUID primary key, so its rowid is
18//! SQLite's implicit one — and `VACUUM` may renumber implicit rowids, which
19//! would silently point every index entry at the wrong message. The backend
20//! never VACUUMs, and `POST /internal/messages/reindex` issues an FTS
21//! `'rebuild'` after its walk, which is the repair if one ever happens.
22//!
23//! Tokenizers differ deliberately: PGroonga's default (TokenBigram +
24//! NormalizerAuto) segments CJK, `unicode61` does not. Dev has no CJK
25//! segmentation; production does.
26
27use sea_orm_migration::prelude::*;
28
29#[derive(DeriveMigrationName)]
30pub struct Migration;
31
32/// Named so it reads like the `sea-orm` index names beside it.
33const PG_INDEX: &str = "idx-community-space-message-search-text";
34
35/// `remove_diacritics 2` is the Unicode-correct setting (the default, `1`,
36/// mishandles codepoints that are two diacritics at once).
37const SQLITE_CREATE_FTS: &str = "\
38CREATE VIRTUAL TABLE IF NOT EXISTS community_space_message_fts USING fts5(
39    search_text,
40    content='community_space_message',
41    content_rowid='rowid',
42    tokenize='unicode61 remove_diacritics 2'
43)";
44
45/// The external-content sync triggers. A delete is expressed as an insert of
46/// the `'delete'` command carrying the *old* text — FTS5's contract for
47/// content tables it does not own.
48///
49/// `UPDATE OF search_text` rather than a bare `UPDATE`: the message write
50/// path upserts with `search_text` always in the SET list (see
51/// `api::space::message::store`), and edits of other columns must not churn
52/// the index.
53const SQLITE_TRIGGERS: [&str; 3] = [
54    "\
55CREATE TRIGGER IF NOT EXISTS community_space_message_fts_insert
56AFTER INSERT ON community_space_message BEGIN
57    INSERT INTO community_space_message_fts(rowid, search_text)
58    VALUES (new.rowid, new.search_text);
59END",
60    "\
61CREATE TRIGGER IF NOT EXISTS community_space_message_fts_delete
62AFTER DELETE ON community_space_message BEGIN
63    INSERT INTO community_space_message_fts(community_space_message_fts, rowid, search_text)
64    VALUES ('delete', old.rowid, old.search_text);
65END",
66    "\
67CREATE TRIGGER IF NOT EXISTS community_space_message_fts_update
68AFTER UPDATE OF search_text ON community_space_message BEGIN
69    INSERT INTO community_space_message_fts(community_space_message_fts, rowid, search_text)
70    VALUES ('delete', old.rowid, old.search_text);
71    INSERT INTO community_space_message_fts(rowid, search_text)
72    VALUES (new.rowid, new.search_text);
73END",
74];
75
76/// Populates the index from whatever is already in the table — this migration
77/// can land on a populated database.
78const SQLITE_REBUILD: &str =
79    "INSERT INTO community_space_message_fts(community_space_message_fts) VALUES('rebuild')";
80
81#[async_trait::async_trait]
82impl MigrationTrait for Migration {
83    async fn up(&self, manager: &SchemaManager) -> Result<(), DbErr> {
84        let db = manager.get_connection();
85        if manager.get_database_backend() == sea_orm::DbBackend::Postgres {
86            db.execute_unprepared("CREATE EXTENSION IF NOT EXISTS pgroonga")
87                .await?;
88            db.execute_unprepared(&format!(
89                r#"CREATE INDEX IF NOT EXISTS "{PG_INDEX}" ON community_space_message USING pgroonga (search_text)"#
90            ))
91            .await?;
92            return Ok(());
93        }
94
95        // One statement per call: `execute_unprepared` is a single prepared
96        // statement on some drivers, and a `;`-joined script is not portable.
97        db.execute_unprepared(SQLITE_CREATE_FTS).await?;
98        for trigger in SQLITE_TRIGGERS {
99            db.execute_unprepared(trigger).await?;
100        }
101        db.execute_unprepared(SQLITE_REBUILD).await?;
102        Ok(())
103    }
104
105    async fn down(&self, manager: &SchemaManager) -> Result<(), DbErr> {
106        let db = manager.get_connection();
107        if manager.get_database_backend() == sea_orm::DbBackend::Postgres {
108            // The index only. `pgroonga` is an extension other schemas in the
109            // same database may be using, and dropping it is not this
110            // migration's to decide.
111            db.execute_unprepared(&format!(r#"DROP INDEX IF EXISTS "{PG_INDEX}""#))
112                .await?;
113            return Ok(());
114        }
115
116        for trigger in [
117            "community_space_message_fts_insert",
118            "community_space_message_fts_delete",
119            "community_space_message_fts_update",
120        ] {
121            db.execute_unprepared(&format!("DROP TRIGGER IF EXISTS {trigger}"))
122                .await?;
123        }
124        db.execute_unprepared("DROP TABLE IF EXISTS community_space_message_fts")
125            .await?;
126        Ok(())
127    }
128}
