1//! # Deterministic inputs for the client media benchmarks
2//!
3//! Every bench target draws its frames from here so a measurement depends only
4//! on the code under test: the generators are pure functions of a frame index —
5//! no clock, no RNG crate, no device — so two runs on the same machine differ
6//! only by the change being measured.
7//!
8//! ## Why not the engine's synthetic capturer
9//!
10//! [`hound_engine`]'s synthetic camera renders the same pattern, but it is a
11//! *capture source*: it sleeps to pace itself to the requested fps, and its
12//! renderer is `pub(crate)`. A bench that called it would measure the sleep.
13//! [`i420_frame`] mirrors its pattern (scrolling gradient + bouncing block +
14//! cycling chroma) so the encoder does the same inter-frame work.
15

24/// Nominal capture cadence the generators stamp at: 30 fps, matching the
25/// client's camera and screen publish rate (`CAM_FPS`/`SCREEN_FPS`).
26pub const FRAME_INTERVAL_US: i64 = 33_333;
27
28/// One deterministic I420 frame at `idx` in a synthetic capture sequence,
29/// stamped on a 30 fps timeline.
30///
31/// The content **moves**, which is the point: a static frame makes every P-frame
32/// a near-empty residual and the encode number a fiction. Three superimposed
33/// motions, all functions of `idx`:
34///
35/// * a diagonal luma gradient scrolling one pixel per frame (global motion,
36///   every macroblock changes);
37/// * a bright block bouncing on a triangle wave (localised motion the encoder
38///   must chase with motion vectors);
39/// * slowly cycling flat U/V (the chroma planes are never constant).
40///
41/// Panics on odd/zero dimensions — [`Dimensions`] enforces the 4:2:0 contract.
42pub fn i420_frame(width: u32, height: u32, idx: usize) -> RawFrame {
43    let dims = Dimensions::new(width, height);
44    let (w, h) = (width as usize, height as usize);
45    let (cw, ch) = (w / 2, h / 2);
46    let y_size = w * h;
47    let c_size = cw * ch;
48
49    let mut data = vec![0u8; y_size + 2 * c_size];
50    let (y_plane, uv) = data.split_at_mut(y_size);
51    let (u_plane, v_plane) = uv.split_at_mut(c_size);
52
53    let t = idx as u64;
54
55    // Scrolling diagonal gradient — the phase advances a pixel per frame, so
56    // successive frames differ everywhere.
57    let scroll = (t % w as u64) as usize;
58    for j in 0..h {
59        let row = j * w;
60        for i in 0..w {
61            let x = (i + scroll) % w;
62            // Limited-range luma ramp varying in x+y.
63            y_plane[row + i] = 16 + (((x + j) * 219 / (w + h)) % 220) as u8;
64        }
65    }
66
67    // Bouncing bright block: localised motion, bounded by a triangle wave so it
68    // reverses at the edges instead of wrapping (a wrap is a scene cut).
69    let bw = (w / 6).max(2);
70    let bh = (h / 6).max(2);
71    let bx = triangle(t, w.saturating_sub(bw).max(1) as u64) as usize;
72    let by = triangle(t / 2, h.saturating_sub(bh).max(1) as u64) as usize;
73    for j in by..(by + bh).min(h) {
74        let row = j * w;
75        for i in bx..(bx + bw).min(w) {
76            y_plane[row + i] = 235;
77        }
78    }
79
80    // Cycling flat chroma, out of phase between U and V.
81    let u_val = (128 + ((t / 3) % 64) as i32 - 32).clamp(16, 240) as u8;
82    let v_val = (128 + ((t / 5) % 64) as i32 - 32).clamp(16, 240) as u8;
83    u_plane.fill(u_val);
84    v_plane.fill(v_val);
85
86    RawFrame {
87        dims,
88        format: PixelFormat::I420,
89        timestamp_us: TimestampUs::from_micros(idx as i64 * FRAME_INTERVAL_US),
90        captured_at: Instant::now(),
91        data,
92        // Portable byte frame: a zero-copy platform surface would make the
93        // measurement host-specific (and unavailable off Apple).
94        surface: None,
95    }
96}

98/// Triangle wave in `0..=span`, period `2*span` — bounces the block between the
99/// frame edges rather than teleporting it back.
100fn triangle(t: u64, span: u64) -> u64 {
101    let period = 2 * span;
102    let phase = t % period;
103    if phase <= span { phase } else { period - phase }
104}

106/// One deterministic 20 ms mono 48 kHz PCM frame at `idx` — 960 samples
107/// ([`SAMPLES_PER_FRAME`]), the exact size Opus and the APM accept.
108///
109/// Pseudo-voice rather than a pure tone: two sines (a 220 Hz fundamental plus a
110/// 660 Hz formant) with a low noise floor, at ~0.4 full scale. Silence or a
111/// single sine would let Opus's rate control settle somewhere no real mic ever
112/// puts it. The phase is continuous across `idx`, so a sequence of frames is a
113/// continuous signal (no per-frame discontinuity for the encoder to spend bits
114/// on).
115pub fn pcm_frame(idx: usize) -> PcmFrame {
116    let base = idx * SAMPLES_PER_FRAME;
117    let rate = SAMPLE_RATE.hz() as f32;
118    let mut noise = Lcg::new(0x5EED_0000 ^ idx as u64);
119    let mut samples = Vec::with_capacity(SAMPLES_PER_FRAME);
120    for i in 0..SAMPLES_PER_FRAME {
121        let t = (base + i) as f32 / rate;
122        let voiced = (std::f32::consts::TAU * 220.0 * t).sin() * 0.30
123            + (std::f32::consts::TAU * 660.0 * t).sin() * 0.12;
124        // ±2% dither — enough that the noise suppressor and Opus's VBR have
125        // something to do, far too little to clip.
126        let hiss = (noise.next_u64() >> 40) as i32 as f32 / (1 << 24) as f32 - 0.5;
127        samples.push(((voiced + hiss * 0.04) * i16::MAX as f32) as i16);
128    }
129    PcmFrame::at_engine_rate(
130        samples,
131        TimestampUs::from_micros(idx as i64 * PCM_FRAME_INTERVAL_US),
132    )
133}

135/// 20 ms in microseconds — the Opus frame duration the audio path is fixed at.
136pub const PCM_FRAME_INTERVAL_US: i64 = 20_000;

138/// A linear congruential generator: the benches' only source of "randomness".
139///
140/// Deterministic and seeded per bench, so jittered arrival times, loss patterns,
141/// and packet gaps are byte-identical between a baseline run and the run being
142/// compared against it — a criterion regression must never be an RNG artefact.
143pub struct Lcg(u64);
144
145impl Lcg {
146    pub const fn new(seed: u64) -> Self {
147        // A zero seed would stick at zero under a pure multiplier; the increment
148        // in `next_u64` rules that out, so any seed is fine.
149        Self(seed)
150    }
151
152    pub fn next_u64(&mut self) -> u64 {
153        // Knuth's MMIX constants.
154        self.0 = self
155            .0
156            .wrapping_mul(6_364_136_223_846_793_005)
157            .wrapping_add(1_442_695_040_888_963_407);
158        self.0
159    }
160
161    /// Uniform in `0..n` (high bits: the low bits of an LCG have short periods).
162    pub fn below(&mut self, n: u64) -> u64 {
163        if n == 0 {
164            0
165        } else {
166            (self.next_u64() >> 33) % n
167        }
168    }
169
170    /// Uniform in `0.0..1.0`.
171    pub fn next_f64(&mut self) -> f64 {
172        (self.next_u64() >> 11) as f64 / (1u64 << 53) as f64
173    }
174}

180/// The checked-in encoded bitstreams the AV1/H.265 decode benches replay.
181///
182/// Those two codecs have **no public software encode path** in this workspace —
183/// AV1 encode is a `cfg(test)` rav1e dev-dependency inside `hound-engine`, and
184/// there is no software HEVC encoder at all — so a bench cannot produce its own
185/// input the way the H.264 benches do. It replays a pre-encoded file instead.
186/// `examples/gen_fixtures.rs` regenerates them.
187pub fn fixture_dir() -> PathBuf {
188    Path::new(env!("CARGO_MANIFEST_DIR")).join("fixtures")
189}
190
191/// Fixture container: a bare sequence of `[u32 little-endian payload length]
192/// [payload bytes]` records, one record per access/temporal unit, and nothing
193/// else — no index, no magic, no per-frame metadata.
194///
195/// Deliberately minimal. A fixture's only job is to hand a decoder the same
196/// units, in the same order, that the wire would; anything richer is a second
197/// container format to keep in sync with the real one for no benefit.
198pub fn write_chunks(path: &Path, chunks: &[Vec<u8>]) -> io::Result<()> {
199    let mut out = Vec::new();
200    for chunk in chunks {
201        let len = u32::try_from(chunk.len())
202            .map_err(|_| io::Error::other("chunk exceeds u32 length field"))?;
203        out.extend_from_slice(&len.to_le_bytes());
204        out.extend_from_slice(chunk);
205    }
206    if let Some(parent) = path.parent() {
207        std::fs::create_dir_all(parent)?;
208    }
209    std::fs::write(path, out)
210}

212/// Read a fixture back into its access units. A truncated record is an error,
213/// not a silent short read — a half-decoded fixture would make a bench measure
214/// a decoder that never bootstraps.
215pub fn read_chunks(path: &Path) -> io::Result<Vec<Vec<u8>>> {
216    let bytes = std::fs::read(path)?;
217    let mut chunks = Vec::new();
218    let mut at = 0usize;
219    while at < bytes.len() {
220        let end = at + 4;
221        let header: [u8; 4] = bytes
222            .get(at..end)
223            .and_then(|s| s.try_into().ok())
224            .ok_or_else(|| io::Error::other("truncated length field"))?;
225        let len = u32::from_le_bytes(header) as usize;
226        let payload = bytes
227            .get(end..end + len)
228            .ok_or_else(|| io::Error::other("truncated payload"))?;
229        chunks.push(payload.to_vec());
230        at = end + len;
231    }
232    Ok(chunks)
233}
