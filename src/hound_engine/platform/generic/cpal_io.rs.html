1//! # cpal microphone capture + speaker playback (CoreAudio on macOS)
2//!
3//! The runnable device I/O for the audio engine. [`CpalCapturer`] opens an
4//! input device and bridges its **real-time callback** to the blocking-pull
5//! [`AudioInput`] trait through a bounded ring buffer; [`CpalPlayback`] opens
6//! an output device and drains a ring buffer the [`AudioOutput`] trait pushes
7//! into. cpal's CoreAudio backend is the macOS/iOS path; the same code is the
8//! Linux (ALSA/Pulse) / Windows (WASAPI) / Android (AAudio) path behind these
9//! traits — which is why this lives in `platform::generic`.
10//!
11//! ## Lifecycle
12//!
13//! Open **is** construct ([`CpalCapturer::open`] / [`CpalPlayback::open`])
14//! and stop is `Drop` — a value that exists is streaming, so a
15//! "capturer not started" runtime error class is unrepresentable.
16//!
17//! ## The real-time contract (the one thing that can go wrong here)
18//!
19//! cpal invokes the data callback on a **CoreAudio real-time thread**. That
20//! callback:
21//! * MUST NOT allocate, lock unboundedly, or block — an xrun is audible.
22//! * Here it only does a bounded `push`/`pop` over an SPSC ring
23//!   ([`rtrb`]-style, but we use a `Mutex<VecDeque>` with a *try* discipline:
24//!   the lock is held for O(burst) memcpy only, never across a syscall, and
25//!   the callback never waits on the consumer). A true lock-free SPSC ring is
26//!   the noted production refinement.
27//!
28//! ## Format negotiation
29//!
30//! We ASK for 48 kHz mono i16. CoreAudio's default device config is often
31//! 48 kHz already, but the channel count and sample FORMAT are
32//! device-dependent (built-in mics frequently expose f32, and some expose
33//! stereo). So:
34//! * We open with the device's **default config's sample format** and convert
35//!   whatever it delivers (f32/i16/u16/…) to i16 in the callback via cpal's
36//!   `Sample`/`FromSample` traits.
37//! * If the device delivers multi-channel, we **downmix to mono** by
38//!   averaging channels in the callback before buffering.
39//! * The device keeps its native RATE; we **resample** to/from the engine's
40//!   48 kHz at the trait boundary (capture `next_frame` / playback `play`,
41//!   via [`super::resample::MonoResampler`]) on worker threads, never in the
42//!   RT callback. AirPods mics (24 kHz HFP) and 44.1 kHz outputs are common;
43//!   mislabelling instead of resampling plays every voice pitch-shifted
44//!   (chipmunk/deep-voice). [`AudioInput::negotiated`] still reports the
45//!   device truth.
46//!
47//! ## Permission / honest-verdict
48//!
49//! On CoreAudio a denied microphone fails at `build_input_stream` / `play`
50//! time. cpal 0.18 maps that to [`cpal::ErrorKind::PermissionDenied`]
51//! (and absence to `DeviceNotAvailable`), which we translate to the typed
52//! [`AudioError::PermissionDenied`] / [`AudioError::NoDevice`] — a DISTINCT
53//! error, never a silent stream of zero-RMS frames.

70/// Ring-buffer capacity (in mono i16 samples). Sized to ~10 frames of 20 ms
71/// (≈200 ms) so a brief consumer hiccup does not drop data, but the buffer can
72/// never grow unbounded: on overflow the oldest samples are dropped (capture)
73/// or the producer drops the frame (playback).
74const RING_CAPACITY_SAMPLES: usize = SAMPLES_PER_FRAME * 10;
75
76/// Linux only: the shared sound-server PCM — `pipewire`, else `pulse`. These
77/// PCMs **multiplex** (non-exclusive access), so opening one gives the shared
78/// access users expect and follows the server's selected default sink/source.
79///
80/// A raw `hw:` / `default` ALSA PCM is **exclusive**: `snd_pcm_open` returns
81/// `EBUSY` ("device temporarily busy") whenever the sound server — or another
82/// app, or a stale process — already holds the card. That is the "no audio at
83/// all" failure on a Linux box whose `default` does not itself route through the
84/// server. Routing through `pipewire`/`pulse` sidesteps it entirely.
85#[cfg(target_os = "linux")]
86fn shared_server_device(host: &cpal::Host, input: bool) -> Option<cpal::Device> {
87    let devices: Vec<cpal::Device> = if input {
88        host.input_devices().ok()?.collect()
89    } else {
90        host.output_devices().ok()?.collect()
91    };
92    for pref in ["pipewire", "pulse"] {
93        if let Some(dev) = devices.iter().find(|d| d.to_string() == pref) {
94            return Some(dev.clone());
95        }
96    }
97    None
98}

100/// Ordered **input** devices to try opening, best first. We try each in turn so a
101/// busy/exclusive primary still yields audio rather than failing the call:
102/// 1. the user's explicit named choice (if present),
103/// 2. **(Linux)** the shared `pipewire`/`pulse` PCM — non-exclusive access (for a
104///    `Default` choice this is the preferred device; for a named choice the
105///    fallback when the named device is busy/exclusive),
106/// 3. the host default PCM.
107fn input_candidates(host: &cpal::Host, choice: &DeviceChoice<MicId>) -> Vec<cpal::Device> {
108    let mut out: Vec<cpal::Device> = Vec::new();
109    if let DeviceChoice::Named(want) = choice
110        && let Ok(devices) = host.input_devices()
111        && let Some(dev) = devices.into_iter().find(|d| d.to_string() == want.as_str())
112    {
113        out.push(dev);
114    }
115    #[cfg(target_os = "linux")]
116    if let Some(dev) = shared_server_device(host, true) {
117        out.push(dev);
118    }
119    if let Some(dev) = host.default_input_device() {
120        out.push(dev);
121    }
122    out
123}

125/// Ordered **output** devices to try opening, best first. Mirror of
126/// [`input_candidates`] (named → Linux shared server PCM → host default).
127fn output_candidates(host: &cpal::Host, choice: &DeviceChoice<SpeakerId>) -> Vec<cpal::Device> {
128    let mut out: Vec<cpal::Device> = Vec::new();
129    if let DeviceChoice::Named(want) = choice
130        && let Ok(devices) = host.output_devices()
131        && let Some(dev) = devices.into_iter().find(|d| d.to_string() == want.as_str())
132    {
133        out.push(dev);
134    }
135    #[cfg(target_os = "linux")]
136    if let Some(dev) = shared_server_device(host, false) {
137        out.push(dev);
138    }
139    if let Some(dev) = host.default_output_device() {
140        out.push(dev);
141    }
142    out
143}

145/// Map a cpal stream-build / control error to our taxonomy, preserving the
146/// permission-vs-absence distinction that the honest-verdict UI depends on.
147/// `kind` names which device direction was being opened (typed permission).
148fn map_cpal_err(context: &str, kind: DeviceKind, e: cpal::Error) -> AudioError {
149    use cpal::ErrorKind;
150    let msg = format!("{context}: {e}");
151    match e.kind() {
152        ErrorKind::PermissionDenied => AudioError::PermissionDenied { kind },
153        ErrorKind::DeviceNotAvailable | ErrorKind::HostUnavailable => AudioError::NoDevice(msg),
154        ErrorKind::UnsupportedConfig
155        | ErrorKind::InvalidInput
156        | ErrorKind::UnsupportedOperation => AudioError::UnsupportedFormat(msg),
157        _ => AudioError::Device(msg),
158    }
159}

162// Shared ring buffer between the RT callback and the worker thread.

165/// A bounded mono-i16 ring shared between a cpal RT callback (one side) and
166/// the trait's blocking-pull / non-blocking-push API (the other). The mutex is
167/// only ever held for a bounded memcpy, never across a syscall, and the
168/// callback never *waits* on it (it would rather drop than block the audio
169/// thread).
170struct Ring {
171    buf: Mutex<VecDeque<i16>>,
172    /// Signalled by the capture callback when it has pushed samples, so a
173    /// blocked `next_frame` wakes promptly instead of polling.
174    not_empty: Condvar,
175    /// Cleared on stop so a blocked consumer unblocks and returns.
176    running: AtomicBool,
177    /// Cumulative capture samples DROPPED on overflow — the send-side audible hole.
178    /// An overflowing capture ring means the consumer thread was descheduled past
179    /// ~200 ms; the assembled frames stay timestamp-contiguous across the hole, so
180    /// this counter is the ONLY evidence it happened. Read+reset by the consumer via
181    /// [`Ring::take_dropped`].
182    dropped_samples: AtomicU64,
183}
184
185impl Ring {
186    fn new() -> Arc<Self> {
187        Arc::new(Self {
188            buf: Mutex::new(VecDeque::with_capacity(RING_CAPACITY_SAMPLES)),
189            not_empty: Condvar::new(),
190            running: AtomicBool::new(true),
191            dropped_samples: AtomicU64::new(0),
192        })
193    }
194
195    /// Producer (capture RT callback): push mono i16 samples, dropping the
196    /// OLDEST on overflow so the buffer is bounded and the callback never
197    /// blocks.
198    fn push_capture(&self, samples: &[i16]) {
199        if let Ok(mut q) = self.buf.lock() {
200            // Bulk-append, then trim the oldest past capacity in one drain (drop-oldest,
201            // never blocks) — a single memcpy instead of a per-sample push_back.
202            q.extend(samples.iter().copied());
203            if q.len() > RING_CAPACITY_SAMPLES {
204                let excess = q.len() - RING_CAPACITY_SAMPLES;
205                q.drain(..excess);
206                self.dropped_samples
207                    .fetch_add(excess as u64, Ordering::Relaxed);
208            }
209            drop(q);
210            self.not_empty.notify_one();
211        }
212    }
213
214    /// Consumer (worker thread): block until `n` samples are available or the
215    /// stream stops, moving them into the caller's reusable `out`; returns `false`
216    /// if the stream stopped before enough data arrived.
217    ///
218    /// `out` is caller-owned and its capacity is reserved BEFORE the lock is taken:
219    /// the RT capture callback blocks on this same mutex in
220    /// [`Ring::push_capture`], so a heap allocation inside the critical section
221    /// would let the allocator (contended by the video encode/decode threads) stall
222    /// a CoreAudio/WASAPI callback — the priority inversion the module header
223    /// forbids. Under the lock this does a bounded memcpy and nothing else.
224    fn pop_blocking_into(&self, n: usize, out: &mut Vec<i16>) -> bool {
225        out.clear();
226        out.reserve(n);
227        let Ok(mut q) = self.buf.lock() else {
228            return false;
229        };
230        while q.len() < n {
231            if !self.running.load(Ordering::Relaxed) {
232                return false;
233            }
234            // Wake periodically so a stop that races the notify is still seen.
235            let Ok((g, _timeout)) = self
236                .not_empty
237                .wait_timeout(q, std::time::Duration::from_millis(100))
238            else {
239                return false;
240            };
241            q = g;
242            if !self.running.load(Ordering::Relaxed) && q.len() < n {
243                return false;
244            }
245        }
246        out.extend(q.drain(..n));
247        true
248    }
249
250    /// Cumulative dropped-sample count since open (capture overflow). Monotonic —
251    /// callers diff it per stats window, like every other bus counter.
252    fn dropped_total(&self) -> u64 {
253        self.dropped_samples.load(Ordering::Relaxed)
254    }
255
256    fn stop(&self) {
257        self.running.store(false, Ordering::Relaxed);
258        self.not_empty.notify_all();
259    }
260}

262// ---------------------------------------------------------------------------
263// Playback ring: wait-free SPSC.
264//
265// The PLAYBACK path has exactly one producer (the mix thread, via `play`) and one
266// consumer (the CoreAudio RT output callback), so it uses a wait-free SPSC ring
267// (`rtrb`) rather than the mutex-guarded `Ring` above. The RT output callback then
268// NEVER takes a lock — a mutex-guarded ring could block it if the mix thread were
269// preempted mid-push, a priority inversion whenever the mix thread is not real-time
270// (RT promotion is best-effort and fails with no audio entitlement). "Audio is
271// sacred": the device thread always makes progress, padding silence on a dry ring
272// instead of waiting. Capture keeps the mutex `Ring` (its consumer blocks on a condvar,
273// which SPSC can't express, and it is not the latency-critical output path).
274// ---------------------------------------------------------------------------
275
276/// Diagnostics shared across the playback ring's three reachers: the mix thread
277/// (owner), the RT output callback (counts underrun), and the stream error callback
278/// (marks death). Plain atomics behind an `Arc`.
279struct PlaybackShared {
280    /// Cumulative output samples the RT callback silence-padded because the ring ran
281    /// dry — the definitive end-of-pipeline AUDIBLE GAP. Steady state is ZERO; a rising
282    /// count IS the stutter, made measurable. Read+reset via [`PlaybackProducer::take_output_underruns`].
283    output_underrun_samples: AtomicU64,
284    /// Cumulative samples `push_playback` DROPPED because the ring was already full
285    /// — the OVERPRODUCTION direction of drift. Without it only the dry direction is
286    /// counted, so a call whose output latency has crept to the whole ring reads
287    /// clean. Read+reset via [`PlaybackProducer::take_output_overruns`].
288    output_drop_samples: AtomicU64,
289    /// Set by the OUTPUT stream's error callback for a FATAL error (the device
290    /// vanished, the config was invalidated, the backend failed): the cpal stream is
291    /// then dead but the `CpalPlayback` lives on, so the mix bus polls
292    /// [`PlaybackProducer::is_output_dead`] each tick and reopens the device.
293    ///
294    /// NOT set for an automatic reroute — see [`Self::route_changed`].
295    output_dead: AtomicBool,
296    /// Set by the OUTPUT stream's error callback on [`cpal::ErrorKind::DeviceChanged`]
297    /// — the OS moved the default output (AirPods connecting/disconnecting) and cpal
298    /// **already rerouted the running stream**, which is still playing. The stream is
299    /// therefore NOT dead; what is stale is our GEOMETRY: [`CpalPlayback::dev_rate`] and
300    /// the 48 kHz→device resampler still describe the OLD device, so `buffered_ms()`
301    /// mis-scales the mix bus's cushion regulator across e.g. a 48 k→24 k hop.
302    ///
303    /// The mix bus debounces this into ONE re-open once the route stops flapping. Read+
304    /// cleared via [`PlaybackProducer::take_route_changed`].
305    route_changed: AtomicBool,
306}
307
308/// The mix-thread (sole producer) half of the playback ring + shared diagnostics.
309struct PlaybackProducer {
310    producer: rtrb::Producer<i16>,
311    capacity: usize,
312    shared: Arc<PlaybackShared>,
313}
314
315/// The RT-output-callback (sole consumer) half + shared diagnostics.
316struct PlaybackConsumer {
317    consumer: rtrb::Consumer<i16>,
318    shared: Arc<PlaybackShared>,
319}
320
321/// Build a wait-free SPSC playback ring of `capacity` mono samples.
322fn playback_ring(capacity: usize) -> (PlaybackProducer, PlaybackConsumer) {
323    let (producer, consumer) = rtrb::RingBuffer::<i16>::new(capacity);
324    let shared = Arc::new(PlaybackShared {
325        output_underrun_samples: AtomicU64::new(0),
326        output_drop_samples: AtomicU64::new(0),
327        output_dead: AtomicBool::new(false),
328        route_changed: AtomicBool::new(false),
329    });
330    (
331        PlaybackProducer {
332            producer,
333            capacity,
334            shared: Arc::clone(&shared),
335        },
336        PlaybackConsumer { consumer, shared },
337    )
338}
339
340impl PlaybackProducer {
341    /// Mix thread: push mono samples for the RT callback to drain, returning how many
342    /// were ACCEPTED. Wait-free; on a full ring (the bus outran the device) the tail is
343    /// dropped rather than blocking, and counted — the caller must not treat dropped
344    /// samples as played (the mix bus's AEC render reference depends on it).
345    fn push_playback(&mut self, samples: &[i16]) -> usize {
346        let mut n = 0;
347        for &s in samples {
348            if self.producer.push(s).is_err() {
349                break; // full: drop the rest (overproduction)
350            }
351            n += 1;
352        }
353        let dropped = samples.len() - n;
354        if dropped > 0 {
355            self.shared
356                .output_drop_samples
357                .fetch_add(dropped as u64, Ordering::Relaxed);
358        }
359        n
360    }
361
362    /// Buffered (unplayed) samples = capacity − free write slots. Coarse (the consumer
363    /// may pop concurrently); the mix bus uses it only for the ms-scale output cushion.
364    fn buffered(&self) -> usize {
365        self.capacity.saturating_sub(self.producer.slots())
366    }
367
368    /// Read and reset the dropped-sample counter (samples the ring was too full to
369    /// accept). Polled by the bus each tick alongside the underrun gauge.
370    fn take_output_overruns(&self) -> u64 {
371        self.shared.output_drop_samples.swap(0, Ordering::Relaxed)
372    }
373
374    /// Read and reset the underrun-sample counter (the silence the RT callback inserted
375    /// on a dry ring). Polled by the bus each tick for the live "playout underruns" stat.
376    fn take_output_underruns(&self) -> u64 {
377        self.shared
378            .output_underrun_samples
379            .swap(0, Ordering::Relaxed)
380    }
381
382    /// Whether the output stream has errored FATALLY (device gone / config invalidated)
383    /// → reopen. An automatic reroute is not a death; see [`Self::take_route_changed`].
384    fn is_output_dead(&self) -> bool {
385        self.shared.output_dead.load(Ordering::Relaxed)
386    }
387
388    /// Read+clear the "the OS rerouted this stream" flag. Edge-triggered on purpose: the
389    /// bus turns each edge into a debounce extension, so a flap burst coalesces into one
390    /// re-sync. See [`PlaybackShared::route_changed`].
391    fn take_route_changed(&self) -> bool {
392        self.shared.route_changed.swap(false, Ordering::Relaxed)
393    }
394}
395
396impl PlaybackConsumer {
397    /// RT output callback: pop up to `out.len()` mono samples (front-filled), silence-pad
398    /// the remainder on underrun, and count the gap. Wait-free — never locks, never blocks.
399    fn pop_playback(&mut self, out: &mut [i16]) {
400        let mut n = 0;
401        while n < out.len() {
402            match self.consumer.pop() {
403                Ok(s) => {
404                    out[n] = s;
405                    n += 1;
406                }
407                Err(_) => break, // ring dry: the rest is silence
408            }
409        }
410        for slot in &mut out[n..] {
411            *slot = 0;
412        }
413        let padded = out.len() - n;
414        if padded > 0 {
415            self.shared
416                .output_underrun_samples
417                .fetch_add(padded as u64, Ordering::Relaxed);
418        }
419    }
420}

422/// Convert one cpal callback buffer of arbitrary sample type `T` and channel
423/// count to mono i16, appending into `scratch`. Downmix = average of channels.
424fn to_mono_i16<T>(data: &[T], channels: usize, scratch: &mut Vec<i16>)
425where
426    T: Sample,
427    i16: cpal::FromSample<T>,
428{
429    scratch.clear();
430    if channels <= 1 {
431        scratch.extend(data.iter().map(|&s| i16::from_sample(s)));
432        return;
433    }
434    for frame in data.chunks_exact(channels) {
435        // Average in i32 to avoid i16 overflow, then clamp.
436        let sum: i32 = frame.iter().map(|&s| i16::from_sample(s) as i32).sum();
437        scratch.push((sum / channels as i32) as i16);
438    }
439}

445/// Microphone capture via cpal. The cpal `Stream` is `!Send` on CoreAudio, so
446/// it is created and held on the thread that calls [`Self::open`]; the ring it
447/// feeds is `Send`/`Sync` and is what [`AudioInput::next_frame`] reads.
448/// Dropping the capturer stops the stream and releases the device (on the
449/// owning thread — the `!Send` contract).
450pub(crate) struct CpalCapturer {
451    ring: Arc<Ring>,
452    /// Held to keep the device stream alive; dropped (released) with `self`.
453    _stream: cpal::Stream,
454    clock: hound_media::CaptureClock,
455    /// The device truth: what the device negotiated (we DELIVER 48 k mono).
456    negotiated: (SampleRate, ChannelCount),
457    /// The name of the device actually opened, for the hot-swap path to log
458    /// what it landed on. `None` if the device name could not be read.
459    opened_name: Option<String>,
460    /// Device rate → 48 kHz conversion (pass-through when the device is 48 k).
461    /// Runs on the consumer thread in [`AudioInput::next_frame`], NEVER in the
462    /// RT callback.
463    resampler: super::resample::MonoResampler,
464    /// Converted 48 kHz samples awaiting frame assembly (FIFO tail of the
465    /// resampler output that did not fill a whole 20 ms frame yet).
466    pending_48k: Vec<i16>,
467    /// Reusable device-rate pull buffer, so the ring pop allocates NOTHING inside
468    /// the mutex the RT capture callback also takes. See [`Ring::pop_blocking_into`].
469    pull_buf: Vec<i16>,
470    /// Drop total at the last emitted warning + when it was emitted — the overflow
471    /// warning is throttled so a wedged consumer does not itself become the log storm.
472    last_logged_drops: u64,
473    last_drop_log: Option<Instant>,
474}

476/// Minimum spacing between capture-overflow warnings. Overflow means the consumer
477/// thread was descheduled past the ring depth; the condition is bursty, so report the
478/// accumulated count on a slow heartbeat rather than per occurrence.
479const DROP_LOG_INTERVAL: std::time::Duration = std::time::Duration::from_secs(5);
480
481impl CpalCapturer {
482    /// Open the best available input device and begin capture, stamping frames on
483    /// `clock`. Tries [`input_candidates`] in order (named → Linux shared server
484    /// PCM → host default), so a busy/exclusive or vanished device falls through
485    /// to the next rather than failing the call — non-exclusive access by default.
486    pub(crate) fn open(
487        choice: &DeviceChoice<MicId>,
488        clock: hound_media::CaptureClock,
489    ) -> Result<Self, AudioError> {
490        let host = cpal::default_host();
491        let candidates = input_candidates(&host, choice);
492        if candidates.is_empty() {
493            return Err(AudioError::NoDevice("no input device".into()));
494        }
495        let mut last_err: Option<AudioError> = None;
496        for device in &candidates {
497            match Self::open_device(device, clock) {
498                Ok(cap) => return Ok(cap),
499                Err(e) => {
500                    tracing::warn!(
501                        target: "audio_endpoint",
502                        device = %device.to_string(),
503                        "input open failed ({e}); trying next candidate"
504                    );
505                    last_err = Some(e);
506                }
507            }
508        }
509        Err(last_err.unwrap_or_else(|| AudioError::NoDevice("no input device".into())))
510    }
511
512    /// Open one specific device at its native format. See [`Self::open`] for the
513    /// candidate-selection wrapper.
514    fn open_device(
515        device: &cpal::Device,
516        clock: hound_media::CaptureClock,
517    ) -> Result<Self, AudioError> {
518        let opened_name = Some(device.to_string());
519
520        // Use the device's default config for the sample FORMAT (built-in mics
521        // are often f32), but request mono and 48 kHz where possible.
522        let default_cfg = device
523            .default_input_config()
524            .map_err(|e| map_cpal_err("query default input config", DeviceKind::Microphone, e))?;
525        let sample_format = default_cfg.sample_format();
526        let dev_channels = default_cfg.channels();
527        let dev_rate = default_cfg.sample_rate();
528
529        // We keep the device's native channel count and rate (downmix to mono
530        // in the RT callback), then RESAMPLE to the engine's 48 kHz on the
531        // consumer thread in `next_frame`. Asking for a hard mono/48k config
532        // that the device does not support would fail the stream build
533        // (AirPods mics are 24 kHz, many defaults 44.1 kHz); mislabelling the
534        // rate instead is the chipmunk/deep-voice bug. The delivered contract
535        // is always 48 k mono.
536        let config = cpal::StreamConfig {
537            channels: dev_channels,
538            sample_rate: dev_rate,
539            buffer_size: cpal::BufferSize::Default,
540        };
541        let negotiated_rate = SampleRate::try_new(dev_rate)
542            .ok_or_else(|| AudioError::UnsupportedFormat("device reports 0 Hz".into()))?;
543        let resampler = super::resample::MonoResampler::new(dev_rate, SAMPLE_RATE.hz())?;
544        if dev_rate != SAMPLE_RATE.hz() {
545            tracing::info!(
546                target: "audio_endpoint",
547                device_rate = dev_rate,
548                "capture device is not 48 kHz — resampling to the engine rate"
549            );
550        }
551
552        let ring = Ring::new();
553        let channels = dev_channels as usize;
554
555        // Unlike the OUTPUT side, capture rebuilds on EVERY error kind — including an
556        // automatic reroute. The device rate and the `MonoResampler` above are captured
557        // at open and drive the 48 kHz delivery contract, so a stream that silently
558        // rerouted to a 24 kHz HFP mic would keep resampling as if it were still 48 k
559        // (the chipmunk/deep-voice bug). Only the LOG LEVEL is classified here; stopping
560        // the ring unblocks the consumer, which reopens.
561        let err_ring = Arc::clone(&ring);
562        let error_cb = move |e: cpal::Error| {
563            use cpal::ErrorKind;
564            match e.kind() {
565                ErrorKind::DeviceChanged => tracing::info!(
566                    target: "audio_endpoint",
567                    "cpal input auto-rerouted ({e}) — reopening to re-derive the capture rate"
568                ),
569                ErrorKind::StreamInvalidated => tracing::warn!(
570                    target: "audio_endpoint",
571                    "cpal input stream invalidated ({e}) — reopening"
572                ),
573                ErrorKind::DeviceNotAvailable
574                | ErrorKind::DeviceBusy
575                | ErrorKind::HostUnavailable
576                | ErrorKind::PermissionDenied
577                | ErrorKind::BackendError => {
578                    tracing::error!(target: "audio_endpoint", "cpal input stream error: {e}")
579                }
580                // `ErrorKind` is `#[non_exhaustive]` — unclassified kinds still reopen.
581                _ => tracing::warn!(
582                    target: "audio_endpoint",
583                    "cpal input stream error ({e}) — reopening"
584                ),
585            }
586            // A device-lost mid-stream unblocks the consumer (returns Err on
587            // the next next_frame via the stopped ring).
588            err_ring.stop();
589        };
590
591        // Build the stream for the device's actual sample format, converting
592        // to mono i16 in the RT callback. Each arm reuses a per-callback
593        // scratch Vec captured by the closure — allocated ONCE at build time
594        // and reused; the only growth is bounded by the burst size.
595        let mut scratch: Vec<i16> = Vec::with_capacity(RING_CAPACITY_SAMPLES);
596        let cb_ring = Arc::clone(&ring);
597        let stream = match sample_format {
598            SampleFormat::F32 => device.build_input_stream(
599                config,
600                move |data: &[f32], _| {
601                    to_mono_i16(data, channels, &mut scratch);
602                    cb_ring.push_capture(&scratch);
603                },
604                error_cb,
605                None,
606            ),
607            SampleFormat::I16 => device.build_input_stream(
608                config,
609                move |data: &[i16], _| {
610                    to_mono_i16(data, channels, &mut scratch);
611                    cb_ring.push_capture(&scratch);
612                },
613                error_cb,
614                None,
615            ),
616            SampleFormat::U16 => device.build_input_stream(
617                config,
618                move |data: &[u16], _| {
619                    to_mono_i16(data, channels, &mut scratch);
620                    cb_ring.push_capture(&scratch);
621                },
622                error_cb,
623                None,
624            ),
625            other => {
626                return Err(AudioError::UnsupportedFormat(format!(
627                    "unsupported input sample format {other:?}"
628                )));
629            }
630        }
631        .map_err(|e| map_cpal_err("build input stream", DeviceKind::Microphone, e))?;
632
633        stream
634            .play()
635            .map_err(|e| map_cpal_err("start input stream", DeviceKind::Microphone, e))?;
636
637        Ok(Self {
638            ring,
639            _stream: stream,
640            clock,
641            negotiated: (negotiated_rate, CHANNELS), // device truth; we DELIVER 48k mono
642            opened_name,
643            resampler,
644            pending_48k: Vec::new(),
645            pull_buf: Vec::with_capacity(RING_CAPACITY_SAMPLES),
646            last_logged_drops: 0,
647            last_drop_log: None,
648        })
649    }
650
651    /// The name of the input device actually opened. `None` if the device
652    /// name could not be read.
653    pub(crate) fn opened_name(&self) -> Option<&str> {
654        self.opened_name.as_deref()
655    }
656
657    /// Emit a throttled warning when the capture ring has dropped more samples since
658    /// the last report. Called once per assembled frame on the (non-RT) consumer
659    /// thread; the cumulative count itself is read via
660    /// [`AudioInput::capture_overflow_samples`].
661    fn note_overflow(&mut self, now: Instant) {
662        let total = self.ring.dropped_total();
663        if total <= self.last_logged_drops {
664            return;
665        }
666        if self
667            .last_drop_log
668            .is_some_and(|at| now.duration_since(at) < DROP_LOG_INTERVAL)
669        {
670            return;
671        }
672        let new_drops = total - self.last_logged_drops;
673        self.last_logged_drops = total;
674        self.last_drop_log = Some(now);
675        tracing::warn!(
676            target: "audio_endpoint",
677            dropped_samples = new_drops,
678            dropped_total = total,
679            "capture ring overflow — the mic worker thread cannot keep up; the FAR end \
680             hears a chopped word (assembled frames stay timestamp-contiguous over the \
681             hole, so no downstream gauge can see it)"
682        );
683    }
684}

688impl AudioInput for CpalCapturer {
689    fn next_frame(&mut self) -> Result<PcmFrame, AudioError> {
690        // Pull device-rate samples and convert until one full 48 kHz frame is
691        // pending. Pass-through devices pull exactly one frame; resampled
692        // devices pull in the resampler's 20 ms input blocks.
693        while self.pending_48k.len() < SAMPLES_PER_FRAME {
694            let pull = if self.resampler.is_passthrough() {
695                SAMPLES_PER_FRAME
696            } else {
697                self.resampler.chunk_in()
698            };
699            // Pop into the reusable buffer — nothing allocates under the ring mutex
700            // the RT capture callback also takes.
701            let mut pull_buf = std::mem::take(&mut self.pull_buf);
702            let ok = self.ring.pop_blocking_into(pull, &mut pull_buf);
703            let converted = if ok {
704                self.resampler
705                    .process_into(&pull_buf, &mut self.pending_48k)
706            } else {
707                Ok(())
708            };
709            self.pull_buf = pull_buf;
710            if !ok {
711                return Err(AudioError::Device("capture stream stopped".into()));
712            }
713            converted?;
714        }
715        // Report any capture-ring overflow (samples the RT callback had to drop
716        // because this thread fell behind) — the send-side audible hole.
717        self.note_overflow(Instant::now());
718        let samples: Vec<i16> = self.pending_48k.drain(..SAMPLES_PER_FRAME).collect();
719        // Stamp at assembly time (within one frame interval of the device
720        // callback; a future refinement carries the callback's host time).
721        Ok(PcmFrame::at_engine_rate(
722            samples,
723            TimestampUs::from_micros(self.clock.now()),
724        ))
725    }
726
727    fn negotiated(&self) -> (SampleRate, ChannelCount) {
728        self.negotiated
729    }
730
731    fn capture_overflow_samples(&self) -> u64 {
732        self.ring.dropped_total()
733    }
734}

748/// Speaker playback via cpal. The output callback drains the ring; the
749/// trait's [`AudioOutput::play`] pushes into it (non-blocking, bounded).
750/// Dropping the sink stops the stream and releases the device.
751pub(crate) struct CpalPlayback {
752    /// Producer half of the wait-free playback ring (the RT callback owns the consumer).
753    out: PlaybackProducer,
754    /// Held to keep the device stream alive; dropped (released) with `self`.
755    _stream: cpal::Stream,
756    /// 48 kHz → device-rate conversion (pass-through on 48 k devices). Runs in
757    /// [`AudioOutput::play`] on the caller's thread, NEVER in the RT callback
758    /// (the ring stays device-rate mono).
759    resampler: super::resample::MonoResampler,
760    /// Reused conversion output buffer (no per-frame allocation).
761    scratch: Vec<i16>,
762    opened_name: Option<String>,
763    /// The device's sample rate — to convert the ring's buffered sample count to
764    /// milliseconds for the mix bus's output-cushion controller.
765    dev_rate: u32,
766}
767
768impl CpalPlayback {
769    /// Open the best available output device at the engine format (resampling to
770    /// the device rate as needed). Tries [`output_candidates`] in order (named →
771    /// Linux shared server PCM → host default), so a busy/exclusive or vanished
772    /// device falls through rather than failing — non-exclusive access by default.
773    pub(crate) fn open(choice: &DeviceChoice<SpeakerId>) -> Result<Self, AudioError> {
774        let host = cpal::default_host();
775        let candidates = output_candidates(&host, choice);
776        if candidates.is_empty() {
777            return Err(AudioError::NoDevice("no output device".into()));
778        }
779        let mut last_err: Option<AudioError> = None;
780        for device in &candidates {
781            match Self::open_device(device) {
782                Ok(pb) => return Ok(pb),
783                Err(e) => {
784                    tracing::warn!(
785                        target: "audio_endpoint",
786                        device = %device.to_string(),
787                        "output open failed ({e}); trying next candidate"
788                    );
789                    last_err = Some(e);
790                }
791            }
792        }
793        Err(last_err.unwrap_or_else(|| AudioError::NoDevice("no output device".into())))
794    }
795
796    /// Open one specific output device at its native format. See [`Self::open`]
797    /// for the candidate-selection wrapper.
798    fn open_device(device: &cpal::Device) -> Result<Self, AudioError> {
799        let opened_name = Some(device.to_string());
800
801        let default_cfg = device
802            .default_output_config()
803            .map_err(|e| map_cpal_err("query default output config", DeviceKind::Speaker, e))?;
804        let sample_format = default_cfg.sample_format();
805        let dev_channels = default_cfg.channels();
806        let dev_rate = default_cfg.sample_rate();
807
808        let config = cpal::StreamConfig {
809            channels: dev_channels,
810            sample_rate: dev_rate,
811            buffer_size: cpal::BufferSize::Default,
812        };
813        // Engine frames are 48 kHz; the ring (and the RT callback that drains
814        // it) runs at the DEVICE rate. `play` converts on the way in — a
815        // 44.1 k output device otherwise plays 48 k samples slow ("deep
816        // voice").
817        let resampler = super::resample::MonoResampler::new(SAMPLE_RATE.hz(), dev_rate)?;
818        // Log the rate on EVERY open, not only the resampled case: after a route change
819        // the reopen's rate is the one fact that says which device we actually landed on
820        // (and whether the cushion geometry moved).
821        tracing::debug!(
822            target: "audio_endpoint",
823            device = opened_name.as_deref().unwrap_or("?"),
824            device_rate = dev_rate,
825            channels = dev_channels,
826            ?sample_format,
827            "output device opened"
828        );
829        if dev_rate != SAMPLE_RATE.hz() {
830            tracing::info!(
831                target: "audio_endpoint",
832                device_rate = dev_rate,
833                "output device is not 48 kHz — resampling from the engine rate"
834            );
835        }
836
837        let (producer, mut consumer) = playback_ring(RING_CAPACITY_SAMPLES);
838        let channels = dev_channels as usize;
839
840        // Classify, don't blanket-kill. `DeviceChanged` is INFORMATIONAL — cpal's own
841        // contract is "the stream was automatically rerouted to a different device […]
842        // the stream remains active and no rebuild is required". With
843        // `DeviceChoice::Default` CoreAudio hands us a DefaultOutput AudioUnit that has
844        // ALREADY followed the route and is still playing. Treating it as death is what
845        // turned one AirPods route flap (5 changes in 50 s) into five blocking device
846        // reopens on the RT mix thread.
847        //
848        // The reroute is not free either: `dev_rate` + the resampler below still describe
849        // the OLD device, so the bus must re-sync ONCE after the flap settles — hence a
850        // separate flag rather than silence. Every other kind is a real fault: the stream
851        // is gone and only a rebuild recovers it.
852        let err_shared = Arc::clone(&producer.shared);
853        let error_cb = move |e: cpal::Error| {
854            use cpal::ErrorKind;
855            match e.kind() {
856                ErrorKind::DeviceChanged => {
857                    tracing::info!(
858                        target: "audio_endpoint",
859                        "cpal output auto-rerouted to the new default device ({e}) — stream \
860                         still live; the mix bus will re-sync its device geometry once the \
861                         route settles"
862                    );
863                    err_shared.route_changed.store(true, Ordering::Relaxed);
864                }
865                // The negotiated config is no longer valid: expected on a genuine device
866                // swap, recovered by the reopen — noisy, not alarming.
867                ErrorKind::StreamInvalidated => {
868                    tracing::warn!(target: "audio_endpoint", "cpal output stream invalidated ({e}) — reopening");
869                    err_shared.output_dead.store(true, Ordering::Relaxed);
870                }
871                // The device or the host itself is gone/unusable. Worth an event (the
872                // Sentry bridge promotes `error!`), because playout is down until it returns.
873                ErrorKind::DeviceNotAvailable
874                | ErrorKind::DeviceBusy
875                | ErrorKind::HostUnavailable
876                | ErrorKind::PermissionDenied
877                | ErrorKind::BackendError => {
878                    tracing::error!(target: "audio_endpoint", "cpal output stream error: {e}");
879                    err_shared.output_dead.store(true, Ordering::Relaxed);
880                }
881                // `ErrorKind` is `#[non_exhaustive]`: an unclassified kind is treated as
882                // fatal (reopen) but logged as a warning, not an event.
883                _ => {
884                    tracing::warn!(target: "audio_endpoint", "cpal output stream error ({e}) — reopening");
885                    err_shared.output_dead.store(true, Ordering::Relaxed);
886                }
887            }
888        };
889
890        // The output callback wants `channels`-interleaved samples; our ring
891        // is mono, so each mono sample is duplicated across the device's
892        // channels. A per-callback mono scratch (allocated once, reused) feeds
893        // the upmix. The consumer is moved into the (single) matched callback.
894        let mut mono_scratch: Vec<i16> = Vec::with_capacity(RING_CAPACITY_SAMPLES);
895        let stream = match sample_format {
896            SampleFormat::F32 => device.build_output_stream(
897                config,
898                move |out: &mut [f32], _| {
899                    let frames = out.len() / channels.max(1);
900                    mono_scratch.resize(frames, 0);
901                    consumer.pop_playback(&mut mono_scratch);
902                    for (frame, &m) in out.chunks_mut(channels).zip(mono_scratch.iter()) {
903                        let v = f32::from_sample(m);
904                        frame.iter_mut().for_each(|s| *s = v);
905                    }
906                },
907                error_cb,
908                None,
909            ),
910            SampleFormat::I16 => device.build_output_stream(
911                config,
912                move |out: &mut [i16], _| {
913                    let frames = out.len() / channels.max(1);
914                    mono_scratch.resize(frames, 0);
915                    consumer.pop_playback(&mut mono_scratch);
916                    for (frame, &m) in out.chunks_mut(channels).zip(mono_scratch.iter()) {
917                        frame.iter_mut().for_each(|s| *s = m);
918                    }
919                },
920                error_cb,
921                None,
922            ),
923            SampleFormat::U16 => device.build_output_stream(
924                config,
925                move |out: &mut [u16], _| {
926                    let frames = out.len() / channels.max(1);
927                    mono_scratch.resize(frames, 0);
928                    consumer.pop_playback(&mut mono_scratch);
929                    for (frame, &m) in out.chunks_mut(channels).zip(mono_scratch.iter()) {
930                        let v = u16::from_sample(m);
931                        frame.iter_mut().for_each(|s| *s = v);
932                    }
933                },
934                error_cb,
935                None,
936            ),
937            other => {
938                return Err(AudioError::UnsupportedFormat(format!(
939                    "unsupported output sample format {other:?}"
940                )));
941            }
942        }
943        .map_err(|e| map_cpal_err("build output stream", DeviceKind::Speaker, e))?;
944
945        stream
946            .play()
947            .map_err(|e| map_cpal_err("start output stream", DeviceKind::Speaker, e))?;
948
949        Ok(Self {
950            out: producer,
951            _stream: stream,
952            resampler,
953            scratch: Vec::new(),
954            opened_name,
955            dev_rate,
956        })
957    }
958
959    /// The name of the output device actually opened.
960    pub(crate) fn opened_name(&self) -> Option<&str> {
961        self.opened_name.as_deref()
962    }
963}

970    }

972    fn take_output_overruns(&self) -> u64 {
973        self.out.take_output_overruns()
974    }
975
976    fn buffered_ms(&self) -> Option<u32> {
977        let rate = self.dev_rate.max(1);
978        Some((self.out.buffered() as u64 * 1000 / rate as u64) as u32)
979    }
980
981    fn is_dead(&self) -> bool {
982        self.out.is_output_dead()
983    }

987    }

990        self.dev_rate

992
993    fn play(&mut self, frame: &PcmFrame) -> Result<(), AudioError> {
994        // Convert the 48 kHz engine frame to the device rate (pass-through on
995        // 48 k devices), then non-blocking push into the bounded ring; the RT
996        // output callback drains it. Mono in, mono ring (upmixed to device
997        // channels in the callback).
998        if self.resampler.is_passthrough() {
999            self.out.push_playback(&frame.samples);
1000        } else {
1001            self.scratch.clear();
1002            self.resampler
1003                .process_into(&frame.samples, &mut self.scratch)?;
1004            self.out.push_playback(&self.scratch);
1005        }
1006        Ok(())
1007    }
1008}

1146        ring.stop();
