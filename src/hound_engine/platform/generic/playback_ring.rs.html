1//! The wait-free SPSC playback ring between the mix thread and the cpal RT
2//! output callback, plus the output-stream error classification/demotion policy
3//! the callback applies ([`apply_output_error`]).
4
5use std::sync::atomic::{AtomicBool, AtomicU64, Ordering};
6use std::sync::{Arc, Mutex};
7
8// ---------------------------------------------------------------------------
9// Playback ring: wait-free SPSC.
10//
11// The PLAYBACK path has exactly one producer (the mix thread, via `play`) and one
12// consumer (the CoreAudio RT output callback), so it uses a wait-free SPSC ring
13// (`rtrb`) rather than the mutex-guarded `Ring` above. The RT output callback then
14// NEVER takes a lock — a mutex-guarded ring could block it if the mix thread were
15// preempted mid-push, a priority inversion whenever the mix thread is not real-time
16// (RT promotion is best-effort and fails with no audio entitlement). "Audio is
17// sacred": the device thread always makes progress, padding silence on a dry ring
18// instead of waiting. Capture keeps the mutex `Ring` (its consumer blocks on a condvar,
19// which SPSC can't express, and it is not the latency-critical output path).
20// ---------------------------------------------------------------------------
21
22/// Diagnostics shared across the playback ring's three reachers: the mix thread
23/// (owner), the RT output callback (counts underrun), and the stream error callback
24/// (marks death). Plain atomics behind an `Arc`.
25pub(super) struct PlaybackShared {
26    /// Cumulative output samples the RT callback silence-padded because the ring ran
27    /// dry — the definitive end-of-pipeline AUDIBLE GAP. Steady state is ZERO; a rising
28    /// count IS the stutter, made measurable. Read+reset via [`PlaybackProducer::take_output_underruns`].
29    output_underrun_samples: AtomicU64,
30    /// Cumulative samples `push_playback` DROPPED because the ring was already full
31    /// — the OVERPRODUCTION direction of drift. Without it only the dry direction is
32    /// counted, so a call whose output latency has crept to the whole ring reads
33    /// clean. Read+reset via [`PlaybackProducer::take_output_overruns`].
34    output_drop_samples: AtomicU64,
35    /// Set by the OUTPUT stream's error callback for a FATAL error (the device
36    /// vanished, the config was invalidated, the backend failed): the cpal stream is
37    /// then dead but the `CpalPlayback` lives on, so the mix bus polls
38    /// [`PlaybackProducer::is_output_dead`] each tick and reopens the device.
39    ///
40    /// NOT set for an automatic reroute — see [`Self::route_changed`].
41    output_dead: AtomicBool,
42    /// Set by the OUTPUT stream's error callback for any NON-FATAL condition — the
43    /// stream is still live, but the sink deserves a debounced re-open. Two sources:
44    ///
45    /// * [`cpal::ErrorKind::DeviceChanged`] — the OS moved the default output (AirPods
46    ///   connecting/disconnecting) and cpal **already rerouted the running stream**,
47    ///   which is still playing. What is stale is our GEOMETRY:
48    ///   [`CpalPlayback::dev_rate`](super::cpal_io::CpalPlayback) and the 48 kHz→device resampler still describe the
49    ///   OLD device, so `buffered_ms()` mis-scales the mix bus's cushion regulator
50    ///   across e.g. a 48 k→24 k hop.
51    /// * [`cpal::ErrorKind::Xrun`] / [`cpal::ErrorKind::RealtimeDenied`] — see
52    ///   [`Self::soft_faults`].
53    ///
54    /// The mix bus debounces these into ONE re-open once they stop arriving. Read+
55    /// cleared via [`PlaybackProducer::take_route_changed`].
56    route_changed: AtomicBool,
57    /// Cumulative NON-FATAL output-stream faults ([`cpal::ErrorKind::Xrun`], a buffer
58    /// under/overrun; [`cpal::ErrorKind::RealtimeDenied`], a refused real-time
59    /// promotion for the device's own callback thread). Both describe audio QUALITY on
60    /// a stream that is still running, so they are DEMOTED into [`Self::route_changed`]
61    /// rather than [`Self::output_dead`]: killing the stream would answer a glitch with
62    /// a blocking device open on the RT mix thread plus a fresh cushion prefill — i.e.
63    /// a bigger glitch. The debounced re-open the bus does eventually perform is a
64    /// plausible remedy (a fresh cushion), and a sustained storm never goes quiet so it
65    /// never triggers one at all, which is the right behaviour.
66    ///
67    /// Steady state 0; a rising count is "the device is glitching without dying".
68    /// Read+reset via [`PlaybackProducer::take_soft_faults`].
69    soft_faults: AtomicU64,
70    /// Latches the first soft-fault LOG line for this stream instance — a backend that
71    /// reports an xrun per period would otherwise storm the log. The count still rises.
72    soft_fault_reported: AtomicBool,
73    /// Latches the first fatal-error report for this stream instance, so a backend
74    /// that re-reports the same fault on every worker iteration (ALSA does, until
75    /// the stream is dropped) logs once instead of storming.
76    error_reported: AtomicBool,
77    /// That first error's message, for the mix bus to quote if the outage it opens
78    /// turns out to be UNRECOVERABLE — the escalation is the only report that
79    /// reaches Sentry, so it has to carry the cause. Written at most once per
80    /// stream (guarded by [`Self::error_reported`]) and only via `try_lock`: the
81    /// error callback can run on the backend's audio worker thread and must never
82    /// wait on the bus. Read+cleared via [`PlaybackProducer::take_stream_error`].
83    first_error: Mutex<Option<String>>,
84}
85
86/// What the output stream's error callback does with one `cpal::Error`.
87///
88/// Split out from the callback so the demotion policy is unit-testable without a live
89/// device — the kinds that must NOT kill the stream are exactly the interesting part.
90#[derive(Debug, Clone, Copy, PartialEq, Eq)]
91pub(super) enum OutputErrorClass {
92    /// The OS moved the default output and cpal already rerouted the LIVE stream.
93    /// Nothing died; the sink's geometry went stale ⇒ debounced re-sync.
94    Rerouted,
95    /// The stream is still live but reported a quality fault: a buffer under/overrun
96    /// (`Xrun`) or a refused real-time promotion (`RealtimeDenied`). Rides the same
97    /// debounce as a reroute — see [`PlaybackShared::soft_faults`] for why.
98    SoftFault,
99    /// Fatal to THIS stream: only a rebuild recovers it.
100    Fatal,
101}
102
103/// Classify one output-stream error.
104///
105/// `ErrorKind` is `#[non_exhaustive]`, so anything unrecognised is FATAL — the
106/// conservative direction, because a needless reopen costs one cushion prefill while a
107/// missed death costs the call's entire audio.
108pub(super) fn classify_output_error(kind: cpal::ErrorKind) -> OutputErrorClass {
109    match kind {
110        cpal::ErrorKind::DeviceChanged => OutputErrorClass::Rerouted,
111        cpal::ErrorKind::Xrun | cpal::ErrorKind::RealtimeDenied => OutputErrorClass::SoftFault,
112        _ => OutputErrorClass::Fatal,
113    }
114}
115
116/// Fold one output-stream error into the shared state the mix bus polls each tick.
117///
118/// Runs on the backend's audio worker thread (CoreAudio's listener, ALSA's poll loop,
119/// the WASAPI event thread), so it only touches atomics and `try_lock`s — it must never
120/// wait on the bus.
121pub(super) fn apply_output_error(shared: &PlaybackShared, e: &cpal::Error) {
122    match classify_output_error(e.kind()) {
123        OutputErrorClass::Rerouted => {
124            tracing::info!(
125                target: "audio_endpoint",
126                "cpal output auto-rerouted to the new default device ({e}) — stream \
127                 still live; the mix bus will re-sync its device geometry once the \
128                 route settles"
129            );
130            shared.route_changed.store(true, Ordering::Relaxed);
131        }
132        OutputErrorClass::SoftFault => {
133            shared.soft_faults.fetch_add(1, Ordering::Relaxed);
134            // Same debounce funnel as a reroute: arm it, let the bus coalesce.
135            shared.route_changed.store(true, Ordering::Relaxed);
136            if !shared.soft_fault_reported.swap(true, Ordering::Relaxed) {
137                tracing::info!(
138                    target: "audio_endpoint",
139                    kind = ?e.kind(),
140                    "cpal output reported a non-fatal fault ({e}) — stream still live, \
141                     NOT reopening on the spot; the mix bus will re-sync once these \
142                     settle (further occurrences are counted, not logged)"
143                );
144            }
145        }
146        OutputErrorClass::Fatal => {
147            // Hand it to the bus's reopen path, which polls `is_dead` every 10 ms tick.
148            shared.output_dead.store(true, Ordering::Relaxed);
149            if shared.error_reported.swap(true, Ordering::Relaxed) {
150                return; // already reported for this stream — see `error_reported`
151            }
152            let msg = e.to_string();
153            tracing::warn!(
154                target: "audio_endpoint",
155                kind = ?e.kind(),
156                "cpal output stream error ({msg}) — the mix bus will reopen the device"
157            );
158            if let Ok(mut slot) = shared.first_error.try_lock() {
159                *slot = Some(msg);
160            }
161        }
162    }
163}

165/// The mix-thread (sole producer) half of the playback ring + shared diagnostics.
166pub(super) struct PlaybackProducer {
167    producer: rtrb::Producer<i16>,
168    capacity: usize,
169    pub(super) shared: Arc<PlaybackShared>,
170}
171
172/// The RT-output-callback (sole consumer) half + shared diagnostics.
173pub(super) struct PlaybackConsumer {
174    consumer: rtrb::Consumer<i16>,
175    shared: Arc<PlaybackShared>,
176}
177
178/// Build a wait-free SPSC playback ring of `capacity` mono samples.
179pub(super) fn playback_ring(capacity: usize) -> (PlaybackProducer, PlaybackConsumer) {
180    let (producer, consumer) = rtrb::RingBuffer::<i16>::new(capacity);
181    let shared = Arc::new(PlaybackShared {
182        output_underrun_samples: AtomicU64::new(0),
183        output_drop_samples: AtomicU64::new(0),
184        output_dead: AtomicBool::new(false),
185        route_changed: AtomicBool::new(false),
186        soft_faults: AtomicU64::new(0),
187        soft_fault_reported: AtomicBool::new(false),
188        error_reported: AtomicBool::new(false),
189        first_error: Mutex::new(None),
190    });
191    (
192        PlaybackProducer {
193            producer,
194            capacity,
195            shared: Arc::clone(&shared),
196        },
197        PlaybackConsumer { consumer, shared },
198    )
199}
200
201impl PlaybackProducer {
202    /// Mix thread: push mono samples for the RT callback to drain, returning how many
203    /// were ACCEPTED. Wait-free; on a full ring (the bus outran the device) the tail is
204    /// dropped rather than blocking, and counted — the caller must not treat dropped
205    /// samples as played (the mix bus's AEC render reference depends on it).
206    pub(super) fn push_playback(&mut self, samples: &[i16]) -> usize {
207        let mut n = 0;
208        for &s in samples {
209            if self.producer.push(s).is_err() {
210                break; // full: drop the rest (overproduction)
211            }
212            n += 1;
213        }
214        let dropped = samples.len() - n;
215        if dropped > 0 {
216            self.shared
217                .output_drop_samples
218                .fetch_add(dropped as u64, Ordering::Relaxed);
219        }
220        n
221    }
222
223    /// Buffered (unplayed) samples = capacity − free write slots. Coarse (the consumer
224    /// may pop concurrently); the mix bus uses it only for the ms-scale output cushion.
225    pub(super) fn buffered(&self) -> usize {
226        self.capacity.saturating_sub(self.producer.slots())
227    }
228
229    /// Read and reset the dropped-sample counter (samples the ring was too full to
230    /// accept). Polled by the bus each tick alongside the underrun gauge.
231    pub(super) fn take_output_overruns(&self) -> u64 {
232        self.shared.output_drop_samples.swap(0, Ordering::Relaxed)
233    }
234
235    /// Read and reset the underrun-sample counter (the silence the RT callback inserted
236    /// on a dry ring). Polled by the bus each tick for the live "playout underruns" stat.
237    pub(super) fn take_output_underruns(&self) -> u64 {
238        self.shared
239            .output_underrun_samples
240            .swap(0, Ordering::Relaxed)
241    }
242
243    /// Whether the output stream has errored FATALLY (device gone / config invalidated)
244    /// → reopen. An automatic reroute is not a death; see [`Self::take_route_changed`].
245    pub(super) fn is_output_dead(&self) -> bool {
246        self.shared.output_dead.load(Ordering::Relaxed)
247    }
248
249    /// Read+clear the "this live stream reported something non-fatal" flag (an OS
250    /// reroute, or a demoted xrun / refused RT promotion). Edge-triggered on purpose:
251    /// the bus turns each edge into a debounce extension, so a burst coalesces into one
252    /// re-sync. See [`PlaybackShared::route_changed`].
253    pub(super) fn take_route_changed(&self) -> bool {
254        self.shared.route_changed.swap(false, Ordering::Relaxed)
255    }
256
257    /// Read+clear the demoted non-fatal fault count (xrun / refused RT promotion). The
258    /// bus folds it into the call-level counter each tick. See
259    /// [`PlaybackShared::soft_faults`].
260    pub(super) fn take_soft_faults(&self) -> u64 {
261        self.shared.soft_faults.swap(0, Ordering::Relaxed)
262    }
263
264    /// Read+clear the message of the fatal stream error that killed this stream, if
265    /// any. Called by the mix bus on the tick that observes the death, so the reason
266    /// travels with the outage rather than living only in a log line the escalation
267    /// cannot quote.
268    pub(super) fn take_stream_error(&self) -> Option<String> {
269        self.shared
270            .first_error
271            .lock()
272            .ok()
273            .and_then(|mut slot| slot.take())
274    }
275}
276
277impl PlaybackConsumer {
278    /// RT output callback: pop up to `out.len()` mono samples (front-filled), silence-pad
279    /// the remainder on underrun, and count the gap. Wait-free — never locks, never blocks.
280    pub(super) fn pop_playback(&mut self, out: &mut [i16]) {
281        let mut n = 0;
282        while n < out.len() {
283            match self.consumer.pop() {
284                Ok(s) => {
285                    out[n] = s;
286                    n += 1;
287                }
288                Err(_) => break, // ring dry: the rest is silence
289            }
290        }
291        for slot in &mut out[n..] {
292            *slot = 0;
293        }
294        let padded = out.len() - n;
295        if padded > 0 {
296            self.shared
297                .output_underrun_samples
298                .fetch_add(padded as u64, Ordering::Relaxed);
299        }
300    }
301}

303/// The output error-callback demotion policy: which cpal kinds kill the stream and
304/// which ones only arm the mix bus's debounced re-sync. Exercised through the same
305/// shared state the live callback writes, so the tests bind the real behaviour rather
306/// than a restatement of the `match`.
307#[cfg(test)]
308mod output_error_demotion_tests {
309    use std::sync::atomic::Ordering;
310
311    use super::super::cpal_io::RING_CAPACITY_SAMPLES;
312    use super::{
313        OutputErrorClass, PlaybackProducer, apply_output_error, classify_output_error,
314        playback_ring,
315    };
316
317    /// Every `ErrorKind` cpal 0.18 defines that is NOT one of the three informational
318    /// ones. Listed explicitly (rather than derived) so a new upstream variant shows up
319    /// as a deliberate classification decision here.
320    const FATAL_KINDS: &[cpal::ErrorKind] = &[
321        cpal::ErrorKind::DeviceBusy,
322        cpal::ErrorKind::DeviceNotAvailable,
323        cpal::ErrorKind::HostUnavailable,
324        cpal::ErrorKind::InvalidInput,
325        cpal::ErrorKind::PermissionDenied,
326        cpal::ErrorKind::ResourceExhausted,
327        cpal::ErrorKind::StreamInvalidated,
328        cpal::ErrorKind::UnsupportedConfig,
329        cpal::ErrorKind::UnsupportedOperation,
330        cpal::ErrorKind::BackendError,
331        cpal::ErrorKind::Other,
332    ];
333
334    /// A producer half whose shared state the tests read through the same private
335    /// accessors the mix bus reaches via `AudioOutput`. The consumer is kept alive so
336    /// the ring is in its normal (un-abandoned) shape.
337    fn ring() -> (PlaybackProducer, super::PlaybackConsumer) {
338        playback_ring(RING_CAPACITY_SAMPLES)
339    }
340
341    /// The three kinds that arrive on a stream that is STILL PLAYING, and what each
342    /// means. `Xrun`/`RealtimeDenied` describe audio quality, not stream death.
343    #[test]
344    fn the_informational_kinds_are_classified_away_from_death() {
345        assert_eq!(
346            classify_output_error(cpal::ErrorKind::DeviceChanged),
347            OutputErrorClass::Rerouted
348        );
349        assert_eq!(
350            classify_output_error(cpal::ErrorKind::Xrun),
351            OutputErrorClass::SoftFault
352        );
353        assert_eq!(
354            classify_output_error(cpal::ErrorKind::RealtimeDenied),
355            OutputErrorClass::SoftFault
356        );
357    }
358
359    /// Everything else stays fatal — including `BackendError` (the ALSA `EIO` case) and
360    /// the `#[non_exhaustive]` catch-all. Demoting a real death would leave playout
361    /// silent for the rest of the call, so the default direction must stay "rebuild".
362    #[test]
363    fn every_other_kind_is_still_a_death() {
364        for &kind in FATAL_KINDS {
365            assert_eq!(
366                classify_output_error(kind),
367                OutputErrorClass::Fatal,
368                "{kind:?} must keep the immediate reopen path"
369            );
370        }
371    }
372
373    /// An xrun arms the debounce and is counted, but leaves the stream ALIVE: no
374    /// `is_dead`, so the bus never tears the device down for it. This is the whole
375    /// change — the old behaviour answered a reported glitch with a blocking device
376    /// open plus a fresh cushion prefill on the RT mix thread.
377    #[test]
378    fn an_xrun_arms_the_resync_without_killing_the_stream() {
379        let (prod, _cons) = ring();
380        apply_output_error(&prod.shared, &cpal::Error::new(cpal::ErrorKind::Xrun));
381
382        assert!(!prod.is_output_dead(), "an xrun is not a stream death");
383        assert!(prod.take_route_changed(), "it arms the debounced re-sync");
384        assert_eq!(prod.take_soft_faults(), 1, "and is counted");
385        assert!(
386            prod.take_stream_error().is_none(),
387            "a demoted fault leaves no fatal cause for the outage escalation to quote"
388        );
389    }
390
391    /// The same demotion for a refused real-time promotion: the device callback thread
392    /// runs at normal priority, which costs latency under load — not a dead stream.
393    #[test]
394    fn a_refused_realtime_promotion_is_demoted_too() {
395        let (prod, _cons) = ring();
396        apply_output_error(
397            &prod.shared,
398            &cpal::Error::new(cpal::ErrorKind::RealtimeDenied),
399        );
400
401        assert!(!prod.is_output_dead());
402        assert!(prod.take_route_changed());
403        assert_eq!(prod.take_soft_faults(), 1);
404    }
405
406    /// A backend that reports a fault per period (ALSA does) must not storm the log:
407    /// the info line latches after the first, while the COUNT keeps every event. The
408    /// counter is read-and-reset, matching the underrun/overrun gauges the bus polls.
409    #[test]
410    fn a_soft_fault_storm_counts_every_event_and_logs_once() {
411        let (prod, _cons) = ring();
412        for _ in 0..500 {
413            apply_output_error(&prod.shared, &cpal::Error::new(cpal::ErrorKind::Xrun));
414        }
415
416        assert!(
417            prod.shared.soft_fault_reported.load(Ordering::Relaxed),
418            "the log latch closes after the first fault"
419        );
420        assert!(!prod.is_output_dead(), "a storm is still not a death");
421        assert_eq!(prod.take_soft_faults(), 500, "every event is counted");
422        assert_eq!(prod.take_soft_faults(), 0, "the counter is read-and-reset");
423    }
424
425    /// A reroute arms the same re-sync but is NOT a quality fault — the two share the
426    /// funnel, not the counter, so `output_soft_faults` stays a clean device-glitch
427    /// signal across a Bluetooth flap.
428    #[test]
429    fn a_reroute_arms_the_resync_but_is_not_a_soft_fault() {
430        let (prod, _cons) = ring();
431        apply_output_error(
432            &prod.shared,
433            &cpal::Error::new(cpal::ErrorKind::DeviceChanged),
434        );
435
436        assert!(prod.take_route_changed());
437        assert_eq!(prod.take_soft_faults(), 0, "a reroute is not a glitch");
438        assert!(!prod.is_output_dead());
439    }
440
441    /// The other direction of the demotion, guarded: a REAL death still takes the
442    /// immediate path — marked dead, never folded into the debounce, and carrying the
443    /// cause so an outage that never recovers can name what broke.
444    #[test]
445    fn a_real_death_still_takes_the_immediate_path() {
446        let (prod, _cons) = ring();
447        apply_output_error(
448            &prod.shared,
449            &cpal::Error::with_message(
450                cpal::ErrorKind::BackendError,
451                "ALSA snd_pcm_avail_delay: I/O error (5)",
452            ),
453        );
454
455        assert!(prod.is_output_dead(), "a fatal error kills the stream");
456        assert!(
457            !prod.take_route_changed(),
458            "a death must not be demoted into the debounce"
459        );
460        assert_eq!(prod.take_soft_faults(), 0);
461        assert!(
462            prod.take_stream_error()
463                .is_some_and(|m| m.contains("I/O error")),
464            "the cause travels with the outage"
465        );
466    }
467}

469#[cfg(test)]
470mod ring_underrun_tests {
471    use super::super::cpal_io::RING_CAPACITY_SAMPLES;
472    use super::playback_ring;
473
474    /// When the playback RT callback asks for more samples than the ring holds, the
475    /// missing tail is silence-padded AND counted — the definitive end-of-pipeline
476    /// audible-gap signal. A fully-fed pop counts zero. The producer counter and the
477    /// consumer's pad share one atomic, so `take_output_underruns` sees the gap.
478    #[test]
479    fn pop_playback_counts_only_the_silence_padded_tail() {
480        let (mut prod, mut cons) = playback_ring(RING_CAPACITY_SAMPLES);
481        prod.push_playback(&[1, 2, 3, 4]);
482
483        // Ask for 4 with 4 available → no underrun.
484        let mut out = [0i16; 4];
485        cons.pop_playback(&mut out);
486        assert_eq!(out, [1, 2, 3, 4]);
487        assert_eq!(
488            prod.take_output_underruns(),
489            0,
490            "exact fill is not an underrun"
491        );
492
493        // Ask for 6 with 2 available → 4 silence-padded samples counted.
494        prod.push_playback(&[5, 6]);
495        let mut out6 = [0i16; 6];
496        cons.pop_playback(&mut out6);
497        assert_eq!(out6, [5, 6, 0, 0, 0, 0], "tail is silence");
498        assert_eq!(
499            prod.take_output_underruns(),
500            4,
501            "the 4 missing samples are the gap"
502        );
503        // take_* resets, so a subsequent fed pop is clean.
504        prod.push_playback(&[7]);
505        let mut out1 = [0i16; 1];
506        cons.pop_playback(&mut out1);
507        assert_eq!(prod.take_output_underruns(), 0, "counter reset on read");
508    }
509
510    /// A completely empty ring drained for a full callback is all-underrun (the
511    /// startup / device-stalled case): every requested sample is silence.
512    #[test]
513    fn empty_ring_pop_is_all_underrun() {
514        let (prod, mut cons) = playback_ring(RING_CAPACITY_SAMPLES);
515        let mut out = [0i16; 480];
516        cons.pop_playback(&mut out);
517        assert!(out.iter().all(|&s| s == 0));
518        assert_eq!(prod.take_output_underruns(), 480);
519    }
520
521    /// The OVERPRODUCTION direction is counted and reported, not silently discarded:
522    /// `push_playback` returns what it accepted and the rejected tail lands in the
523    /// overrun gauge. Without this a call whose output latency has crept to the full
524    /// ring still reads clean (only the dry direction had a counter).
525    #[test]
526    fn push_playback_counts_the_dropped_tail() {
527        let (mut prod, mut cons) = playback_ring(8);
528        assert_eq!(prod.push_playback(&[1, 2, 3, 4, 5, 6, 7, 8]), 8);
529        assert_eq!(
530            prod.take_output_overruns(),
531            0,
532            "an exact fill drops nothing"
533        );
534
535        // Ring full: every further sample is dropped and counted.
536        assert_eq!(prod.push_playback(&[9, 10, 11]), 0);
537        assert_eq!(prod.take_output_overruns(), 3);
538        assert_eq!(prod.take_output_overruns(), 0, "counter reset on read");
539
540        // Freeing two slots accepts exactly two of the next three.
541        let mut out = [0i16; 2];
542        cons.pop_playback(&mut out);
543        assert_eq!(out, [1, 2]);
544        assert_eq!(prod.push_playback(&[12, 13, 14]), 2);
545        assert_eq!(prod.take_output_overruns(), 1);
546    }
