1//! Level-aware multi-channel → mono capture downmix: averages by default, and
2//! emits a single dominant channel alone once it has held the floor for a full
3//! settle window (see [`Downmix`] for the −6 dB one-hot-device rationale).
4
5use cpal::Sample;
6
7/// Sustained per-channel level difference that means one channel IS the mic and
8/// the others are dead. 12 dB is far past any real stereo image (a hard-panned
9/// source still leaks into the other channel, and a real stereo capsule pair
10/// differs by a few dB at most), so only a genuinely unused channel gets here.
11const DOMINANCE_ENTER_DB: f64 = 12.0;
12
13/// Difference at which single-channel mode is abandoned again. Below
14/// [`DOMINANCE_ENTER_DB`] on purpose: a source hovering exactly at the entry
15/// threshold must not be able to toggle a 6 dB level step on and off.
16const DOMINANCE_EXIT_DB: f64 = 9.0;
17
18/// How long the evidence must hold before the downmix changes. ~1 s — longer than
19/// any syllable or inter-word gap, so nothing inside one utterance can flip it.
20const DOWNMIX_SETTLE_MS: f64 = 1000.0;
21
22/// Time constant of the per-channel energy estimate. Fast enough to characterise a
23/// device well inside [`DOWNMIX_SETTLE_MS`], slow enough that a pause between words
24/// is not mistaken for a dead channel.
25const DOWNMIX_ENERGY_TAU_MS: f64 = 200.0;
26
27/// Mean-square level below which no dominance judgement is made: both channels are
28/// effectively silent and the ratio is noise against noise. `(10^(−70/20)·32768)²`
29/// — i.e. −70 dBFS, under any real room's floor.
30const DOWNMIX_SILENCE_ENERGY: f64 = 107.0;
31
32/// How [`to_mono_i16`] folds a multi-channel device buffer to mono.
33#[derive(Clone, Copy, PartialEq, Eq, Debug)]
34enum DownmixMode {
35    /// Average every channel — the balanced (true stereo / dual-mono) case.
36    Average,
37    /// Emit this channel alone — one channel carries the signal.
38    Channel(usize),
39}
40
41/// Chooses [`to_mono_i16`]'s downmix from a slow per-channel energy estimate.
42///
43/// A plain channel average is a silent **−6 dB** whenever a multi-channel device
44/// carries the mic on ONE channel: USB interfaces with a single XLR occupied,
45/// "Stereo Mix", a mono capsule wired to the left input. The dead channel
46/// contributes only its own noise and halves the level — invisible in every gauge,
47/// and exactly the headroom the loudness chain is fighting for.
48///
49/// Real-time discipline: this runs in the cpal capture callback. All state
50/// (including the per-buffer scratch) is sized once at stream open, so the
51/// callback only reads and writes it — no allocation, no locking, no syscall. The
52/// mode is re-evaluated once per callback buffer and applied to the WHOLE buffer,
53/// so a switch can only ever land on a buffer boundary, never mid-waveform.
54pub(super) struct Downmix {
55    /// Smoothed mean-square per device channel, in FloatS16² units.
56    energy: Vec<f64>,
57    /// This buffer's per-channel sum-of-squares — the evidence `energy` integrates.
58    /// Owned (not a local array) so the channel count is unbounded and the callback
59    /// still allocates nothing.
60    sums: Vec<f64>,
61    /// The downmix in force.
62    mode: DownmixMode,
63    /// The mode recent evidence argues for, and how many sample-frames it has
64    /// argued for it. Equal to [`Self::mode`] ⇒ nothing pending.
65    pending: DownmixMode,
66    pending_frames: f64,
67    /// Sample-frames of sustained evidence needed to change [`Self::mode`].
68    settle_frames: f64,
69    /// EMA weight per sample-frame (device rate → [`DOWNMIX_ENERGY_TAU_MS`]).
70    alpha: f64,
71}
72
73impl Downmix {
74    /// Size the state for a device with `channels` channels at `sample_rate`.
75    /// Starts in [`DownmixMode::Average`]: unbalanced is the exception, and the
76    /// settle window has to elapse before the exception is believed.
77    pub(super) fn new(channels: usize, sample_rate: u32) -> Self {
78        let rate = f64::from(sample_rate.max(1));
79        Self {
80            energy: vec![0.0; channels],
81            sums: vec![0.0; channels],
82            mode: DownmixMode::Average,
83            pending: DownmixMode::Average,
84            pending_frames: 0.0,
85            settle_frames: rate * DOWNMIX_SETTLE_MS / 1000.0,
86            alpha: (1000.0 / (DOWNMIX_ENERGY_TAU_MS * rate)).min(1.0),
87        }
88    }
89
90    /// Integrate [`Self::sums`] over `frames` sample-frames and return the mode to
91    /// use for THIS buffer.
92    fn decide(&mut self, frames: usize) -> DownmixMode {
93        if frames == 0 {
94            return self.mode;
95        }
96        let frames_f = frames as f64;
97        // One EMA step covering the whole buffer: `alpha` is per-sample-frame, so a
98        // buffer of n frames moves the estimate n steps' worth (capped at a full
99        // replacement). Linear in n rather than exp(-n·alpha) — the same thing to
100        // within a few percent over a callback, without a transcendental per buffer.
101        let w = (self.alpha * frames_f).min(1.0);
102        for (e, &sum) in self.energy.iter_mut().zip(self.sums.iter()) {
103            *e += w * (sum / frames_f - *e);
104        }
105
106        let want = self.candidate();
107        if want == self.mode {
108            // Evidence agrees with the status quo: any pending switch is abandoned,
109            // so a flapping source can never accumulate its way across the window.
110            self.pending = self.mode;
111            self.pending_frames = 0.0;
112        } else {
113            if want != self.pending {
114                self.pending = want;
115                self.pending_frames = 0.0;
116            }
117            self.pending_frames += frames_f;
118            if self.pending_frames >= self.settle_frames {
119                self.mode = want;
120                self.pending_frames = 0.0;
121            }
122        }
123        self.mode
124    }
125
126    /// The mode the current energy estimates argue for. Hysteresis is applied here,
127    /// not in [`Self::decide`]: entering single-channel mode needs
128    /// [`DOMINANCE_ENTER_DB`], staying in it only [`DOMINANCE_EXIT_DB`].
129    fn candidate(&self) -> DownmixMode {
130        let mut top = (0usize, f64::MIN);
131        let mut second = f64::MIN;
132        for (i, &e) in self.energy.iter().enumerate() {
133            if e > top.1 {
134                second = top.1;
135                top = (i, e);
136            } else if e > second {
137                second = e;
138            }
139        }
140        // Nothing audible on any channel: no evidence either way, hold.
141        if top.1 < DOWNMIX_SILENCE_ENERGY {
142            return self.mode;
143        }
144        // A digitally-silent runner-up is infinite dominance, not a division by zero.
145        let ratio_db = if second <= 0.0 {
146            f64::INFINITY
147        } else {
148            10.0 * (top.1 / second).log10()
149        };
150        let threshold = match self.mode {
151            DownmixMode::Channel(c) if c == top.0 => DOMINANCE_EXIT_DB,
152            _ => DOMINANCE_ENTER_DB,
153        };
154        if ratio_db >= threshold {
155            DownmixMode::Channel(top.0)
156        } else {
157            DownmixMode::Average
158        }
159    }
160}

162/// Convert one cpal callback buffer of arbitrary sample type `T` and channel
163/// count to mono i16, appending into `scratch`.
164///
165/// Downmix is the average of the channels UNLESS one channel has dominated the
166/// others for a full settle window, in which case that channel is emitted alone —
167/// see [`Downmix`] for why averaging a one-hot device costs 6 dB.
168pub(super) fn to_mono_i16<T>(
169    data: &[T],
170    channels: usize,
171    state: &mut Downmix,
172    scratch: &mut Vec<i16>,
173) where
174    T: Sample,
175    i16: cpal::FromSample<T>,
176{
177    scratch.clear();
178    if channels <= 1 {
179        scratch.extend(data.iter().map(|&s| i16::from_sample(s)));
180        return;
181    }
182    // Per-channel energy for THIS buffer, in FloatS16². `zip` bounds the write to
183    // the tracked channel count, so a device that renegotiated more channels than
184    // the state was sized for degrades to tracking the first few, never panics.
185    state.sums.fill(0.0);
186    for frame in data.chunks_exact(channels) {
187        for (acc, &s) in state.sums.iter_mut().zip(frame) {
188            let v = f64::from(i16::from_sample(s));
189            *acc += v * v;
190        }
191    }
192    match state.decide(data.len() / channels) {
193        DownmixMode::Channel(c) => {
194            scratch.extend(
195                data.chunks_exact(channels)
196                    .map(|frame| i16::from_sample(frame[c])),
197            );
198        }
199        DownmixMode::Average => {
200            for frame in data.chunks_exact(channels) {
201                // Average in i32 to avoid i16 overflow, then clamp.
202                let sum: i32 = frame.iter().map(|&s| i16::from_sample(s) as i32).sum();
203                scratch.push((sum / channels as i32) as i16);
204            }
205        }
206    }
207}

209#[cfg(test)]
210mod downmix_tests {
211    use super::{Downmix, to_mono_i16};
212
213    const RATE: u32 = 48_000;
214    /// One 10 ms callback buffer at [`RATE`] — the cpal default is in this range.
215    const FRAMES: usize = 480;
216
217    /// Interleave `frames` sample-frames from a per-channel sample value.
218    fn interleave(per_channel: &[i16], frames: usize) -> Vec<i16> {
219        std::iter::repeat_n(per_channel, frames)
220            .flatten()
221            .copied()
222            .collect()
223    }
224
225    /// Push `buffers` callback buffers of the same interleaved content through the
226    /// downmix and return the LAST buffer's mono output.
227    fn run(state: &mut Downmix, per_channel: &[i16], buffers: usize) -> Vec<i16> {
228        let data = interleave(per_channel, FRAMES);
229        let mut scratch = Vec::new();
230        for _ in 0..buffers {
231            to_mono_i16(&data, per_channel.len(), state, &mut scratch);
232        }
233        scratch
234    }
235
236    /// Buffers spanning well past the settle window, so a sustained condition has
237    /// had every chance to switch the mode: 300 × 10 ms = 3 s, 3× `DOWNMIX_SETTLE_MS`.
238    const PAST_SETTLE: usize = 300;
239
240    /// A balanced device is untouched: dual-mono in, the same value out. This is the
241    /// common case (a real stereo mic, or a mono capsule the driver duplicates), and
242    /// it must stay bit-for-bit the arithmetic average it always was.
243    #[test]
244    fn balanced_channels_stay_on_the_average() {
245        let mut state = Downmix::new(2, RATE);
246        let out = run(&mut state, &[9000, 9000], PAST_SETTLE);
247        assert!(
248            out.iter().all(|&s| s == 9000),
249            "dual-mono must pass through at unity, got {:?}",
250            &out[..4]
251        );
252    }
253
254    /// The −6 dB bug: a 2-channel device carrying the mic on ONE channel. Averaging
255    /// halves it; after the settle window the dominant channel is emitted alone and
256    /// the full level is recovered.
257    #[test]
258    fn one_hot_device_recovers_the_full_level() {
259        let mut state = Downmix::new(2, RATE);
260
261        // Before the window elapses the average still rules — the switch is
262        // deliberately slow, and this pins that it is not instantaneous.
263        let early = run(&mut state, &[9000, 0], 2);
264        assert!(
265            early.iter().all(|&s| s == 4500),
266            "the mode must not switch on the first buffers, got {:?}",
267            &early[..4]
268        );
269
270        let out = run(&mut state, &[9000, 0], PAST_SETTLE);
271        assert!(
272            out.iter().all(|&s| s == 9000),
273            "a one-hot device must recover its full level, not sit 6 dB low, got {:?}",
274            &out[..4]
275        );
276    }
277
278    /// Hysteresis: content that alternates which channel is hot must never switch —
279    /// each candidate keeps resetting the other's evidence, so the mode stays on the
280    /// average and the level never steps mid-utterance.
281    #[test]
282    fn alternating_dominance_never_switches() {
283        let mut state = Downmix::new(2, RATE);
284        let mut scratch = Vec::new();
285        let left = interleave(&[9000, 0], FRAMES);
286        let right = interleave(&[0, 9000], FRAMES);
287        for i in 0..PAST_SETTLE {
288            let data = if i % 2 == 0 { &left } else { &right };
289            to_mono_i16(data, 2, &mut state, &mut scratch);
290            assert!(
291                scratch.iter().all(|&s| s == 4500),
292                "alternating content must stay on the average (buffer {i}), got {:?}",
293                &scratch[..4]
294            );
295        }
296    }
297
298    /// Silence decides nothing. A device idling below the judgement floor must not
299    /// acquire a mode off channel noise — the level ratio there is meaningless.
300    #[test]
301    fn silence_does_not_pick_a_channel() {
302        let mut state = Downmix::new(2, RATE);
303        // Channel 0 has a hair of noise, channel 1 is digital silence: an infinite
304        // ratio, but both are far under the floor.
305        let out = run(&mut state, &[1, 0], PAST_SETTLE);
306        assert!(
307            out.iter().all(|&s| s == 0),
308            "silence must stay on the average, got {:?}",
309            &out[..4]
310        );
311    }
312
313    /// Mono devices skip the machinery entirely (no channels to compare).
314    #[test]
315    fn mono_is_a_straight_copy() {
316        let mut state = Downmix::new(1, RATE);
317        let mut scratch = Vec::new();
318        to_mono_i16(&[1i16, -2, 3, -4], 1, &mut state, &mut scratch);
319        assert_eq!(scratch, vec![1, -2, 3, -4]);
320    }
