1//! # Pixel-format conversion (capture → encoder input)
2//!
3//! Cameras emit MJPEG / YUYV / NV12 / etc.; the encoders want planar **I420**
4//! (8-bit YUV 4:2:0). Sources that produce packed RGB (`PixelFormat::Rgb`
5//! frames, e.g. the synthetic test source) convert via [`rgb_to_i420`] here.
6//!
7//! This conversion is a hot-path allocation + per-pixel arithmetic cost: it is
8//! exactly the work the hardware VideoToolbox path avoids by feeding a
9//! `CVPixelBuffer` straight into the encoder. Where the device can offer YUV
10//! natively (YUYV / NV12), the capturer prefers a cheap structural repack
11//! ([`yuyv_to_i420`] / [`nv12_bytes_to_i420`]) over the RGB round-trip, since
12//! that skips the color-matrix math entirely.
13//!
14//! Coefficients are BT.601 limited-range (the de-facto default for camera
15//! capture / H.264 / AV1 main-profile SD-HD). Chroma is subsampled by a simple
16//! 2×2 box average — cheap and visually adequate at half-resolution chroma; a
17//! higher-quality filter would cost more CPU per frame for little gain here.
18
19/// Convert packed RGB (`width*height*3` bytes, R,G,B) to planar I420.
20///
21/// Returns `Y (w*h) || U (w/2 * h/2) || V (w/2 * h/2)`. `width` and `height`
22/// must be even (I420 chroma is half-resolution); odd inputs are floored, which
23/// is safe because the capturer only requests even resolutions.
24///
25/// BT.601 limited range:
26/// * `Y  =  0.257R + 0.504G + 0.098B + 16`
27/// * `U  = -0.148R - 0.291G + 0.439B + 128`
28/// * `V  =  0.439R - 0.368G - 0.071B + 128`
29// Only the nokhwa camera path (live on Linux) consumes these RGB→I420 helpers;
30// on macOS the camera uses the direct-AVFoundation NV12 path instead, so RGB→I420
31// is dead code there. Allow rather than delete so the one shared helper stays in
32// place for the platforms that do use it.
33#[cfg_attr(target_os = "macos", allow(dead_code))]
34pub(crate) fn rgb_to_i420(rgb: &[u8], width: usize, height: usize) -> Vec<u8> {
35    let cw = width / 2;
36    let ch = height / 2;
37    let y_size = width * height;
38    let c_size = cw * ch;
39    let mut out = vec![0u8; y_size + 2 * c_size];
40    let (y_plane, uv) = out.split_at_mut(y_size);
41    let (u_plane, v_plane) = uv.split_at_mut(c_size);
42
43    // Luma: one sample per pixel.
44    for j in 0..height {
45        for i in 0..width {
46            let p = (j * width + i) * 3;
47            // Guard against a short RGB buffer (defensive; nokhwa gives full).
48            if p + 2 >= rgb.len() {
49                continue;
50            }
51            let (r, g, b) = (rgb[p] as i32, rgb[p + 1] as i32, rgb[p + 2] as i32);
52            let y = ((66 * r + 129 * g + 25 * b + 128) >> 8) + 16;
53            y_plane[j * width + i] = y.clamp(0, 255) as u8;
54        }
55    }
56
57    // Chroma: one sample per 2×2 block, averaged.
58    for j in 0..ch {
59        for i in 0..cw {
60            let mut rs = 0i32;
61            let mut gs = 0i32;
62            let mut bs = 0i32;
63            for dy in 0..2 {
64                for dx in 0..2 {
65                    let px = i * 2 + dx;
66                    let py = j * 2 + dy;
67                    let p = (py * width + px) * 3;
68                    if p + 2 >= rgb.len() {
69                        continue;
70                    }
71                    rs += rgb[p] as i32;
72                    gs += rgb[p + 1] as i32;
73                    bs += rgb[p + 2] as i32;
74                }
75            }
76            let (r, g, b) = (rs / 4, gs / 4, bs / 4);
77            let u = ((-38 * r - 74 * g + 112 * b + 128) >> 8) + 128;
78            let v = ((112 * r - 94 * g - 18 * b + 128) >> 8) + 128;
79            u_plane[j * cw + i] = u.clamp(0, 255) as u8;
80            v_plane[j * cw + i] = v.clamp(0, 255) as u8;
81        }
82    }
83
84    out
85}

87/// Convert packed **YUYV** (a.k.a. YUY2 / `V4L2_PIX_FMT_YUYV`) to planar I420.
88///
89/// YUYV is a packed 4:2:2 format: each 4-byte group encodes two horizontally
90/// adjacent pixels as `Y0 U Y1 V` — full luma per pixel, one shared U and one
91/// shared V per pixel pair (chroma is subsampled horizontally but NOT
92/// vertically). I420 is 4:2:0 — chroma subsampled in BOTH dimensions — so the
93/// conversion keeps every luma sample and reduces the chroma vertically: the
94/// crate keeps each 2×2 block's top-row chroma (nearest-neighbour), the extra
95/// vertical halving the formats differ by.
96///
97/// This is the dominant cheap path for USB Video Class (UVC) webcams on Linux,
98/// which overwhelmingly expose YUYV (and MJPEG); preferring YUYV lets the V4L2
99/// capturer feed the encoder with a simple repack and no JPEG-decode dependency.
100///
101/// `data` must hold at least `width * height * 2` bytes (the packed YUYV frame
102/// at no row padding); `width`/`height` must be even (I420 chroma is half-res).
103/// Returns `None` on a malformed input (odd/zero dimension, or a short buffer)
104/// so the caller skips the frame rather than feeding the encoder corrupt planes.
105///
106/// The actual repack runs through [`yuvutils_rs::yuyv422_to_yuv420`] — a
107/// maintained, runtime-SIMD-accelerated, pure-Rust converter — rather than a
108/// hand-rolled per-pixel loop. `yuyv422_to_yuv420` is a *structural* repack (it
109/// applies no color matrix; the bytes are already YUV). It keeps each block's
110/// top-row chroma for the vertical 4:2:2→4:2:0 reduction (nearest-neighbour)
111/// rather than averaging the two rows; both are valid 4:2:2→4:2:0 reductions and
112/// the difference is visually negligible on a half-vertical-resolution chroma
113/// plane (pinned by the `yuyv_to_i420_chroma_takes_top_row` test). This thin
114/// wrapper exists to preserve the engine's validate-or-`None` contract and to
115/// hand back a tightly-packed `Vec<u8>` in the `Y||U||V` layout the encoder
116/// expects (yuvutils' `alloc` already uses width/half-width strides, so the
117/// planes concatenate with no padding to strip).
118///
119/// Pure function (no platform dependency) so it is unit-tested here on macOS even
120/// though its only live caller is the Linux V4L2 camera.
121#[cfg_attr(any(target_os = "macos", target_os = "ios"), allow(dead_code))]
122pub(crate) fn yuyv_to_i420(data: &[u8], width: usize, height: usize) -> Option<Vec<u8>> {
123    if width == 0 || height == 0 || !width.is_multiple_of(2) || !height.is_multiple_of(2) {
124        return None;
125    }
126    // Each row is `width` pixels × 2 bytes/pixel (YUYV packs 2 bytes per pixel).
127    let row_bytes = width * 2;
128    if data.len() < row_bytes * height {
129        return None;
130    }
131
132    // The YUYV source as a packed image: stride is in *components per row*, which
133    // for packed YUYV is the byte width (2 bytes/pixel = width*2 components).
134    let packed = yuvutils_rs::YuvPackedImage {
135        yuy: &data[..row_bytes * height],
136        yuy_stride: row_bytes as u32,
137        width: width as u32,
138        height: height as u32,
139    };
140    // Tightly-packed I420 target (alloc uses stride == width / width/2, so the
141    // three planes concatenate into our exact `Y||U||V` layout — no padding).
142    let mut planar = yuvutils_rs::YuvPlanarImageMut::<u8>::alloc(
143        width as u32,
144        height as u32,
145        yuvutils_rs::YuvChromaSubsampling::Yuv420,
146    );
147    yuvutils_rs::yuyv422_to_yuv420(&mut planar, &packed).ok()?;
148
149    let y_size = width * height;
150    let c_size = (width / 2) * (height / 2);
151    let mut out = Vec::with_capacity(y_size + 2 * c_size);
152    out.extend_from_slice(planar.y_plane.borrow());
153    out.extend_from_slice(planar.u_plane.borrow());
154    out.extend_from_slice(planar.v_plane.borrow());
155    Some(out)
156}

158/// Convert packed **BGRA** (8 bits per channel, the byte order Windows Graphics
159/// Capture and most PipeWire screen streams deliver) with an explicit row
160/// `stride` to planar, tightly-packed I420 (BT.601 limited range, matching the
161/// rest of the pipeline). The alpha byte is ignored, so this also covers the
162/// `BGRx` / `BGR0` layouts (4 bytes/pixel, 4th byte unused).
163///
164/// `stride` is bytes per row in the source (≥ `width * 4`); pass `width * 4`
165/// for a tightly-packed buffer. `width`/`height` must be even. Returns `None`
166/// on a malformed input (odd/zero dimension, short buffer, or `stride < width*4`).
167///
168/// Live caller is the portal/PipeWire screen backend (Linux);
169/// unit-tested on macOS.
170#[cfg_attr(any(target_os = "macos", target_os = "ios"), allow(dead_code))]
171pub(crate) fn bgra_to_i420(
172    data: &[u8],
173    width: usize,
174    height: usize,
175    stride: usize,
176) -> Option<Vec<u8>> {
177    if width == 0 || height == 0 || !width.is_multiple_of(2) || !height.is_multiple_of(2) {
178        return None;
179    }
180    if stride < width * 4 || data.len() < stride * height {
181        return None;
182    }
183
184    let mut planar = yuvutils_rs::YuvPlanarImageMut::<u8>::alloc(
185        width as u32,
186        height as u32,
187        yuvutils_rs::YuvChromaSubsampling::Yuv420,
188    );
189    // `bgra_stride` is in bytes per row; SIMD-accelerated inside yuvutils.
190    yuvutils_rs::bgra_to_yuv420(
191        &mut planar,
192        &data[..stride * height],
193        stride as u32,
194        yuvutils_rs::YuvRange::Limited,
195        yuvutils_rs::YuvStandardMatrix::Bt601,
196        yuvutils_rs::YuvConversionMode::default(),
197    )
198    .ok()?;
199
200    let y_size = width * height;
201    let c_size = (width / 2) * (height / 2);
202    let mut out = Vec::with_capacity(y_size + 2 * c_size);
203    out.extend_from_slice(planar.y_plane.borrow());
204    out.extend_from_slice(planar.u_plane.borrow());
205    out.extend_from_slice(planar.v_plane.borrow());
206    Some(out)
207}

209/// Convert packed **NV12** (Y plane followed by interleaved CbCr, the V4L2
210/// `NV12` / `V4L2_PIX_FMT_NV12` layout) with explicit plane strides to planar,
211/// tightly-packed I420. The Linux V4L2 capturer's preferred path when a camera
212/// offers NV12 natively (the same plane layout AVFoundation hands us on Apple),
213/// but here the source is a raw V4L2 mmap buffer with a known row stride rather
214/// than a `CVPixelBuffer`, so this is a plain-bytes helper (no CoreVideo).
215///
216/// `y_stride` is the bytes per Y row; the CbCr plane starts at `y_stride*height`
217/// and shares the same row stride (V4L2 NV12 uses one stride for both planes).
218/// Pass `width` for a tightly-packed source. `width`/`height` must be even.
219/// Returns `None` on a malformed input (odd/zero dimension, short buffer, or a
220/// stride narrower than `width`).
221///
222/// Pure function, unit-tested on macOS; live caller is the Linux V4L2 camera.
223#[cfg_attr(any(target_os = "macos", target_os = "ios"), allow(dead_code))]
224pub(crate) fn nv12_bytes_to_i420(
225    data: &[u8],
226    width: usize,
227    height: usize,
228    y_stride: usize,
229) -> Option<Vec<u8>> {
230    // No crop: the coded height equals the output height (the common camera case
231    // where the source already delivers the display size).
232    nv12_bytes_to_i420_cropped(data, height, width, height, y_stride)
233}

235/// NV12 → tightly-packed I420 with a **display crop**.
236///
237/// The source planes are laid out at `coded_height` rows (the codec's
238/// macroblock-aligned height) with `y_stride` bytes/row; only the top-left
239/// `out_width × out_height` is real picture — H.264/HEVC pad the alignment to the
240/// bottom/right and carry the true size as the display aperture
241/// (`MF_MT_MINIMUM_DISPLAY_APERTURE`). The interleaved CbCr plane begins at
242/// `y_stride * coded_height` (NOT `* out_height`), so the coded height is what
243/// places the chroma; the output is cropped to `out_*`. `out_width`/`out_height`
244/// must be even and within the coded frame. Returns `None` on malformed input.
245pub(crate) fn nv12_bytes_to_i420_cropped(
246    data: &[u8],
247    coded_height: usize,
248    out_width: usize,
249    out_height: usize,
250    y_stride: usize,
251) -> Option<Vec<u8>> {
252    if out_width == 0
253        || out_height == 0
254        || !out_width.is_multiple_of(2)
255        || !out_height.is_multiple_of(2)
256        || y_stride < out_width
257        || coded_height < out_height
258    {
259        return None;
260    }
261    let cw = out_width / 2;
262    let ch = out_height / 2;
263    let y_size = out_width * out_height;
264    let c_size = cw * ch;
265    // The CbCr plane is placed by the CODED height (the full encoded frame), then we
266    // read only the display rows out of each plane.
267    let y_plane_bytes = y_stride * coded_height;
268    let uv_plane_bytes = y_stride * (coded_height / 2);
269    if data.len() < y_plane_bytes + uv_plane_bytes {
270        return None;
271    }
272
273    let mut out = vec![0u8; y_size + 2 * c_size];
274    let (y_dst, rest) = out.split_at_mut(y_size);
275    let (u_dst, v_dst) = rest.split_at_mut(c_size);
276
277    // Y: copy the display rows, dropping the row padding (and any coded-vs-display
278    // extra rows below `out_height`).
279    for row in 0..out_height {
280        y_dst[row * out_width..row * out_width + out_width]
281            .copy_from_slice(&data[row * y_stride..row * y_stride + out_width]);
282    }
283    // CbCr: de-interleave U,V from the second plane (stride-respecting), display rows.
284    let uv = &data[y_plane_bytes..];
285    for row in 0..ch {
286        deinterleave_cbcr(
287            &uv[row * y_stride..row * y_stride + cw * 2],
288            &mut u_dst[row * cw..row * cw + cw],
289            &mut v_dst[row * cw..row * cw + cw],
290        );
291    }
292
293    Some(out)
294}

296/// Split one `CbCrCbCr…` row into its U and V halves. `src` drives the length: it is
297/// read in pairs, and `u`/`v` are trimmed to that pair count (extra capacity is left
298/// untouched).
299///
300/// This is one row of the NV12→I420 chroma de-interleave, split out with the lengths
301/// re-derived from `src` *on purpose*. The whole routine lives or dies on LLVM
302/// autovectorizing this loop into the NEON de-interleaving load (`ld2`), and that
303/// only happens if the bounds checks are eliminated first — a check is an extra loop
304/// exit, and the vectorizer needs a single-exit countable loop. Deriving the trip
305/// count and both destination lengths from one value (`n`) makes every check
306/// trivially provable. Indexing the *planes* instead (`u_dst[row * cw + col]`) leaves
307/// LLVM to prove it from the surrounding loop nest, which it does at -O3 but stops
308/// doing under `-Cprofile-use`; the loop then goes scalar, one byte at a time, and a
309/// 1080p frame costs 430 µs instead of 85 µs.
310///
311/// `pub(crate)` so the Apple sys layer's `CVImageBuffer` extractor shares this one
312/// row loop — that path has the same shape and the same hazard.
313#[inline]
314pub(crate) fn deinterleave_cbcr(src: &[u8], u: &mut [u8], v: &mut [u8]) {
315    let n = src.len() / 2;
316    let (u, v) = (&mut u[..n], &mut v[..n]);
317    for i in 0..n {
318        u[i] = src[i * 2];
319        v[i] = src[i * 2 + 1];
320    }
321}

323/// Pack tightly-packed I420 (Y plane ‖ U plane ‖ V plane, stride == width) into
324/// tightly-packed **NV12** (Y plane ‖ interleaved CbCr, stride == width) at the
325/// same `width`×`height`. The inverse of [`nv12_bytes_to_i420`]; the Windows /
326/// Linux HW encoders take NV12 input, while the engine's frames are I420.
327/// Returns `None` on odd dimensions or a short buffer (a frame to skip, never a
328/// panic). Gated to the configs that consume it (the Windows MFT encoder and the
329/// Linux VAAPI encoder under `linux-vaapi`) plus `test`, so it is exercised on
330/// every host without warning where no HW encoder takes NV12 input (macOS feeds
331/// VideoToolbox planar/surface input; a default Linux build uses the SW floor).
332#[cfg(any(
333    target_os = "windows",
334    all(target_os = "linux", feature = "linux-vaapi"),
335    test
336))]
337pub(crate) fn i420_to_nv12(data: &[u8], width: usize, height: usize) -> Option<Vec<u8>> {
338    if width == 0 || height == 0 || !width.is_multiple_of(2) || !height.is_multiple_of(2) {
339        return None;
340    }
341    let (cw, ch) = (width / 2, height / 2);
342    let y_size = width * height;
343    let c_size = cw * ch;
344    if data.len() < y_size + 2 * c_size {
345        return None;
346    }
347    let mut out = vec![0u8; y_size + 2 * c_size];
348    // Y plane copies straight across.
349    out[..y_size].copy_from_slice(&data[..y_size]);
350    // Interleave U,V into the CbCr plane.
351    let u = &data[y_size..y_size + c_size];
352    let v = &data[y_size + c_size..y_size + 2 * c_size];
353    let uv = &mut out[y_size..];
354    for i in 0..c_size {
355        uv[i * 2] = u[i];
356        uv[i * 2 + 1] = v[i];
357    }
358    Some(out)
359}

361/// Convert a decoded/captured NV12 (`420v` / `420f`) `CVImageBuffer` to a planar
362/// I420 [`RawFrame`]. The **single** validated NV12→I420 helper shared by the
363/// VideoToolbox decoder and the AVFoundation capturer — they produce
364/// byte-identical NV12 buffers, so the conversion is one helper, not two that can
365/// drift.
366///
367/// The implementation lives in the Apple sys layer ([`super::apple`]): it locks
368/// the buffer through the RAII guard (which **always** unlocks, on every path)
369/// and copies through the guard's bounds-checked plane views, so no `unsafe`
370/// pixel-buffer handling remains in this module. This thin re-export keeps the
371/// call sites (`videotoolbox_decoder`, `avfoundation_capture`) unchanged. The
372/// helper is fully fallible — both callers run it on an Apple-owned thread where
373/// a panic would abort the process.
374// (Test-only shim: the live callers — the VT decoder sink and the camera
375// delegate — call the apple sys-layer helper directly; the converter tests
376// below exercise it through this module to pin the shared-converter contract.)
377#[cfg(all(test, any(target_os = "macos", target_os = "ios")))]
378pub(crate) fn nv12_image_buffer_to_i420(
379    pb: &objc2_core_video::CVImageBuffer,
380    context: &str,
381) -> Result<crate::types::RawFrame, String> {
382    crate::platform::apple::nv12_image_buffer_to_i420(pb, context)
383}

444    /// YUYV chroma vertical subsampling (4:2:2 → 4:2:0). The conversion runs
445    /// through `yuvutils_rs::yuyv422_to_yuv420`, which takes the **top** row's
446    /// chroma for each 2×2 block (nearest-neighbour) rather than averaging the two
447    /// rows. Both are valid 4:2:2→4:2:0 reductions; the difference is
448    /// confined to a half-vertical-resolution chroma plane and is visually
449    /// negligible, and the maintained, SIMD-accelerated crate beats a per-pixel
450    /// loop. This test pins the crate's actual behaviour so
451    /// any future change to it is caught.
452    #[test]
453    fn yuyv_to_i420_chroma_takes_top_row() {
454        let (w, h) = (2usize, 2usize);
455        // Row 0: U=100 V=20 ; Row 1: U=200 V=80. One group per row (w/2 = 1).
456        let yuyv = vec![
457            10, 100, 11, 20, // row 0: Y0=10 U=100 Y1=11 V=20
458            12, 200, 13, 80, // row 1: Y0=12 U=200 Y1=13 V=80
459        ];

461        let c_off = w * h; // chroma after the 2×2 Y plane
462        // Top-row chroma is kept (U=100, V=20), not the (100+200)/2 / (20+80)/2
463        // average the hand-rolled converter produced.
464        assert_eq!(out[c_off], 100, "U should be the top row's chroma");
465        assert_eq!(out[c_off + 1], 20, "V should be the top row's chroma");
466    }

470    /// NV12 plain-bytes → I420 round-trips a known frame, dropping row padding

473    fn nv12_bytes_to_i420_round_trips_with_stride() {
474        let (w, h) = (4usize, 2usize);
475        let stride = 6usize; // padded: stride > width
476        let cw = w / 2;
477        let ch = h / 2;
478        // Y plane: stride*h bytes, first `w` of each row meaningful (a ramp).
479        let mut data = vec![0u8; stride * h + stride * ch];
480        for row in 0..h {
481            for col in 0..w {
482                data[row * stride + col] = (row * w + col) as u8;
483            }
484        }
485        // CbCr plane starts at stride*h; interleave U,V (U=70, V=210).
486        let uv_off = stride * h;
487        for row in 0..ch {
488            for col in 0..cw {
489                data[uv_off + row * stride + col * 2] = 70;
490                data[uv_off + row * stride + col * 2 + 1] = 210;
491            }
492        }
493        let out = nv12_bytes_to_i420(&data, w, h, stride).expect("convert");
494        assert_eq!(out.len(), w * h + 2 * cw * ch);
495        // Y is the ramp with padding dropped.
496        let expected_y: Vec<u8> = (0..(w * h) as u8).collect();
497        assert_eq!(&out[..w * h], &expected_y[..], "Y not preserved");
498        let c_off = w * h;
499        let c_size = cw * ch;
500        assert!(
501            out[c_off..c_off + c_size].iter().all(|&u| u == 70),
502            "U wrong"
503        );
504        assert!(out[c_off + c_size..].iter().all(|&v| v == 210), "V wrong");
505    }

507    /// I420 → NV12 → I420 round-trips exactly (the encoder-input path): Y straight
508    /// across, U/V interleaved into CbCr and de-interleaved back.
509    #[test]
510    fn i420_to_nv12_round_trips() {
511        let (w, h) = (4usize, 2usize);
512        let (cw, ch) = (w / 2, h / 2);
513        let (y_size, c_size) = (w * h, cw * ch);
514        let mut i420 = vec![0u8; y_size + 2 * c_size];
515        for (i, b) in i420[..y_size].iter_mut().enumerate() {
516            *b = i as u8; // Y ramp
517        }
518        i420[y_size..y_size + c_size].fill(70); // U
519        i420[y_size + c_size..].fill(210); // V
520
521        let nv12 = i420_to_nv12(&i420, w, h).expect("i420→nv12");
522        assert_eq!(nv12.len(), y_size + 2 * c_size);
523        assert_eq!(&nv12[..y_size], &i420[..y_size], "Y not preserved");
524        // CbCr is interleaved U,V.
525        for i in 0..c_size {
526            assert_eq!(nv12[y_size + i * 2], 70, "Cb wrong");
527            assert_eq!(nv12[y_size + i * 2 + 1], 210, "Cr wrong");
528        }
529        // And it inverts back to the original I420 (tightly packed, stride == width).
530        let back = nv12_bytes_to_i420(&nv12, w, h, w).expect("nv12→i420");
531        assert_eq!(back, i420, "round-trip changed the frame");
532    }

636    // ---- NV12 CVPixelBuffer → I420 (the shared, panic-safety-critical helper) --
637    //
638    // These build a real NV12 `CVPixelBuffer` via `CVPixelBufferCreate` (no
639    // camera, no decode session — fully headless), fill known Y/UV bytes
640    // respecting the buffer's actual per-plane stride (CoreVideo pads rows), and
641    // round-trip through the shared `nv12_image_buffer_to_i420`. This is the
642    // direct device-independent test of the converter both the VT decoder and the
643    // AVFoundation capturer share.
644
645    /// Allocate an NV12 `CVPixelBuffer` of `w`×`h` and fill it from a known I420
646    /// source (Y plane + de-interleaved U/V → interleaved CbCr), respecting the
647    /// allocator's per-plane stride. Built entirely through the safe sys-layer
648    /// [`super::super::apple::pixel_buffer::PixelBuffer`] (no `unsafe` here).
649    #[cfg(any(target_os = "macos", target_os = "ios"))]
650    fn make_test_nv12(
651        w: usize,
652        h: usize,
653        i420: &[u8],
654    ) -> crate::platform::apple::pixel_buffer::PixelBuffer {
655        use crate::platform::apple::pixel_buffer::PixelBuffer;
656
657        let cw = w / 2;
658        let ch = h / 2;
659        let y_size = w * h;
660        let c_size = cw * ch;
661        let (y_plane, rest) = i420.split_at(y_size);
662        let (u_plane, v_plane) = rest.split_at(c_size);
663
664        let mut pb = PixelBuffer::new_nv12(w, h).expect("CVPixelBufferCreate");
665        {
666            let mut guard = pb.lock_writable().expect("lock");
667            // Plane 0: Y, stride-respecting row copy (scoped so the borrow ends).
668            {
669                let mut y = guard.plane_mut(0).expect("Y plane");
670                for row in 0..h {
671                    let dst = y.row_mut(row, w).expect("Y row");
672                    dst.copy_from_slice(&y_plane[row * w..row * w + w]);
673                }
674            }
675            // Plane 1: interleave U,V into CbCr.
676            let mut uv = guard.plane_mut(1).expect("UV plane");
677            for row in 0..ch {
678                let dst = uv.row_mut(row, cw * 2).expect("UV row");
679                for col in 0..cw {
680                    dst[col * 2] = u_plane[row * cw + col];
681                    dst[col * 2 + 1] = v_plane[row * cw + col];
682                }
683            }
684        }
685        pb
686    }

688    /// Build an I420 buffer with a per-pixel-distinct Y ramp and flat U/V so the
689    /// NV12 round-trip can assert the Y plane survives byte-exact through the
690    /// stride-respecting copy (NV12↔I420 is a lossless layout shuffle).
691    #[cfg(any(target_os = "macos", target_os = "ios"))]
692    fn ramp_i420(w: usize, h: usize) -> Vec<u8> {
693        let cw = w / 2;
694        let ch = h / 2;
695        let mut buf = vec![0u8; w * h + 2 * cw * ch];
696        for row in 0..h {
697            for col in 0..w {
698                buf[row * w + col] = ((row * 7 + col * 13) % 256) as u8;
699            }
700        }
701        let y_size = w * h;
702        let c_size = cw * ch;
703        buf[y_size..y_size + c_size].fill(90); // U
704        buf[y_size + c_size..].fill(170); // V
705        buf
706    }

709    /// layout shuffle is lossless), including correct handling of CoreVideo's
710    /// per-plane stride padding. Uses 66×34: deliberately not a multiple of 16 so
711    /// the allocator pads each row and the converter must drop that padding.
712    #[cfg(any(target_os = "macos", target_os = "ios"))]
713    #[test]
714    fn nv12_pixel_buffer_round_trips_to_i420() {
