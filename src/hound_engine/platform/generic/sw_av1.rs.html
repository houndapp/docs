1//! The software-AV1 seam: **one** `cfg` boundary for the whole engine.
2//!
3//! The dav1d decoder is behind an optional dependency (see
4//! `super::dav1d_decoder` for why — not linked, it only exists under its
5//! feature), but the three places that *select* a
6//! decoder — the Apple HW-or-SW composite, the portable SW backend, and the
7//! `decode_caps` probe — should not each carry a feature `cfg`. They ask this
8//! module instead:
9//!
10//! * [`available`] — is there a real AV1 decoder in this build? The single input
11//!   to `decode_caps().av1` on the software side, so a client can never advertise
12//!   AV1 to peers and then fail to construct a decoder for it.
13//! * [`open`] — build one, or a typed [`CaptureError::Unsupported`].
14//! * [`SwAv1Decoder`] — the decoder type, **uninhabited** when the feature is
15//!   off, exactly like [`crate::platform::stub`]'s never-backends. That is what
16//!   lets the composite decoders hold it in an enum arm unconditionally: the arm
17//!   exists in the type, no value can ever occupy it, and it costs nothing.
18
19use crate::error::CaptureError;
20// Only the uninhabited stand-in below needs the SPI vocabulary; under the feature
21// `SwAv1Decoder` is a type alias whose impls live with the decoder itself.
22#[cfg(not(feature = "av1-sw-decode"))]
23use crate::backend::{VideoDecode, sealed};
24#[cfg(not(feature = "av1-sw-decode"))]
25use crate::types::{RawFrame, VideoCodec};
26
27/// The software AV1 decoder: dav1d where the `av1-sw-decode` feature is on.
28#[cfg(feature = "av1-sw-decode")]
29pub(crate) type SwAv1Decoder = super::dav1d_decoder::Dav1dDecoder;
30
31/// Uninhabited stand-in where the feature is off — no value can be constructed,
32/// so every arm holding one is statically unreachable.
33#[cfg(not(feature = "av1-sw-decode"))]
34pub(crate) enum SwAv1Decoder {}
35
36#[cfg(not(feature = "av1-sw-decode"))]
37impl sealed::Sealed for SwAv1Decoder {}
38
39#[cfg(not(feature = "av1-sw-decode"))]
40impl VideoDecode for SwAv1Decoder {
41    fn codec(&self) -> VideoCodec {
42        match *self {}
43    }
44    fn decode(&mut self, _chunk: &[u8]) -> Result<Option<RawFrame>, CaptureError> {
45        match *self {}
46    }
47    fn backend(&self) -> &'static str {
48        match *self {}
49    }
50}
51
52/// Whether this build can decode AV1 in software **right now** — the honest input
53/// to `decode_caps().av1` alongside the hardware probe.
54///
55/// A compile-time answer, unlike the software H.264 twin
56/// ([`super::openh264::sw_h264_available`]), which is a *runtime* question because
57/// Cisco's binary is downloaded onto the device rather than linked. dav1d is
58/// linked at build time under a permissive licence, so if it is in the build it is
59/// available, full stop.
60pub(crate) const fn available() -> bool {
61    cfg!(feature = "av1-sw-decode")
62}
63
64/// Build a software AV1 decoder, or report the typed capability miss. Callers
65/// gate on [`available`] first where they need to choose an arm; this returning
66/// `Unsupported` is the belt-and-braces path.
67pub(crate) fn open() -> Result<SwAv1Decoder, CaptureError> {
68    #[cfg(feature = "av1-sw-decode")]
69    {
70        super::dav1d_decoder::Dav1dDecoder::new()
71    }
72    #[cfg(not(feature = "av1-sw-decode"))]
73    {
74        Err(CaptureError::Unsupported {
75            what: "software AV1 decode (build with the engine's `av1-sw-decode` feature; \
76                   `mise run av1deps` installs libdav1d)",
77        })
78    }
79}

81#[cfg(test)]
82mod tests {
83    use super::*;
84
85    /// **Caps honesty**, pinned as a law rather than a value: the capability
86    /// answer and the constructor must agree in both directions. Advertising AV1
87    /// with no decoder behind it makes peers send a stream that will never
88    /// decode (a permanently black tile with a healthy-looking downlink);
89    /// refusing to build one we *do* have would silently drop the codec off the
90    /// ladder.
91    #[test]
92    fn availability_and_the_constructor_never_disagree() {
93        match open() {
94            Ok(_) => assert!(
95                available(),
96                "a decoder was built while `available()` said there is none — peers would never \
97                 be offered AV1 despite this client being able to decode it"
98            ),
99            Err(e) => assert!(
100                !available(),
101                "`available()` promises AV1 to every peer but the decoder would not build: {e}"
102            ),
103        }
104    }
105}
