1//! Software video codec backends (openh264) — the non-Apple codec path.
2//!
3//! These thin `open(...)`/`caps()` wrappers are the SW H.264 floor. They are the
4//! **selected** backend on Windows (encoder), on Linux **without** the
5//! `linux-vaapi` feature (both ways), and on the bare-fallback targets. Apple
6//! (always) and Linux **with** `linux-vaapi` reach the SW floor through their own
7//! HW-or-SW dispatch enums, holding an [`Openh264Encoder`]/[`Openh264Decoder`]
8//! directly — so in that one configuration these wrappers are compiled but unused;
9//! the `cfg_attr` keeps that build warning-clean without narrowing the module cfg.
10#![cfg_attr(all(target_os = "linux", feature = "linux-vaapi"), allow(dead_code))]
11
12//!
13//! **Encode is H.264 only.** A device with no HW video encoder cannot run SW
14//! AV1/HEVC encode in real time, so those return `Unsupported` and the codec
15//! ladder's universal floor is H.264.
16//!
17//! **Decode is H.264 + AV1.** Decode is far cheaper than encode, and AV1's
18//! hardware floor is high enough (Intel Arc / Meteor Lake, RTX 30-series) that a
19//! large share of receivers would otherwise have to refuse the codec outright.
20//! The AV1 arm is [`super::sw_av1`] (dav1d), present only in a build carrying the
21//! `av1-sw-decode` feature; where it is absent this is exactly the H.264-only
22//! backend it was. HEVC has no software decoder anywhere.
23//!
24//! These wrap [`Openh264Encoder`] / [`Openh264Decoder`] / the dav1d decoder behind
25//! the uniform `open(codec, …)` / `caps()` interface the platform facade
26//! dispatches through, so [`crate::api`] never branches on target.
27

36/// SW (openh264) video encoder backend — the non-Apple `VideoEncoderBackend`.
37/// Boxed: openh264 carries large inline buffers; built once per track, so the
38/// indirection is free on the hot path.
39pub(crate) struct SwVideoEncoder(Box<Openh264Encoder>);
40
41impl SwVideoEncoder {
42    /// Open a SW encoder for `codec`. Only H.264 (the universal floor) is
43    /// supported; HEVC/AV1 have no SW path and return `Unsupported`.
44    /// [`Self::open`]; this backend has no temporal-layer path, so the request is
45    /// accepted and ignored and the encoder marks every frame layer 0 (a valid
46    /// L1T1 stream). Present so the platform facade never branches on target.
47    pub(crate) fn open_temporal(
48        codec: VideoCodec,
49        dims: Dimensions,
50        fps: Fps,
51        bitrate: Bitrate,
52        cadence: KeyframeCadence,
53    ) -> Result<Self, CaptureError> {
54        Self::open(codec, dims, fps, bitrate, cadence)
55    }
56
57    pub(crate) fn open(
58        codec: VideoCodec,
59        dims: Dimensions,
60        fps: Fps,
61        bitrate: Bitrate,
62        cadence: KeyframeCadence,
63    ) -> Result<Self, CaptureError> {
64        match codec {
65            VideoCodec::H264 => Ok(Self(Box::new(Openh264Encoder::new(
66                dims, fps, bitrate, cadence,
67            )?))),
68            VideoCodec::H265 => Err(CaptureError::Unsupported {
69                what: "HEVC video encode",
70            }),
71            VideoCodec::Av1 => Err(CaptureError::Unsupported {
72                what: "AV1 video encode",
73            }),
74        }
75    }
76
77    /// What this device can encode: H.264 only — and only while a SW H.264
78    /// implementation actually exists (see [`super::openh264::sw_h264_available`]:
79    /// a shipped build whose Cisco binary has not downloaded yet, or whose user
80    /// turned it off, has none, and must not advertise the codec).
81    pub(crate) fn caps() -> VideoCodecCaps {
82        VideoCodecCaps {
83            av1: false,
84            h265: false,
85            h264: super::openh264::sw_h264_available(),
86        }
87    }
88}
89
90impl sealed::Sealed for SwVideoEncoder {}

113/// SW video decoder backend — the non-Apple `VideoDecoderBackend`. An enum, not
114/// `dyn`, so the per-AU `decode` stays static dispatch like every other composite
115/// backend.
116pub(crate) enum SwVideoDecoder {
117    /// openh264 — H.264, the universal wire floor.
118    H264(Box<Openh264Decoder>),
119    /// dav1d — AV1. Uninhabited without the `av1-sw-decode` feature, so this arm
120    /// is statically unreachable in a build that does not carry it.
121    Av1(Box<SwAv1Decoder>),
122}
123
124impl SwVideoDecoder {
125    /// Open a SW decoder for `codec`: H.264 (openh264) and AV1 (dav1d, where the
126    /// build has it). HEVC — and AV1 without dav1d — return `Unsupported`, and the
127    /// receive pipeline drains-and-drops so audio still plays.
128    pub(crate) fn open(codec: VideoCodec) -> Result<Self, CaptureError> {
129        match codec {
130            VideoCodec::H264 => Ok(Self::H264(Box::new(Openh264Decoder::new()?))),
131            VideoCodec::H265 => Err(CaptureError::Unsupported {
132                what: "HEVC video decode",
133            }),
134            VideoCodec::Av1 => sw_av1::open().map(|d| Self::Av1(Box::new(d))),
135        }
136    }
137
138    /// What this device can decode: H.264 while a SW H.264 implementation exists
139    /// (see [`SwVideoEncoder::caps`]), AV1 while dav1d is in the build. Both
140    /// answers are exactly the condition [`Self::open`] tests, so a peer is never
141    /// offered a codec whose decoder will not construct.
142    pub(crate) fn caps() -> VideoCodecCaps {
143        VideoCodecCaps {
144            av1: sw_av1::available(),
145            h265: false,
146            h264: super::openh264::sw_h264_available(),
147        }
148    }
149}
150
151impl sealed::Sealed for SwVideoDecoder {}
152
153impl VideoDecode for SwVideoDecoder {
154    fn codec(&self) -> VideoCodec {
155        match self {
156            Self::H264(_) => VideoCodec::H264,
157            Self::Av1(_) => VideoCodec::Av1,
158        }
159    }
160    fn decode(&mut self, chunk: &[u8]) -> Result<Option<RawFrame>, CaptureError> {
161        match self {
162            Self::H264(d) => d.decode(chunk),
163            Self::Av1(d) => d.decode(chunk),
164        }
165    }
166    fn backend(&self) -> &'static str {
167        match self {
168            Self::H264(d) => d.backend(),
169            Self::Av1(d) => d.backend(),
170        }
171    }
172}

174#[cfg(test)]
175mod tests {
176    use super::*;
177
178    /// **Caps honesty**: every codec this backend advertises must actually open,
179    /// and every codec it refuses must be one it does not advertise. Advertising a
180    /// codec with no decoder behind it is the expensive failure — the sender picks
181    /// it off the ladder and every frame lands on a black tile with a healthy
182    /// downlink — and the AV1 arm is the one that made this a live question.
183    #[test]
184    fn decode_caps_match_what_open_will_actually_build() {
185        let caps = SwVideoDecoder::caps();
186        for (codec, advertised) in [
187            (VideoCodec::H264, caps.h264),
188            (VideoCodec::H265, caps.h265),
189            (VideoCodec::Av1, caps.av1),
190        ] {
191            let opened = SwVideoDecoder::open(codec);
192            assert_eq!(
193                opened.is_ok(),
194                advertised,
195                "{codec:?}: caps say {advertised} but open said {}",
196                opened.is_ok()
197            );
198        }
199        assert!(
200            !caps.h265,
201            "there is no software HEVC decoder on any target"
202        );
203    }
