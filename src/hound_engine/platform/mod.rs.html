1//! # Per-platform backends
2//!
3//! The concrete implementations of the [`crate::backend`] SPI, selected at
4//! **compile time** per target. Nothing in here is visible outside the crate
5//! (`pub(crate)`): the app reaches devices and codecs only through
6//! [`crate::api`], which wraps these in platform-agnostic handles.
7//!
8//! * [`generic`] — the portable pieces that already build on every desktop
9//!   target: cpal audio I/O (CoreAudio / WASAPI / ALSA-Pulse / AAudio),
10//!   device-rate resampling, audio device enumeration, the SW H.264 codec
11//!   (openh264, both ways), the synthetic test source, and pixel-format
12//!   conversion.
13//! * `apple` — the Apple-only camera + HW codec backend (AVFoundation
14//!   capture, VideoToolbox H.264/HEVC/AV1, and the safe sys layer under them).
15//! * `linux` — the Linux camera backend (V4L2 via the `v4l` crate). Video codecs
16//!   default to the portable SW openh264 H.264 floor; a VAAPI HW backend
17//!   (cros-libva / cros-codecs: HW H.264 encode, H.264 + HEVC decode) is gated
18//!   behind the off-by-default `linux-vaapi` feature (cros-libva's system-libva
19//!   build coupling — see the feature note). HEVC encode is unwired regardless.
20//! * `windows` — the Windows camera backend (Media Foundation via the official
21//!   `windows` crate, `IMFSourceReader` synchronous pull) plus the Media
22//!   Foundation Transform HW video codecs: HW H.264/HEVC encode (async MFT) and
23//!   H.264/HEVC/AV1 decode (DXVA), with the SW openh264 H.264 floor retained.
24//! * [`stub`] — the uninhabited per-role backends for capabilities a target
25//!   lacks (a typed `Unsupported`, never a panic).
26//!
27//! ## Backend selection lives HERE
28//!
29//! The per-role backend **types** are resolved by `cfg` in this module
30//! (`CameraBackend`, `ScreenBackend`, `AppAudioBackend`, `VideoDecoderBackend`),
31//! so [`crate::api`] stays platform-agnostic — it names `platform::CameraBackend`
32//! and never carries a `#[cfg(target_os = …)]` of its own. A capability a target
33//! lacks resolves to the matching [`stub`] type.
34//!
35//! ## Adding a backend (the Windows/Linux parity path)
36//!
37//! Add a `windows`/`linux` module gated on its `target_os`, implement the
38//! [`crate::backend`] traits for the native capture/codec APIs, then point the
39//! `cfg`-resolved alias below at it (and, for a new HW codec, wire the runtime
40//! [`crate::api::Engine::encode_caps`] / [`crate::api::Engine::decode_caps`]
41//! probes). Platform deps go in this crate's `[target.'cfg(...)'.dependencies]`
42//! — the app crate never gains one.
43

47#[cfg(target_os = "linux")]

105/// The video **decoder** backend, by platform:
106/// * **Apple** — HW VideoToolbox (H.264 / HEVC / AV1) with a SW openh264 fallback
107///   for H.264. Unlike every other backend the Apple one switches arms on the
108///   DECODE path, not at open: VideoToolbox builds its decode session lazily on
109///   the first keyframe, so a session-create failure is a mid-stream event (see
110///   [`apple::video_codec::AppleVideoDecoder`]).

112///   SW openh264 fallback for H.264.
113/// * **Linux** — SW openh264 (H.264) by default; HW VAAPI (H.264 / HEVC via
114///   cros-codecs) with a SW H.264 fallback when built with the `linux-vaapi`
115///   feature (see the encoder note above for why it is off by default).
116/// * **else** — SW openh264 (H.264 only).
117///
118/// All expose the uniform `open(codec)` / `caps()` the facade dispatches through.

120pub(crate) type VideoDecoderBackend = apple::video_codec::AppleVideoDecoder;
121#[cfg(target_os = "windows")]

123#[cfg(all(target_os = "linux", feature = "linux-vaapi"))]

133// ── The decode-session budget (one number per decoder backend) ──────────────
134// Lives here, next to the `VideoDecoderBackend` alias above, because it is a
135// property OF that backend: a target that changes decoders without restating its
136// session budget would otherwise silently inherit another backend's number.
137
138/// Concurrent decode sessions a HARDWARE decoder backend is good for.
139///
140/// **16**, which is the conservative floor across the three HW paths rather than
141/// any one of their maxima: Intel QuickSync tops out around 16 simultaneous
142/// decode sessions, Apple Silicon is soft-limited by its (1–2) video decode
143/// engines rather than by a published session count, and the MFT/VAAPI limits are
144/// driver-dependent.
145///
146/// On Apple it is the FIRST protection, and for HEVC/AV1 still the only one.
147/// Apple's TN2267 is explicit that decode-session creation may simply fail under
148/// contention. That failure does NOT surface as a decoder that will not build:
149/// VideoToolbox's `open` is infallible because the `VTDecompressionSession` is
150/// created lazily on the first keyframe, so it surfaces mid-stream, on the decode
151/// path ([`apple::vt_decoder::VideoToolboxDecoder::install_session`]). For H.264
152/// that is now caught and degraded onto the SW openh264 floor
153/// ([`apple::video_codec::AppleVideoDecoder`]); for HEVC/AV1 there is no software
154/// arm, so it is still a black tile. Capping the number of sessions a receiver
155/// ever asks for is the receive-side way to not reach that wall at all.
156const HW_DECODE_SESSIONS: usize = 16;
157
158/// Threads left alone by the software decode budget: the RT audio mix thread and
159/// the app's main/UI thread. Both must keep their core under decode load — the
160/// audio one is the whole reason this budget exists.
161const SW_DECODE_HEADROOM_THREADS: usize = 2;
162
163/// Floor on any platform's budget — Jitsi's `MIN_LAST_N`. A receiver that can
164/// decode fewer than three remote videos is not in a video call, and cutting
165/// below this trades a call for a CPU saving nobody asked for.
166const MIN_DECODE_SESSIONS: usize = 3;
167
168/// The software (openh264) decode budget for a host with `threads` hardware
169/// threads. Each SW decoder is one decode thread's worth of CPU, so this is a
170/// **CPU prior** — not an OS session limit, of which the SW path has none.
171///
172/// Pure and separated from [`decode_session_budget`] so the clamp arithmetic is
173/// testable on every target, not only on one that resolves to the SW backend.
174// Dead on a HW target's non-test build, and deliberately still compiled there: the
175// arithmetic is target-independent policy, and its test runs everywhere.
176#[cfg_attr(not(test), allow(dead_code))]
177fn sw_decode_sessions(threads: usize) -> usize {
178    threads
179        .saturating_sub(SW_DECODE_HEADROOM_THREADS)
180        .clamp(MIN_DECODE_SESSIONS, HW_DECODE_SESSIONS)
181}
182
183/// How many concurrent decode sessions this build's [`VideoDecoderBackend`] is
184/// good for — see [`crate::api::Engine::decode_session_budget`] for what the
185/// number means and who consumes it.
186pub(crate) fn decode_session_budget() -> usize {
187    // HW paths: Apple VideoToolbox, the Windows MFT and Linux VAAPI — all three
188    // now with a SW H.264 fallback. The fallback does not raise the budget — a box
189    // that has fallen back to openh264 for every tile is CPU-bound at least as
190    // early as the SW arm below assumes.
191    #[cfg(any(
192        target_os = "macos",
193        target_os = "ios",
194        target_os = "windows",
195        all(target_os = "linux", feature = "linux-vaapi")
196    ))]
197    {
198        HW_DECODE_SESSIONS
199    }
200    // The portable SW openh264 floor — every other target, including a default
201    // Linux build.
202    #[cfg(not(any(
203        target_os = "macos",
204        target_os = "ios",
205        target_os = "windows",
206        all(target_os = "linux", feature = "linux-vaapi")
207    )))]
208    {
209        // An unknowable parallelism (a container with no cgroup answer) reads as
210        // the floor's worth of threads, which clamps to the floor.
211        let threads = std::thread::available_parallelism().map_or(0, |n| n.get());
212        sw_decode_sessions(threads)
213    }
214}

253// ── Device privacy gates (TCC on Apple; nothing to ask elsewhere) ───────────
254// The `cfg` lives here, next to backend selection, so the facade stays agnostic;
255// on a platform with no per-device grant these are a no-op / `Unknown`.

257/// This app's privacy grant for `kind`, **without prompting**. Apple reads the
258/// AVFoundation authorization status (camera + microphone); every other platform
259/// has no equivalent gate and reports
260/// [`DevicePermission::Unknown`](crate::types::DevicePermission::Unknown).
261#[cfg_attr(
262    not(any(target_os = "macos", target_os = "ios")),
263    allow(unused_variables)
264)]
265pub(crate) fn device_permission(kind: crate::types::DeviceKind) -> crate::types::DevicePermission {
266    #[cfg(any(target_os = "macos", target_os = "ios"))]
267    {
268        match apple::tcc::AvMedia::of_device(kind) {
269            Some(media) => apple::tcc::status(media),
270            None => crate::types::DevicePermission::Unknown,
271        }
272    }
273    #[cfg(not(any(target_os = "macos", target_os = "ios")))]
274    {
275        crate::types::DevicePermission::Unknown
276    }
277}

279/// Preflight the microphone grant before the device is opened, prompting (and
280/// blocking) when the user has never been asked.
281///
282/// The camera capturer preflights inside its own backend; the mic cannot, because
283/// its backend is the portable cpal path — which just opens CoreAudio and fails
284/// opaquely (or delivers silence) when TCC has denied us. So the Apple gate is
285/// applied HERE, at the platform seam, leaving `generic::cpal_io` free of
286/// Apple-specific code. A no-op everywhere else.
287pub(crate) fn ensure_microphone_authorized() -> Result<(), crate::error::AudioError> {
288    #[cfg(any(target_os = "macos", target_os = "ios"))]
289    {
290        use apple::tcc::{AvMedia, TccDenial};
291        apple::tcc::ensure_authorized(AvMedia::Audio).map_err(|d| match d {
292            TccDenial::Denied => crate::error::AudioError::PermissionDenied {
293                kind: crate::types::DeviceKind::Microphone,
294            },
295            TccDenial::Unavailable(what) => crate::error::AudioError::Device(what),
296        })
297    }
298    #[cfg(not(any(target_os = "macos", target_os = "ios")))]
299    {
300        Ok(())
301    }
302}

409#[cfg(test)]
410mod tests {
411    use super::*;
412
413    /// The SW budget is "one thread each, minus the threads audio and the UI must
414    /// keep" — and it is clamped at both ends, so neither a two-core box nor a
415    /// 64-core workstation escapes the range the client's `DecodeBudget` walks.
416    #[test]
417    fn the_software_decode_budget_leaves_audio_and_ui_a_thread_each() {
418        assert_eq!(sw_decode_sessions(8), 6, "8 threads − audio − UI");
419        assert_eq!(sw_decode_sessions(10), 8);
420        // The floor: a dual-core (and the degenerate "parallelism unknown" 0) still
421        // gets MIN_DECODE_SESSIONS, because fewer is not a video call.
422        for threads in [0, 1, 2, 3, 4, 5] {
423            assert_eq!(
424                sw_decode_sessions(threads),
425                MIN_DECODE_SESSIONS,
426                "{threads} threads must clamp up to the floor"
427            );
428        }
429        // …and the ceiling: a big machine does not get to open 62 decoders, because
430        // the budget is a receive-side policy and not a CPU measurement.
431        for threads in [64, 128, usize::MAX] {
432            assert_eq!(sw_decode_sessions(threads), HW_DECODE_SESSIONS);
433        }
434        // Monotone in the thread count — a bigger box never gets a smaller budget.
435        for threads in 0..64usize {
436            assert!(sw_decode_sessions(threads) <= sw_decode_sessions(threads + 1));
437        }
438    }
439
440    /// Whatever backend this target resolved to, the budget it reports is inside
441    /// the range every consumer assumes: at least the floor (so a receiver always
442    /// gets a call) and never above the HW ceiling (so the VideoToolbox no-fallback
443    /// wall is never approached from a *static* number).
444    #[test]
445    fn every_platforms_decode_budget_is_within_the_shared_bounds() {
446        let k = decode_session_budget();
447        assert!(
448            (MIN_DECODE_SESSIONS..=HW_DECODE_SESSIONS).contains(&k),
449            "this target's decode session budget is {k}"
450        );
451    }
