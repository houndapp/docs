1//! Portal screen-source enumeration — the sources half of
2//! [`super::portal_screen`] (the running-capture backend), matching the
3//! per-platform `screen` / `screen_sources` split. Wayland's portal dialog owns
4//! the real selection, so enumeration is one sentinel row per castable KIND and
5//! thumbnails are typed-unsupported (capturing one would prompt the dialog).
6
7use super::portal;
8use crate::error::CaptureError;
9use crate::types::{
10    Dimensions, DisplayId, ScreenSource, ScreenSourceInfo, ThumbnailImage, WindowId,
11};
12
13/// The sentinel ids [`enumerate_sources`] hands out. The portal dialog does the
14/// real selection, so the id only encodes which *kind* the picker is
15/// restricted to; a stale/unknown id of the right variant opens the same way.
16const PORTAL_SCREEN_ID: &str = "portal-screen";
17const PORTAL_WINDOW_ID: &str = "portal-window";
18
19/// List the castable source kinds as picker entries — one "Screen" and/or one
20/// "Window" row (the compositor's dialog picks the concrete source at open).
21/// An absent portal surfaces [`CaptureError::Unsupported`] so the UI degrades
22/// the same way as a platform without a screen backend.
23pub(crate) fn enumerate_sources() -> Result<Vec<ScreenSourceInfo>, CaptureError> {
24    let kinds = portal::available_kinds()?;
25    if kinds.is_empty() {
26        return Err(CaptureError::Unsupported {
27            what: "screen capture (portal offers no castable sources)",
28        });
29    }
30    // Placeholder geometry: the true size is known only after the portal
31    // dialog; the share opens at the source's native size regardless.
32    let placeholder = Dimensions::even_clamped(1920, 1080);
33    Ok(kinds
34        .into_iter()
35        .map(|kind| match kind {
36            portal::SourceKinds::Monitor => ScreenSourceInfo {
37                source: ScreenSource::Display(DisplayId::new(PORTAL_SCREEN_ID.to_string())),
38                title: "Screen (choose in system dialog)".to_string(),
39                app_name: None,
40                dimensions: placeholder,
41                is_on_screen: true,
42            },
43            portal::SourceKinds::Window => ScreenSourceInfo {
44                source: ScreenSource::Window(WindowId::new(PORTAL_WINDOW_ID.to_string())),
45                title: "Window (choose in system dialog)".to_string(),
46                app_name: None,
47                dimensions: placeholder,
48                is_on_screen: true,
49            },
50        })
51        .collect())
52}

54/// Thumbnails require capturing, and capturing prompts the portal dialog — a
55/// per-row prompt storm. Typed `Unsupported`; the picker degrades to
56/// title-only cards (its documented fallback).
57pub(crate) fn capture_thumbnail(
58    _source: &ScreenSource,
59    _max_w: u32,
60    _max_h: u32,
61) -> Result<ThumbnailImage, CaptureError> {
62    Err(CaptureError::Unsupported {
63        what: "screen thumbnails (the portal dialog owns source selection)",
64    })
