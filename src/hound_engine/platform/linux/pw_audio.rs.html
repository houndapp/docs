1//! # PipeWire loopback capture → screen-share audio (Linux)
2//!
3//! The Linux audio leg of a screen share, scoped by [`AudioScope`]:
4//!
5//! * **Display share → whole-system audio** ([`AudioScope::SystemMonitor`]): one
6//!   capture stream with `stream.capture.sink`, which WirePlumber connects to the
7//!   **default sink's monitor** — everything audible is on the screen being
8//!   shared, the same contract as the Windows WASAPI render loopback. The stream
9//!   follows default-sink changes (session-manager policy), so a device swap
10//!   mid-share keeps audio flowing.
11//! * **App-scoped audio** ([`AudioScope::App`]): the per-application model every
12//!   shipping Linux capturer uses (OBS's app capture, Discord's venmic): watch
13//!   the PipeWire registry for `Stream/Output/Audio` nodes whose identity
14//!   matches the target — process id **within the target's process tree** (a
15//!   browser renders audio in child processes), or application binary/name —
16//!   and run one capture stream per matching node (`target.object`), mixed to a
17//!   single leg. Nodes are picked up and released **live** as the app starts and
18//!   stops streams. Wayland's portal never reveals which app owns a shared
19//!   window, so the caller supplies the identity explicitly; a scope is never
20//!   silently widened to system audio (the privacy contract shared with the
21//!   Windows process-loopback path). Our own process is always excluded — a
22//!   share must not capture the call's own playout as an echo.
23//!
24//! Either way the leg delivers 48 kHz mono 20 ms [`PcmFrame`]s — the same
25//! contract as the microphone path and the macOS/Windows share-audio legs, so
26//! the Opus encode + publish plumbing is reused verbatim.
27//!
28//! ## Pipeline & threading
29//!
30//! One dedicated **bridge thread** owns every PipeWire object (`MainLoop`,
31//! context, registry, streams — none are `Send`): it iterates the loop, and all
32//! stream `process` callbacks run in that iteration (streams are connected
33//! *without* `RT_PROCESS`, so delivery is marshalled off PipeWire's RT data
34//! thread onto the loop — the engine never runs Rust in the graph's RT path).
35//! Each stream asks the graph for **48 kHz mono S16** and PipeWire's stream
36//! adapter does the rate/channel conversion, so the bridge only mixes and
37//! frames: per-node sample queues → [`AppMixer`](super::app_mixer::AppMixer) (aligned summing with a
38//! liveness window, so one stalled app can't hold back the leg) → the shared
39//! [`PcmFramer`] 20 ms cadence → a drop-oldest [`BoundedRing`] popped by
40//! [`AppAudio::next_frame`].
41//!
42//! An idle scope (no matching app streams; a silent suspended sink) produces no
43//! frames and the framer re-anchors on resume — the receiver's `screen_audio`
44//! track is gappy during silence exactly like macOS/Windows.
45//!
46//! ## Where the testable logic lives
47//!
48//! The graph plumbing needs a live PipeWire daemon (covered by the self-skipping
49//! integration tests below, which spawn real `pw-play` producers). The
50//! bug-prone parts are pure and unit-tested on every host: node identity
51//! matching ([`node_matches`](super::app_match::node_matches)), process-tree
52//! expansion ([`descendants`](super::app_match::descendants),
53//! [`parse_stat_ppid`](super::app_match)), and the multi-stream mixer
54//! ([`AppMixer`](super::app_mixer::AppMixer)).
55

75/// Bounded PCM ring depth (20 ms frames) — ~200 ms, so a brief consumer stall
76/// never starves while the buffer stays bounded. Drop-oldest on overflow.
77const RING_FRAMES: usize = 10;
78
79/// Loop iteration timeout: the mix/frame cadence when no graph events arrive.
80const LOOP_TICK: Duration = Duration::from_millis(10);
81
82/// Re-expand the target's process tree at most this often when unknown pids
83/// appear (browsers spawn audio child processes after the share starts).
84const TREE_RESCAN_US: i64 = 1_000_000;

86/// What the leg captures: the default sink's monitor (display share) or one
87/// application's output streams (the app-scoped, privacy-bounded capture).
88#[derive(Clone, Debug)]
89pub(crate) enum AudioScope {
90    /// Everything playing on the default sink.
91    SystemMonitor,
92    /// Only streams owned by the application identified by [`AppTarget`].
93    App(AppTarget),
94}

96/// The identity an app-scoped capture matches `Stream/Output/Audio` nodes
97/// against. Any populated criterion admits a node (pid-tree first, then the
98/// name fallbacks — the OBS matching order); an empty target matches nothing.
99#[derive(Clone, Debug, Default)]
100pub(crate) struct AppTarget {
101    /// Root process id; the match includes its whole descendant tree.
102    pub(crate) root_pid: Option<u32>,
103    /// `application.process.binary` (case-insensitive).
104    pub(crate) binary: Option<String>,
105    /// `application.name` (case-insensitive).
106    pub(crate) app_name: Option<String>,
107}

109/// The running PipeWire share-audio leg. Holds the drop-oldest ring the bridge
110/// thread fills; [`AppAudio::next_frame`] pops it. Dropping it stops the bridge
111/// thread, which releases every PipeWire object on its owning thread.
112pub(crate) struct PipeWireAppAudio {
113    ring: Arc<BoundedRing<PcmFrame>>,
114    stop: Arc<AtomicBool>,
115    bridge: Option<JoinHandle<()>>,
116}
117
118impl PipeWireAppAudio {
119    /// Open the capture for `scope` and begin delivery. Graph connection happens
120    /// on the bridge thread and its result is reported back, so a no-daemon /
121    /// no-default-sink failure surfaces here as a typed [`AudioError`] (never
122    /// silent). An app scope with no streams *yet* opens successfully and starts
123    /// delivering when the app plays (that is the normal share-then-play order).
124    pub(crate) fn open(
125        clock: hound_media::CaptureClock,
126        scope: AudioScope,
127    ) -> Result<Self, AudioError> {
128        let ring = BoundedRing::<PcmFrame>::new(RING_FRAMES);
129        let stop = Arc::new(AtomicBool::new(false));
130        let (ready_tx, ready_rx) = std::sync::mpsc::channel::<Result<(), AudioError>>();
131
132        let bridge = std::thread::Builder::new()
133            .name("pw-share-audio".into())
134            .spawn({
135                let ring = Arc::clone(&ring);
136                let stop = Arc::clone(&stop);
137                move || bridge_loop(clock, scope, ring, stop, ready_tx)
138            })
139            .map_err(|e| AudioError::Device(format!("pipewire audio thread spawn failed: {e}")))?;
140
141        match ready_rx.recv() {
142            Ok(Ok(())) => Ok(Self {
143                ring,
144                stop,
145                bridge: Some(bridge),
146            }),
147            Ok(Err(e)) => {
148                let _ = bridge.join();
149                Err(e)
150            }
151            Err(_) => {
152                let _ = bridge.join();
153                Err(AudioError::Device(
154                    "pipewire audio thread exited before reporting open".into(),
155                ))
156            }
157        }
158    }
159}

161impl sealed::Sealed for PipeWireAppAudio {}

163impl AppAudio for PipeWireAppAudio {
164    fn next_frame(&mut self) -> Result<PcmFrame, AudioError> {
165        self.ring
166            .pop_blocking()
167            .ok_or_else(|| AudioError::Device("screen-share audio ended".into()))
168    }
169}

171impl Drop for PipeWireAppAudio {
172    fn drop(&mut self) {
173        self.stop.store(true, Ordering::Relaxed);
174        self.ring.stop();
175        if let Some(h) = self.bridge.take() {
176            let _ = h.join();
177        }
178    }
179}

183/// Mutable state shared between the loop body and the stream/registry
184/// callbacks (all on the bridge thread — `RefCell`, no locks).
185struct Shared {
186    mixer: AppMixer,
187    /// Matched nodes awaiting a capture stream: `(node_id, object.serial)`.
188    /// Streams are created between loop iterations, never inside a registry
189    /// callback (no re-entrant PipeWire calls from its own dispatch).
190    pending_add: Vec<(u32, String)>,
191    /// Nodes whose global vanished; their stream + leg are torn down.
192    pending_remove: Vec<u32>,
193    /// The system-scope stream hit an error state (fatal for the leg).
194    failed: bool,
195}
196
197/// Bridge-thread body: connect to the graph (reporting success/failure), pump
198/// the loop + mixer until stopped, then stop the ring so the consumer ends.
199fn bridge_loop(
200    clock: hound_media::CaptureClock,
201    scope: AudioScope,
202    ring: Arc<BoundedRing<PcmFrame>>,
203    stop: Arc<AtomicBool>,
204    ready_tx: std::sync::mpsc::Sender<Result<(), AudioError>>,
205) {
206    super::pw_init_once();
207
208    let mainloop = match pw::main_loop::MainLoopRc::new(None) {
209        Ok(l) => l,
210        Err(e) => {
211            let _ = ready_tx.send(Err(AudioError::Device(format!("pw main loop: {e}"))));
212            return;
213        }
214    };
215    let context = match pw::context::ContextRc::new(&mainloop, None) {
216        Ok(c) => c,
217        Err(e) => {
218            let _ = ready_tx.send(Err(AudioError::Device(format!("pw context: {e}"))));
219            return;
220        }
221    };
222    let core = match context.connect_rc(None) {
223        Ok(c) => c,
224        Err(e) => {
225            let _ = ready_tx.send(Err(AudioError::NoDevice(format!(
226                "pipewire daemon unavailable: {e}"
227            ))));
228            return;
229        }
230    };
231
232    let shared = Rc::new(RefCell::new(Shared {
233        mixer: AppMixer::default(),
234        pending_add: Vec::new(),
235        pending_remove: Vec::new(),
236        failed: false,
237    }));
238
239    // Live capture streams by node id (leg id). The listener must outlive the
240    // stream, so both are held together.
241    let mut streams: HashMap<u32, (pw::stream::StreamRc, pw::stream::StreamListener<()>)> =
242        HashMap::new();
243
244    // Scope setup: one monitor stream, or a registry watch that feeds
245    // `pending_add`/`pending_remove`.
246    let mut registry_guard = None;
247    match &scope {
248        AudioScope::SystemMonitor => {
249            const MONITOR_LEG: u32 = 0;
250            match open_capture_stream(&core, MONITOR_LEG, None, true, &clock, &shared) {
251                Ok(s) => {
252                    streams.insert(MONITOR_LEG, s);
253                }
254                Err(e) => {
255                    let _ = ready_tx.send(Err(e));
256                    return;
257                }
258            }
259        }
260        AudioScope::App(target) => {
261            let registry = match core.get_registry_rc() {
262                Ok(r) => r,
263                Err(e) => {
264                    let _ = ready_tx.send(Err(AudioError::Device(format!("pw registry: {e}"))));
265                    return;
266                }
267            };
268            let watch = RegistryWatch::new(target.clone(), Rc::clone(&shared));
269            let listener = registry
270                .add_listener_local()
271                .global({
272                    let watch = watch.clone();
273                    move |global| watch.on_global(global)
274                })
275                .global_remove({
276                    let watch = watch.clone();
277                    move |id| watch.on_global_remove(id)
278                })
279                .register();
280            registry_guard = Some((registry, listener));
281        }
282    }
283    let _ = ready_tx.send(Ok(()));
284
285    let mut framer = PcmFramer::new();
286    let mut mixed: Vec<i16> = Vec::new();
287    let mut frames: Vec<PcmFrame> = Vec::new();
288
289    while !stop.load(Ordering::Relaxed) {
290        mainloop
291            .loop_()
292            .iterate(pw::loop_::Timeout::Finite(LOOP_TICK));
293
294        // Service registry-driven stream churn outside the dispatch.
295        let (adds, removes, failed) = {
296            let mut s = shared.borrow_mut();
297            (
298                std::mem::take(&mut s.pending_add),
299                std::mem::take(&mut s.pending_remove),
300                s.failed,
301            )
302        };
303        if failed {
304            break;
305        }
306        for id in removes {
307            if streams.remove(&id).is_some() {
308                shared.borrow_mut().mixer.remove(id);
309                tracing::info!(target: "pw_audio", node = id, "app audio stream ended");
310            }
311        }
312        for (id, serial) in adds {
313            if streams.contains_key(&id) {
314                continue;
315            }
316            match open_capture_stream(&core, id, Some(&serial), false, &clock, &shared) {
317                Ok(s) => {
318                    streams.insert(id, s);
319                    tracing::info!(target: "pw_audio", node = id, "app audio stream captured");
320                }
321                Err(e) => {
322                    tracing::warn!(target: "pw_audio", node = id, "app stream open failed: {e}");
323                }
324            }
325        }
326
327        // Mix whatever the callbacks delivered this tick and frame it out.
328        let now = clock.now();
329        shared.borrow_mut().mixer.take(now, &mut mixed);
330        if !mixed.is_empty() {
331            framer.push(&mixed, now, &mut frames);
332            for f in frames.drain(..) {
333                ring.push(f);
334            }
335        }
336    }
337
338    drop(streams);
339    drop(registry_guard);
340    ring.stop();
341}

343/// Registry-side matching state for an app scope: the target identity, the
344/// (lazily re-expanded) process tree, and the **client identity table**. A
345/// stream node does not reliably carry `application.*` props itself — they
346/// live on the owning *Client* object, referenced by the node's `client.id` —
347/// so Client globals are recorded and merged into each node's identity (the
348/// OBS matching model). Runs inside the `global` callback, so it only
349/// *records* matches; stream creation happens in the loop body.
350#[derive(Clone)]
351struct RegistryWatch {
352    target: AppTarget,
353    shared: Rc<RefCell<Shared>>,
354    tree: Rc<RefCell<HashSet<u32>>>,
355    clients: Rc<RefCell<HashMap<u32, ClientIdentity>>>,
356    /// Output-stream nodes whose owning client the registry has not announced
357    /// yet (announce order is id order, and ids are reused — a node can arrive
358    /// before its client). Re-evaluated when their client shows up.
359    unresolved: Rc<RefCell<HashMap<u32, NodeIdentity>>>,
360    last_scan_us: Rc<RefCell<i64>>,
361    own_pid: u32,
362}
363
364/// The `application.*` identity of one PipeWire client (connection).
365#[derive(Clone, Default, Debug)]
366struct ClientIdentity {
367    pid: Option<u32>,
368    binary: Option<String>,
369    app_name: Option<String>,
370}
371
372impl RegistryWatch {
373    fn new(target: AppTarget, shared: Rc<RefCell<Shared>>) -> Self {
374        let tree = match target.root_pid {
375            Some(root) => descendants(root, &proc_parent_pairs()),
376            None => HashSet::new(),
377        };
378        Self {
379            target,
380            shared,
381            tree: Rc::new(RefCell::new(tree)),
382            clients: Rc::new(RefCell::new(HashMap::new())),
383            unresolved: Rc::new(RefCell::new(HashMap::new())),
384            last_scan_us: Rc::new(RefCell::new(0)),
385            own_pid: std::process::id(),
386        }
387    }
388
389    fn on_global(&self, global: &pw::registry::GlobalObject<&spa::utils::dict::DictRef>) {
390        let Some(props) = global.props.as_ref() else {
391            return;
392        };
393        match global.type_ {
394            pw::types::ObjectType::Client => {
395                // `pipewire.sec.pid` is the kernel-verified peer pid
396                // (SO_PEERCRED) and is present in the registry props;
397                // `application.process.id` is the client's self-report and may
398                // only exist on the bound object. Prefer the verified one.
399                self.clients.borrow_mut().insert(
400                    global.id,
401                    ClientIdentity {
402                        pid: props
403                            .get("pipewire.sec.pid")
404                            .or_else(|| props.get("application.process.id"))
405                            .and_then(|p| p.parse().ok()),
406                        binary: props.get("application.process.binary").map(str::to_owned),
407                        app_name: props.get("application.name").map(str::to_owned),
408                    },
409                );
410                // Nodes that arrived ahead of this client can be judged now.
411                let owned: Vec<u32> = self
412                    .unresolved
413                    .borrow()
414                    .iter()
415                    .filter(|(_, n)| n.client_id == Some(global.id))
416                    .map(|(&id, _)| id)
417                    .collect();
418                for node_id in owned {
419                    if let Some(identity) = self.unresolved.borrow_mut().remove(&node_id) {
420                        self.evaluate(node_id, identity);
421                    }
422                }
423            }
424            pw::types::ObjectType::Node => {
425                let identity = NodeIdentity::from_props(props);
426                if identity.is_output_stream() {
427                    self.evaluate(global.id, identity);
428                }
429            }
430            _ => {}
431        }
432    }
433
434    /// Judge one output-stream node against the target, after merging the
435    /// owning client's identity; park it as unresolved when that client is
436    /// still unannounced.
437    fn evaluate(&self, node_id: u32, mut identity: NodeIdentity) {
438        {
439            let clients = self.clients.borrow();
440            match identity.client_id.map(|id| clients.get(&id)) {
441                Some(Some(client)) => {
442                    identity.pid = identity.pid.or(client.pid);
443                    identity.binary = identity.binary.take().or_else(|| client.binary.clone());
444                    identity.app_name =
445                        identity.app_name.take().or_else(|| client.app_name.clone());
446                }
447                // The owning client is not in the registry snapshot yet; judge
448                // this node when it arrives.
449                Some(None) => {
450                    drop(clients);
451                    self.unresolved.borrow_mut().insert(node_id, identity);
452                    return;
453                }
454                None => {}
455            }
456        }
457
458        // A candidate pid outside the known tree may be a child spawned after
459        // the last scan — re-expand (throttled) before judging it.
460        if let (Some(root), Some(pid)) = (self.target.root_pid, identity.pid)
461            && !self.tree.borrow().contains(&pid)
462        {
463            let now = monotonic_us();
464            let mut last = self.last_scan_us.borrow_mut();
465            if now - *last >= TREE_RESCAN_US {
466                *last = now;
467                *self.tree.borrow_mut() = descendants(root, &proc_parent_pairs());
468            }
469        }
470
471        if node_matches(&identity, &self.target, &self.tree.borrow(), self.own_pid) {
472            tracing::debug!(
473                target: "pw_audio",
474                node = node_id, ?identity,
475                "app-scope match"
476            );
477            if let Some(serial) = &identity.serial {
478                self.shared
479                    .borrow_mut()
480                    .pending_add
481                    .push((node_id, serial.clone()));
482            }
483        }
484    }
485
486    fn on_global_remove(&self, id: u32) {
487        self.clients.borrow_mut().remove(&id);
488        self.unresolved.borrow_mut().remove(&id);
489        self.shared.borrow_mut().pending_remove.push(id);
490    }
491}

493/// Build + connect one S16/48 kHz/mono capture stream. `capture_sink` selects
494/// the default-sink monitor; otherwise `target_serial` pins the stream to one
495/// node (`target.object`). Samples land in the shared mixer under `leg`.
496fn open_capture_stream(
497    core: &pw::core::CoreRc,
498    leg: u32,
499    target_serial: Option<&str>,
500    capture_sink: bool,
501    clock: &hound_media::CaptureClock,
502    shared: &Rc<RefCell<Shared>>,
503) -> Result<(pw::stream::StreamRc, pw::stream::StreamListener<()>), AudioError> {
504    let mut props = pw::properties::properties! {
505        *pw::keys::MEDIA_TYPE => "Audio",
506        *pw::keys::MEDIA_CATEGORY => "Capture",
507        *pw::keys::MEDIA_ROLE => "Screen",
508        *pw::keys::NODE_NAME => "hound-share-audio",
509    };
510    if capture_sink {
511        props.insert(*pw::keys::STREAM_CAPTURE_SINK, "true");
512    }
513    if let Some(serial) = target_serial {
514        props.insert(*pw::keys::TARGET_OBJECT, serial);
515    }
516
517    let stream = pw::stream::StreamRc::new(core.clone(), "hound-share-audio", props)
518        .map_err(|e| AudioError::Device(format!("pw stream: {e}")))?;
519
520    let listener = stream
521        .add_local_listener::<()>()
522        .state_changed({
523            let shared = Rc::clone(shared);
524            let fatal = capture_sink; // app legs churn; the monitor leg is the share's lifeline
525            move |_, _, _old, new| {
526                if let pw::stream::StreamState::Error(e) = new {
527                    tracing::warn!(target: "pw_audio", leg, "stream error: {e}");
528                    let mut s = shared.borrow_mut();
529                    if fatal {
530                        s.failed = true;
531                    } else {
532                        s.pending_remove.push(leg);
533                    }
534                }
535            }
536        })
537        .process({
538            let shared = Rc::clone(shared);
539            let clock = *clock;
540            move |stream, _| {
541                if let Some(mut buffer) = stream.dequeue_buffer() {
542                    let datas = buffer.datas_mut();
543                    if let Some(d) = datas.first_mut() {
544                        let (offset, size) =
545                            (d.chunk().offset() as usize, d.chunk().size() as usize);
546                        if let Some(bytes) = d.data() {
547                            let end = (offset + size).min(bytes.len());
548                            if offset < end {
549                                let mono: Vec<i16> = bytes[offset..end]
550                                    .chunks_exact(2)
551                                    .map(|s| i16::from_le_bytes([s[0], s[1]]))
552                                    .collect();
553                                shared.borrow_mut().mixer.push(leg, &mono, clock.now());
554                            }
555                        }
556                    }
557                }
558            }
559        })
560        .register()
561        .map_err(|e| AudioError::Device(format!("pw stream listener: {e}")))?;
562
563    let pod_bytes = capture_format_pod();
564    let pod = spa::pod::Pod::from_bytes(&pod_bytes)
565        .ok_or_else(|| AudioError::Device("pw format pod build failed".into()))?;
566
567    // No RT_PROCESS: `process` must run on the loop thread (RefCell state), not
568    // PipeWire's RT data thread.
569    stream
570        .connect(
571            spa::utils::Direction::Input,
572            None,
573            pw::stream::StreamFlags::AUTOCONNECT | pw::stream::StreamFlags::MAP_BUFFERS,
574            &mut [pod],
575        )
576        .map_err(|e| AudioError::Device(format!("pw stream connect: {e}")))?;
577
578    Ok((stream, listener))
579}

581/// The one capture format asked of every stream: S16LE, 48 kHz, mono — the
582/// engine's PCM contract. PipeWire's stream adapter converts whatever the
583/// source node carries, so the bridge never resamples. Pure (a serialized
584/// `EnumFormat` pod), unit-tested below.
585fn capture_format_pod() -> Vec<u8> {
586    let mut info = spa::param::audio::AudioInfoRaw::new();
587    info.set_format(spa::param::audio::AudioFormat::S16LE);
588    info.set_rate(SAMPLE_RATE.hz());
589    info.set_channels(1);
590    let obj = spa::pod::Object {
591        type_: spa::utils::SpaTypes::ObjectParamFormat.as_raw(),
592        id: spa::param::ParamType::EnumFormat.as_raw(),
593        properties: info.into(),
594    };
595    spa::pod::serialize::PodSerializer::serialize(
596        std::io::Cursor::new(Vec::new()),
597        &spa::pod::Value::Object(obj),
598    )
599    .map(|(c, _)| c.into_inner())
600    .unwrap_or_default()
601}

603/// Monotonic microseconds for callback-side throttling (never the capture
604/// clock — that belongs to frame timestamps only).
605fn monotonic_us() -> i64 {
606    use std::time::Instant;
607    static EPOCH: std::sync::OnceLock<Instant> = std::sync::OnceLock::new();
608    EPOCH.get_or_init(Instant::now).elapsed().as_micros() as i64
609}

632    // These need a running PipeWire daemon with a default sink and the
633    // `pw-play` client (any stock desktop; headless CI skips). They spawn real
634    // playback processes and pin the two scope contracts end to end: the
635    // default-sink monitor hears what plays, and an app scope hears ONLY its

641    /// Serializes the live tests: they spawn `pw-play` producers into the ONE
642    /// user PipeWire graph, so running them concurrently lets one test's
643    /// playback and teardown timing bleed into the other's RMS assertions.
644    fn live_lock() -> std::sync::MutexGuard<'static, ()> {
645        static LOCK: std::sync::Mutex<()> = std::sync::Mutex::new(());
646        LOCK.lock().unwrap_or_else(|p| p.into_inner())
647    }

649    /// True iff a PipeWire daemon answers on this host; tests skip otherwise.
650    fn have_pipewire() -> bool {
651        super::super::pw_init_once();
652        let Ok(ml) = pw::main_loop::MainLoopRc::new(None) else {
653            return false;
654        };
655        let Ok(ctx) = pw::context::ContextRc::new(&ml, None) else {
656            return false;
657        };
658        ctx.connect_rc(None).is_ok()
659    }

661    fn have_pw_play() -> bool {
662        std::process::Command::new("pw-play")
663            .arg("--help")
664            .stdout(std::process::Stdio::null())
665            .stderr(std::process::Stdio::null())
666            .status()
667            .is_ok_and(|s| s.success())
668    }

670    /// Write a 48 kHz mono s16 sine WAV (the little 44-byte RIFF header + PCM).
671    fn write_tone_wav(path: &std::path::Path, secs: f32, freq: f32, amp: i16) {
672        let rate = 48_000u32;
673        let n = (secs * rate as f32) as usize;
674        let mut pcm = Vec::with_capacity(n * 2);
675        for i in 0..n {
676            let s = (amp as f32
677                * (2.0 * std::f32::consts::PI * freq * i as f32 / rate as f32).sin())
678                as i16;
679            pcm.extend_from_slice(&s.to_le_bytes());
680        }
681        let data_len = pcm.len() as u32;
682        let mut wav = Vec::with_capacity(44 + pcm.len());
683        wav.extend_from_slice(b"RIFF");
684        wav.extend_from_slice(&(36 + data_len).to_le_bytes());
685        wav.extend_from_slice(b"WAVEfmt ");
686        wav.extend_from_slice(&16u32.to_le_bytes());
687        wav.extend_from_slice(&1u16.to_le_bytes()); // PCM
688        wav.extend_from_slice(&1u16.to_le_bytes()); // mono
689        wav.extend_from_slice(&rate.to_le_bytes());
690        wav.extend_from_slice(&(rate * 2).to_le_bytes());
691        wav.extend_from_slice(&2u16.to_le_bytes());
692        wav.extend_from_slice(&16u16.to_le_bytes());
693        wav.extend_from_slice(b"data");
694        wav.extend_from_slice(&data_len.to_le_bytes());
695        wav.extend_from_slice(&pcm);
696        std::fs::write(path, wav).expect("write tone wav");
697    }

699    fn spawn_pw_play(path: &std::path::Path) -> std::process::Child {
700        std::process::Command::new("pw-play")
701            .arg(path)
702            .stdout(std::process::Stdio::null())
703            .stderr(std::process::Stdio::null())
704            .spawn()
705            .expect("spawn pw-play")
706    }

708    fn rms(frame: &PcmFrame) -> f64 {
709        let sum: f64 = frame.samples.iter().map(|&s| (s as f64) * (s as f64)).sum();
710        (sum / frame.samples.len().max(1) as f64).sqrt()
711    }
712
713    /// Pull frames off the leg's ring until one clears the RMS bar or the
714    /// deadline passes; returns the loudest RMS seen.
715    fn loudest_within(leg: &PipeWireAppAudio, window: Duration, bar: f64) -> f64 {
716        let deadline = Instant::now() + window;
717        let mut loudest = 0.0f64;
718        while Instant::now() < deadline {
719            if let crate::platform::generic::ring::PopWait::Frame(f) =
720                leg.ring.pop_timeout(Duration::from_millis(100))
721            {
722                assert_eq!(f.samples.len(), SAMPLES_PER_FRAME, "20 ms frame contract");
723                loudest = loudest.max(rms(&f));
724                if loudest >= bar {
725                    break;
726                }
727            }
728        }
729        loudest
730    }

732    /// THE display-share audio proof: a system-monitor leg hears what plays on
733    /// the default sink, framed to the 20 ms mic contract. Skips with no
734    /// daemon/`pw-play`.
735    #[test]
736    fn system_monitor_hears_default_sink_playback() {
737        if !have_pipewire() || !have_pw_play() {
738            eprintln!("SKIP system_monitor_hears_default_sink_playback: no pipewire/pw-play");
739            return;
740        }
741        let _live = live_lock();
742        let dir = std::env::temp_dir();
743        let wav = dir.join("hound_pw_test_tone.wav");
744        write_tone_wav(&wav, 4.0, 440.0, 12_000);
745
746        let clock = hound_media::CaptureClock::start();
747        let leg = PipeWireAppAudio::open(clock, AudioScope::SystemMonitor)
748            .expect("system-monitor leg should open against a live daemon");
749        let mut child = spawn_pw_play(&wav);
750
751        let loudest = loudest_within(&leg, Duration::from_secs(6), 1_000.0);
752        let _ = child.kill();
753        let _ = child.wait();
754        drop(leg);
755        assert!(
756            loudest >= 1_000.0,
757            "monitor leg never heard the tone (loudest RMS {loudest:.0})"
758        );
759    }

761    /// THE privacy proof for app scope: a leg scoped to one process hears that
762    /// process, and goes silent when it exits — even while ANOTHER app keeps
763    /// playing on the same sink. Skips with no daemon/`pw-play`.
764    #[test]
765    fn app_scope_hears_only_its_target_process() {
766        if !have_pipewire() || !have_pw_play() {
767            eprintln!("SKIP app_scope_hears_only_its_target_process: no pipewire/pw-play");
768            return;
769        }
770        let _live = live_lock();
771        let dir = std::env::temp_dir();
772        let long = dir.join("hound_pw_test_long.wav");
773        let short = dir.join("hound_pw_test_short.wav");
774        write_tone_wav(&long, 20.0, 330.0, 12_000); // the decoy keeps playing
775        write_tone_wav(&short, 5.0, 550.0, 12_000); // the target
776
777        let mut decoy = spawn_pw_play(&long);
778        let mut target = spawn_pw_play(&short);
779
780        let clock = hound_media::CaptureClock::start();
781        let leg = PipeWireAppAudio::open(
782            clock,
783            AudioScope::App(AppTarget {
784                root_pid: Some(target.id()),
785                ..Default::default()
786            }),
787        )
788        .expect("app-scoped leg should open against a live daemon");
789
790        let heard = loudest_within(&leg, Duration::from_secs(6), 1_000.0);
791        assert!(
792            heard >= 1_000.0,
793            "app scope never heard its target process (loudest RMS {heard:.0})"
794        );
795
796        // Kill the target; the decoy is still playing, and must NOT leak in.
797        let _ = target.kill();
798        let _ = target.wait();
799        // Drain the tail the leg may still hold for the dead target.
800        let _ = loudest_within(&leg, Duration::from_millis(1_500), f64::INFINITY);
801        let leaked = loudest_within(&leg, Duration::from_millis(1_500), 500.0);
802        let _ = decoy.kill();
803        let _ = decoy.wait();
804        drop(leg);
805        assert!(
806            leaked < 500.0,
807            "app scope leaked another process's audio after its target exited (RMS {leaked:.0})"
808        );
809    }
