1//! # AV1 OBU parsing + `av1C` configuration-record synthesis
2//!
3//! VideoToolbox can decode AV1 in hardware (A17 Pro / M3 and later), but unlike
4//! H.264/HEVC there is **no** `CMVideoFormatDescriptionCreateFromAV1ParameterSets`
5//! in objc2-core-media 0.3.2. The AV1 format description must instead be built
6//! with the generic `CMVideoFormatDescriptionCreate`, carrying an **`av1C`**
7//! configuration-record (the AV1 Codec ISO-BMFF binding, [`AV1CodecConfigurationRecord`])
8//! as a sample-description extension atom. This module is the codec-only half of
9//! that: it has **no** VideoToolbox / objc2 dependency, so it is unit-testable on
10//! any host and shared cleanly with the decoder.
11//!
12//! ## What we parse, and what we deliberately don't
13//! AV1's "low-overhead bitstream format" (what `rav1e` emits, and what goes on the
14//! MoQ wire) is a sequence of OBUs:
15//!
16//! ```text
17//!  obu_header + (optional) leb128 obu_size + payload
18//! ```
19//!
20//! To synthesise an `av1C` we need exactly the fields the config record
21//! carries, all of which live near the **front** of the sequence-header OBU:
22//! `seq_profile`, `seq_level_idx[0]`, `seq_tier[0]`, the bit-depth flags
23//! (`high_bitdepth`/`twelve_bit`), `mono_chrome`, the chroma subsampling flags, and
24//! `chroma_sample_position`. We parse just far enough to reach those — through the
25//! (skippable) `timing_info` / `decoder_model_info` / `operating_points` preamble —
26//! and stop. We do **not** decode frames; libaom/dav1d/VideoToolbox does that.
27//!
28//! We also keep `max_frame_width_minus_1` / `max_frame_height_minus_1`, which the
29//! walk crosses anyway. Those are **not** part of the config record — the `av1C`
30//! layout below has no width/height field at all — but the ISOBMFF sample entry
31//! that carries an `av1C` always does, and so does the `CMVideoFormatDescription`
32//! VideoToolbox wants (see [`Av1Config`]).
33//!
34//! ## `av1C` layout (AV1-ISOBMFF §2.3.3, `AV1CodecConfigurationRecord`)
35//! ```text
36//! marker(1)=1, version(7)=1
37//! seq_profile(3), seq_level_idx_0(5)
38//! seq_tier_0(1), high_bitdepth(1), twelve_bit(1), monochrome(1),
39//!   chroma_subsampling_x(1), chroma_subsampling_y(1), chroma_sample_position(2)
40//! reserved(3)=0, initial_presentation_delay_present(1)=0,
41//!   initial_presentation_delay_minus_one(4) OR reserved(4)=0
42//! configOBUs[]  // the sequence-header OBU, verbatim (header + size + payload)
43//! ```
44//! VideoToolbox wants this atom under the `"av1C"` key inside
45//! `kCMFormatDescriptionExtension_SampleDescriptionExtensionAtoms`.
46//!
47//! ## The record does not carry the geometry — the format description must
48//! Note what is *absent* above: there is no width or height anywhere in an
49//! `av1C`. In a real MP4 the geometry travels in the `av1C`'s enclosing `stsd`
50//! sample entry, and every muxer/demuxer pair moves it that way; the AV1
51//! equivalent of "the decoder re-derives it from the parameter sets" does not
52//! exist, because the only copy of the frame size inside the record is buried in
53//! the `configOBUs` sequence header, which VideoToolbox's AV1 decoder does not
54//! mine for it. `CMVideoFormatDescriptionCreate` must therefore be given real
55//! dimensions — a `0×0` AV1 description is accepted by the *constructor* and then
56//! rejected by `VTDecompressionSessionCreate` with `-12911`
57//! (`kVTVideoDecoderUnsupportedDataFormatErr`). Hence [`Av1Config`], which pairs
58//! the record with the geometry parsed out of the same sequence header.
59//!
60//! (H.264/HEVC never hit this: their `CreateFrom*ParameterSets` constructors read
61//! the SPS themselves, and even the generic constructor at `0×0` yields a session
62//! those decoders happily drive.)
63
64/// OBU type for the sequence header (`OBU_SEQUENCE_HEADER`).
65const OBU_SEQUENCE_HEADER: u8 = 1;
66/// OBU type for a temporal delimiter (`OBU_TEMPORAL_DELIMITER`) — zero-length
67/// payload, marks the start of a temporal unit.
68const OBU_TEMPORAL_DELIMITER: u8 = 2;

70/// One OBU located inside an AV1 bitstream, as a borrowed slice of the *whole*
71/// OBU (header + size field + payload) plus its decoded type. Keeping the whole
72/// OBU lets us copy the sequence header verbatim into the `av1C` `configOBUs`.
73#[derive(Debug, Clone, Copy)]
74pub struct Obu<'a> {
75    pub obu_type: u8,
76    /// `obu_has_size_field`. The low-overhead format lets the **last** OBU of a
77    /// stream omit its size (it then runs to the end of the stream); a
78    /// `configOBUs` entry may not, so [`build_av1c`] re-frames such an OBU
79    /// instead of copying it.
80    pub has_size_field: bool,
81    /// Just the `obu_header`: 1 byte, or 2 when `obu_extension_flag` is set.
82    pub header: &'a [u8],
83    /// The complete OBU bytes (header + leb128 size + payload), exactly as they
84    /// appear in the stream — what `av1C.configOBUs` contains verbatim.
85    pub whole: &'a [u8],
86    /// Just the payload (after header + size field). Used to parse the sequence
87    /// header fields for the `av1C` record.
88    pub payload: &'a [u8],
89}

91/// The fields extracted from a sequence-header OBU that the `av1C` record needs,
92/// plus the frame geometry the *format description* needs (see the module docs —
93/// the record itself has no width/height).
94#[derive(Debug, Clone, Copy, PartialEq, Eq)]
95pub struct SeqHeaderConfig {
96    pub seq_profile: u8,
97    pub seq_level_idx_0: u8,
98    pub seq_tier_0: u8,
99    pub high_bitdepth: u8,
100    pub twelve_bit: u8,
101    pub monochrome: u8,
102    pub chroma_subsampling_x: u8,
103    pub chroma_subsampling_y: u8,
104    pub chroma_sample_position: u8,
105    /// `max_frame_width_minus_1 + 1` — the coded width the sequence declares.
106    /// Not an `av1C` field; parsed for the format description.
107    pub max_frame_width: u32,
108    /// `max_frame_height_minus_1 + 1`.
109    pub max_frame_height: u32,
110}

112/// Everything a VideoToolbox AV1 format description needs, derived from one
113/// sequence header: the `av1C` record **and** the geometry that is not in it.
114///
115/// The two travel together because they are only correct together — a record from
116/// one sequence header paired with dimensions from another describes no stream at
117/// all. The session-rebuild check compares `record`: the geometry comes from the
118/// same bytes, so equal records imply equal dimensions.
119#[derive(Debug, Clone, PartialEq, Eq)]
120pub struct Av1Config {
121    /// The `av1C` configuration-record bytes.
122    pub record: Vec<u8>,
123    /// Coded width in luma samples.
124    pub width: u32,
125    /// Coded height in luma samples.
126    pub height: u32,
127}

129/// A minimal MSB-first bit reader over a byte slice. AV1 syntax is big-endian
130/// bit-packed (`f(n)` reads n bits MSB-first); this reads exactly that. Every
131/// read is bounds-checked and returns `None` past the end rather than panicking
132/// — the parser runs on attacker-influenced wire data.
133struct BitReader<'a> {
134    data: &'a [u8],
135    bit_pos: usize,
136}
137
138impl<'a> BitReader<'a> {
139    fn new(data: &'a [u8]) -> Self {
140        Self { data, bit_pos: 0 }
141    }
142
143    /// Read a single bit (`f(1)`).
144    fn bit(&mut self) -> Option<u8> {
145        let byte = self.bit_pos / 8;
146        if byte >= self.data.len() {
147            return None;
148        }
149        let shift = 7 - (self.bit_pos % 8);
150        let v = (self.data[byte] >> shift) & 1;
151        self.bit_pos += 1;
152        Some(v)
153    }
154
155    /// Read `n` bits MSB-first into a u32 (`f(n)`, n ≤ 32).
156    fn f(&mut self, n: u32) -> Option<u32> {
157        let mut v = 0u32;
158        for _ in 0..n {
159            v = (v << 1) | self.bit()? as u32;
160        }
161        Some(v)
162    }
163
164    /// `uvlc()` — unsigned variable-length code (Exp-Golomb-ish), used by
165    /// `timing_info` for `num_units_in_display_tick` etc. We only need to *skip*
166    /// these, but reading them correctly keeps the bit cursor aligned.
167    fn uvlc(&mut self) -> Option<u32> {
168        let mut leading_zeros = 0u32;
169        loop {
170            let done = self.bit()?;
171            if done == 1 {
172                break;
173            }
174            leading_zeros += 1;
175            if leading_zeros >= 32 {
176                return Some(u32::MAX);
177            }
178        }
179        let value = self.f(leading_zeros)?;
180        Some(value + (1u32 << leading_zeros) - 1)
181    }
182}

184/// Decode an unsigned LEB128 value at `data[pos..]`, returning `(value, bytes_read)`.
185/// AV1 uses LEB128 for `obu_size` and the temporal-unit/frame-unit framing of the
186/// length-delimited "Annex-B"-ish variants. Bounded to 8 bytes per the spec.
187fn read_leb128(data: &[u8], pos: usize) -> Option<(u64, usize)> {
188    let mut value = 0u64;
189    let mut i = 0usize;
190    loop {
191        if i >= 8 || pos + i >= data.len() {
192            return None;
193        }
194        let b = data[pos + i];
195        value |= ((b & 0x7f) as u64) << (i * 7);
196        i += 1;
197        if b & 0x80 == 0 {
198            break;
199        }
200    }
201    Some((value, i))
202}
203
204/// Append `value` as unsigned LEB128 — the inverse of [`read_leb128`], used only
205/// when [`build_av1c`] has to re-frame a size-less sequence-header OBU.
206fn write_leb128(out: &mut Vec<u8>, mut value: u64) {
207    loop {
208        let mut byte = (value & 0x7f) as u8;
209        value >>= 7;
210        if value != 0 {
211            byte |= 0x80;
212        }
213        out.push(byte);
214        if value == 0 {
215            return;
216        }
217    }
218}

220/// Split an AV1 **low-overhead** OBU stream into its OBUs (the format `rav1e`
221/// emits and that travels on our wire). Each OBU is `obu_header` (1–2 bytes) +
222/// optional `leb128 obu_size` + payload. We require `obu_has_size_field` (rav1e
223/// sets it; the low-overhead format mandates it for all but the last), and parse
224/// defensively: a malformed length stops the walk rather than panicking.
225///
226/// Returns the OBUs in stream order. Temporal delimiters are included (the caller
227/// filters), because the *whole* OBU slices are what `av1C`/`configOBUs` and the
228/// CMSampleBuffer payload are built from.
229pub fn parse_obus(stream: &[u8]) -> Vec<Obu<'_>> {
230    let mut obus = Vec::new();
231    let mut pos = 0usize;
232    while pos < stream.len() {
233        let header_start = pos;
234        let first = stream[pos];
235        // obu_header: forbidden(1)=0, type(4), extension_flag(1), has_size_field(1),
236        // reserved(1).
237        let obu_type = (first >> 3) & 0x0f;
238        let extension_flag = (first >> 2) & 1;
239        let has_size_field = (first >> 1) & 1;
240        pos += 1;
241        if extension_flag == 1 {
242            // temporal_id(3), spatial_id(2), reserved(3)
243            if pos >= stream.len() {
244                break;
245            }
246            pos += 1;
247        }
248        let header = &stream[header_start..pos];
249        let payload_len = if has_size_field == 1 {
250            let (size, n) = match read_leb128(stream, pos) {
251                Some(v) => v,
252                None => break,
253            };
254            pos += n;
255            size as usize
256        } else {
257            // No size field → this OBU runs to the end of the stream (only valid
258            // for the final OBU in low-overhead format).
259            stream.len() - pos
260        };
261        let payload_start = pos;
262        let payload_end = payload_start.checked_add(payload_len);
263        let Some(payload_end) = payload_end else {
264            break;
265        };
266        if payload_end > stream.len() {
267            break; // truncated/malformed — stop rather than read OOB
268        }
269        obus.push(Obu {
270            obu_type,
271            has_size_field: has_size_field == 1,
272            header,
273            whole: &stream[header_start..payload_end],
274            payload: &stream[payload_start..payload_end],
275        });
276        pos = payload_end;
277    }
278    obus
279}

281/// Find the first sequence-header OBU in a stream, if present.
282pub fn find_sequence_header(stream: &[u8]) -> Option<Obu<'_>> {
283    parse_obus(stream)
284        .into_iter()
285        .find(|o| o.obu_type == OBU_SEQUENCE_HEADER)
286}

288/// Parse the fields the `av1C` record needs out of a sequence-header OBU payload.
289///
290/// This walks the `sequence_header_obu()` syntax (AV1 spec §5.5) only as far as
291/// the color config, skipping the timing/decoder-model/operating-points preamble
292/// correctly so the bit cursor lands on the right fields. Returns `None` on any
293/// truncation or an unsupported shape we can't align past (we then refuse AV1
294/// rather than emit a wrong `av1C`).
295pub fn parse_sequence_header(payload: &[u8]) -> Option<SeqHeaderConfig> {
296    let mut r = BitReader::new(payload);
297
298    let seq_profile = r.f(3)? as u8;
299    let _still_picture = r.bit()?;
300    let reduced_still_picture_header = r.bit()?;
301
302    let mut seq_level_idx_0 = 0u8;
303    let mut seq_tier_0 = 0u8;
304
305    if reduced_still_picture_header == 1 {
306        // timing_info_present=0, decoder_model_info=0, etc.
307        // seq_level_idx[0] = f(5); seq_tier[0] = 0.
308        seq_level_idx_0 = r.f(5)? as u8;
309        seq_tier_0 = 0;
310    } else {
311        let timing_info_present_flag = r.bit()?;
312        let mut decoder_model_info_present_flag = 0u8;
313        let mut buffer_delay_length_minus_1 = 0u32;
314        if timing_info_present_flag == 1 {
315            // timing_info(): num_units_in_display_tick f(32), time_scale f(32),
316            // equal_picture_interval f(1), [if 1] num_ticks_per_picture_minus_1 uvlc.
317            r.f(32)?; // num_units_in_display_tick
318            r.f(32)?; // time_scale
319            let equal_picture_interval = r.bit()?;
320            if equal_picture_interval == 1 {
321                r.uvlc()?; // num_ticks_per_picture_minus_1
322            }
323            decoder_model_info_present_flag = r.bit()?;
324            if decoder_model_info_present_flag == 1 {
325                // decoder_model_info(): buffer_delay_length_minus_1 f(5),
326                // num_units_in_decoding_tick f(32), buffer_removal_time_length_minus_1 f(5),
327                // frame_presentation_time_length_minus_1 f(5).
328                buffer_delay_length_minus_1 = r.f(5)?;
329                r.f(32)?;
330                r.f(5)?;
331                r.f(5)?;
332            }
333        }
334
335        let initial_display_delay_present_flag = r.bit()?;
336        let operating_points_cnt_minus_1 = r.f(5)?;
337        for i in 0..=operating_points_cnt_minus_1 {
338            r.f(12)?; // operating_point_idc[i]
339            let level = r.f(5)?; // seq_level_idx[i]
340            let mut tier = 0u32;
341            if level > 7 {
342                tier = r.bit()? as u32; // seq_tier[i]
343            }
344            if decoder_model_info_present_flag == 1 {
345                let decoder_model_present_for_this_op = r.bit()?;
346                if decoder_model_present_for_this_op == 1 {
347                    // operating_parameters_info(): decoder_buffer_delay,
348                    // encoder_buffer_delay each (buffer_delay_length_minus_1+1) bits,
349                    // then low_delay_mode_flag f(1).
350                    let n = buffer_delay_length_minus_1 + 1;
351                    r.f(n)?;
352                    r.f(n)?;
353                    r.bit()?;
354                }
355            }
356            if initial_display_delay_present_flag == 1 {
357                let initial_display_delay_present_for_this_op = r.bit()?;
358                if initial_display_delay_present_for_this_op == 1 {
359                    r.f(4)?; // initial_display_delay_minus_1[i]
360                }
361            }
362            if i == 0 {
363                seq_level_idx_0 = level as u8;
364                seq_tier_0 = tier as u8;
365            }
366        }
367    }
368
369    // frame_width_bits_minus_1 f(4), frame_height_bits_minus_1 f(4),
370    // max_frame_width_minus_1 f(n), max_frame_height_minus_1 f(n).
371    let frame_width_bits_minus_1 = r.f(4)?;
372    let frame_height_bits_minus_1 = r.f(4)?;
373    let max_frame_width = r.f(frame_width_bits_minus_1 + 1)? + 1;
374    let max_frame_height = r.f(frame_height_bits_minus_1 + 1)? + 1;
375
376    // frame_id_numbers_present_flag (0 unless reduced_still_picture_header).
377    let mut frame_id_numbers_present_flag = 0u8;
378    if reduced_still_picture_header == 0 {
379        frame_id_numbers_present_flag = r.bit()?;
380    }
381    if frame_id_numbers_present_flag == 1 {
382        r.f(4)?; // delta_frame_id_length_minus_2
383        r.f(3)?; // additional_frame_id_length_minus_1
384    }
385
386    r.bit()?; // use_128x128_superblock
387    r.bit()?; // enable_filter_intra
388    r.bit()?; // enable_intra_edge_filter
389
390    if reduced_still_picture_header == 1 {
391        // enable_* flags forced; seq_force_* set to defaults. Nothing more to read
392        // before the color config in this branch beyond what the spec fixes.
393    } else {
394        r.bit()?; // enable_interintra_compound
395        r.bit()?; // enable_masked_compound
396        r.bit()?; // enable_warped_motion
397        r.bit()?; // enable_dual_filter
398        let enable_order_hint = r.bit()?;
399        if enable_order_hint == 1 {
400            r.bit()?; // enable_jnt_comp
401            r.bit()?; // enable_ref_frame_mvs
402        }
403        let seq_choose_screen_content_tools = r.bit()?;
404        let seq_force_screen_content_tools = if seq_choose_screen_content_tools == 1 {
405            2u32 // SELECT_SCREEN_CONTENT_TOOLS
406        } else {
407            r.bit()? as u32
408        };
409        if seq_force_screen_content_tools > 0 {
410            let seq_choose_integer_mv = r.bit()?;
411            if seq_choose_integer_mv == 0 {
412                r.bit()?; // seq_force_integer_mv
413            }
414        }
415        if enable_order_hint == 1 {
416            r.f(3)?; // order_hint_bits_minus_1
417        }
418    }
419
420    r.bit()?; // enable_superres
421    r.bit()?; // enable_cdef
422    r.bit()?; // enable_restoration
423
424    // color_config():
425    let high_bitdepth = r.bit()?;
426    let mut twelve_bit = 0u8;
427    let bit_depth;
428    if seq_profile == 2 && high_bitdepth == 1 {
429        twelve_bit = r.bit()?;
430        bit_depth = if twelve_bit == 1 { 12 } else { 10 };
431    } else {
432        // profile <= 2: bit_depth = high_bitdepth ? 10 : 8
433        bit_depth = if high_bitdepth == 1 { 10 } else { 8 };
434    }
435    let _ = bit_depth;
436
437    let monochrome = if seq_profile == 1 {
438        0u8 // mono_chrome not coded for profile 1
439    } else {
440        r.bit()?
441    };
442
443    let color_description_present_flag = r.bit()?;
444    let (mut color_primaries, mut transfer_characteristics, mut matrix_coefficients) =
445        (2u32, 2u32, 2u32); // CP/TC/MC_UNSPECIFIED
446    if color_description_present_flag == 1 {
447        color_primaries = r.f(8)?;
448        transfer_characteristics = r.f(8)?;
449        matrix_coefficients = r.f(8)?;
450    }
451
452    let chroma_subsampling_x;
453    let chroma_subsampling_y;
454    let mut chroma_sample_position = 0u8; // CSP_UNKNOWN
455
456    if monochrome == 1 {
457        r.bit()?; // color_range
458        // subsampling 1,1; chroma_sample_position 0.
459        chroma_subsampling_x = 1;
460        chroma_subsampling_y = 1;
461    } else if color_primaries == 1 && transfer_characteristics == 13 && matrix_coefficients == 0 {
462        // sRGB: color_range=1, subsampling 0,0.
463        chroma_subsampling_x = 0;
464        chroma_subsampling_y = 0;
465    } else {
466        r.bit()?; // color_range
467        if seq_profile == 0 {
468            chroma_subsampling_x = 1;
469            chroma_subsampling_y = 1;
470        } else if seq_profile == 1 {
471            chroma_subsampling_x = 0;
472            chroma_subsampling_y = 0;
473        } else {
474            // profile 2
475            if bit_depth == 12 {
476                chroma_subsampling_x = r.bit()?;
477                if chroma_subsampling_x == 1 {
478                    chroma_subsampling_y = r.bit()?;
479                } else {
480                    chroma_subsampling_y = 0;
481                }
482            } else {
483                chroma_subsampling_x = 1;
484                chroma_subsampling_y = 0;
485            }
486        }
487        if chroma_subsampling_x == 1 && chroma_subsampling_y == 1 {
488            chroma_sample_position = r.f(2)? as u8;
489        }
490    }
491
492    Some(SeqHeaderConfig {
493        seq_profile,
494        seq_level_idx_0,
495        seq_tier_0,
496        high_bitdepth,
497        twelve_bit,
498        monochrome,
499        chroma_subsampling_x,
500        chroma_subsampling_y,
501        chroma_sample_position,
502        max_frame_width,
503        max_frame_height,
504    })
505}

507/// Build the `av1C` configuration-record bytes from the sequence-header OBU.
508///
509/// `seq_header` is the sequence-header OBU as located in the stream, normally
510/// copied into `configOBUs` **verbatim** (byte-for-byte what a muxer would write
511/// — cross-checked against ffmpeg's `av1C` box for the committed fixture, see the
512/// tests); `cfg` is the parsed config. The returned bytes are exactly what goes
513/// under the `"av1C"` key in the format description's
514/// `SampleDescriptionExtensionAtoms`.
515///
516/// The one departure from a verbatim copy: an OBU with `obu_has_size_field = 0`
517/// is re-framed with an explicit size. That flag is legal in the low-overhead
518/// bitstream for the last OBU of a stream, where "no size" means "runs to the
519/// end" — a meaning `configOBUs` cannot express, and which would otherwise make
520/// [`parse_obus`] swallow every following OBU into the record.
521pub fn build_av1c(cfg: &SeqHeaderConfig, seq_header: &Obu<'_>) -> Vec<u8> {
522    let config_obu = if seq_header.has_size_field {
523        seq_header.whole.to_vec()
524    } else {
525        let mut framed = seq_header.header.to_vec();
526        framed[0] |= 0b0000_0010; // set obu_has_size_field
527        write_leb128(&mut framed, seq_header.payload.len() as u64);
528        framed.extend_from_slice(seq_header.payload);
529        framed
530    };
531
532    let mut out = Vec::with_capacity(4 + config_obu.len());
533
534    // Byte 0: marker(1)=1, version(7)=1  →  0b1_0000001 = 0x81.
535    out.push(0x81);
536    // Byte 1: seq_profile(3), seq_level_idx_0(5).
537    out.push(((cfg.seq_profile & 0x07) << 5) | (cfg.seq_level_idx_0 & 0x1f));
538    // Byte 2: seq_tier_0(1), high_bitdepth(1), twelve_bit(1), monochrome(1),
539    //         chroma_subsampling_x(1), chroma_subsampling_y(1), chroma_sample_position(2).
540    out.push(
541        ((cfg.seq_tier_0 & 1) << 7)
542            | ((cfg.high_bitdepth & 1) << 6)
543            | ((cfg.twelve_bit & 1) << 5)
544            | ((cfg.monochrome & 1) << 4)
545            | ((cfg.chroma_subsampling_x & 1) << 3)
546            | ((cfg.chroma_subsampling_y & 1) << 2)
547            | (cfg.chroma_sample_position & 0x03),
548    );
549    // Byte 3: reserved(3)=0, initial_presentation_delay_present(1)=0,
550    //         initial_presentation_delay_minus_one(4) → all zero.
551    out.push(0x00);
552
553    // configOBUs: the sequence-header OBU.
554    out.extend_from_slice(&config_obu);
555    out
556}

558/// Convenience: from a full AV1 temporal-unit byte stream that contains a
559/// sequence-header OBU, synthesise the `av1C` record **and** the coded geometry
560/// that goes with it. Returns `None` if there is no sequence header, it can't be
561/// parsed, or it declares a degenerate size — the callers then refuse AV1 rather
562/// than build a format description that `VTDecompressionSessionCreate` would
563/// reject (see the module docs).
564pub fn av1_config_from_stream(stream: &[u8]) -> Option<Av1Config> {
565    let sh = find_sequence_header(stream)?;
566    let cfg = parse_sequence_header(sh.payload)?;
567    if cfg.max_frame_width == 0 || cfg.max_frame_height == 0 {
568        return None;
569    }
570    Some(Av1Config {
571        record: build_av1c(&cfg, &sh),
572        width: cfg.max_frame_width,
573        height: cfg.max_frame_height,
574    })
575}
576
577/// True if a temporal-delimiter-only / empty stream (nothing to decode).
578pub fn is_temporal_delimiter_only(stream: &[u8]) -> bool {
579    let obus = parse_obus(stream);
580    !obus.is_empty() && obus.iter().all(|o| o.obu_type == OBU_TEMPORAL_DELIMITER)
581}

607    fn leb128_roundtrip() {
608        // 0x80 0x01 = 128; single byte 0x7f = 127.
609        assert_eq!(read_leb128(&[0x7f], 0), Some((127, 1)));
610        assert_eq!(read_leb128(&[0x80, 0x01], 0), Some((128, 2)));
611        assert_eq!(read_leb128(&[0xff, 0x7f], 0), Some((16383, 2)));
612        // Truncated continuation byte → None, no panic.
613        assert_eq!(read_leb128(&[0x80], 0), None);
614        // …and back out again.
615        for value in [0u64, 1, 127, 128, 300, 16383, 16384, 1 << 40] {
616            let mut buf = Vec::new();
617            write_leb128(&mut buf, value);
618            assert_eq!(read_leb128(&buf, 0), Some((value, buf.len())), "{value}");
619        }
