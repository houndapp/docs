1//! The discrete audible-gap events the bus records ([`AudioGlitch`]) and the
2//! per-push status a producer reports back ([`PushStatus`]).
3
4use std::collections::VecDeque;
5use std::sync::Mutex;
6
7/// What a [`BusProducer::push`](super::BusProducer::push) reports back so the
8/// caller can surface per-peer audio-health stats without a second lock —
9/// everything the debug report needs to attribute a glitch to the jitter-buffer
10/// stage.
11pub struct PushStatus {
12    /// Current smoothed inter-arrival jitter estimate (µs, RFC 3550 `J`).
13    pub jitter_us: i64,
14    /// Frames currently queued in the jitter buffer (the playout depth gauge).
15    pub buffered_frames: u32,
16    /// Current jitter-buffer depth (µs) — what playout is delayed by.
17    pub target_delay_us: i64,
18    /// Depth the estimator recommends (`floor + 4·J`). `recommended ≫ target` =
19    /// buffer too shallow for the link (the grow step can't keep up) → underflows.
20    pub recommended_delay_us: i64,
21    /// Cumulative frames dropped as arrived-too-late (network late-arrival / loss).
22    pub late_drops: u64,
23    /// Cumulative frames dropped because the buffer was full (backed-up / clock skew).
24    pub overflow_drops: u64,
25    /// Cumulative mix-tick underflows for THIS slot (a primed slot that came up short
26    /// when the RT bus drained it — the per-peer audible gap). Read from the slot's
27    /// running starve count, so it reflects the latest value as of this push.
28    pub underflow_total: u64,
29}
30
31/// A sentinel `peer_tag` for a producer registered without attribution (the plain
32/// [`PlayoutBus::register`](super::PlayoutBus::register)) or a bus-global event
33/// (a late tick) — the glitch is recorded but not tied to a specific peer.
34pub const NO_PEER_TAG: u32 = u32::MAX;
35
36/// What an [`AudioGlitch`] is — the discrete audible-gap events the bus records, each
37/// already attributed to a stage so the debug report reads a cause, not a raw count.
38#[derive(Debug, Clone, Copy, PartialEq, Eq)]
39pub enum GlitchKind {
40    /// A primed slot came up short when the RT bus drained it: the per-peer jitter
41    /// buffer had nothing due (network starvation / buffer too shallow).
42    Underflow,
43    /// The 10 ms mix tick fired late (the RT feeder thread was descheduled) — a
44    /// device-side gap affecting ALL audio. `detail_us` is how late.
45    LateTick,
46    /// The output sink was torn down and re-opened (the stream died, or an automatic
47    /// route change settled and the device geometry had to be re-derived). The open is
48    /// a BLOCKING device call on the RT mix thread and is followed by a fresh
49    /// [`OUTPUT_CUSHION_MS`](super::output::OUTPUT_CUSHION_MS) silence prefill,
50    /// so it shows up as a late tick + output
51    /// underruns — without this event those get misattributed to CPU contention.
52    /// Affects ALL audio; `detail_us` is how long the reopen took.
53    DeviceReopen,
54}
55
56/// One discrete audio-glitch event captured at the instant it happened in the RT mix
57/// loop. The client drains these into a rolling history the debug report renders as a
58/// timestamped, cause-attributed glitch log.
59#[derive(Debug, Clone, Copy)]
60pub struct AudioGlitch {
61    /// Capture-clock time (µs) the glitch occurred (the bus's `clock.now()`).
62    pub at_us: i64,
63    /// The peer this is attributed to (the control-plane peer id passed to
64    /// [`PlayoutBus::register_tagged`](super::PlayoutBus::register_tagged)), or [`NO_PEER_TAG`].
65    pub peer_tag: u32,
66    /// The producer label (`"mic"` / `"screen_audio"`), or `"tick"` for a late tick.
67    pub label: &'static str,
68    pub kind: GlitchKind,
69    /// Kind-specific magnitude in µs (e.g. how late a [`GlitchKind::LateTick`] was);
70    /// 0 when not applicable.
71    pub detail_us: i64,
72}
73
74/// Max glitch events the bus retains between drains (bounded, drop-oldest). The
75/// client drains every stats window (1 Hz), so this only needs one window's events
76/// plus headroom.
77pub(super) const GLITCH_RING_CAP: usize = 256;
78
79/// Append one glitch to the bus's bounded (drop-oldest) ring. Used for the events
80/// recorded OUTSIDE the per-tick batch drain — currently the device reopen, which
81/// happens after the batch has already been flushed.
82pub(super) fn record_glitch(ring: &Mutex<VecDeque<AudioGlitch>>, glitch: AudioGlitch) {
83    if let Ok(mut r) = ring.lock() {
84        if r.len() >= GLITCH_RING_CAP {
85            r.pop_front();
86        }
87        r.push_back(glitch);
88    }
89}
