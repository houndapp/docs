22/// One decoded audio frame in a slot's jitter buffer: the sender's capture-clock
23/// timestamp (for playout scheduling) plus its 48 kHz mono PCM. The pump pushes
24/// these; the RT thread pops the due ones into the carry-queue.
25struct MixFrame {
26    ts_us: i64,
27    samples: Vec<i16>,
28}
29
30impl Timestamped for MixFrame {
31    fn timestamp_us(&self) -> i64 {
32        self.ts_us
33    }
34}

36/// The per-peer jitter state the PUMP mutates (push) and the RT thread reads (pop),
37/// behind one mutex held only for the bounded buffer op. Owning the
38/// [`AdaptiveJitter`] here keeps push (fold arrival → retarget depth) and pop (the
39/// release schedule) operating on one consistent buffer under one lock.
40struct PlayoutState {
41    buffer: PlayoutBuffer<MixFrame>,
42    adapt: AdaptiveJitter,
43}
44
45/// What [`Slot::pull_into`] reports back to the bus loop for one tick.
46struct PullOutcome {
47    /// Samples this slot contributed to the mix this tick. (Read by tests asserting
48    /// the drain contract; the production loop only acts on `starved`.)
49    #[allow(dead_code)]
50    contributed: usize,
51    /// Whether a primed slot came up short (an underflow this tick).
52    starved: bool,
53}

55/// One producer's per-peer slot. The PUMP pushes decoded, timestamped PCM into
56/// `state.buffer` (the jitter buffer); the BUS THREAD pops the due frames on its
57/// own real-time clock into `carry` and drains [`MIX_CHUNK_SAMPLES`] per tick. The
58/// two mutexes are each held only for a bounded op, never across IO; the producer
59/// is the async per-peer pump, the consumer is the bus's own thread (neither is a
60/// cpal RT callback).
61pub(super) struct Slot {
62    /// Jitter buffer + adaptive depth. Pump writes (push), bus reads (pop_due_into).
63    state: Mutex<PlayoutState>,
64    /// Sample carry-queue between the 20 ms popped frames and the 10 ms drained
65    /// chunks. Touched ONLY by the bus thread, so it is uncontended — the mutex is
66    /// for the `Arc<Slot>` interior mutability, not cross-thread coordination.
67    carry: Mutex<VecDeque<i16>>,
68    /// Shared bus counter of mix STARVATION events: a tick where this slot was
69    /// primed (not idle) but its carry came up SHORT of a full chunk — a per-peer
70    /// audible gap. Held as a shared `Arc` so a dropped slot's events still count
71    /// toward the bus total. THIS is the per-peer stutter signal (distinct from the
72    /// output-ring underrun, which is the device-side gap). With the pull model a
73    /// rising count means the JITTER BUFFER itself ran dry (real network loss /
74    /// total starvation), not a producer-scheduling race.
75    starves: Arc<AtomicU64>,
76    /// THIS slot's own cumulative starve count (the shared `starves` above is the
77    /// bus-wide sum). Used only to attribute the aggregate to a source kind
78    /// ([`Self::label`]) so the stutter log can say WHICH audio starves — e.g. a
79    /// screen-share's intermittent app audio vs the remote mic (voice).
80    pub(super) own_starves: AtomicU64,
81    /// A human label for the slot's source kind ("mic" / "screen_audio"), for the
82    /// per-kind starve breakdown. Static — the call sites pass a string literal.
83    pub(super) label: &'static str,
84    /// DTX-aware starvation classifier (warmup gate + gap-onset window + recent-energy
85    /// silence discriminator). Touched ONLY by the bus thread each tick, so the mutex
86    /// is uncontended interior mutability over the `Arc<Slot>`, never cross-thread
87    /// coordination. See [`crate::audio::starve`].
88    starve: Mutex<StarveTracker>,
89    /// The control-plane peer id this slot's audio belongs to ([`NO_PEER_TAG`](super::NO_PEER_TAG) if
90    /// registered untagged), so a recorded underflow [`AudioGlitch`] can name the peer.
91    peer_tag: u32,
92    /// Free-list of PCM buffers recycled from the bus thread's drain back to the
93    /// pump's push, so neither the allocation nor the free lands on the RT tick.
94    /// Bounded by [`SLOT_POOL_CAP`]; the RT side `try_lock`s and simply frees the
95    /// buffer when contended, so it can never block on the pump.
96    pool: Mutex<Vec<Vec<i16>>>,
97}

128    /// Producer side (pump): fold this frame's arrival into the adaptive jitter depth
129    /// and push it into the jitter buffer, both under one lock so the depth retarget
130    /// and the insert stay consistent. Returns the post-push depth + jitter estimate
131    /// for the caller's per-peer stats.
132    ///
133    /// The two clocks are deliberately separate, and both are on the shared
134    /// [`CaptureClock`](hound_media::CaptureClock) the bus thread pops against:
135    ///
136    /// * `arrival_us` — when the frame came off the WIRE, or `None` for a frame that
137    ///   never crossed it (locally CONCEALED audio; see
138    ///   [`BusProducer::push_concealed`](super::BusProducer::push_concealed)),
139    ///   which is folded into the buffer but never into the estimator. Only the jitter
140    ///   estimator sees this instant, and it must be the wire one, because that estimator
141    ///   is differential (`D = arrivalΔ − tsΔ`): feed it anything the receive pipeline
142    ///   delayed — or synthesized — and it measures the pipeline instead of the link.
143    /// * `now_us` — when the frame became available for PLAYOUT, i.e. now. This is what
144    ///   the buffer schedules and computes cushion against, so the cushion stays honest
145    ///   about how long the frame really has before it is due.
146    ///
147    /// They differ whenever something between the wire and here holds a frame — the
148    /// audio path's encoded reorder hold being the case that matters. See
149    /// [`BusProducer::push_at`](super::BusProducer::push_at).
150    pub(super) fn push(
151        &self,
152        arrival_us: Option<i64>,
153        now_us: i64,
154        ts_us: i64,
155        samples: &[i16],
156    ) -> PushStatus {
157        // Own the samples BEFORE taking the lock: this copy must not happen inside
158        // `state.lock()`, which the RT mix pull also takes every 10 ms tick — an alloc
159        // under that lock would let a fill push briefly block the sacred playout
160        // thread. A recycled buffer from the pool (returned by the RT drain) usually
161        // makes it allocation-free entirely.
162        let mut buf = self
163            .pool
164            .lock()
165            .ok()
166            .and_then(|mut p| p.pop())
167            .unwrap_or_default();
168        buf.clear();
169        buf.extend_from_slice(samples);
170        let frame = MixFrame {
171            ts_us,
172            samples: buf,
173        };
174        // `ever_fed` is set when audio first REACHES the carry (the first release
175        // below), NOT here: a frame pushed now sits in the jitter buffer for ~target
176        // ms before it is due, and that intentional warmup delay must not count as
177        // starvation. Setting it on push would flag the whole warmup as a gap.
178        let Ok(mut st) = self.state.lock() else {
179            // poisoned ⇒ a prior op panicked; degrade rather than propagate.
180            return PushStatus {
181                jitter_us: 0,
182                buffered_frames: 0,
183                target_delay_us: 0,
184                recommended_delay_us: 0,
185                late_drops: 0,
186                overflow_drops: 0,
187                underflow_total: 0,
188            };
189        };
190        let PlayoutState { buffer, adapt } = &mut *st;
191        if let Some(arrival_us) = arrival_us {
192            adapt.on_frame(buffer, arrival_us, ts_us);
193        }
194        buffer.push(now_us, frame);
195        PushStatus {
196            jitter_us: adapt.jitter_us(),
197            buffered_frames: buffer.len() as u32,
198            target_delay_us: buffer.target_delay_us(),
199            recommended_delay_us: adapt.recommended_delay_us(),
200            late_drops: buffer.late_drops(),
201            overflow_drops: buffer.overflow_drops(),
202            // The slot's running starve count (updated by the RT pop) — read here so
203            // the per-peer underflow gauge advances without a second lock.
204            underflow_total: self.own_starves.load(Ordering::Relaxed),
205        }
206    }
207
208    /// Bus side (one tick): release every frame whose playout time has arrived (up
209    /// to `now_us + MIX_PULL_LOOKAHEAD_US`) into the carry-queue, then accumulate up
210    /// to `out.len()` carry samples into `out` (summing in i32), leaving the rest of
211    /// `out` untouched. Returns a [`PullOutcome`] (samples contributed + whether this
212    /// was an underflow), so the bus loop can record a per-peer glitch. Bumps the
213    /// shared starvation counter when a primed, non-idle slot comes up short — which,
214    /// under the pull model, means the jitter buffer genuinely had nothing due
215    /// (network starvation), not a producer-scheduling race.
216    ///
217    /// `due` is the bus loop's release scratch ([`MixScratch::due`]), reused across
218    /// every slot and every tick: this thread must not allocate, so the frames come
219    /// out via `pop_due_into` rather than a fresh `Vec` per call. The invariant is
220    /// **clear on entry**, which is what keeps one slot's released frames out of the
221    /// next slot's carry; draining them is best-effort, since a poisoned carry lock
222    /// leaves them in the scratch to be dropped by the next call's clear.
223    fn pull_into(&self, now_us: i64, out: &mut [i32], due: &mut Vec<MixFrame>) -> PullOutcome {
224        // 1) Release due frames from the jitter buffer into the carry (bounded).
225        // The jitter lock is held for the drain ONLY — the carry (and the pool
226        // `try_lock`) are touched after it is released, never nested under it.
227        due.clear();
228        if let Ok(mut st) = self.state.lock() {
229            st.buffer.pop_due_into(now_us + MIX_PULL_LOOKAHEAD_US, due);
230        }
231        if !due.is_empty()
232            && let Ok(mut q) = self.carry.lock()
233        {
234            for frame in due.drain(..) {
235                // Bulk-append the frame's samples, then trim the oldest past
236                // capacity in ONE drain (drop-oldest backpressure) — a single
237                // memcpy + bounds check per frame instead of one per sample.
238                let mut buf = frame.samples;
239                q.extend(buf.iter().copied());
240                if q.len() > SLOT_CAPACITY_SAMPLES {
241                    let excess = q.len() - SLOT_CAPACITY_SAMPLES;
242                    q.drain(..excess);
243                }
244                // Hand the buffer back to the pump instead of freeing it on this
245                // (RT) thread. Best-effort: `try_lock` so the RT tick never waits
246                // on the pump — a contended tick just drops the buffer as before.
247                buf.clear();
248                if let Ok(mut pool) = self.pool.try_lock()
249                    && pool.len() < SLOT_POOL_CAP
250                {
251                    pool.push(buf);
252                }
253            }
254        }
255
256        // 2) Drain a chunk from the carry into the mix accumulator, tracking the peak
257        // |sample| released so the starve classifier can tell speech from silence.
258        let want = out.len();
259        let mut peak_abs: i32 = 0;
260        let n = {
261            let Ok(mut q) = self.carry.lock() else {
262                return PullOutcome {
263                    contributed: 0,
264                    starved: false,
265                };
266            };
267            let n = want.min(q.len());
268            // Bulk-drain the oldest n samples into the mix accumulator (sum in i32),
269            // one pass instead of a per-sample pop_front.
270            for (slot, s) in out.iter_mut().zip(q.drain(..n)) {
271                *slot += s as i32;
272                let a = (s as i32).abs();
273                if a > peak_abs {
274                    peak_abs = a;
275                }
276            }
277            n
278        };
279
280        // 3) Starvation accounting (bus thread only) via the DTX-aware classifier: a
281        // dry/short carry counts as audible stutter only if real audio was recently
282        // playing — a silent peer's comfort-noise gaps are not stutter. The tracker
283        // owns the warmup gate + gap-onset window + silence discrimination.
284        let counted = self
285            .starve
286            .lock()
287            .map(|mut tr| {
288                tr.observe(Tick {
289                    contributed: n,
290                    want,
291                    peak_abs,
292                })
293            })
294            .unwrap_or(false);
295        if counted {
296            self.starves.fetch_add(1, Ordering::Relaxed);
297            self.own_starves.fetch_add(1, Ordering::Relaxed);
298        }
299        PullOutcome {
300            contributed: n,
301            starved: counted,
302        }
303    }
304
305    /// Test-only: seed the carry-queue directly (bypassing the jitter buffer) so the
306    /// pure sum/clamp/drain unit tests can drive the accumulator without scheduling.
307    /// The first `pull_into` that drains a sample marks the slot fed (via the
308    /// classifier), so starvation accounting goes live exactly as in production.
309    #[cfg(test)]
310    pub(super) fn seed_carry(&self, samples: &[i16]) {
311        let mut q = self.carry.lock().unwrap();
312        for &s in samples {
313            if q.len() >= SLOT_CAPACITY_SAMPLES {
314                q.pop_front();
315            }
316            q.push_back(s);
317        }
318    }
319}

321/// Pull one mixed [`MIX_CHUNK_SAMPLES`]-sample chunk from the snapshot's slots at
322/// local time `now_us`: each slot releases its due jitter-buffered frames (on the
323/// RT clock) into its carry-queue and accumulates a chunk in i32 (no overflow on N
324/// loud peers), then **soft-limits** the sum and converts to i16. A slot with
325/// nothing due contributes silence. Returns the mixed chunk.
326///
327/// The limiter is libwebrtc's mixer shape (`frame_combiner.cc`): straight sum →
328/// AGC2 limiter → saturating convert. Without it the i16 conversion IS the
329/// overload behaviour — a hard clip, whose flat tops are broadband distortion
330/// across every peer at once. That was survivable only while every source was
331/// quiet; it is not once the capture chain hits its loudness target, which is why
332/// this landed with the gain raise (see `audio::apm`).
333///
334/// Allocation-free given a warm [`MixScratch`] and a `glitches` buffer with capacity
335/// (the bus loop pre-sizes both — see
336/// [`TICK_GLITCH_CAPACITY`](super::TICK_GLITCH_CAPACITY)), except for one
337/// 1-element `Vec` of channel views that `Limiter::process` builds internally per
338/// call. Not pure: the limiter carries gain state ACROSS chunks (its whole point —
339/// a per-chunk-independent gain would step audibly at chunk boundaries), so the
340/// bus loop's several catch-up chunks must run through the same [`MixScratch`], in
341/// order. `now_us` still makes the release schedule explicit/testable.
342pub(super) fn mix_chunk(
343    slots: &[Arc<Slot>],
344    now_us: i64,
345    scratch: &mut MixScratch,
346    glitches: &mut Vec<AudioGlitch>,
347) -> [i16; MIX_CHUNK_SAMPLES] {
348    let MixScratch {
349        acc,
350        due,
351        limit,
352        limiter,
353    } = scratch;
354    acc.fill(0);
355    for slot in slots {
356        let outcome = slot.pull_into(now_us, acc, due);
357        if outcome.starved {
358            // Record a per-peer underflow at the moment it happened, attributed to
359            // the slot's peer + source kind (the bus loop drains these into the ring).
360            glitches.push(AudioGlitch {
361                at_us: now_us,
362                peer_tag: slot.peer_tag,
363                label: slot.label,
364                kind: GlitchKind::Underflow,
365                detail_us: 0,
366            });
367        }
368    }
369    // The limiter works in libwebrtc's FloatS16 domain — float values on the i16
370    // scale — so the i32 accumulator crosses to f32 for the gain pass and back.
371    // Below the knee (~0 dBFS) it looks up an exact 1.0 and this is a no-op round
372    // trip; above it, it spends gain instead of flat-topping the waveform.
373    for (f, &sum) in limit.iter_mut().zip(acc.iter()) {
374        *f = sum as f32;
375    }
376    // RT caveat: `Limiter::process` collects a 1-element `Vec` of channel views per
377    // call — the only allocator touch left on the tick, and it comes from inside
378    // the crate. Bounded and tiny (one 16-byte alloc per 10 ms chunk), but it IS an
379    // allocation on an RT-promoted thread; if the tick-lateness histogram ever
380    // implicates it, the fix is a fork or a hand-rolled gain-curve pass.
381    limiter.process(&mut [limit.as_mut_slice()]);
382    let mut out = [0i16; MIX_CHUNK_SAMPLES];
383    for (o, &f) in out.iter_mut().zip(limit.iter()) {
384        // The limiter already clamps to the FloatS16 rails, so this normally only
385        // rounds — but it stays SATURATING as the backstop: no change upstream may
386        // ever wrap a sample to the opposite sign.
387        *o = f.round().clamp(i16::MIN as f32, i16::MAX as f32) as i16;
388    }
389    out
390}

392/// Everything one mix tick reuses so the RT bus thread allocates nothing. Owned by
393/// the bus loop for the life of the thread and threaded through [`mix_chunk`].
394pub(super) struct MixScratch {
395    /// Per-tick i32 accumulator the slots sum into (i32 so N loud peers cannot wrap
396    /// before the limiter sees the true sum).
397    acc: [i32; MIX_CHUNK_SAMPLES],
398    /// Frames released from ONE slot's jitter buffer this tick — `pop_due_into`'s
399    /// destination, cleared and drained per slot. Pre-sized ([`MIX_DUE_SCRATCH_FRAMES`])
400    /// so a steady tick never grows it; only a descheduled tick's catch-up burst can,
401    /// and only until it has grown once.
402    due: Vec<MixFrame>,
403    /// FloatS16 view of [`Self::acc`] the limiter gain-scales in place (it works in
404    /// float on the i16 scale, not in normalised float).
405    limit: [f32; MIX_CHUNK_SAMPLES],
406    /// The mix bus's output limiter — AGC2's, the same one the capture chain ends
407    /// in. STATEFUL across chunks (gain smoothing, attack/decay envelope), so there
408    /// is exactly one per bus thread and every chunk of every tick goes through it
409    /// in order. [`MIX_CHUNK_SAMPLES`] is its frame size: 480 = 10 ms @ 48 kHz,
410    /// which is also its maximum and divides its 20 sub-frames evenly.
411    limiter: Limiter,
412}
413
414impl MixScratch {
415    pub(super) fn new() -> Self {
416        Self {
417            acc: [0i32; MIX_CHUNK_SAMPLES],
418            due: Vec::with_capacity(MIX_DUE_SCRATCH_FRAMES),
419            limit: [0.0f32; MIX_CHUNK_SAMPLES],
420            limiter: Limiter::new(MIX_CHUNK_SAMPLES),
421        }
422    }
423}

425/// Wrap a mixed 10 ms chunk in the [`PcmFrame`] shape the output sink and
426/// `Apm::process_render` take. The timestamp is unused on the render/playout side
427/// (the APM render path ignores it; the cpal ring is time-agnostic), so 0 is fine.
428pub(super) fn chunk_to_frame(samples: Vec<i16>) -> PcmFrame {
429    PcmFrame::at_engine_rate(samples, TimestampUs::ZERO)
430}

438    /// The mix summing/clamping operates on each slot's carry-queue (what the RT
439    /// thread drains after releasing due frames). A huge `now` makes the jitter-pop

444    /// Two known sine producers sum sample-for-sample into the mixed chunk.
445    #[test]
446    fn mixes_two_producers_by_summing() {
447        let registry = Arc::new(Registry::default());
448        let a = Arc::new(Slot::unmetered());
449        let b = Arc::new(Slot::unmetered());
450        registry.slots.lock().unwrap().insert(0, a.clone());
451        registry.slots.lock().unwrap().insert(1, b.clone());

496    /// A 400 Hz sine chunk at peak `amp`, phase-continuous across chunk index `n`
497    /// (400 Hz is exactly 4 periods per 480-sample chunk). `amp` above full scale
498    /// yields the HARD-CLIPPED waveform, because the `as i16` cast saturates — which
499    /// is precisely the control signal [`limiter_soft_limits_a_hot_sum`] needs.
500    fn sine_chunk(n: usize, amp: f32) -> [i16; MIX_CHUNK_SAMPLES] {
501        let mut out = [0i16; MIX_CHUNK_SAMPLES];
502        for (i, s) in out.iter_mut().enumerate() {
503            let t = (n * MIX_CHUNK_SAMPLES + i) as f32 / SAMPLE_RATE.hz() as f32;
504            *s = (amp * (std::f32::consts::TAU * 400.0 * t).sin()) as i16;
505        }
506        out
507    }

509    /// Longest run of consecutive samples pinned to an i16 rail — the signature of
510    /// hard clipping. A limiter touches the rail at isolated peaks; a clipper
511    /// flattens whole arcs of the waveform, and it is the flat top (not the peak
512    /// itself) that is audible as broadband distortion.
513    fn longest_rail_run(samples: &[i16]) -> usize {
514        let (mut longest, mut run) = (0usize, 0usize);
515        for &s in samples {
516            if s == i16::MAX || s == i16::MIN {

845    // ── "Audio is sacred" end-to-end invariants (jitter buffer → carry → starve) ──
846    //
847    // These drive the REAL Slot — adaptive jitter buffer, carry, and the DTX-aware
848    // StarveTracker — through a scripted arrival schedule on a synthetic 10 ms pull
849    // clock (no device, no thread, fully deterministic). They assert the invariant the
850    // wire-priority + DTX fixes exist to uphold: with on-time delivery active audio
851    // NEVER starves, and a silent peer never false-positives. They fail loudly if a
852    // change to the buffer/carry/starve cores reintroduces playout stutter.
853
854    /// 20 ms Opus frame at 48 kHz mono.
855    const FRAME_SAMPLES_20MS: usize = (SAMPLE_RATE.hz() as usize / 1000) * 20;
856    /// 10 ms synthetic pull tick.
857    const TICK_US: i64 = MIX_TICK_MS as i64 * 1_000;
858
859    /// 20 ms in capture-clock µs (one frame's timestamp step).
860    const FRAME_TS_US: i64 = FRAME_SAMPLES_20MS as i64 * 1_000_000 / 48_000;
861
862    /// Drive a slot: every 10 ms tick, push whatever frames `arrive_at(tick)` says
863    /// land then (by frame index — frame N has capture ts N·20 ms), then pull a chunk.
864    /// `peak` sets the level (active vs silence). Returns total starves over `ticks`.
865    fn run_arrival_schedule(
866        floor_us: i64,
867        ticks: usize,
868        peak: i16,
869        mut arrive_at: impl FnMut(usize) -> Vec<usize>,
870    ) -> u64 {
871        let starves = Arc::new(AtomicU64::new(0));
872        let slot = Slot::new(starves.clone(), floor_us, "mic", 1);
873        let samples = vec![peak; FRAME_SAMPLES_20MS];
874        for tick in 0..ticks {
875            let now = tick as i64 * TICK_US;
876            for frame_idx in arrive_at(tick) {
877                slot.push(Some(now), now, frame_idx as i64 * FRAME_TS_US, &samples);
878            }
879            let mut out = [0i32; MIX_CHUNK_SAMPLES];
880            let _ = slot.pull_into(now, &mut out, &mut Vec::new());
881        }
882        starves.load(Ordering::Relaxed)
883    }

949    // ── End-to-end audio playout-chain continuity (the crackle guard) ────────────
950    //
951    // Deterministic, sans-IO: drive a peer's audio through the REAL pull-based chain
952    // — the PUMP pushes 20 ms tone frames into the slot's jitter buffer at their
953    // (possibly delayed) arrival, and the MIX thread pulls exactly one 10 ms chunk
954    // per STEADY tick via `pull_into` (the production cadence). Both sides share one
955    // timeline (the `CaptureClock` invariant). Audio crackle, as a number, is a
956    // silence/short chunk mid-stream — a `Slot` starvation. The decisive property the
957    // pull model gives us: the mix's release clock is independent of the pump's push
958    // timing, so pump jitter/descheduling (the live crackle cause) only delays a FILL,
959    // which the jitter buffer absorbs — it can NOT race the release. These pin that.
960
961    const CHAIN_FRAME_US: i64 = 20_000; // 20 ms Opus frame
962    const CHAIN_FRAME_SAMPLES: usize = 960; // 20 ms @ 48 kHz mono
963    const CHAIN_TARGET_US: i64 = 60_000; // jitter cushion (PLAYOUT_TARGET_US)
964    const CHAIN_TICK_US: i64 = (MIX_TICK_MS as i64) * 1_000; // 10 ms mix tick
965    const CHAIN_TONE: i16 = 5_000;
966
967    /// Drive ONE peer through the real pull chain for `ticks` mix ticks. `arrival(k)`
968    /// returns the shared-clock time at which the pump PUSHES frame `k` (its ideal
969    /// arrival is `k·20 ms`; returning later models pump descheduling/jitter). The mix
970    /// pulls exactly one chunk per steady 10 ms tick. Returns the per-tick mixed chunks
971    /// and the slot starvation count.
972    fn run_pull_chain(
973        ticks: i64,
974        target_us: i64,
975        arrival: impl Fn(i64) -> i64,
976    ) -> (Vec<[i16; MIX_CHUNK_SAMPLES]>, u64) {
977        let starves = Arc::new(AtomicU64::new(0));
978        let slot = Arc::new(Slot::new(starves.clone(), target_us, "mic", NO_PEER_TAG));
979        let mut scratch = MixScratch::new();
980        let slots = [slot.clone()];
981        let mut out = Vec::new();
982        let mut next_frame = 0i64;
983        for t in 0..ticks {
984            let now = t * CHAIN_TICK_US;
985            // Pump: push every frame whose (possibly delayed) arrival has come, at its
986            // arrival time on the shared clock — exactly what the live pump does.
987            while arrival(next_frame) <= now {
988                let ts = next_frame * CHAIN_FRAME_US;
989                let at = arrival(next_frame);
990                slot.push(Some(at), at, ts, &[CHAIN_TONE; CHAIN_FRAME_SAMPLES]);
991                next_frame += 1;
992            }
993            // Mix: pull exactly one chunk on the steady RT clock.
994            out.push(mix_chunk(&slots, now, &mut scratch, &mut Vec::new()));
995        }
996        (out, starves.load(Ordering::Relaxed))
997    }

1010    /// THE core guard: across EVERY frame-vs-tick phase alignment, an on-time stream
1011    /// pulled one-chunk-per-tick never starves and is continuous after warmup. A 20 ms
1012    /// frame straddles two 10 ms ticks, so its due-instant lands at every offset in the
1013    /// tick grid as the phase sweeps; the RT-side lookahead must cover all of them. (The
1014    /// old push-based slot with a 20 ms lead failed exactly here — a tick landing in the
1015    /// inter-frame trough played silence.)
1016    #[test]
1017    fn pull_chain_is_continuous_across_all_phases() {
1018        for phase_ms in 0..20i64 {
1019            let phase = phase_ms * 1_000;
1020            // Every frame (incl. the first, which sets the offset) arrives at ideal +
1021            // phase, so the frame-due phase relative to the tick grid sweeps 0..20 ms.
1022            let (mixed, starves) =
1023                run_pull_chain(200, CHAIN_TARGET_US, |k| k * CHAIN_FRAME_US + phase);
1024            assert_eq!(
1025                starves, 0,
1026                "phase {phase_ms} ms: a steady pulled chain must never starve"
1027            );
1028            // Warmup: first frame due at ~target; +phase; give generous slack.
1029            assert_continuous_after(&mixed, 10);
1030        }
1031    }

1033    /// The pull model's core guarantee, isolated: a non-real-time fill task
1034    /// descheduled under load (modelled as the pump running a CONSTANT delay behind
1035    /// real time — every frame pushed 45 ms late, ≪ the 60 ms buffer) must not gap
1036    /// playout. A *constant* delay is not jitter (the inter-arrival deviation is
1037    /// zero), so the buffer stays at its floor and every frame still beats its
1038    /// stable playout deadline. Under a push model a late pump feeds the slot late
1039    /// → starvation; under pull the RT thread pops the jitter buffer on its own
1040    /// clock, so a consistently-late pump is just a harmless late FILL — ZERO
1041    /// starvation, fully continuous.
1042    #[test]
1043    fn pull_chain_absorbs_constant_pump_delay() {
1044        let (mixed, starves) =
1045            run_pull_chain(200, CHAIN_TARGET_US, |k| k * CHAIN_FRAME_US + 45_000);
1046        assert_eq!(
1047            starves, 0,
1048            "a constantly-descheduled pump must not starve (the crackle root cause)"
1049        );
1050        assert_continuous_after(&mixed, 12);
1051    }

1053    /// The metric still catches REAL starvation: an arrival delayed beyond the buffer
1054    /// depth (network loss / total stall, not a scheduling race) genuinely empties the
1055    /// jitter buffer, so the slot comes up short — the gap is surfaced, not masked.
1056    #[test]
1057    fn pull_chain_starves_when_arrival_exceeds_buffer() {
1058        // Frame 20 arrives 200 ms late (≫ TARGET): its slot must run dry around then.
1059        let arrival = |k: i64| {
1060            let base = k * CHAIN_FRAME_US;
1061            if k >= 20 { base + 200_000 } else { base }
1062        };
1063        let (_mixed, starves) = run_pull_chain(120, CHAIN_TARGET_US, arrival);
1064        assert!(
1065            starves > 0,
1066            "an arrival past the buffer depth is real starvation the metric must report"
1067        );
1068    }
