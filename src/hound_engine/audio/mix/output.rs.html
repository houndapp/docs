15/// Opens the output sink for a [`DeviceChoice`] on the bus thread, returning a
16/// boxed [`AudioOutput`] or the typed failure. Production passes
17/// [`cpal_sink_factory`] ([`CpalPlayback`]); tests inject a fake so they never
18/// touch a real (possibly sandbox-blocked) audio device. Runs ONLY on the bus
19/// thread, so the `!Send` `cpal::Stream` never crosses a thread boundary.
20///
21/// The failure is returned rather than swallowed because the bus — not the
22/// factory — decides whether a failed open is a hiccup or an outage worth an
23/// event, and the escalation has to be able to name the cause ([`OutputHealth`]).
24pub(super) type SinkFactory =
25    Box<dyn FnMut(&DeviceChoice<SpeakerId>) -> Result<Box<dyn AudioOutput>, AudioError> + Send>;
26
27/// The production sink factory: open a [`CpalPlayback`] for the choice (with the
28/// vanish-fallback inside `open`), boxed as [`AudioOutput`].
29///
30/// A failure is logged at WARN, not error: the bus keeps ticking the AEC
31/// reference and retries on its backoff ladder, and most failures here are a
32/// device that is momentarily busy or mid-reroute. Only an outage that survives
33/// the ladder is escalated — see [`OutputHealth`].
34pub(super) fn cpal_sink_factory() -> SinkFactory {
35    Box::new(
36        |choice: &DeviceChoice<SpeakerId>| match CpalPlayback::open(choice) {
37            Ok(playback) => {
38                tracing::info!(
39                    opened = playback.opened_name().unwrap_or("<unknown>"),
40                    "voice: mix-bus playback opened"
41                );
42                Ok(Box::new(playback) as Box<dyn AudioOutput>)
43            }
44            Err(e) => {
45                tracing::warn!("voice: mix-bus playback open failed ({e}) — retrying");
46                Err(e)
47            }
48        },
49    )
50}

52/// The synthetic sink factory: open a [`SyntheticAudioOutput`] (speaker-free
53/// null playout) boxed as [`AudioOutput`], ignoring the device choice — there is
54/// no real device to select. For a headless/CI run (a bot) so the mix bus drives
55/// a full playout path (including the honest end-of-pipeline underrun gauge)
56/// without opening a real DAC. Errors only on the (practically infallible) open
57/// failure, matching [`cpal_sink_factory`]'s shape.
58#[cfg(feature = "synthetic")]
59pub(super) fn synthetic_sink_factory() -> SinkFactory {
60    use crate::platform::generic::synthetic_audio_output::SyntheticAudioOutput;
61    Box::new(
62        |_choice: &DeviceChoice<SpeakerId>| match SyntheticAudioOutput::open() {
63            Ok(sink) => {
64                tracing::info!("voice: mix-bus SYNTHETIC playback opened (no real device)");
65                Ok(Box::new(sink) as Box<dyn AudioOutput>)
66            }
67            Err(e) => {
68                tracing::warn!("voice: mix-bus synthetic playback open failed ({e}) — retrying");
69                Err(e)
70            }
71        },
72    )
73}

75/// Target depth of the output device ring the bus keeps filled (~3 chunks). The
76/// device drains the ring on its hardware clock while this thread refills it on a
77/// software clock; without a cushion the ring sits at ~0 and underruns on the
78/// slightest pacing jitter (or when encode load deschedules this thread). 30 ms
79/// absorbs normal jitter at a small, constant added output latency (AEC3 tolerates
80/// the fixed delay); the thread-priority promotion handles longer stalls.
81///
82/// The cushion is *regulated*, not merely seeded — see [`chunks_for_tick`].
83pub(super) const OUTPUT_CUSHION_MS: u32 = 30;
84
85/// Dead band around [`OUTPUT_CUSHION_MS`] within which the tick pushes exactly one
86/// chunk. It must exceed one device callback (a cpal default buffer is ~10–21 ms),
87/// because the depth sampled at tick time swings by that much purely from the
88/// tick-vs-callback phase — a narrower band would toggle the regulator on phase
89/// noise instead of on real drift.
90pub(super) const CUSHION_HYSTERESIS_MS: u32 = 15;
91
92/// Initial rate-limit on device-reopen attempts when no speaker is open (a down
93/// device must not hammer `start` every tick); first retry after ~500 ms.
94pub(super) const REOPEN_BACKOFF: Duration = Duration::from_millis(500);
95
96/// Ceiling for the reopen backoff. A *persistently* unavailable output device
97/// (e.g. an ALSA `hw:` held exclusively by another process on Linux — where each
98/// failed open/close churn can itself prolong the `EBUSY`) must not retry at
99/// 2 Hz forever, flooding the log and thrashing the device. The interval doubles
100/// from [`REOPEN_BACKOFF`] up to this cap, then holds — a slow heartbeat that
101/// still recovers automatically once the device frees, without the storm.
102pub(super) const REOPEN_BACKOFF_MAX: Duration = Duration::from_secs(5);
103
104/// How long the non-fatal output-stream events must stay QUIET before the bus re-syncs
105/// its sink.
106///
107/// cpal's `ErrorKind::DeviceChanged` is informational — the stream was already rerouted
108/// and is still playing — so nothing is urgent. But the sink's geometry (`device_rate`,
109/// its 48 k→device resampler, and hence the scale of `buffered_ms`) still describes the
110/// OLD device, so a 48 k→24 k hop silently halves the cushion regulator's units. One
111/// reopen fixes that. `Xrun` / `RealtimeDenied` arrive on the same (still-live) stream
112/// and are demoted into this same window, where the eventual reopen's fresh cushion
113/// prefill is the only remedy on offer — and, since a persistent fault storm keeps
114/// pushing the deadline out, one that a sustained problem never triggers.
115///
116/// The debounce exists because a single Bluetooth connect/disconnect makes macOS move
117/// the default output repeatedly (5 changes in 50 s observed on AirPods). Each change
118/// PUSHES this deadline, so a whole flap burst costs ONE reopen instead of one per
119/// event — and each reopen is a blocking device open (up to ~1 s inside CoreAudio) on
120/// this real-time thread plus a fresh [`OUTPUT_CUSHION_MS`] prefill. 2 s is comfortably
121/// longer than the gaps within an observed burst and short enough that the stale
122/// geometry never outlives a call's cushion tuning.
123pub(super) const ROUTE_SETTLE: Duration = Duration::from_secs(2);
124
125/// Pre-fill a freshly-opened output sink's ring with [`OUTPUT_CUSHION_MS`] of
126/// SILENCE — the cushion's initial condition, from which [`chunks_for_tick`] holds it
127/// closed-loop. Silence (not slot audio) means the pre-fill costs nothing from any
128/// peer's stream; it is a fixed ~cushion of added output latency, which AEC absorbs.
129pub(super) fn prefill_output(playback: &mut Option<Box<dyn AudioOutput>>) {
130    if let Some(p) = playback.as_mut() {
131        let silent = chunk_to_frame(vec![0i16; MIX_CHUNK_SAMPLES]);
132        for _ in 0..(OUTPUT_CUSHION_MS / MIX_TICK_MS) {
133            let _ = p.play(&silent);
134        }
135    }
136}

138/// How many mixed chunks this tick should push, from the sink's reported ring depth
139/// (`depth_ms`, sampled BEFORE the push) and how late the tick fired (`late_us`).
140///
141/// This is the closed loop around [`OUTPUT_CUSHION_MS`]. The bus produces on a
142/// software 10 ms clock while the DAC drains on its own crystal, so one chunk per
143/// tick is only right if the two rates match exactly. They never do: tens of ppm is
144/// normal (far worse on an OS-resampled BT/HFP route), and open-loop the ring walks
145/// to one of the two ends within minutes — dry (continuous fine-grained dropouts,
146/// which `diagnose()` then misattributes to the device/route) or full (output latency
147/// pinned at the ring depth, with the overproduction silently discarded).
148///
149/// * depth ABOVE the dead band ⇒ push **nothing**. The chunk is not mixed at all, so
150///   no audio is lost: the slots simply keep it and the whole playout timeline slips
151///   10 ms later, which is exactly the correction a slow DAC needs.
152/// * depth BELOW it ⇒ push the deficit as extra chunks (bounded by
153///   [`MAX_CHUNKS_PER_TICK`]). The extra chunks come from the slots' carry, which is
154///   where a stall's un-drained backlog sits — so catching the ring up also drains
155///   the backlog instead of leaving it as permanent added latency.
156/// * a LATE tick means the device drained for `late_us` with no refill; replay those
157///   chunks even when the depth reading has not caught up yet, so a scheduling stall
158///   costs a one-time gap rather than the cushion for the rest of the call.
159///
160/// `depth_ms == None` (a sink that cannot report depth — the synthetic/null sinks)
161/// keeps the previous strictly-open-loop behaviour: exactly one chunk per tick.
162pub(super) fn chunks_for_tick(depth_ms: Option<u32>, late_us: u128) -> usize {
163    let Some(depth) = depth_ms else {
164        return 1;
165    };
166    if depth > OUTPUT_CUSHION_MS + CUSHION_HYSTERESIS_MS {
167        return 0;
168    }
169    // Below the band: refill the whole shortfall to target (floored to whole chunks,
170    // so the correction converges on the cushion instead of overshooting past it).
171    let deficit_chunks = if depth + CUSHION_HYSTERESIS_MS < OUTPUT_CUSHION_MS {
172        ((OUTPUT_CUSHION_MS - depth) / MIX_TICK_MS) as usize
173    } else {
174        0
175    };
176    let late_chunks = (late_us / (MIX_TICK_MS as u128 * 1_000)) as usize;
177    (1 + deficit_chunks.max(late_chunks)).min(MAX_CHUNKS_PER_TICK)
178}

180/// Arm the one-shot [`crate::audio::ROUTE_WARMUP_DELAY`] timer iff the just-opened sink runs at a
181/// non-engine rate (a resampled device — in practice a Bluetooth HFP headset whose
182/// SCO link may not be carrying audio yet). 48 kHz devices return `None` (no warmup
183/// reopen), so the common path is untouched. See [`crate::audio::ROUTE_WARMUP_DELAY`].
184pub(super) fn warmup_reopen_at(
185    playback: &Option<Box<dyn AudioOutput>>,
186    now: Instant,
187) -> Option<Instant> {
188    match playback.as_ref() {
189        Some(p) if p.device_rate() != SAMPLE_RATE.hz() => {
190            Some(now + crate::audio::ROUTE_WARMUP_DELAY)
191        }
192        _ => None,
193    }
194}

196/// Debounce for the non-fatal output-stream events: automatic route changes (cpal
197/// `ErrorKind::DeviceChanged`) and the demoted quality faults (`Xrun`,
198/// `RealtimeDenied`). All of them reach the bus as one edge —
199/// [`AudioOutput::take_route_changed`].
200///
201/// A pure `Instant` state machine — the RT mix thread only compares timestamps in its
202/// existing tick, it never queries a device to decide. [`Self::note_change`] (re-)arms a
203/// [`ROUTE_SETTLE`] deadline, so a burst collapses into ONE firing after the events go
204/// quiet; [`Self::fire`] returns true exactly once at the deadline; [`Self::disarm`]
205/// drops a pending re-sync when a REAL death already forced a reopen.
206#[derive(Default)]
207pub(super) struct RouteSettle {
208    pub(super) due: Option<Instant>,
209}
210
211impl RouteSettle {
212    /// Observe one route change: (re-)arm the deadline `ROUTE_SETTLE` from `now`. Each
213    /// further change pushes it out, which is what coalesces a flap burst.
214    pub(super) fn note_change(&mut self, now: Instant) {
215        self.due = Some(now + ROUTE_SETTLE);
216    }
217
218    /// True exactly once, when the route has been quiet for [`ROUTE_SETTLE`].
219    pub(super) fn fire(&mut self, now: Instant) -> bool {
220        match self.due {
221            Some(t) if now >= t => {
222                self.due = None;
223                true
224            }
225            _ => false,
226        }
227    }
228
229    /// Cancel a pending re-sync — the sink was rebuilt for another reason (a real death,
230    /// a user hot-swap, the warmup reopen), so its geometry is already fresh.
231    pub(super) fn disarm(&mut self) {
232        self.due = None;
233    }
234}

236/// Output-device health: WHEN a device failure stops being a hiccup and becomes an
237/// operator event.
238///
239/// The stream error callback cannot make that call — it fires before recovery has
240/// been attempted, and the reopen a tick later fixes nearly every one of them (a
241/// USB headset unplugged, a suspend/resume, an ALSA `EIO` from a PipeWire node that
242/// went away). Reporting there filed a Sentry issue per recovered hiccup while a
243/// device whose reopens quietly failed forever produced nothing. So the callback
244/// only ever warns, and the escalation lives here: the shared
245/// [`DeviceOutage`] latch, plus the last cause so the one report that DOES reach
246/// Sentry can name it.
247#[derive(Default)]
248pub(super) struct OutputHealth {
249    outage: DeviceOutage,
250    /// The most recent stream death or open failure, carried into the escalation.
251    cause: Option<String>,
252}
253
254impl OutputHealth {
255    /// The sink is open. Closes any outage; announces the recovery only if the
256    /// outage had already been escalated, so the filed event gets a visible end and
257    /// ordinary hiccups stay quiet.
258    pub(super) fn opened(&mut self, now: Instant) {
259        if let Some(o) = self.outage.recovered(now) {
260            tracing::info!(
261                target: "audio_endpoint",
262                down_ms = o.elapsed.as_millis() as u64,
263                failed_reopens = o.failed_attempts,
264                "voice: output device recovered — playout restored after {:?}",
265                o.elapsed
266            );
267        }
268        self.cause = None;
269    }
270
271    /// The live stream died (its error callback marked it dead). `cause` is the
272    /// backend's message for that error, if it captured one.
273    pub(super) fn died(&mut self, now: Instant, cause: Option<String>) {
274        if let Some(c) = cause {
275            self.cause = Some(c);
276        }
277        self.outage.down(now);
278    }
279
280    /// A reopen attempt failed — the outage continues and the ladder backs off.
281    pub(super) fn open_failed(&mut self, now: Instant, e: &AudioError) {
282        self.cause = Some(e.to_string());
283        self.outage.attempt_failed(now);
284    }
285
286    /// Called every tick while there is NO open sink. Keeps the outage clock
287    /// running and emits the single `error!` once recovery has been failing for
288    /// [`OUTAGE_ESCALATE_AFTER`](crate::audio::outage::OUTAGE_ESCALATE_AFTER).
289    pub(super) fn poll_down(&mut self, now: Instant) {
290        self.outage.down(now);
291        if let Some(o) = self.outage.escalate(now) {
292            tracing::error!(
293                target: "audio_endpoint",
294                down_ms = o.elapsed.as_millis() as u64,
295                failed_reopens = o.failed_attempts,
296                cause = self.cause.as_deref().unwrap_or("<no error captured>"),
297                "voice: output device is not recovering — no playout for {:?} across {} \
298                 failed reopens; the user hears nobody in the call",
299                o.elapsed,
300                o.failed_attempts
301            );
302        }
303    }
304}
305
306/// One (re)open attempt on the bus thread: run the factory and fold the outcome
307/// into [`OutputHealth`]. `None` leaves the endpoint down for the tick loop's
308/// backoff ladder to retry.
309pub(super) fn open_sink(
310    factory: &mut SinkFactory,
311    choice: &DeviceChoice<SpeakerId>,
312    health: &mut OutputHealth,
313    now: Instant,
314) -> Option<Box<dyn AudioOutput>> {
315    match factory(choice) {
316        Ok(sink) => {
317            health.opened(now);
318            Some(sink)
319        }
320        Err(e) => {
321            health.open_failed(now, &e);
322            None
323        }
324    }
325}

339    /// The output cushion is established by a one-time SILENCE pre-fill at open —
340    /// one full mix chunk per cushion tick — so the device has a buffer immediately
341    /// and the steady one-chunk-per-tick mix never has to over-drain the per-peer
342    /// slots to refill it (an over-drain is audible as per-peer crackle).
343    #[test]
344    fn prefill_seeds_the_output_cushion_with_silence() {
345        let sink = RecordingSink::default();
346        let mut playback: Option<Box<dyn AudioOutput>> = Some(Box::new(sink.clone()));
347        prefill_output(&mut playback);
348
349        let frames = sink.frames.lock().unwrap();
350        let want = (OUTPUT_CUSHION_MS / MIX_TICK_MS) as usize;
351        assert_eq!(frames.len(), want, "one silent chunk per cushion tick");
352        assert!(
353            frames
354                .iter()
355                .all(|f| f.len() == MIX_CHUNK_SAMPLES && f.iter().all(|&s| s == 0)),
356            "the cushion pre-fill is full chunks of silence"
357        );
358    }

362    /// In the dead band the loop is a no-op: exactly one chunk per tick, and a sink
363    /// that cannot report depth stays strictly open-loop at one chunk (the synthetic
364    /// / null sinks must behave exactly as before).
365    #[test]
366    fn cushion_holds_one_chunk_inside_the_dead_band() {
367        assert_eq!(chunks_for_tick(Some(OUTPUT_CUSHION_MS), 0), 1);
368        assert_eq!(
369            chunks_for_tick(Some(OUTPUT_CUSHION_MS + CUSHION_HYSTERESIS_MS), 0),
370            1,
371            "the band edge is still one chunk"

373        assert_eq!(
374            chunks_for_tick(Some(OUTPUT_CUSHION_MS - CUSHION_HYSTERESIS_MS), 0),
375            1

377        assert_eq!(chunks_for_tick(None, 0), 1, "no depth signal ⇒ open-loop");

385    /// Ring too DEEP (a DAC slower than the tick, or a catch-up overshoot): push
386    /// nothing. Nothing is mixed either, so no audio is lost — the playout timeline
387    /// simply slips a chunk, which is the correction a slow DAC needs.
388    #[test]
389    fn cushion_skips_when_the_ring_is_too_deep() {
390        assert_eq!(
391            chunks_for_tick(Some(OUTPUT_CUSHION_MS + CUSHION_HYSTERESIS_MS + 1), 0),

393        );

395        assert_eq!(chunks_for_tick(Some(200), 500_000), 0);

398    /// Ring too SHALLOW (a DAC faster than the tick): push the deficit as extra
399    /// chunks, bounded — a dry ring is refilled to the cushion in one tick.
400    #[test]
401    fn cushion_catches_up_when_the_ring_runs_shallow() {
402        // 10 ms deep, 30 ms target ⇒ 2 chunks of deficit + the tick's own = 3.
403        assert_eq!(chunks_for_tick(Some(10), 0), 3);

408    /// A LATE tick replays what the device drained while this thread was descheduled,
409    /// even when the depth reading has not caught up yet — so a stall costs a one-time
410    /// gap instead of permanently spending the cushion (the open-loop failure).

412    fn cushion_replays_after_a_late_tick() {

416            chunks_for_tick(Some(OUTPUT_CUSHION_MS), 500_000),

422    /// Drive the regulator against a modelled DAC draining `drain_ms` per 10 ms tick
423    /// (the clock-drift the open-loop bus could not correct), from `start_ms` of
424    /// initial depth. Returns the (min, max) depth seen over `ticks`.
425    fn simulate_cushion(start_ms: i64, drain_ms: i64, ticks: usize) -> (i64, i64) {
426        let mut depth = start_ms;
427        let (mut lo, mut hi) = (depth, depth);
428        for _ in 0..ticks {
429            let n = chunks_for_tick(Some(depth.max(0) as u32), 0) as i64;
430            depth = (depth + n * MIX_TICK_MS as i64 - drain_ms).max(0);
431            lo = lo.min(depth);
432            hi = hi.max(depth);
433        }
434        (lo, hi)
435    }

437    /// The loop actually REGULATES: against a DAC 20 % fast or 20 % slow (drift far
438    /// beyond any real crystal) the ring neither bottoms out nor climbs toward the
439    /// 200 ms cap — the two open-loop end states the finding describes.
440    #[test]
441    fn cushion_regulates_against_clock_drift() {
442        for drain_ms in [8, 10, 12] {
443            let (lo, hi) = simulate_cushion(OUTPUT_CUSHION_MS as i64, drain_ms, 2_000);
444            assert!(
445                lo > 0,
446                "drain {drain_ms}: the ring must never run dry (lo {lo})"
447            );
448            assert!(
449                hi <= (OUTPUT_CUSHION_MS + CUSHION_HYSTERESIS_MS + MIX_TICK_MS) as i64,
450                "drain {drain_ms}: the ring must not climb toward the cap (hi {hi})"
451            );
452        }
453    }

525    /// A burst of route changes costs exactly ONE re-sync, and only after the route has
526    /// been quiet for the full [`ROUTE_SETTLE`]. This is the whole point of the debounce:
527    /// macOS moved the default output 5× in 50 s on a single AirPods connect, and each
528    /// event used to force a blocking device reopen on the RT mix thread.
529    #[test]
530    fn route_changes_coalesce_into_one_resync() {
531        let t0 = Instant::now();
532        let mut settle = RouteSettle::default();
533        assert!(!settle.fire(t0), "idle: nothing to re-sync");
534
535        // Five changes spread over 2 s, each closer together than ROUTE_SETTLE.
536        let mut fires = 0;
537        for i in 0..5 {
538            let at = t0 + Duration::from_millis(500 * i);
539            settle.note_change(at);
540            // Nothing fires DURING the burst — each change pushes the deadline out.
541            if settle.fire(at + Duration::from_millis(400)) {
542                fires += 1;
543            }
544        }
545        assert_eq!(fires, 0, "the burst must not re-sync per event");
546
547        let last_change = t0 + Duration::from_millis(2000);
548        assert!(
549            !settle.fire(last_change + ROUTE_SETTLE - Duration::from_millis(1)),
550            "still inside the settle window"
551        );
552        assert!(
553            settle.fire(last_change + ROUTE_SETTLE),
554            "one re-sync once the route goes quiet"
555        );
556        assert!(
557            !settle.fire(last_change + ROUTE_SETTLE + Duration::from_secs(60)),
558            "and exactly once — it does not re-fire"
559        );
560    }

562    /// A real death (or a hot-swap / warmup reopen) already rebuilds the sink, so a
563    /// pending re-sync must be dropped rather than costing a second reopen right after.
564    #[test]
565    fn disarm_cancels_a_pending_resync() {
566        let t0 = Instant::now();
567        let mut settle = RouteSettle::default();
568        settle.note_change(t0);
569        settle.disarm();
570        assert!(
571            !settle.fire(t0 + ROUTE_SETTLE * 10),
572            "a disarmed debounce never fires"
573        );
574
575        // …and it can be re-armed afterwards by a later route change.
576        settle.note_change(t0);
577        assert!(settle.fire(t0 + ROUTE_SETTLE));
578    }

580    /// A CONTINUOUS stream of non-fatal events (the case a demoted `Xrun` storm
581    /// produces) never fires the re-sync at all: every arrival pushes the deadline out.
582    /// That is the desired outcome, not a hole — answering sustained xruns with a
583    /// blocking device open per event on the RT mix thread is what makes them worse.
584    /// The `output_soft_faults` counter is what keeps such a storm visible.
585    #[test]
586    fn a_soft_fault_storm_never_fires_the_resync() {
587        let t0 = Instant::now();
588        let mut settle = RouteSettle::default();
589
590        // One fault every 100 ms for a minute — 20× more often than ROUTE_SETTLE.
591        let mut fires = 0;
592        for i in 0..600u64 {
593            let at = t0 + Duration::from_millis(100 * i);
594            settle.note_change(at);
595            if settle.fire(at) {
596                fires += 1;
597            }
598        }
599        assert_eq!(fires, 0, "a sustained storm must never trigger a reopen");
600
601        // …and when the device finally settles, exactly one re-sync lands.
602        let last = t0 + Duration::from_millis(100 * 599);
603        assert!(
604            settle.fire(last + ROUTE_SETTLE),
605            "one re-sync once the faults stop"
606        );
607    }

609    /// THE demotion guard, end to end through the live loop: a sink reporting `Xrun` /
610    /// `RealtimeDenied` keeps playing. The bus counts every fault and does NOT tear the
611    /// device down — the reopen these used to force was a blocking device open plus a
612    /// fresh cushion prefill on the RT mix thread, i.e. a real glitch in answer to a
613    /// reported one. (The debounced re-sync is 2 s out, well past this window.)
614    #[test]
615    fn a_demoted_soft_fault_is_counted_without_reopening_the_device() {
616        let (_tx, rx) = watch::channel(DeviceChoice::<SpeakerId>::Default);
617        let sink = SoftFaultSink::with_faults(3);
618        let opens = Arc::new(AtomicU64::new(0));
619        let (inner, opens_seen) = (sink.clone(), opens.clone());
620        let bus = PlayoutBus::start_with_sink(
621            rx,
622            None,
623            CaptureClock::start(),
624            0,
625            Box::new(move |_choice: &DeviceChoice<SpeakerId>| {
626                opens_seen.fetch_add(1, Ordering::Relaxed);
627                Ok(Box::new(inner.clone()) as Box<dyn AudioOutput>)
628            }),
629        );
630        std::thread::sleep(Duration::from_millis(120));
631        let ticks = bus.tick_count();
632        bus.stop();
633
634        assert!(
635            ticks >= 5,
636            "the bus must keep ticking through a demoted fault (got {ticks})"
637        );
638        assert_eq!(
639            bus.output_soft_faults(),
640            3,
641            "every demoted fault reaches the call-level counter"
642        );
643        assert_eq!(
644            opens.load(Ordering::Relaxed),
645            1,
646            "an xrun / RT denial must not cost a device reopen — it rides the debounce"
647        );
648    }

650    /// A device that dies and then comes back must produce NO error-level report:
651    /// the outage is closed long before
652    /// [`OUTAGE_ESCALATE_AFTER`](crate::audio::outage::OUTAGE_ESCALATE_AFTER), so
653    /// `recovered` stays silent. This is the regression the Linux `EIO` issue was —
654    /// the error callback filed an event for a fault the reopen had already fixed.
655    #[test]
656    fn a_dead_then_reopened_sink_never_escalates() {
657        let t0 = Instant::now();
658        let mut health = OutputHealth::default();
659        health.died(t0, Some("ALSA snd_pcm_avail_delay: I/O error (5)".into()));
660        assert!(health.outage.is_down());
661        // The reopen lands on the next tick, far inside the escalation window.
662        health.opened(t0 + Duration::from_millis(10));
663        assert!(!health.outage.is_down());
664        assert!(
665            health.cause.is_none(),
666            "a recovered outage clears its cause"
667        );
668        // And a later tick with a sink open cannot escalate anything.
669        assert_eq!(health.outage.escalate(t0 + Duration::from_secs(60)), None);
670    }
671
672    /// The other direction: reopens that keep failing DO escalate, exactly once,
673    /// and the escalation carries the cause the (warn-level) stream error captured.
674    #[test]
675    fn a_sink_that_never_reopens_escalates_once() {
676        let t0 = Instant::now();
677        let mut health = OutputHealth::default();
678        health.died(t0, Some("device disconnected".into()));
679        for i in 1..=5 {
680            health.open_failed(
681                t0 + Duration::from_secs(i),
682                &AudioError::NoDevice("no output device".into()),
683            );
684        }
685        assert_eq!(
686            health.outage.escalate(t0 + Duration::from_secs(1)),
687            None,
688            "no event while the ladder is still running"
689        );
690        let fired = health
691            .outage
692            .escalate(t0 + crate::audio::outage::OUTAGE_ESCALATE_AFTER)
693            .expect("an unrecovered outage escalates");
694        assert_eq!(fired.failed_attempts, 5);
695        assert!(
696            health
697                .cause
698                .as_deref()
699                .is_some_and(|c| c.contains("no output device")),
700            "the escalation names the latest cause, got {:?}",
701            health.cause
702        );
703    }
