1//! # Device-outage escalation — report the failure that did NOT recover
2//!
3//! Every audio device loop in the stack recovers from a dead stream by
4//! reopening: the mix bus's output sink ([`super::mix`]) and the client's mic
5//! capture loop both drop the endpoint and re-open it on a backoff. The device
6//! errors that trigger that are overwhelmingly *transient and self-healing* — a
7//! USB headset unplugged, a suspend/resume, a PipeWire node restarted under the
8//! stream, an ALSA `EIO` — and the reopen has fixed them within a tick.
9//!
10//! So the thing worth an operator EVENT is not the stream error. It is an outage
11//! that **outlives the recovery**: no device for N seconds across M failed
12//! reopens. Reporting at the error callback instead files an issue for every
13//! successfully-recovered hiccup while a genuinely-down device (whose reopens
14//! quietly fail forever) produces nothing at all.
15//!
16//! This module is that latch. It owns no clock — the caller passes `Instant`s —
17//! so the policy is unit-tested rather than inferred from a live device, and the
18//! real-time mix thread can drive it from the `Instant` it already sampled.
19
20use std::time::{Duration, Instant};
21
22/// How long an audio endpoint may stay down before the outage is escalated from
23/// a `warn!` (recovery in progress) to an `error!` (recovery is not working —
24/// the Sentry bridge promotes first-party `error!` to an event).
25///
26/// Sized well above every recovery path that actually works: the mix bus's
27/// reopen ladder tops out at a 5 s heartbeat, the route-change debounce settles
28/// in 2 s, and the BT/HFP warmup reopen fires at 1.2 s. 8 s therefore means
29/// "the ladder has run and is still failing", not "a reopen is mid-flight".
30pub const OUTAGE_ESCALATE_AFTER: Duration = Duration::from_secs(8);
31
32/// What an escalation (or a recovery) reports: how long the endpoint was down
33/// and how many reopen attempts failed inside that window.
34#[derive(Debug, Clone, Copy, PartialEq, Eq)]
35pub struct Outage {
36    pub elapsed: Duration,
37    pub failed_attempts: u32,
38}
39
40/// Tracks one audio endpoint's down/up state and decides when the outage stops
41/// being a recoverable hiccup. Pure: `Default` is "up", every transition takes
42/// the caller's `now`.
43#[derive(Default, Debug)]
44pub struct DeviceOutage {
45    /// When the endpoint went down; `None` while it is up.
46    since: Option<Instant>,
47    failed_attempts: u32,
48    /// Latch so one outage costs at most one event, however many ticks observe it.
49    escalated: bool,
50}
51
52impl DeviceOutage {
53    /// Mark the endpoint down as of `now` (the stream died, or there is simply no
54    /// device open). Idempotent: only the FIRST call starts the clock, so a tick
55    /// loop can call it unconditionally while the endpoint is down.
56    pub fn down(&mut self, now: Instant) {
57        self.since.get_or_insert(now);
58    }
59
60    /// A reopen attempt failed: keep the outage running and count the attempt, so
61    /// the escalation can say how hard recovery tried.
62    pub fn attempt_failed(&mut self, now: Instant) {
63        self.down(now);
64        self.failed_attempts = self.failed_attempts.saturating_add(1);
65    }
66
67    /// Whether the endpoint is currently considered down.
68    pub fn is_down(&self) -> bool {
69        self.since.is_some()
70    }
71
72    /// Returns `Some` exactly once per outage, at the point recovery has been
73    /// failing for [`OUTAGE_ESCALATE_AFTER`] — the caller's cue to log an
74    /// `error!`. Every later tick of the same outage returns `None`.
75    pub fn escalate(&mut self, now: Instant) -> Option<Outage> {
76        let since = self.since?;
77        let elapsed = now.saturating_duration_since(since);
78        if self.escalated || elapsed < OUTAGE_ESCALATE_AFTER {
79            return None;
80        }
81        self.escalated = true;
82        Some(Outage {
83            elapsed,
84            failed_attempts: self.failed_attempts,
85        })
86    }
87
88    /// The endpoint is back: clear the outage. Returns `Some` only if the outage
89    /// had already been escalated, so the caller can close the loop on an event
90    /// it filed ("recovered after …") and stay silent about the ordinary hiccups.
91    pub fn recovered(&mut self, now: Instant) -> Option<Outage> {
92        let since = self.since.take()?;
93        let escalated = std::mem::replace(&mut self.escalated, false);
94        let failed_attempts = std::mem::take(&mut self.failed_attempts);
95        escalated.then(|| Outage {
96            elapsed: now.saturating_duration_since(since),
97            failed_attempts,
98        })
99    }
100}

102#[cfg(test)]
103mod tests {
104    use super::*;
105
106    /// The common case: a stream error whose reopen works. The outage opens and
107    /// closes well inside the escalation window, so NOTHING is reported — this is
108    /// the whole point (an EIO that self-heals must not file a Sentry issue).
109    #[test]
110    fn a_recovered_hiccup_reports_nothing() {
111        let t0 = Instant::now();
112        let mut outage = DeviceOutage::default();
113        outage.down(t0);
114        assert!(outage.is_down());
115        assert_eq!(outage.escalate(t0 + Duration::from_millis(10)), None);
116        assert_eq!(
117            outage.recovered(t0 + Duration::from_millis(20)),
118            None,
119            "a hiccup that never escalated closes silently"
120        );
121        assert!(!outage.is_down());
122    }
123
124    /// An outage that outlives the reopen ladder escalates ONCE, carrying how long
125    /// and how many attempts — and the recovery behind it is reported too, so the
126    /// filed event has a visible end.
127    #[test]
128    fn a_sustained_outage_escalates_once_then_reports_recovery() {
129        let t0 = Instant::now();
130        let mut outage = DeviceOutage::default();
131        // The clock starts at the DEATH; the reopen attempts that follow only add
132        // to the count (`attempt_failed` never re-anchors `since`).
133        outage.down(t0);
134        for i in 1..=3 {
135            outage.attempt_failed(t0 + Duration::from_secs(i));
136        }
137        assert_eq!(
138            outage.escalate(t0 + OUTAGE_ESCALATE_AFTER - Duration::from_millis(1)),
139            None,
140            "still inside the recovery window"
141        );
142
143        let fired = outage
144            .escalate(t0 + OUTAGE_ESCALATE_AFTER)
145            .expect("escalates at the threshold");
146        assert_eq!(fired.failed_attempts, 3);
147        assert_eq!(fired.elapsed, OUTAGE_ESCALATE_AFTER);
148        assert_eq!(
149            outage.escalate(t0 + OUTAGE_ESCALATE_AFTER * 10),
150            None,
151            "one event per outage, however long it runs"
152        );
153
154        let back = outage
155            .recovered(t0 + Duration::from_secs(20))
156            .expect("an escalated outage reports its recovery");
157        assert_eq!(back.elapsed, Duration::from_secs(20));
158        assert_eq!(back.failed_attempts, 3);
159    }
160
161    /// Recovery re-arms: a second outage on the same tracker can escalate again
162    /// (a device that flaps down/up/down must not go silent after the first).
163    #[test]
164    fn recovery_re_arms_the_latch() {
165        let t0 = Instant::now();
166        let mut outage = DeviceOutage::default();
167        outage.down(t0);
168        assert!(outage.escalate(t0 + OUTAGE_ESCALATE_AFTER).is_some());
169        outage.recovered(t0 + OUTAGE_ESCALATE_AFTER);
170
171        let t1 = t0 + Duration::from_secs(60);
172        outage.attempt_failed(t1);
173        assert_eq!(
174            outage.escalate(t1).map(|o| o.failed_attempts),
175            None,
176            "the second outage starts its own clock"
177        );
178        assert_eq!(
179            outage
180                .escalate(t1 + OUTAGE_ESCALATE_AFTER)
181                .map(|o| o.failed_attempts),
182            Some(1),
183            "and its own attempt count"
184        );
185    }
186
187    /// An `escalate`/`recovered` on an endpoint that was never marked down is a
188    /// no-op — the tick loop calls these unconditionally.
189    #[test]
190    fn an_up_endpoint_is_inert() {
191        let now = Instant::now();
192        let mut outage = DeviceOutage::default();
193        assert_eq!(outage.escalate(now), None);
194        assert_eq!(outage.recovered(now), None);
195        assert!(!outage.is_down());
196    }
197}
