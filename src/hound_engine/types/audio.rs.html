1//! The engine's fixed audio format (48 kHz mono, 20 ms) and the PCM/Opus units
2//! the audio pipelines exchange.
3
4use std::time::Instant;
5
6use super::units::{ChannelCount, SampleRate, TimestampUs};
7
8/// Sample rate for the whole audio engine. 48 kHz is Opus's native internal
9/// rate (no internal resample) and the CoreAudio default on Macs — picking it
10/// end to end avoids a resample on either the capture or the playback side.
11pub const SAMPLE_RATE: SampleRate = SampleRate::HZ_48K;
12
13/// Channel count. Mono for voice — half the bitrate of stereo and what AEC
14/// expects on the capture side. If a device only offers stereo input we
15/// downmix to mono before encode.
16pub const CHANNELS: ChannelCount = ChannelCount::MONO;
17
18/// Opus frame duration. 20 ms is the voice sweet spot: 960 samples @ 48 kHz,
19/// low algorithmic delay, the default the WebRTC/Opus VoIP stacks use.
20pub const FRAME_MS: u32 = 20;
21
22/// Samples per channel in one [`PcmFrame`] / one Opus frame: 48000 × 20 / 1000
23/// = **960**. Opus only accepts exact frame sizes; the capture side must hand
24/// it exactly this.
25pub const SAMPLES_PER_FRAME: usize = (SAMPLE_RATE.hz() as usize * FRAME_MS as usize) / 1000;
26
27/// One block of captured/decoded PCM, stamped on the shared capture clock.
28///
29/// Interleaved `i16` samples (here always mono, so interleaving is a no-op).
30/// `samples.len()` is exactly [`SAMPLES_PER_FRAME`] × [`CHANNELS`] for a frame
31/// destined for the encoder.
32#[derive(Debug, Clone)]
33pub struct PcmFrame {
34    /// Interleaved i16 PCM. Length is `samples_per_channel × channels`.
35    pub samples: Vec<i16>,
36    pub sample_rate: SampleRate,
37    pub channels: ChannelCount,
38    /// Capture timestamp on the session's `CaptureClock` — the SAME timeline
39    /// as video ([`crate::types::RawFrame::timestamp_us`]) for A/V sync.
40    pub timestamp_us: TimestampUs,
41    /// When this frame was assembled, for capture→playout latency measurement.
42    /// Local instrumentation only; not on the wire.
43    pub captured_at: Instant,
44}
45
46impl PcmFrame {
47    /// A frame at the engine's fixed format (48 kHz mono) — the shape every
48    /// internal producer (capture, decode, mix) hands around.
49    pub fn at_engine_rate(samples: Vec<i16>, timestamp_us: TimestampUs) -> Self {
50        Self {
51            samples,
52            sample_rate: SAMPLE_RATE,
53            channels: CHANNELS,
54            timestamp_us,
55            captured_at: Instant::now(),
56        }
57    }
58
59    /// Root-mean-square amplitude over the block, normalised to `0.0..=1.0`
60    /// (i16 full-scale = 1.0). Distinguishes "silent room" from "no mic /
61    /// permission denied" (the latter yields *no frames at all*, not a quiet
62    /// RMS).
63    pub fn rms(&self) -> f32 {
64        if self.samples.is_empty() {
65            return 0.0;
66        }
67        let sum_sq: f64 = self
68            .samples
69            .iter()
70            .map(|&s| {
71                let f = s as f64;
72                f * f
73            })
74            .sum();
75        let mean_sq = sum_sq / self.samples.len() as f64;
76        (mean_sq.sqrt() / i16::MAX as f64) as f32
77    }
78}
79
80/// One encoded Opus packet (one [`FRAME_MS`] frame of audio). This is the unit
81/// the transport carries on the wire.
82#[derive(Debug, Clone)]
83pub struct OpusPacket {
84    pub data: Vec<u8>,
85    /// Source capture timestamp (A/V sync), carried so the wire header is
86    /// stamped straight from it, mirroring [`crate::types::EncodedChunk`].
87    pub timestamp_us: TimestampUs,
88}

90#[cfg(test)]
91mod tests {
92    use super::*;
93
94    #[test]
95    fn samples_per_frame_is_960() {
96        assert_eq!(SAMPLES_PER_FRAME, 960);
97        assert_eq!(SAMPLE_RATE.hz(), 48_000);
98        assert_eq!(CHANNELS.count(), 1);
99    }
100
101    #[test]
102    fn pcm_frame_rms_full_scale_and_silence() {
103        let silent = PcmFrame::at_engine_rate(vec![0; 960], TimestampUs::ZERO);
104        assert_eq!(silent.rms(), 0.0);
105        let full = PcmFrame::at_engine_rate(vec![i16::MAX; 960], TimestampUs::ZERO);
106        assert!((full.rms() - 1.0).abs() < 1e-4);
107        assert_eq!(full.sample_rate, SAMPLE_RATE);
108        assert_eq!(full.channels, CHANNELS);
109    }
