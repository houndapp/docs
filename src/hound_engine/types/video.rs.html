8/// Pixel layout of a [`RawFrame`].
9///
10/// The pipeline normalises every captured frame to [`PixelFormat::I420`]
11/// (planar YUV 4:2:0, 8-bit) because that is what the encoders consume and the
12/// render seam displays. The variants for the formats cameras actually *emit*
13/// (NV12, RGB) are kept so a future zero-copy path can hand a HW encoder the
14/// native buffer without a planar conversion.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum PixelFormat {
17    /// Planar Y, then U, then V; chroma at half width/height. Encoder input.
18    I420,
19    /// Planar Y followed by interleaved UV; common AVFoundation/MediaCodec
20    /// output. Reserved for the zero-copy HW path.
21    Nv12,
22    /// Packed 8-bit R,G,B. Reserved for capture paths that decode to RGB.
23    Rgb,
24}

26/// Which codec a video encoder produces / a video decoder consumes.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub enum VideoCodec {
29    /// AV1 — HW encode on M5 Pro/Max + modern discrete GPUs (VideoToolbox);
30    /// HW decode broadly (A17 Pro / M3+). No SW path — a device without any HW
31    /// video codec falls back to SW H.264, never SW AV1.
32    Av1,
33    /// H.265 / HEVC — HW encode+decode (VideoToolbox today; MediaCodec later).
34    H265,
35    /// H.264 — HW where present (VideoToolbox / MFT / VAAPI), else SW openh264,
36    /// which is ALWAYS Cisco's downloaded binary: no build of ours compiles a
37    /// H.264 codec in, because only Cisco's own binary carries their patent
38    /// grant. The universal *wire* floor — but see
39    /// [`pick_encode_codec`](crate::api::pick_encode_codec) for the
40    /// state in which a client cannot build the codec locally at all (no HW, blob
41    /// not downloaded yet or switched off).
42    H264,
43}
44
45impl VideoCodec {
46    /// Map to the wire codec enum in `hound-media` so encoded frames packetize
47    /// with the correct header byte.
48    pub fn to_wire(self) -> hound_media::Codec {
49        match self {
50            VideoCodec::Av1 => hound_media::Codec::Av1,
51            VideoCodec::H265 => hound_media::Codec::H265,
52            VideoCodec::H264 => hound_media::Codec::H264,
53        }
54    }
55
56    /// Map back from the wire codec, for receive-side decoder selection.
57    /// `None` for non-video wire codecs (Opus).
58    pub fn from_wire(codec: hound_media::Codec) -> Option<Self> {
59        match codec {
60            hound_media::Codec::Av1 => Some(VideoCodec::Av1),
61            hound_media::Codec::H265 => Some(VideoCodec::H265),
62            hound_media::Codec::H264 => Some(VideoCodec::H264),
63            _ => None,
64        }
65    }
66}

68/// Which of the three video codecs a device supports, as a small copyable record.
69/// Used both as a device's local encode capabilities and as the running
70/// intersection of every receiver's decode capabilities — the two inputs to the
71/// codec ladder ([`pick_encode_codec`](crate::api::pick_encode_codec)).
72#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
73pub struct VideoCodecCaps {
74    pub av1: bool,
75    pub h265: bool,
76    pub h264: bool,
77}
78
79impl VideoCodecCaps {
80    /// Does this set include `codec`?
81    pub fn supports(self, codec: VideoCodec) -> bool {
82        match codec {
83            VideoCodec::Av1 => self.av1,
84            VideoCodec::H265 => self.h265,
85            VideoCodec::H264 => self.h264,
86        }
87    }
88
89    /// The codecs **both** sets support — used to fold each receiver's decode caps
90    /// into the common set a sender may target.
91    pub fn intersect(self, other: Self) -> Self {
92        Self {
93            av1: self.av1 && other.av1,
94            h265: self.h265 && other.h265,
95            h264: self.h264 && other.h264,
96        }
97    }
98}

100/// One captured raw video frame, stamped on the shared capture clock.
101///
102/// `data` layout depends on `format`. For [`PixelFormat::I420`] it is
103/// `Y (w*h) || U (w/2*h/2) || V (w/2*h/2)` with no row padding (stride ==
104/// width per plane) — the contract every encoder/render consumer expects.
105/// The geometry is a validated [`Dimensions`], so 4:2:0 plane math on this
106/// frame can never see an odd size.
107#[derive(Debug, Clone)]
108pub struct RawFrame {
109    pub dims: Dimensions,
110    pub format: PixelFormat,
111    /// Capture timestamp on the session's `CaptureClock` (A/V sync timeline).
112    pub timestamp_us: TimestampUs,
113    /// When this frame was handed to us, for capture→encode latency
114    /// measurement. Not on the wire — local instrumentation only.
115    pub captured_at: Instant,
116    /// Tightly-packed planar pixels (Y‖U‖V for I420, Y‖UVUV… for NV12).
117    ///
118    /// **Empty when [`Self::surface`] is `Some`** — a zero-copy capture frame
119    /// keeps its pixels in the platform buffer. Read pixels via [`Self::as_i420`]
120    /// (which materializes from the surface on demand), never `data` directly,
121    /// unless you have already checked `surface.is_none()`.
122    pub data: Vec<u8>,
123    /// `Some` for a zero-copy capture frame whose pixels live in the platform
124    /// capture buffer (an Apple IOSurface-backed `CVPixelBuffer`) — the hardware
125    /// encoder consumes it directly. `None` for portable byte frames (decoders,
126    /// software-scaled frames, non-Apple capture), where `data` holds the pixels.
127    pub surface: Option<CapturedSurface>,
128}
129
130/// An opaque, zero-copy handle to a captured frame still living in its platform
131/// capture buffer (an Apple IOSurface-backed `CVPixelBuffer`).
132///
133/// Present as [`RawFrame::surface`] on the hardware-encode fast path: the
134/// capture buffer travels to the encoder uncopied, and a byte consumer locks +
135/// converts it only when it genuinely needs planar I420 ([`RawFrame::as_i420`]).
136/// Opaque to portable code; on platforms without a zero-copy backend the inner
137/// type is uninhabited, so `Option<CapturedSurface>` is always `None` and costs
138/// nothing.
139#[derive(Debug, Clone)]
140pub struct CapturedSurface(SurfaceInner);
141
142#[cfg(any(target_os = "macos", target_os = "ios"))]
143type SurfaceInner = crate::platform::apple::AppleSurface;
144#[cfg(target_os = "windows")]
145type SurfaceInner = crate::platform::windows::surface::WindowsSurface;
146#[cfg(not(any(target_os = "macos", target_os = "ios", target_os = "windows")))]
147type SurfaceInner = std::convert::Infallible;
148
149impl CapturedSurface {
150    /// Materialize the surface's pixels as tightly-packed planar I420 (locks the
151    /// platform buffer and converts). `None` on a malformed buffer.
152    pub(crate) fn to_i420(&self) -> Option<Vec<u8>> {
153        #[cfg(any(target_os = "macos", target_os = "ios", target_os = "windows"))]
154        {
155            self.0.to_i420()
156        }
157        #[cfg(not(any(target_os = "macos", target_os = "ios", target_os = "windows")))]
158        {
159            match self.0 {}
160        }
161    }
162
163    /// The raw platform GPU-surface pointer for the zero-copy DECODE→render seam
164    /// (macOS: a `CVPixelBufferRef`, IOSurface-backed NV12). `None` on a platform
165    /// with no GPU-surface backend (the renderer then takes the I420 fallback).
166    /// Valid only while `self` is alive; the renderer takes its own +1 retain.
167    pub fn cv_pixel_buffer_ptr(&self) -> Option<usize> {
168        #[cfg(any(target_os = "macos", target_os = "ios"))]
169        {
170            Some(self.0.cv_pixel_buffer_ptr())
171        }
172        // Windows exposes its GPU surface via `d3d11_texture_ptr` instead.
173        #[cfg(target_os = "windows")]
174        {
175            None
176        }
177        #[cfg(not(any(target_os = "macos", target_os = "ios", target_os = "windows")))]
178        {
179            match self.0 {}
180        }
181    }
182
183    /// The raw `ID3D11Texture2D` pointer (an NV12 frame on the engine's shared
184    /// device) for the zero-copy DECODE→render seam on Windows; `None` elsewhere.
185    /// **Borrowed, not retained**: valid only while `self` is alive, so the
186    /// renderer must consume it synchronously within the call it was passed to —
187    /// the surface's pool lease is what keeps the texture from being rewritten.
188    pub fn d3d11_texture_ptr(&self) -> Option<usize> {
189        #[cfg(target_os = "windows")]
190        {
191            Some(self.0.texture_ptr())
192        }
193        #[cfg(not(target_os = "windows"))]
194        {
195            None
196        }
197    }
198}

227// Deliberate layering inversion: these three conversions reach DOWN into
228// `crate::platform` (convert/scale, and the Apple surface scaler). Keeping them
229// as inherent methods is what lets every consumer say `frame.as_i420()` instead
230// of threading a platform helper through the pipelines; the vocabulary stays
231// platform-agnostic because the platform seam owns all the `cfg`s.
232impl RawFrame {
233    /// Luma width in pixels (convenience for `dims.width()`).
234    pub const fn width(&self) -> u32 {
235        self.dims.width()
236    }
237
238    /// Luma height in pixels (convenience for `dims.height()`).
239    pub const fn height(&self) -> u32 {
240        self.dims.height()
241    }
242
243    /// This frame as planar I420 — borrowed when it already is I420 (the common
244    /// case, zero-copy), converted when it is NV12 or RGB.
245    ///
246    /// The capture fast path keeps Apple frames in NV12 all the way to the
247    /// hardware encoder; consumers that genuinely need I420 — the software
248    /// encoder and the self-preview (an I420 render channel) — call this
249    /// at their boundary so the NV12→I420 de-interleave happens *only* when
250    /// something downstream can't take NV12. `None` on a malformed buffer (a
251    /// skippable frame, never a panic).
252    pub fn as_i420(&self) -> Option<std::borrow::Cow<'_, RawFrame>> {
253        use crate::platform::generic::convert;
254        use std::borrow::Cow;
255        let (w, h) = (self.width() as usize, self.height() as usize);
256        let data = match &self.surface {
257            // Zero-copy capture frame: lock + convert the platform buffer.
258            Some(surface) => surface.to_i420()?,
259            None => match self.format {
260                PixelFormat::I420 => return Some(Cow::Borrowed(self)),
261                // Tightly-packed NV12 → I420 (Y stride == width).
262                PixelFormat::Nv12 => convert::nv12_bytes_to_i420(&self.data, w, h, w)?,
263                PixelFormat::Rgb => convert::rgb_to_i420(&self.data, w, h),
264            },
265        };
266        Some(Cow::Owned(RawFrame {
267            dims: self.dims,
268            format: PixelFormat::I420,
269            timestamp_us: self.timestamp_us,
270            captured_at: self.captured_at,
271            data,
272            surface: None,
273        }))
274    }
275
276    /// A copy of this frame scaled to `dims` as tightly-packed planar I420
277    /// (materializing a zero-copy surface first). `None` if `dims` are invalid (zero
278    /// or odd — 4:2:0 needs even extents) or the pixels can't be read. The result is
279    /// a portable byte frame (`surface: None`), so the zero-copy hardware-encode path
280    /// is traded for one software scale: callers use this only when the quality
281    /// ladder degrades resolution, and pass the frame through untouched at full res.
282    pub fn scaled_to(&self, dims: Dimensions) -> Option<RawFrame> {
283        let src = self.as_i420()?;
284        let scaled = crate::platform::generic::scale::scale_i420(
285            &src.data,
286            src.width() as usize,
287            src.height() as usize,
288            dims.width() as usize,
289            dims.height() as usize,
290        )?;
291        Some(RawFrame {
292            dims,
293            format: PixelFormat::I420,
294            timestamp_us: self.timestamp_us,
295            captured_at: self.captured_at,
296            data: scaled,
297            surface: None,
298        })
299    }
300
301    /// A copy of this frame downscaled to `dims` **into a platform surface**, so a
302    /// hardware encoder can take the result uncopied.
303    ///
304    /// The counterpart to [`Self::scaled_to`] for the simulcast low rung. That one
305    /// materializes I420, scales on the CPU, and hands the hardware encoder bytes
306    /// it must pack back into a platform buffer — three full-frame touches. This
307    /// one scales in the capture layout (Apple: NV12→NV12 straight into a pooled
308    /// IOSurface-backed `CVPixelBuffer`) and the encoder consumes it directly, so
309    /// only the unavoidable read-and-write remain.
310    ///
311    /// `Some` only when this frame is surface-backed **and** the platform has a
312    /// surface scaler — i.e. the Apple capture fast path, and only for a genuine
313    /// downscale. `None` for a byte frame, off Apple, or on any geometry the
314    /// kernel declines; callers fall back to [`Self::scaled_to`], which is also
315    /// what a software encoder wants.
316    pub fn scaled_to_surface(&self, dims: Dimensions) -> Option<RawFrame> {
317        #[cfg(any(target_os = "macos", target_os = "ios"))]
318        {
319            let scaled = crate::platform::apple::surface_scale::downscale_surface(
320                self.surface.as_ref()?.apple(),
321                dims,
322            )?;
323            Some(RawFrame {
324                dims,
325                // Capture-native NV12, straight into the low VideoToolbox encoder.
326                format: PixelFormat::Nv12,
327                timestamp_us: self.timestamp_us,
328                captured_at: self.captured_at,
329                data: Vec::new(), // pixels live in the surface
330                surface: Some(CapturedSurface::from_apple(scaled)),
331            })
332        }
333        #[cfg(not(any(target_os = "macos", target_os = "ios")))]
334        {
335            let _ = dims;
336            None
337        }
338    }
339}

341/// One encoded output unit from a video encoder (an AV1 temporal unit / an
342/// H.264 access unit). Carries the keyframe flag and the source timestamp so
343/// the packetizer can stamp the wire header straight from it.
344#[derive(Debug, Clone)]
345pub struct EncodedChunk {
346    pub data: Vec<u8>,
347    pub keyframe: bool,
348    /// Source capture timestamp (A/V sync). Producers that cannot thread the
349    /// source timestamp (HW callbacks) leave this zero and the pipeline stamps
350    /// the wire from the originating [`RawFrame`].
351    pub timestamp_us: TimestampUs,
352    /// The L1T2 temporal layer the encoder assigned this frame: 0 = base
353    /// (self-decodable, referenced by later frames), 1 = enhancement
354    /// (droppable without breaking decode). An encoder without temporal-layer
355    /// support emits 0 for every frame, which is a valid L1T1 stream. Rides
356    /// the MoQ object envelope's reserved byte, so the relay can derive a
357    /// base-layer-only track without parsing the bitstream.
358    pub temporal_id: u8,
359}

365    #[test]
366    fn as_i420_borrows_i420_and_converts_nv12() {
367        use std::borrow::Cow;
368        let dims = Dimensions::new(4, 2);
369        let (w, h) = (4usize, 2usize);
370        let (cw, ch) = (w / 2, h / 2);
371        let frame = |format, data| RawFrame {
372            dims,
373            format,
374            timestamp_us: TimestampUs::ZERO,
375            captured_at: std::time::Instant::now(),
376            data,
377            surface: None,
378        };
379
380        // I420 → borrowed (zero-copy), same bytes.
381        let i420 = frame(PixelFormat::I420, vec![7u8; w * h + 2 * cw * ch]);
382        let got = i420.as_i420().expect("i420 as_i420");
383        assert!(
384            matches!(got, Cow::Borrowed(_)),
385            "I420 must borrow, not copy"
386        );
387
388        // NV12 (Y ‖ UVUV…) → owned I420 of the right size; chroma de-interleaved.
389        let mut nv12 = vec![0u8; w * h]; // Y
390        for _ in 0..(cw * ch) {
391            nv12.push(70); // U
392            nv12.push(200); // V
393        }
394        let nv12_frame = frame(PixelFormat::Nv12, nv12);
395        let got = nv12_frame.as_i420().expect("nv12 as_i420");
396        assert!(matches!(got, Cow::Owned(_)), "NV12 must convert (owned)");
397        assert_eq!(got.format, PixelFormat::I420);
398        assert_eq!(got.data.len(), w * h + 2 * cw * ch);
399        // De-interleaved planes: U all 70, V all 200.
400        let u_off = w * h;
401        assert!(got.data[u_off..u_off + cw * ch].iter().all(|&u| u == 70));
402        assert!(got.data[u_off + cw * ch..].iter().all(|&v| v == 200));
403    }

405    #[test]
406    fn scaled_to_downscales_to_i420_of_target_size() {
407        // A 16x8 flat I420 frame downscaled to 8x4 → tightly-packed I420 at the new
408        // geometry, surface dropped (portable byte frame).
409        let (w, h) = (16usize, 8usize);
410        let src = RawFrame {
411            dims: Dimensions::new(w as u32, h as u32),
412            format: PixelFormat::I420,
413            timestamp_us: TimestampUs::from_micros(123),
414            captured_at: std::time::Instant::now(),
415            data: vec![128u8; w * h + 2 * (w / 2) * (h / 2)],
416            surface: None,
417        };
418        let dst = Dimensions::new(8, 4);
419        let scaled = src.scaled_to(dst).expect("downscale 16x8 → 8x4");
420        assert_eq!(scaled.dims, dst);
421        assert_eq!(scaled.format, PixelFormat::I420);
422        assert_eq!(
423            scaled.data.len(),
424            8 * 4 + 2 * 4 * 2,
425            "tightly-packed I420 at 8x4"
426        );
427        assert_eq!(scaled.timestamp_us, src.timestamp_us, "timestamp preserved");
428        assert!(
429            scaled.surface.is_none(),
430            "scaled frame is a portable byte frame"
431        );
432
433        // The surface downscale has nothing to work with here (a byte frame): it
434        // must decline rather than invent a surface, so the caller falls back to
435        // the portable scale above. Holds on every platform, Apple included.
436        assert!(
437            src.scaled_to_surface(dst).is_none(),
438            "a byte frame has no surface to scale"
439        );
440    }
