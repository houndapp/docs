1//! The sans-IO relying party: a thin wrapper over `webauthn_rs::Webauthn`
2//! exposing the four ceremony calls plus the RP-config accessors the
3//! `.well-known` endpoints read. It owns NO state beyond
4//! the configured `Webauthn` — the in-flight ceremony state (`PasskeyRegistration`
5//! / `PasskeyAuthentication`) is returned to and stored by the caller, and the
6//! `Passkey` credentials live in the caller's database.
7//!
8//! We wrap `webauthn-rs`; we do not reimplement any crypto.
9
10use uuid::Uuid;
11use webauthn_rs::prelude::{Url, Webauthn, WebauthnBuilder};
12
13use crate::error::PasskeyError;
14use crate::proto::{
15    CreationChallengeResponse, PublicKeyCredential, RegisterPublicKeyCredential,
16    RequestChallengeResponse,
17};

19// The relying-party crypto types the caller must store across a ceremony or in its
20// database. Re-exported so the backend imports them from `hound-passkey` rather
21// than depending on `webauthn-rs` directly.
22pub use webauthn_rs::prelude::{
23    AuthenticationResult, Passkey, PasskeyAuthentication, PasskeyRegistration,
24};
25
26/// A configured WebAuthn relying party. Sans-IO: every method is synchronous and
27/// pure over the wrapped `Webauthn`.
28#[derive(Clone, Debug)]
29pub struct RelyingParty {
30    rp_id: String,
31    webauthn: Webauthn,
32}
33
34impl RelyingParty {
35    /// Build the relying party for `rp_id` / `rp_origin`, appending any
36    /// `extra_origins` (e.g. dev origins, native app origins).
37    pub fn new(rp_id: &str, rp_origin: &Url, extra_origins: &[Url]) -> Result<Self, PasskeyError> {
38        let mut builder = WebauthnBuilder::new(rp_id, rp_origin)?;
39        for origin in extra_origins {
40            builder = builder.append_allowed_origin(origin);
41        }
42        Ok(Self {
43            rp_id: rp_id.to_owned(),
44            webauthn: builder.build()?,
45        })
46    }
47
48    /// Begin registration for a (provisional) user. Returns the creation challenge
49    /// to send to the client and the [`PasskeyRegistration`] state the caller must
50    /// persist until `finish_registration`.
51    pub fn start_registration(
52        &self,
53        user_id: Uuid,
54        email: &str,
55        display_name: &str,
56    ) -> Result<(CreationChallengeResponse, PasskeyRegistration), PasskeyError> {
57        Ok(self
58            .webauthn
59            .start_passkey_registration(user_id, email, display_name, None)?)
60    }
61
62    /// Complete registration: verify the client's created credential against the
63    /// stored [`PasskeyRegistration`], yielding the [`Passkey`] to persist.
64    pub fn finish_registration(
65        &self,
66        credential: &RegisterPublicKeyCredential,
67        state: &PasskeyRegistration,
68    ) -> Result<Passkey, PasskeyError> {
69        Ok(self
70            .webauthn
71            .finish_passkey_registration(credential, state)?)
72    }
73
74    /// Begin authentication against the user's known passkeys. Returns the request
75    /// challenge to send to the client and the [`PasskeyAuthentication`] state the
76    /// caller must persist until `finish_authentication`.
77    pub fn start_authentication(
78        &self,
79        passkeys: &[Passkey],
80    ) -> Result<(RequestChallengeResponse, PasskeyAuthentication), PasskeyError> {
81        Ok(self.webauthn.start_passkey_authentication(passkeys)?)
82    }
83
84    /// Complete authentication: verify the client's assertion against the stored
85    /// [`PasskeyAuthentication`], yielding the [`AuthenticationResult`] (which the
86    /// caller inspects for `needs_update()` to roll the credential counter).
87    pub fn finish_authentication(
88        &self,
89        credential: &PublicKeyCredential,
90        state: &PasskeyAuthentication,
91    ) -> Result<AuthenticationResult, PasskeyError> {
92        Ok(self
93            .webauthn
94            .finish_passkey_authentication(credential, state)?)
95    }
96
97    /// The configured relying-party id (`rp_id`). Served by the `/webauthn`
98    /// RP-config endpoint.
99    pub fn rp_id(&self) -> &str {
100        &self.rp_id
101    }
102
103    /// The allowed origins. Served (as bare `scheme://host[:port]`) by the
104    /// `/webauthn` RP-config endpoint.
105    pub fn get_allowed_origins(&self) -> &[Url] {
106        self.webauthn.get_allowed_origins()
107    }
108}

121    /// The dev stack's primary origin: the `hound-auth` page the
122    /// browser-deferred ceremony runs in (`services/backend/compose.dev.yaml`,
123    /// mirrored in `tools/stack/src/backend.rs`'s `RP_ORIGIN`).
124    const DEV_BROWSER_ORIGIN: &str = "http://localhost:3100";
125
126    /// The origin a NATIVE macOS/iOS ceremony presents. Apple's app-based
127    /// WebAuthn (`ASAuthorization*PublicKeyCredential*`) has no web page to
128    /// take an origin from, so it synthesises `https://<rpId>` — always https,
129    /// never a port, whatever the relying party actually listens on. With
130    /// `rp_id = localhost` (dev) that is this string; in prod it is
131    /// `https://api.hound.chat`, which already equals `RP_ORIGIN` and so needs
132    /// no extra entry. Dev is the case that needs one.
133    const NATIVE_APPLE_ORIGIN: &str = "https://localhost";
134
135    /// Long enough that the software token never times out; it does no IO.
136    const TIMEOUT_MS: u32 = 5_000;
137
138    /// A full register + authenticate round trip against `rp`, with the
139    /// authenticator signing `origin` into `clientDataJSON` — the only knob
140    /// this test needs, since origin validation is the whole subject.
141    ///
142    /// `SoftToken::new(true)` (`falsify_uv`): webauthn-rs's *passkey* ceremony
143    /// always demands user verification, so a token that reports itself
144    /// UV-incapable fails inside the authenticator before the RP ever sees an
145    /// origin. Authenticator-side calls therefore `expect` — a failure there is
146    /// a broken test, not the assertion under test; only the two `RelyingParty`
147    /// finish calls are allowed to fail, and both fail with the same
148    /// `WebauthnError::InvalidRPOrigin`.
149    fn round_trip(rp: &RelyingParty, origin: &str) -> Result<(), PasskeyError> {
150        let origin = Url::parse(origin).expect("test origin parses");
151        let (mut token, _ca) = SoftToken::new(true).expect("build software token");
152
153        let (challenge, reg_state) =
154            rp.start_registration(Uuid::now_v7(), "user@hound.chat", "Test User")?;
155        let credential = token
156            .perform_register(origin.clone(), challenge.public_key, TIMEOUT_MS)
157            .expect("software token creates a credential");
158        let passkey = rp.finish_registration(&credential, &reg_state)?;
159
160        let (challenge, auth_state) = rp.start_authentication(&[passkey])?;
161        let assertion = token
162            .perform_auth(origin, challenge.public_key, TIMEOUT_MS)
163            .expect("software token signs an assertion");
164        rp.finish_authentication(&assertion, &auth_state).map(drop)
165    }
166
167    /// The dev stack's configuration: the native app's `https://localhost` is
168    /// an EXTRA origin next to the browser page's `http://localhost:3100`, and
169    /// both complete a ceremony. Without this the native macOS sheet succeeds
170    /// and then the backend rejects the result.
171    #[test]
172    fn both_the_browser_page_and_the_native_app_origin_are_accepted() {
173        let rp = RelyingParty::new(
174            "localhost",
175            &DEV_BROWSER_ORIGIN.parse().unwrap(),
176            &[NATIVE_APPLE_ORIGIN.parse().unwrap()],
177        )
178        .unwrap();
179
180        round_trip(&rp, DEV_BROWSER_ORIGIN).expect("browser-page ceremony");
181        round_trip(&rp, NATIVE_APPLE_ORIGIN).expect("native macOS/iOS ceremony");
182    }
183
184    /// The guard the entry above pays for: `https://localhost` differs from
185    /// `http://localhost:3100` in both scheme and port, and webauthn-rs matches
186    /// origins exactly (no `allow_any_port`, no scheme coercion), so dropping
187    /// the extra origin silently breaks native sign-in.
188    #[test]
189    fn the_native_app_origin_is_rejected_when_not_configured() {
190        let rp = RelyingParty::new("localhost", &DEV_BROWSER_ORIGIN.parse().unwrap(), &[]).unwrap();
191
192        let err = round_trip(&rp, NATIVE_APPLE_ORIGIN)
193            .expect_err("an unlisted origin must not complete a ceremony");
194        assert!(
195            matches!(
196                err,
197                PasskeyError::WebAuthn(webauthn_rs::prelude::WebauthnError::InvalidRPOrigin)
198            ),
199            "expected InvalidRPOrigin, got: {err:?}",
200        );
201    }

203    // ---------------------------------------------------------------------
204    // Stored-credential compatibility (webauthn-rs 0.5.5 -> 0.6.x-dev).
205    //
206    // PERMANENT REGRESSION GUARD — do not delete on a version bump; re-run it.
207    //
208    // The backend persists `Passkey` as a JSON column, and prod holds rows
209    // written by webauthn-rs 0.5.5. The 0.6 line replaced OpenSSL with
210    // RustCrypto (`crypto-glue`) under the same public types, so nothing in
211    // the compiler stops a serde-shape or key-interpretation change from
212    // locking every existing user out — only these fixtures do.
213    //
214    // `tests/fixtures/*-v0.5.5.json` were captured under 0.5.5 and are frozen:
215    // - `passkey-es256-v0.5.5.json` + `assertion-es256-v0.5.5.json` come from
216    //   one real softtoken ceremony (register, then authenticate) that 0.5.5
217    //   itself accepted, so a failure here is 0.6's.
218    // - `passkey-rs256-v0.5.5.json` is hand-built in the 0.5.5 shape and was
219    //   round-trip-verified under 0.5.5: RS256 (-257) is advertised alongside
220    //   ES256 (-7) by both lines and is what Windows Hello / TPM keys use, but
221    //   the softtoken only ever mints ES256, so it cannot be captured live.
222    // ---------------------------------------------------------------------
223
224    const ES256_PASSKEY: &str = include_str!("../tests/fixtures/passkey-es256-v0.5.5.json");
225    const ES256_ASSERTION: &str = include_str!("../tests/fixtures/assertion-es256-v0.5.5.json");
226    const RS256_PASSKEY: &str = include_str!("../tests/fixtures/passkey-rs256-v0.5.5.json");
227
228    /// The bytes WebAuthn signs over: `authenticatorData || SHA-256(clientDataJSON)`
229    /// (§7.2 step 19). Re-derived here so the guard can verify a captured
230    /// assertion directly against the stored key, with no in-flight ceremony
231    /// state to serialise.
232    fn signed_bytes(assertion: &PublicKeyCredential) -> Vec<u8> {
233        use sha2::{Digest, Sha256};
234
235        let mut data = assertion.response.authenticator_data.clone();
236        data.extend_from_slice(&Sha256::digest(&assertion.response.client_data_json));
237        data
238    }
239
240    /// A 0.5.5-written ES256 row still loads, still enters a ceremony, and its
241    /// public key still verifies the signature its own authenticator produced
242    /// back then. The last clause is the one that matters: it proves the stored
243    /// key material means the same thing to RustCrypto that it meant to OpenSSL.
244    #[test]
245    fn a_v055_es256_credential_still_loads_and_verifies() {
246        let passkey: Passkey =
247            serde_json::from_str(ES256_PASSKEY).expect("0.5.5 ES256 row deserialises");
248        let assertion: PublicKeyCredential =
249            serde_json::from_str(ES256_ASSERTION).expect("0.5.5 assertion deserialises");
250
251        assert_eq!(*passkey.cred_algorithm(), COSEAlgorithm::ES256);
252        assert_eq!(
253            passkey.cred_id().as_slice(),
254            assertion.raw_id.as_slice(),
255            "fixtures must come from the same credential",
256        );
257
258        // The row is usable as ceremony input, not merely as bytes.
259        let rp = RelyingParty::new("localhost", &DEV_BROWSER_ORIGIN.parse().unwrap(), &[]).unwrap();
260        let (challenge, _state) = rp
261            .start_authentication(std::slice::from_ref(&passkey))
262            .expect("a stored credential can open an authentication ceremony");
263        let allowed = challenge.public_key.allow_credentials;
264        assert_eq!(allowed.len(), 1);
265        assert_eq!(allowed[0].id.as_slice(), passkey.cred_id().as_slice());
266
267        assert!(
268            passkey
269                .get_public_key()
270                .verify_signature(&assertion.response.signature, &signed_bytes(&assertion))
271                .expect("the stored ES256 key is usable for verification"),
272            "the 0.5.5 assertion must still verify against the 0.5.5 public key",
273        );
274    }

276    /// The RS256 half of the same guard. No captured assertion (see the module
277    /// comment), so instead of verifying a real signature this drives
278    /// `verify_signature` — which parses the stored modulus/exponent BEFORE it
279    /// looks at the signature — with a correctly-sized dud. `Ok(false)` means
280    /// the key parsed and merely didn't match; a stored key RustCrypto cannot
281    /// read would come back `Err(RsaParametersInvalid)` instead, which is
282    /// exactly the lockout this guard exists to catch.
283    #[test]
284    fn a_v055_rs256_credential_still_loads() {
285        let passkey: Passkey =
286            serde_json::from_str(RS256_PASSKEY).expect("0.5.5 RS256 row deserialises");
287
288        assert_eq!(*passkey.cred_algorithm(), COSEAlgorithm::RS256);
289        let COSEKeyType::RSA(rsa) = &passkey.get_public_key().key else {
290            panic!(
291                "expected an RSA key, got {:?}",
292                passkey.get_public_key().key
293            );
294        };
295        assert_eq!(rsa.n.len(), 256, "2048-bit modulus survives the round trip");
296        assert_eq!(rsa.e, [0x01, 0x00, 0x01]);
297        assert!(
298            !passkey
299                .get_public_key()
300                .verify_signature(&[0u8; 256], b"not the signed bytes")
301                .expect("the stored RSA key is usable for verification"),
302            "a dud signature must not verify",
303        );
304
305        let rp = RelyingParty::new("localhost", &DEV_BROWSER_ORIGIN.parse().unwrap(), &[]).unwrap();
306        let (challenge, _state) = rp
307            .start_authentication(std::slice::from_ref(&passkey))
308            .expect("a stored credential can open an authentication ceremony");
309        assert_eq!(challenge.public_key.allow_credentials.len(), 1);
310    }

312    /// Re-serialising a 0.5.5 row must reproduce it byte for byte. The backend
313    /// rewrites the whole column on every counter roll (`update_credential` ->
314    /// `update_passkey`), so a one-way-compatible shape would silently rewrite
315    /// prod into a form only 0.6 can read — an upgrade that cannot be rolled
316    /// back. Pretty-printed because that is what the fixtures store.
317    #[test]
318    fn rewriting_a_v055_row_does_not_change_it() {
319        for (name, fixture) in [("es256", ES256_PASSKEY), ("rs256", RS256_PASSKEY)] {
320            let passkey: Passkey = serde_json::from_str(fixture).expect("row deserialises");
321            let rewritten = serde_json::to_string_pretty(&passkey).expect("row reserialises");
322            assert_eq!(rewritten.trim(), fixture.trim(), "{name} row changed shape");
323        }
324    }
