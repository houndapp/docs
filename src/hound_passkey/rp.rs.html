1//! The sans-IO relying party: a thin wrapper over `webauthn_rs::Webauthn`
2//! exposing the four ceremony calls plus the RP-config accessors the
3//! `.well-known` endpoints read. It owns NO state beyond
4//! the configured `Webauthn` — the in-flight ceremony state (`PasskeyRegistration`
5//! / `PasskeyAuthentication`) is returned to and stored by the caller, and the
6//! `Passkey` credentials live in the caller's database.
7//!
8//! We wrap `webauthn-rs`; we do not reimplement any crypto.
9
10use uuid::Uuid;
11use webauthn_rs::prelude::{Url, Webauthn, WebauthnBuilder};
12
13use crate::error::PasskeyError;
14use crate::proto::{
15    CreationChallengeResponse, PublicKeyCredential, RegisterPublicKeyCredential,
16    RequestChallengeResponse,
17};
18
19// The relying-party crypto types the caller must store across a ceremony or in its
20// database. Re-exported so the backend imports them from `hound-passkey` rather
21// than depending on `webauthn-rs` directly.
22pub use webauthn_rs::prelude::{
23    AuthenticationResult, Passkey, PasskeyAuthentication, PasskeyRegistration,
24};
25
26/// A configured WebAuthn relying party. Sans-IO: every method is synchronous and
27/// pure over the wrapped `Webauthn`.
28#[derive(Clone, Debug)]
29pub struct RelyingParty {
30    rp_id: String,
31    webauthn: Webauthn,
32}
33
34impl RelyingParty {
35    /// Build the relying party for `rp_id` / `rp_origin`, appending any
36    /// `extra_origins` (e.g. dev origins, native app origins).
37    pub fn new(rp_id: &str, rp_origin: &Url, extra_origins: &[Url]) -> Result<Self, PasskeyError> {
38        let mut builder = WebauthnBuilder::new(rp_id, rp_origin)?;
39        for origin in extra_origins {
40            builder = builder.append_allowed_origin(origin);
41        }
42        Ok(Self {
43            rp_id: rp_id.to_owned(),
44            webauthn: builder.build()?,
45        })
46    }
47
48    /// Begin registration for a (provisional) user. Returns the creation challenge
49    /// to send to the client and the [`PasskeyRegistration`] state the caller must
50    /// persist until `finish_registration`.
51    pub fn start_registration(
52        &self,
53        user_id: Uuid,
54        email: &str,
55        display_name: &str,
56    ) -> Result<(CreationChallengeResponse, PasskeyRegistration), PasskeyError> {
57        Ok(self
58            .webauthn
59            .start_passkey_registration(user_id, email, display_name, None)?)
60    }
61
62    /// Complete registration: verify the client's created credential against the
63    /// stored [`PasskeyRegistration`], yielding the [`Passkey`] to persist.
64    pub fn finish_registration(
65        &self,
66        credential: &RegisterPublicKeyCredential,
67        state: &PasskeyRegistration,
68    ) -> Result<Passkey, PasskeyError> {
69        Ok(self
70            .webauthn
71            .finish_passkey_registration(credential, state)?)
72    }
73
74    /// Begin authentication against the user's known passkeys. Returns the request
75    /// challenge to send to the client and the [`PasskeyAuthentication`] state the
76    /// caller must persist until `finish_authentication`.
77    pub fn start_authentication(
78        &self,
79        passkeys: &[Passkey],
80    ) -> Result<(RequestChallengeResponse, PasskeyAuthentication), PasskeyError> {
81        Ok(self.webauthn.start_passkey_authentication(passkeys)?)
82    }
83
84    /// Complete authentication: verify the client's assertion against the stored
85    /// [`PasskeyAuthentication`], yielding the [`AuthenticationResult`] (which the
86    /// caller inspects for `needs_update()` to roll the credential counter).
87    pub fn finish_authentication(
88        &self,
89        credential: &PublicKeyCredential,
90        state: &PasskeyAuthentication,
91    ) -> Result<AuthenticationResult, PasskeyError> {
92        Ok(self
93            .webauthn
94            .finish_passkey_authentication(credential, state)?)
95    }
96
97    /// The configured relying-party id (`rp_id`). Served by the `/webauthn`
98    /// RP-config endpoint.
99    pub fn rp_id(&self) -> &str {
100        &self.rp_id
101    }
102
103    /// The allowed origins. Served (as bare `scheme://host[:port]`) by the
104    /// `/webauthn` RP-config endpoint.
105    pub fn get_allowed_origins(&self) -> &[Url] {
106        self.webauthn.get_allowed_origins()
107    }
108}

110#[cfg(test)]
111mod tests {
112    use uuid::Uuid;
113    use webauthn_authenticator_rs::AuthenticatorBackend;
114    use webauthn_authenticator_rs::softtoken::SoftToken;
115
116    use super::*;
117
118    /// The dev stack's primary origin: the `hound-auth` page the
119    /// browser-deferred ceremony runs in (`services/backend/compose.dev.yaml`,
120    /// mirrored in `tools/stack/src/backend.rs`'s `RP_ORIGIN`).
121    const DEV_BROWSER_ORIGIN: &str = "http://localhost:3100";
122
123    /// The origin a NATIVE macOS/iOS ceremony presents. Apple's app-based
124    /// WebAuthn (`ASAuthorization*PublicKeyCredential*`) has no web page to
125    /// take an origin from, so it synthesises `https://<rpId>` — always https,
126    /// never a port, whatever the relying party actually listens on. With
127    /// `rp_id = localhost` (dev) that is this string; in prod it is
128    /// `https://api.hound.chat`, which already equals `RP_ORIGIN` and so needs
129    /// no extra entry. Dev is the case that needs one.
130    const NATIVE_APPLE_ORIGIN: &str = "https://localhost";
131
132    /// Long enough that the software token never times out; it does no IO.
133    const TIMEOUT_MS: u32 = 5_000;
134
135    /// A full register + authenticate round trip against `rp`, with the
136    /// authenticator signing `origin` into `clientDataJSON` — the only knob
137    /// this test needs, since origin validation is the whole subject.
138    ///
139    /// `SoftToken::new(true)` (`falsify_uv`): webauthn-rs's *passkey* ceremony
140    /// always demands user verification, so a token that reports itself
141    /// UV-incapable fails inside the authenticator before the RP ever sees an
142    /// origin. Authenticator-side calls therefore `expect` — a failure there is
143    /// a broken test, not the assertion under test; only the two `RelyingParty`
144    /// finish calls are allowed to fail, and both fail with the same
145    /// `WebauthnError::InvalidRPOrigin`.
146    fn round_trip(rp: &RelyingParty, origin: &str) -> Result<(), PasskeyError> {
147        let origin = Url::parse(origin).expect("test origin parses");
148        let (mut token, _ca) = SoftToken::new(true).expect("build software token");
149
150        let (challenge, reg_state) =
151            rp.start_registration(Uuid::now_v7(), "user@hound.chat", "Test User")?;
152        let credential = token
153            .perform_register(origin.clone(), challenge.public_key, TIMEOUT_MS)
154            .expect("software token creates a credential");
155        let passkey = rp.finish_registration(&credential, &reg_state)?;
156
157        let (challenge, auth_state) = rp.start_authentication(&[passkey])?;
158        let assertion = token
159            .perform_auth(origin, challenge.public_key, TIMEOUT_MS)
160            .expect("software token signs an assertion");
161        rp.finish_authentication(&assertion, &auth_state).map(drop)
162    }
163
164    /// The dev stack's configuration: the native app's `https://localhost` is
165    /// an EXTRA origin next to the browser page's `http://localhost:3100`, and
166    /// both complete a ceremony. Without this the native macOS sheet succeeds
167    /// and then the backend rejects the result.
168    #[test]
169    fn both_the_browser_page_and_the_native_app_origin_are_accepted() {
170        let rp = RelyingParty::new(
171            "localhost",
172            &DEV_BROWSER_ORIGIN.parse().unwrap(),
173            &[NATIVE_APPLE_ORIGIN.parse().unwrap()],
174        )
175        .unwrap();
176
177        round_trip(&rp, DEV_BROWSER_ORIGIN).expect("browser-page ceremony");
178        round_trip(&rp, NATIVE_APPLE_ORIGIN).expect("native macOS/iOS ceremony");
179    }
180
181    /// The guard the entry above pays for: `https://localhost` differs from
182    /// `http://localhost:3100` in both scheme and port, and webauthn-rs matches
183    /// origins exactly (no `allow_any_port`, no scheme coercion), so dropping
184    /// the extra origin silently breaks native sign-in.
185    #[test]
186    fn the_native_app_origin_is_rejected_when_not_configured() {
187        let rp = RelyingParty::new("localhost", &DEV_BROWSER_ORIGIN.parse().unwrap(), &[]).unwrap();
188
189        let err = round_trip(&rp, NATIVE_APPLE_ORIGIN)
190            .expect_err("an unlisted origin must not complete a ceremony");
191        assert!(
192            matches!(
193                err,
194                PasskeyError::WebAuthn(webauthn_rs::prelude::WebauthnError::InvalidRPOrigin)
195            ),
196            "expected InvalidRPOrigin, got: {err:?}",
197        );
198    }
