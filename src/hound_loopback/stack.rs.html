1//! [`LoopbackStack`]: relay → backend → SFU, each started only once the one
2//! before it can serve, so a scenario begins against a stack whose SFU is
3//! already routable.
4
5use std::net::{Ipv4Addr, SocketAddrV4};
6use std::sync::Arc;
7use std::time::Duration;
8
9use clap::Parser;

11use hound_sfu::sfu::Sfu;

19const LOOPBACK: Ipv4Addr = Ipv4Addr::LOCALHOST;
20
21/// Shared by the backend (`--relay-hmac-secret`, which also bearer-gates
22/// `/internal`) and the SFU (relay-token verification) — they must agree.
23const RELAY_SECRET: &str = "loopback-relay-secret";
24
25/// How long the backend gets to migrate, provision and answer HTTP.
26const BACKEND_READY_BUDGET: Duration = Duration::from_secs(30);
27
28/// Readiness poll cadence. Everything is in-process on loopback.
29const READY_POLL: Duration = Duration::from_millis(50);
30
31/// SFU re-registration cadence — the production SFU's heartbeat interval,
32/// well inside the backend registry's 90 s liveness TTL, so a scenario of any
33/// length keeps a routable node.
34const REGISTER_INTERVAL: Duration = Duration::from_secs(30);
35
36/// A running relay + backend + SFU. Dropping it tears all three down
37/// abruptly; [`LoopbackStack::shutdown`] closes connections first.
38pub struct LoopbackStack {
39    backend_url: String,
40    inbox: Arc<MemoryInbox>,
41    /// Aborted on drop: the backend server and the SFU accept/heartbeat loops.
42    tasks: Vec<JoinHandle<()>>,
43    sfu: Arc<Sfu>,
44    /// Stops on drop; `None` once [`LoopbackStack::shutdown`] has stopped it.
45    relay: Option<iroh_relay::server::Server>,
46    /// The backend's local upload root.
47    _storage: tempfile::TempDir,
48}
49
50impl LoopbackStack {
51    /// Start the stack. Returns once the SFU is registered and is the node
52    /// the backend routes new voice joins to.
53    ///
54    /// Must run on a multi-threaded runtime: the services share it with the
55    /// scenario's own awaits.
56    pub async fn start() -> eyre::Result<Self> {
57        // Before the backend's `drive` tries to install its own subscriber:
58        // the global one is first-wins, and the testkit's routes to the
59        // per-test capture.
60        hound_testkit::env::init_test_tracing();
61
62        let relay = start_relay().await?;
63        let relay_url = format!(
64            "http://{}",
65            relay
66                .http_addr()
67                .ok_or_else(|| eyre::eyre!("relay has no http listener"))?
68        );
69
70        let inbox = Arc::new(MemoryInbox::default());
71        let storage = tempfile::tempdir()?;
72        let (backend_url, backend) = start_backend(&relay_url, inbox.clone(), &storage).await?;
73        let mut tasks = vec![backend];
74
75        let sfu = Arc::new(
76            Sfu::bind(
77                relay_url.parse()?,
78                iroh::SecretKey::generate(),
79                RELAY_SECRET.to_owned(),
80                SocketAddrV4::new(LOOPBACK, free_udp_port()?),
81                RelayConfig::default(),
82            )
83            .await?,
84        );
85        tasks.push(tokio::spawn({
86            let sfu = sfu.clone();
87            async move {
88                if let Err(e) = sfu.run().await {
89                    tracing::error!("loopback SFU accept loop failed: {e}");
90                }
91            }
92        }));
93        let node_id = sfu.node_id().to_string();
94        let http = reqwest::Client::new();
95        register_sfu(&http, &backend_url, &node_id).await?;
96        tasks.push(tokio::spawn(heartbeat(
97            http.clone(),
98            backend_url.clone(),
99            node_id.clone(),
100        )));
101
102        let nodes = sfu_nodes(&http, &backend_url).await?;
103        if nodes.best.as_deref() != Some(node_id.as_str()) {
104            eyre::bail!("backend does not route joins to the loopback SFU {node_id}: {nodes:?}");
105        }
106
107        Ok(Self {
108            backend_url,
109            inbox,
110            tasks,
111            sfu,
112            relay: Some(relay),
113            _storage: storage,
114        })
115    }
116
117    pub fn backend_url(&self) -> &str {
118        &self.backend_url
119    }
120
121    /// The backend's outbound mail, as a testkit [`MailSource`].
122    pub fn mail(&self) -> MailSource {
123        MailSource::Inbox(self.inbox.clone())
124    }
125
126    /// A scenario [`Env`] against this stack (bootstraps the default
127    /// community's space pair).
128    pub async fn env(&self) -> eyre::Result<Env> {
129        Env::with_mail_source(self.backend_url.clone(), self.mail()).await
130    }
131
132    /// Tear down gracefully: connected peers get a QUIC close from the SFU and
133    /// the relay, rather than the abort a drop during runtime shutdown causes.
134    pub async fn shutdown(mut self) -> eyre::Result<()> {
135        for task in self.tasks.drain(..) {
136            task.abort();
137        }
138        self.sfu.close().await;
139        if let Some(relay) = self.relay.take() {
140            relay.shutdown().await?;
141        }
142        Ok(())
143    }
144}
145
146impl Drop for LoopbackStack {
147    fn drop(&mut self) {
148        for task in &self.tasks {
149            task.abort();
150        }
151    }
152}

154/// An iroh relay on `127.0.0.1:0`, plain HTTP (the dev stack's relay is too).
155/// No QUIC address-discovery listener: peers here share one host.
156async fn start_relay() -> eyre::Result<iroh_relay::server::Server> {
157    let mut config = iroh_relay::server::ServerConfig::default();
158    config.relay = Some(iroh_relay::server::RelayConfig::new(
159        std::net::SocketAddr::from((LOOPBACK, 0)),
160    ));
161    Ok(iroh_relay::server::Server::spawn(config).await?)
162}

164/// Run the backend's own `Cli::drive` on a spawned task and wait until it
165/// serves HTTP — which `drive` only does after migrating and provisioning.
166///
167/// Every setting that matters is passed as an argument: clap falls back to
168/// the process env for anything omitted, and an explicit argument wins over
169/// a stray `DATABASE_URL`/`PORT` in the caller's environment.
170async fn start_backend(
171    relay_url: &str,
172    inbox: Arc<MemoryInbox>,
173    storage: &tempfile::TempDir,
174) -> eyre::Result<(String, JoinHandle<()>)> {
175    let port = free_tcp_port()?;
176    let wt_port = free_udp_port()?;
177    let mut cli = backend::Cli::try_parse_from([
178        "backend".to_owned(),
179        "--database-url=sqlite::memory:".to_owned(),
180        format!("--host={LOOPBACK}"),
181        format!("--port={port}"),
182        "--quiet".to_owned(),
183        "--environment=dev".to_owned(),
184        "--migrate=fresh".to_owned(),
185        "--secret-key=loopback-secret-key".to_owned(),
186        "--rp-id=localhost".to_owned(),
187        "--rp-origin=http://localhost:3100".to_owned(),
188        format!("--relay-url={relay_url}"),
189        format!("--relay-hmac-secret={RELAY_SECRET}"),
190        "--magic-base-url=http://localhost:3100".to_owned(),
191        "--default-community-name=loopback".to_owned(),
192        "--storage-backend=local".to_owned(),
193        format!("--storage-local-dir={}", storage.path().display()),
194        // The realtime plane's primary leg, as in the dev stack. The URL is
195        // the IPv4 literal: the endpoint binds `--host` only.
196        "--rt-wt-self-signed".to_owned(),
197        format!("--rt-wt-port={wt_port}"),
198        format!("--rt-wt-url=https://{LOOPBACK}:{wt_port}/realtime"),
199    ])?;
200    cli.mailer = Some(inbox);
201
202    let backend_url = format!("http://{LOOPBACK}:{port}");
203    let mut task = tokio::spawn(async move {
204        if let Err(e) = cli.drive().await {
205            tracing::error!("loopback backend exited: {e:?}");
206        }
207    });
208
209    let http = reqwest::Client::new();
210    let deadline = tokio::time::Instant::now() + BACKEND_READY_BUDGET;
211    loop {
212        if task.is_finished() {
213            let _ = (&mut task).await;
214            eyre::bail!("loopback backend exited during startup (see its error above)");
215        }
216        if sfu_nodes(&http, &backend_url).await.is_ok() {
217            return Ok((backend_url, task));
218        }
219        if tokio::time::Instant::now() >= deadline {
220            task.abort();
221            eyre::bail!("loopback backend not serving after {BACKEND_READY_BUDGET:?}");
222        }
223        tokio::time::sleep(READY_POLL).await;
224    }
225}

227/// `POST /internal/sfu/register` — the production SFU's registration body.
228async fn register_sfu(
229    http: &reqwest::Client,
230    backend_url: &str,
231    node_id: &str,
232) -> eyre::Result<()> {
233    http.post(format!("{backend_url}/internal/sfu/register"))
234        .bearer_auth(RELAY_SECRET)
235        .json(&serde_json::json!({
236            "node_id": node_id,
237            "draining": false,
238            "sessions": 0,
239        }))
240        .send()
241        .await?
242        .error_for_status()?;
243    Ok(())
244}
245
246async fn heartbeat(http: reqwest::Client, backend_url: String, node_id: String) {
247    loop {
248        tokio::time::sleep(REGISTER_INTERVAL).await;
249        if let Err(e) = register_sfu(&http, &backend_url, &node_id).await {
250            tracing::warn!("loopback SFU re-registration failed: {e}");
251        }
252    }
253}
254
255async fn sfu_nodes(http: &reqwest::Client, backend_url: &str) -> eyre::Result<SfuNodesResponse> {
256    Ok(http
257        .get(format!("{backend_url}/internal/sfu/nodes"))
258        .bearer_auth(RELAY_SECRET)
259        .send()
260        .await?
261        .error_for_status()?
262        .json()
263        .await?)
264}

266/// A loopback port free at the time of the call. The service binds it
267/// moments later; a collision with another process in that window fails the
268/// bind loudly rather than misrouting.
269fn free_tcp_port() -> std::io::Result<u16> {
270    Ok(std::net::TcpListener::bind((LOOPBACK, 0))?
271        .local_addr()?
272        .port())
273}
274
275fn free_udp_port() -> std::io::Result<u16> {
276    Ok(std::net::UdpSocket::bind((LOOPBACK, 0))?
277        .local_addr()?
278        .port())
