1//! `InCall`: a joined participant's call handle. Every `expect_*` is
2//! deadline-bounded and watch-driven — it polls a `tokio::sync::watch`
3//! channel to a condition (never a blind `sleep`-then-assert), and fails with
4//! the last-observed state on timeout.

18/// Which of a peer's two inbound video streams a per-peer video expectation
19/// reads, and which tile the demand verbs address. Mirrors the `source`
20/// string `HoundClient::call_set_tile_quality` takes (`"camera"` /
21/// `"screen"`) and the camera-vs-screen split in `InboundSnapshot` — the two
22/// are the same axis, so a scenario names it once.
23///
24/// Note both spatial LAYERS of a source (`cam` and `cam_l0`) count into the
25/// SAME snapshot stream: the receive pipeline buckets by source, not by
26/// layer. That is what makes a layer switch observable as a change in the
27/// decoded `width`/`height` gauges on an otherwise continuous
28/// `frames_received` counter.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub enum VideoSource {
31    Camera,
32    Screen,
33}

37    fn as_client_str(self) -> &'static str {
38        match self {
39            Self::Camera => "camera",
40            Self::Screen => "screen",
41        }
42    }

45/// A joined call. Dropping this without calling [`InCall::leave`] leaves the
46/// call running server-side until the participant's `HoundClient` is dropped
47/// or another `join` supersedes it — tests should always `leave()` explicitly.
48///
49/// Holds the `watch_call`/`watch_stats` receivers from *before* the join
50/// (see [`crate::participant::Participant::join`]): a receiver that exists
51/// from the start can never miss a publish, whatever the sender's
52/// `send`-vs-`send_replace` discipline.
53pub struct InCall {
54    participant: Participant,
55    call_rx: watch::Receiver<CallView>,
56    stats_rx: watch::Receiver<StatsSnapshot>,
57}

110    /// Wait for inbound audio frames (summed across every remote peer) to
111    /// grow by at least `min_delta` from the count observed when this call is
112    /// made — the "media is still flowing" half of the mute/unmute assertion.
113    ///
114    /// Returns how long that took, so a scenario can report the measured latency
115    /// next to the deadline that bounds it. As with
116    /// [`Self::expect_video_growth`], the figure carries up to one `call:stats`
117    /// interval (1 s) of quantization — the counters are sampled by a 1 Hz
118    /// emitter — and any budget asserted through `deadline` must include it.
119    pub async fn expect_audio_growth(
120        &self,
121        min_delta: u64,
122        deadline: Option<Duration>,
123    ) -> eyre::Result<Duration> {
124        let mut rx = self.stats_rx.clone();
125        let start = inbound_audio_total(&rx.borrow_and_update());
126        let started = tokio::time::Instant::now();
127        wait_for(
128            deadline,
129            &mut rx,
130            move |s: &StatsSnapshot| inbound_audio_total(s).saturating_sub(start) >= min_delta,
131            move |s| {
132                format!(
133                    "inbound_audio_frames={} (start {start})",
134                    inbound_audio_total(s)
135                )
136            },
137        )
138        .await?;
139        Ok(started.elapsed())
140    }

142    /// Assert that a remote peer's audio QUIESCES without going away: its frame
143    /// rate must fall to at most `max_fps` and stay there for `window`, while
144    /// still delivering something.
145    ///
146    /// This is the assertion a silent peer needs, and it is deliberately not
147    /// [`Self::expect_audio_stops`]. A peer who is merely quiet does not stop:
148    /// the publisher's silence suppression withholds the silence but keeps
149    /// publishing a keepalive roughly twice a second, precisely so the
150    /// subscription stays warm and the receiver's stall watchdog stays fed. So
151    /// this asserts BOTH halves, and both are load-bearing:
152    ///
153    /// * **rate collapsed** — the ~50 frames/s of a talking peer is gone. Without
154    ///   this a suppression gate that never engages passes.
155    /// * **still alive** — the count keeps moving at a keepalive CADENCE, not merely
156    ///   at all. Without a floor a torn-down subscription, a wedged pull or a dead
157    ///   client passes the same gate, which is the exact confusion "no frames" would
158    ///   otherwise create — and `grown > 0` is not enough for that, because one
159    ///   straggler arriving a moment after the window opens satisfies it for a
160    ///   publisher that then dies. See [`MIN_ALIVE_FPS`].
161    ///
162    /// Returns the measured rate (frames/s) over the window.
163    pub async fn expect_audio_quiesces(
164        &self,
165        max_fps: f64,
166        settle: Duration,
167        window: Duration,
168    ) -> eyre::Result<f64> {
169        /// Shortest window over which a ~2/s keepalive cadence is measurable
170        /// against the 1 Hz snapshot at all. Rejected outright rather than
171        /// silently weakened.
172        const MIN_WINDOW: Duration = Duration::from_secs(3);
173        /// Liveness FLOOR: the rate below which the peer is not quiet, it is gone.
174        ///
175        /// Bracketed from both sides. A suppressing publisher delivers ~2/s
176        /// (`policy::KEEPALIVE_INTERVAL`, plus a real mic's comfort-noise refresh on
177        /// top), so 1/s sits comfortably under the real cadence — it will not fail a
178        /// healthy quiet peer whose keepalives bunch across the 1 Hz snapshot grid —
179        /// while being far above the zero a torn-down subscription produces and above
180        /// the handful of stragglers a publisher that dies mid-window can emit. Over
181        /// the 3 s minimum window that is ≥3 frames; over the scenario's 6 s, ≥6.
182        const MIN_ALIVE_FPS: f64 = 1.0;
183        eyre::ensure!(
184            window >= MIN_WINDOW,
185            "expect_audio_quiesces needs a window of at least {MIN_WINDOW:?} to \
186             measure a keepalive cadence against the 1 Hz snapshot, got {window:?}"
187        );
188
189        let mut rx = self.stats_rx.clone();
190        // Phase 1 — settle. Wait (bounded) for a whole stats tick whose growth is
191        // already at or under the target rate; suppression engages asynchronously
192        // (the encoder's own hangover, then the gate's confirmation window), and
193        // frames in flight when it does are legitimately still arriving.
194        let settle_by = tokio::time::Instant::now() + settle;
195        let mut last = inbound_audio_total(&rx.borrow_and_update());
196        let mut last_at = tokio::time::Instant::now();
197        loop {
198            let now = tokio::time::Instant::now();
199            let never_quiesced = || {
200                eyre::eyre!(
201                    "peer audio never quiesced within {settle:?}: still arriving faster \
202                     than {max_fps:.1}/s (total {last}) — the publisher is putting its \
203                     silence on the wire"
204                )
205            };
206            if now >= settle_by {
207                return Err(never_quiesced());
208            }
209            if tokio::time::timeout(settle_by - now, rx.changed())
210                .await
211                .is_err()
212            {
213                return Err(never_quiesced());
214            }
215            let cur = inbound_audio_total(&rx.borrow());
216            let now = tokio::time::Instant::now();
217            let secs = (now - last_at).as_secs_f64().max(f64::EPSILON);
218            let rate = cur.saturating_sub(last) as f64 / secs;
219            last = cur;
220            last_at = now;
221            if rate <= max_fps {
222                break;
223            }
224        }
225
226        // Phase 2 — hold. Measure across the whole window rather than per tick, so
227        // one late-delivered keepalive pair can't fail an otherwise-quiet peer.
228        let start_total = last;
229        let started = tokio::time::Instant::now();
230        let deadline = started + window;
231        let mut ticks = 0u32;
232        loop {
233            let now = tokio::time::Instant::now();
234            if now >= deadline
235                || tokio::time::timeout(deadline - now, rx.changed())
236                    .await
237                    .is_err()
238            {
239                break;
240            }
241            ticks += 1;
242        }
243        eyre::ensure!(
244            ticks > 0,
245            "no stats tick during the {window:?} quiet window — the 1 Hz snapshot went \
246             silent, so the client stopped reporting rather than staying quiet. This is \
247             NOT evidence that audio quiesced."
248        );
249        let grown = inbound_audio_total(&rx.borrow()).saturating_sub(start_total);
250        let elapsed = started.elapsed().as_secs_f64();
251        let rate = grown as f64 / elapsed;
252        eyre::ensure!(
253            rate <= max_fps,
254            "peer audio resumed while the sender was silent: {grown} frames in \
255             {elapsed:.1}s = {rate:.1}/s, over the {max_fps:.1}/s ceiling"
256        );
257        eyre::ensure!(
258            rate >= MIN_ALIVE_FPS,
259            "peer audio did not stay alive: only {grown} frames in {elapsed:.1}s = \
260             {rate:.2}/s, under the {MIN_ALIVE_FPS:.1}/s floor. A suppressing publisher \
261             must keep the subscription warm with keepalives (~2/s); a rate this low \
262             means the track died, the pull wedged, or the client stopped — not that \
263             the sender is quiet."
264        );
265        Ok(rate)
266    }

268    /// The deepest inbound audio jitter buffer across all remote peers right now, as
269    /// `(target_delay_us, recommended_delay_us)`.
270    ///
271    /// The receive-side DEPTH, which no frame-count assertion can see: a peer can be
272    /// delivering perfectly while their audio plays half a second late because the
273    /// adaptive estimator was fed something that looked like transit jitter. `target`
274    /// is what the buffer is running at, `recommended` what the estimator is asking
275    /// for; both should sit at the client's `PLAYOUT_TARGET_US` floor on a healthy
276    /// link, and a `recommended` pinned near the 450 ms adaptive cap is the signature
277    /// of an estimator being fed pipeline delay rather than link jitter.
278    pub fn inbound_audio_jitterbuf_us(&self) -> (u32, u32) {
279        let snap = self.stats_rx.borrow();
280        snap.inbound
281            .iter()
282            .map(|p| (p.audio.target_delay_us, p.audio.recommended_delay_us))
283            .max()
284            .unwrap_or((0, 0))
285    }

287    /// Make this participant's synthetic mic emit digital SILENCE (or resume its
288    /// speech-like signal) — a SOURCE-level toggle, not
289    /// [`mute`](Self::mute).
290    ///
291    /// The distinction is the whole point. Mute gates before the encoder, so a
292    /// muted peer publishes literally nothing and exercises none of the silence
293    /// path. This feeds real zeros into a live encoder, so Opus's VAD marks
294    /// non-speech, DTX collapses the payload, and the publisher's silence
295    /// suppression engages for real — which is the only way an e2e run can
296    /// observe the behaviour a quiet-but-unmuted participant actually produces.
297    /// (The synthetic source is deliberately VAD-active otherwise; see
298    /// `hound_engine`'s `dtx_never_suppresses_the_synthetic_capture_source`.)
299    pub fn set_source_silent(&self, silent: bool) {
300        self.participant.client().set_source_silent(silent);
301    }

303    /// Assert that muting STOPS inbound audio: it must go quiet within
304    /// [`MUTE_SETTLE`], and then stay quiet for `quiet_for`. Watch-driven
305    /// throughout — it wakes on every stats tick rather than blind-sleeping.
306    ///
307    /// ## Why there is a settle phase
308    /// The obvious form of this assertion — sample a baseline, then fail on any
309    /// growth — is not well posed, and it flaked in CI for exactly that reason
310    /// (`frames grew 52 -> 53`, one frame over a 3 s window). The baseline can
311    /// only be read from the 1 Hz stats snapshot the caller already has, which
312    /// is published BEFORE `mute()` is even issued. Everything in the gap counts
313    /// as "growth": the test task waking, the mute reaching the capture thread
314    /// (checked once per 20 ms frame), and the frame already in flight through
315    /// encode → publish → network. At 50 frames/s that gap contains zero or one
316    /// frame essentially at random, so the old assertion was a coin flip.
317    ///
318    /// The product side of this was real and is fixed separately — the mic used
319    /// to gate mute only at capture, leaving up to 8 queued PCM frames (~160 ms)
320    /// to publish after mute; it now gates at the encoder too
321    /// (`client/src/call/audio_tx.rs`). But no product change can drive the
322    /// residual to zero: a frame mid-flight when the flag flips is legitimately
323    /// pre-mute audio.
324    ///
325    /// So the contract asserted here is the one that actually means "mute
326    /// works", and it is not weaker where it matters. A mute that never takes
327    /// effect keeps growing every tick and fails the settle phase; a mute that
328    /// takes effect and then leaks fails the quiet phase. Only the unobservable
329    /// one-frame boundary is forgiven.
330    ///
331    /// ## A silent stats channel is a failure, not a pass
332    ///
333    /// Both phases below distinguish "ticks arrived and the count held" from "no
334    /// tick arrived at all". Only the first is evidence of anything. The stats
335    /// snapshot is published at 1 Hz unconditionally, so a settle or quiet window
336    /// measured in seconds that observes **zero** ticks means the client or its
337    /// stats plumbing has stopped — and treating that as "audio stopped" would let
338    /// a dead client pass the very gate that exists to prove mute works.
339    pub async fn expect_audio_stops(&self, quiet_for: Duration) -> eyre::Result<()> {
340        /// Shortest quiet window that can observe the 1 Hz snapshot often enough for
341        /// "ticks arrived and showed no growth" to mean anything. Below this the
342        /// zero-tick check could not tell a muted client from a dead one, so a
343        /// too-short window is rejected outright rather than silently weakened.
344        const MIN_QUIET: Duration = Duration::from_secs(2);
345        /// How long audio may take to quiesce after `mute()` before we call it
346        /// broken. Mute is asynchronous by construction, and the count is only
347        /// visible at 1 Hz, so settling costs ~2 stats ticks; this is that plus
348        /// margin, and is still far below the "never stops" failure it guards.
349        const MUTE_SETTLE: Duration = Duration::from_secs(4);
350
351        eyre::ensure!(
352            quiet_for >= MIN_QUIET,
353            "expect_audio_stops needs a quiet window of at least {MIN_QUIET:?} to \
354             observe the 1 Hz stats snapshot, got {quiet_for:?}"
355        );
356
357        let mut rx = self.stats_rx.clone();
358
359        // Phase 1 — settle. Advance until a FULL stats tick passes with no new
360        // frames; that is the observable definition of "the stream stopped".
361        let settle_deadline = tokio::time::Instant::now() + MUTE_SETTLE;
362        let mut last = inbound_audio_total(&rx.borrow_and_update());
363        let mut ticks = 0u32;
364        let settled = loop {
365            let now = tokio::time::Instant::now();
366            if now >= settle_deadline {
367                break None;
368            }
369            if tokio::time::timeout(settle_deadline - now, rx.changed())
370                .await
371                .is_err()
372            {
373                // Deadline hit inside the wait. With ticks already seen this is the
374                // same "never stopped" verdict as the loop head; with none it is a
375                // dead stats channel, which the check after the loop separates.
376                break None;
377            }
378            ticks += 1;
379            let current = inbound_audio_total(&rx.borrow());
380            if current == last {
381                break Some(current);
382            }
383            last = current;
384        };
385        if settled.is_none() && ticks == 0 {
386            eyre::bail!(
387                "no stats tick at all within {MUTE_SETTLE:?} of muting — the 1 Hz \
388                 snapshot should have published ~{} times, so the client or its stats \
389                 plumbing is broken. This is NOT evidence that audio stopped.",
390                MUTE_SETTLE.as_secs()
391            );
392        }
393        let Some(start) = settled else {
394            eyre::bail!(
395                "inbound audio never stopped within {MUTE_SETTLE:?} of muting — \
396                 frames were still arriving every stats tick ({ticks} ticks, latest \
397                 total {last}), so mute is not gating the encoder"
398            );
399        };
400
401        // Phase 2 — the quiet window, now anchored AFTER the stream went quiet.
402        // Passing needs ticks that showed no growth; zero ticks proves nothing.
403        let deadline = tokio::time::Instant::now() + quiet_for;
404        let mut quiet_ticks = 0u32;
405        loop {
406            let now = tokio::time::Instant::now();
407            let timed_out = now >= deadline
408                || tokio::time::timeout(deadline - now, rx.changed())
409                    .await
410                    .is_err();
411            if timed_out {
412                if quiet_ticks == 0 {
413                    eyre::bail!(
414                        "no stats tick during the {quiet_for:?} quiet window — the 1 Hz \
415                         snapshot went silent right after audio settled at {start} \
416                         frames, so the client stopped reporting rather than staying \
417                         muted. This is NOT evidence that audio stayed stopped."
418                    );
419                }
420                return Ok(());
421            }
422            quiet_ticks += 1;
423            let current = inbound_audio_total(&rx.borrow());
424            if current > start {
425                eyre::bail!(
426                    "inbound audio resumed while still muted: it went quiet at \
427                     {start} frames, then grew to {current} within {quiet_for:?}"
428                );
429            }
430        }
431    }

433    /// The warmup-baseline audio-health gate — see
434    /// [`hound_client::call_health`] for the shared monitor (the same one
435    /// `houndbot` gates its verdict on). Observes for `window`, anchoring the
436    /// starve-counter baseline once `min_peers` have been present for
437    /// [`hound_client::call_health::WARMUP`], then fails if any starvation
438    /// counter grew past that baseline.
439    pub async fn expect_steady_audio(&self, min_peers: u32, window: Duration) -> eyre::Result<()> {
440        let sample = crate::sample::sample_call(self.participant.client(), min_peers, window).await;
441        if sample.starved {
442            eyre::bail!(
443                "audio starvation detected over {window:?} \
444                 (peers_seen={}, audio_frames_rx={})",
445                sample.peers_seen,
446                sample.audio_frames_rx
447            );
448        }
449        Ok(())
450    }

498    /// After an EXPECTED disruption (e.g. an SFU migration's brief media
499    /// gap), poll for a clean [`Self::expect_steady_audio`] window within
500    /// `within`. Each attempt re-anchors its own warmup baseline from the
501    /// counter totals current at that attempt's start, so a disruption
502    /// caught mid-window doesn't poison the next attempt. The invariant
503    /// under test is BOUNDED recovery, not zero interruption — a single
504    /// zero-tolerance starve gate spanning the disruption itself would
505    /// always fail on an expected, transient blip.
506    pub async fn expect_audio_recovers(
507        &self,
508        min_peers: u32,
509        within: Duration,
510    ) -> eyre::Result<()> {
511        const ATTEMPT_WINDOW: Duration = Duration::from_secs(6);
512        let deadline = tokio::time::Instant::now() + within;
513        loop {
514            match self.expect_steady_audio(min_peers, ATTEMPT_WINDOW).await {
515                Ok(()) => return Ok(()),
516                Err(e) if tokio::time::Instant::now() >= deadline => {
517                    eyre::bail!(
518                        "audio did not recover within {within:?} after the expected disruption: {e}"
519                    );
520                }
521                Err(_) => {} // expected mid-disruption blip — retry with a fresh window
522            }
523        }
524    }

526    /// Wait until the roster names exactly ONE peer besides self, and return
527    /// its `peer_id` — the handle every per-peer expectation and every
528    /// demand verb below needs.
529    ///
530    /// Two-party scenarios only: it fails rather than guess once a second
531    /// remote peer appears, because "the remote peer" is then ambiguous and a
532    /// silently-picked one would make the assertions read on an arbitrary
533    /// stream. Watch-driven on the roster, like [`Self::expect_peer_count`].
534    pub async fn remote_peer(&self, deadline: Option<Duration>) -> eyre::Result<u32> {
535        let mut rx = self.call_rx.clone();
536        wait_for(
537            deadline,
538            &mut rx,
539            |v: &CallView| remote_peers(v).len() == 1,
540            |v| format!("self_id={:?} peers={:?}", v.self_id, v.peers),
541        )
542        .await?;
543        let peers = remote_peers(&rx.borrow());
544        match peers.as_slice() {
545            [only] => Ok(*only),
546            other => eyre::bail!("expected exactly one remote peer, got {other:?}"),
547        }
548    }

550    /// Wait until the roster names exactly `count` peers besides self, and
551    /// return their `peer_id`s **sorted**, so a scenario with more than one
552    /// remote can name them stably across runs.
553    ///
554    /// [`Self::remote_peer`] deliberately refuses to guess past one remote;
555    /// this is the explicit multi-party form, for scenarios whose property is
556    /// about telling two remotes apart (e.g. reporting a tile for one and never
557    /// for the other). Which sorted slot is which USER is not defined — every
558    /// participant in these scenarios publishes the same synthetic media, so a
559    /// scenario must only rely on the peers being distinct.
560    pub async fn remote_peers(
561        &self,
562        count: usize,
563        deadline: Option<Duration>,
564    ) -> eyre::Result<Vec<u32>> {
565        let mut rx = self.call_rx.clone();
566        wait_for(
567            deadline,
568            &mut rx,
569            |v: &CallView| remote_peers(v).len() == count,

571        )
572        .await?;
573        let mut peers = remote_peers(&rx.borrow());
574        peers.sort_unstable();
575        Ok(peers)
576    }

612    /// Wait for `peer`'s inbound `source` video to grow by at least
613    /// `min_delta` frames from the count observed when this call is made —
614    /// the per-peer, per-source video counterpart of
615    /// [`Self::expect_audio_growth`]. Returns how long that took, so a
616    /// scenario can report the measured latency next to the deadline that
617    /// bounds it.
618    ///
619    /// The measured figure carries up to one `call:stats` interval (1 s) of
620    /// quantization: the counters are sampled by a 1 Hz emitter, so growth
621    /// that happened at t+0.1 s first becomes observable at the next tick.
622    /// Any budget asserted through `deadline` must include that second.
623    ///
624    /// Reset-tolerant: the baseline re-anchors DOWNWARD if the counter drops
625    /// (a full re-establish clears the per-peer stats row), so a reconnect
626    /// landing mid-window can't wedge this into a false timeout.
627    pub async fn expect_video_growth(
628        &self,
629        peer: u32,
630        source: VideoSource,
631        min_delta: u64,
632        deadline: Option<Duration>,
633    ) -> eyre::Result<Duration> {
634        let mut rx = self.stats_rx.clone();
635        let base = AtomicU64::new(peer_video_frames(&rx.borrow_and_update(), peer, source));
636        let started = tokio::time::Instant::now();
637        wait_for(
638            deadline,
639            &mut rx,
640            |s: &StatsSnapshot| {
641                let cur = peer_video_frames(s, peer, source);
642                if cur < base.load(Ordering::Relaxed) {
643                    base.store(cur, Ordering::Relaxed); // counters were cleared — re-anchor
644                    return false;
645                }
646                cur - base.load(Ordering::Relaxed) >= min_delta
647            },
648            |s| {
649                format!(
650                    "peer {peer} {source:?} frames_received={} (baseline {}, wanted +{min_delta})",
651                    peer_video_frames(s, peer, source),
652                    base.load(Ordering::Relaxed)
653                )
654            },
655        )
656        .await?;
657        Ok(started.elapsed())
658    }

660    /// The paused half of a pause/resume assertion: `peer`'s inbound `source`
661    /// video must STOP.
662    ///
663    /// Two bounded phases, both watch-driven (it wakes on every stats tick
664    /// rather than blind-sleeping a window and checking once):
665    ///
666    /// 1. **settle** — frames already in flight when the pause request goes
667    ///    out are expected to land, and the teardown is asynchronous, so this
668    ///    first waits (bounded by `settle`) for a whole stats tick in which
669    ///    the counter did not move. Never seeing one is a failure: the pause
670    ///    did not take effect.
671    /// 2. **quiet** — with the counter anchored at that settled value, video
672    ///    must stay stopped for `quiet_for`.
673    ///
674    /// Anchoring the baseline *after* the settle rather than at call time is
675    /// what keeps this from flaking on in-flight frames while still failing
676    /// loudly on a pause that never actually stops the stream.
677    pub async fn expect_no_video_growth(
678        &self,
679        peer: u32,
680        source: VideoSource,
681        settle: Duration,
682        quiet_for: Duration,
683    ) -> eyre::Result<()> {
684        let mut rx = self.stats_rx.clone();
685        let mut last = peer_video_frames(&rx.borrow_and_update(), peer, source);
686        // Phase 1: bounded settle.
687        let settle_by = tokio::time::Instant::now() + settle;
688        loop {
689            let now = tokio::time::Instant::now();
690            let still_arriving = || {
691                eyre::eyre!(
692                    "peer {peer} {source:?} video never stopped within {settle:?} of the pause \
693                     — frames_received still climbing (last {last})"
694                )
695            };
696            if now >= settle_by {
697                return Err(still_arriving());
698            }
699            if tokio::time::timeout(settle_by - now, rx.changed())
700                .await
701                .is_err()
702            {
703                return Err(still_arriving());
704            }
705            let cur = peer_video_frames(&rx.borrow(), peer, source);
706            if cur <= last {
707                last = cur;
708                break; // a whole tick with no new frames — the pause landed
709            }
710            last = cur;
711        }
712        // Phase 2: it must STAY stopped.
713        let quiet_by = tokio::time::Instant::now() + quiet_for;
714        loop {
715            let now = tokio::time::Instant::now();
716            if now >= quiet_by {
717                return Ok(());
718            }
719            if tokio::time::timeout(quiet_by - now, rx.changed())
720                .await
721                .is_err()
722            {
723                return Ok(());
724            }
725            let cur = peer_video_frames(&rx.borrow(), peer, source);
726            if cur > last {
727                eyre::bail!(
728                    "expected no inbound video from peer {peer} {source:?} while paused, \
729                     but frames grew {last} -> {cur} within {quiet_for:?}"
730                );
731            }
732        }
733    }

736    /// least `min_width` wide — i.e. the receiver is on the SOURCE spatial
737    /// layer (`cam` / `screen`). Returns the elapsed time so a scenario can
738    /// report the measured layer-switch latency; the `deadline` is the
739    /// budget. See [`Self::expect_video_growth`] on the 1 Hz quantization

761    pub async fn expect_inbound_width_at_most(
762        &self,

764        source: VideoSource,
765        max_width: u32,
766        deadline: Option<Duration>,
767    ) -> eyre::Result<Duration> {

769            peer,

778    /// Shared core of the geometry expectations. A width of 0 (no frame
779    /// decoded yet for this peer/source, or the row is absent entirely) NEVER
780    /// satisfies the predicate — otherwise an at-most bound would pass
781    /// vacuously on a stream that is simply dead, which is the exact failure
782    /// these scenarios exist to catch.
783    async fn expect_inbound_width(
784        &self,
785        peer: u32,
786        source: VideoSource,
787        pred: impl Fn(u32) -> bool,
788        want: &str,
789        deadline: Option<Duration>,
790    ) -> eyre::Result<Duration> {
791        let mut rx = self.stats_rx.clone();
792        let started = tokio::time::Instant::now();
793        wait_for(
794            deadline,
795            &mut rx,
796            |s: &StatsSnapshot| {
797                let w = peer_video(s, peer, source).map_or(0, |v| v.width);
798                w != 0 && pred(w)
799            },
800            |s| {
801                let v = peer_video(s, peer, source);
802                format!(
803                    "peer {peer} {source:?} decoded {}x{} (wanted width {want}), \
804                     frames_received={} decoded_fps={:.1}",
805                    v.map_or(0, |v| v.width),
806                    v.map_or(0, |v| v.height),
807                    v.map_or(0, |v| v.frames_received),
808                    v.map_or(0.0, |v| v.decoded_fps),
809                )
810            },
811        )
812        .await?;
813        Ok(started.elapsed())
814    }

842/// Everyone in the roster except this client. Empty until the server has
843/// assigned our own `self_id` (a roster we can't subtract ourselves from
844/// tells us nothing about who the *remote* peers are).
845fn remote_peers(v: &CallView) -> Vec<u32> {
846    v.self_id.map_or_else(Vec::new, |me| {
847        v.peers.iter().copied().filter(|p| *p != me).collect()
848    })
849}

851/// One peer's inbound stream for `source`, or `None` when that peer has no
852/// inbound row yet (rows are created lazily by the first pipeline).
853fn peer_video(s: &StatsSnapshot, peer: u32, source: VideoSource) -> Option<&InboundVideoSnapshot> {
854    s.inbound
855        .iter()
856        .find(|p| p.peer_id == peer)
857        .map(|p| match source {
858            VideoSource::Camera => &p.video,
859            VideoSource::Screen => &p.screen_video,
860        })
861}

875/// Shared polling core for a watch-driven `expect_*`: check `pred` on every
876/// change (and once up front), bail with `describe`'s last-observed state on
877/// timeout. `deadline` defaults to [`DEFAULT_DEADLINE`].
878async fn wait_for<T>(
879    deadline: Option<Duration>,
880    rx: &mut watch::Receiver<T>,
881    pred: impl Fn(&T) -> bool,
882    describe: impl Fn(&T) -> String,
883) -> eyre::Result<()> {
884    let deadline = deadline.unwrap_or(DEFAULT_DEADLINE);
885    let wait = async {
886        loop {
887            if pred(&rx.borrow_and_update()) {
888                return;
889            }
890            if rx.changed().await.is_err() {
891                return; // sender dropped — the final check below settles it
892            }
893        }
894    };
895    if tokio::time::timeout(deadline, wait).await.is_ok() && pred(&rx.borrow()) {
896        return Ok(());
897    }
898    eyre::bail!(
899        "timed out after {deadline:?} waiting for condition; last observed: {}",
900        describe(&rx.borrow())
901    )
902}
