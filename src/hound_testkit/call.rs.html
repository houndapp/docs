1//! `InCall`: a joined participant's call handle. Every `expect_*` is
2//! deadline-bounded and watch-driven — it polls a `tokio::sync::watch`
3//! channel to a condition (never a blind `sleep`-then-assert), and fails with
4//! the last-observed state on timeout.
5
6use std::sync::atomic::{AtomicU64, Ordering};
7use std::time::Duration;

15/// Default deadline for an `expect_*` call when the caller passes `None`.
16const DEFAULT_DEADLINE: Duration = Duration::from_secs(20);

18/// Which of a peer's two inbound video streams a per-peer video expectation
19/// reads, and which tile the demand verbs address. Mirrors the `source`
20/// string `HoundClient::call_set_tile_quality` takes (`"camera"` /
21/// `"screen"`) and the camera-vs-screen split in `InboundSnapshot` — the two
22/// are the same axis, so a scenario names it once.
23///
24/// Note both spatial LAYERS of a source (`cam` and `cam_l0`) count into the
25/// SAME snapshot stream: the receive pipeline buckets by source, not by
26/// layer. That is what makes a layer switch observable as a change in the
27/// decoded `width`/`height` gauges on an otherwise continuous
28/// `frames_received` counter.
29#[derive(Clone, Copy, Debug, PartialEq, Eq)]
30pub enum VideoSource {
31    Camera,
32    Screen,
33}
34
35impl VideoSource {
36    /// The wire string `HoundClient::call_set_tile_quality` matches on.
37    fn as_client_str(self) -> &'static str {
38        match self {
39            Self::Camera => "camera",
40            Self::Screen => "screen",
41        }
42    }
43}

45/// A joined call. Dropping this without calling [`InCall::leave`] leaves the
46/// call running server-side until the participant's `HoundClient` is dropped
47/// or another `join` supersedes it — tests should always `leave()` explicitly.
48///
49/// Holds the `watch_call`/`watch_stats` receivers from *before* the join
50/// (see [`crate::participant::Participant::join`]): a receiver that exists
51/// from the start can never miss a publish, whatever the sender's
52/// `send`-vs-`send_replace` discipline.
53pub struct InCall {
54    participant: Participant,
55    call_rx: watch::Receiver<CallView>,
56    stats_rx: watch::Receiver<StatsSnapshot>,
57}

110    /// Wait for inbound audio frames (summed across every remote peer) to
111    /// grow by at least `min_delta` from the count observed when this call is
112    /// made — the "media is still flowing" half of the mute/unmute assertion.
113    pub async fn expect_audio_growth(
114        &self,
115        min_delta: u64,
116        deadline: Option<Duration>,
117    ) -> eyre::Result<()> {
118        let mut rx = self.stats_rx.clone();
119        let start = inbound_audio_total(&rx.borrow_and_update());
120        wait_for(
121            deadline,
122            &mut rx,
123            move |s: &StatsSnapshot| inbound_audio_total(s).saturating_sub(start) >= min_delta,
124            move |s| {
125                format!(
126                    "inbound_audio_frames={} (start {start})",
127                    inbound_audio_total(s)
128                )
129            },
130        )
131        .await
132    }

134    /// Wait `quiet_for`, asserting inbound audio frames never grew — the
135    /// "muted means silence" half of the mute assertion. A negative
136    /// expectation, but still watch-driven: it wakes on every stats tick
137    /// rather than blind-sleeping the whole window before checking once.
138    pub async fn expect_no_audio_growth(&self, quiet_for: Duration) -> eyre::Result<()> {
139        let mut rx = self.stats_rx.clone();
140        let start = inbound_audio_total(&rx.borrow_and_update());
141        let deadline = tokio::time::Instant::now() + quiet_for;
142        loop {
143            let now = tokio::time::Instant::now();
144            if now >= deadline {
145                return Ok(());
146            }
147            if tokio::time::timeout(deadline - now, rx.changed())
148                .await
149                .is_err()
150            {
151                return Ok(());
152            }
153            let current = inbound_audio_total(&rx.borrow());
154            if current > start {
155                eyre::bail!(
156                    "expected no inbound audio growth for {quiet_for:?} while muted, \
157                     but frames grew {start} -> {current}"
158                );
159            }
160        }
161    }

163    /// The warmup-baseline audio-health gate — see
164    /// [`hound_client::call_health`] for the shared monitor (the same one
165    /// `houndbot` gates its verdict on). Observes for `window`, anchoring the
166    /// starve-counter baseline once `min_peers` have been present for
167    /// [`hound_client::call_health::WARMUP`], then fails if any starvation
168    /// counter grew past that baseline.
169    pub async fn expect_steady_audio(&self, min_peers: u32, window: Duration) -> eyre::Result<()> {
170        let sample = crate::sample::sample_call(self.participant.client(), min_peers, window).await;
171        if sample.starved {
172            eyre::bail!(
173                "audio starvation detected over {window:?} \
174                 (peers_seen={}, audio_frames_rx={})",
175                sample.peers_seen,
176                sample.audio_frames_rx
177            );
178        }
179        Ok(())
180    }

228    /// After an EXPECTED disruption (e.g. an SFU migration's brief media
229    /// gap), poll for a clean [`Self::expect_steady_audio`] window within
230    /// `within`. Each attempt re-anchors its own warmup baseline from the
231    /// counter totals current at that attempt's start, so a disruption
232    /// caught mid-window doesn't poison the next attempt. The invariant
233    /// under test is BOUNDED recovery, not zero interruption — a single
234    /// zero-tolerance starve gate spanning the disruption itself would
235    /// always fail on an expected, transient blip.
236    pub async fn expect_audio_recovers(
237        &self,
238        min_peers: u32,
239        within: Duration,
240    ) -> eyre::Result<()> {
241        const ATTEMPT_WINDOW: Duration = Duration::from_secs(6);
242        let deadline = tokio::time::Instant::now() + within;
243        loop {
244            match self.expect_steady_audio(min_peers, ATTEMPT_WINDOW).await {
245                Ok(()) => return Ok(()),
246                Err(e) if tokio::time::Instant::now() >= deadline => {
247                    eyre::bail!(
248                        "audio did not recover within {within:?} after the expected disruption: {e}"
249                    );
250                }
251                Err(_) => {} // expected mid-disruption blip — retry with a fresh window
252            }
253        }
254    }

256    /// Wait until the roster names exactly ONE peer besides self, and return
257    /// its `peer_id` — the handle every per-peer expectation and every
258    /// demand verb below needs.
259    ///
260    /// Two-party scenarios only: it fails rather than guess once a second
261    /// remote peer appears, because "the remote peer" is then ambiguous and a
262    /// silently-picked one would make the assertions read on an arbitrary
263    /// stream. Watch-driven on the roster, like [`Self::expect_peer_count`].
264    pub async fn remote_peer(&self, deadline: Option<Duration>) -> eyre::Result<u32> {
265        let mut rx = self.call_rx.clone();
266        wait_for(
267            deadline,
268            &mut rx,
269            |v: &CallView| remote_peers(v).len() == 1,
270            |v| format!("self_id={:?} peers={:?}", v.self_id, v.peers),
271        )
272        .await?;
273        let peers = remote_peers(&rx.borrow());
274        match peers.as_slice() {
275            [only] => Ok(*only),
276            other => eyre::bail!("expected exactly one remote peer, got {other:?}"),
277        }
278    }

314    /// Wait for `peer`'s inbound `source` video to grow by at least
315    /// `min_delta` frames from the count observed when this call is made —
316    /// the per-peer, per-source video counterpart of
317    /// [`Self::expect_audio_growth`]. Returns how long that took, so a
318    /// scenario can report the measured latency next to the deadline that
319    /// bounds it.
320    ///
321    /// The measured figure carries up to one `call:stats` interval (1 s) of
322    /// quantization: the counters are sampled by a 1 Hz emitter, so growth
323    /// that happened at t+0.1 s first becomes observable at the next tick.
324    /// Any budget asserted through `deadline` must include that second.
325    ///
326    /// Reset-tolerant: the baseline re-anchors DOWNWARD if the counter drops
327    /// (a full re-establish clears the per-peer stats row), so a reconnect
328    /// landing mid-window can't wedge this into a false timeout.
329    pub async fn expect_video_growth(
330        &self,
331        peer: u32,
332        source: VideoSource,
333        min_delta: u64,
334        deadline: Option<Duration>,
335    ) -> eyre::Result<Duration> {
336        let mut rx = self.stats_rx.clone();
337        let base = AtomicU64::new(peer_video_frames(&rx.borrow_and_update(), peer, source));
338        let started = tokio::time::Instant::now();
339        wait_for(
340            deadline,
341            &mut rx,
342            |s: &StatsSnapshot| {
343                let cur = peer_video_frames(s, peer, source);
344                if cur < base.load(Ordering::Relaxed) {
345                    base.store(cur, Ordering::Relaxed); // counters were cleared — re-anchor
346                    return false;
347                }
348                cur - base.load(Ordering::Relaxed) >= min_delta
349            },
350            |s| {
351                format!(
352                    "peer {peer} {source:?} frames_received={} (baseline {}, wanted +{min_delta})",
353                    peer_video_frames(s, peer, source),
354                    base.load(Ordering::Relaxed)
355                )
356            },
357        )
358        .await?;
359        Ok(started.elapsed())
360    }

362    /// The paused half of a pause/resume assertion: `peer`'s inbound `source`
363    /// video must STOP.
364    ///
365    /// Two bounded phases, both watch-driven (it wakes on every stats tick
366    /// rather than blind-sleeping a window and checking once):
367    ///
368    /// 1. **settle** — frames already in flight when the pause request goes
369    ///    out are expected to land, and the teardown is asynchronous, so this
370    ///    first waits (bounded by `settle`) for a whole stats tick in which
371    ///    the counter did not move. Never seeing one is a failure: the pause
372    ///    did not take effect.
373    /// 2. **quiet** — with the counter anchored at that settled value, video
374    ///    must stay stopped for `quiet_for`.
375    ///
376    /// Anchoring the baseline *after* the settle rather than at call time is
377    /// what keeps this from flaking on in-flight frames while still failing
378    /// loudly on a pause that never actually stops the stream.
379    pub async fn expect_no_video_growth(
380        &self,
381        peer: u32,
382        source: VideoSource,
383        settle: Duration,
384        quiet_for: Duration,
385    ) -> eyre::Result<()> {
386        let mut rx = self.stats_rx.clone();
387        let mut last = peer_video_frames(&rx.borrow_and_update(), peer, source);
388        // Phase 1: bounded settle.
389        let settle_by = tokio::time::Instant::now() + settle;
390        loop {
391            let now = tokio::time::Instant::now();
392            let still_arriving = || {
393                eyre::eyre!(
394                    "peer {peer} {source:?} video never stopped within {settle:?} of the pause \
395                     — frames_received still climbing (last {last})"
396                )
397            };
398            if now >= settle_by {
399                return Err(still_arriving());
400            }
401            if tokio::time::timeout(settle_by - now, rx.changed())
402                .await
403                .is_err()
404            {
405                return Err(still_arriving());
406            }
407            let cur = peer_video_frames(&rx.borrow(), peer, source);
408            if cur <= last {
409                last = cur;
410                break; // a whole tick with no new frames — the pause landed
411            }
412            last = cur;
413        }
414        // Phase 2: it must STAY stopped.
415        let quiet_by = tokio::time::Instant::now() + quiet_for;
416        loop {
417            let now = tokio::time::Instant::now();
418            if now >= quiet_by {
419                return Ok(());
420            }
421            if tokio::time::timeout(quiet_by - now, rx.changed())
422                .await
423                .is_err()
424            {
425                return Ok(());
426            }
427            let cur = peer_video_frames(&rx.borrow(), peer, source);
428            if cur > last {
429                eyre::bail!(
430                    "expected no inbound video from peer {peer} {source:?} while paused, \
431                     but frames grew {last} -> {cur} within {quiet_for:?}"
432                );
433            }
434        }
435    }

480    /// Shared core of the geometry expectations. A width of 0 (no frame
481    /// decoded yet for this peer/source, or the row is absent entirely) NEVER
482    /// satisfies the predicate — otherwise an at-most bound would pass
483    /// vacuously on a stream that is simply dead, which is the exact failure
484    /// these scenarios exist to catch.
485    async fn expect_inbound_width(
486        &self,
487        peer: u32,
488        source: VideoSource,
489        pred: impl Fn(u32) -> bool,
490        want: &str,
491        deadline: Option<Duration>,
492    ) -> eyre::Result<Duration> {
493        let mut rx = self.stats_rx.clone();
494        let started = tokio::time::Instant::now();
495        wait_for(
496            deadline,
497            &mut rx,
498            |s: &StatsSnapshot| {
499                let w = peer_video(s, peer, source).map_or(0, |v| v.width);
500                w != 0 && pred(w)
501            },
502            |s| {
503                let v = peer_video(s, peer, source);
504                format!(
505                    "peer {peer} {source:?} decoded {}x{} (wanted width {want}), \
506                     frames_received={} decoded_fps={:.1}",
507                    v.map_or(0, |v| v.width),
508                    v.map_or(0, |v| v.height),
509                    v.map_or(0, |v| v.frames_received),
510                    v.map_or(0.0, |v| v.decoded_fps),
511                )
512            },
513        )
514        .await?;
515        Ok(started.elapsed())
516    }

544/// Everyone in the roster except this client. Empty until the server has
545/// assigned our own `self_id` (a roster we can't subtract ourselves from
546/// tells us nothing about who the *remote* peers are).
547fn remote_peers(v: &CallView) -> Vec<u32> {
548    v.self_id.map_or_else(Vec::new, |me| {
549        v.peers.iter().copied().filter(|p| *p != me).collect()
550    })
551}

553/// One peer's inbound stream for `source`, or `None` when that peer has no
554/// inbound row yet (rows are created lazily by the first pipeline).
555fn peer_video(s: &StatsSnapshot, peer: u32, source: VideoSource) -> Option<&InboundVideoSnapshot> {
556    s.inbound
557        .iter()
558        .find(|p| p.peer_id == peer)
559        .map(|p| match source {
560            VideoSource::Camera => &p.video,
561            VideoSource::Screen => &p.screen_video,
562        })
563}
564
565fn peer_video_frames(s: &StatsSnapshot, peer: u32, source: VideoSource) -> u64 {
566    peer_video(s, peer, source).map_or(0, |v| v.frames_received)
567}
568
569fn inbound_video_total(s: &StatsSnapshot) -> u64 {
570    s.inbound.iter().map(|p| p.video.frames_received).sum()
571}
572
573fn inbound_audio_total(s: &StatsSnapshot) -> u64 {
574    s.inbound.iter().map(|p| p.audio.frames_received).sum()
575}

577/// Shared polling core for a watch-driven `expect_*`: check `pred` on every
578/// change (and once up front), bail with `describe`'s last-observed state on
579/// timeout. `deadline` defaults to [`DEFAULT_DEADLINE`].
580async fn wait_for<T>(
581    deadline: Option<Duration>,
582    rx: &mut watch::Receiver<T>,
583    pred: impl Fn(&T) -> bool,
584    describe: impl Fn(&T) -> String,
585) -> eyre::Result<()> {
586    let deadline = deadline.unwrap_or(DEFAULT_DEADLINE);
587    let wait = async {
588        loop {
589            if pred(&rx.borrow_and_update()) {
590                return;
591            }
592            if rx.changed().await.is_err() {
593                return; // sender dropped — the final check below settles it
594            }
595        }
596    };
597    if tokio::time::timeout(deadline, wait).await.is_ok() && pred(&rx.borrow()) {
598        return Ok(());
599    }
600    eyre::bail!(
601        "timed out after {deadline:?} waiting for condition; last observed: {}",
602        describe(&rx.borrow())
603    )
