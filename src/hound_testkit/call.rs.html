1//! `InCall`: a joined participant's call handle. Every `expect_*` is
2//! deadline-bounded and watch-driven — it polls a `tokio::sync::watch`
3//! channel to a condition (never a blind `sleep`-then-assert), and fails with
4//! the last-observed state on timeout.
5//!
6//! ## Run scoping
7//!
8//! The dev stack is persistent and scenarios share one voice space, and the
9//! backend keeps a disconnected peer's voice presence for an eviction grace —
10//! so a roster can hold peers this scenario never created (the previous run's,
11//! a houndbot's). Every roster count and every summed inbound counter here
12//! therefore reads only the REMOTE peers whose user this run's [`crate::Env`]
13//! signed in, joined through `CallView::participants` (peer id → user id). A
14//! peer whose identity has not arrived yet is not counted until it does; the
15//! waits absorb that like any other propagation delay.
16

30/// Which of a peer's two inbound video streams a per-peer video expectation
31/// reads, and which tile the demand verbs address. Mirrors the `source`
32/// string `HoundClient::call_set_tile_quality` takes (`"camera"` /
33/// `"screen"`) and the camera-vs-screen split in `InboundSnapshot` — the two
34/// are the same axis, so a scenario names it once.
35///
36/// Note both spatial LAYERS of a source (`cam` and `cam_l0`) count into the
37/// SAME snapshot stream: the receive pipeline buckets by source, not by
38/// layer. That is what makes a layer switch observable as a change in the
39/// decoded `width`/`height` gauges on an otherwise continuous
40/// `frames_received` counter.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub enum VideoSource {
43    Camera,
44    Screen,
45}

50        match self {
51            Self::Camera => "camera",
52            Self::Screen => "screen",
53        }
54    }

57/// A joined call. Dropping this without calling [`InCall::leave`] leaves the
58/// call running server-side until the participant's `HoundClient` is dropped
59/// or another `join` supersedes it — tests should always `leave()` explicitly.
60///
61/// Holds the `watch_call`/`watch_stats` receivers from *before* the join
62/// (see [`crate::participant::Participant::join`]): a receiver that exists
63/// from the start can never miss a publish, whatever the sender's
64/// `send`-vs-`send_replace` discipline.
65pub struct InCall {
66    participant: Participant,
67    call_rx: watch::Receiver<CallView>,
68    stats_rx: watch::Receiver<StatsSnapshot>,
69}

88    /// This run's remote peers in `v` — see the module doc's "Run scoping".
89    fn run_remotes(&self, v: &CallView) -> HashSet<u32> {
90        let users = self.participant.run_users();
91        v.participants
92            .iter()
93            .filter(|p| Some(p.peer_id) != v.self_id && users.contains(&p.user_id))
94            .map(|p| p.peer_id)
95            .collect()
96    }

98    /// [`Self::run_remotes`] of the roster as it stands now — what the
99    /// stats-driven expectations sum over.
100    fn run_remotes_now(&self) -> HashSet<u32> {
101        self.run_remotes(&self.call_rx.borrow())
102    }

104    /// The peer set a windowed audio expectation sums over, fixed at its start:
105    /// a peer entering the sum mid-window would count its whole history as
106    /// growth. Empty is an error, not a vacuous window — call these once the
107    /// peers are in (`expect_peer_count` / `peer_of`).
108    fn window_peers(&self, what: &str) -> eyre::Result<HashSet<u32>> {
109        let peers = self.run_remotes_now();
110        eyre::ensure!(
111            !peers.is_empty(),
112            "{what}: none of this run's peers is in the call yet (roster {:?}) — \
113             wait for them first",
114            self.call_rx.borrow().participants
115        );
116        Ok(peers)
117    }

119    /// Wait until the call has at least `n` of this run's peers (including
120    /// self). Foreign peers in the room never count toward `n`.
121    pub async fn expect_peer_count(
122        &self,
123        n: usize,
124        deadline: Option<Duration>,
125    ) -> eyre::Result<()> {
126        let mut rx = self.call_rx.clone();
127        wait_for(
128            deadline,
129            &mut rx,
130            |v: &CallView| usize::from(v.self_id.is_some()) + self.run_remotes(v).len() >= n,
131            |v| {
132                format!(
133                    "self_id={:?} run_remotes={:?} participants={:?} peers={:?}",
134                    v.self_id,
135                    self.run_remotes(v),
136                    v.participants,
137                    v.peers
138                )
139            },
140        )
141        .await
142    }

144    /// Wait until `other` is in the call, and return their `peer_id` — the
145    /// handle every per-peer expectation and demand verb takes. Selected by
146    /// user id, so it is unambiguous whoever else shares the room.
147    pub async fn peer_of(
148        &self,
149        other: &Participant,
150        deadline: Option<Duration>,
151    ) -> eyre::Result<u32> {
152        let user = other.user_id();
153        let find = |v: &CallView| {
154            v.participants
155                .iter()
156                .find(|p| p.user_id == user && v.peers.contains(&p.peer_id))
157                .map(|p| p.peer_id)
158        };
159        let mut rx = self.call_rx.clone();
160        wait_for(
161            deadline,
162            &mut rx,
163            |v: &CallView| find(v).is_some(),
164            |v| {
165                format!(
166                    "{} ({user}) not in participants={:?}",
167                    other.name(),
168                    v.participants
169                )
170            },
171        )
172        .await?;
173        find(&rx.borrow()).ok_or_else(|| eyre::eyre!("{} left the call again", other.name()))
174    }

225    /// Wait for inbound audio frames (summed across this run's remote peers) to
226    /// grow by at least `min_delta` from the count observed when this call is
227    /// made — the "media is still flowing" half of the mute/unmute assertion.
228    ///
229    /// Returns how long that took, so a scenario can report the measured latency
230    /// next to the deadline that bounds it. As with
231    /// [`Self::expect_video_growth`], the figure carries up to one `call:stats`
232    /// interval (1 s) of quantization — the counters are sampled by a 1 Hz
233    /// emitter — and any budget asserted through `deadline` must include it.
234    pub async fn expect_audio_growth(
235        &self,
236        min_delta: u64,
237        deadline: Option<Duration>,
238    ) -> eyre::Result<Duration> {
239        let peers = self.window_peers("expect_audio_growth")?;
240        let mut rx = self.stats_rx.clone();
241        let start = inbound_audio_total(&rx.borrow_and_update(), &peers);
242        let started = tokio::time::Instant::now();
243        wait_for(
244            deadline,
245            &mut rx,
246            |s: &StatsSnapshot| inbound_audio_total(s, &peers).saturating_sub(start) >= min_delta,
247            |s| {
248                format!(
249                    "inbound_audio_frames={} from run peers {peers:?} (start {start})",
250                    inbound_audio_total(s, &peers)
251                )
252            },
253        )
254        .await?;
255        Ok(started.elapsed())
256    }

258    /// Assert that a remote peer's audio QUIESCES without going away: its frame
259    /// rate must fall to at most `max_fps` and stay there for `window`, while
260    /// still delivering something.
261    ///
262    /// This is the assertion a silent peer needs, and it is deliberately not
263    /// [`Self::expect_audio_stops`]. A peer who is merely quiet does not stop:
264    /// the publisher's silence suppression withholds the silence but keeps
265    /// publishing a keepalive roughly twice a second, precisely so the
266    /// subscription stays warm and the receiver's stall watchdog stays fed. So
267    /// this asserts BOTH halves, and both are load-bearing:
268    ///
269    /// * **rate collapsed** — the ~50 frames/s of a talking peer is gone. Without
270    ///   this a suppression gate that never engages passes.
271    /// * **still alive** — the count keeps moving at a keepalive CADENCE, not merely
272    ///   at all. Without a floor a torn-down subscription, a wedged pull or a dead
273    ///   client passes the same gate, which is the exact confusion "no frames" would
274    ///   otherwise create — and `grown > 0` is not enough for that, because one
275    ///   straggler arriving a moment after the window opens satisfies it for a
276    ///   publisher that then dies. See [`MIN_ALIVE_FPS`].
277    ///
278    /// Returns the measured rate (frames/s) over the window.
279    pub async fn expect_audio_quiesces(
280        &self,
281        max_fps: f64,
282        settle: Duration,
283        window: Duration,
284    ) -> eyre::Result<f64> {
285        /// Shortest window over which a ~2/s keepalive cadence is measurable
286        /// against the 1 Hz snapshot at all. Rejected outright rather than
287        /// silently weakened.
288        const MIN_WINDOW: Duration = Duration::from_secs(3);
289        /// Liveness FLOOR: the rate below which the peer is not quiet, it is gone.
290        ///
291        /// Bracketed from both sides. A suppressing publisher delivers ~2/s
292        /// (`policy::KEEPALIVE_INTERVAL`, plus a real mic's comfort-noise refresh on
293        /// top), so 1/s sits comfortably under the real cadence — it will not fail a
294        /// healthy quiet peer whose keepalives bunch across the 1 Hz snapshot grid —
295        /// while being far above the zero a torn-down subscription produces and above
296        /// the handful of stragglers a publisher that dies mid-window can emit. Over
297        /// the 3 s minimum window that is ≥3 frames; over the scenario's 6 s, ≥6.
298        const MIN_ALIVE_FPS: f64 = 1.0;
299        eyre::ensure!(
300            window >= MIN_WINDOW,
301            "expect_audio_quiesces needs a window of at least {MIN_WINDOW:?} to \
302             measure a keepalive cadence against the 1 Hz snapshot, got {window:?}"
303        );
304
305        let peers = self.window_peers("expect_audio_quiesces")?;
306        let mut rx = self.stats_rx.clone();
307        // Phase 1 — settle. Wait (bounded) for a whole stats tick whose growth is
308        // already at or under the target rate; suppression engages asynchronously
309        // (the encoder's own hangover, then the gate's confirmation window), and
310        // frames in flight when it does are legitimately still arriving.
311        let settle_by = tokio::time::Instant::now() + settle;
312        let mut last = inbound_audio_total(&rx.borrow_and_update(), &peers);
313        let mut last_at = tokio::time::Instant::now();
314        loop {
315            let now = tokio::time::Instant::now();
316            let never_quiesced = || {
317                eyre::eyre!(
318                    "peer audio never quiesced within {settle:?}: still arriving faster \
319                     than {max_fps:.1}/s (total {last}) — the publisher is putting its \
320                     silence on the wire"
321                )
322            };
323            if now >= settle_by {
324                return Err(never_quiesced());
325            }
326            if tokio::time::timeout(settle_by - now, rx.changed())
327                .await
328                .is_err()
329            {
330                return Err(never_quiesced());
331            }
332            let cur = inbound_audio_total(&rx.borrow(), &peers);
333            let now = tokio::time::Instant::now();
334            let secs = (now - last_at).as_secs_f64().max(f64::EPSILON);
335            let rate = cur.saturating_sub(last) as f64 / secs;
336            last = cur;
337            last_at = now;
338            if rate <= max_fps {
339                break;
340            }
341        }
342
343        // Phase 2 — hold. Measure across the whole window rather than per tick, so
344        // one late-delivered keepalive pair can't fail an otherwise-quiet peer.
345        let start_total = last;
346        let started = tokio::time::Instant::now();
347        let deadline = started + window;
348        let mut ticks = 0u32;
349        loop {
350            let now = tokio::time::Instant::now();
351            if now >= deadline
352                || tokio::time::timeout(deadline - now, rx.changed())
353                    .await
354                    .is_err()
355            {
356                break;
357            }
358            ticks += 1;
359        }
360        eyre::ensure!(
361            ticks > 0,
362            "no stats tick during the {window:?} quiet window — the 1 Hz snapshot went \
363             silent, so the client stopped reporting rather than staying quiet. This is \
364             NOT evidence that audio quiesced."
365        );
366        let grown = inbound_audio_total(&rx.borrow(), &peers).saturating_sub(start_total);
367        let elapsed = started.elapsed().as_secs_f64();
368        let rate = grown as f64 / elapsed;
369        eyre::ensure!(
370            rate <= max_fps,
371            "peer audio resumed while the sender was silent: {grown} frames in \
372             {elapsed:.1}s = {rate:.1}/s, over the {max_fps:.1}/s ceiling"
373        );
374        eyre::ensure!(
375            rate >= MIN_ALIVE_FPS,
376            "peer audio did not stay alive: only {grown} frames in {elapsed:.1}s = \
377             {rate:.2}/s, under the {MIN_ALIVE_FPS:.1}/s floor. A suppressing publisher \
378             must keep the subscription warm with keepalives (~2/s); a rate this low \
379             means the track died, the pull wedged, or the client stopped — not that \
380             the sender is quiet."
381        );
382        Ok(rate)
383    }

385    /// The deepest inbound audio jitter buffer across this run's remote peers right
386    /// now, as `(target_delay_us, recommended_delay_us)`.
387    ///
388    /// The receive-side DEPTH, which no frame-count assertion can see: a peer can be
389    /// delivering perfectly while their audio plays half a second late because the
390    /// adaptive estimator was fed something that looked like transit jitter. `target`
391    /// is what the buffer is running at, `recommended` what the estimator is asking
392    /// for; both should sit at the client's `PLAYOUT_TARGET_US` floor on a healthy
393    /// link, and a `recommended` pinned near the 450 ms adaptive cap is the signature
394    /// of an estimator being fed pipeline delay rather than link jitter.
395    pub fn inbound_audio_jitterbuf_us(&self) -> (u32, u32) {
396        let peers = self.run_remotes_now();
397        let snap = self.stats_rx.borrow();
398        snap.inbound
399            .iter()
400            .filter(|p| peers.contains(&p.peer_id))
401            .map(|p| (p.audio.target_delay_us, p.audio.recommended_delay_us))
402            .max()
403            .unwrap_or((0, 0))
404    }

406    /// Make this participant's synthetic mic emit digital SILENCE (or resume its
407    /// speech-like signal) — a SOURCE-level toggle, not
408    /// [`mute`](Self::mute).
409    ///
410    /// The distinction is the whole point. Mute gates before the encoder, so a
411    /// muted peer publishes literally nothing and exercises none of the silence
412    /// path. This feeds real zeros into a live encoder, so Opus's VAD marks
413    /// non-speech, DTX collapses the payload, and the publisher's silence
414    /// suppression engages for real — which is the only way an e2e run can
415    /// observe the behaviour a quiet-but-unmuted participant actually produces.
416    /// (The synthetic source is deliberately VAD-active otherwise; see
417    /// `hound_engine`'s `dtx_never_suppresses_the_synthetic_capture_source`.)
418    pub fn set_source_silent(&self, silent: bool) {
419        self.participant.client().set_source_silent(silent);
420    }

422    /// Assert that muting STOPS inbound audio: it must go quiet within
423    /// [`MUTE_SETTLE`], and then stay quiet for `quiet_for`. Watch-driven
424    /// throughout — it wakes on every stats tick rather than blind-sleeping.
425    ///
426    /// ## Why there is a settle phase
427    /// The obvious form of this assertion — sample a baseline, then fail on any
428    /// growth — is not well posed, and it flaked in CI for exactly that reason
429    /// (`frames grew 52 -> 53`, one frame over a 3 s window). The baseline can
430    /// only be read from the 1 Hz stats snapshot the caller already has, which
431    /// is published BEFORE `mute()` is even issued. Everything in the gap counts
432    /// as "growth": the test task waking, the mute reaching the capture thread
433    /// (checked once per 20 ms frame), and the frame already in flight through
434    /// encode → publish → network. At 50 frames/s that gap contains zero or one
435    /// frame essentially at random, so the old assertion was a coin flip.
436    ///
437    /// The product side of this was real and is fixed separately — the mic used
438    /// to gate mute only at capture, leaving up to 8 queued PCM frames (~160 ms)
439    /// to publish after mute; it now gates at the encoder too
440    /// (`client/src/call/audio_tx.rs`). But no product change can drive the
441    /// residual to zero: a frame mid-flight when the flag flips is legitimately
442    /// pre-mute audio.
443    ///
444    /// So the contract asserted here is the one that actually means "mute
445    /// works", and it is not weaker where it matters. A mute that never takes
446    /// effect keeps growing every tick and fails the settle phase; a mute that
447    /// takes effect and then leaks fails the quiet phase. Only the unobservable
448    /// one-frame boundary is forgiven.
449    ///
450    /// ## A silent stats channel is a failure, not a pass
451    ///
452    /// Both phases below distinguish "ticks arrived and the count held" from "no
453    /// tick arrived at all". Only the first is evidence of anything. The stats
454    /// snapshot is published at 1 Hz unconditionally, so a settle or quiet window
455    /// measured in seconds that observes **zero** ticks means the client or its
456    /// stats plumbing has stopped — and treating that as "audio stopped" would let
457    /// a dead client pass the very gate that exists to prove mute works.
458    pub async fn expect_audio_stops(&self, quiet_for: Duration) -> eyre::Result<()> {
459        /// Shortest quiet window that can observe the 1 Hz snapshot often enough for
460        /// "ticks arrived and showed no growth" to mean anything. Below this the
461        /// zero-tick check could not tell a muted client from a dead one, so a
462        /// too-short window is rejected outright rather than silently weakened.
463        const MIN_QUIET: Duration = Duration::from_secs(2);
464        /// How long audio may take to quiesce after `mute()` before we call it
465        /// broken. Mute is asynchronous by construction, and the count is only
466        /// visible at 1 Hz, so settling costs ~2 stats ticks; this is that plus
467        /// margin, and is still far below the "never stops" failure it guards.
468        const MUTE_SETTLE: Duration = Duration::from_secs(4);
469
470        eyre::ensure!(
471            quiet_for >= MIN_QUIET,
472            "expect_audio_stops needs a quiet window of at least {MIN_QUIET:?} to \
473             observe the 1 Hz stats snapshot, got {quiet_for:?}"
474        );
475
476        let peers = self.window_peers("expect_audio_stops")?;
477        let mut rx = self.stats_rx.clone();
478
479        // Phase 1 — settle. Advance until a FULL stats tick passes with no new
480        // frames; that is the observable definition of "the stream stopped".
481        let settle_deadline = tokio::time::Instant::now() + MUTE_SETTLE;
482        let mut last = inbound_audio_total(&rx.borrow_and_update(), &peers);
483        let mut ticks = 0u32;
484        let settled = loop {
485            let now = tokio::time::Instant::now();
486            if now >= settle_deadline {
487                break None;
488            }
489            if tokio::time::timeout(settle_deadline - now, rx.changed())
490                .await
491                .is_err()
492            {
493                // Deadline hit inside the wait. With ticks already seen this is the
494                // same "never stopped" verdict as the loop head; with none it is a
495                // dead stats channel, which the check after the loop separates.
496                break None;
497            }
498            ticks += 1;
499            let current = inbound_audio_total(&rx.borrow(), &peers);
500            if current == last {
501                break Some(current);
502            }
503            last = current;
504        };
505        if settled.is_none() && ticks == 0 {
506            eyre::bail!(
507                "no stats tick at all within {MUTE_SETTLE:?} of muting — the 1 Hz \
508                 snapshot should have published ~{} times, so the client or its stats \
509                 plumbing is broken. This is NOT evidence that audio stopped.",
510                MUTE_SETTLE.as_secs()
511            );
512        }
513        let Some(start) = settled else {
514            eyre::bail!(
515                "inbound audio never stopped within {MUTE_SETTLE:?} of muting — \
516                 frames were still arriving every stats tick ({ticks} ticks, latest \
517                 total {last}), so mute is not gating the encoder"
518            );
519        };
520
521        // Phase 2 — the quiet window, now anchored AFTER the stream went quiet.
522        // Passing needs ticks that showed no growth; zero ticks proves nothing.
523        let deadline = tokio::time::Instant::now() + quiet_for;
524        let mut quiet_ticks = 0u32;
525        loop {
526            let now = tokio::time::Instant::now();
527            let timed_out = now >= deadline
528                || tokio::time::timeout(deadline - now, rx.changed())
529                    .await
530                    .is_err();
531            if timed_out {
532                if quiet_ticks == 0 {
533                    eyre::bail!(
534                        "no stats tick during the {quiet_for:?} quiet window — the 1 Hz \
535                         snapshot went silent right after audio settled at {start} \
536                         frames, so the client stopped reporting rather than staying \
537                         muted. This is NOT evidence that audio stayed stopped."
538                    );
539                }
540                return Ok(());
541            }
542            quiet_ticks += 1;
543            let current = inbound_audio_total(&rx.borrow(), &peers);
544            if current > start {
545                eyre::bail!(
546                    "inbound audio resumed while still muted: it went quiet at \
547                     {start} frames, then grew to {current} within {quiet_for:?}"
548                );
549            }
550        }
551    }

553    /// The warmup-baseline audio-health gate — see
554    /// [`hound_client::call_health`] for the shared monitor (the same one
555    /// `houndbot` gates its verdict on). Observes for `window`, anchoring the

557    /// [`hound_client::call_health::WARMUP`], then fails if any starvation

560        let sample = crate::sample::sample_call(self.participant.client(), min_peers, window).await;

565                sample.peers_seen,
566                sample.audio_frames_rx

569        Ok(())

618    /// After an EXPECTED disruption (e.g. an SFU migration's brief media
619    /// gap), poll for a clean [`Self::expect_steady_audio`] window within
620    /// `within`. Each attempt re-anchors its own warmup baseline from the
621    /// counter totals current at that attempt's start, so a disruption
622    /// caught mid-window doesn't poison the next attempt. The invariant
623    /// under test is BOUNDED recovery, not zero interruption — a single
624    /// zero-tolerance starve gate spanning the disruption itself would
625    /// always fail on an expected, transient blip.
626    pub async fn expect_audio_recovers(
627        &self,
628        min_peers: u32,
629        within: Duration,
630    ) -> eyre::Result<()> {
631        const ATTEMPT_WINDOW: Duration = Duration::from_secs(6);
632        let deadline = tokio::time::Instant::now() + within;
633        loop {
634            match self.expect_steady_audio(min_peers, ATTEMPT_WINDOW).await {
635                Ok(()) => return Ok(()),
636                Err(e) if tokio::time::Instant::now() >= deadline => {
637                    eyre::bail!(
638                        "audio did not recover within {within:?} after the expected disruption: {e}"
639                    );
640                }
641                Err(_) => {} // expected mid-disruption blip — retry with a fresh window
642            }
643        }
644    }

680    /// Wait for `peer`'s inbound `source` video to grow by at least
681    /// `min_delta` frames from the count observed when this call is made —
682    /// the per-peer, per-source video counterpart of
683    /// [`Self::expect_audio_growth`]. Returns how long that took, so a
684    /// scenario can report the measured latency next to the deadline that
685    /// bounds it.
686    ///
687    /// The measured figure carries up to one `call:stats` interval (1 s) of
688    /// quantization: the counters are sampled by a 1 Hz emitter, so growth
689    /// that happened at t+0.1 s first becomes observable at the next tick.
690    /// Any budget asserted through `deadline` must include that second.
691    ///
692    /// Reset-tolerant: the baseline re-anchors DOWNWARD if the counter drops
693    /// (a full re-establish clears the per-peer stats row), so a reconnect
694    /// landing mid-window can't wedge this into a false timeout.
695    pub async fn expect_video_growth(
696        &self,
697        peer: u32,
698        source: VideoSource,
699        min_delta: u64,
700        deadline: Option<Duration>,
701    ) -> eyre::Result<Duration> {
702        let mut rx = self.stats_rx.clone();
703        let base = AtomicU64::new(peer_video_frames(&rx.borrow_and_update(), peer, source));
704        let started = tokio::time::Instant::now();
705        wait_for(
706            deadline,
707            &mut rx,
708            |s: &StatsSnapshot| {
709                let cur = peer_video_frames(s, peer, source);
710                if cur < base.load(Ordering::Relaxed) {
711                    base.store(cur, Ordering::Relaxed); // counters were cleared — re-anchor
712                    return false;
713                }
714                cur - base.load(Ordering::Relaxed) >= min_delta
715            },
716            |s| {
717                format!(
718                    "peer {peer} {source:?} frames_received={} (baseline {}, wanted +{min_delta})",
719                    peer_video_frames(s, peer, source),
720                    base.load(Ordering::Relaxed)
721                )
722            },
723        )
724        .await?;
725        Ok(started.elapsed())
726    }

728    /// The paused half of a pause/resume assertion: `peer`'s inbound `source`
729    /// video must STOP.
730    ///
731    /// Two bounded phases, both watch-driven (it wakes on every stats tick
732    /// rather than blind-sleeping a window and checking once):
733    ///
734    /// 1. **settle** — frames already in flight when the pause request goes
735    ///    out are expected to land, and the teardown is asynchronous, so this
736    ///    first waits (bounded by `settle`) for a whole stats tick in which
737    ///    the counter did not move. Never seeing one is a failure: the pause
738    ///    did not take effect.
739    /// 2. **quiet** — with the counter anchored at that settled value, video
740    ///    must stay stopped for `quiet_for`.
741    ///
742    /// Anchoring the baseline *after* the settle rather than at call time is
743    /// what keeps this from flaking on in-flight frames while still failing
744    /// loudly on a pause that never actually stops the stream.
745    pub async fn expect_no_video_growth(
746        &self,
747        peer: u32,
748        source: VideoSource,
749        settle: Duration,
750        quiet_for: Duration,
751    ) -> eyre::Result<()> {
752        let mut rx = self.stats_rx.clone();
753        let mut last = peer_video_frames(&rx.borrow_and_update(), peer, source);
754        // Phase 1: bounded settle.
755        let settle_by = tokio::time::Instant::now() + settle;
756        loop {
757            let now = tokio::time::Instant::now();
758            let still_arriving = || {
759                eyre::eyre!(
760                    "peer {peer} {source:?} video never stopped within {settle:?} of the pause \
761                     — frames_received still climbing (last {last})"
762                )
763            };
764            if now >= settle_by {
765                return Err(still_arriving());
766            }
767            if tokio::time::timeout(settle_by - now, rx.changed())
768                .await
769                .is_err()
770            {
771                return Err(still_arriving());
772            }
773            let cur = peer_video_frames(&rx.borrow(), peer, source);
774            if cur <= last {
775                last = cur;
776                break; // a whole tick with no new frames — the pause landed
777            }
778            last = cur;
779        }
780        // Phase 2: it must STAY stopped.
781        let quiet_by = tokio::time::Instant::now() + quiet_for;
782        loop {
783            let now = tokio::time::Instant::now();
784            if now >= quiet_by {
785                return Ok(());
786            }
787            if tokio::time::timeout(quiet_by - now, rx.changed())
788                .await
789                .is_err()
790            {
791                return Ok(());
792            }
793            let cur = peer_video_frames(&rx.borrow(), peer, source);
794            if cur > last {
795                eyre::bail!(
796                    "expected no inbound video from peer {peer} {source:?} while paused, \
797                     but frames grew {last} -> {cur} within {quiet_for:?}"
798                );
799            }
800        }
801    }

846    /// Shared core of the geometry expectations. A width of 0 (no frame
847    /// decoded yet for this peer/source, or the row is absent entirely) NEVER
848    /// satisfies the predicate — otherwise an at-most bound would pass
849    /// vacuously on a stream that is simply dead, which is the exact failure
850    /// these scenarios exist to catch.
851    async fn expect_inbound_width(
852        &self,
853        peer: u32,
854        source: VideoSource,
855        pred: impl Fn(u32) -> bool,
856        want: &str,
857        deadline: Option<Duration>,
858    ) -> eyre::Result<Duration> {
859        let mut rx = self.stats_rx.clone();
860        let started = tokio::time::Instant::now();
861        wait_for(
862            deadline,
863            &mut rx,
864            |s: &StatsSnapshot| {
865                let w = peer_video(s, peer, source).map_or(0, |v| v.width);
866                w != 0 && pred(w)
867            },
868            |s| {
869                let v = peer_video(s, peer, source);
870                format!(
871                    "peer {peer} {source:?} decoded {}x{} (wanted width {want}), \
872                     frames_received={} decoded_fps={:.1}",
873                    v.map_or(0, |v| v.width),
874                    v.map_or(0, |v| v.height),
875                    v.map_or(0, |v| v.frames_received),
876                    v.map_or(0.0, |v| v.decoded_fps),
877                )
878            },
879        )
880        .await?;
881        Ok(started.elapsed())
882    }

942/// Shared polling core for a watch-driven `expect_*`: check `pred` on every
943/// change (and once up front), bail with `describe`'s last-observed state on
944/// timeout. `deadline` defaults to [`DEFAULT_DEADLINE`].
945async fn wait_for<T>(
946    deadline: Option<Duration>,
947    rx: &mut watch::Receiver<T>,
948    pred: impl Fn(&T) -> bool,
949    describe: impl Fn(&T) -> String,
950) -> eyre::Result<()> {
951    let deadline = deadline.unwrap_or(DEFAULT_DEADLINE);
952    let wait = async {
953        loop {
954            if pred(&rx.borrow_and_update()) {
955                return;
956            }
957            if rx.changed().await.is_err() {
958                return; // sender dropped — the final check below settles it
959            }
960        }
961    };
962    if tokio::time::timeout(deadline, wait).await.is_ok() && pred(&rx.borrow()) {
963        return Ok(());
964    }
965    eyre::bail!(
966        "timed out after {deadline:?} waiting for condition; last observed: {}",
967        describe(&rx.borrow())
968    )
