1//! Magic-link sign-in: trigger the email over REST (bypassing the loopback
2//! listener `HoundClient`'s interactive flow binds — a headless caller never
3//! clicks the emailed link, so that listener is dead weight, and skipping it
4//! makes concurrent sign-ins trivially port-safe), poll Mailpit for the code,
5//! and adopt the session. Ported out of `tools/houndbot` so both `houndbot` and
6//! the scenario suite (`crate::env::Env::user`) share one implementation.
7
8use std::sync::Arc;
9use std::time::Duration;
10
11use hound_client::model::AuthFlowOutcome;
12use hound_client::{HoundClient, MemorySessionStore, RestClient, SessionStore};
13
14use crate::mailpit;

16/// How long to wait for the magic-link code to appear in Mailpit.
17pub const MAIL_WAIT_TIMEOUT: Duration = Duration::from_secs(30);
18
19/// Build a fresh, signed-out client bound to `backend_url` (its own in-memory
20/// session store — never shares tokens with another client).
21pub fn fresh_client(backend_url: &str) -> eyre::Result<Arc<HoundClient>> {
22    let store: Arc<dyn SessionStore> = Arc::new(MemorySessionStore::new());
23    HoundClient::new(backend_url, store, None)
24        .map_err(|e| eyre::eyre!("failed to build client: {e}"))
25}
26
27/// Trigger a magic-link email for `email` over REST, using a throwaway
28/// [`RestClient`] rather than `HoundClient::magic_login_begin` (which binds a
29/// loopback callback listener no headless caller ever needs).
30pub async fn request_code(backend_url: &str, email: &str) -> eyre::Result<()> {
31    let rest = RestClient::new(backend_url)
32        .map_err(|e| eyre::eyre!("failed to build bootstrap rest client: {e}"))?;
33    rest.magic_start(email, 0, "hound-testkit")
34        .await
35        .map_err(|e| eyre::eyre!("magic/start failed: {e}"))
36}
37
38/// Poll Mailpit for the code a prior [`request_code`] triggered.
39pub async fn wait_for_code(mailpit_url: &str, email: &str) -> eyre::Result<String> {
40    let http = crate::remote::http_client();
41    mailpit::wait_for_code(&http, mailpit_url, email, MAIL_WAIT_TIMEOUT)
42        .await
43        .map_err(|e| eyre::eyre!("mailpit poll failed: {e}"))
44}

46/// Mail captured in the test process itself — a backend embedded in-process
47/// hands its outbound mail here instead of to SMTP.
48pub trait Inbox: Send + Sync {
49    /// Plain-text bodies of every message sent to `email` so far, oldest first.
50    fn texts_to(&self, email: &str) -> Vec<String>;
51}
52
53/// Where the magic-link emails a backend sends can be read back.
54#[derive(Clone)]
55pub enum MailSource {
56    /// A Mailpit v1 HTTP API (the compose stack's SMTP sink).
57    Mailpit(String),
58    /// An in-process [`Inbox`].
59    Inbox(Arc<dyn Inbox>),
60}
61
62/// How often an [`Inbox`] is re-read. In-process, so a read is a lock and a
63/// scan; the cadence only bounds the added sign-in latency.
64const INBOX_POLL: Duration = Duration::from_millis(20);
65
66impl MailSource {
67    /// Wait up to [`MAIL_WAIT_TIMEOUT`] for the newest magic-link code sent
68    /// to `email`.
69    pub async fn wait_for_code(&self, email: &str) -> eyre::Result<String> {
70        let inbox = match self {
71            Self::Mailpit(url) => return wait_for_code(url, email).await,
72            Self::Inbox(inbox) => inbox,
73        };
74        let deadline = tokio::time::Instant::now() + MAIL_WAIT_TIMEOUT;
75        loop {
76            if let Some(code) = inbox
77                .texts_to(email)
78                .iter()
79                .rev()
80                .find_map(|text| mailpit::extract_code(text))
81            {
82                return Ok(code);
83            }
84            if tokio::time::Instant::now() >= deadline {
85                eyre::bail!(
86                    "no magic-link code reached the in-process inbox for {email} within {MAIL_WAIT_TIMEOUT:?}"
87                );
88            }
89            tokio::time::sleep(INBOX_POLL).await;
90        }
91    }
92}

94/// Full sign-in: request the code, poll Mailpit, verify, registering a fresh
95/// account (`display_name`) if the email is unknown. Returns the signed-in
96/// client, ready to drive calls/messaging.
97pub async fn sign_in_magic_link(
98    backend_url: &str,
99    mailpit_url: &str,
100    email: &str,
101    display_name: &str,
102) -> eyre::Result<Arc<HoundClient>> {
103    sign_in_magic_link_via(
104        backend_url,
105        &MailSource::Mailpit(mailpit_url.to_owned()),
106        email,
107        display_name,
108    )
109    .await
110}

112/// [`sign_in_magic_link`] reading the code from any [`MailSource`].
113pub async fn sign_in_magic_link_via(
114    backend_url: &str,
115    mail: &MailSource,
116    email: &str,
117    display_name: &str,
118) -> eyre::Result<Arc<HoundClient>> {
119    // Also the entry point `hound-bot` uses (it never builds an `Env`), so the
120    // remote-target DNS pins get installed here too — a no-op when unset, and
121    // write-once, so the `Env` path installing them first is fine.
122    crate::remote::install_dns_overrides();
123    let client = fresh_client(backend_url)?;
124    request_code(backend_url, email).await?;
125    let code = mail.wait_for_code(email).await?;
126
127    let outcome = client
128        .magic_login_code(&code)
129        .await
130        .map_err(|e| eyre::eyre!("magic/verify failed: {e}"))?;
131    match outcome {
132        AuthFlowOutcome::SignedIn(_) => Ok(client),
133        AuthFlowOutcome::EmailUnknown { .. } => {
134            client
135                .magic_register(display_name)
136                .await
137                .map_err(|e| eyre::eyre!("magic/register failed: {e}"))?;
138            Ok(client)
139        }
140        other => eyre::bail!("unexpected magic-link auth outcome: {other:?}"),
141    }
142}
