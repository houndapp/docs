4use std::collections::HashSet;
5use std::sync::{Arc, Mutex};

7use hound_client::HoundClient;

14/// A space handle a [`Participant`] can join/open — carries just enough to
15/// route `call_join`/`open_space` without the caller re-threading a bare
16/// `Uuid` (and to catch a text/voice mismatch before the network call).
17#[derive(Clone, Copy, Debug)]
18pub struct Space {
19    id: Uuid,
20    kind: SpaceKind,
21}
22
23impl Space {
24    pub(crate) fn new(id: Uuid, kind: SpaceKind) -> Self {
25        Self { id, kind }
26    }
27
28    pub fn id(&self) -> Uuid {
29        self.id
30    }
31}
32
33/// The user ids one [`crate::Env`] has signed in — the scope every [`InCall`]
34/// expectation filters the call roster by. Shared (not snapshotted) so a user
35/// signed in after a call starts is in scope for it too.
36#[derive(Clone, Default)]
37pub(crate) struct RunUsers(Arc<Mutex<HashSet<Uuid>>>);
38
39impl RunUsers {
40    pub(crate) fn insert(&self, user: Uuid) {
41        self.lock().insert(user);
42    }
43
44    pub(crate) fn contains(&self, user: &Uuid) -> bool {
45        self.lock().contains(user)
46    }
47
48    /// Poison-tolerant: the set is insert-only, so a panicked writer cannot
49    /// have left it inconsistent.
50    fn lock(&self) -> std::sync::MutexGuard<'_, HashSet<Uuid>> {
51        self.0.lock().unwrap_or_else(|e| e.into_inner())
52    }
53}

55/// A signed-in participant. Cheap to clone (wraps `Arc<HoundClient>`).
56#[derive(Clone)]
57pub struct Participant {
58    client: Arc<HoundClient>,
59    name: String,
60    user_id: Uuid,
61    run_users: RunUsers,
62}
63
64impl Participant {
65    pub(crate) fn new(
66        client: Arc<HoundClient>,
67        name: String,
68        user_id: Uuid,
69        run_users: RunUsers,
70    ) -> Self {
71        Self {
72            client,
73            name,
74            user_id,
75            run_users,
76        }
77    }
78
79    /// This participant's account id — how a call roster names them
80    /// (`CallView::participants`).
81    pub fn user_id(&self) -> Uuid {
82        self.user_id
83    }
84
85    pub(crate) fn run_users(&self) -> &RunUsers {
86        &self.run_users
87    }
88
89    /// The underlying client — escape hatch for anything not yet wrapped by a
90    /// semantic scenario verb (e.g. a raw `watch_auth()`/`watch_stats()`
91    /// assertion a test wants to make directly).
92    ///
93    /// Contract: reach through this at most ONCE for a given purpose. The
94    /// second scenario that needs the same raw access is the signal to
95    /// promote it to a verb on [`Participant`]/[`InCall`] — that promotion
96    /// discipline is what keeps the suite semantic instead of eroding into
97    /// raw-client calls that break on every internal refactor.
98    pub fn client(&self) -> &Arc<HoundClient> {
99        &self.client
100    }
101
102    pub fn name(&self) -> &str {
103        &self.name
104    }
105
106    /// Join a voice/video call. `space` must be [`SpaceKind::Voice`]. Opens
107    /// the engine's synthetic (camera-free/mic-free) capture sources — every
108    /// scenario participant is headless.
109    pub async fn join(&self, space: &Space) -> eyre::Result<InCall> {
110        if space.kind != SpaceKind::Voice {
111            eyre::bail!("space {} is not a voice space", space.id);
112        }
113        self.client
114            .set_capture_source(hound_client::call::CaptureSource::Synthetic)
115            .await;
116        // Subscribe the observables BEFORE joining so no publish is ever
117        // missed (see the `InCall` doc).
118        let call_rx = self.client.watch_call();
119        let stats_rx = self.client.watch_stats();
120        self.client
121            .call_join(space.id, true)
122            .await
123            .map_err(|e| eyre::eyre!("{} failed to join call {}: {e}", self.name, space.id))?;
124        Ok(InCall::new(self.clone(), call_rx, stats_rx))
125    }
126
127    /// Open a text space for messaging: loads history and subscribes its
128    /// realtime room so `message:*` events flow into the returned handle's
129    /// watch stream. `space` must be [`SpaceKind::Text`].
130    ///
131    /// Returning means the backend has **confirmed** the room join, so a
132    /// scenario may send immediately afterwards and expect the echo. That is a
133    /// product guarantee (`HoundClient::open_space`), not a testkit-side wait:
134    /// a person opening a channel needs the same thing, and a DSL that
135    /// synchronized on its own would have hidden the loss instead of fixing it.
136    pub async fn open_text(&self, space: &Space) -> eyre::Result<TextSpace> {
137        if space.kind != SpaceKind::Text {
138            eyre::bail!("space {} is not a text space", space.id);
139        }
140        self.client
141            .open_space(space.id)
142            .await
143            .map_err(|e| eyre::eyre!("{} failed to open space {}: {e}", self.name, space.id))?;
144        Ok(TextSpace::new(self.clone(), space.id))
145    }
146}
